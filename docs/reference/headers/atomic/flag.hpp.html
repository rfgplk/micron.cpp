void
  ttas(memory_order success, memory_order failure) noexcept
  {
    __pause_backoff bo;
    for ( ;; ) {
      while ( tk.get(memory_order::acquire) ) bo.relax();
      bool exp = ATOMIC_OPEN;
      if ( tk.compare_exchange_weak(exp, ATOMIC_LOCKED, success, failure) ) break;
      bo.relax();
    }
  }
