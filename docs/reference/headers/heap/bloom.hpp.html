{
  const double k = (static_cast<double>(m) / static_cast<double>(n)) * 0.6931471805599453;
  // round-half-up; k is always >= 0 here so no negative branch is needed
  usize r = static_cast<usize>(k + 0.5);
  if ( r < 1 ) r = 1;
  if ( r > cap ) r = cap;
