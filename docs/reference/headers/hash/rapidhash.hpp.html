inline __attribute__((always_inline)) void
rapid_short(const u8 *p, usize len, u64 &seed, u64 &a, u64 &b) noexcept
{
  if ( len >= 4 ) {
    seed ^= len;
    if ( len >= 8 ) {
      a = rapid_read64(p);
      b = rapid_read64(p + len - 8);
    } else {
      a = rapid_read32(p);
      b = rapid_read32(p + len - 4);
    }
  } else if ( len > 0 ) {
    a = (((u64)p[0]) << 45) | p[len - 1];
    b = p[len >> 1];
  } else
    a = b = 0;
}
