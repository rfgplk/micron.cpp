constexpr u32
__parse_release(const char *s) noexcept
{
  u32 part[3]{ 0, 0, 0 };
  usize i = 0;
  for ( usize p = 0; p < 3; p++ ) {
    if ( s[i] < '0' || s[i] > '9' ) break;
    u32 acc = 0;
    for ( ; s[i] >= '0' && s[i] <= '9'; i++ ) {
      acc = acc * 10 + static_cast<u32>(s[i] - '0');
      if ( acc > 255 ) acc = 255;
    }
    part[p] = acc;
    if ( s[i] != '.' ) break;
    i++;
  }
  if ( part[0] == 0 ) return 0;
  return ver(part[0], part[1], part[2]);
}
