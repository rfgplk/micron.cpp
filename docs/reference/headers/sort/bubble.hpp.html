__bubble(T &arr, Cmp comp)
{
  max_t n = static_cast<max_t>(arr.size());
  while ( n >= 1 ) {
    max_t k = 0;
    for ( max_t i = 1; i < n; ++i ) {
      if ( comp(arr[i], arr[i - 1]) ) {
        micron::swap(arr[i - 1], arr[i]);
        k = i;
      }
    }
    n = k;
  }
  return arr;
