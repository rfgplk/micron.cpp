Equivalent definition
__bb_slow void
  release_sheet(u32 sheet) noexcept
  {
    if ( !live || sheet == __null_off ) return;
    byte *p = buddy.base + sheet;
    if ( !buddy.__in_range(p) || !buddy.is_sheet_at(sheet) ) return;
    if ( !tlsf.remove_sheet(sheet) ) return;
    buddy.unmark_sheet(p);
    if ( buddy.deallocate(p) != ret_flag::ok ) {
      buddy.mark_sheet(p);
      (void)tlsf.add_sheet(sheet);
      return;
    }
    if ( spare == sheet ) spare = __null_off;
    collect_stats<stat_type::sheet_released>();
  }
