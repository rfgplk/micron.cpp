__bb_cold bool
  __attach(byte *mem, usize len, u8 *tags, usize tag_cap, bool zeroed) noexcept
  {
    if ( count >= __max_regions || mem == nullptr || len == 0 ) return false;
    const uintptr_t a = reinterpret_cast<uintptr_t>(mem);
    if ( a + len < a ) return false;

      if ( !__clear_of_all(tags, tag_cap) ) return false;
    }
    if ( !regions[count].attach(mem, len, tags, tag_cap, zeroed) ) return false;
    ++count;
    inited = true;
    return true;
  }

  __grow(usize bytes) noexcept
  {
    if constexpr ( !__growable ) {
