[[nodiscard]] __MICRON_ALLOC_INTERPOSE void *
operator new(usize size)
{
  ALLOC_MESSAGE("new(", size, ")");
  // [expr.new]/[basic.stc.dynamic.allocation]: a zero-size request must still
  // yield a unique non-null pointer
  // ; the arena hands back null for 0, so ask for one byte like libstdc++
  if ( size == 0 ) size = 1;
  if ( void *ptr = micron::__alloc(size) ) {
    ALLOC_MESSAGE("returning(", ptr, ")");
    return ptr;
  }
  micron::exc<micron::except::memory_error_new_scalar>("micron::operator new(): micron::__alloc failed");
}

[[nodiscard, gnu::always_inline]] inline void *
__do_alloc(usize size, usize align)
{
#if defined(__micron_arch_wasm32)
  if ( align == 0 || (align & (align - 1)) != 0 || size > static_cast<usize>(-1) - (align - 1) ) return nullptr;
#endif
  const usize padded = (size + align - 1) & ~(align - 1);
  return micron::__alloc_aligned(align, padded ? padded : align);
}
