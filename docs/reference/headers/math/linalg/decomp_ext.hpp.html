micron::math::linalg::decomp::eigen_sym_result

Defined in header <micron/math/linalg/decomp_ext.hpp>.

Real symmetric eigensystem with values, eigenvector columns and converged flag.

Algorithms and dynamic form

eigen_sym returns Jacobi sweep-order values after at most 50 sweeps. eigen_sym_qr returns ascending values after reduction and QL iteration. eigen_sym_result_dyn is the dynamic values/vectors record returned by eigen_sym_qr(dynmat). A false flag retains the current iterate rather than clearing all coefficients; symmetry and a square dynamic shape remain caller preconditions.

micron::math::linalg::decomp::schur_result

Defined in header <micron/math/linalg/decomp_ext.hpp>.

Real Schur result containing quasi-upper-triangular T, orthogonal Z and converged flag, with A=Z*T*Z^T.

Blocks and failure

Nonzero first-subdiagonal entries identify real 2-by-2 blocks. The routine reduces to Hessenberg form and applies Francis double-shift steps with exceptional shifts on stagnation; iteration is capped at 30*N+30 steps. Isolated 2-by-2 blocks are accepted without diagonalization. On a false flag T/Z remain the current approximants.

micron::math::linalg::decomp::eigen_result

Defined in header <micron/math/linalg/decomp_ext.hpp>.

General real-matrix eigenvalue record containing values_re, values_im, Schur Z/T and convergence flag.

Meaning of vectors

The implementation extracts values from Schur 1-by-1/2-by-2 blocks, emitting conjugate imaginary signs for a complex pair. Z contains Schur vectors rather than individual eigenvectors. Values are not globally sorted, and they are still extracted when Schur iteration fails; inspect converged before relying on them.

micron::math::linalg::decomp::svd_result

Defined in header <micron/math/linalg/decomp_ext.hpp>.

Fixed tall-matrix Jacobi SVD result with U[R,R], S[min(R,C)], V[C,C] and convergence flag.

Layout and completion

The svd function requires R>=C, performs at most 30 sweeps and sorts singular values descending. U is initially zero and only its first C columns are populated; zero singular columns and columns C..R-1 remain zero. Thus its physical square U is not a guaranteed completed orthogonal basis. V columns correspond to S; the input is copied and remains unchanged.

template<ieee754_floating F, usize N>
[[nodiscard]] inline eigen_sym_result<F, N>
eigen_sym(const mat<F, N, N> &A) noexcept
{
  eigen_sym_result<F, N> r{};
  if constexpr ( N == 1 ) {
    r.values.data[0] = A.data[0];
    r.vectors = mat<F, N, N>::identity();
    r.converged = true;
    return r;
  } else {
    mat<F, N, N> M = A;
    mat<F, N, N> V = mat<F, N, N>::identity();
    constexpr int max_sweeps = 50;
    bool converged = false;

    for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
      usize rotations = 0;

      for ( usize p = 0; p + 1 < N; ++p ) {
        for ( usize q = p + 1; q < N; ++q ) {
          F apq = M.data[p * N + q];
          F app = M.data[p * N + p];
          F aqq = M.data[q * N + q];
          // scale-free skip
          const F dd = math::fabs(app) + math::fabs(aqq);
          if ( math::fabs(apq) + dd == dd ) continue;
          ++rotations;
          F theta = (aqq - app) / (F(2) * apq);
          F t;
          if ( theta >= F(0) )
            t = F(1) / (theta + math::fsqrt(F(1) + theta * theta));
          else
            t = F(1) / (theta - math::fsqrt(F(1) + theta * theta));
          F c = F(1) / math::fsqrt(F(1) + t * t);
          F s = t * c;

          M.data[p * N + p] = app - t * apq;
          M.data[q * N + q] = aqq + t * apq;
          M.data[p * N + q] = F(0);
          M.data[q * N + p] = F(0);

          for ( usize rr = 0; rr < N; ++rr ) {
            if ( rr == p || rr == q ) continue;
            F arp = M.data[rr * N + p];
            F arq = M.data[rr * N + q];
            F new_arp = c * arp - s * arq;
            F new_arq = s * arp + c * arq;
            M.data[rr * N + p] = new_arp;
            M.data[p * N + rr] = new_arp;
            M.data[rr * N + q] = new_arq;
            M.data[q * N + rr] = new_arq;
          }
          for ( usize rr = 0; rr < N; ++rr ) {
            F vrp = V.data[rr * N + p];
            F vrq = V.data[rr * N + q];
            V.data[rr * N + p] = c * vrp - s * vrq;
            V.data[rr * N + q] = s * vrp + c * vrq;
          }
        }
      }

      if ( rotations == 0 ) {
        converged = true;
        break;
      }
    }

    for ( usize i = 0; i < N; ++i ) r.values.data[i] = M.data[i * N + i];
    r.vectors = V;
    r.converged = converged;
    return r;
  }
}

template<ieee754_floating F, usize N>
[[nodiscard]] inline schur_result<F, N>
schur(const mat<F, N, N> &A) noexcept
{
  schur_result<F, N> r{};
  if constexpr ( N == 1 ) {
    r.T = A;
    r.Z = mat<F, 1, 1>::identity();
    r.converged = true;
    return r;
  } else {
    auto h = hessenberg<F, N>(A);
    r.T = h.H;
    r.Z = h.Q;
    r.converged = false;

    const F eps = default_eps<F>();
    const int max_iter = int(30 * N) + 30;
    int iter = 0;
    int stagnant = 0;
    usize hi = N - 1;

    while ( true ) {

      while ( hi > 0 ) {
        F sub = math::fabs(r.T.data[hi * N + (hi - 1)]);
        F dd = math::fabs(r.T.data[(hi - 1) * N + (hi - 1)]) + math::fabs(r.T.data[hi * N + hi]);
        if ( sub <= eps * dd || sub == F(0) ) {
          r.T.data[hi * N + (hi - 1)] = F(0);
          --hi;
          stagnant = 0;
          continue;
        }
        break;
      }
      if ( hi == 0 ) {
        r.converged = true;
        break;
      }

      usize lo = hi;
      while ( lo > 0 ) {
        F sub = math::fabs(r.T.data[lo * N + (lo - 1)]);
        F dd = math::fabs(r.T.data[(lo - 1) * N + (lo - 1)]) + math::fabs(r.T.data[lo * N + lo]);
        if ( sub <= eps * dd ) {
          r.T.data[lo * N + (lo - 1)] = F(0);
          break;
        }
        --lo;
      }

      if ( hi - lo == 1 ) {

        if ( lo == 0 ) {
          r.converged = true;
          break;
        }
        hi = lo - 1;
        stagnant = 0;
        continue;
      }

      F shift_s, shift_t;
      const F p = r.T.data[(hi - 1) * N + (hi - 1)];
      const F q = r.T.data[hi * N + hi];
      const F rr_off = r.T.data[(hi - 1) * N + hi];
      const F ss_off = r.T.data[hi * N + (hi - 1)];
      if ( stagnant >= 10 && stagnant % 10 == 0 ) {

        F s_excep = math::fabs(r.T.data[hi * N + (hi - 1)]) + (hi >= 2 ? math::fabs(r.T.data[(hi - 1) * N + (hi - 2)]) : F(0));
        s_excep *= F(1.5);
        shift_s = F(2) * s_excep;
        shift_t = s_excep * s_excep;
      } else {
        shift_s = p + q;
        shift_t = p * q - rr_off * ss_off;
      }
      __impl_decomp_schur::francis_step<F, N>(r.T, r.Z, lo, hi, shift_s, shift_t);
      ++iter;
      ++stagnant;
      if ( iter >= max_iter ) {
        r.converged = false;
        break;
      }
    }
    return r;
  }
}

  constexpr int max_sweeps = 30;

  for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
    usize rotations = 0;
    for ( usize p = 0; p + 1 < C_; ++p ) {
      for ( usize q = p + 1; q < C_; ++q ) {

        F a = F(0), b = F(0), c = F(0);
        for ( usize i = 0; i < R_; ++i ) {
          const F wp = W.data[i * C_ + p];
          const F wq = W.data[i * C_ + q];
          a = math::fma<F>(wp, wp, a);
          b = math::fma<F>(wq, wq, b);
          c = math::fma<F>(wp, wq, c);
