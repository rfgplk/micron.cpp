micron::math::cr::ziv_result

Defined in header <micron/math/cr.hpp>.

Double-double approximation y and asserted absolute-error bound abs_err for a rounding-test stage.

Driver behavior

drive_f64/drive_f64_2arg run a first stage, test, then a second stage if needed and return that stage even if its test remains inconclusive. Binary32 drivers round the supplied double-double stage through a round-to-odd helper. Specific sin/cos/log/exp CR routines use bounded double-double kernels rather than an extensible precision ladder. This record is useful for the stage protocol, not an independent proof that an arbitrary supplied error bound is valid.

template<ieee754_floating F>
  [[nodiscard, gnu::always_inline]] inline constexpr F
  ulp_of(F x) noexcept
  {
    using T = ieee::traits<F>;
    using U = typename T::uint_type;
    U b = ieee::to_bits(x) & ~T::sign_mask;
    if ( b == 0 ) [[unlikely]] {
      return ieee::from_bits<F>(U(1));
    }
    if ( b >= T::exp_mask ) [[unlikely]] {
      return ieee::inf_v<F>(0);
    }
    U raw_e = (b & T::exp_mask) >> T::mant_bits;
    if ( raw_e == 0 ) [[unlikely]] {
      return ieee::from_bits<F>(U(1));
    }
    if ( raw_e <= U(T::mant_bits) ) {
      return ieee::from_bits<F>(U(1));
    }
    U ulp_raw = (raw_e - U(T::mant_bits)) << T::mant_bits;
    return ieee::from_bits<F>(ulp_raw);
  }

[[nodiscard, gnu::always_inline]] inline constexpr f32
  round_dd_to_f32(dd64 y) noexcept
  {
    const dd64 t = dd::fast_two_sum(y.hi, y.lo);
    const f64 s = t.hi;
    const f64 resid = t.lo;
    if ( resid == 0.0 ) {
      return f32(s);
    }
    // round to odd
    if ( ieee::to_bits(s) & u64(1) ) {
      return f32(s);
    }
    return f32(resid > 0.0 ? ieee::next_up(s) : ieee::next_down(s));
  }

[[nodiscard, gnu::always_inline]] inline constexpr f64
  abs_err_bound(dd64 y) noexcept
  {
    const f64 ah = mkbits::manip::fabs(y.hi);
    // 2^-72 · max(|hi|, 1).  -72 is the worst-case dd64 polynomial bound
    // we trust the kernels to deliver; well below half-ulp_f64 for any
    // |y| ≥ 2^-1022.
    return (ah > 1.0 ? ah : 1.0) * 0x1.0p-72;
  }
