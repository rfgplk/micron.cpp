template<ieee754_floating F>
  [[nodiscard]] inline constexpr F
  cbrt(F x) noexcept
  {
#if defined(__micron_arch_wasm_any)
    if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::cbrt(x);
#endif
    if ( ieee::is_nan(x) || ieee::is_inf(x) || x == F(0) ) return x;
    // NOTE: the iteration forms a**3, so the argument is first pulled into [1,8)
    const int ex = manip::ilogb<F>(x);
    int k = ex / 3;
    if ( ex % 3 < 0 ) --k;
    const F m = manip::ldexp<F>(manip::fabs(x), -3 * k);
    F a;
    if constexpr ( sizeof(F) == 4 )
      a = F(__cbrt_impl::seed_f32(f32(m)));
    else
      a = F(__cbrt_impl::seed_f64(f64(m)));
    // halley's iteration: a * (a**3 + 2m) / (2*(a**3) + m)
    for ( int i = 0; i < (sizeof(F) == 4 ? 2 : 3); ++i ) {
      F a3 = a * a * a;
      a = a * (a3 + F(2) * m) / (F(2) * a3 + m);
    }
    return manip::copysign<F>(manip::ldexp<F>(a, k), x);
  }

template<ieee754_floating F>
  [[nodiscard]] inline constexpr F
  hypot(F a, F b) noexcept
  {
#if defined(__micron_arch_wasm_any)
    if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::hypot(a, b);
#endif
    if ( ieee::is_inf(a) || ieee::is_inf(b) ) return ieee::inf_v<F>(0);
    if ( ieee::is_nan(a) || ieee::is_nan(b) ) return ieee::qnan_v<F>();
    a = manip::fabs(a);
    b = manip::fabs(b);
    if ( a == F(0) ) return b;
    if ( b == F(0) ) return a;

    F m = (a > b) ? a : b;
    int e = manip::ilogb<F>(m);
    if constexpr ( sizeof(F) == 8 ) {
      F sa = manip::ldexp<F>(a, -e);
      F sb = manip::ldexp<F>(b, -e);
      auto a2 = dd::two_prod(f64(sa), f64(sa));
      auto b2 = dd::two_prod(f64(sb), f64(sb));
      auto s = dd::add(a2, b2);
      F root = F(hw::sqrt_sd(f64(s.hi)));
      auto r2 = dd::two_prod(f64(root), f64(root));
      f64 resid_hi = (s.hi - r2.hi) - r2.lo + s.lo;
      F corr = F(resid_hi / (f64(2) * f64(root)));
      root = F(root + corr);
      return manip::ldexp<F>(root, e);
    } else {
      F sa = manip::ldexp<F>(a, -e);
      F sb = manip::ldexp<F>(b, -e);
      F s = sa * sa + sb * sb;
      return manip::ldexp<F>(F(hw::sqrt<F>(s)), e);
    }
  }
