    const u64 m = (b & TR::mant_mask) | TR::implicit_one;
    const int e = int((b & TR::exp_mask) >> TR::mant_bits) - TR::exp_bias;

      if ( shift >= 0 && shift < 128 ) {
        acc = acc + (prod << shift);
      } else if ( shift < 0 && shift > -77 ) {
        acc = acc + (prod >> (-shift));
      } else if ( shift <= -77 ) {
