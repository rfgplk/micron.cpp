micron::math::rational

Defined in header <micron/math/rational.hpp>.

Canonical exact fraction stored as an owned arbint<> numerator and arbuint<> denominator. The default value is zero over one.

Construction and operations

make_rational(numerator,denominator,limits) accepts arbitrary-precision signed inputs or i64 values, transfers the denominator sign to the numerator, divides by their greatest common divisor, and returns a rational_result. Zero denominators are rejected. A zero numerator is normalized to the default zero value.

numerator() and denominator() return const references. sign(), is_zero(), and is_integer() inspect the canonical components. Free rational_negate, rational_add, rational_subtract, rational_multiply, rational_divide, and rational_power return checked results; equality compares the canonical components directly. rational_compare returns a checked ordering instead of a Boolean less-than operation.

Addition reduces a shared denominator factor before forming products. Multiplication cancels numerator/denominator cross factors before multiplying. Division forms the reciprocal of the divisor and uses multiplication. Power uses repeated squaring, supports negative exponents, and rejects zero to a nonpositive exponent, including 0^0.

Storage, complexity, and lifetime

Arithmetic can allocate arbitrary-precision component and temporary storage. Its cost depends on limb lengths, greatest-common-divisor calculations, multiplication, and division. It is not constant-time scalar arithmetic. Power performs O(log |exponent|) checked multiplications, whose operand sizes can grow. Equal-denominator comparison avoids cross products; a general comparison forms reduced cross products.

Copies duplicate the component values; moves use the component move operations. References returned by component accessors belong to the rational object and must not outlive it or be retained across replacement of its state. Operations produce new values and preserve the input fractions.

inline rational_result
rational_add(const rational &a, const rational &b, rational_limits limits = {})
{
  using namespace __impl;
  if ( !fits(a, limits) || !fits(b, limits) ) return fail(rational_error::limb_limit);
  if ( a.is_zero() ) return { b };
  if ( b.is_zero() ) return { a };
  if ( a.denominator() == b.denominator() && a.numerator() == -b.numerator() ) return {};
  const auto common = micron::math::gcd(a.denominator(), b.denominator());
  const auto ad = a.denominator() / common;
  const auto bd = b.denominator() / common;
  if ( !product_fits(a.numerator().size(), bd.size(), limits.max_work_limbs)
       || !product_fits(b.numerator().size(), ad.size(), limits.max_work_limbs)
       || !product_fits(ad.size(), b.denominator().size(), limits.max_work_limbs) )
    return fail(rational_error::limb_limit);
  auto left = a.numerator() * arbint<>(bd, false);
  const auto right = b.numerator() * arbint<>(ad, false);
  if ( left.sign() == right.sign() && (left.size() >= limits.max_work_limbs || right.size() >= limits.max_work_limbs) )
    return fail(rational_error::limb_limit);
  left += right;
  return normalize(micron::move(left), ad * b.denominator(), limits);
}

rational_multiply(const rational &a, const rational &b, rational_limits limits = {})

inline rational_result
rational_power(const rational &a, i64 exponent, rational_limits limits = {})
{
  using namespace __impl;
  const u64 power = exponent < 0 ? static_cast<u64>(-(exponent + 1)) + 1 : static_cast<u64>(exponent);
  if ( power > limits.max_power ) return fail(rational_error::exponent_limit);
  if ( !fits(a, limits) ) return fail(rational_error::limb_limit);
  // 0^0 is deliberately undefined, rather than silently introducing a value
  if ( a.is_zero() && exponent <= 0 ) return fail(rational_error::zero_denominator);
  auto base = exponent < 0 ? rational_divide(make_rational(1).value, a, limits) : rational_result{ a };
  if ( !base ) return base;
  auto result = make_rational(1, 1, limits);
  u64 remaining = power;
  while ( remaining ) {
    if ( remaining & 1 ) {
      result = rational_multiply(result.value, base.value, limits);
      if ( !result ) return result;
    }
    remaining >>= 1;
    if ( remaining ) {
      base = rational_multiply(base.value, base.value, limits);
      if ( !base ) return base;
    }
  }
  return result;
}

inline rational_order_result
rational_compare(const rational &a, const rational &b, rational_limits limits = {})
{
  using namespace __impl;
  if ( !fits(a, limits) || !fits(b, limits) ) return { 0, rational_error::limb_limit };
  if ( a.sign() != b.sign() ) return { a.sign() < b.sign() ? -1 : 1 };

  const auto right = b.numerator() * arbint<>(ad, false);

normalize(arbint<> numerator, arbuint<> denominator, rational_limits limits)
{
  if ( denominator.is_zero() ) return fail(rational_error::zero_denominator);
  if ( !limits.max_limbs || denominator.size() > limits.max_work_limbs || numerator.size() > limits.max_work_limbs )
    return fail(rational_error::limb_limit);
  if ( numerator.is_zero() ) return {};

  if ( common != 1 ) {
    numerator /= arbint<>(common, false);
    denominator /= common;
