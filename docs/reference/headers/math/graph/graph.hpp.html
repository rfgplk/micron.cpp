  [[nodiscard]] static constexpr bool
  __endpoint(U value, Index &out) noexcept
  {

    if constexpr ( micron::is_signed_v<U> )

    using UU = micron::make_unsigned_t<U>;
    using UI = micron::make_unsigned_t<Index>;
