Type reference

micron::math::splines::bspline

Defined in header <micron/math/splines/bspline.hpp>.

Owns a scalar B-spline's degree, nondecreasing knot vector, scalar controls, and mutable span cache.

Invariants and choice

The public maximum degree is bspline_max_degree=16. make_bspline_from_ctrl accepts degrees 1..16, control_count>=degree+1, knot_count=control_count+degree+1, and nondecreasing knots; it does not require a nonempty active knot interval or comprehensive finite-value validation. Data are copied in O(control_count+degree) work/storage. The active domain is [knots[degree],knots[control_count]]. evaluate clamps into it and uses De Boor recursion, O(p²+log n) worst-case with O(p) automatic scratch; degrees 1..5 have specialized loops.

make_bspline_interpolating uses averaged clamped knots, one basis row per sample, and a no-pivot general band LU solve to obtain scalar controls. It requires strictly increasing equal-size samples and n>=p+1. Assembly/factorization cost O(n*p²), with O(n*p) banded storage. make_lsq_bspline accepts a caller knot sequence and positive optional sample weights, checks sample-domain membership and n_samples>=n_ctrl, and chooses banded Givens QR or normal-equation Cholesky. lsq_method::householder_qr aliases givens_qr=0; normal_equations=1. There is no Householder implementation behind that alias. Both fitting paths cost O((m+n)*p²) for m samples and n controls, with O(n*p) banded storage.

Basis, calculus, and allocation

basis returns n_ctrl entries with at most p+1 local nonzeros; pointer basis first clears out_count entries and returns without basis values when out_count<n_ctrl. basis_derivatives returns row-major (min(order,p)+1)*n_ctrl entries. design_matrix allocates a u32-indexed CSR matrix with p+1 stored entries per row, including local zeros; counts/indices must fit u32.

derivative_spline drops the outer knots and forms n-1 difference controls in O(n). For a degree-1 input its result has degree 0, but evaluate(bspline) returns zero for degree 0; it therefore does not evaluate the intended piecewise-constant derivative. antiderivative_spline adds endpoint knots and cumulative controls, with a supplied initial constant; degree 16 returns empty. integral constructs an antiderivative on every call and evaluates its clamped endpoints; it allocates O(n), and the degree-16 case returns zero.

make_uniform_clamped_knots, make_averaged_clamped_knots, make_unclamped_uniform_knots, make_periodic_uniform_knots, make_knots, and greville_abscissae allocate returned vectors. knot_style enumerates clamped_uniform=0,averaged=1,unclamped_uniform=2,periodic_uniform=3. Averaged/Greville generation costs O(n*p); uniform generation is O(n+p). periodic_uniform is the same uniform knot sequence as unclamped_uniform and does not alone repeat controls or create a periodic curve.

make_bspline_from_ctrl(raw_slice<const F> knots, raw_slice<const F> ctrl, u32 degree, build_info<F> *info = nullptr) noexcept
{
  bspline<F> s{};
  s.degree = degree;
  if ( degree == 0 || degree > bspline_max_degree ) {
    if ( info ) info->status = build_status::invalid_argument;
    return s;
  }
  if ( knots.size() != ctrl.size() + degree + 1 ) {
    if ( info ) info->status = build_status::size_mismatch;
    return s;
  }
  if ( ctrl.size() < degree + 1 ) {
    if ( info ) info->status = build_status::too_few_points;
    return s;
  }
  for ( usize i = 1; i < knots.size(); ++i )
    if ( knots[i] < knots[i - 1] ) {
      if ( info ) info->status = build_status::non_monotonic_x;
      return s;
    }

  s.knots.reserve(knots.size());
  s.ctrl.reserve(ctrl.size());
  for ( usize i = 0; i < knots.size(); ++i ) s.knots.emplace_back(knots[i]);
  for ( usize i = 0; i < ctrl.size(); ++i ) s.ctrl.emplace_back(ctrl[i]);

  if ( info ) info->status = build_status::ok;
  return s;
}

[[nodiscard, gnu::flatten]] inline F
evaluate(const bspline<F> &s, F x) noexcept
{
  const usize n_ctrl = s.ctrl.size();

  if ( s.degree == 0 ) return F(0);

  const u32 p = s.degree;
  const F *__restrict__ u = s.knots.data();
  const F *__restrict__ P = s.ctrl.data();

  if ( x > u[n_ctrl] ) x = u[n_ctrl];
  const usize k = __impl_bspline::bspline_span<F>(u, n_ctrl, p, x, s.last_hit);

  switch ( p ) {
  case 1:
    return __impl_bspline::deboor_fixed<1, F>(u, P, k, x);
  case 2:
    return __impl_bspline::deboor_fixed<2, F>(u, P, k, x);
  case 3:
    return __impl_bspline::deboor_fixed<3, F>(u, P, k, x);
  case 4:
    return __impl_bspline::deboor_fixed<4, F>(u, P, k, x);
  case 5:
    return __impl_bspline::deboor_fixed<5, F>(u, P, k, x);
  default:
    break;
  }

  F d[bspline_max_degree + 1];
  for ( u32 j = 0; j <= p; ++j ) d[j] = P[k - p + j];

  for ( u32 r = 1; r <= p; ++r ) {
    for ( u32 j = p; j >= r; --j ) {
      const F denom = u[k + 1 + j - r] - u[k + j - p];
      F alpha = F(0);

      d[j] = math::fma<F>(alpha, d[j] - d[j - 1], d[j - 1]);
    }
  }
  return d[p];
}

derivative_spline(const bspline<F> &s) noexcept
{
  bspline<F> out{};
  if ( s.degree == 0 ) return out;
  const u32 p = s.degree;
  const usize n_ctrl = s.ctrl.size();
  if ( n_ctrl < 2 ) return out;
  out.degree = p - 1;
  out.knots.reserve(s.knots.size() - 2);
  for ( usize i = 1; i + 1 < s.knots.size(); ++i ) out.knots.emplace_back(s.knots[i]);

  const F *P = s.ctrl.data();
  for ( usize i = 0; i + 1 < n_ctrl; ++i ) {
    const F denom = u[i + p + 1] - u[i + 1];
    F q = F(0);
    if ( denom > F(0) ) q = F(p) * (P[i + 1] - P[i]) / denom;
    out.ctrl.emplace_back(q);
  }
  return out;
}

{

  s.degree = degree;
  if ( degree == 0 || degree > bspline_max_degree ) {
    if ( info ) info->status = build_status::invalid_argument;
    return s;
  }
  if ( xs.size() != ys.size() ) {
    if ( info ) info->status = build_status::size_mismatch;
    return s;
  }
  if ( xs.size() < degree + 1 ) {
    if ( info ) info->status = build_status::too_few_points;
    return s;
  }
  if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
    if ( info ) info->status = build_status::non_monotonic_x;
    return s;
  }

  const usize n = xs.size();
  const u32 p = degree;
  const usize kl = p, ku = p;
  const usize nb = kl + ku + 1;

  s.knots = make_averaged_clamped_knots<F>(xs.ptr, n, p);
  if ( s.knots.size() != n + p + 1 ) {
    if ( info ) info->status = build_status::degenerate;
    return s;
  }
  const F *__restrict__ u = s.knots.data();

  vector<F> AB(nb * n, F(0));
  AB.set_size(nb * n);
  for ( usize i = 0; i < nb * n; ++i ) AB[i] = F(0);

  vector<F> rhs(n, F(0));
  rhs.set_size(n);

  usize last_hit = 0;
  F basis[bspline_max_degree + 1];
  for ( usize i = 0; i < n; ++i ) {
    const F xi = xs[i];
    const usize k = __impl_bspline::bspline_span<F>(u, n, p, xi, last_hit);
    __impl_bspline::bspline_basis<F>(u, k, p, xi, basis);
    for ( u32 j = 0; j <= p; ++j ) {
      const usize col = k - p + j;
      const usize band = (i + ku) - col;
      AB[band * n + col] = basis[j];
    }
    rhs[i] = ys[i];
  }

  if ( !linalg::gband_lu_solve<F>(AB.data(), kl, ku, n, rhs.data()) ) {
    if ( info ) info->status = build_status::singular_system;
    s.knots.set_size(0);
    return s;
  }

  s.ctrl.reserve(n);
  for ( usize i = 0; i < n; ++i ) s.ctrl.emplace_back(rhs[i]);
  if ( info ) info->status = build_status::ok;
  return s;
}
