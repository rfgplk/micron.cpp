gcd(T a, T b)
{

  U ub = b < 0 ? U(-b) : U(b);

  const int i = countr_zero(ua);
  ua >>= i;

  const int j = countr_zero(ub);
  ub >>= j;

  const int k = i < j ? i : j;

  for ( ;; ) {
    if ( ua > ub ) {
      U tmp = ua;
      ua = ub;
      ub = tmp;
    }
    ub -= ua;
    if ( !ub ) return T(ua << k);
    ub >>= countr_zero(ub);
  }
}
