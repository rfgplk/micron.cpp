[[nodiscard]] static SEn
  inverse(const SEn &g) noexcept
  {
    mat<F, H, H> Tinv = mat<F, H, H>::zero();
    for ( usize r = 0; r < N; ++r )
      for ( usize c = 0; c < N; ++c ) Tinv.data[r * H + c] = g.T.data[c * H + r];
    for ( usize r = 0; r < N; ++r ) {
      F acc{};
      for ( usize c = 0; c < N; ++c ) acc -= g.T.data[c * H + r] * g.T.data[c * H + N];
      Tinv.data[r * H + N] = acc;
    }
    Tinv.data[N * H + N] = F(1);
    return SEn{ Tinv };
  }
