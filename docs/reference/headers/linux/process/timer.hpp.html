Type reference

micron::timerfd_t

Defined in header <micron/linux/process/timer.hpp>.

Move-only timer descriptor selected by system_clocks, with arm, disarm, and expiration-count reads.

Description

Construction creates a close-on-exec descriptor and throws on failure. nonblocking creates a replacement descriptor with the nonblock flag. The realtime/monotonic/boottime aliases select clock domains. Relative arm can set equal initial/period intervals or separate values; arm_once uses a zero period, and arm_abs sets absolute-time flags. A zero initial expiry disarms the timer.

wait retries EINTR and returns the accumulated u64 expiration count, throwing on a negative read. try_wait performs one read, returns zero and ok=false unless exactly eight bytes were read; on a blocking descriptor that operation can still block. gettime queries current remaining time, while the public current field is only the last submitted specification. Copy is disabled; moving transfers the descriptor, although move assignment lacks a self-move guard.

Equivalent definition
static constexpr timespec_t
  __to_timespec(fduration_t value, unit u) noexcept
  {
    timespec_t ts{};
    fduration_t secs = 0;
    switch ( u ) {
    case unit::nanoseconds:
      ts.tv_sec = static_cast<time_t>(value / 1'000'000'000LL);
      ts.tv_nsec = static_cast<long>(static_cast<i64>(value) % 1'000'000'000LL);
      return ts;
    case unit::microseconds:
      ts.tv_sec = static_cast<time_t>(value / 1'000'000LL);
      ts.tv_nsec = static_cast<long>(static_cast<i64>(value) % 1'000'000LL) * 1'000L;
      return ts;
    case unit::milliseconds:
      ts.tv_sec = static_cast<time_t>(value / 1'000LL);
      ts.tv_nsec = static_cast<long>(static_cast<i64>(value) % 1'000LL) * 1'000'000L;
      return ts;
    case unit::seconds:
      secs = value;
      break;
    case unit::minutes:
      secs = value * __dur_sec_per_min;
      break;
    case unit::hours:
      secs = value * __dur_sec_per_hr;
      break;
    case unit::days:
      secs = value * __dur_sec_per_day;
      break;
    default:
      return ts;      // disarmed, which is at least a defined answer
    }
    if ( secs < 0 ) secs = 0;
    const time_t whole = static_cast<time_t>(secs);
    ts.tv_sec = whole;
    ts.tv_nsec = static_cast<long>((secs - static_cast<fduration_t>(whole)) * 1'000'000'000.0);
    if ( ts.tv_nsec > 999'999'999L ) ts.tv_nsec = 999'999'999L;
    if ( ts.tv_nsec < 0 ) ts.tv_nsec = 0;
    return ts;
  }
