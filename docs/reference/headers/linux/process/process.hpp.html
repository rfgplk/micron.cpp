micron::uprocess_t

Defined in header <micron/linux/process/process.hpp>.

Copyable executable configuration and process metadata with owned path/argument strings.

Description

The default constructor observes the current process through /proc, identities, limits, and affinity. Path constructors create a launch configuration and include the path as argv[0]. process_list_t is fvector<uprocess_t>; create_processes builds one configuration per name. spawn(uprocess_t&) updates PID, selected identities, and the spawn status.

This type does not own a running process or reap on destruction. Its envp field is a stored snapshot/configuration field, but the provided spawn/run_processes overloads pass global environ instead. The stored lims and affinity are not automatically applied by plain spawn; use explicit limit overloads and affinity operations. vector_to_argv owns the pointer array while borrowing each source string's characters; mutation or destruction of those strings invalidates the pointers.

micron::proc_info_t

Defined in header <micron/linux/process/process.hpp>.

Extended process snapshot adding capabilities, executable path, and observed stack/heap addresses.

Description

Addresses in runtime_t are numeric addresses in the observed process; retaining them neither maps nor owns that process's memory. child_result_t combines a PID, decoded exit/signal flags, wait4 resource usage, and a pre-reap resource snapshot. wait_and_collect reads /proc before its blocking wait, so those resource fields need not represent final child usage.

get_proc_info currently passes status.cap_bnd into ucap_set_t's ambient argument and status.cap_amb into its bounding argument; interpret those two resulting fields accordingly. These convenience snapshots are not a validity-checked process handle.

inline runtime_t
load_stack_heap(void)
{
  micron::string dt;
  io::open_file("/proc/self/maps") >> dt;
  micron::string::iterator stck = micron::format::find(dt, "[stack]");
  micron::string::iterator heap = micron::format::find(dt, "[heap]");
  // NOTE: the reason this is here is because we might not always have a heap, in which case heap will be nullptr
  if ( !stck and !heap ) {
    runtime_t rt = { .stack = nullptr, .heap = nullptr };
    return rt;
  } else if ( heap ) {
    micron::string::iterator stck_nl = micron::format::find_reverse(dt, stck, "\n") + 1;
    micron::string::iterator heap_nl = micron::format::find_reverse(dt, heap, "\n") + 1;
    micron::string::iterator stck_ptr = micron::format::find(dt, stck_nl, "-");
    micron::string::iterator heap_ptr = micron::format::find(dt, heap_nl, "-");
    byte *sptr = micron::format::to_pointer_addr<byte, micron::string>(stck_nl, stck_ptr);
    byte *hptr = micron::format::to_pointer_addr<byte, micron::string>(heap_nl, heap_ptr);
    runtime_t rt = { .stack = sptr, .heap = hptr };
    return rt;
  } else {
    micron::string::iterator stck_nl = micron::format::find_reverse(dt, stck, "\n") + 1;
    micron::string::iterator stck_ptr = micron::format::find(dt, stck_nl, "-");
    byte *sptr = micron::format::to_pointer_addr<byte, micron::string>(stck_nl, stck_ptr);
    runtime_t rt = { .stack = sptr, .heap = nullptr };
    return rt;
  }
  runtime_t rt = { .stack = nullptr, .heap = nullptr };
  return rt;
}

      : pids{ posix::getuid(), posix::geteuid(), posix::getgid(), posix::getpid(), posix::getppid() }, path(), argv{}, envp{}, lims(0),
        affinity(), status(0)
  {
    // programatically get argv and environ
    micron::string str;
    io::open_file("/proc/self/cmdline") >> str;
    umax_t k = 0;
    umax_t t = 0;
    // stored as null term strings, so we're iterating according to what we read
    for ( usize i = 0; i < str.size(); ++i ) {
      if ( str[i] == 0x0 ) {
        t = i;
        argv.emplace_back(str.begin() + k, str.begin() + t);
        k = t + 1;
      }
    }
    // argv[0] is always path, not absolute but works without reparsing
    if ( !argv.empty() ) path = argv[0];
    str.clear();
    k = 0;
    t = 0;
    io::open_file("/proc/self/environ") >> str;
    for ( usize i = 0; i < str.size(); ++i ) {
      if ( str[i] == 0x0 ) {
        t = i;
        envp.emplace_back(str.begin() + k, str.begin() + t);
        // our vector can only hold 64 elements, and as it turns out environ could easily exceed that
        if ( envp.full_or_overflowed() ) break;
        k = t + 1;
      }
    }
    posix::get_affinity(pids.pid, affinity);
  }

inline void
run_processes(process_list_t &n)
{
  for ( auto &t : n ) {
    micron::svector<char *> argv;
    if ( t.argv.empty() ) argv.push_back(&t.path[0]);      // t.argv carries argv[0]; only seed when empty
    for ( usize i = 0; i < t.argv.size(); i++ ) argv.push_back(&t.argv[i][0]);
    argv.push_back(nullptr);
    t.pids.uid = posix::getuid();
    t.pids.gid = posix::getgid();
    if ( micron::spawn(t.pids.pid, t.path.c_str(), &argv[0], environ) ) {
      exc<except::system_error>("micron process failed to start spawn");
    }
  }
}

template<int Stack = default_stack_size, typename F, typename... Args>
  requires micron::is_function_v<F> or micron::is_function_v<micron::remove_pointer_t<F>>
int
daemon(F f, Args &&...args)
{
  int pid = micron::fork();
  if ( pid < 0 ) exc<except::runtime_error>("micron process daemon: first fork failed");
  if ( pid > 0 ) micron::posix::exit(0);      // parent exits

  if ( posix::setsid() < 0 ) exc<except::runtime_error>("micron process daemon failed to create new session");

  // double-fork: the second child is NOT a session leader, so it can never reacquire a tty
  int pid2 = micron::fork();
  if ( pid2 < 0 ) exc<except::runtime_error>("micron process daemon: second fork failed");
  if ( pid2 > 0 ) micron::posix::exit(0);

  // don't change dir
  micron::umask(0);
  micron::close(stdin_fileno);
  micron::close(stdout_fileno);
  micron::close(stderr_fileno);

  int nfd = micron::open("/dev/null", posix::o_rdwr);      // lowest free fd == 0 (just closed)
  if ( nfd == 0 ) {
    micron::dup(0);      // -> 1
    micron::dup(0);      // -> 2
  }
  f(micron::forward<Args>(args)...);      // forward, don't decay to lvalues
  return 0;
}

inline child_result_t
wait_and_collect(pid_t pid)
{
  child_result_t r{};
  r.pid = pid;

  // read /proc/PID resources while the child is still a (possibly zombie) process, THEN reap
  r.resources = get_process_resources(pid);

  int wstatus = 0;
  micron::wait4(pid, &wstatus, 0, &r.usage);      // blocking reap (EINTR-retried inside wait4)

  if ( wifexited(wstatus) ) {
    r.exited_normally = true;
    r.exit_code = wexitstatus(wstatus);
  } else if ( wifsignaled(wstatus) ) {
    r.signaled = true;
    r.term_signal = wtermsig(wstatus);
    r.exit_code = -r.term_signal;
  }
  return r;
