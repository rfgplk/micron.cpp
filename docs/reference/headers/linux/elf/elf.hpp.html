micron::elf::handle_t

Defined in header <micron/linux/elf/elf.hpp>.

Move-only owner of one loaded module and a vector snapshot of its dynamic symbols.

Operations and invalidation

open/open_path map the object, link it into the legacy loaded-module list, and build O(symbol count) symbol rows. valid()/Boolean conversion test the load base. sym searches this object's dynamic hash tables; sym_as casts its result. symbols/dyn/path/soname/base return borrowed data or references. Move transfers the mapping and updates list links; destruction or replacement invalidates mapping pointers and snapshot references. The legacy list is not a general independently synchronized shared registry.

Equivalent definition
void
  __build_symbols()
  {
    const dyn_info_t &d = __mod.dyn;
    if ( !d.symtab || !d.strtab || d.symcount == 0 ) return;
    __syms.reserve(static_cast<usize>(d.symcount));
    for ( xword i = 0; i < d.symcount; ++i ) {
      const nsym_t &s = d.symtab[i];
      symbol_info_t row;
      row.name = d.strtab + s.name;
      row.size = s.size;
      row.type = elf_st_type(s.info);
      row.binding = elf_st_bind(s.info);
      row.visibility = elf_st_visibility(s.other);
      row.defined = (s.shndx != shn_undef);
      row.address = row.defined ? reinterpret_cast<void *>(__mod.load_base + s.value) : nullptr;
      __syms.push_back(row);
    }
  }

void
  __unlink_self() noexcept
  {
    if ( !__ilist ) return;
    if ( __loaded_modules == &__mod ) {
      __loaded_modules = __mod.next;
    } else {
      for ( module_t *p = __loaded_modules; p && p->next; p = p->next ) {
        if ( p->next == &__mod ) {
