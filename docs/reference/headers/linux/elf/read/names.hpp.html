constexpr const char *
dt_tag_name(half machine, i64 tag) noexcept
{
  if ( tag < dt_loproc || tag > dt_hiproc ) return dt_tag_name(tag);
  switch ( machine ) {
  case em_arm:
    switch ( tag ) {
    case dt_arm_exidx:
      return "DT_ARM_EXIDX";
    case dt_arm_exidxsz:
      return "DT_ARM_EXIDXSZ";
    default:
      return nullptr;
    }
  case em_x86_64:
    switch ( tag ) {
    case 0x70000000:
      return "DT_X86_64_PLT";
    case 0x70000001:
      return "DT_X86_64_PLTSZ";
    case 0x70000003:
      return "DT_X86_64_PLTENT";
    default:
      return nullptr;
    }
  default:
    return nullptr;
  }
}
