Type reference

micron::sysinfo

Defined in header <micron/linux/sys/sysinfo.hpp>.

Kernel sysinfo snapshot requested during construction.

Description

Fields contain uptime, fixed-point load averages, memory/swap/high-memory quantities, process count, and mem_unit. Memory quantities must be multiplied by mem_unit to obtain bytes. Construction discards syscall errors; initialized fields remain zero on failure. resources copies selected fields and can reload, but likewise preserves raw quantities and their separate mem_unit rather than converting them to bytes.

getpagesize returns the compile-time architecture default, including the conservative AArch64 page-size choice. getpagesizelive scans /proc/self/auxv for AT_PAGESZ once and caches a fallback on failure. That cache is an unsynchronized mutable static. get_phys_pages combines the queried size/unit with the live page size.

micron::getpagesizelive(void) noexcept

Equivalent definition
inline usize
getpagesizelive(void) noexcept
{
  static usize cached = 0;
  if ( cached ) return cached;
  usize ps = static_cast<usize>(getpagesize());      // fallback if the probe fails
  long fd = micron::syscall(SYS_openat, -100 /* AT_FDCWD */, "/proc/self/auxv", 0 /* O_RDONLY */, 0);
  if ( !micron::syscall_failed(fd) ) {
    unsigned long e[2];      // each auxv entry is { a_type, a_val }, both unsigned long
    for ( ;; ) {
      long n = micron::syscall(SYS_read, fd, &e[0], sizeof(e));
      if ( n != static_cast<long>(sizeof(e)) ) break;
      if ( e[0] == 6UL ) {      // AT_PAGESZ
        if ( e[1] ) ps = static_cast<usize>(e[1]);
        break;
      }
      if ( e[0] == 0UL ) break;      // AT_NULL terminates the auxv
    }
    micron::syscall(SYS_close, fd);
  }
  cached = ps;
  return ps;
}

Equivalent to the following definition. Expressions, calls, initialization, and conditional branches are evaluated in the order specified here.

Equivalent definition
inline long int
get_phys_pages(void)
{
  sysinfo sys;
  unsigned long int _page_size = getpagesizelive();
  while ( sys.mem_unit > 1 && _page_size > 1 ) {
    sys.mem_unit >>= 1;
    _page_size >>= 1;
  }
  sys.totalram *= sys.mem_unit;
  while ( _page_size > 1 ) {
    _page_size >>= 1;
    sys.totalram >>= 1;
  }
  return static_cast<long int>(sys.totalram);
}
