micron::posix::spawn_action

Defined in header <micron/linux/sys/spawn.hpp>.

Borrowed file-action description interpreted in the child before exec.

Description

spawn_action_type selects OPEN(path, flags, mode, target fd), CLOSE(fd), or DUP2(fd,newfd). spawn_file_actions_t contains allocated/used counts and a borrowed action pointer; iteration uses used directly without checking allocated. All pointed-to paths/actions must remain valid through the spawn handshake.

spawnattr_t holds optional session/process-group, signal mask/defaults, and scheduler settings. Initialize it with spawnattr_init before setting flags. spawn_ctx borrows path, argv, envp, actions, attributes, and an error descriptor. These are local protocol records, not owning process objects or general validation layers.

__attribute__((noreturn)) int
spawn_process(spawn_ctx *ctx)
{
  long err = 0;
  if ( ctx->fa ) {
    err = __apply_file_actions(*ctx->fa);
    if ( err < 0 ) goto fail;
  }
  if ( ctx->attr ) {
    err = __apply_spawnattr(*ctx->attr);
    if ( err < 0 ) goto fail;
  }
  err = posix::execve(ctx->path, ctx->argv, ctx->envp);      // only returns on failure (raw -errno)

fail:
  int e = micron::syscall_errno(err);      // positive errno for the parent
  if ( ctx->errfd > 0 ) micron::write(ctx->errfd, &e, sizeof(e));
  micron::posix::exit(127);
  __builtin_unreachable();
}

Equivalent definition
int
__apply_file_actions(const spawn_file_actions_t &fa)
{
  for ( int i = 0; i < fa.used; ++i ) {
    const spawn_action &a = fa.__actions[i];

    switch ( a.type ) {
    case SPAWN_ACTION_OPEN: {
      auto fd = micron::openat(at_fdcwd, a.path, a.oflag, a.mode);
      if ( fd < 0 ) return static_cast<int>(fd);
      if ( fd != a.fd ) {
        micron::dup3(fd, a.fd, 0);
        micron::close(fd);
      }
      break;
    }
    case SPAWN_ACTION_CLOSE:
      micron::close(a.fd);
      break;

    case SPAWN_ACTION_DUP2:
      micron::dup2(a.fd, a.newfd);
      break;
    }
  }
  return 0;
}

int
__apply_spawnattr(const spawnattr_t &attr)
{
  // TODO: push as separate file
  if ( attr.__flags & posix_spawn_setsid ) {
    long r = micron::syscall(SYS_setsid);      // new session + process group leader; do this before setpgid
    if ( micron::syscall_failed(r) ) return static_cast<int>(r);
  }
  if ( attr.__flags & posix_spawn_setpgroup ) {
    long r = micron::syscall(SYS_setpgid, 0, attr.pgrp);
    if ( micron::syscall_failed(r) ) return static_cast<int>(r);
  }
  if ( attr.__flags & posix_spawn_setsigmask ) {
    long r = micron::syscall(SYS_rt_sigprocmask, sig_setmask, &attr.ss, nullptr, __sig_syscall_size);
    if ( micron::syscall_failed(r) ) return static_cast<int>(r);
  }
  if ( attr.__flags & posix_spawn_setsigdef ) {
    for ( int s = 1; s <= 64; ++s ) {
      if ( !sigismember(attr.sd, s) ) continue;
      __syscall_sigaction_t act{};
      act.k_sa_handler = reinterpret_cast<sighandler_t>(0);      // SIG_DFL
      long r = micron::syscall(SYS_rt_sigaction, s, &act, nullptr, __sig_syscall_size);
      if ( micron::syscall_failed(r) ) return static_cast<int>(r);
    }
  }
  if ( attr.__flags & posix_spawn_setscheduler ) {
    long r = micron::syscall(SYS_sched_setscheduler, 0, attr.policy, &attr.__sp);
    if ( micron::syscall_failed(r) ) return static_cast<int>(r);
  }
  if ( attr.__flags & posix_spawn_setschedparam ) {
    long r = micron::syscall(SYS_sched_setparam, 0, &attr.__sp);
    if ( micron::syscall_failed(r) ) return static_cast<int>(r);
  }

  //   if ( micron::syscall_failed(r) ) return static_cast<int>(r);

  return 0;
}
