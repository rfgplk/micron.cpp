Effects

Equivalent to the following definition. Expressions, calls, initialization, and conditional branches are evaluated in the order specified here.

Equivalent definition
unsigned int
_pext_u32(unsigned int v, unsigned int mask) noexcept
{
#if defined(__micron_x86_bmi2)
  return __builtin_ia32_pext_si(v, mask);
#else
  unsigned r = 0;
  int k = 0;
  while ( mask ) {
    unsigned lo = mask & (~mask + 1u);      // lowest set bit
    if ( v & lo ) r |= (1u << k);
    ++k;
    mask &= mask - 1u;
  }
  return r;
#endif
}
