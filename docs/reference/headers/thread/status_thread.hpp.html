Type reference

micron::status_thread

Defined in header <micron/thread/status_thread.hpp>.

An immovable bounded freestanding callback thread with inline launch state and explicit negative-error setup.

Bounds and lifecycle

Stack_Size defaults to 262144 bytes and must be at least four default pages and a page multiple. TLS_Limit defaults to 1048576 bytes and must lie between one page and 16 MiB. reservation_bytes is their sum; mapped_bytes() reports the actual mapped stack plus TLS size when active.

start(void(*)(void*), context) returns 0 on success, -busy for an occupied controller, -invalid_arg for a null callback or unsuitable page geometry, and negative setup errors otherwise. Hosted use returns -op_not_supported. Attached freestanding execution is rejected. TLS-template size and alignment limits are checked before/after mapping; partial failure releases resources and leaves an empty reusable owner.

The callback and context are borrowed until join; no argument box or typed result is allocated. try_join() returns -busy while running or 0 after release; join() and destruction wait then release. Callback cancellation is the callback's own protocol. The public type provides no generic thread cancel/terminate methods.

[[nodiscard]] int
  start(void (*callback)(void *), void *context) noexcept
  {
    if ( __tid || __stack ) return -error::busy;
    if ( !callback ) return -error::invalid_arg;
#if defined(__micron_freestanding)
#if defined(__micron_attach_capable)
    if ( __micron_attach_info != nullptr ) return -error::op_not_supported;
#endif
    if ( !__tls_ensure_template() ) return -error::op_not_supported;
    const auto &t = __micron_tls_template;
    if ( !t.pagesz || t.pagesz > Stack_Size / 4 || Stack_Size % t.pagesz ) return -error::invalid_arg;
    // The frame adds at most two pages for its TCB, alignment and rounding.
    if ( t.pagesz > TLS_Limit / 2 || t.memsz > TLS_Limit - 2 * t.pagesz ) return -error::out_of_memory;
    __stack = micron::addrmap(Stack_Size);
    if ( !__stack || micron::mmap_failed(__stack) ) {
      __stack = nullptr;
      return -error::out_of_memory;
    }
    const int guarded = micron::mprotect(__stack, t.pagesz, micron::prot_none);
    if ( guarded < 0 ) {
      __release();
      return guarded;
    }
    __tls = micron::__tls_make_child_frame();
    if ( !__tls.base || __tls.size > TLS_Limit ) {
      __release();
      return -error::out_of_memory;
    }
    __callback = callback;
    __context = context;
    __payload.alive.store(true, memory_order_release);
    const auto id = posix::clone3_spawn(__entry, this, reinterpret_cast<byte *>(__stack) + t.pagesz, Stack_Size - t.pagesz,
                                        reinterpret_cast<unsigned long>(__tls.tp), nullptr, &__ctid);
    if ( id < 0 ) {
      __release();
      return id;
    }
    __tid = id;
    return 0;
#else
    (void)context;
    return -error::op_not_supported;
#endif
  }
