template<typename Fn, typename... Args>
void
when(atomic_token<bool> *result, Fn fn, Args... args)
{
  go([result, fn = micron::move(fn), ... args = micron::move(args)]() mutable {
    while ( result->get(memory_order::acquire) != true ) cpu_pause<500>();
    fn(args...);
  });
