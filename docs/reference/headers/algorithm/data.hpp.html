cut(const T &obj, const T &bins)
{
  T out;
  out.resize(obj.size());
  for ( umax_t i = 0; i < obj.size(); i++ ) {
    umax_t bin_index = 0;
    while ( bin_index < bins.size() - 1 && obj[i] > bins[bin_index + 1] ) bin_index++;
    out[i] = bin_index;

template<typename I>
I
rotate(I first, I n_first, I last)
{
  if ( first == n_first ) return last;
  if ( n_first == last ) return first;

  auto rev = [](I a, I b) {
    while ( a < b ) {
      --b;
      if ( !(a < b) ) {
        auto tmp = *a;
        *a = *b;
        *b = tmp;
        break;
      }
      auto tmp = *a;
      *a = *b;
      *b = tmp;
      ++a;
    }
  };
  rev(first, n_first);
  rev(n_first, last);
  rev(first, last);
  return first + (last - n_first);
}

Equivalent definition
template<typename I>
I
cycle_rotate(I first, I n_first, I last)
{
  if ( first == n_first || n_first == last ) return last;

  using diff_t = ptrdiff_t;
  diff_t n = last - first;
  diff_t k = n_first - first;
  diff_t g = math::gcd(n, k);

  for ( diff_t i = 0; i < g; ++i ) {
    auto tmp = *(first + i);
    diff_t j = i;

    while ( true ) {
      diff_t m = j + k;
      if ( m >= n ) m -= n;
      if ( m == i ) break;
      *(first + j) = *(first + m);
      j = m;
    }

    *(first + j) = tmp;
  }

  return first + (last - n_first);
}

template<is_iterable_container C>
constexpr void
sort_heap(C &c) noexcept
{
  // Default sort_heap consumes a MAX-heap (built by make_heap default) and
  // produces ASCENDING output. __sift_down with comp = (a < b) maintains
  // the max-heap invariant on the shrinking prefix.
  using T = typename C::value_type;
  auto comp = [](const T &a, const T &b) { return a < b; };
  umax_t n = c.size();
  auto *first = c.begin();

  while ( n > 1 ) {
    auto tmp = micron::move(first[0]);
    first[0] = micron::move(first[n - 1]);
    first[n - 1] = micron::move(tmp);

    --n;
    __sift_down(first, 0, n, comp);
  }
}

__sift_down(T *first, umax_t start, umax_t end, Cmp comp)
{
  umax_t root = start;
  while ( true ) {
    umax_t child = 2 * root + 1;
    if ( child >= end ) break;
    if ( child + 1 < end && comp(first[child], first[child + 1]) ) child++;
    if ( comp(first[root], first[child]) ) {
      micron::swap(first[root], first[child]);
      root = child;

    }
  }
