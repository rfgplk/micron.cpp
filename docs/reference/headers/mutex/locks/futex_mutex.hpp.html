static bool
  __remaining(const timespec_t &a, const timespec_t &b, timespec_t &out) noexcept
  {
    time64_t s = a.tv_sec - b.tv_sec;
    long ns = static_cast<long>(a.tv_nsec) - static_cast<long>(b.tv_nsec);
    if ( ns < 0 ) {

      --s;
