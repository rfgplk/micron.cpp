template<typename T>
inline void
__write_unsigned(T n)
{
  char buf[32];
  int i = 0;
  if ( n == 0 ) {
    buf[i++] = '0';
  } else {
    while ( n > 0 ) {
      buf[i++] = static_cast<char>('0' + (n % 10));
      n /= 10;
    }
    // reverse
    for ( int j = 0; j < i / 2; ++j ) {
      char tmp = buf[j];
      buf[j] = buf[i - j - 1];
      buf[i - j - 1] = tmp;
    }
  }
  __write(buf, static_cast<usize>(i));
}
