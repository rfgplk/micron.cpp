micron::wasm::math::format

Defined in header <micron/bits/wasm/math/real.hpp>.

Compile-time binary floating-format description used by the software arithmetic helpers.

Representation and helper values

For 4-byte and 8-byte floating types the description selects u32/u64 storage, 24/53 significand precision, and 8/11 exponent bits. The extended case uses unsigned __int128, precision 113, and 15 exponent bits. Bias, fraction, sign, infinity, and quiet-NaN masks follow those layouts. The helpers assume an appropriate binary representation; this is not a general arbitrary-size user format.

wide aliases unsigned __int128. parts stores a decomposed unsigned significand and exponent; sign is handled separately. accumulator is an aggregate of eight u64 limbs, operated on by the free term/compare/add/subtract helpers and the FMA normalization loop. These scratch values own inline storage and perform no allocation; their constant-size representation does not make all operations a single machine instruction.

inline constexpr F
pack(wide sign, int exponent, wide significand) noexcept
{
  using T = format<F>;
  if ( exponent >= int(T::emax) ) return from<F>(sign | T::infinity);
  if ( exponent <= 0 ) {

template<typename F>
inline constexpr F
sqrt(F x) noexcept
{
  using T = format<F>;
  if ( isnan(x) ) return from<F>(bits(x) | T::quiet);
  if ( x == F(0) ) return x;
  if ( x < F(0) ) return nan<F>();
  if ( isinf(x) ) return x;
  auto p = decompose(x);
  int leading = msb(p.significand), exponent = p.exponent + leading;
  p.significand <<= T::mantissa - leading;
  if ( exponent & 1 ) {
    p.significand <<= 1;
    --exponent;
  }
  constexpr unsigned shift = T::mantissa + 4;
  wide root = 0, remainder = 0;
  for ( int pair = int(T::precision + 1); pair >= 0; --pair ) {
    unsigned digits = 0;
    for ( int bit = 1; bit >= 0; --bit ) {
      const int pos = 2 * pair + bit - int(shift);
      digits = (digits << 1) | (pos >= 0 && pos < 128 ? unsigned((p.significand >> pos) & 1) : 0);
    }
    remainder = (remainder << 2) | digits;
    root <<= 1;
    const wide trial = (root << 1) | 1;
    if ( remainder >= trial ) {
      remainder -= trial;
      ++root;
    }
  }
  return pack<F>(0, exponent / 2 + T::bias, (root << 1) | wide(remainder != 0));
}
