micron::timespec_t

Defined in header <micron/bits/wasm/time.hpp>.

WebAssembly time record containing signed seconds and nanoseconds.

Time record family

timeval_t stores seconds and microseconds; itimerspec_t and itimerval_t each pair interval and current-value records in their respective units. The records are trivial values with no timer ownership, scheduling, or automatic normalization. Supply normalized fractions and representable durations to the corresponding wrappers.

Clock support is limited to the host-backed realtime/monotonic families. A matching Linux record name does not make Linux timer descriptors or every clock identifier available on this target. Copying these fixed-size records is constant-time.

clock_gettime(clockid_t id, timespec_t &out) noexcept
{
  unsigned which;

inline ssize_t
clock_getres(clockid_t id, timespec_t &out) noexcept
{
  timespec_t probe;
  const auto r = clock_gettime(id, probe);
  if ( r < 0 ) return r;
  // The host reports its guaranteed resolution, not a fabricated nanosecond clock.
  const i64 n = __micron_wasm_host_clock(id == clock_realtime || id == clock_realtime_coarse ? 2 : 3);
  if ( n <= 0 ) return -95;
  out = { n / 1000000000LL, n % 1000000000LL };
  return 0;
}

inline ssize_t
clock_nanosleep(clockid_t id, int flags, const timespec_t &ts, timespec_t *rem = nullptr) noexcept
{
  if ( ts.tv_sec < 0 || ts.tv_nsec < 0 || ts.tv_nsec >= 1000000000LL || (flags & ~timer_abstime) ) return -22;
  if ( ts.tv_sec > (0x7fffffffffffffffLL - ts.tv_nsec) / 1000000000LL ) return -22;
  i64 ns = ts.tv_sec * 1000000000LL + ts.tv_nsec;
  timespec_t now;
  const auto r = clock_gettime(id, now);
  if ( r < 0 ) return r;
  if ( flags == timer_abstime ) ns -= now.tv_sec * 1000000000LL + now.tv_nsec;
  if ( rem ) *rem = {};
  if ( ns <= 0 ) return 0;
  const auto wait = wasm::wait32(&__wasm_sleep_word, 0, ns);
  return wait == -110 ? 0 : wait;
}
