inline max_t
unifput(const char32_t *p, fd_t fd)
{
  max_t result = 0;
  while ( *p ) {
    unsigned c = static_cast<unsigned>(*p++);
    char buf[4];
    unsigned n;
    if ( c <= 0x7f ) {
      buf[0] = c;
      n = 1;
    } else if ( c <= 0x7ff ) {
      buf[0] = 0xc0 | (c >> 6);
      buf[1] = 0x80 | (c & 63);
      n = 2;
    } else if ( c <= 0xffff && !(c >= 0xd800 && c <= 0xdfff) ) {
      buf[0] = 0xe0 | (c >> 12);
      buf[1] = 0x80 | ((c >> 6) & 63);
      buf[2] = 0x80 | (c & 63);
      n = 3;
    } else if ( c >= 0x10000 && c <= 0x10ffff ) {
      buf[0] = 0xf0 | (c >> 18);
      buf[1] = 0x80 | ((c >> 12) & 63);
      buf[2] = 0x80 | ((c >> 6) & 63);
      buf[3] = 0x80 | (c & 63);
      n = 4;
    } else {
      buf[0] = '\xef';
      buf[1] = '\xbf';
      buf[2] = '\xbd';
      n = 3;
    }
    const auto r = fwrite(buf, n, fd);
    if ( r < 0 ) return r;
    result += r;
  }
  return result;
}
