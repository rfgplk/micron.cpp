micron::wasm::thread_handle

Defined in header <micron/bits/wasm/threads.hpp>.

Pair of slot and generation numbers identifying a particular worker-slot occupant.

Handle and slot protocol

A zero slot denotes an empty handle. Validation checks both the slot and its generation, so a retained handle does not silently identify the next thread to reuse that slot. Copying the pair is constant-time and does not duplicate a join right or extend callable/argument lifetime.

thread_slot stores state, generation, cancellation/park words, permanent stack/TLS addresses, the thread_entry function pointer, its borrowed void* argument, and the integer result. States distinguish free, starting, running, complete, and failed. thread_entry is int (*)(void*). These records describe the runtime protocol; raw record copies bypass its atomic publication and lifetime rules.

{

template<void (*BeforeState)(thread_handle) = nullptr>
inline int
__join(thread_handle &handle, int *result = nullptr) noexcept
{
  if ( !handle.slot || handle.slot >= MICRON_WASM_MAX_THREADS ) return -22;
  if ( handle.slot + 1 == __thread_id ) return -35;
  auto &t = __threads[handle.slot];
  unsigned state;
  for ( ;; ) {
    if ( __atomic_load_n(&t.generation, __ATOMIC_ACQUIRE) != handle.generation ) return -22;
    if constexpr ( BeforeState != nullptr ) BeforeState(handle);
    state = __atomic_load_n(&t.state, __ATOMIC_ACQUIRE);
    if ( state != 1 && state != 2 ) break;
    // A copied handle can be reaped and its slot reused between the two loads.
    // Bound the state-only wait so it cannot wait indefinitely on that successor.
    const int r = wait32(&t.state, state, __join_poll_ns);
    if ( r < 0 && r != -11 && r != -110 ) return r;
  }
  __threads_lock.lock();
  state = __atomic_load_n(&t.state, __ATOMIC_ACQUIRE);
  if ( __atomic_load_n(&t.generation, __ATOMIC_RELAXED) != handle.generation || (state != 3 && state != 4) ) {
    __threads_lock.unlock();
    return -22;
  }
  if ( result ) *result = t.result;
  const int status = state == 4 ? -5 : 0;
  __atomic_store_n(&t.state, 0u, __ATOMIC_RELEASE);
  __atomic_fetch_sub(&__live_threads, 1u, __ATOMIC_ACQ_REL);
  __threads_lock.unlock();
  handle = {};
  return status;
}
