24 template<is_map_class M, typename Fn>
25   requires micron::is_invocable_v<Fn, const typename M::key_type &, const typename M::mapped_type &>
26 const typename M::mapped_type *
27 find_if_not(const M &m, Fn fn) noexcept
28 {
29   const typename M::mapped_type *p = nullptr;
30   __impl::visit_kv(m, [&](const auto &k, const auto &v) {
31     if ( !p && !fn(k, v) ) p = micron::addressof(v);
32   });
33   return p;
34 }

100 template<is_map_class M, typename Fn>
101   requires micron::is_invocable_v<Fn, const typename M::key_type &, const typename M::mapped_type &>
102 micron::pair<micron::remove_cvref_t<M>, micron::remove_cvref_t<M>>
103 partition(Fn fn, const M &m)
104 {
105   using MM = micron::remove_cvref_t<M>;
106   MM yes = __impl::rebuild_map<MM>(m, [&](const auto &k, const auto &v) { return fn(k, v); }, __impl::__kv_value);
107   MM no = __impl::rebuild_map<MM>(m, [&](const auto &k, const auto &v) { return !fn(k, v); }, __impl::__kv_value);
108   return micron::pair<MM, MM>(micron::move(yes), micron::move(no));
109 }

111 template<is_mutable_map M, typename Fn>
112   requires micron::is_invocable_v<Fn, const typename M::key_type &, const typename M::mapped_type &>
113 usize
114 erase_if(M &m, Fn fn)
115 {
116   using K = typename M::key_type;
117   micron::fvector<K> doomed;
118   __impl::visit_kv(m, [&](const K &k, const auto &v) {
119     if ( fn(k, v) ) doomed.push_back(k);
120   });
121   for ( const K &k : doomed ) m.erase(k);
122   return doomed.size();
123 }

128 typename M::mapped_type
129 max(const M &m) noexcept
130 {
131   typename M::mapped_type best{};
132   bool first = true;
133   __impl::visit_kv(m, [&](const auto &, const auto &v) {
134     if ( first ) {
135       best = v;
136       first = false;
137     } else if ( best < v )
138       best = v;
139   });
140   return best;
141 }

229 template<typename R = flong, is_map_class M>
230 R
231 geomean(const M &m) noexcept
232 {
233   R prod = 1;
234   bool first = true;
235   __impl::visit_kv(m, [&](const auto &, const auto &v) {
236     if ( first ) {
237       prod = static_cast<R>(v);
238       first = false;
239     } else
240       prod *= static_cast<R>(v);
241   });
242   return math::powerflong(prod, static_cast<R>(R(1) / R(m.size())));
243 }

245 template<typename R = flong, is_map_class M>
246 R
247 harmonicmean(const M &m) noexcept
248 {
249   R recsum = 0;
250   __impl::visit_kv(m, [&](const auto &, const auto &v) { recsum += (R(1) / static_cast<R>(v)); });
251   return static_cast<R>(m.size()) / recsum;
252 }
