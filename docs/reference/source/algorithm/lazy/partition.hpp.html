32 template<typename C, typename P, typename V>
33 constexpr micron::tuple<C, C>
34 partition_into(V &&__v, const P &__p)
35 {
36   using B = micron::remove_cvref_t<V>;
37   static_assert(kind_of<B> != size_kind::endless, "micron::lz::partition: this pipeline never terminates");
38 
39   C __yes, __no;
40   const usize __hint = static_cast<usize>(micron::ranges::reserve_hint(__v));
41   __impl::__reserve_if(__yes, __hint);
42   __impl::__reserve_if(__no, __hint);
43 
44   __impl::__each(__v, [&](auto &&__x) {
45     if ( __p(__x) )
46       __yes.push_back(static_cast<typename C::value_type>(__x));
47     else
48       __no.push_back(static_cast<typename C::value_type>(__x));
49   });
50   return micron::make_tuple(micron::move(__yes), micron::move(__no));
51 }

107 template<typename C, typename F, typename V>
108 constexpr auto
109 traverse_into(V &&__v, const F &__f)
110 {
111   using B = micron::remove_cvref_t<V>;
112   using Res = micron::remove_cvref_t<micron::invoke_result_t<const F &, micron::ranges::range_reference_t<B>>>;
113   using E = typename Res::second_type;
114   using O = micron::option<C, E>;
115   static_assert(kind_of<B> != size_kind::endless, "micron::lz::traverse: this pipeline never terminates");
116 
117   C __out;
118   __impl::__reserve_if(__out, static_cast<usize>(micron::ranges::reserve_hint(__v)));
119 
120   bool __failed = false;
121   E __err{};
122   __impl::__each_while(__v, [&](auto &&__x) {
123     Res __r = __f(micron::forward<decltype(__x)>(__x));
124     if ( !__r.is_first() ) {
125       __err = __r.template cast<E>();
126       __failed = true;
127       return false;
128     }
129     __out.push_back(static_cast<typename C::value_type>(__r.template cast<typename Res::first_type>()));
130     return true;
131   });
132 
133   if ( __failed ) return O{ __err };
134   return O{ micron::move(__out) };
