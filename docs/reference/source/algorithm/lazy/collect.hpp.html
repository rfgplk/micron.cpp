13 //  exact   -> resize(n) then a raw-pointer fill
14 //  bounded -> resize(hint), fill, resize(written)
15 //  unknown -> reserve(hint) if possible, then push_back

22 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
23 //   micron::vector      resize + reserve + push_back
24 //   micron::fvector     same, but move only
25 //   micron::svector     resize/reserve are = delete'd; push_back only, and it throws on overflow
26 //   micron::array<T,N>  none of the three; fixed N, written through operator[]
27 template<typename O>
28 concept sizable = requires(O &__o, usize __n) { __o.resize(__n); };
29 
30 template<typename O>
31 concept reservable = requires(O &__o, usize __n) { __o.reserve(__n); };
32 
33 template<typename O>
34 concept pushable = requires(O &__o, typename O::value_type &&__v) { __o.push_back(micron::move(__v)); };
35 
36 template<typename O>
37 concept size_constructible = sizable<O> && requires(usize __n) { O(__n); };
38

42 template<typename O>
43 constexpr O
44 __sized(usize __n)
45 {
46   if constexpr ( size_constructible<O> ) {
47     return O(__n);
48   } else {
49     O __o;
50     __o.resize(__n);
51     return __o;
52   }
53 }

65 template<typename Out, typename V>
66 constexpr Out
67 collect_into(V &&__v)
68 {
69   using B = micron::remove_cvref_t<V>;
70   static_assert(kind_of<B> != size_kind::endless,
71                 "micron::lz::collect: this pipeline never terminates -- put a take(n) or take_while(p) before it");
72 
73   if constexpr ( kind_of<B> == size_kind::exact && sizable<Out> ) {
74     const usize __n = static_cast<usize>(micron::ranges::reserve_hint(__v));
75     Out __out = __impl::__sized<Out>(__n);
76     auto *__restrict __dst = __out.begin();
77     auto __i = micron::ranges::begin(__v);
78     const auto __e = micron::ranges::end(__v);
79     for ( ; !(__i == __e); ++__i, ++__dst ) *__dst = *__i;
80     return __out;
81   }
82 
83   else if constexpr ( micron::has_static_size<Out> && !sizable<Out> && !pushable<Out> ) {
84     Out __out;
85     static_assert(static_size_of<B> == micron::remove_cvref_t<Out>::static_size,
86                   "micron::lz::collect<array<T,N>>: the pipeline's compile-time extent does not match N. "
87                   "filter / take(n) / generators erase it -- collect into a vector instead");
88     if constexpr ( static_size_of<B> != no_static_size && static_size_of<B> <= micron::__unroll_max ) {
89       __impl::__fill_unrolled(__out, __v, micron::make_index_sequence<static_size_of<B>>{});
90     } else {
91       auto __i = micron::ranges::begin(__v);
92       for ( usize __k = 0; __k < micron::remove_cvref_t<Out>::static_size; ++__k, ++__i ) __out[__k] = *__i;
93     }
94     return __out;
95   }
96 
97   else if constexpr ( kind_of<B> == size_kind::bounded ) {
98     const usize __n = static_cast<usize>(micron::ranges::reserve_hint(__v));
99     if constexpr ( sizable<Out> ) {
100       Out __out = __impl::__sized<Out>(__n);
101       auto *__restrict __dst = __out.begin();
102 
103       if constexpr ( requires(const B &__b, decltype(__dst) __p) { __b.__drain_into(__p); } ) {
104         __dst = __v.__drain_into(__dst);
105       } else {
106         auto __i = micron::ranges::begin(__v);
107         const auto __e = micron::ranges::end(__v);
108         for ( ; !(__i == __e); ++__i, ++__dst ) *__dst = *__i;
109       }
110       __out.resize(static_cast<usize>(__dst - __out.begin()));
111       return __out;
112     } else {
113       static_assert(pushable<Out>, "micron::lz::collect: Out has neither resize() nor push_back()");
114       Out __out;
115       if constexpr ( reservable<Out> ) __out.reserve(__n);
116       auto __i = micron::ranges::begin(__v);
117       const auto __e = micron::ranges::end(__v);
118       for ( ; !(__i == __e); ++__i ) __out.push_back(*__i);
119       return __out;
120     }
121   }
122 
123   else {
124     static_assert(pushable<Out>, "micron::lz::collect: an unknown-length pipeline needs an Out with push_back()");
125     Out __out;
126     if constexpr ( reservable<Out> ) {
127       const usize __n = static_cast<usize>(micron::ranges::reserve_hint(__v));
128       if ( __n ) __out.reserve(__n);
129     }
130     auto __i = micron::ranges::begin(__v);
131     const auto __e = micron::ranges::end(__v);
132     for ( ; !(__i == __e); ++__i ) __out.push_back(*__i);
133     return __out;
134   }
135 }

137 template<typename Out> struct __collect_fn {
138   template<typename R>
139   constexpr Out
140   operator()(R &&__r) const
141   {
142     return collect_into<Out>(__as_view(micron::forward<R>(__r)));
143   }
