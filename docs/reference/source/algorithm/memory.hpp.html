49 template<typename T>

51 destroy(T &mem)
52   requires(!micron::is_array<T>::value)
53 {
54   if constexpr ( micron::is_class<T>::value ) {
55     mem.~T();
56   } else if constexpr ( micron::is_pointer<T>::value && !micron::is_null_pointer<T>::value && !micron::is_array<T>::value ) {
57     delete mem;
58   }

62 inline void
63 destroy(T &mem)
64   requires micron::is_array<T>::value
65 {
66   if constexpr ( micron::is_class<T>::value ) {
67     for ( usize i = 0; i < N; i++ ) mem[i].~T();
68   } else if constexpr ( micron::is_pointer<T>::value && !micron::is_null_pointer<T>::value ) {
69     delete[] mem;
70   }
71 }

108 template<typename T>
109 inline bool
110 is_zero(const T *src)
111 {
112   if ( src == nullptr ) exc<except::library_error>("micron::is_zero invalid pointer.");
113   const byte *b_ptr = reinterpret_cast<const byte *>(src);
114   for ( usize i = 0; i < sizeof(T); i++ )
115     if ( *b_ptr++ != 0x0 ) return false;
116   return true;
117 }

268 template<typename T>
269 void
270 scopy(const T &restrict src, T &restrict dst)
271 {
272   auto smlr = src.size() < dst.size() ? src.size() : dst.size();
273   if constexpr ( micron::is_class_v<T> or micron::is_object_v<T> ) {
274     for ( usize i = 0; i < smlr; i++ ) dst[i] = src[i];
275     return;
276   }
277   smlr = smlr * sizeof(typename T::value_type);
278 #if __micron_x86_simd_width >= 256
279   if ( smlr <= 32 )
280     __memcpy_32(dst, src, smlr);
281   else if ( smlr % 32 == 0 )
282     simd::memcpy256(dst, src, smlr);
283   else if ( smlr % 16 == 0 )
284     simd::memcpy128(dst, src, smlr);
285   else
286     memcpy(dst, src, smlr);
287 #elif __micron_x86_simd_width >= 128
288   if ( smlr <= 32 )
289     __memcpy_32(dst, src, smlr);
290   else if ( smlr % 16 == 0 )
291     simd::memcpy128(dst, src, smlr);
292   else
293     memcpy(dst, src, smlr);
294 #else
295   memcpy(dst, src, smlr);
296 #endif
297 }

583 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
584 // swap_ns
585 
586 template<typename N, typename T>
587 void
588 swap_n(T *restrict a, T *restrict b, const N cnt)
589 {
590   if ( a == nullptr or b == nullptr ) exc<except::library_error>("micron::swap_n invalid pointers.");
591   byte tmp[32];
592   byte *pa = reinterpret_cast<byte *>(a);
593   byte *pb = reinterpret_cast<byte *>(b);
594   N remaining = cnt;
595   while ( remaining >= 32 ) {
596     __memcpy_32(tmp, pa, 32);
597     __memcpy_32(pa, pb, 32);
598     __memcpy_32(pb, tmp, 32);
599     pa += 32;
600     pb += 32;
601     remaining -= 32;
602   }
603   for ( N i = 0; i < remaining; i++ ) {
604     byte t = pa[i];
605     pa[i] = pb[i];
606     pb[i] = t;
607   }
608 }
