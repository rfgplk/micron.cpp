85 template<typename T, class P>
86 constexpr void
87 fill(T *first, T *end, const P &value) noexcept
88 {
89   const usize n = static_cast<usize>(end - first);
90   // return early on a null test, avoids UB on nonnull
91   if ( n == 0 ) return;
92   if constexpr ( !micron::is_class_v<T> ) {
93     if ( !__builtin_is_constant_evaluated() ) {
94       micron::typeset(first, static_cast<T>(value), n);
95       return;
96     }
97   }
98   for ( usize i = 0; i < n; ++i ) first[i] = value;
99 }

432 template<typename T>
433 constexpr T *
434 shift_left(T *first, T *end, usize n) noexcept
435 {
436   if ( n == 0 || first == end ) return end;
437   const usize len = static_cast<usize>(end - first);
438   if ( n >= len ) {
439     for ( T *p = first; p != end; ++p ) *p = T{};
440     return first;
441   }
442   for ( usize i = 0; i + n < len; ++i ) first[i] = micron::move(first[i + n]);
443   for ( usize i = len - n; i < len; ++i ) first[i] = T{};
444   return first + (len - n);
445 }

447 template<typename T>
448 constexpr T *
449 shift_right(T *first, T *end, usize n) noexcept
450 {
451   if ( n == 0 || first == end ) return first;
452   const usize len = static_cast<usize>(end - first);
453   if ( n >= len ) {
454     for ( T *p = first; p != end; ++p ) *p = T{};
455     return end;
456   }
457   for ( usize i = len; i-- > n; ) first[i] = micron::move(first[i - n]);
458   for ( usize i = 0; i < n; ++i ) first[i] = T{};
459   return first + n;
460 }

478 template<typename T>
479 constexpr void
480 rotate_left(T *first, T *end, usize n) noexcept
481 {
482   const usize len = static_cast<usize>(end - first);
483   if ( len == 0 ) return;
484   n %= len;
485   if ( n == 0 ) return;
486   auto rev = [](T *a, T *b) {
487     while ( a < b ) {
488       auto tmp = *a;
489       *a = *b;
490       *b = tmp;
491       ++a;
492       --b;
493     }
494   };
495   rev(first, first + n - 1);
496   rev(first + n, end - 1);
497   rev(first, end - 1);
498 }

530 __micron_push_options __micron_optimize_no_fast_math namespace __impl{
531   [[gnu::always_inline]] constexpr void __neumaier_add(f64 & acc, f64 &comp, f64 v) noexcept { const f64 t = acc + v;
532 // whichever operand is larger in magnitude keeps its bits; the other loses the low end
533 comp += (__builtin_fabs(acc) >= __builtin_fabs(v)) ? ((acc - t) + v) : ((v - t) + acc);
534 acc = t;

543 template<typename A>
544 constexpr f64
545 __sum_lanes(const A &a, usize n) noexcept
546 {
547   f64 s[4] = { 0, 0, 0, 0 };
548   f64 c[4] = { 0, 0, 0, 0 };
549 
550   usize i = 0;
551   for ( ; i + 4 <= n; i += 4 ) {
552     __neumaier_add(s[0], c[0], static_cast<f64>(a[i + 0]));
553     __neumaier_add(s[1], c[1], static_cast<f64>(a[i + 1]));
554     __neumaier_add(s[2], c[2], static_cast<f64>(a[i + 2]));
555     __neumaier_add(s[3], c[3], static_cast<f64>(a[i + 3]));
556   }
557   for ( ; i < n; ++i ) __neumaier_add(s[0], c[0], static_cast<f64>(a[i]));
558 
559   f64 acc = 0, comp = 0;
560   for ( usize k = 0; k < 4; ++k ) {
561     __neumaier_add(acc, comp, s[k]);
562     comp += c[k];
563   }
564   return acc + comp;
565 }

600                                        constexpr umax_t __sum_int(const A &a, usize n) noexcept { // four accumulators
601                                                                                                   umax_t w = 0, x = 0, y = 0, z = 0;
602 usize i = 0;
603 for ( ; i + 4 <= n; i += 4 ) {
604   w += static_cast<umax_t>(a[i + 0]);
605   x += static_cast<umax_t>(a[i + 1]);
606   y += static_cast<umax_t>(a[i + 2]);
607   z += static_cast<umax_t>(a[i + 3]);
608 }

610 return (w + x) + (y + z);
611 }
