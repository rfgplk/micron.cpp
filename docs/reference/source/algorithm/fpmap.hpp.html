31 // keys / values -> sequence (insertion order of the map's traversal)
32 template<is_map_class M>
33 micron::fvector<typename micron::remove_cvref_t<M>::key_type>
34 keys(const M &m)

36   micron::fvector<typename micron::remove_cvref_t<M>::key_type> out;
37   micron::__impl::visit_kv(m, [&](const auto &k, const auto &) { out.push_back(k); });

50 // unzip
51 template<is_map_class M>
52 micron::pair<micron::fvector<typename micron::remove_cvref_t<M>::key_type>,
53              micron::fvector<typename micron::remove_cvref_t<M>::mapped_type>>
54 unzip(const M &m)
55 {
56   micron::fvector<typename micron::remove_cvref_t<M>::key_type> ks;
57   micron::fvector<typename micron::remove_cvref_t<M>::mapped_type> vs;
58   micron::__impl::visit_kv(m, [&](const auto &k, const auto &v) {
59     ks.push_back(k);
60     vs.push_back(v);
61   });
62   return micron::pair<micron::fvector<typename micron::remove_cvref_t<M>::key_type>,
63                       micron::fvector<typename micron::remove_cvref_t<M>::mapped_type>>(micron::move(ks), micron::move(vs));
64 }

68 bool
69 elem(const M &m, const typename micron::remove_cvref_t<M>::mapped_type &x)
70 {
71   bool found = false;
72   micron::__impl::visit_kv(m, [&](const auto &, const auto &v) {

78 // find_first: first value satisfying fn(k,v), else empty
79 template<is_map_class M, typename Fn>
80   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<M>::key_type &,
81                                   const typename micron::remove_cvref_t<M>::mapped_type &>
82 micron::option<typename micron::remove_cvref_t<M>::mapped_type, empty_container_error>
83 find_first(const M &m, Fn fn)
84 {
85   using V = typename micron::remove_cvref_t<M>::mapped_type;
86   const V *hit = nullptr;
87   micron::__impl::visit_kv(m, [&](const auto &k, const auto &v) {
88     if ( !hit && fn(k, v) ) hit = micron::addressof(v);
89   });
90   if ( hit ) return micron::option<V, empty_container_error>{ *hit };
91   return micron::option<V, empty_container_error>{ empty_container_error{} };
92 }

94 // nub: distinct values, in traversal order -> sequence
95 template<is_map_class M>
96 micron::fvector<typename micron::remove_cvref_t<M>::mapped_type>
97 nub(const M &m)
98 {
99   using V = typename micron::remove_cvref_t<M>::mapped_type;
100   micron::fvector<V> out;
101   micron::fp::__impl::seen_set<V> seen;
102   seen.reserve(m.size());
103   micron::__impl::visit_kv(m, [&](const auto &, const auto &v) {
104     if ( seen.add(v) ) out.push_back(v);
105   });
106   return out;
107 }

109 // flat_map: (k,v) -> container, concatenated -> sequence
110 template<is_map_class M, typename Fn>
111   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<M>::key_type &,
112                                   const typename micron::remove_cvref_t<M>::mapped_type &>
113 auto
114 flat_map(const M &m, Fn fn)
115 {
116   using K = typename micron::remove_cvref_t<M>::key_type;
117   using V = typename micron::remove_cvref_t<M>::mapped_type;
118   using R = micron::remove_cvref_t<decltype(fn(micron::declval<const K &>(), micron::declval<const V &>()))>;
119   using E = typename R::value_type;
120   micron::fvector<E> out;
121   micron::__impl::visit_kv(m, [&](const auto &k, const auto &v) {
122     auto sub = fn(k, v);
123     for ( const auto &e : sub ) out.push_back(e);
124   });
125   return out;
126 }

174 // concat: key-union of two maps
175 template<is_map_class M>
176 micron::remove_cvref_t<M>
177 concat(const M &a, const M &b)
178 {
179   using MM = micron::remove_cvref_t<M>;
180   MM out = micron::__impl::rebuild_map<MM>(a, [](const auto &, const auto &) { return true; }, micron::__impl::__kv_value);
181   if constexpr ( is_persistent_map<MM> ) {
182     micron::__impl::visit_kv(b, [&](const auto &k, const auto &v) { out = out.insert(k, v); });
183   } else {
184     micron::__impl::visit_kv(b, [&](const auto &k, const auto &v) { out.insert(k, v); });
185   }
186   return out;
187 }

189 // traverse : v -> option<U,E>, short-circuit on first error -> option<map, E>
190 template<is_map_class M, typename Fn>
191   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<M>::mapped_type &>
192 auto
193 traverse(const M &m, Fn fn)
194 {
195   using MM = micron::remove_cvref_t<M>;
196   using V = typename MM::mapped_type;
197   using Opt = micron::remove_cvref_t<decltype(fn(micron::declval<const V &>()))>;
198   using E = typename Opt::second_type;
199 
200   bool ok = true;
201   E err{};
202   MM out = micron::__impl::make_empty_like<MM>(m.size());
203   if constexpr ( is_persistent_map<MM> ) out = MM{};
204 
205   micron::__impl::visit_kv(m, [&](const auto &k, const auto &v) {
206     if ( !ok ) return;
207     Opt r = fn(v);
208     if ( r.is_first() ) {
209       if constexpr ( is_persistent_map<MM> )
210         out = out.insert(k, r.template cast<V>());
211       else
212         out.insert(k, r.template cast<V>());
213     } else {
214       ok = false;
215       err = r.template cast<E>();
216     }
217   });
218 
219   if ( ok ) return micron::option<MM, E>{ micron::move(out) };
220   return micron::option<MM, E>{ micron::move(err) };
221 }
