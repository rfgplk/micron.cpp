33 // ensure hash enforces equality due to the wide range of hashes provided
34 template<typename T>
35 concept __claims_hash_equality = requires { requires micron::remove_cvref_t<T>::hash_equality_consistent::value; };
36 
37 template<typename T>
38 concept hash_matches_equality
39     = micron::is_integral_v<T> || micron::is_enum_v<T>
40       || (micron::is_pointer_v<T> && !micron::is_char_ptr<micron::remove_cvref_t<T>>) || micron::is_string_ascii<T>
41       || __claims_hash_equality<T>;
42 
43 template<typename T>
44 concept hash_dedupable = hash_matches_equality<T> && requires(const T &v) {
45   { micron::hash<hash64_t>(v) } -> micron::convertible_to<hash64_t>;
46 };
47 
48 template<typename T> class seen_set
49 {
50   static constexpr bool __hashed = hash_dedupable<T>;
51   micron::conditional_t<__hashed, micron::heap_swiss_set<T>, micron::fvector<T>> __s;
52 
53 public:
54   void
55   reserve(usize n)
56   {
57     __s.reserve(n);
58   }
59 
60   // true the first time v is offered
61   bool
62   add(const T &v)
63   {
64     if constexpr ( __hashed ) {
65       return __s.insert(v);
66     } else {
67       for ( const auto &e : __s )
68         if ( e == v ) return false;
69       __s.push_back(v);
70       return true;
71     }
72   }
73 };
74 
75 };      // namespace __impl

284 {
285   if ( n >= c.size() ) {

287     return empty;
288   }
289   const usize sz = c.size() - n;

292   const auto *first = c.begin() + n;

294   for ( usize i = 0; i < sz; ++i ) *dst++ = first[i];
295   return out;

502 template<is_iterable_container C>
503 C
504 unique(const C &c)
505 {
506   if ( c.size() == 0 ) return c;
507   C out;
508   out.resize(c.size());
509   auto *dst = out.begin();
510   const auto *first = c.begin();
511   const auto *last = c.end();
512   *dst++ = *first++;
513   for ( ; first != last; ++first )
514     if ( !(*first == *(dst - 1)) ) *dst++ = *first;
515   out.resize(static_cast<usize>(dst - out.begin()));
516   return out;
517 }

597   requires fn_binary_predicate<EqFn, typename C::value_type>
598 C
599 nub_by(const C &c, EqFn eq)
600 {
601   C out;
602   out.resize(c.size());
603   auto *dst = out.begin();
604   const auto *first = c.begin();
605   const auto *last = c.end();
606   usize out_count = 0;
607   for ( const auto *it = first; it != last; ++it ) {
608     bool found = false;
609     const auto *ob = out.begin();
610     for ( usize k = 0; k < out_count; ++k ) {
611       if ( eq(ob[k], *it) ) {
612         found = true;
613         break;
614       }
615     }
616     if ( !found ) {
617       *dst++ = *it;
618       ++out_count;
619     }
620   }
621   out.resize(out_count);
622   return out;
623 }
