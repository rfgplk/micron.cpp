118 constexpr T &
119 log10(T &cont) noexcept
120 {
121   // micron::math has log10f32/log10f64 but no unqualified log10. Use the
122   // identity log10(x) = log(x) / log(10) via the generic log overload.

124   for ( auto *it = cont.begin(); it != cont.end(); ++it ) *it = math::log(static_cast<double>(*it)) / ln10;
125   return cont;

441 // gradient of function f: R^n -> R
442 template<typename F, typename V>
443 V
444 jacobian(F f, const V &x, typename V::value_type h = typename V::value_type(1e-6)) noexcept
445 {
446   using T = typename V::value_type;
447 
448   V grad = x;
449   const usize n = x.size();
450 
451   for ( usize i = 0; i < n; ++i ) {
452     V xp = x;
453     V xm = x;
454 
455     xp[i] += h;
456     xm[i] -= h;
457 
458     grad[i] = (f(xp) - f(xm)) / (T(2) * h);
459   }
460 
461   return grad;
462 }

464 // hessian of function f: R^n -> R
465 template<typename F, typename V>
466 auto
467 hessian(F f, const V &x, typename V::value_type h = typename V::value_type(1e-5)) noexcept
468 {
469   using T = typename V::value_type;
470   const usize n = x.size();
471 
472   typename V::template rebind<V>::type H(n, V(n));
473 
474   for ( usize i = 0; i < n; ++i ) {
475     for ( usize j = 0; j < n; ++j ) {
476 
477       V xpp = x;
478       V xpm = x;
479       V xmp = x;
480       V xmm = x;
481 
482       xpp[i] += h;
483       xpp[j] += h;
484       xpm[i] += h;
485       xpm[j] -= h;
486       xmp[i] -= h;
487       xmp[j] += h;
488       xmm[i] -= h;
489       xmm[j] -= h;
490 
491       H[i][j] = (f(xpp) - f(xpm) - f(xmp) + f(xmm)) / (T(4) * h * h);
492     }
493   }
494 
495   return H;
