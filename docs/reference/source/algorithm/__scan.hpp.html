22 template<typename T>
23 concept lane_scannable = (micron::is_integral_v<micron::remove_cv_t<T>> or micron::is_enum_v<micron::remove_cv_t<T>>
24                           or micron::is_pointer_v<micron::remove_cv_t<T>>)
25                          and (sizeof(T) == 1 or sizeof(T) == 2 or sizeof(T) == 4 or sizeof(T) == 8);

30 template<typename T>
31 [[gnu::always_inline]] constexpr usize
32 scan_find(const T *p, usize n, const T &v) noexcept
33 {
34   if constexpr ( lane_scannable<T> ) {
35     if ( !__builtin_is_constant_evaluated() ) return micron::simd::find_first_elem(p, n, v);
36   }
37   for ( usize i = 0; i < n; ++i )
38     if ( p[i] == v ) return i;
39   return n;
40 }

88 template<typename T>
89 [[gnu::always_inline]] constexpr bool
90 scan_equal(const T *a, const T *b, usize n) noexcept
91 {
92   usize i = 0;
93   if constexpr ( lane_scannable<T> ) {
94     if ( !__builtin_is_constant_evaluated() ) {
95       const usize bulk_end = n - n % 8;
96       for ( ; i < bulk_end; i += 8 ) {
97         bool same = true;
98         for ( usize k = 0; k < 8; ++k ) same &= (a[i + k] == b[i + k]);
99         if ( !same ) return false;
100       }
101     }
102   }
103   for ( ; i < n; ++i )
104     if ( !(a[i] == b[i]) ) return false;
105   return true;
106 }

111 struct byte_set {
112   u64 w[4];
113 
114   constexpr byte_set() noexcept : w{ 0, 0, 0, 0 } { }
115 
116   constexpr void
117   add(u8 c) noexcept
118   {
119     w[c >> 6] |= (u64(1) << (c & 63));
120   }
121 
122   constexpr bool
123   has(u8 c) const noexcept
124   {
125     return (w[c >> 6] >> (c & 63)) & 1;
126   }
127 };

129 template<typename T>
130 concept byte_wide = sizeof(T) == 1 and (micron::is_integral_v<micron::remove_cv_t<T>> or micron::is_enum_v<micron::remove_cv_t<T>>);

132 template<typename T, typename P>
133 [[gnu::always_inline]] constexpr usize
134 scan_find_first_of(const T *p, usize n, const P *set, usize k) noexcept
135 {
136   if ( k == 0 ) return n;
137 
138   if constexpr ( byte_wide<T> ) {
139     byte_set bs;
140     for ( usize c = 0; c < k; ++c ) bs.add(static_cast<u8>(static_cast<T>(set[c])));
141     for ( usize i = 0; i < n; ++i )
142       if ( bs.has(static_cast<u8>(p[i])) ) return i;
143     return n;
144   } else if constexpr ( lane_scannable<T> and micron::is_same_v<micron::remove_cv_t<T>, micron::remove_cv_t<P>> ) {
145     if ( !__builtin_is_constant_evaluated() ) return micron::simd::find_first_of_elem(p, n, set, k, 0);
146   }
147 
148   for ( usize i = 0; i < n; ++i )
149     for ( usize c = 0; c < k; ++c )
150       if ( p[i] == static_cast<T>(set[c]) ) return i;
151   return n;
152 }

162 template<typename T, typename P>
163 constexpr void
164 kmp_table(const P *pat, usize m, kmp_slot *tbl) noexcept
165 {
166   if ( m == 0 ) return;
167   tbl[0] = 0;
168   usize k = 0;
169   for ( usize i = 1; i < m; ) {
170     if ( static_cast<T>(pat[i]) == static_cast<T>(pat[k]) ) {
171       tbl[i] = static_cast<kmp_slot>(k + 1);
172       ++i;
173       ++k;
174     } else if ( k ) {
175       k = tbl[k - 1];
176     } else {
177       tbl[i] = 0;
178       ++i;
179     }
180   }
181 }

204 template<typename T, typename P>
205 constexpr usize
206 kmp_scan(const T *hay, usize n, const P *pat, usize m, const kmp_slot *tbl) noexcept
207 {
208   if ( m == 0 ) return 0;
209   if ( m > n ) return n;
210   usize k = 0;
211   for ( usize i = 0; i < n; ) {
212     if ( hay[i] == static_cast<T>(pat[k]) ) {
213       ++k;
214       ++i;
215       if ( k == m ) return i - k;
216     } else if ( k ) {
217       k = tbl[k - 1];
218     } else {
219       ++i;
220     }
221   }
222   return n;
223 }

228 {
229   if ( m == 0 or m > n ) return n;

234       ++i;

236     } else if ( k ) {
237       k = tbl[k - 1];

239       ++i;

241   }

245 template<typename T, typename P>
246 constexpr usize
247 skip_scan(const T *hay, usize n, const P *pat, usize m) noexcept
248 {
249   if ( m == 0 ) return 0;
250   if ( m > n ) return n;
251   const T lead = static_cast<T>(pat[0]);
252   usize i = 0;
253   while ( i + m <= n ) {
254     const usize hop = scan_find(hay + i, n - i, lead);
255     if ( hop == n - i ) return n;
256     i += hop;
257     if ( i + m > n ) return n;
258     usize j = 1;
259     for ( ; j < m; ++j )
260       if ( !(hay[i + j] == static_cast<T>(pat[j])) ) break;
261     if ( j == m ) return i;
262     ++i;
263   }
264   return n;
265 }

267 template<typename T, typename P>
268 constexpr usize
269 skip_rscan(const T *hay, usize n, const P *pat, usize m) noexcept
270 {
271   if ( m == 0 or m > n ) return n;
272   for ( usize i = n - m + 1; i-- > 0; ) {
273     usize j = 0;
274     for ( ; j < m; ++j )
275       if ( !(hay[i + j] == static_cast<T>(pat[j])) ) break;
276     if ( j == m ) return i;
277   }
278   return n;
