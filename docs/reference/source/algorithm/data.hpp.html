24 template<is_iterable_container T>
25   requires is_iterable_container<typename T::value_type>
26 T
27 melt(const T &obj)
28 {
29   T out;
30   using Row = typename T::value_type;
31   for ( umax_t i = 0; i < obj[0].size(); i++ ) {
32     for ( umax_t j = 0; j < obj.size(); j++ ) {
33       Row r(2);
34       r[0] = obj[j][i];
35       r[1] = static_cast<typename Row::value_type>(j);
36       out.push_back(r);
37     }
38   }
39   return out;
40 }

42 template<is_iterable_container T>
43   requires micron::is_arithmetic_v<typename T::value_type>
44 T
45 cut(const T &obj, const T &bins)
46 {
47   T out;
48   out.resize(obj.size());
49   for ( umax_t i = 0; i < obj.size(); i++ ) {
50     umax_t bin_index = 0;
51     while ( bin_index < bins.size() - 1 && obj[i] > bins[bin_index + 1] ) bin_index++;
52     out[i] = bin_index;
53   }
54   return out;
55 }

57 template<typename I, typename J, typename O>
58 constexpr O
59 merge(I first1, I last1, J first2, J last2, O d_first)
60 {
61   while ( first1 != last1 && first2 != last2 ) {
62     if ( *first2 < *first1 ) {
63       *d_first++ = *first2++;
64     } else {
65       *d_first++ = *first1++;
66     }
67   }
68   while ( first1 != last1 ) *d_first++ = *first1++;
69   while ( first2 != last2 ) *d_first++ = *first2++;
70   return d_first;
71 }

89 template<typename I>
90 void
91 reverse(I first, I last)
92 {
93   while ( (first != last) && (first != --last) ) {
94     auto tmp = *first;
95     *first = *last;
96     *last = tmp;
97     ++first;
98   }

101 template<typename I>
102 I
103 rotate(I first, I n_first, I last)
104 {
105   if ( first == n_first ) return last;
106   if ( n_first == last ) return first;
107 
108   auto rev = [](I a, I b) {
109     while ( a < b ) {
110       --b;
111       if ( !(a < b) ) {
112         auto tmp = *a;
113         *a = *b;
114         *b = tmp;
115         break;
116       }
117       auto tmp = *a;
118       *a = *b;
119       *b = tmp;
120       ++a;
121     }
122   };
123   rev(first, n_first);
124   rev(n_first, last);
125   rev(first, last);
126   return first + (last - n_first);
127 }

129 template<typename I>
130 I
131 cycle_rotate(I first, I n_first, I last)
132 {
133   if ( first == n_first || n_first == last ) return last;
134 
135   using diff_t = ptrdiff_t;
136   diff_t n = last - first;
137   diff_t k = n_first - first;
138   diff_t g = math::gcd(n, k);
139 
140   for ( diff_t i = 0; i < g; ++i ) {
141     auto tmp = *(first + i);
142     diff_t j = i;
143 
144     while ( true ) {
145       diff_t m = j + k;
146       if ( m >= n ) m -= n;
147       if ( m == i ) break;
148       *(first + j) = *(first + m);
149       j = m;
150     }
151 
152     *(first + j) = tmp;
153   }
154 
155   return first + (last - n_first);
156 }

158 template<typename T, typename Cmp>
159 constexpr void
160 __sift_down(T *first, umax_t start, umax_t end, Cmp comp)
161 {
162   umax_t root = start;
163   while ( true ) {
164     umax_t child = 2 * root + 1;
165     if ( child >= end ) break;
166     if ( child + 1 < end && comp(first[child], first[child + 1]) ) child++;
167     if ( comp(first[root], first[child]) ) {
168       micron::swap(first[root], first[child]);
169       root = child;
170     } else {
171       break;
172     }
173   }
174 }

176 template<typename T, typename Cmp>
177 constexpr void
178 __sift_up(T *first, umax_t idx, Cmp comp) noexcept
179 {
180   while ( idx > 0 ) {
181     umax_t parent = (idx - 1) >> 1;
182 
183     if ( !comp(first[parent], first[idx]) ) return;
184 
185     auto tmp = micron::move(first[parent]);
186     first[parent] = micron::move(first[idx]);
187     first[idx] = micron::move(tmp);
188 
189     idx = parent;
190   }
191 }

194 constexpr void
195 make_heap(C &c) noexcept
196 {
197   using T = typename C::value_type;
198   auto comp = [](const T &a, const T &b) { return a < b; };
199 
200   umax_t n = c.size();
201   if ( n < 2 ) return;
202 
203   for ( umax_t i = (n >> 1); i > 0; ) {
204     --i;
205     __sift_down(c.begin(), i, n, comp);
206   }
207 }

361 template<is_iterable_container C>
362 constexpr void
363 sort_heap(C &c) noexcept
364 {
365   // Default sort_heap consumes a MAX-heap (built by make_heap default) and
366   // produces ASCENDING output. __sift_down with comp = (a < b) maintains
367   // the max-heap invariant on the shrinking prefix.
368   using T = typename C::value_type;
369   auto comp = [](const T &a, const T &b) { return a < b; };
370   umax_t n = c.size();
371   auto *first = c.begin();
372 
373   while ( n > 1 ) {
374     auto tmp = micron::move(first[0]);
375     first[0] = micron::move(first[n - 1]);
376     first[n - 1] = micron::move(tmp);
377 
378     --n;
379     __sift_down(first, 0, n, comp);
380   }
381 }

419 template<is_iterable_container C>
420 constexpr bool
421 is_heap(const C &c) noexcept
422 {
423   using T = typename C::value_type;
424   auto comp = [](const T &a, const T &b) { return a < b; };
425 
426   umax_t n = c.size();
427   auto *first = c.begin();
428 
429   for ( umax_t i = 1; i < n; ++i ) {
430     umax_t parent = (i - 1) >> 1;
431     if ( comp(first[parent], first[i]) ) return false;
432   }
433 
434   return true;
435 }
