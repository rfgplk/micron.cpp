85 // find_first -> option (set: element; map: value)
86 template<is_set_tree Tree, typename Fn>
87   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<Tree>::value_type &>
88 micron::option<typename micron::remove_cvref_t<Tree>::value_type, empty_container_error>
89 find_first(const Tree &t, Fn fn)
90 {
91   using V = typename micron::remove_cvref_t<Tree>::value_type;
92   const V *hit = nullptr;
93   t.for_each([&](const auto &e) {
94     if ( !hit && fn(e) ) hit = micron::addressof(e);
95   });
96   if ( hit ) return micron::option<V, empty_container_error>{ *hit };
97   return micron::option<V, empty_container_error>{ empty_container_error{} };

144 // flat_map -> sequence (concat of per-element containers)
145 template<is_set_tree Tree, typename Fn>
146   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<Tree>::value_type &>
147 auto
148 flat_map(const Tree &t, Fn fn)
149 {
150   using V = typename micron::remove_cvref_t<Tree>::value_type;
151   using R = micron::remove_cvref_t<decltype(fn(micron::declval<const V &>()))>;
152   using E = typename R::value_type;
153   micron::fvector<E> out;
154   t.for_each([&](const auto &e) {
155     auto sub = fn(e);
156     for ( const auto &x : sub ) out.push_back(x);
157   });
158   return out;
159 }

162   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<Tree>::key_type &,
163                                   const typename micron::remove_cvref_t<Tree>::mapped_type &>
164 auto
165 flat_map(const Tree &t, Fn fn)
166 {
167   using K = typename micron::remove_cvref_t<Tree>::key_type;
168   using V = typename micron::remove_cvref_t<Tree>::mapped_type;
169   using R = micron::remove_cvref_t<decltype(fn(micron::declval<const K &>(), micron::declval<const V &>()))>;
170   using E = typename R::value_type;
171   micron::fvector<E> out;
172   t.for_each([&](const auto &k, const auto &v) {
173     auto sub = fn(k, v);
174     for ( const auto &x : sub ) out.push_back(x);
175   });
176   return out;
177 }

289 // traverse (tree_map values) : v -> option<U,E>, short-circuit -> option<tree, E>
290 template<is_tree_map Tree, typename Fn>
291   requires micron::is_invocable_v<Fn, const typename micron::remove_cvref_t<Tree>::mapped_type &>
292 auto
293 traverse(const Tree &t, Fn fn)
294 {
295   using TT = micron::remove_cvref_t<Tree>;
296   using V = typename TT::mapped_type;
297   using Opt = micron::remove_cvref_t<decltype(fn(micron::declval<const V &>()))>;
298   using E = typename Opt::second_type;
299 
300   bool ok = true;
301   E err{};
302   TT out;
303   t.for_each([&](const auto &k, const auto &v) {
304     if ( !ok ) return;
305     Opt r = fn(v);
306     if ( r.is_first() )
307       out.insert(k, r.template cast<V>());
308     else {
309       ok = false;
310       err = r.template cast<E>();
311     }
312   });
313   if ( ok ) return micron::option<TT, E>{ micron::move(out) };
314   return micron::option<TT, E>{ micron::move(err) };
315 }
