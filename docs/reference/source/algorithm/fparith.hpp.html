192 // element-wise zip arithmetic
193 template<is_iterable_container C>
194 C
195 add_zip(C a, const C &b) noexcept
196 {
197   if constexpr ( micron::unrollable<C> ) {
198     micron::__impl::__unroll_zip_add(a.begin(), b.begin(), make_index_sequence<C::static_size>{});
199   } else {
200     const usize __n = a.size() < b.size() ? a.size() : b.size();
201     auto *fa = a.begin();
202     const auto *fb = b.begin();
203     for ( usize __i = 0; __i < __n; ++__i, ++fa, ++fb ) *fa = *fa + *fb;
204   }
205   return a;
206 }

238 template<is_iterable_container C>
239 micron::option<C, division_by_zero_error>
240 divide_zip(C a, const C &b) noexcept
241 {
242   const usize __n = a.size() < b.size() ? a.size() : b.size();
243   auto *fa = a.begin();
244   const auto *fb = b.begin();
245   for ( usize __i = 0; __i < __n; ++__i, ++fa, ++fb ) {
246     if ( *fb == typename C::value_type{} ) return micron::option<C, division_by_zero_error>{ division_by_zero_error{} };
247     *fa = *fa / *fb;
248   }
249   return micron::option<C, division_by_zero_error>{ micron::move(a) };
250 }

253 // inner_product
254 template<is_iterable_container C, typename R = typename C::value_type>
255 R
256 inner_product(const C &a, const C &b, R init = R{}) noexcept
257 {
258   if constexpr ( micron::unrollable<C> ) {
259     return micron::__impl::__unroll_inner_product(a.begin(), b.begin(), init, make_index_sequence<C::static_size>{});
260   } else {
261     const auto *fa = a.begin();
262     const auto *ea = a.end();
263     const auto *fb = b.begin();
264     for ( ; fa != ea; ++fa, ++fb ) init = init + (*fa * *fb);
265     return init;
266   }
267 }

307 // negate
308 template<is_iterable_container C>
309 C
310 negate(C cont) noexcept
311 {
312   if constexpr ( micron::unrollable<C> ) {
313     micron::__impl::__unroll_negate(cont.begin(), make_index_sequence<C::static_size>{});
314   } else {
315     auto *first = cont.begin();
316     const auto *end = cont.end();
317     for ( ; first != end; ++first ) *first = -(*first);
318   }
319   return cont;
320 }

333   requires micron::is_signed_v<typename C::value_type>
334 C
335 abs(C cont) noexcept
336 {
337   auto *first = cont.begin();
338   const auto *end = cont.end();
339   for ( ; first != end; ++first ) *first = (*first < typename C::value_type{}) ? -(*first) : *first;
340   return cont;
341 }
