37 // NOTE: uintptr_t and size_t must be defined off sizeof(), NEVER __UINTPTR_TYPE__;
38 // arm uapi header asm/types.h does #undef __UINTPTR_TYPE__ / #define __UINTPTR_TYPE__ unsigned long,
39 // while gcc's arm32-glibc builtin is unsigned int; any TU that saw a kernel header before a
40 // micron one therefore got a size_t that conflicted with glibc's, and stopped compiling
41 using ptrdiff_t = __PTRDIFF_TYPE__;
42 using intptr_t = __INTPTR_TYPE__;
