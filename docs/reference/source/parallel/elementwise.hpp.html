39 // in place; maps the first half, swaps symmetric elements
40 template<class It>
41 [[nodiscard]] micron::task<void>
42 reverse(It __first, It __last)
43 {
44   const usize __n = static_cast<usize>(__last - __first);
45   It __half = __first + __n / 2;
46   auto __leaf = [__first, __n](It __a, It __b) {
47     for ( It __p = __a; __p != __b; ++__p ) {
48       const usize __i = static_cast<usize>(__p - __first);
49       It __m = __first + (__n - 1u - __i);
50       auto __t = *__p;
51       *__p = *__m;
52       *__m = __t;
53     }
54   };
55   return __pmap<It, decltype(__leaf)>(__first, __half, __leaf, __grain_for(__n / 2u));
56 }

58 // out[i] = in[n-1-i]
59 template<class It, class Out>
60 [[nodiscard]] micron::task<void>
61 reverse_copy(It __first, It __last, Out __out)
62 {
63   const usize __n = static_cast<usize>(__last - __first);
64   auto __leaf = [__first, __out, __n](It __a, It __b) {
65     for ( It __p = __a; __p != __b; ++__p ) {
66       const usize __i = static_cast<usize>(__p - __first);
67       __out[__i] = *(__first + (__n - 1u - __i));
68     }
69   };
70   return __pmap<It, decltype(__leaf)>(__first, __last, __leaf, __grain_for(__n));
71 }
