19 // keep(i) is a predicate over the global index i in [0, n)
20 template<class In, class Out, class Keep>
21 micron::task<usize>
22 __pcompact_idx(In __in, Out __out, usize __n, Keep __keep)
23 {
24   if ( __n == 0 ) co_return 0;
25   const usize __B = __grain_for(__n);
26   const usize __nb = (__n + __B - 1u) / __B;
27   if ( __nb <= 1u ) {      // single block: serial compaction
28     usize __k = 0;
29     for ( usize __i = 0; __i < __n; ++__i )
30       if ( __keep(__i) ) __out[__k++] = __in[__i];
31     co_return __k;
32   }
33 
34   micron::vector<usize> __cnt(__nb);
35   usize *__counts = &__cnt[0];
36   {
37     auto __body = [__n, __B, __counts, __keep](usize __b) {
38       const usize __s = __b * __B;
39       const usize __e = (__s + __B < __n) ? __s + __B : __n;
40       usize __c = 0;
41       for ( usize __i = __s; __i < __e; ++__i )
42         if ( __keep(__i) ) ++__c;
43       __counts[__b] = __c;
44     };
45     co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
46   }
47   micron::vector<usize> __off(__nb);
48   usize *__offsets = &__off[0];
49   usize __total = 0;
50   for ( usize __b = 0; __b < __nb; ++__b ) {
51     __offsets[__b] = __total;
52     __total += __counts[__b];
53   }
54   {
55     auto __body = [__in, __out, __n, __B, __offsets, __keep](usize __b) {
56       const usize __s = __b * __B;
57       const usize __e = (__s + __B < __n) ? __s + __B : __n;
58       usize __pos = __offsets[__b];
59       for ( usize __i = __s; __i < __e; ++__i )
60         if ( __keep(__i) ) __out[__pos++] = __in[__i];
61     };
62     co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
63   }
64   co_return __total;
65 }

67 template<class It, class Out, class Pred>
68 [[nodiscard]] micron::task<usize>
69 where(It __first, It __last, Out __out, Pred __pred)
70 {
71   auto __keep = [__first, __pred](usize __i) -> bool { return __pred(__first[__i]); };
72   return __pcompact_idx<It, Out, decltype(__keep)>(__first, __out, static_cast<usize>(__last - __first), __keep);
73 }

77 filter(It __first, It __last, Out __out, Pred __pred)
78 {
79   return where<It, Out, Pred>(__first, __last, __out, __pred);
80 }

84 unique(It __first, It __last, Out __out)
85 {
86   auto __keep = [__first](usize __i) -> bool { return __i == 0 || !(__first[__i] == __first[__i - 1u]); };
87   return __pcompact_idx<It, Out, decltype(__keep)>(__first, __out, static_cast<usize>(__last - __first), __keep);
