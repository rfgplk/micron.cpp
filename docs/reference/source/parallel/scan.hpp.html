20 // out[i] = op(in[0], in[1], ..., in[i]
21 template<class In, class Out, class Op>
22 micron::task<void>
23 __pscan_incl(In __in, Out __out, usize __n, Op __op)
24 {
25   using T = micron::remove_cvref_t<decltype(*__in)>;
26   if ( __n == 0 ) co_return;
27   const usize __B = __grain_for(__n);
28   const usize __nb = (__n + __B - 1u) / __B;
29   if ( __nb <= 1u ) {      // single block: plain serial scan
30     T __acc = __in[0];
31     __out[0] = __acc;
32     for ( usize __i = 1; __i < __n; ++__i ) {
33       __acc = __op(__acc, __in[__i]);
34       __out[__i] = __acc;
35     }
36     co_return;
37   }
38 
39   micron::vector<T> __bs(__nb);
40   T *__block = &__bs[0];
41   {
42     auto __body = [__in, __n, __B, __block, __op](usize __b) {
43       const usize __s = __b * __B;
44       const usize __e = (__s + __B < __n) ? __s + __B : __n;
45       T __acc = __in[__s];
46       for ( usize __i = __s + 1; __i < __e; ++__i ) __acc = __op(__acc, __in[__i]);
47       __block[__b] = __acc;
48     };
49     co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
50   }
51   micron::vector<T> __cr(__nb);
52   T *__carry = &__cr[0];
53   {
54     T __acc = __block[0];
55     if ( __nb >= 2u ) __carry[1] = __acc;
56     for ( usize __b = 2; __b < __nb; ++__b ) {
57       __acc = __op(__acc, __block[__b - 1u]);
58       __carry[__b] = __acc;
59     }
60   }
61   {
62     auto __body = [__in, __out, __n, __B, __carry, __op](usize __b) {
63       const usize __s = __b * __B;
64       const usize __e = (__s + __B < __n) ? __s + __B : __n;
65       T __acc = (__b == 0) ? T(__in[__s]) : __op(__carry[__b], __in[__s]);
66       __out[__s] = __acc;
67       for ( usize __i = __s + 1; __i < __e; ++__i ) {
68         __acc = __op(__acc, __in[__i]);
69         __out[__i] = __acc;
70       }
71     };
72     co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
73   }
74 }

76 template<class It, class Out, class Op>
77 [[nodiscard]] micron::task<void>
78 scan(It __first, It __last, Out __out, Op __op)
79 {
80   return __pscan_incl<It, Out, Op>(__first, __out, static_cast<usize>(__last - __first), __op);
81 }

83 template<class It, class Out, class T, class Op>
84 [[nodiscard]] micron::task<void>
85 scanl(It __first, It __last, Out __out, T __init, Op __op)
86 {
87   const usize __n = static_cast<usize>(__last - __first);
88   __out[0] = __init;
89   if ( __n == 0 ) co_return;
90   co_await __pscan_incl<It, Out, Op>(__first, __out + 1, __n, __op);      // out[1+i] = op-fold(in[0..i])
91   auto __seed = __init;
92   auto __leaf = [__seed, __op](Out __a, Out __b) {
93     for ( ; __a != __b; ++__a ) *__a = __op(__seed, *__a);
94   };
95   co_await __pmap<Out, decltype(__leaf)>(__out + 1, __out + 1 + __n, __leaf, __grain_for(__n));
96 }
