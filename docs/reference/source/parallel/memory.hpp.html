19 // dst[i] = src[i]
20 template<class T, class F>
21 [[nodiscard]] micron::task<void>
22 copy_n(const T *__src, F *__dst, usize __n)
23 {
24   auto __leaf = [__src, __dst](const T *__a, const T *__b) {
25     const usize __off = static_cast<usize>(__a - __src);
26     micron::copy_n(__a, __dst + __off, static_cast<usize>(__b - __a));
27   };
28   return __pmap<const T *, decltype(__leaf)>(__src, __src + __n, __leaf, __grain_for(__n));
29 }

39   };
40   return __pmap<T *, decltype(__leaf)>(__src, __src + __n, __leaf, __grain_for(__n));

43 // count is in bytes
44 template<class T>
45 [[nodiscard]] micron::task<void>
46 set_n(T *__dst, unsigned char __val, usize __nbytes)
47 {
48   using BP = unsigned char *;
49   BP __base = reinterpret_cast<BP>(__dst);
50   auto __leaf = [__val](BP __a, BP __b) { micron::set_n(__a, __val, static_cast<usize>(__b - __a)); };
51   return __pmap<BP, decltype(__leaf)>(__base, __base + __nbytes, __leaf, __grain_for(__nbytes));
52 }

64 template<class T>
65 [[nodiscard]] micron::task<void>
66 swap_n(T *__a0, T *__b0, usize __nbytes)
67 {
68   using BP = unsigned char *;
69   BP __abase = reinterpret_cast<BP>(__a0);
70   BP __bbase = reinterpret_cast<BP>(__b0);
71   auto __leaf = [__abase, __bbase](BP __a, BP __b) {
72     const usize __off = static_cast<usize>(__a - __abase);
73     micron::swap_n(__a, __bbase + __off, static_cast<usize>(__b - __a));
74   };
75   return __pmap<BP, decltype(__leaf)>(__abase, __abase + __nbytes, __leaf, __grain_for(__nbytes));
76 }
