33 template<class T, class Cmp>
34 micron::task<void>
35 __pmergesort(T *__arr, max_t __lo, max_t __hi, Cmp __comp, usize __grain)
36 {
37   const usize __n = static_cast<usize>(__hi - __lo + 1);
38   if ( __n <= __grain ) {
39     micron::sort::__merge_sort(__arr, __lo, __hi, __comp);      // serial stable leaf
40     co_return;
41   }
42   const max_t __mid = __lo + (__hi - __lo) / 2;
43   co_await micron::coro::fork(micron::coro::discard, __pmergesort<T, Cmp>)(__arr, __lo, __mid, __comp, __grain);
44   co_await micron::coro::call(__pmergesort<T, Cmp>, __arr, __mid + 1, __hi, __comp, __grain);
45   co_await micron::coro::join;
46   micron::sort::__merge(__arr, __lo, __mid, __hi, __comp);      // serial merge of the two sorted halves
47 }

51 [[nodiscard]] micron::task<void>
52 merge(It __first, It __last, Cmp __comp = Cmp{})
53 {
54   using T = micron::remove_cvref_t<decltype(*__first)>;
55   const usize __n = static_cast<usize>(__last - __first);
56   if ( __n < 2 ) co_return;
57   co_await __pmergesort<T, Cmp>(__first, 0, static_cast<max_t>(__n) - 1, __comp, __grain_for(__n));
58 }

67 template<class T, class Cmp>
68 micron::task<void>
69 __pquicksort(T *__arr, max_t __lo, max_t __hi, max_t __depth, Cmp __comp, usize __grain)
70 {
71   if ( __lo >= __hi ) co_return;      // 0 or 1 element
72   const usize __n = static_cast<usize>(__hi - __lo + 1);
73   if ( __n <= __grain || __depth <= 0 ) {
74     micron::sort::__merge_sort(__arr, __lo, __hi, __comp);      // serial leaf (complete + stable)
75     co_return;
76   }
77   micron::sort::__q_med3_to_high(__arr, __lo, __hi, __comp);      // median-of-three pivot -> __arr[hi]
78   const max_t __p = micron::sort::__q_partition(__arr, __lo, __hi, __comp);
79   co_await micron::coro::fork(micron::coro::discard, __pquicksort<T, Cmp>)(__arr, __lo, __p - 1, __depth - 1, __comp, __grain);
80   co_await micron::coro::call(__pquicksort<T, Cmp>, __arr, __p + 1, __hi, __depth - 1, __comp, __grain);
81   co_await micron::coro::join;
82 }

84 template<class It, class Cmp = __pless>
85 [[nodiscard]] micron::task<void>
86 quick(It __first, It __last, Cmp __comp = Cmp{})
87 {
88   using T = micron::remove_cvref_t<decltype(*__first)>;
89   const usize __n = static_cast<usize>(__last - __first);
90   if ( __n < 2 ) co_return;
91   max_t __depth = 0;
92   for ( max_t __t = static_cast<max_t>(__n); __t > 1; __t >>= 1 ) __depth += 2;      // 2 * floor(log2 n)
93   co_await __pquicksort<T, Cmp>(__first, 0, static_cast<max_t>(__n) - 1, __depth, __comp, __grain_for(__n));
94 }

103 template<class It, class KeyFn>
104 micron::task<void>
105 __pradix(It __arr, usize __n, KeyFn __key)
106 {
107   using T = micron::remove_cvref_t<decltype(*__arr)>;
108   using K = micron::remove_cvref_t<decltype(__key(*__arr))>;
109   if ( __n < 2 ) co_return;
110 
111   K __kmin = __key(__arr[0]), __kmax = __key(__arr[0]);      // serial min/max (one pass)
112   for ( usize __i = 1; __i < __n; ++__i ) {
113     K __k = __key(__arr[__i]);
114     if ( __k < __kmin ) __kmin = __k;
115     if ( __kmax < __k ) __kmax = __k;
116   }
117   const umax_t __bias = static_cast<umax_t>(__kmin);
118   const umax_t __span = static_cast<umax_t>(__kmax) - static_cast<umax_t>(__kmin);
119 
120   const usize __B = __grain_for(__n);
121   const usize __nb = (__n + __B - 1u) / __B;
122   micron::vector<usize> __histv(__nb * 256u);
123   micron::vector<usize> __offv(__nb * 256u);
124   micron::vector<T> __scratchv(__n);
125   usize *__hist = &__histv[0];
126   usize *__off = &__offv[0];
127   T *__src = &__arr[0];
128   T *__dst = &__scratchv[0];
129 
130   for ( i64 __shift = 0; __shift < static_cast<i64>(sizeof(umax_t)) * 8; __shift += 8 ) {
131     if ( (__span >> __shift) == 0 ) break;      // no more significant bytes
132     // pass 1: per-block histograms (parallel)
133     {
134       auto __body = [__src, __n, __B, __hist, __shift, __bias, __key](usize __b) {
135         usize *__h = __hist + __b * 256u;
136         for ( int __i = 0; __i < 256; ++__i ) __h[__i] = 0;
137         const usize __s = __b * __B;
138         const usize __e = (__s + __B < __n) ? __s + __B : __n;
139         for ( usize __i = __s; __i < __e; ++__i ) {
140           const umax_t __k = static_cast<umax_t>(__key(__src[__i])) - __bias;
141           ++__h[(__k >> __shift) & 0xFFu];
142         }
143       };
144       co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
145     }
146     {
147       usize __running = 0;
148       for ( int __d = 0; __d < 256; ++__d )
149         for ( usize __b = 0; __b < __nb; ++__b ) {
150           __off[__b * 256u + (usize)__d] = __running;
151           __running += __hist[__b * 256u + (usize)__d];
152         }
153     }
154     {
155       auto __body = [__src, __dst, __n, __B, __off, __shift, __bias, __key](usize __b) {
156         usize *__o = __off + __b * 256u;
157         const usize __s = __b * __B;
158         const usize __e = (__s + __B < __n) ? __s + __B : __n;
159         for ( usize __i = __s; __i < __e; ++__i ) {
160           const umax_t __k = static_cast<umax_t>(__key(__src[__i])) - __bias;
161           __dst[__o[(__k >> __shift) & 0xFFu]++] = __src[__i];
162         }
163       };
164       co_await __pblocks<decltype(__body)>(0, __nb, __body, 1);
165     }
166     T *__tmp = __src;
167     __src = __dst;
168     __dst = __tmp;
169   }
170 
171   if ( __src != &__arr[0] ) {      // odd number of passes: sorted data is in scratch -> copy back (parallel)
172     T *__from = __src;
173     T *__to = &__arr[0];
174     auto __copy = [__from, __to](T *__a, T *__b) {
175       for ( T *__p = __a; __p != __b; ++__p ) __to[__p - __from] = *__p;
176     };
177     co_await __pmap<T *, decltype(__copy)>(__from, __from + __n, __copy, __grain_for(__n));
178   }
179 }

181 template<class It>
182 [[nodiscard]] micron::task<void>
183 radix(It __first, It __last)
184 {
185   using T = micron::remove_cvref_t<decltype(*__first)>;
186   if constexpr ( micron::is_floating_point_v<T> ) {
187     static_assert(sizeof(T) == 4 || sizeof(T) == 8, "parallel::sort::radix: only 32/64-bit IEEE-754 floats");
188     using U = micron::__conditional_t<sizeof(T) == 4, u32, u64>;
189     return __pradix<It>(__first, static_cast<usize>(__last - __first), [](T __e) -> U {
190       constexpr int __top = static_cast<int>(sizeof(T)) * 8 - 1;
191       const U __signbit = static_cast<U>(1) << __top;
192       const U __u = __builtin_bit_cast(U, __e);
193       return __u ^ ((static_cast<U>(0) - (__u >> __top)) | __signbit);      // order-preserving forward transform
194     });
195   } else {
196     return __pradix<It>(__first, static_cast<usize>(__last - __first), [](auto __e) { return __e; });
197   }
198 }
