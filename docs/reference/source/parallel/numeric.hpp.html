161 inner_product(It1 __f1, It1 __l1, It2 __f2, T __init)
162 {
163   if ( __f1 == __l1 ) co_return __init;

174                                                                                __grain_for(static_cast<usize>(__l1 - __f1)));

216 template<class It>
217 [[nodiscard]] micron::task<bool>
218 is_sorted(It __first, It __last)
219 {
220   using A = __sorted_acc<It>;
221   if ( __first == __last ) co_return true;
222   auto __leaf = [](It __a, It __b) -> A {
223     A __s;
224     __s.__fv = *__a;
225     __pe_t<It> __prev = *__a;
226     __s.__ok = true;
227     for ( It __p = __a + 1; __p != __b; ++__p ) {
228       if ( *__p < __prev ) __s.__ok = false;
229       __prev = *__p;
230     }
231     __s.__lv = __prev;
232     return __s;
233   };
234   auto __comb = [](A __l, A __r) -> A {
235     A __s;
236     __s.__fv = __l.__fv;
237     __s.__lv = __r.__lv;
238     __s.__ok = __l.__ok && __r.__ok && !(__r.__fv < __l.__lv);      // boundary: l.last <= r.first
239     return __s;
240   };
241   A __res = co_await __pmapreduce<It, decltype(__leaf), decltype(__comb), A>(__first, __last, __leaf, __comb,
242                                                                              __grain_for(static_cast<usize>(__last - __first)));
243   co_return __res.__ok;
244 }
