23 // an immutable / persistent LIFO stack
24 // refcounted, immutable singly-linked cons-list
25 template<is_movable_object t> class istack
26 {
27   struct node {
28     t value;
29     node *next;
30     usize refs;      // number of incoming references (heads + successor .next)
31   };
32 
33   node *head = nullptr;
34   usize len = 0;
35 
36   template<class U>
37   node *
38   __cons(U &&v, node *tail)
39   {
40     node *m = micron::alloc<node>(sizeof(node));
41     if ( !m ) [[unlikely]]
42       exc<except::library_error>("micron::istack: node allocation failed");
43 #if !defined(__micron_freestanding) || defined(__micron_eh)
44     try {
45       new (micron::addr(m->value)) t(micron::forward<U>(v));
46     } catch ( ... ) {
47       micron::free(m);
48       throw;
49     }
50 #else
51     new (micron::addr(m->value)) t(micron::forward<U>(v));
52 #endif
53     m->next = tail;
54     m->refs = 1;
55     return m;
56   }
57 
58   node *
59   __cons_default(node *tail)
60   {
61     node *m = micron::alloc<node>(sizeof(node));
62     if ( !m ) [[unlikely]]
63       exc<except::library_error>("micron::istack: node allocation failed");
64 #if !defined(__micron_freestanding) || defined(__micron_eh)
65     try {
66       new (micron::addr(m->value)) t();
67     } catch ( ... ) {
68       micron::free(m);
69       throw;
70     }
71 #else
72     new (micron::addr(m->value)) t();
73 #endif
74     m->next = tail;
75     m->refs = 1;
76     return m;
77   }
78 
79   static void
80   __drop(node *p) noexcept
81   {
82     while ( p ) {
83       if ( --p->refs != 0 ) break;
84       node *nx = p->next;
85       if constexpr ( micron::is_class<t>::value ) p->value.~t();
86       micron::free(p);
87       p = nx;
88     }
89   }
90 
91   static node *
92   __share(node *p) noexcept
93   {
94     if ( p ) ++p->refs;
95     return p;
96   }
97 
98   istack(node *h, usize l) noexcept : head(h), len(l) { }
99 
100 public:
101   using category_type = buffer_tag;
102   using mutability_type = immutable_tag;
103   using memory_type = heap_tag;
104   typedef t value_type;
105   typedef const t &reference;
106   typedef const t &ref;
107   typedef const t &const_reference;
108   typedef const t &const_ref;
109   typedef const t *pointer;
110   typedef const t *const_pointer;
111   typedef const t *iterator;
112   typedef const t *const_iterator;
113 
114   ~istack() { __drop(head); }
115 
116   istack() noexcept : head(nullptr), len(0) { }
117 
118   explicit istack(const umax_t n) : head(nullptr), len(0)
119   {
120     for ( umax_t i = 0; i < n; ++i ) {
121       head = __cons_default(head);
122       ++len;
123     }
124   }
125 
126   istack(const std::initializer_list<t> &lst) : head(nullptr), len(0)
127   {
128     for ( const t &v : lst ) {
129       head = __cons(v, head);
130       ++len;
131     }
132   }
133 
134   template<typename K = t, usize SN, class SA>
135     requires(micron::is_convertible_v<K, t>)
136   istack(const stack<K, SN, SA> &o) : head(nullptr), len(0)
137   {
138     for ( usize i = o.size(); i-- > 0; ) {
139       head = __cons(o[i], head);      // o[size-1] (bottom) first ... o[0] (top) last
140       ++len;
141     }
142   }
143 
144   istack(const istack &o) noexcept : head(__share(o.head)), len(o.len) { }
145 
146   istack(istack &&o) noexcept : head(o.head), len(o.len)
147   {
148     o.head = nullptr;
149     o.len = 0;
150   }
151 
152   istack &
153   operator=(const istack &o) noexcept
154   {
155     if ( this == &o ) return *this;
156     node *old = head;
157     head = __share(o.head);
158     len = o.len;
159     __drop(old);
160     return *this;
161   }
162 
163   istack &
164   operator=(istack &&o) noexcept
165   {
166     if ( this == &o ) return *this;
167     __drop(head);
168     head = o.head;
169     len = o.len;
170     o.head = nullptr;
171     o.len = 0;
172     return *this;
173   }
174 
175   const t &
176   top() const
177   {
178     if ( !head ) [[unlikely]]
179       exc<except::library_error>("micron::istack top() called on empty stack");
180     return head->value;
181   }
182 
183   const t &
184   operator()() const
185   {
186     return top();
187   }
188 
189   const t &
190   operator[](const umax_t n) const
191   {
192     if ( n >= len ) [[unlikely]]
193       exc<except::library_error>("micron::istack operator[] out of range");
194     const node *p = head;
195     for ( umax_t i = 0; i < n; ++i ) p = p->next;
196     return p->value;
197   }
198 
199   [[nodiscard]] istack
200   push(const t &v) const
201   {
202     node *m = const_cast<istack *>(this)->__cons(v, head);
203     __share(head);      // *this keeps head; the new node also references it
204     return istack(m, len + 1);
205   }
206 
207   [[nodiscard]] istack
208   push(t &&v) const
209   {
210     node *m = const_cast<istack *>(this)->__cons(micron::move(v), head);
211     __share(head);
212     return istack(m, len + 1);
213   }
214 
215   [[nodiscard]] istack
216   push() const
217   {
218     node *m = const_cast<istack *>(this)->__cons_default(head);
219     __share(head);
220     return istack(m, len + 1);
221   }
222 
223   [[nodiscard]] istack
224   pop() const
225   {
226     if ( !head ) [[unlikely]]
227       exc<except::library_error>("micron::istack pop() called on empty stack");
228     return istack(__share(head->next), len - 1);
229   }
230 
231   [[nodiscard]] istack
232   clear() const
233   {
234     return istack();
235   }
236 
237   void
238   swap(istack &o) noexcept
239   {
240     micron::swap(head, o.head);
241     micron::swap(len, o.len);
242   }
243 
244   [[nodiscard]] bool
245   empty() const noexcept
246   {
247     return head == nullptr;
248   }
249 
250   [[nodiscard]] usize
251   size() const noexcept
252   {
253     return len;
254   }
255 
256   [[nodiscard]] constexpr usize
257   max_size() const noexcept
258   {
259     return static_cast<usize>(-1);
260   }
261 
262   static constexpr bool
263   is_pod()
264   {
265     return micron::is_pod_v<t>;
266   }
267 };
