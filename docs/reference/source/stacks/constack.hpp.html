28 // coarse-grained, mutex-synchronized LIFO stack
29 template<is_regular_object T, usize N = micron::alloc_auto_sz, class Alloc = micron::allocator_serial<>>
30 class constack: public __mutable_memory_resource<T, Alloc>
31 {
32   using __mem = __mutable_memory_resource<T, Alloc>;
33   mutable micron::fast_mutex __mtx;
34 
35   using __defer = micron::unique_lock<micron::lock_starts::defer, micron::fast_mutex>;

37   struct __hold {
38     micron::fast_mutex &m;
39 
40     [[gnu::always_inline]] explicit __hold(micron::fast_mutex &mm) noexcept : m(mm) { m.lock(); }
41 
42     [[gnu::always_inline]] ~__hold() noexcept { m.unlock(); }
43 
44     __hold(const __hold &) = delete;
45     __hold &operator=(const __hold &) = delete;
46   };

48   // NOTE: lock two deferred guards in a deadlock-free order
49   static inline void
50   __lock_ordered(micron::fast_mutex &a, __defer &la, micron::fast_mutex &b, __defer &lb)
51   {
52     if ( micron::addr(a) == micron::addr(b) ) {
53       la.lock();
54     } else if ( static_cast<const void *>(micron::addr(a)) < static_cast<const void *>(micron::addr(b)) ) {
55       la.lock();
56       lb.lock();
57     } else {
58       lb.lock();
59       la.lock();
60     }
61   }

63   inline void
64   __reserve_unsafe(const usize n)
65   {
66     if ( n < __mem::capacity ) return;
67     __mem::expand(n);
68   }

70   inline void
71   __ensure_one_unsafe()
72   {
73     if ( __mem::length >= __mem::capacity )
74       __reserve_unsafe(__mem::recommended_capacity(__mem::capacity, allocation_add_or_throw(__mem::length, 1)));
75   }

77   inline void
78   __clear_unsafe()
79   {
80     if ( !__mem::length ) return;
81     __impl_container::destroy(micron::addr(__mem::memory[0]), __mem::length);
82     __mem::length = 0;
83   }

85   inline void
86   __push_unsafe(const T &v)
87   {
88     __ensure_one_unsafe();
89     new (micron::addr(__mem::memory[__mem::length++])) T(v);
90   }

92   inline void
93   __push_unsafe(T &&v)
94   {
95     __ensure_one_unsafe();
96     new (micron::addr(__mem::memory[__mem::length++])) T(micron::move(v));
97   }

99   inline T
100   __pop_unsafe()
101   {
102     T val = micron::move(__mem::memory[__mem::length - 1]);
103     if constexpr ( micron::is_class_v<T> ) __mem::memory[__mem::length - 1].~T();
104     czero<sizeof(T) / sizeof(byte)>((byte *)micron::voidify(&__mem::memory[__mem::length-- - 1]));
105     return val;
106   }

108   static inline umax_t
109   __checked_count(const umax_t n)
110   {
111     if constexpr ( sizeof(T) > 1 ) {
112       if ( n > (static_cast<umax_t>(-1) / sizeof(T)) ) [[unlikely]]
113         exc<except::library_error>("micron::constack size overflow");
114     }
115     return n;
116   }

132     if ( __mem::is_zero() ) return;

138   explicit constack(const umax_t n) : __mem(__checked_count(n))
139   {

143   constack(const std::initializer_list<T> &lst) : __mem(lst.size())
144   {
145     if constexpr ( micron::is_class_v<T> ) {
146       usize i = 0;
147       for ( const T &value : lst ) new (micron::addr(__mem::memory[i++])) T(value);
148       __mem::length = lst.size();
149     } else {
150       usize i = 0;
151       for ( T value : lst ) __mem::memory[i++] = value;
152       __mem::length = lst.size();
153     }
154   }

156   constack(const constack &o) : __mem(nullptr)
157   {
158     __hold lo(o.__mtx);
159     if ( o.length ) {
160       __reserve_unsafe(o.length);
161       __impl_container::copy(__mem::memory, o.memory, o.length);
162       __mem::length = o.length;
163     }
164   }
165 
166   constack(constack &&o) : __mem(nullptr)
167   {
168     __hold lo(o.__mtx);
169     __mem::memory = o.memory;
170     __mem::length = o.length;
171     __mem::capacity = o.capacity;
172     o.memory = nullptr;
173     o.length = 0;
174     o.capacity = 0;
175   }

177   constack &
178   operator=(const constack &o)
179   {
180     if ( this == micron::addr(o) ) return *this;
181     __defer la(__mtx), lb(o.__mtx);
182     __lock_ordered(__mtx, la, o.__mtx, lb);
183     __clear_unsafe();      // destroy our current elements; slots become raw storage
184     if ( o.length > __mem::capacity ) __reserve_unsafe(o.length);
185     __impl_container::copy(__mem::memory, o.memory, o.length);      // placement-construct
186     __mem::length = o.length;
187     return *this;
188   }
189 
190   constack &
191   operator=(constack &&o)
192   {
193     if ( this == micron::addr(o) ) return *this;
194     __defer la(__mtx), lb(o.__mtx);
195     __lock_ordered(__mtx, la, o.__mtx, lb);
196     __clear_unsafe();      // destroy our live elements before releasing the buffer (free() would leak them)
197     if ( __mem::memory ) __mem::free();
198     __mem::memory = o.memory;
199     __mem::length = o.length;
200     __mem::capacity = o.capacity;
201     o.memory = nullptr;
202     o.length = 0;
203     o.capacity = 0;
204     return *this;
205   }

289   inline void
290   pop_range(Args &...args)
291   {
292     static_assert((micron::is_same_v<micron::remove_cvref_t<Args>, T> && ...),
293                   "micron::constack pop_range(): all output types must match value_type");
294     __hold lk(__mtx);
295     if ( sizeof...(Args) > __mem::length ) [[unlikely]]
296       exc<except::library_error>("micron::constack pop_range(): not enough elements");

314   inline void
315   swap(constack &o) noexcept
316   {
317     if ( this == micron::addr(o) ) return;
318     __defer la(__mtx), lb(o.__mtx);
319     __lock_ordered(__mtx, la, o.__mtx, lb);
320     micron::swap(__mem::memory, o.memory);
321     micron::swap(__mem::length, o.length);
322     micron::swap(__mem::capacity, o.capacity);
323   }

364   bool
365   operator==(const constack &o) const noexcept
366   {
367     if ( this == micron::addr(o) ) return true;
368     __defer la(__mtx), lb(o.__mtx);
369     __lock_ordered(__mtx, la, o.__mtx, lb);
370     if ( __mem::length != o.length ) return false;
371     for ( usize i = 0; i < __mem::length; ++i )
372       if ( __mem::memory[i] != o.memory[i] ) return false;
373     return true;
374   }

382   bool
383   operator<(const constack &o) const noexcept
384   {
385     if ( this == micron::addr(o) ) return false;
386     __defer la(__mtx), lb(o.__mtx);
387     __lock_ordered(__mtx, la, o.__mtx, lb);
388     usize n = __mem::length < o.length ? __mem::length : o.length;
389     for ( usize i = 0; i < n; ++i ) {
390       if ( __mem::memory[i] < o.memory[i] ) return true;
391       if ( o.memory[i] < __mem::memory[i] ) return false;
392     }
393     return __mem::length < o.length;
394   }
