13 // reference this for exact ABI compatibility rather than the VERSIONs below
14 constexpr static const int MICRON_ABI = 9;

17 constexpr static const int MICRON_VERSION_MAJOR = 0x1000;
18 constexpr static const int MICRON_VERSION_MINOR = 0x0A00;
19 constexpr static const int MICRON_VERSION_PATCH = 0x0020;
20 constexpr static const int MICRON_VERSION_HOTFIX = 0x0000;
21 
22 enum kernel_support : int { kernel_linux = 0, kernel_darwin = 1, kernel_obsd, kernel_netbsd, kernel_freebsd, kernel_none };
23 
24 constexpr static const int MICRON_KERNEL_TYPE = kernel_linux;

26 template<int __major, int __minor, int __patch>
27 constexpr bool
28 is_version()
29 {
30   if constexpr ( __major == MICRON_VERSION_MAJOR and __minor == MICRON_VERSION_MINOR and __patch == MICRON_VERSION_PATCH ) return true;
31   return false;
32 }
33 
34 template<int __hotfix>
35 constexpr bool
36 is_hotfix()
37 {
38   if constexpr ( __hotfix == MICRON_VERSION_HOTFIX ) return true;
39   return false;
40 }
41 
42 constexpr int
43 get_version(void)
44 {
45   return MICRON_VERSION_MAJOR | MICRON_VERSION_MINOR | MICRON_VERSION_PATCH | MICRON_VERSION_HOTFIX;
46 }
