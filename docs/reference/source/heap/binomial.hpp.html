99   template<typename... A>
100   [[gnu::always_inline]] inline node_type *
101   __acquire(A &&...a)
102   {
103     if ( __pool ) {
104       node_type *slot = __pool;
105       __pool = *reinterpret_cast<node_type **>(slot);

119   inline void
120   __drain_pool() noexcept
121   {
122     while ( __pool ) {
123       node_type *nxt = *reinterpret_cast<node_type **>(__pool);

125       __pool = nxt;

129   inline void
130   link(node_type *y, node_type *z)
131   {
132     y->parent = z;
133     y->sibling = z->child;
134     z->child = y;
135     z->degree++;

138   inline node_type *
139   merge_root_lists(node_type *h1, node_type *h2)
140   {
141     if ( !h1 ) return h2;
142     if ( !h2 ) return h1;
143 
144     node_type *head;
145     node_type **pos = &head;
146 
147     while ( h1 && h2 ) {
148       if ( h1->degree <= h2->degree ) {
149         *pos = h1;
150         h1 = h1->sibling;
151       } else {
152         *pos = h2;
153         h2 = h2->sibling;
154       }
155       pos = &((*pos)->sibling);
156     }
157 
158     *pos = h1 ? h1 : h2;
159     return head;
160   }

162   inline void
163   union_heap(__binomial_heap &other)
164   {
165     if ( !other.root_list ) return;
166 
167     root_list = merge_root_lists(root_list, other.root_list);
168     other.root_list = nullptr;
169 
170     if ( !root_list ) {
171       min_node = nullptr;
172       node_count += other.node_count;
173       other.node_count = 0;
174       return;
175     }
176 
177     node_type *prev = nullptr;
178     node_type *curr = root_list;
179     node_type *next = curr->sibling;
180 
181     while ( next ) {
182       if ( (curr->degree != next->degree) || (next->sibling && next->sibling->degree == curr->degree) ) {
183         prev = curr;
184         curr = next;
185       } else {
186         if ( comp(curr->value, next->value) ) {
187           curr->sibling = next->sibling;
188           link(next, curr);
189         } else {
190           if ( !prev )
191             root_list = next;
192           else
193             prev->sibling = next;
194           link(curr, next);
195           curr = next;
196         }
197       }
198       next = curr->sibling;
199     }
200 
201     update_min();
202     node_count += other.node_count;
203     other.node_count = 0;
204     other.min_node = nullptr;
205   }

207   inline void
208   update_min()
209   {
210     if ( !root_list ) {
211       min_node = nullptr;
212       return;
213     }
214 
215     min_node = root_list;
216     node_type *curr = root_list->sibling;
217 
218     while ( curr ) {
219       if ( comp(curr->value, min_node->value) ) min_node = curr;
220       curr = curr->sibling;
221     }
222   }

271   inline T
272   extract_min()
273   {
274     if ( !min_node ) exc<except::library_error>("micron::__binomial_heap::extract_min() is empty");
275 
276     T min_value = micron::move(min_node->value);
277 
278     node_type *prev = nullptr;
279     node_type *curr = root_list;
280 
281     while ( curr != min_node ) {
282       prev = curr;
283       curr = curr->sibling;
284     }
285 
286     if ( !prev )
287       root_list = min_node->sibling;
288     else
289       prev->sibling = min_node->sibling;
290 
291     node_type *child = min_node->child;
292     node_type *reversed = nullptr;
293 
294     while ( child ) {
295       node_type *next = child->sibling;
296       child->sibling = reversed;
297       child->parent = nullptr;
298       reversed = child;
299       child = next;
300     }
301 
302     __binomial_heap child_heap(comp);
303     child_heap.root_list = reversed;
304     child_heap.node_count = 0;
305 
306     __recycle(min_node);
307     node_count--;
308 
309     union_heap(child_heap);
310 
311     if ( !reversed ) update_min();
312 
313     return min_value;
314   }

316   inline void
317   decrease_key(node_type *node, const T &new_value)
318   {
319     if ( !comp(new_value, node->value) ) return;
320 
321     node->value = new_value;
322 
323     node_type *curr = node;
324     node_type *parent = curr->parent;
325 
326     while ( parent && comp(curr->value, parent->value) ) {
327       micron::swap(curr->value, parent->value);
328       curr = parent;
329       parent = curr->parent;
330     }
331 
332     if ( !min_node || comp(curr->value, min_node->value) ) min_node = curr;
333   }

335   inline void
336   delete_node(node_type *node)
337   {
338     if ( !node ) return;
339 
340     node_type *curr = node;
341     node_type *parent = curr->parent;
342 
343     while ( parent ) {
344       micron::swap(curr->value, parent->value);
345       curr = parent;
346       parent = curr->parent;
347     }
348 
349     min_node = curr;
350     extract_min();
351   }

359   inline node_type *
360   find_recursive(node_type *root, const T &value)
361   {
362     if ( !root ) return nullptr;

364     if ( root->value == value ) return root;
365
