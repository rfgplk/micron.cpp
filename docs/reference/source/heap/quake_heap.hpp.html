98   [[gnu::always_inline]] inline node *
99   __acquire_node()
100   {
101     if ( __node_pool ) {
102       __pool_link *slot = __node_pool;
103       __node_pool = slot->next;
104       const usize allocation_len = slot->allocation_len;
105       return new (static_cast<void *>(slot)) node{ nullptr, nullptr, nullptr, nullptr, 0, 0, allocation_len };
106     }
107     chunk<byte> storage = __allocator_create<Alloc, alignof(node)>(sizeof(node));
108     return new (static_cast<void *>(storage.ptr)) node{ nullptr, nullptr, nullptr, nullptr, 0, 0, storage.len };
109   }

111   [[gnu::always_inline]] inline void
112   __recycle_node(node *n) noexcept
113   {
114     const usize allocation_len = n->allocation_len;
115     n->~node();
116     __node_pool = new (static_cast<void *>(n)) __pool_link{ __node_pool, allocation_len };
117   }

119   template<typename... A>
120   [[gnu::always_inline]] inline entry *
121   __acquire_entry(A &&...a)
122   {
123     entry *e;
124     if ( __entry_pool ) {
125       __pool_link *slot = __entry_pool;
126       __entry_pool = slot->next;
127       e = reinterpret_cast<entry *>(slot);
128     } else {
129       chunk<byte> storage = __allocator_create<Alloc, alignof(entry)>(sizeof(entry));
130       e = reinterpret_cast<entry *>(storage.ptr);
131       new (static_cast<void *>(e)) __pool_link{ nullptr, storage.len };
132     }
133     const usize allocation_len = reinterpret_cast<__pool_link *>(e)->allocation_len;
134 #if !defined(__micron_freestanding) || defined(__micron_eh)
135     try {
136 #endif
137       new (static_cast<void *>(e)) entry{ T(micron::forward<A>(a)...), nullptr, allocation_len };
138 #if !defined(__micron_freestanding) || defined(__micron_eh)
139     } catch ( ... ) {
140       __allocator_destroy<Alloc, alignof(entry)>({ reinterpret_cast<byte *>(e), allocation_len });
141       throw;
142     }
143 #endif
144     return e;
145   }

155   inline void
156   __drain_pools() noexcept
157   {
158     while ( __node_pool ) {
159       __pool_link *nx = __node_pool->next;
160       const usize allocation_len = __node_pool->allocation_len;
161       __allocator_destroy<Alloc, alignof(node)>({ reinterpret_cast<byte *>(__node_pool), allocation_len });
162       __node_pool = nx;
163     }
164     while ( __entry_pool ) {
165       __pool_link *nx = __entry_pool->next;
166       const usize allocation_len = __entry_pool->allocation_len;
167       __allocator_destroy<Alloc, alignof(entry)>({ reinterpret_cast<byte *>(__entry_pool), allocation_len });
168       __entry_pool = nx;
169     }
170   }

172   [[gnu::always_inline]] inline void
173   __push_root(node *n)
174   {
175     n->fpos = static_cast<u32>(forest.size());
176     n->parent = nullptr;
177     forest.push_back(n);
178     if constexpr ( __simd_on ) fkeys.push_back(n->e->value);
179   }

181   [[gnu::always_inline]] inline void
182   __erase_root(u32 pos)
183   {
184     const u32 last = static_cast<u32>(forest.size()) - 1;
185     if ( pos != last ) {
186       node *m = forest[last];
187       forest[pos] = m;
188       m->fpos = pos;
189       if constexpr ( __simd_on ) fkeys[pos] = fkeys[last];
190     }
191     forest.pop_back();
192     if constexpr ( __simd_on ) fkeys.pop_back();
193   }

195   inline void
196   __bump_levels(u32 h) noexcept
197   {
198     if ( h + 1 > levels ) levels = h + 1;
199   }

201   node *
202   __link(node *a, node *b)
203   {
204     node *w, *l;
205     if ( comp(b->e->value, a->e->value) ) {
206       w = b;
207       l = a;
208     } else {
209       w = a;
210       l = b;
211     }
212     node *p = __acquire_node();
213     p->e = w->e;
214     p->left = w;
215     p->right = l;
216     p->parent = nullptr;
217     p->height = w->height + 1;
218     w->parent = p;
219     l->parent = p;
220     w->e->high = p;
221     if ( p->height < kMaxLevels ) counts[p->height]++;
222     __bump_levels(p->height);
223     return p;
224   }

226   void
227   __shake(node *r)
228   {
229     __erase_root(r->fpos);
230     entry *te = r->e;
231     node *cur = r;
232     while ( cur ) {
233       if ( cur->height < kMaxLevels && counts[cur->height] ) counts[cur->height]--;
234       node *L = cur->left;
235       node *R = cur->right;
236       node *win = nullptr;
237       if ( L && L->e == te ) {
238         win = L;
239         if ( R ) __push_root(R);
240       } else if ( R && R->e == te ) {
241         win = R;
242         if ( L ) __push_root(L);
243       } else {      // leaf (no child carries te)
244         if ( L ) __push_root(L);
245         if ( R ) __push_root(R);
246       }
247       __recycle_node(cur);
248       cur = win;
249     }
250   }

252   void
253   __consolidate()
254   {
255     node *by_h[kMaxLevels];
256     for ( u32 i = 0; i < kMaxLevels; ++i ) by_h[i] = nullptr;
257 
258     const u32 cnt = static_cast<u32>(forest.size());
259     for ( u32 i = 0; i < cnt; ++i ) {
260       node *r = forest[i];
261       r->parent = nullptr;
262       u32 h = r->height;
263       while ( h < kMaxLevels && by_h[h] ) {
264         node *o = by_h[h];
265         by_h[h] = nullptr;
266         r = __link(r, o);
267         h = r->height;
268       }
269       if ( h < kMaxLevels ) by_h[h] = r;
270     }
271 
272     forest.clear();
273     if constexpr ( __simd_on ) fkeys.clear();
274     for ( u32 h = 0; h < kMaxLevels; ++h )
275       if ( by_h[h] ) __push_root(by_h[h]);
276   }

278   void
279   __decap(node *n, u32 keep)
280   {
281     if ( n->height <= keep ) {
282       n->parent = nullptr;
283       n->e->high = n;
284       __push_root(n);
285       return;
286     }
287     if ( n->left ) __decap(n->left, keep);
288     if ( n->right ) __decap(n->right, keep);
289     __recycle_node(n);
290   }

292   void
293   __quake()
294   {
295     if ( levels < 2 ) return;
296     const usize viol = micron::simd::first_quake_violation(counts, levels, AlphaNum, AlphaDen);
297     if ( viol >= levels ) return;      // invariant holds everywhere
298 
299     const u32 keep = static_cast<u32>(viol);
300 
301     node *roots[kMaxLevels];
302     u32 rc = 0;
303     const u32 cnt = static_cast<u32>(forest.size());
304     for ( u32 i = 0; i < cnt && rc < kMaxLevels; ++i ) roots[rc++] = forest[i];
305 
306     forest.clear();
307     if constexpr ( __simd_on ) fkeys.clear();
308     for ( u32 h = keep + 1; h < levels; ++h ) counts[h] = 0;
309     levels = keep + 1;
310 
311     for ( u32 i = 0; i < rc; ++i ) __decap(roots[i], keep);
312   }

314   inline void
315   __update_min()
316   {
317     if ( forest.size() == 0 ) {
318       min_node = nullptr;
319       return;
320     }
321     usize idx;
322     if constexpr ( __simd_on ) {
323       idx = __simd_min ? micron::simd::argmin<T>(fkeys.data(), fkeys.size()) : micron::simd::argmax<T>(fkeys.data(), fkeys.size());
324     } else {
325       idx = 0;
326       const usize m = forest.size();
327       for ( usize k = 1; k < m; ++k )
328         if ( comp(forest[k]->e->value, forest[idx]->e->value) ) idx = k;
329     }
330     min_node = forest[idx];
331   }

333   void
334   __free_subtree(node *n) noexcept
335   {
336     if ( !n ) return;
337     __free_subtree(n->left);
338     __free_subtree(n->right);
339     if ( n->e->high == n ) __recycle_entry(n->e);
340     __recycle_node(n);
341   }

370   handle
371   __insert_entry(entry *e)
372   {
373     node *n = __acquire_node();
374     n->e = e;
375     n->left = nullptr;
376     n->right = nullptr;
377     n->parent = nullptr;
378     n->height = 0;
379     e->high = n;
380     __push_root(n);
381     if ( counts[0]++ == 0 && levels < 1 ) levels = 1;
382     __bump_levels(0);
383     ++total;
384     if ( !min_node || comp(e->value, min_node->e->value) ) min_node = n;
385     return handle{ e };

487   void
488   decrease_key(handle h, const T &nv)
489   {
490     entry *e = h.e;
491     if ( !e ) return;
492     if ( !comp(nv, e->value) ) return;
493     e->value = nv;
494     node *x = e->high;
495     if ( x->parent ) {
496       node *p = x->parent;
497       if ( p->left == x )
498         p->left = nullptr;
499       else if ( p->right == x )
500         p->right = nullptr;
501       __push_root(x);
502     } else if constexpr ( __simd_on ) {
503       fkeys[x->fpos] = e->value;
504     }
505     if ( !min_node || comp(e->value, min_node->e->value) ) min_node = x;
506   }

508   T
509   extract_min()
510   {
511     if ( !min_node ) exc<except::library_error>("quake_heap::extract_min() empty");
512     node *r = min_node;
513     entry *e = r->e;
514     T result = micron::move(e->value);
515     __shake(r);
516     __recycle_entry(e);
517     --total;
518     min_node = nullptr;
519     __consolidate();
520     __quake();
521     __update_min();
522     return result;
523   }

525   void
526   erase(handle h)
527   {
528     entry *e = h.e;
529     if ( !e || total == 0 ) return;
530     node *x = e->high;
531     if ( x->parent ) {
532       node *p = x->parent;
533       if ( p->left == x )
534         p->left = nullptr;
535       else if ( p->right == x )
536         p->right = nullptr;
537       __push_root(x);
538     }
539     min_node = x;
540     (void)extract_min();
541   }

543   void
544   meld(quake_heap &&o)
545   {
546     if ( o.total == 0 ) return;
547     const u32 ocnt = static_cast<u32>(o.forest.size());
548     node *omin = o.min_node;
549     for ( u32 i = 0; i < ocnt; ++i ) __push_root(o.forest[i]);
550     for ( u32 h = 0; h < o.levels; ++h ) counts[h] += o.counts[h];
551     if ( o.levels > levels ) levels = o.levels;
552     total += o.total;
553     if ( !min_node || (omin && comp(omin->e->value, min_node->e->value)) ) min_node = omin;
554 
555     o.forest.set_size(0);
556     if constexpr ( __simd_on ) o.fkeys.set_size(0);
557     o.min_node = nullptr;
558     o.total = 0;
559     for ( u32 i = 0; i < kMaxLevels; ++i ) o.counts[i] = 0;
560     o.levels = 0;
561   }

563   void
564   clear() noexcept
565   {
566     const usize m = forest.size();
567     for ( usize i = 0; i < m; ++i ) __free_subtree(forest[i]);
568     forest.clear();
569     if constexpr ( __simd_on ) fkeys.clear();
570     __zero_counts();
571     total = 0;
572     min_node = nullptr;
