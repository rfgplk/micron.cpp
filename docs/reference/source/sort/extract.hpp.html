25 template<typename C>
26 concept __range_harvestable = !is_extractable_heap<C> && requires(const micron::remove_cvref_t<C> &c) {
27   c.begin();
28   c.end();
29   { *c.begin() };
30 };
31 
32 template<typename C>
33 concept __for_each_harvestable = !is_extractable_heap<C> && requires {
34   typename micron::remove_cvref_t<C>::value_type;
35 } && !requires(const micron::remove_cvref_t<C> &c) {
36   c.begin();
37   c.end();
38 } && requires(const micron::remove_cvref_t<C> &c) { c.for_each([](const typename micron::remove_cvref_t<C>::value_type &) { }); };
39 
40 template<__range_harvestable C, typename Cmp>
41 auto
42 sorted(const C &c, Cmp comp) -> micron::vector<micron::remove_cvref_t<decltype(*c.begin())>>
43 {
44   using E = micron::remove_cvref_t<decltype(*c.begin())>;
45   micron::vector<E> out;
46   for ( auto it = c.begin(); it != c.end(); ++it ) out.push_back(*it);
47   sort(out, comp);
48   return out;
49 }

51 template<__range_harvestable C>
52 auto
53 sorted(const C &c) -> micron::vector<micron::remove_cvref_t<decltype(*c.begin())>>
54 {
55   return sorted(c, [](const auto &a, const auto &b) { return a < b; });
56 }

58 template<__for_each_harvestable C, typename Cmp>
59 auto
60 sorted(const C &c, Cmp comp) -> micron::vector<typename micron::remove_cvref_t<C>::value_type>
61 {
62   using E = typename micron::remove_cvref_t<C>::value_type;
63   micron::vector<E> out;
64   c.for_each([&](const E &e) { out.push_back(e); });
65   sort(out, comp);
66   return out;
67 }

94 template<is_extractable_heap H>
95 auto
96 drain_sorted(H &h) -> micron::vector<typename micron::remove_cvref_t<H>::value_type>
97 {
98   using E = typename micron::remove_cvref_t<H>::value_type;
99   micron::vector<E> out;
100   while ( h.size() > 0 ) {
101     if constexpr ( requires { h.extract_min(); } )
102       out.push_back(h.extract_min());
103     else if constexpr ( requires { h.get(); } )
104       out.push_back(h.get());
105     else
106       out.push_back(h.pop());
107   }
108   return out;
109 }
