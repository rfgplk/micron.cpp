8 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
9 // bitonic sort
10 //   time:  O(n log^2 n) all cases
11 //   space: O(1) in-place
12 //   stable: no
13 //   power-of-two lengths only (non-pow2 -> insertion fallback)
14 //
15 //   default ascending path is SIMD-vectorised ~2.5-3.6x (AVX2/NEON) for fundamental keys
16

29 template<is_iterable_container T, typename Cmp>
30 T &
31 __bitonic(T &arr, Cmp comp)
32 {
33   const max_t n = static_cast<max_t>(arr.size());
34   if ( n < 2 ) return arr;
35   if ( (n & (n - 1)) != 0 ) {      // not a power of two
36     __insertion(arr, 0, arr.size(), comp);
37     return arr;
38   }
39   for ( max_t k = 2; k <= n; k <<= 1 ) {
40     for ( max_t j = k >> 1; j > 0; j >>= 1 ) {
41       for ( max_t i = 0; i < n; ++i ) {
42         const max_t l = i ^ j;
43         if ( l > i ) {
44           const bool ascending = ((i & k) == 0);
45           const bool out_of_order = ascending ? comp(arr[l], arr[i]) : comp(arr[i], arr[l]);
46           if ( out_of_order ) micron::swap(arr[i], arr[l]);
47         }
48       }
49     }
50   }
51   return arr;
52 }

54 template<is_iterable_container T>
55 T &
56 bitonic(T &arr)
57 {
58   using V = typename T::value_type;
59   if constexpr ( micron::simd::__net_has_simd<V>() ) {
60     const usize n = arr.size();
61     if ( n >= 2 && (n & (n - 1)) == 0 ) {      // power of two + SIMD key -> vectorised network
62       micron::simd::bitonic_sort_pow2(arr.data(), n);
63       return arr;
64     }
65   }
66   return __bitonic(arr, [](const auto &a, const auto &b) { return a < b; });
67 }

69 template<is_iterable_container T, is_valid_comp<T> Cmp>
70 T &
71 bitonic(T &arr, Cmp comp)
72 {
73   return __bitonic(arr, comp);
