30 template<typename U> struct __sort_scratch {
31   U *p;
32 
33   explicit __sort_scratch(usize n) : p(n ? new U[n] : nullptr) { }
34 
35   ~__sort_scratch() { delete[] p; }
36 
37   __sort_scratch(const __sort_scratch &) = delete;
38   __sort_scratch &operator=(const __sort_scratch &) = delete;

41   get() noexcept
42   {
43     return p;
44   }

47 template<is_iterable_container T, typename KeyFn>
48 void
49 __radix_pass(T &arr, T &carr, usize n, i64 shift, umax_t bias_min, KeyFn key)
50 {
51   usize bckt[256] = {};
52 
53   for ( usize i = 0; i < n; ++i ) {
54     umax_t k = static_cast<umax_t>(static_cast<umax_t>(key(arr[i])) - bias_min);
55     bckt[(k >> shift) & 0xFFu]++;
56   }
57   for ( i32 i = 1; i < 256; ++i ) bckt[i] += bckt[i - 1];
58 
59   for ( usize k = n; k-- > 0; ) {
60     umax_t kk = static_cast<umax_t>(static_cast<umax_t>(key(arr[k])) - bias_min);
61     carr[--bckt[(kk >> shift) & 0xFFu]] = arr[k];
62   }
63   micron::memcpy(&arr[0], &carr[0], n);      // element-count copy-back
64 }

66 template<is_iterable_container T, typename KeyFn>
67 void
68 __radix_int(T &arr, KeyFn key)
69 {
70   using key_t = micron::remove_cvref_t<decltype(key(arr[0]))>;
71   const usize n = arr.size();
72   if ( n < 2 ) return;      // empty / single: sorted
73 
74   key_t kmin = key(arr[0]), kmax = key(arr[0]);
75   for ( usize i = 1; i < n; ++i ) {
76     key_t k = key(arr[i]);
77     if ( k < kmin ) kmin = k;
78     if ( kmax < k ) kmax = k;
79   }
80   const umax_t bias_min = static_cast<umax_t>(kmin);
81   const umax_t span = static_cast<umax_t>(static_cast<umax_t>(kmax) - static_cast<umax_t>(kmin));      // biased max
82 
83   T carr = arr;      // scratch (overwritten each pass)
84   for ( i64 shift = 0; shift < static_cast<i64>(sizeof(umax_t)) * 8; shift += 8 ) {
85     if ( (span >> shift) == 0 ) break;      // no more significant bytes
86     __radix_pass(arr, carr, n, shift, bias_min, key);
87   }
88 }

106 template<is_iterable_container T>
107   requires micron::is_integral_v<typename T::value_type>
108 void
109 radix(typename T::iterator start, typename T::iterator end)
110 {
111   const max_t cnt = static_cast<max_t>(end - start);
112   if ( cnt < 2 ) return;
113 
114   using value_t = typename T::value_type;
115   value_t vmin = start[0], vmax = start[0];
116   for ( max_t i = 1; i < cnt; ++i ) {
117     if ( start[i] < vmin ) vmin = start[i];
118     if ( vmax < start[i] ) vmax = start[i];
119   }
120   const umax_t bias_min = static_cast<umax_t>(vmin);
121   const umax_t span = static_cast<umax_t>(static_cast<umax_t>(vmax) - static_cast<umax_t>(vmin));
122 
123   T carr(static_cast<typename T::size_type>(cnt));
124   for ( i64 shift = 0; shift < static_cast<i64>(sizeof(umax_t)) * 8; shift += 8 ) {
125     if ( (span >> shift) == 0 ) break;
126     usize bckt[256] = {};
127     for ( max_t i = 0; i < cnt; ++i ) bckt[(static_cast<umax_t>(static_cast<umax_t>(start[i]) - bias_min) >> shift) & 0xFFu]++;
128     for ( i32 i = 1; i < 256; ++i ) bckt[i] += bckt[i - 1];
129     for ( max_t k = cnt; k-- > 0; )
130       carr[--bckt[(static_cast<umax_t>(static_cast<umax_t>(start[k]) - bias_min) >> shift) & 0xFFu]] = start[k];
131     micron::memcpy(start, &carr[0], static_cast<u64>(cnt));
132   }
133 }

135 template<is_iterable_container T>
136   requires(micron::is_floating_point_v<typename T::value_type>
137            && (sizeof(typename T::value_type) == 4 || sizeof(typename T::value_type) == 8))
138 void
139 fradix(T &arr)
140 {
141   using V = typename T::value_type;
142   using U = micron::__conditional_t<sizeof(V) == 4, u32, u64>;
143   constexpr i64 W = static_cast<i64>(sizeof(V));
144   constexpr i64 TOP = W * 8 - 1;
145   const U signbit = static_cast<U>(1) << TOP;
146 
147   const usize n = arr.size();
148   if ( n < 2 ) return;
149 
150   __sort_scratch<U> kbuf(n), tbuf(n);
151   U *keys = kbuf.get();
152   U *tmp = tbuf.get();
153 
154   for ( usize i = 0; i < n; ++i ) {
155     U u = 0;
156     micron::bytecpy(&u, &arr[i], W);                                 // bit-pun float -> unsigned
157     keys[i] = u ^ ((static_cast<U>(0) - (u >> TOP)) | signbit);      // order-preserving forward transform
158   }
159 
160   for ( i64 b = 0; b < W; ++b ) {
161     i32 cnt[256] = {};
162     for ( usize i = 0; i < n; ++i ) cnt[static_cast<byte>(keys[i] >> (b * 8))]++;
163     for ( i32 i = 1; i < 256; ++i ) cnt[i] += cnt[i - 1];
164     for ( usize i = n; i-- > 0; ) tmp[--cnt[static_cast<byte>(keys[i] >> (b * 8))]] = keys[i];
165     U *swap = keys;
166     keys = tmp;
167     tmp = swap;
168   }
169 
170   for ( usize i = 0; i < n; ++i ) {
171     U u = keys[i] ^ (((keys[i] >> TOP) - static_cast<U>(1)) | signbit);      // inverse transform
172     micron::bytecpy(&arr[i], &u, W);                                         // bit-pun back to float
173   }
174 }

176 template<is_iterable_container T, typename KeyFn>
177   requires(is_floating_key_fn<KeyFn, T> && micron::is_trivially_copyable_v<typename T::value_type>)
178 void
179 fradix(T &arr, KeyFn key)
180 {
181   using V = micron::remove_cvref_t<decltype(key(arr[0]))>;
182   using U = micron::__conditional_t<sizeof(V) == 4, u32, u64>;
183   constexpr i64 W = static_cast<i64>(sizeof(V));
184   constexpr i64 TOP = W * 8 - 1;
185   const U signbit = static_cast<U>(1) << TOP;
186 
187   const usize n = arr.size();
188   if ( n < 2 ) return;
189 
190   auto tkey = [&](const typename T::value_type &e) -> U {
191     V v = key(e);
192     U u = 0;
193     micron::bytecpy(&u, &v, W);
194     return u ^ ((static_cast<U>(0) - (u >> TOP)) | signbit);
195   };
196 
197   T carr = arr;
198   for ( i64 b = 0; b < W; ++b ) {
199     usize cnt[256] = {};
200     for ( usize i = 0; i < n; ++i ) cnt[static_cast<byte>(tkey(arr[i]) >> (b * 8))]++;
201     for ( i32 i = 1; i < 256; ++i ) cnt[i] += cnt[i - 1];
202     for ( usize k = n; k-- > 0; ) carr[--cnt[static_cast<byte>(tkey(arr[k]) >> (b * 8))]] = arr[k];
203     micron::memcpy(&arr[0], &carr[0], n);
204   }
205 }
