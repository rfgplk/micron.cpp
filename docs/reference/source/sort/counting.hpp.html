8 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
9 // counting sort
10 //   time:  O(n + range)
11 //   space: O(n + range)
12 //   stable: yes
13 //
14 //   best when key span = O(n); wide / sparse / domain-overflowing spans auto-fall
15 //   back to radix (O(n) memory, no span-sized alloc). key-projection overload
16 //   sorts structs by a derived integer key
17

28 template<is_iterable_container T, typename KeyFn>
29 void
30 __counting_impl(T &arr, KeyFn key)
31 {
32   using key_t = micron::remove_cvref_t<decltype(key(arr[0]))>;
33   const usize n = arr.size();
34   if ( n < 2 ) return;      // empty / single element: already sorted
35 
36   key_t kmin = key(arr[0]), kmax = key(arr[0]);
37   for ( usize i = 1; i < n; ++i ) {
38     key_t k = key(arr[i]);
39     if ( k < kmin ) kmin = k;
40     if ( kmax < k ) kmax = k;
41   }
42 
43   const umax_t span = static_cast<umax_t>(static_cast<umax_t>(kmax) - static_cast<umax_t>(kmin));      // biased max
44   const umax_t n_u = static_cast<umax_t>(n);
45   constexpr umax_t CAP = static_cast<umax_t>(1) << 27;      // hard bucket-count ceiling (~1 GiB of umax_t)
46   if ( span == ~static_cast<umax_t>(0) || span > 4u * n_u || span >= CAP ) {
47     radix(arr, key);      // wide / sparse / full-domain: radix is the right tool
48     return;
49   }
50 
51   const umax_t range = span + 1u;
52   const umax_t bias_min = static_cast<umax_t>(kmin);
53 
54   __sort_scratch<umax_t> bbuf(static_cast<usize>(range));
55   umax_t *bucket = bbuf.get();
56   for ( usize i = 0; i < static_cast<usize>(range); ++i ) bucket[i] = 0;
57   for ( usize i = 0; i < n; ++i ) ++bucket[static_cast<umax_t>(static_cast<umax_t>(key(arr[i])) - bias_min)];
58   for ( umax_t i = 1; i < range; ++i ) bucket[i] += bucket[i - 1];
59 
60   T carr = arr;      // output buffer
61   // scatter right-to-left for stability; index variable is a position, not a key
62   for ( usize k = n; k-- > 0; ) {
63     const umax_t idx = static_cast<umax_t>(static_cast<umax_t>(key(arr[k])) - bias_min);
64     carr[--bucket[idx]] = arr[k];
65   }
66 
67   arr = carr;
68 }
69 
70 template<is_iterable_container T>
71   requires micron::is_integral_v<typename T::value_type>
72 void
73 counting(T &arr)
74 {
75   __counting_impl(arr, [](const auto &e) { return e; });
76 }
77 
78 template<is_iterable_container T, typename KeyFn>
79   requires(is_integral_key_fn<KeyFn, T> && micron::is_trivially_copyable_v<typename T::value_type>)
80 void
81 counting(T &arr, KeyFn key)
82 {
83   __counting_impl(arr, key);
