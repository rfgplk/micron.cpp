24 template<typename It, typename Cmp>
25 void
26 __heap_range(It a, max_t lo, max_t hi, Cmp comp)
27 {
28   const max_t n = hi - lo + 1;
29   auto sift = [&](max_t root, max_t cnt) {
30     for ( ;; ) {
31       max_t child = (root << 1) + 1;
32       if ( child >= cnt ) break;
33       if ( child + 1 < cnt && comp(a[lo + child], a[lo + child + 1]) ) ++child;
34       if ( !comp(a[lo + root], a[lo + child]) ) break;
35       micron::swap(a[lo + root], a[lo + child]);
36       root = child;
37     }
38   };
39   for ( max_t start = n / 2; start-- > 0; ) sift(start, n);
40   for ( max_t cnt = n; cnt > 1; ) {
41     --cnt;
42     micron::swap(a[lo], a[lo + cnt]);
43     sift(0, cnt);
44   }
45 }

47 template<typename It, typename Cmp>
48 void
49 __introsort(It a, max_t lo, max_t hi, max_t depth, Cmp comp)
50 {
51   constexpr max_t THRESH = 24;
52   while ( hi - lo > THRESH ) {
53     if ( depth == 0 ) {
54       __heap_range(a, lo, hi, comp);
55       return;
56     }
57     --depth;
58     if ( hi - lo >= 2 ) __q_med3_to_high(a, lo, hi, comp);
59     max_t p = __q_partition(a, lo, hi, comp);
60     if ( p - lo < hi - p ) {
61       if ( lo < p - 1 ) __introsort(a, lo, p - 1, depth, comp);
62       lo = p + 1;
63     } else {
64       if ( p + 1 < hi ) __introsort(a, p + 1, hi, depth, comp);
65       hi = p - 1;
66     }
67   }
68 }

70 template<is_iterable_container T, typename Cmp>
71 T &
72 __sort_dispatch(T &arr, Cmp comp)
73 {
74   const max_t n = static_cast<max_t>(arr.size());
75   if ( n < 2 ) return arr;
76   max_t depth = 0;
77   for ( max_t t = n; t > 1; t >>= 1 ) depth += 2;      // 2 * floor(log2 n)
78   __introsort(arr.begin(), 0, n - 1, depth, comp);
79   __insertion(arr, 0, arr.size(), comp);      // finish the <= THRESH chunks
80   return arr;
81 }

83 template<is_iterable_container T>
84 T &
85 sort(T &arr)
86 {
87   return __sort_dispatch(arr, [](const auto &a, const auto &b) { return a < b; });
88 }
