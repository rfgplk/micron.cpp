9 // bubble sort
10 //   time:  O(n) best (sorted)  /  O(n^2) average + worst
11 //   space: O(1) in-place
12 //   stable: yes
13 //   use: tiny / nearly-sorted inputs, teaching; prefer insertion/sort otherwise

23 template<is_iterable_container T, typename Cmp>
24 T &
25 __bubble(T &arr, Cmp comp)
26 {
27   max_t n = static_cast<max_t>(arr.size());
28   while ( n >= 1 ) {
29     max_t k = 0;
30     for ( max_t i = 1; i < n; ++i ) {
31       if ( comp(arr[i], arr[i - 1]) ) {
32         micron::swap(arr[i - 1], arr[i]);
33         k = i;
34       }
35     }
36     n = k;
37   }
38   return arr;
39 }

41 template<is_iterable_container T>
42 T &
43 bubble(T &arr)
44 {
45   return __bubble(arr, [](const auto &a, const auto &b) { return a < b; });

55 template<is_iterable_container T, typename Cmp>
56 typename T::iterator
57 __bubble_it(typename T::iterator start, typename T::iterator end, Cmp comp)
58 {
59   max_t n = static_cast<max_t>(end - start);
60   while ( n >= 1 ) {
61     max_t k = 0;
62     for ( max_t i = 1; i < n; ++i ) {
63       if ( comp(start[i], start[i - 1]) ) {
64         micron::swap(start[i - 1], start[i]);
65         k = i;
66       }
67     }
68     n = k;
69   }
70   return start;
71 }
