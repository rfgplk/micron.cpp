9 // quicksort
10 //   time:  O(n log n) average
11 //   O(n^2) worst (low-cardinality / adversarial)
12 //   space: O(log n) stack (depth-bounded)
13 //   stable: no
14 //   in-place: yes
15 //
16 //   WARNING: degrades to ~O(n^2) on many equal keys
17 //  prefer sort::sort (introsort) as the default;
18 //   raw quick is here for the introsort core and explicit use on known-random data

31 // median-of-three
32 template<typename It, typename Cmp>
33 [[gnu::always_inline]] static inline void
34 __q_med3_to_high(It a, max_t lo, max_t hi, Cmp comp)
35 {
36   max_t mid = lo + ((hi - lo) >> 1);
37   if ( comp(a[mid], a[lo]) ) micron::swap(a[mid], a[lo]);
38   if ( comp(a[hi], a[lo]) ) micron::swap(a[hi], a[lo]);
39   if ( comp(a[hi], a[mid]) ) micron::swap(a[hi], a[mid]);
40   micron::swap(a[mid], a[hi]);      // median (a[mid]) becomes the pivot at a[hi]
41 }

43 // Lomuto partition
44 template<typename It, typename Cmp>
45 static inline max_t
46 __q_partition(It a, max_t lo, max_t hi, Cmp comp)
47 {
48   auto pivot = a[hi];
49   max_t i = lo - 1;
50   for ( max_t j = lo; j < hi; ++j )
51     if ( comp(a[j], pivot) ) {
52       ++i;
53       if ( i != j ) micron::swap(a[i], a[j]);
54     }
55   ++i;
56   if ( i != hi ) micron::swap(a[i], a[hi]);
57   return i;
58 }

60 template<is_iterable_container T, is_valid_comp<T> Cmp>
61 void
62 __quick(typename T::iterator start, max_t low, max_t high, Cmp comp)
63 {
64   while ( low < high ) {
65     if ( high - low >= 2 ) __q_med3_to_high(start, low, high, comp);
66     max_t p = __q_partition(start, low, high, comp);
67     if ( p - low < high - p ) {
68       if ( low < p - 1 ) __quick<T>(start, low, p - 1, comp);
69       low = p + 1;
70     } else {
71       if ( p + 1 < high ) __quick<T>(start, p + 1, high, comp);
72       high = p - 1;
73     }
74   }
75 }
