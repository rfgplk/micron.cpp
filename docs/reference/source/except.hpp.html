42 class __base_exception
43 {
44 public:
45   __base_exception() noexcept = default;
46   virtual ~__base_exception() = default;
47 
48   // NOTE: thrown objects are released by __cxa_free_exception;
49   // the deleting dtor needs this ABI slot even when no global new/delete
50   // header is part of the translation unit
51   static void
52   operator delete(void *, __SIZE_TYPE__) noexcept
53   {
54   }
55 
56   virtual const char *
57   what() const noexcept
58   {
59     return "__base_exception";
60   };
61 };
