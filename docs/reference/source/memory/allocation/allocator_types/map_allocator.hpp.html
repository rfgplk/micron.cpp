9 // map_allocator
10 //
11 // use for large or independently releasable regions that should bypass abcmalloc and map directly through Linux; every allocation is
12 // page-rounded; over-page alignment is supported by trimming a larger mapping; destroy returns the whole region with munmap
13 
14 namespace micron
15 {
16 
17 inline chunk<byte>
18 __map_create_aligned(usize capacity, usize alignment, i32 extra_flags = 0)
19 {
20   allocation_validate_alignment(alignment);
21   capacity = allocation_round_up_or_throw(capacity, page_size);
22   if ( capacity == 0 ) return { nullptr, 0 };
23 
24   usize mapping_len = capacity;
25   if ( alignment > page_size ) mapping_len = allocation_add_or_throw(capacity, alignment);
26 
27   addr_t *mapping = micron::mmap(nullptr, mapping_len, prot_read | prot_write, map_private | map_anonymous | extra_flags, -1, 0);
28   if ( micron::mmap_failed(mapping) ) exc<except::memory_error>("allocator: mmap failed");
29 
30   if ( alignment <= page_size ) return { reinterpret_cast<byte *>(mapping), capacity };
31 
32   const uintptr_t raw = reinterpret_cast<uintptr_t>(mapping);
33   const uintptr_t aligned = (raw + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
34   const usize prefix = aligned - raw;
35   const usize suffix = mapping_len - prefix - capacity;
36   if ( prefix != 0 ) micron::munmap(mapping, prefix);
37   if ( suffix != 0 ) micron::munmap(reinterpret_cast<addr_t *>(aligned + capacity), suffix);
38   return { reinterpret_cast<byte *>(aligned), capacity };
39 }

41 template<is_policy P = serial_allocation_policy> class map_allocator
42 {
43   static usize
44   __capacity(usize bytes)
45   {
46     return allocation_round_up_or_throw(__allocation_policy_capacity<P>(bytes), page_size);
47   }
48 
49 public:
50   static constexpr bool allocator_trusted = true;
51 
52   ~map_allocator() = default;
53   map_allocator() = default;
54   map_allocator(const map_allocator &) = default;
55   map_allocator(map_allocator &&) = default;
56   map_allocator &operator=(const map_allocator &) = default;
57   map_allocator &operator=(map_allocator &&) = default;
58 
59   [[nodiscard]] static constexpr usize
60   auto_size() noexcept
61   {
62     return page_size;
63   }
64 
65   [[nodiscard]] static usize
66   allocation_extent(usize bytes, usize)
67   {
68     return __capacity(bytes);
69   }
70 
71   [[nodiscard]] static chunk<byte>
72   create(usize bytes, usize alignment)
73   {
74     return __map_create_aligned(__capacity(bytes), alignment);
75   }
76 
77   [[nodiscard]] static chunk<byte>
78   create(usize bytes)
79   {
80     return create(bytes, page_size);
81   }
82 
83   [[nodiscard]] static chunk<byte>
84   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
85   {
86     if ( old.ptr == nullptr ) return create(bytes, alignment);
87     allocation_validate_alignment(alignment);
88     const usize capacity = __capacity(bytes);
89     const usize overlap = micron::min(old.len, capacity);
90 
91     if ( alignment <= page_size && preserve_bytes >= overlap ) {
92       addr_t *result = micron::mremap(micron::ptr_cast<addr_t *>(old.ptr), old.len, capacity, mremap_maymove);
93       if ( !micron::mmap_failed(result) ) return { reinterpret_cast<byte *>(result), capacity };
94     }
95 
96     chunk<byte> next = create(bytes, alignment);
97     const usize copied = micron::min(preserve_bytes, old.len, next.len);
98     if ( copied != 0 ) micron::memcpy(next.ptr, old.ptr, copied);
99     destroy(old, alignment);
100     return next;
101   }
102 
103   [[nodiscard]] static chunk<byte>
104   grow(chunk<byte> old, usize minimum)
105   {
106     const usize target = recommend(old.len, minimum);
107     if ( target == __allocation_max ) exc<except::length_error>("map_allocator: growth overflow");
108     return resize(old, target, old.len, page_size);
109   }
110 
111   static void
112   destroy(chunk<byte> memory, usize) noexcept
113   {
114     if ( memory.ptr == nullptr ) return;
115     micron::munmap(micron::ptr_cast<addr_t *>(memory.ptr), memory.len);
116   }
117 
118   static void
119   destroy(chunk<byte> memory) noexcept
120   {
121     destroy(memory, page_size);
122   }
123 
124   [[nodiscard]] static constexpr usize
125   recommend(usize current, usize minimum) noexcept
126   {
127     return __allocation_policy_recommend<P>(current, minimum);
128   }
129 
130   [[nodiscard]] static constexpr f32
131   get_grow() noexcept
132   {
133     return P::on_grow;
134   }
135 
136   byte *share(void) = delete;
137 };
