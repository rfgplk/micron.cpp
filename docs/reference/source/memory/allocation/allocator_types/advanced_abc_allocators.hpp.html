8 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
9 // temporal, retiring, and persistent ABC allocation
10 //
11 // use allocator_temporal for explicitly aliased scratch storage;
12 // repeated launder calls return the same live address and must be paired
13 // with retire;
14 // use allocator_retiring when freed addresses must be tombstoned instead of immediately recycled;
15 // resize always allocates and retires the old block
16 
17 namespace micron
18 {
19 
20 class allocator_temporal
21 {
22 public:
23   template<usize Alignment = 16>
24   [[nodiscard]] static chunk<byte>
25   launder(usize bytes)
26   {
27     static_assert(allocation_is_power_of_two(Alignment), "allocator_temporal: alignment must be a non-zero power of two");
28     chunk<byte> memory = abc::aligned_launder(Alignment, bytes);
29     if ( bytes != 0 && memory.ptr == nullptr ) exc<except::memory_error>("allocator_temporal: laundering failed");
30     return memory;
31   }
32 
33   [[nodiscard]] static chunk<byte>
34   launder(usize bytes, usize alignment)
35   {
36     allocation_validate_alignment(alignment);
37     chunk<byte> memory = abc::aligned_launder(alignment, bytes);
38     if ( bytes != 0 && memory.ptr == nullptr ) exc<except::memory_error>("allocator_temporal: laundering failed");
39     return memory;
40   }
41 
42   template<usize Alignment = 16>
43   static void
44   retire(chunk<byte> memory)
45   {
46     static_assert(allocation_is_power_of_two(Alignment), "allocator_temporal: alignment must be a non-zero power of two");
47     abc::aligned_retire(memory, Alignment);
48   }
49 
50   static void
51   retire(chunk<byte> memory, usize alignment)
52   {
53     allocation_validate_alignment(alignment);
54     abc::aligned_retire(memory, alignment);
55   }
56 };

58 template<is_policy P = serial_allocation_policy> class allocator_retiring: public __abc_policy_allocator<P, 16>
59 {
60   using base = __abc_policy_allocator<P, 16>;
61 
62   template<usize Alignment>
63   static void
64   __retire(byte *memory) noexcept
65   {
66     if ( memory == nullptr ) return;
67 #if defined(__micron_sanitizer_owns_heap)
68     base::template destroy<Alignment>(memory);
69 #else
70     abc::aligned_retire(memory, Alignment);
71 #endif
72   }
73 
74   static void
75   __retire(byte *memory, usize alignment) noexcept
76   {
77     if ( memory == nullptr ) return;
78 #if defined(__micron_sanitizer_owns_heap)
79     base::destroy(memory, alignment);
80 #else
81     abc::aligned_retire(memory, alignment);
82 #endif
83   }
84 
85 public:
86   static constexpr bool allocator_trusted = true;
87 
88   using base::allocate;
89   using base::auto_size;
90   using base::create;
91   using base::get_grow;
92   using base::recommend;
93 
94   template<usize Alignment>
95   [[nodiscard]] static chunk<byte>
96   resize(chunk<byte> old, usize bytes, usize preserve_bytes)
97   {
98     chunk<byte> next = base::template create<Alignment>(bytes);
99     const usize copied = micron::min(preserve_bytes, old.len, next.len);
100     if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
101     if ( old.ptr ) base::__telemetry_deallocate(old.len);
102     __retire<Alignment>(old.ptr);
103     return next;
104   }
105 
106   [[nodiscard]] static chunk<byte>
107   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
108   {
109     chunk<byte> next = base::create(bytes, alignment);
110     const usize copied = micron::min(preserve_bytes, old.len, next.len);
111     if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
112     if ( old.ptr ) base::__telemetry_deallocate(old.len);
113     __retire(old.ptr, alignment);
114     return next;
115   }
116 
117   [[nodiscard]] static chunk<byte>
118   grow(chunk<byte> old, usize minimum)
119   {
120     const usize target = recommend(old.len, minimum);
121     if ( target == __allocation_max ) exc<except::length_error>("allocator_retiring: growth overflow");
122     return resize(old, target, old.len, 16);
123   }
124 
125   template<usize Alignment>
126   static void
127   destroy(chunk<byte> memory) noexcept
128   {
129     if ( memory.ptr ) base::__telemetry_deallocate(memory.len);
130     __retire<Alignment>(memory.ptr);
131   }
132 
133   static void
134   destroy(chunk<byte> memory, usize alignment) noexcept
135   {
136     if ( memory.ptr ) base::__telemetry_deallocate(memory.len);
137     __retire(memory.ptr, alignment);
138   }
139 
140   static void
141   destroy(chunk<byte> memory) noexcept
142   {
143     if ( memory.ptr ) base::__telemetry_deallocate(memory.len);
144     __retire<16>(memory.ptr);
145   }
146 
147   template<usize Alignment>
148   static void
149   destroy(byte *memory) noexcept
150   {
151     if ( memory ) base::__telemetry_deallocate(0);
152     __retire<Alignment>(memory);
153   }
154 
155   static void
156   destroy(byte *memory, usize alignment) noexcept
157   {
158     if ( memory ) base::__telemetry_deallocate(0);
159     __retire(memory, alignment);
160   }
161 
162   template<usize Alignment>
163   static void
164   deallocate(chunk<byte> memory) noexcept
165   {
166     destroy<Alignment>(memory);
167   }
168 
169   static void
170   deallocate(chunk<byte> memory, usize alignment) noexcept
171   {
172     destroy(memory, alignment);
173   }
174 
175   template<usize Alignment>
176   static void
177   deallocate(byte *memory) noexcept
178   {
179     destroy<Alignment>(memory);
180   }
181 
182   static void
183   deallocate(byte *memory, usize alignment) noexcept
184   {
185     destroy(memory, alignment);
186   }
