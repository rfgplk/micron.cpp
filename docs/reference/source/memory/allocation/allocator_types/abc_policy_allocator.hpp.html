22   template<usize Alignment>
23   [[gnu::always_inline]] static inline chunk<byte>
24   __create_capacity(usize capacity)
25   {
26     static_assert(allocation_is_power_of_two(Alignment), "allocator: alignment must be a non-zero power of two");
27     if ( capacity == 0 ) return { nullptr, 0 };
28 #if defined(__micron_sanitizer_owns_heap)
29     chunk<byte> memory = abc_allocator<byte>::allocate_aligned(capacity, Alignment);
30 #else
31     chunk<byte> memory = Alignment <= abc::native_block_alignment ? abc::balloc(capacity) : abc::aligned_balloc(Alignment, capacity);
32     if ( memory.ptr == nullptr ) [[unlikely]]
33       exc<except::critical_error>("allocator: abcmalloc failed to satisfy request");
34 #endif
35     return { memory.ptr, capacity };
36   }

38   static chunk<byte>
39   __create_capacity(usize capacity, usize alignment)
40   {
41     if ( capacity == 0 ) return { nullptr, 0 };
42     allocation_validate_alignment(alignment);
43 #if defined(__micron_sanitizer_owns_heap)
44     chunk<byte> memory = abc_allocator<byte>::allocate_aligned(capacity, alignment);
45 #else
46     chunk<byte> memory = alignment <= abc::native_block_alignment ? abc_allocator<byte>::allocate(capacity)
47                                                                   : abc_allocator<byte>::allocate_aligned(capacity, alignment);
48 #endif
49     return { memory.ptr, capacity };
50   }

52   template<usize Alignment>
53   [[gnu::always_inline]] static inline void
54   __destroy_ptr(byte *memory) noexcept
55   {
56     if ( memory == nullptr ) return;
57 #if defined(__micron_sanitizer_owns_heap)
58     abc_allocator<byte>::dealloc_aligned(memory, Alignment);
59 #else
60     if constexpr ( Alignment <= abc::native_block_alignment )
61       abc::dealloc(memory);
62     else
63       abc::aligned_dealloc(memory, Alignment);
64 #endif
65   }

139   [[nodiscard]] static chunk<byte>
140   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
141   {
142     const usize capacity = __allocation_policy_capacity<P>(bytes);
143     if ( old.ptr == nullptr ) {
144       chunk<byte> next = __create_capacity(capacity, alignment);
145       __telemetry::__telemetry_resize(bytes, 0, next.len, 0);
146       return next;
147     }
148     if ( capacity == 0 ) {
149       __destroy_ptr(old.ptr, alignment);
150       __telemetry::__telemetry_resize(bytes, old.len, 0, 0);
151       return { nullptr, 0 };
152     }
153 #if !defined(__micron_sanitizer_owns_heap)
154     if ( alignment <= abc::native_block_alignment ) {
155       chunk<byte> next = abc::resize_chunk(old, capacity, micron::min(preserve_bytes, old.len));
156       if ( next.ptr == nullptr ) exc<except::memory_error>("allocator: abcmalloc resize failed");
157       __telemetry::__telemetry_resize(bytes, old.len, capacity, next.ptr == old.ptr ? 0 : micron::min(preserve_bytes, old.len, capacity));
158       return { next.ptr, capacity };
159     }
160 #endif
161     chunk<byte> next = __create_capacity(capacity, alignment);
162     const usize copied = micron::min(preserve_bytes, old.len, next.len);
163     if ( copied != 0 ) micron::memcpy(next.ptr, old.ptr, copied);
164     __destroy_ptr(old.ptr, alignment);
165     __telemetry::__telemetry_resize(bytes, old.len, next.len, copied);
166     return next;
167   }

169   template<usize Alignment>
170   [[nodiscard]] static chunk<byte>
171   resize(chunk<byte> old, usize bytes, usize preserve_bytes)
172   {
173     static_assert(allocation_is_power_of_two(Alignment), "allocator: alignment must be a non-zero power of two");
174     const usize capacity = __allocation_policy_capacity<P>(bytes);
175     if ( old.ptr == nullptr ) {
176       chunk<byte> next = __create_capacity<Alignment>(capacity);
177       __telemetry::__telemetry_resize(bytes, 0, next.len, 0);
178       return next;
179     }
180     if ( capacity == 0 ) {
181       __destroy_ptr<Alignment>(old.ptr);
182       __telemetry::__telemetry_resize(bytes, old.len, 0, 0);
183       return { nullptr, 0 };
184     }
185 #if !defined(__micron_sanitizer_owns_heap)
186     if constexpr ( Alignment <= abc::native_block_alignment ) {
187       chunk<byte> next = abc::resize_chunk(old, capacity, micron::min(preserve_bytes, old.len));
188       if ( next.ptr == nullptr ) exc<except::memory_error>("allocator: abcmalloc resize failed");
189       __telemetry::__telemetry_resize(bytes, old.len, capacity, next.ptr == old.ptr ? 0 : micron::min(preserve_bytes, old.len, capacity));
190       return { next.ptr, capacity };
191     }
192 #endif
193     chunk<byte> next = __create_capacity<Alignment>(capacity);
194     const usize copied = micron::min(preserve_bytes, old.len, next.len);
195     if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
196     __destroy_ptr<Alignment>(old.ptr);
197     __telemetry::__telemetry_resize(bytes, old.len, next.len, copied);
198     return next;
199   }

201   [[nodiscard]] static chunk<byte>
202   grow(chunk<byte> old, usize minimum)
203   {
204     const usize target = recommend(old.len, minimum);
205     if ( target == __allocation_max ) exc<except::length_error>("allocator: growth overflow");
206     return resize(old, target, old.len, DefaultAlignment);
207   }

209   static void
210   destroy(chunk<byte> memory, usize alignment) noexcept
211   {
212     if ( memory.ptr ) __telemetry::__telemetry_deallocate(memory.len);
213     if ( memory.ptr == nullptr ) return;
214 #if defined(__micron_sanitizer_owns_heap)
215     abc_allocator<byte>::dealloc_aligned(memory.ptr, alignment);
216 #else
217     if ( alignment <= abc::native_block_alignment )
218       abc::dealloc(memory.ptr, memory.len);
219     else
220       abc::aligned_dealloc(memory.ptr, alignment);
221 #endif
222   }
