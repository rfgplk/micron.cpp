9 // allocator_immutable
10 //
11 // use for data built once and read thereafter, such as tables or decoded metadata;
12 // storage is a private, page-rounded mapping returned with requested logical length;
13 // seal verifies allocator provenance and changes the entire mapping to read-only
14 
15 namespace micron
16 {
17 
18 class allocator_immutable
19 {
20   static usize
21   __mapping_len(usize bytes)
22   {
23     return allocation_round_up_or_throw(bytes, page_size);
24   }
25 
26 public:
27   static constexpr bool allocator_trusted = true;
28 
29   [[nodiscard]] static constexpr usize
30   auto_size() noexcept
31   {
32     return page_size;
33   }
34 
35   [[nodiscard]] static constexpr usize
36   allocation_extent(usize bytes, usize) noexcept
37   {
38     return bytes;
39   }
40 
41   [[nodiscard]] static chunk<byte>
42   create(usize bytes, usize alignment)
43   {
44     allocation_validate_alignment(alignment);
45     if ( bytes == 0 ) return { nullptr, 0 };
46     chunk<byte> mapping = __map_create_aligned(__mapping_len(bytes), alignment);
47     if ( !abc::register_external(mapping.ptr, mapping.len, abc::external_provenance::immutable) ) {
48       micron::munmap(micron::ptr_cast<addr_t *>(mapping.ptr), mapping.len);
49       exc<except::memory_error>("allocator_immutable: provenance registration failed");
50     }
51     return { mapping.ptr, bytes };
52   }
53 
54   [[nodiscard]] static chunk<byte>
55   create(usize bytes)
56   {
57     return create(bytes, 16);
58   }
59 
60   [[nodiscard]] static chunk<byte>
61   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
62   {
63     chunk<byte> next = create(bytes, alignment);
64     const usize copied = micron::min(preserve_bytes, old.len, next.len);
65     if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
66     destroy(old, alignment);
67     return next;
68   }
69 
70   [[nodiscard]] static chunk<byte>
71   grow(chunk<byte> old, usize minimum)
72   {
73     const usize target = recommend(old.len, minimum);
74     if ( target == __allocation_max ) exc<except::length_error>("allocator_immutable: growth overflow");
75     return resize(old, target, old.len, 16);
76   }
77 
78   [[nodiscard]] static bool
79   seal(chunk<byte> memory) noexcept
80   {
81     if ( memory.ptr == nullptr || memory.len == 0 ) return false;
82     usize mapping_len;
83     if ( !allocation_checked_round_up(memory.len, page_size, mapping_len) ) return false;
84     if ( abc::external_query_provenance(memory.ptr) != abc::external_provenance::immutable ) return false;
85     if ( abc::external_query_size(memory.ptr) != mapping_len ) return false;
86     return micron::mprotect(micron::ptr_cast<addr_t *>(memory.ptr), mapping_len, prot_read) == 0;
87   }
88 
89   [[nodiscard]] static bool
90   seal(byte *memory, usize bytes) noexcept
91   {
92     return seal({ memory, bytes });
93   }
94 
95   static void
96   destroy(chunk<byte> memory, usize) noexcept
97   {
98     if ( memory.ptr == nullptr || memory.len == 0 ) return;
99     usize mapping_len;
100     if ( !allocation_checked_round_up(memory.len, page_size, mapping_len) ) return;
101     (void)abc::unregister_external(memory.ptr, mapping_len);
102     micron::munmap(micron::ptr_cast<addr_t *>(memory.ptr), mapping_len);
103   }
104 
105   static void
106   destroy(chunk<byte> memory) noexcept
107   {
108     destroy(memory, 16);
109   }
110 
111   [[nodiscard]] static constexpr usize
112   recommend(usize current, usize minimum) noexcept
113   {
114     usize result;
115     return allocation_checked_growth(current, minimum, 2, 1, result) ? result : __allocation_max;
116   }
117 
118   [[nodiscard]] static constexpr f32
119   get_grow() noexcept
120   {
121     return 2.0f;
122   }
123 };
124 
125 };      // namespace micron
