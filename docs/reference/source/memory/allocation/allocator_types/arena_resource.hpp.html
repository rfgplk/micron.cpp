291   [[nodiscard, gnu::always_inline]] static inline chunk<byte>
292   __try_allocate(__block *block, usize bytes, usize alignment) noexcept
293   {
294     const uintptr_t current = reinterpret_cast<uintptr_t>(block->cursor);
295     const uintptr_t end = reinterpret_cast<uintptr_t>(block->end);
296     if ( current > end ) return { nullptr, 0 };
297     if ( current > micron::numeric_limits<uintptr_t>::max() - (alignment - 1) ) return { nullptr, 0 };
298     const uintptr_t aligned = (current + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
299     if ( aligned > end || bytes > end - aligned ) return { nullptr, 0 };
300     block->cursor = reinterpret_cast<byte *>(aligned + bytes);
301     return { reinterpret_cast<byte *>(aligned), bytes };
302   }

304   template<usize Alignment>
305   [[nodiscard, gnu::always_inline]] static inline bool
306   __try_allocate(__block *block, usize bytes, byte *&result) noexcept
307   {
308     const uintptr_t current = reinterpret_cast<uintptr_t>(block->cursor);
309     const uintptr_t end = reinterpret_cast<uintptr_t>(block->end);
310     if constexpr ( Alignment > page_size ) {
311       chunk<byte> memory = __try_allocate(block, bytes, Alignment);
312       result = memory.ptr;
313       return result != nullptr;
314     }
315     const usize padding = static_cast<usize>(-current) & (Alignment - 1);
316     const uintptr_t aligned = current + padding;
317     if ( aligned > end || bytes > end - aligned ) return false;
318     result = reinterpret_cast<byte *>(aligned);
319     block->cursor = result + bytes;
320     return true;
321   }

330   [[nodiscard]] __block *
331   __append_owned(usize bytes, usize alignment)
332   {
333     const usize usable = __max(__block_bytes, bytes);
334     if ( __head == nullptr && !__external_origin ) {
335       const usize upstream_alignment = __max(alignment, alignof(__block));
336       chunk<byte> storage = __allocator_create<Upstream>(usable, upstream_alignment);
337       __first = { storage, nullptr, storage.ptr, storage.ptr + storage.len, storage.ptr, upstream_alignment, false, false };
338       __head = __tail = __current = &__first;
339       this->__stat_block_add(storage.len);
340       return &__first;
341     }
342 
343     const usize overhead = allocation_add_or_throw(sizeof(__block), alignment - 1);
344     const usize total = allocation_add_or_throw(usable, overhead);
345     const usize upstream_alignment = __max(alignment, alignof(__block));
346     chunk<byte> storage = __allocator_create<Upstream>(total, upstream_alignment);
347     auto *block = new (static_cast<void *>(storage.ptr)) __block{};
348     const uintptr_t first = reinterpret_cast<uintptr_t>(storage.ptr) + sizeof(__block);
349     const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
350     block->storage = storage;
351     block->next = nullptr;
352     block->begin = reinterpret_cast<byte *>(aligned);
353     block->end = storage.ptr + storage.len;
354     block->cursor = block->begin;
355     block->upstream_alignment = upstream_alignment;
356     block->external = false;
357     block->embedded = true;
358     if ( __tail )
359       __tail->next = block;
360     else
361       __head = block;
362     __tail = __current = block;
363     this->__stat_block_add(static_cast<usize>(block->end - block->begin));
364     return block;
365   }

367   [[nodiscard, gnu::noinline]] chunk<byte>
368   __allocate_slow(usize bytes, usize alignment)
369   {
370     __block *block = __current ? __current->next : __head;
371     for ( ; block; block = block->next ) {
372       chunk<byte> result = __try_allocate(block, bytes, alignment);
373       if ( result.ptr ) {
374         __current = block;
375         __record_allocation(result, bytes);
376         return result;
377       }
378     }
379 
380     if ( __external_origin && __overflow == arena_overflow::fail ) exc<except::memory_error>("arena_resource: external span exhausted");
381     block = __append_owned(bytes, alignment);
382     chunk<byte> result = __try_allocate(block, bytes, alignment);
383     if ( result.ptr == nullptr ) exc<except::memory_error>("arena_resource: upstream block cannot satisfy allocation");
384     __current = block;
385     __record_allocation(result, bytes);
386     return result;
387   }

390   __allocate_locked(usize bytes, usize alignment)
391   {
392     if ( bytes == 0 ) return { nullptr, 0 };

394     if ( block ) [[likely]] {
395       chunk<byte> result = __try_allocate(block, bytes, alignment);
396       if ( result.ptr ) [[likely]] {

457   void
458   __invalidate_markers(u64 first, u64 last) noexcept
459   {
460     if ( first > last ) return;
461     for ( u8 i = 0; i < __invalid_count; ++i ) {
462       __invalid_marker_range &range = __invalid_markers[i];
463       if ( last + (last != micron::numeric_limits<u64>::max()) < range.first
464            || range.last + (range.last != micron::numeric_limits<u64>::max()) < first )
465         continue;
466       if ( first < range.first ) range.first = first;
467       if ( last > range.last ) range.last = last;
468       return;
469     }
470     if ( __invalid_count < 8 ) {
471       __invalid_markers[__invalid_count++] = { first, last };
472       return;
473     }
474     ++__generation;
475     __marker_next = 0;
476     __invalid_count = 0;
477   }

479   [[nodiscard]] bool
480   __marker_valid(const marker &value) const noexcept
481   {
482     if ( value.resource != this || value.generation != __generation || value.serial == 0 || value.serial > __marker_next ) return false;
483     for ( u8 i = 0; i < __invalid_count; ++i )
484       if ( value.serial >= __invalid_markers[i].first && value.serial <= __invalid_markers[i].last ) return false;
485     return true;
486   }

503   void
504   __release_locked(bool count_release) noexcept
505   {
506     __block *block = __head;
507     __block *kept_external = nullptr;
508     while ( block ) {
509       __block *next = block->next;
510       if ( block->external ) {
511         block->cursor = block->begin;
512         block->next = nullptr;
513         kept_external = block;
514       } else {
515         const chunk<byte> storage = block->storage;
516         const usize alignment = block->upstream_alignment;
517         if ( block->embedded ) block->~__block();
518         __allocator_destroy<Upstream>(storage, alignment);
519       }
520       block = next;
521     }
522     __head = __tail = __current = kept_external;
523     __clear_last();
524     __new_generation();
525     this->__stat_used(0);
526     this->__stat_blocks(kept_external ? 1 : 0);
527     if ( count_release ) this->__stat_release();
528   }

593   [[nodiscard]] chunk<byte>
594   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
595   {
596     allocation_validate_alignment(alignment);
597     __guard guard(this);
598     if ( old.ptr == nullptr ) return __allocate_locked(bytes, alignment);
599     if ( !__owns_locked(old.ptr) ) exc<except::invalid_argument>("arena_resource: resize pointer is not owned by this arena");
600 
601     __block *last_block = __current;
602     if ( old.ptr == __last_ptr && last_block && (reinterpret_cast<uintptr_t>(old.ptr) & (alignment - 1)) == 0 ) {
603       const usize last_offset = static_cast<usize>(reinterpret_cast<uintptr_t>(old.ptr) - reinterpret_cast<uintptr_t>(last_block->begin));
604       const usize last_capacity = static_cast<usize>(last_block->end - last_block->begin);
605       if ( last_offset <= last_capacity && old.len <= last_capacity - last_offset && last_block->cursor == old.ptr + old.len
606            && bytes <= last_capacity - last_offset ) {
607         last_block->cursor = old.ptr + bytes;
608         this->__stat_resize(bytes);
609         this->__stat_used(__used_locked());
610         if ( bytes == 0 ) {
611           __clear_last();
612           return { nullptr, 0 };
613         }
614         return { old.ptr, bytes };
615       }
616     }
617 
618     chunk<byte> next = __allocate_locked(bytes, alignment);
619     const usize copied = micron::min(preserve_bytes, old.len, next.len);
620     if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
621     this->__stat_resize(bytes);
622     this->__stat_copy(copied);
623     return next;
624   }

656   [[nodiscard]] bool
657   rewind(const marker &value) noexcept
658   {
659     __guard guard(this);
660     if ( !__marker_valid(value) ) return false;
661     __block *target = reinterpret_cast<__block *>(value.block);
662     if ( target ) {
663       bool found = false;
664       for ( __block *block = __head; block; block = block->next ) {
665         if ( block == target ) {
666           found = true;
667           if ( value.cursor > static_cast<usize>(block->end - block->begin) ) return false;
668           block->cursor = block->begin + value.cursor;
669           for ( block = block->next; block; block = block->next ) block->cursor = block->begin;
670           break;
671         }
672       }
673       if ( !found ) return false;
674       __current = target;
675     } else {
676       for ( __block *block = __head; block; block = block->next ) block->cursor = block->begin;
677       __current = __head;
678     }
679     __invalidate_markers(value.serial + 1, __marker_next);
680     __clear_last();
681     this->__stat_used(__used_locked());
682     this->__stat_rewind();
683     return true;
684   }

730   }
731 
732   [[nodiscard]] usize
733   available() const noexcept
734   {
735     __guard guard(this);
736     const usize total = __capacity_locked();
737     const usize consumed = __used_locked();
738     return total - consumed;
739   }
740 
741   [[nodiscard]] usize
742   block_count() const noexcept
743   {
744     __guard guard(this);
745     return __blocks_locked();
746   }
747 
748   [[nodiscard]] usize
749   block_size() const noexcept
750   {
751     return __block_bytes;
752   }
753 
754   [[nodiscard]] allocator_stats_snapshot
755   stats() const noexcept
756   {
757     __guard guard(this);
758     return this->__stats_base::__stat_snapshot();
759   }
760 
761   void
762   reset_stats() noexcept
763   {
764     __guard guard(this);
765     this->__stats_base::__stat_clear();
766     this->__stats_base::__stat_blocks(__blocks_locked());
767     this->__stats_base::__stat_used(__used_locked());
768   }
769 };
