28 [[nodiscard]] constexpr bool
29 allocation_is_power_of_two(usize n) noexcept
30 {
31   return n != 0 && (n & (n - 1)) == 0;

34 [[nodiscard]] constexpr bool
35 allocation_checked_add(usize a, usize b, usize &result) noexcept
36 {
37   if ( b > __allocation_max - a ) return false;
38   result = a + b;
39   return true;
40 }

42 [[nodiscard]] constexpr bool
43 allocation_checked_multiply(usize a, usize b, usize &result) noexcept
44 {
45   if ( a != 0 && b > __allocation_max / a ) return false;
46   result = a * b;
47   return true;
48 }

50 [[nodiscard]] constexpr bool
51 allocation_checked_round_up(usize value, usize granularity, usize &result) noexcept
52 {
53   if ( granularity == 0 ) return false;
54   const usize remainder = value % granularity;
55   if ( remainder == 0 ) {
56     result = value;
57     return true;
58   }
59   return allocation_checked_add(value, granularity - remainder, result);
60 }

62 [[nodiscard]] constexpr bool
63 allocation_checked_growth(usize current, usize minimum, usize numerator, usize denominator, usize &result) noexcept
64 {
65   if ( denominator == 0 || numerator < denominator ) return false;
66   const usize whole = numerator / denominator;
67   const usize fractional = numerator % denominator;
68   usize grown;
69   if ( !allocation_checked_multiply(current, whole, grown) ) return false;
70 
71   if ( fractional != 0 ) {
72     const usize quotient = current / denominator;
73     const usize remainder = current % denominator;
74     usize fractional_whole;
75     usize fractional_tail;
76     if ( !allocation_checked_multiply(quotient, fractional, fractional_whole)
77          || !allocation_checked_multiply(remainder, fractional, fractional_tail) )
78       return false;
79     usize fractional_result;
80     if ( !allocation_checked_add(fractional_whole, fractional_tail / denominator, fractional_result) ) return false;
81     if ( fractional_tail % denominator != 0 ) {
82       if ( fractional_result == __allocation_max ) return false;
83       ++fractional_result;
84     }
85     if ( !allocation_checked_add(grown, fractional_result, grown) ) return false;
86   }
87   result = grown < minimum ? minimum : grown;
88   return true;
89 }

92 allocation_add_or_throw(usize a, usize b)

94   usize result;
95   if ( !allocation_checked_add(a, b, result) ) exc<except::length_error>("allocator: size addition overflow");
96   return result;

122 template<is_policy P>
123 [[nodiscard]] inline usize
124 __allocation_policy_capacity(usize bytes)
125 {
126   if ( bytes == 0 ) return 0;
127   const usize requested = bytes < P::minimum_bytes ? P::minimum_bytes : bytes;
128   return allocation_round_up_or_throw(requested, P::granularity);
129 }

131 template<is_policy P>
132 [[nodiscard]] constexpr usize
133 __allocation_policy_recommend(usize current, usize minimum) noexcept
134 {
135   usize grown;
136   if ( !allocation_checked_growth(current, minimum, P::growth_numerator, P::growth_denominator, grown) ) return __allocation_max;
137   if ( grown < P::minimum_bytes ) grown = P::minimum_bytes;
138   usize rounded;
139   return allocation_checked_round_up(grown, P::granularity, rounded) ? rounded : __allocation_max;
140 }

374   static void
375   __validate_result(chunk<byte> result, usize bytes, usize alignment)
376   {
377     if ( bytes != 0 && result.ptr == nullptr ) exc<except::memory_error>("allocator: returned null storage for a non-zero request");
378     if ( result.ptr && (reinterpret_cast<uintptr_t>(result.ptr) & (alignment - 1)) != 0 ) {
379       __deallocate_chunk(result, alignment);
380       exc<except::memory_error_core_unaligned>("allocator: returned storage is under-aligned");
381     }
382     if ( result.len < bytes ) {
383       __deallocate_chunk(result, alignment);
384       exc<except::memory_error>("allocator: returned capacity is smaller than requested");
385     }
386   }

502   [[nodiscard]] static chunk<byte>
503   resize(chunk<byte> old, usize bytes, usize preserve_bytes)
504   {
505     static_assert(allocation_is_power_of_two(Alignment), "allocator: alignment must be a non-zero power of two");
506     chunk<byte> result;
507     if constexpr ( requires { Alloc::template resize<Alignment>(old, bytes, preserve_bytes); } ) {
508       result = Alloc::template resize<Alignment>(old, bytes, preserve_bytes);
509     } else if constexpr ( requires { Alloc::resize(old, bytes, preserve_bytes, Alignment); } ) {
510       result = Alloc::resize(old, bytes, preserve_bytes, Alignment);
511     } else if constexpr ( requires { Alloc::grow(old, bytes); } ) {
512       if ( bytes >= old.len && preserve_bytes >= old.len ) {
513         result = Alloc::grow(old, bytes);
514       } else {
515         result = allocate<Alignment>(bytes);
516         const usize copied = micron::min(preserve_bytes, old.len, result.len);
517         if ( copied ) micron::memcpy(result.ptr, old.ptr, copied);
518         __deallocate_chunk<Alignment>(old);
519       }
520     } else {
521       result = allocate<Alignment>(bytes);
522       const usize copied = micron::min(preserve_bytes, old.len, result.len);
523       if ( copied ) micron::memcpy(result.ptr, old.ptr, copied);
524       __deallocate_chunk<Alignment>(old);
525     }
526     if constexpr ( !trusted || allocator_checks_enabled ) __validate_result<Alignment>(result, bytes);
527     return result;
528   }

622 template<class Alloc, typename T>
623 [[nodiscard]] inline T *
624 __allocator_allocate_array(usize count)
625 {
626   const usize bytes = allocation_multiply_or_throw(count, sizeof(T));
627   if constexpr ( allocator_traits<Alloc>::template has_unsized_deallocate<alignof(T)> ) {
628     return micron::ptr_cast<T *>(__allocator_create<Alloc, alignof(T)>(bytes).ptr);
629   }
630 
631   constexpr usize alignment = alignof(T) < alignof(__allocator_array_header) ? alignof(__allocator_array_header) : alignof(T);
632   const usize overhead = allocation_add_or_throw(sizeof(__allocator_array_header), alignment - 1);
633   chunk<byte> storage = __allocator_create<Alloc, alignment>(allocation_add_or_throw(bytes, overhead));
634   const uintptr_t first = reinterpret_cast<uintptr_t>(storage.ptr) + sizeof(__allocator_array_header);
635   const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
636   T *result = reinterpret_cast<T *>(aligned);
637   auto *header = micron::ptr_cast<__allocator_array_header *>(reinterpret_cast<byte *>(result) - sizeof(__allocator_array_header));
638   new (static_cast<void *>(header)) __allocator_array_header{ storage, alignment };
639   return result;
640 }
641 
642 template<class Alloc, typename T>
643 [[nodiscard]] inline T *
644 __allocator_allocate_array(usize count, usize alignment)
645 {
646   allocation_validate_alignment(alignment);
647   if ( alignment < alignof(T) ) alignment = alignof(T);
648   if ( alignment < alignof(__allocator_array_header) ) alignment = alignof(__allocator_array_header);
649   const usize bytes = allocation_multiply_or_throw(count, sizeof(T));
650   const usize overhead = allocation_add_or_throw(sizeof(__allocator_array_header), alignment - 1);
651   chunk<byte> storage = __allocator_create<Alloc>(allocation_add_or_throw(bytes, overhead), alignment);
652   const uintptr_t first = reinterpret_cast<uintptr_t>(storage.ptr) + sizeof(__allocator_array_header);
653   const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
654   T *result = reinterpret_cast<T *>(aligned);
655   auto *header = reinterpret_cast<__allocator_array_header *>(reinterpret_cast<byte *>(result) - sizeof(__allocator_array_header));
656   new (static_cast<void *>(header)) __allocator_array_header{ storage, alignment };
657   return result;
658 }
659 
660 template<class Alloc, typename T>
661 inline void
662 __allocator_deallocate_array(T *memory) noexcept
663 {
664   if ( memory == nullptr ) return;
665   if constexpr ( allocator_traits<Alloc>::template has_unsized_deallocate<alignof(T)> ) {
666     allocator_traits<Alloc>::template deallocate<alignof(T)>(reinterpret_cast<byte *>(memory));
667     return;
668   }
669   auto *header = micron::ptr_cast<__allocator_array_header *>(reinterpret_cast<byte *>(memory) - sizeof(__allocator_array_header));
670   __allocator_destroy<Alloc>(header->storage, header->alignment);
671 }
