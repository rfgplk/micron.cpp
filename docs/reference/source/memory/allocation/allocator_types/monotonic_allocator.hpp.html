64   static __block *
65   __new_block(usize bytes, usize alignment)
66   {
67     const usize usable = bytes < BlockBytes ? BlockBytes : bytes;
68     const usize overhead = allocation_add_or_throw(sizeof(__block), alignment - 1);
69     const usize total = allocation_add_or_throw(usable, overhead);
70     const usize block_alignment = alignment < alignof(__block) ? alignof(__block) : alignment;
71     chunk<byte> storage = __allocator_create<Upstream>(total, block_alignment);
72     auto *block = new (static_cast<void *>(storage.ptr)) __block{};
73     const uintptr_t first = reinterpret_cast<uintptr_t>(storage.ptr) + sizeof(__block);
74     const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
75     block->storage = storage;
76     block->next = nullptr;
77     block->begin = reinterpret_cast<byte *>(aligned);
78     block->end = storage.ptr + storage.len;
79     block->cursor = block->begin;
80     block->allocation_alignment = block_alignment;
81     if ( __tail )
82       __tail->next = block;
83     else
84       __head = block;
85     __tail = block;
86     __current = block;
87 #if defined(MICRON_ALLOCATOR_STATS)
88     ++__stats.blocks;
89     if ( __stats.blocks > __stats.peak_blocks ) __stats.peak_blocks = __stats.blocks;
90 #endif
91     return block;
92   }

94   static chunk<byte>
95   __try_allocate(__block *block, usize bytes, usize alignment) noexcept
96   {
97     const uintptr_t current = reinterpret_cast<uintptr_t>(block->cursor);
98     const uintptr_t end = reinterpret_cast<uintptr_t>(block->end);
99     if ( current > micron::numeric_limits<uintptr_t>::max() - (alignment - 1) ) return { nullptr, 0 };
100     const uintptr_t aligned = (current + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
101     if ( aligned > end || bytes > end - aligned ) return { nullptr, 0 };
102     block->cursor = reinterpret_cast<byte *>(aligned + bytes);
103     return { reinterpret_cast<byte *>(aligned), bytes };
104   }

106   template<usize Alignment>
107   [[gnu::always_inline]] static inline bool
108   __try_allocate(__block *block, usize bytes, byte *&result) noexcept
109   {
110     if constexpr ( Alignment > page_size ) {
111       chunk<byte> memory = __try_allocate(block, bytes, Alignment);
112       result = memory.ptr;
113       return result != nullptr;
114     }
115     const uintptr_t current = reinterpret_cast<uintptr_t>(block->cursor);
116     const uintptr_t end = reinterpret_cast<uintptr_t>(block->end);
117     const usize padding = static_cast<usize>(-current) & (Alignment - 1);
118     const uintptr_t aligned = current + padding;
119     if ( aligned > end || bytes > end - aligned ) return false;
120     result = reinterpret_cast<byte *>(aligned);
121     block->cursor = result + bytes;
122     return true;
123   }

125   [[gnu::always_inline]] static inline void
126   __record_allocation(chunk<byte> result, usize bytes) noexcept
127   {
128     __last_ptr = result.ptr;
129 #if defined(MICRON_ALLOCATOR_STATS)
130     ++__stats.allocations;
131     __stats.bytes_requested += bytes;
132     __stats.bytes_granted += result.len;
133     __stats.current_bytes += result.len;
134     if ( __stats.current_bytes > __stats.peak_bytes ) __stats.peak_bytes = __stats.current_bytes;
135 #else
136     (void)bytes;
137 #endif

140   [[gnu::noinline]] static chunk<byte>
141   __allocate_slow(usize bytes, usize alignment)
142   {
143     for ( __block *block = __current ? __current->next : __head; block; block = block->next ) {
144       chunk<byte> result = __try_allocate(block, bytes, alignment);
145       if ( result.ptr ) {
146         __current = block;
147         __record_allocation(result, bytes);
148         return result;
149       }
150     }
151     __block *block = __new_block(bytes, alignment);
152     chunk<byte> result = __try_allocate(block, bytes, alignment);
153     __record_allocation(result, bytes);
154     return result;
155   }

172   [[nodiscard]] static chunk<byte>
173   create(usize bytes, usize alignment)
174   {
175     allocation_validate_alignment(alignment);
176     if ( bytes == 0 ) return { nullptr, 0 };
177     __guard guard;
178     __block *block = __current;
179     if ( block ) [[likely]] {
180       chunk<byte> result = __try_allocate(block, bytes, alignment);
181       if ( result.ptr ) [[likely]] {
182         __record_allocation(result, bytes);
183         return result;
184       }
185     }
186     return __allocate_slow(bytes, alignment);
187   }

189   template<usize Alignment>
190   [[nodiscard, gnu::always_inline]] static inline chunk<byte>
191   create(usize bytes)
192   {
193     static_assert(allocation_is_power_of_two(Alignment), "allocator_monotonic: alignment must be a non-zero power of two");
194     if ( bytes == 0 ) return { nullptr, 0 };
195     __guard guard;
196     __block *block = __current;
197     if ( block ) [[likely]] {
198       byte *ptr;
199       if ( __try_allocate<Alignment>(block, bytes, ptr) ) [[likely]] {
200         chunk<byte> result{ ptr, bytes };
201         __record_allocation(result, bytes);
202         return result;
203       }
204     }
205     return __allocate_slow(bytes, Alignment);
206   }

214   [[nodiscard]] static chunk<byte>
215   resize(chunk<byte> old, usize bytes, usize preserve_bytes, usize alignment)
216   {
217     allocation_validate_alignment(alignment);
218     if ( old.ptr == nullptr ) return create(bytes, alignment);
219     {
220       __guard guard;
221       __block *block = __current;
222       if ( old.ptr == __last_ptr && block && (reinterpret_cast<uintptr_t>(old.ptr) & (alignment - 1)) == 0 ) {
223         const usize offset = static_cast<usize>(reinterpret_cast<uintptr_t>(old.ptr) - reinterpret_cast<uintptr_t>(block->begin));
224         const usize capacity = static_cast<usize>(block->end - block->begin);
225         if ( offset <= capacity && old.len <= capacity - offset && block->cursor == old.ptr + old.len && bytes <= capacity - offset ) {
226           block->cursor = old.ptr + bytes;
227 #if defined(MICRON_ALLOCATOR_STATS)
228           ++__stats.resizes;
229           __stats.bytes_requested += bytes;
230           usize used = 0;
231           for ( __block *it = __head; it; it = it->next ) used += static_cast<usize>(it->cursor - it->begin);
232           __stats.current_bytes = used;
233 #endif
234           if ( bytes == 0 ) {
235             __last_ptr = nullptr;
236             return { nullptr, 0 };
237           }
238           return { old.ptr, bytes };
239         }
240       }
241     }
242     chunk<byte> next = create(bytes, alignment);
243     const usize copied = micron::min(preserve_bytes, old.len, next.len);
244     if ( copied != 0 ) micron::memcpy(next.ptr, old.ptr, copied);
245 #if defined(MICRON_ALLOCATOR_STATS)
246     {
247       __guard guard;
248       ++__stats.resizes;
249       __stats.bytes_copied += copied;
250     }
251 #endif
252     return next;
253   }

317   static void
318   release() noexcept
319   {
320     __guard guard;
321     while ( __head ) {
322       __block *block = __head;
323       __head = block->next;
324       chunk<byte> storage = block->storage;
325       const usize alignment = block->allocation_alignment;
326       block->~__block();
327       __allocator_destroy<Upstream>(storage, alignment);
328     }
329     __tail = nullptr;
330     __current = nullptr;
331     __last_ptr = nullptr;
332 #if defined(MICRON_ALLOCATOR_STATS)
333     ++__stats.releases;
334     __stats.blocks = 0;
335     __stats.current_bytes = 0;
336 #endif
337   }
