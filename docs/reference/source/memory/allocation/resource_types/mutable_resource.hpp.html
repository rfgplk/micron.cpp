14 template<typename T, typename Alloc> struct __owned_memory_resource: public __core_memory_resource<T> {
15   using __core = __core_memory_resource<T>;
16   using typename __core::size_type;
17 
18   size_type length;
19 
20 private:
21   [[nodiscard]] static chunk<byte>
22   __allocation(T *memory, usize capacity) noexcept
23   {
24     if ( memory == nullptr ) return { nullptr, 0 };
25     const usize usable_bytes = capacity * sizeof(T);
26     return { reinterpret_cast<byte *>(memory), __allocator_extent<Alloc, alignof(T)>(usable_bytes) };
27   }
28 
29   [[nodiscard]] static bool
30   __extent_matches(chunk<byte> memory, usize required_bytes = 0) noexcept
31   {
32     if ( memory.ptr == nullptr ) return memory.len == 0 && required_bytes == 0;
33     if ( memory.len < required_bytes || memory.len == 0 ) return false;
34     if ( (reinterpret_cast<uintptr_t>(memory.ptr) & (alignof(T) - 1)) != 0 ) return false;
35     const usize usable_bytes = (memory.len / sizeof(T)) * sizeof(T);
36     return __allocator_extent<Alloc, alignof(T)>(usable_bytes) == memory.len;
37   }
38 
39   [[nodiscard]] static chunk<byte>
40   __create_checked(usize bytes)
41   {
42     chunk<byte> memory = __allocator_create<Alloc, alignof(T)>(bytes);
43     if ( !__extent_matches(memory, bytes) ) {
44       __allocator_destroy<Alloc, alignof(T)>(memory);
45       exc<except::memory_error>("allocator: allocation extent cannot be reconstructed from element capacity");
46     }
47     return memory;
48   }
49 
50   [[nodiscard]] static chunk<byte> &&
51   __adopt_checked(chunk<byte> &memory)
52   {
53     if ( !__extent_matches(memory) )
54       exc<except::memory_error>("allocator: adopted allocation extent cannot be reconstructed from element capacity");
55     return micron::move(memory);
56   }
57 
58   static void
59   __destroy_range(T *memory, usize count) noexcept
60   {
61     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
62       for ( usize i = 0; i < count; ++i ) memory[i].~T();
63     }
64   }
65 
66   void
67   __release_raw() noexcept
68   {
69     if ( __core::memory ) __allocator_destroy<Alloc, alignof(T)>(__allocation(__core::memory, __core::capacity));
70     __core::memory = nullptr;
71     __core::capacity = 0;
72   }
73 
74   void
75   __relocate_to(usize elements)
76   {
77     const usize bytes = allocation_multiply_or_throw(elements, sizeof(T));
78     const usize retained = length < elements ? length : elements;
79 
80     if constexpr ( micron::is_trivially_copyable_v<T> ) {
81       chunk<byte> next = __allocator_resize_bytes<Alloc, alignof(T)>(__allocation(__core::memory, __core::capacity), bytes,
82                                                                      allocation_multiply_or_throw(retained, sizeof(T)));
83       __core::memory = micron::ptr_cast<T *>(next.ptr);
84       __core::capacity = next.len / sizeof(T);
85       length = retained;
86       if ( !__extent_matches(next, bytes) ) {
87         __allocator_destroy<Alloc, alignof(T)>(next);
88         __core::memory = nullptr;
89         __core::capacity = 0;
90         length = 0;
91         exc<except::memory_error>("allocator: resized extent cannot be reconstructed from element capacity");
92       }
93       return;
94     }
95 
96     chunk<byte> next = __create_checked(bytes);
97     T *destination = micron::ptr_cast<T *>(next.ptr);
98     usize constructed = 0;
99 #if !defined(__micron_freestanding) || defined(__micron_eh)
100     try {
101 #endif
102       for ( ; constructed < retained; ++constructed ) {
103         if constexpr ( micron::is_nothrow_move_constructible_v<T> || !micron::is_copy_constructible_v<T> )
104           micron::construct_at(destination + constructed, micron::move(__core::memory[constructed]));
105         else
106           micron::construct_at(destination + constructed, __core::memory[constructed]);
107       }
108 #if !defined(__micron_freestanding) || defined(__micron_eh)
109     } catch ( ... ) {
110       __destroy_range(destination, constructed);
111       __allocator_destroy<Alloc, alignof(T)>(next);
112       throw;
113     }
114 #endif
115 
116     __destroy_range(__core::memory, length);
117     __release_raw();
118     __core::memory = destination;
119     __core::capacity = next.len / sizeof(T);
120     length = retained;
121   }
122 
123 public:
124   ~__owned_memory_resource() { __release_raw(); }
125 
126   __owned_memory_resource(nullptr_t) noexcept : __core(), length(0) { }
127 
128   __owned_memory_resource() : __core(), length(0)
129   {
130     const usize minimum = Alloc::auto_size() < sizeof(T) ? sizeof(T) : Alloc::auto_size();
131     const usize bytes = allocation_round_up_or_throw(minimum, sizeof(T));
132     chunk<byte> memory = __create_checked(bytes);
133     __core::accept(micron::move(memory));
134   }
135 
136   explicit __owned_memory_resource(usize elements) : __core(__create_checked(allocation_multiply_or_throw(elements, sizeof(T)))), length(0)
137   {
138   }
139 
140   explicit __owned_memory_resource(chunk<byte> &&memory) : __core(__adopt_checked(memory)), length(0) { }
141 
142   __owned_memory_resource(const __owned_memory_resource &) = delete;
143   __owned_memory_resource &operator=(const __owned_memory_resource &) = delete;
144 
145   void accept(const chunk<byte> &) = delete;
146 
147   void
148   accept(chunk<byte> &&memory)
149   {
150     __core::accept(__adopt_checked(memory));
151   }
152 
153   __owned_memory_resource(__owned_memory_resource &&other) noexcept : __core(micron::move(other)), length(other.length)
154   {
155     other.length = 0;
156   }
157 
158   __owned_memory_resource &
159   operator=(__owned_memory_resource &&other) noexcept
160   {
161     if ( this == micron::addressof(other) ) return *this;
162     __destroy_range(__core::memory, length);
163     __release_raw();
164     __core::operator=(micron::move(other));
165     length = other.length;
166     other.length = 0;
167     return *this;
168   }
169 
170   __owned_memory_resource &
171   swap(__owned_memory_resource &other) noexcept
172   {
173     micron::swap(__core::capacity, other.capacity);
174     micron::swap(__core::memory, other.memory);
175     micron::swap(length, other.length);
176     return *this;
177   }
178 
179   [[nodiscard]] bool
180   is_zero() const noexcept
181   {
182     return !__core::alive();
183   }
184 
185   [[nodiscard]] bool
186   has_space(usize elements) const noexcept
187   {
188     return length <= __core::capacity && elements <= __core::capacity - length;
189   }
190 
191   [[nodiscard]] chunk<byte>
192   data() const noexcept
193   {
194     return __allocation(__core::memory, __core::capacity);
195   }
196 
197   [[nodiscard]] chunk<byte>
198   operator*() const noexcept
199   {
200     return data();
201   }
202 
203   [[nodiscard]] chunk<byte>
204   operator*() noexcept
205   {
206     return data();
207   }
208 
209   void
210   free() noexcept
211   {
212     __release_raw();
213     length = 0;
214   }
215 
216   void
217   realloc(usize elements)
218   {
219     if ( elements == __core::capacity ) return;
220     __relocate_to(elements);
221   }
222 
223   void
224   expand(usize elements)
225   {
226     if ( elements <= __core::capacity ) return;
227     __relocate_to(elements);
228   }
229 
230   [[nodiscard]] static usize
231   recommended_capacity(usize current, usize minimum)
232   {
233     const usize current_bytes = allocation_multiply_or_throw(current, sizeof(T));
234     const usize minimum_bytes = allocation_multiply_or_throw(minimum, sizeof(T));
235     const usize recommended = __allocator_recommend<Alloc>(current_bytes, minimum_bytes);
236     if ( recommended == __allocation_max ) exc<except::length_error>("allocator: growth recommendation overflow");
237     return recommended / sizeof(T) + (recommended % sizeof(T) != 0);
238   }
