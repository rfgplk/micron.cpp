36 class __external_guard
37 {
38 public:
39   __external_guard() noexcept
40   {
41     while ( __external_lock.test_and_set(micron::memory_order::acquire) ) __cpu_pause();

50 [[nodiscard]] inline bool
51 __external_bounds(byte *base, usize len, uintptr_t &lo, uintptr_t &hi) noexcept
52 {
53   if ( base == nullptr || len == 0 ) return false;
54   lo = reinterpret_cast<uintptr_t>(base);
55   if ( len > micron::numeric_limits<uintptr_t>::max() - lo ) return false;
56   hi = lo + len;
57   return hi > lo;
58 }

60 [[nodiscard]] inline bool
61 __external_overlap(uintptr_t alo, uintptr_t ahi, const __external_range &range) noexcept
62 {
63   if ( range.base == nullptr ) return false;
64   const uintptr_t blo = reinterpret_cast<uintptr_t>(range.base);
65   const uintptr_t bhi = blo + range.len;
66   return alo < bhi && blo < ahi;
67 }

69 [[nodiscard, gnu::cold, gnu::noinline]] inline __external_page *
70 __external_new_page() noexcept
71 {
72   constexpr usize header = __builtin_offsetof(__external_page, ranges);
73   const usize mapping_len = micron::page_size < header + sizeof(__external_range) ? header + sizeof(__external_range) : micron::page_size;
74   addr_t *memory
75       = micron::mmap(nullptr, mapping_len, micron::prot_read | micron::prot_write, micron::map_private | micron::map_anonymous, -1, 0);
76   if ( micron::mmap_failed(memory) ) return nullptr;
77   auto *page = reinterpret_cast<__external_page *>(memory);
78   page->next = nullptr;
79   page->capacity = (mapping_len - header) / sizeof(__external_range);
80   page->used = 0;
81   page->high_water = 1;
82   for ( usize i = 0; i < page->capacity; ++i ) page->ranges[i] = { nullptr, 0, external_provenance::none };
83   return page;
84 }

86 [[nodiscard, gnu::noinline]] inline bool
87 __external_register_slow(byte *base, usize len, external_provenance provenance, uintptr_t lo, uintptr_t hi) noexcept
88 {
89   __external_range *vacant = nullptr;
90   __external_page *vacant_page = nullptr;
91   __external_page *tail_vacant = nullptr;
92   for ( __external_page *page = __external_pages; page; page = page->next ) {
93     for ( usize i = 0; i < page->high_water; ++i ) {
94       __external_range &range = page->ranges[i];
95       if ( __external_overlap(lo, hi, range) ) return false;
96       if ( vacant == nullptr && range.base == nullptr ) {
97         vacant = &range;
98         vacant_page = page;
99       }
100     }
101     if ( vacant == nullptr && tail_vacant == nullptr && page->high_water < page->capacity ) tail_vacant = page;
102   }
103 
104   if ( vacant == nullptr && tail_vacant ) {
105     vacant = &tail_vacant->ranges[tail_vacant->high_water++];
106     vacant_page = tail_vacant;
107   }
108 
109   if ( vacant == nullptr ) {
110     __external_page *page = __external_new_page();
111     if ( page == nullptr ) return false;
112     page->next = __external_pages;
113     __external_pages = page;
114     vacant = &page->ranges[0];
115     vacant_page = page;
116   }
117 
118   *vacant = { base, len, provenance };
119   ++vacant_page->used;
120   ++__external_count;
121   return true;
122 }

124 [[nodiscard, gnu::always_inline]] inline bool
125 register_external(byte *base, usize len, external_provenance provenance = external_provenance::caller) noexcept
126 {
127   uintptr_t lo;
128   uintptr_t hi;
129   if ( !__external_bounds(base, len, lo, hi) ) return false;
130 
131   __external_guard guard;
132   if ( __external_count == 0 && __external_pages ) {
133     __external_page *page = __external_pages;
134     page->ranges[0] = { base, len, provenance };
135     page->used = 1;
136     page->high_water = 1;
137     __external_count = 1;
138     return true;
139   }
140   return __external_register_slow(base, len, provenance, lo, hi);
141 }

143 [[nodiscard, gnu::noinline]] inline bool
144 __external_unregister_slow(byte *base, usize len) noexcept
145 {
146   for ( __external_page *page = __external_pages; page; page = page->next ) {
147     for ( usize i = 0; i < page->high_water; ++i ) {
148       __external_range &range = page->ranges[i];
149       if ( range.base != base ) continue;
150       if ( range.len != len ) return false;
151       range = { nullptr, 0, external_provenance::none };
152       --page->used;
153       --__external_count;
154       if ( i + 1 == page->high_water )
155         while ( page->high_water && page->ranges[page->high_water - 1].base == nullptr ) --page->high_water;
156       return true;
157     }
158   }
159   return false;
160 }

162 [[nodiscard, gnu::always_inline]] inline bool
163 unregister_external(byte *base, usize len) noexcept
164 {
165   if ( base == nullptr || len == 0 ) return false;
166   __external_guard guard;
167   __external_page *page = __external_pages;
168   if ( page && page->high_water && page->ranges[0].base == base ) {
169     __external_range &range = page->ranges[0];
170     if ( range.len != len ) return false;
171     range = { nullptr, 0, external_provenance::none };
172     --page->used;
173     --__external_count;
174     if ( page->high_water == 1 ) page->high_water = 0;
175     return true;
176   }
177   return __external_unregister_slow(base, len);
178 }

180 [[nodiscard, gnu::cold]] inline bool
181 external_is_present(const byte *base) noexcept
182 {
183   if ( base == nullptr ) return false;
184   __external_guard guard;
185   for ( const __external_page *page = __external_pages; page; page = page->next )
186     for ( usize i = 0; i < page->high_water; ++i )
187       if ( page->ranges[i].base == base ) return true;
188   return false;
189 }

191 [[nodiscard, gnu::cold]] inline bool
192 external_within(const byte *address) noexcept
193 {
194   if ( address == nullptr ) return false;
195   const uintptr_t target = reinterpret_cast<uintptr_t>(address);
196   __external_guard guard;
197   for ( const __external_page *page = __external_pages; page; page = page->next ) {
198     for ( usize i = 0; i < page->high_water; ++i ) {
199       const __external_range &range = page->ranges[i];
200       if ( range.base == nullptr ) continue;
201       const uintptr_t lo = reinterpret_cast<uintptr_t>(range.base);
202       if ( target >= lo && target - lo < range.len ) return true;
203     }
204   }
205   return false;
206 }

208 [[nodiscard, gnu::cold]] inline usize
209 external_query_size(const byte *base) noexcept
210 {
211   if ( base == nullptr ) return 0;
212   __external_guard guard;
213   for ( const __external_page *page = __external_pages; page; page = page->next )

216   return 0;

219 [[nodiscard, gnu::cold]] inline external_provenance
220 external_query_provenance(const byte *base) noexcept
221 {
222   if ( base == nullptr ) return external_provenance::none;
223   __external_guard guard;
224   for ( const __external_page *page = __external_pages; page; page = page->next )
225     for ( usize i = 0; i < page->high_water; ++i )
226       if ( page->ranges[i].base == base ) return page->ranges[i].provenance;
227   return external_provenance::none;
