30 inline auto
31 check_oom(void) -> bool
32 {
33   if constexpr ( __default_oom_enable ) {
34 #if defined(__micron_freestanding)
35     static u32 __oom_budget = 0;
36 #else
37     static thread_local u32 __oom_budget = 0;
38 #endif
39     if ( ++__oom_budget < __default_oom_check_interval ) [[likely]]
40       return false;
41     __oom_budget = 0;
42     micron::resources rs;
43     usize total_ram = rs.total_memory;
44     usize free_ram = rs.free_memory;
45     if ( ((float)free_ram / (float)total_ram) <= __default_oom_limit_error ) {
46       return true;
47     } else if ( ((float)free_ram / (float)total_ram) <= __default_oom_limit_warn ) {
48       // TODO: add an actual warning here, false for now
49       return false;
50     }
51   }
52   return false;
53 }
