33 namespace abc

35 
36 struct __mpsc_free_payload {
37   byte *ptr;
38   usize size;
39 };
40 
41 template<usize N>
42   requires(N > 0 and (N & (N - 1)) == 0)
43 class __mpsc_free_queue
44 {
45 
46   friend struct __ins_probe;
47 
48   static constexpr u64 __cache_line = micron::cache_line_size();
49   static constexpr usize __capacity = N;
50   static constexpr usize __mask = N - 1;
51 
52   struct alignas(__cache_line) __cell {
53     micron::atomic_token<usize> seq;
54     __mpsc_free_payload payload;
55   };
56 
57   alignas(__cache_line) __cell __cells[__capacity];
58   alignas(__cache_line) micron::atomic_token<usize> __tail;
59   alignas(__cache_line) usize __head;
60 
61 public:
62   __mpsc_free_queue() noexcept : __tail(0), __head(0)
63   {
64     for ( usize i = 0; i < __capacity; ++i ) {
65       __cells[i].seq.store(i, micron::memory_order_relaxed);
66     }
67   }
68 
69   __mpsc_free_queue(const __mpsc_free_queue &) = delete;
70   __mpsc_free_queue(__mpsc_free_queue &&) = delete;
71   __mpsc_free_queue &operator=(const __mpsc_free_queue &) = delete;
72   __mpsc_free_queue &operator=(__mpsc_free_queue &&) = delete;
73 
74   static constexpr usize
75   capacity() noexcept
76   {
77     return __capacity;
78   }
79 
80   [[gnu::always_inline]] inline bool
81   maybe_nonempty() const noexcept
82   {
83     return __tail.get(micron::memory_order_relaxed) != __head;
84   }
85 
86   [[gnu::always_inline]] inline bool
87   push(byte *p, usize sz) noexcept
88   {
89     usize tail = __tail.get(micron::memory_order_relaxed);
90     unsigned backoff = 1u;
91     for ( ;; ) {
92       __cell &c = __cells[tail & __mask];
93       const usize seq = c.seq.get(micron::memory_order_acquire);
94       const long long diff = static_cast<long long>(seq) - static_cast<long long>(tail);
95       if ( diff == 0 ) {
96         if ( __tail.compare_exchange_weak(tail, tail + 1u, micron::memory_order_relaxed, micron::memory_order_relaxed) ) {
97           c.payload.ptr = p;
98           c.payload.size = sz;
99           c.seq.store(tail + 1u, micron::memory_order_release);
100           return true;
101         }
102         backoff = micron::__spin_backoff(backoff);
103         tail = __tail.get(micron::memory_order_relaxed);
104       } else if ( diff < 0 ) {
105         return false;
106       } else {
107         backoff = micron::__spin_backoff(backoff);
108         tail = __tail.get(micron::memory_order_relaxed);
109       }
110     }
111   }
112 
113   [[gnu::always_inline]] inline bool
114   pop(__mpsc_free_payload &out) noexcept
115   {
116     const usize head = __head;
117     __cell &c = __cells[head & __mask];
118     const usize seq = c.seq.get(micron::memory_order_acquire);
119     const long long diff = static_cast<long long>(seq) - static_cast<long long>(head + 1u);
120     if ( diff != 0 ) return false;
121     out = c.payload;
122     c.seq.store(head + __capacity, micron::memory_order_release);
123     __head = head + 1u;
124     return true;
125   }
126 
127   template<typename Fn>
128   [[gnu::always_inline]] inline u32
129   drain(Fn &&fn) noexcept
130   {
131     u32 n = 0;
132     __mpsc_free_payload p;
133     while ( pop(p) ) {
134       fn(p.ptr, p.size);
135       ++n;
136     }
137     return n;
138   }
139 };
