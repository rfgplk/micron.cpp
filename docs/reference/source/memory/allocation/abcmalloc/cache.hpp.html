21 #pragma once
22 #include "book.hpp"
23 #include "config.hpp"
24 #include "harden.hpp"
25 #include "hooks.hpp"
26 #include "stats.hpp"
27 
28 namespace abc
29 {
30 struct cache {
31   micron::__chunk<byte>
32   __heap_grow(const usize sz)
33   {
34     return { reinterpret_cast<byte *>(micron::sbrk(static_cast<intptr_t>(sz))), sz };
35   }
36 
37   void
38   __heap_shrink(const max_t sz)
39   {
40     micron::sbrk(static_cast<intptr_t>((-1) * sz));
41   }
42 
43   micron::__chunk<byte>
44   grow(const usize sz)
45   {
46     collect_stats<stat_type::alloc>();
47     collect_stats<stat_type::total_memory_req>(sz);
48     micron::__chunk<byte> memory;
49     if ( memory = __heap_grow(sz); !memory.zero() ) {
50       zero_on_alloc(memory.ptr, memory.len);
51       sanitize_on_alloc(memory.ptr, memory.len);
52       collect_stats<stat_type::total_memory_throughput>(memory.len);
53       return memory;
54     }
55     return { (byte *)-1, micron::numeric_limits<usize>::max() };
56   }
57 
58   void
59   shrink(const max_t sz)
60   {
61     collect_stats<stat_type::dealloc>();
62     collect_stats<stat_type::total_memory_freed>(static_cast<usize>(sz));
63     __heap_shrink(sz);
64   }
65 };
66 };      // namespace abc
