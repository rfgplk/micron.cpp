35 //^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
36 //  __tlsf_list: O(1) two level segregated fit
37 // our buddy list greatly underperformed on lots of small tiny allocations
38 // this has been implemented to alleviate that pressure and speed up performance for general purpose code
39 // all ops are o(1)
40 //
41 //  block layout (within __hdr_offset of 32b, reused for compat across systems):
42 //    offset 0:  total block size in bytes (incl header)
43 //    offset 4:  __block_alloc / __block_free / __block_tombstone
44 //    offset 8:  previous physical block
45 //    offset 16: free-list link
46 //    offset 24: free-list link
47 //    offset 32: meta
48 
49 template<typename T, i64 Min, i32 Mx = 64>
50   requires(micron::is_trivially_constructible_v<T> and micron::is_trivially_destructible_v<T> and (bool)((Min & (Min - 1)) == 0))
51 struct __tlsf_list {

53   static constexpr usize __block_align = __hdr_offset;         // 32
54   static constexpr usize __min_block = __block_align * 2;      // 64
55   static constexpr i32 __sl_log2 = 4;
56   static constexpr i32 __sl_count = 1 << __sl_log2;                 // 16
57   static constexpr i32 __fl_shift = 6;                              // log2(64)
58   static constexpr i32 __fl_count = 26;                             // indices 0..25
59   static constexpr i32 __list_count = __fl_count * __sl_count;      // 416
60   static constexpr i32 __temporal_ring = 2;
61 
62   static_assert(__block_align >= 32, "header must fit in __hdr_offset");
63   static_assert((1ULL << __fl_shift) == __min_block, "__fl_shift must equal log2(__min_block)");
64   static_assert(__fl_count <= 32, "__fl_count must fit in u32 bitmap");
65   static_assert(Mx <= 64, "Mx must be <= 64");

107   __attribute__((always_inline)) static inline void
108   mapping_insert(usize size, i32 &fi, i32 &si) noexcept
109   {
110     i32 fl = fls64(size);
111     fi = fl - __fl_shift;
112     i32 shift = fl - __sl_log2;
113     si = (i32)((size >> shift) & (__sl_count - 1));
114   }

116   __attribute__((always_inline)) static inline void
117   mapping_search(usize size, i32 &fi, i32 &si) noexcept
118   {
119     i32 fl = fls64(size);
120     i32 shift = fl - __sl_log2;
121     size += ((usize)1 << shift) - 1;
122     fl = fls64(size);
123     fi = fl - __fl_shift;
124     shift = fl - __sl_log2;
125     si = (i32)((size >> shift) & (__sl_count - 1));
126   }

128   __attribute__((always_inline)) inline tlsf_hdr *

131     return micron::ptr_cast<tlsf_hdr *>(reinterpret_cast<byte *>(b) + (usize)b->bsize);

134   __attribute__((always_inline)) inline void
135   fl_insert(tlsf_hdr *block) noexcept
136   {
137     i32 fi, si;
138     mapping_insert((usize)block->bsize, fi, si);
139     i32 i = idx(fi, si);
140 
141     tlsf_hdr *head = heads[i];
142     block->next_free = head;
143     block->prev_free = nullptr;
144     if ( head ) head->prev_free = block;
145     heads[i] = block;
146 
147     block->flags = __block_free;
148     fl_bitmap |= (1u << fi);
149     sl_bitmap[fi] |= (1u << si);
150   }
151 
152   __attribute__((always_inline)) inline void
153   fl_remove(tlsf_hdr *block) noexcept
154   {
155     i32 fi, si;
156     mapping_insert((usize)block->bsize, fi, si);
157     i32 i = idx(fi, si);
158 
159     tlsf_hdr *prev = block->prev_free;
160     tlsf_hdr *next = block->next_free;
161     if ( prev )
162       prev->next_free = next;
163     else
164       heads[i] = next;
165 
166     if ( next ) next->prev_free = prev;
167 
168     if ( !heads[i] ) {
169       sl_bitmap[fi] &= ~(1u << si);
170       if ( !sl_bitmap[fi] ) fl_bitmap &= ~(1u << fi);
171     }
172   }

174   __attribute__((always_inline)) inline tlsf_hdr *
175   find_free(usize needed) noexcept
176   {
177     i32 fi, si;
178     mapping_search(needed, fi, si);
179     if ( fi >= fl_count ) return nullptr;
180 
181     u32 sl_map = sl_bitmap[fi] & (~0u << si);
182     if ( !sl_map ) {
183 
184       u32 fl_map = fl_bitmap & (~0u << (fi + 1));
185       if ( !fl_map ) return nullptr;
186       fi = __builtin_ctz(fl_map);
187       sl_map = sl_bitmap[fi];
188     }
189     si = __builtin_ctz(sl_map);
190 
191     const i32 i = idx(fi, si);
192     tlsf_hdr *block = heads[i];
193     tlsf_hdr *next = block->next_free;
194     heads[i] = next;
195     if ( next ) {
196       next->prev_free = nullptr;
197     } else {
198       const u32 new_sl = sl_bitmap[fi] & ~(1u << si);
199       sl_bitmap[fi] = new_sl;
200       if ( new_sl == 0 ) fl_bitmap &= ~(1u << fi);
201     }
202     return block;
203   }

205   __attribute__((always_inline)) inline void
206   try_split(tlsf_hdr *block, usize needed) noexcept
207   {
208     usize remain = (usize)block->bsize - needed;
209     if ( remain < __min_block ) return;
210 
211     block->bsize = (u32)needed;
212 
213     tlsf_hdr *rem = micron::ptr_cast<tlsf_hdr *>(reinterpret_cast<byte *>(block) + needed);
214     rem->bsize = (u32)remain;
215     rem->prev_phys = block;
216 
217     tlsf_hdr *after = next_phys(rem);
218     after->prev_phys = rem;
219 
220     fl_insert(rem);
221   }

223   __attribute__((always_inline)) inline void
224   coalesce_and_free(tlsf_hdr *block) noexcept
225   {
226     bool merged = false;
227     // forward merge
228     tlsf_hdr *next = next_phys(block);
229     if ( next->flags == __block_free ) {
230       fl_remove(next);
231       block->bsize = (u32)((usize)block->bsize + (usize)next->bsize);
232       merged = true;
233     }
234 
235     // backward merge
236     tlsf_hdr *prev = block->prev_phys;
237     if ( prev && prev->flags == __block_free ) {
238       fl_remove(prev);
239       prev->bsize = (u32)((usize)prev->bsize + (usize)block->bsize);
240       block = prev;
241       merged = true;
242     }
243 
244     if ( merged ) next_phys(block)->prev_phys = block;
245     fl_insert(block);
246   }

248   void
249   __impl_zero_arrays() noexcept
250   {
251     fl_bitmap = 0;
252     for ( i32 i = 0; i < __fl_count; ++i ) sl_bitmap[i] = 0;
253     for ( i32 i = 0; i < __list_count; ++i ) {
254       heads[i] = nullptr;
255       for ( i32 r = 0; r < __temporal_ring; ++r ) temporal_active[i][r] = nullptr;
256       temporal_rotor[i] = 0;
257     }
258   }

260   void
261   __impl_init_memory(byte *_ptr, usize _len)
262   {
263     uintptr_t p = (uintptr_t)_ptr;
264     uintptr_t r = align_up(p, __block_align);
265     usize adjust = r - p;
266     if ( _len <= adjust ) goto fail;
267     {
268       byte *aligned = (byte *)r;
269       usize usable = _len - adjust;
270       usable = (usable / __block_align) * __block_align;
271 
272       // need start-sentinel + at least __min_block + end-sentinel
273       if ( usable < 2 * __block_align + __min_block ) goto fail;
274 
275       // reject pools > 4 GiB
276       usize data = usable - 2 * __block_align;
277       if ( data > (usize)0xFFFFFFFFu ) goto fail;
278 
279       base = aligned;
280       total = data;
281 
282       i32 fl = fls64(total);
283       fl_count = fl - __fl_shift + 1;
284       if ( fl_count > __fl_count ) fl_count = __fl_count;
285       if ( fl_count < 1 ) fl_count = 1;
286 
287       __impl_zero_arrays();
288 
289       tlsf_hdr *ss = micron::ptr_cast<tlsf_hdr *>(base);
290       ss->bsize = (u32)__block_align;
291       ss->flags = __block_alloc;
292       ss->prev_phys = nullptr;
293       ss->next_free = nullptr;
294       ss->prev_free = nullptr;
295 
296       tlsf_hdr *es = micron::ptr_cast<tlsf_hdr *>(base + __block_align + total);
297       es->bsize = (u32)__block_align;
298       es->flags = __block_alloc;
299       es->prev_phys = nullptr;
300       es->next_free = nullptr;
301       es->prev_free = nullptr;
302 
303       tlsf_hdr *blk = micron::ptr_cast<tlsf_hdr *>(base + __block_align);
304       blk->bsize = (u32)total;
305       blk->flags = __block_free;
306       blk->prev_phys = ss;
307 
308       es->prev_phys = blk;
309 
310       fl_insert(blk);
311       return;
312     }
313   fail:
314     base = nullptr;
315     total = 0;
316     fl_count = 0;
317   }

396   __attribute__((always_inline)) static inline usize
397   adjusted_block_size(usize user_n) noexcept
398   {
399     constexpr usize overhead = sizeof(micron::simd::i256) + __hdr_offset;
400     constexpr usize align_mask = __block_align - 1;
401     if ( user_n > micron::numeric_limits<usize>::max() - overhead - align_mask ) return 0;
402     usize needed = (user_n + overhead + align_mask) & ~align_mask;
403     return needed < __min_block ? __min_block : needed;
404   }

406   __attribute__((always_inline)) inline bool
407   __clear_temporal_class(tlsf_hdr *block, usize size) noexcept
408   {
409     i32 fi, si;
410     mapping_search(size, fi, si);
411     if ( fi < 0 || fi >= fl_count ) return false;
412     const i32 i = idx(fi, si);
413     for ( i32 r = 0; r < __temporal_ring; ++r ) {
414       if ( temporal_active[i][r] == block ) {
415         temporal_active[i][r] = nullptr;
416         return true;
417       }
418     }
419     return false;
420   }

429   T
430   allocate(usize n) noexcept
431   {
432     if ( !base ) return { nullptr, 0 };
433 
434     usize needed = adjusted_block_size(n);
435     if ( needed == 0 ) return { nullptr, 0 };
436 
437     tlsf_hdr *block = find_free(needed);
438     if ( !block ) return { nullptr, 0 };
439 
440     try_split(block, needed);
441     block->flags = __block_alloc;
442     const usize block_size = (usize)block->bsize;
443     allocated_bytes += block_size;
444 
445     return { reinterpret_cast<byte *>(block) + __hdr_offset, block_size - __hdr_offset };
446   }

448   T
449   temporal_allocate(usize n) noexcept
450   {
451     if ( !base ) return { nullptr, 0 };
452 
453     usize needed = adjusted_block_size(n);
454     if ( needed == 0 ) return { nullptr, 0 };
455 
456     i32 fi, si;
457     mapping_search(needed, fi, si);
458     if ( fi >= 0 && fi < fl_count ) {
459       i32 i = idx(fi, si);
460       u8 r = temporal_rotor[i] & (__temporal_ring - 1);
461       tlsf_hdr *blk = temporal_active[i][r];
462       if ( !blk ) {
463         r ^= 1u;
464         blk = temporal_active[i][r];
465       }
466       if ( blk ) {
467         temporal_rotor[i] = r ^ 1u;
468         return { reinterpret_cast<byte *>(blk) + __hdr_offset, (usize)blk->bsize - __hdr_offset };
469       }
470     }
471 
472     tlsf_hdr *block = find_free(needed);
473     if ( !block ) return { nullptr, 0 };
474 
475     try_split(block, needed);
476     block->flags = __block_alloc | __block_temporal;
477     const usize block_size = (usize)block->bsize;
478     allocated_bytes += block_size;
479 
480     if ( fi >= 0 && fi < fl_count ) {
481       i32 i = idx(fi, si);
482       const u8 r = temporal_rotor[i];
483       temporal_active[i][r] = block;
484       temporal_rotor[i] = r ^ 1u;
485     }
486 
487     return { reinterpret_cast<byte *>(block) + __hdr_offset, block_size - __hdr_offset };
488   }

490   T
491   allocate_exact(usize n) noexcept
492   {
493     if ( !base ) return { nullptr, 0 };
494     if ( n < __min_block ) return { nullptr, 0 };
495     if ( (n & (n - 1)) != 0 ) return { nullptr, 0 };
496 
497     i32 fi, si;
498     mapping_insert(n, fi, si);
499     if ( fi < 0 || fi >= fl_count ) return { nullptr, 0 };
500 
501     i32 i = idx(fi, si);
502     if ( !heads[i] ) return { nullptr, 0 };
503 
504     tlsf_hdr *block = heads[i];
505     fl_remove(block);
506     block->flags = __block_alloc;
507     allocated_bytes += (usize)block->bsize;
508 
509     return { reinterpret_cast<byte *>(block) + __hdr_offset, (usize)block->bsize - __hdr_offset };
510   }

512   ret_flag
513   tombstone(byte *ptr) noexcept
514   {
515     tlsf_hdr *hdr = micron::ptr_cast<tlsf_hdr *>(ptr - __hdr_offset);
516     if ( !(hdr->flags & __block_alloc) ) return { __flag_invalid };
517 
518     usize bsz = (usize)hdr->bsize;
519     hdr->flags = __block_tombstone;
520     allocated_bytes -= bsz;
521     tombstoned_bytes += bsz;
522 
523     return __flag_tombstoned;
524   }

547   ret_flag
548   deallocate(byte *ptr) noexcept
549   {
550     if ( !ptr || !base ) return __flag_failure;
551 
552     tlsf_hdr *block = micron::ptr_cast<tlsf_hdr *>(ptr - __hdr_offset);
553     usize bsz = (usize)block->bsize;
554     const i32 flags = block->flags;
555 
556     if ( flags == __block_free ) return { __flag_invalid };
557 
558     allocated_bytes -= bsz;
559     if ( flags & __block_tombstone ) tombstoned_bytes -= bsz;
560 
561     if ( flags & __block_temporal ) [[unlikely]] {
562       __clear_temporal(block, bsz);
563     }
564 
565     coalesce_and_free(block);
566     return { __flag_ok };
567   }

576   T
577   reallocate(T node, usize new_size) noexcept
578   {
579     if ( !base ) return { nullptr, 0 };
580     if ( !node.ptr ) return allocate(new_size);
581     if ( new_size == 0 ) {
582       deallocate(node);
583       return { nullptr, 0 };
584     }
585 
586     if ( node.len >= new_size && new_size > (node.len >> 1) ) return node;
587 
588     T nnode = allocate(new_size);
589     if ( !nnode.ptr ) return { nullptr, 0 };
590 
591     usize to_copy = (node.len < nnode.len) ? node.len : nnode.len;
592     micron::memcpy(nnode.ptr, node.ptr, to_copy);
593     deallocate(node);
594     return nnode;
595   }

647   usize
648   allocated_size(byte *ptr) const noexcept
649   {
650     if ( !is_allocated(ptr) ) return 0;
651     return block_size(ptr);
652   }

654 #if defined(ABCMALLOC_DOCTOR_HELP)
655   // deep corruption walk
656   template<class V>
657   void
658   __doctor_walk(V &v)
659   {
660     if ( !base ) return;
661     byte *data_lo = base + __block_align;
662     byte *es_addr = base + __block_align + total;
663 
664     // sentinels
665     tlsf_hdr *ss = reinterpret_cast<tlsf_hdr *>(base);
666     if ( ss->bsize != (u32)__block_align || ss->flags != __block_alloc ) v.note("tlsf: start sentinel corrupt", base);
667     tlsf_hdr *es = reinterpret_cast<tlsf_hdr *>(es_addr);
668     if ( es->bsize != (u32)__block_align || es->flags != __block_alloc ) v.note("tlsf: end sentinel corrupt", es_addr);
669 
670     // physical block chain via next_phys
671     tlsf_hdr *cur = reinterpret_cast<tlsf_hdr *>(data_lo);
672     tlsf_hdr *prev = ss;
673     usize guard = 0;
674     const usize maxg = (total / __min_block) + 8;
675     while ( reinterpret_cast<byte *>(cur) < es_addr && guard++ < maxg ) {
676       byte *cb = reinterpret_cast<byte *>(cur);
677       ++v.blocks;
678       u32 bs = cur->bsize;
679       if ( bs == 0 || (bs & (__block_align - 1)) != 0 ) {
680         v.note("tlsf: block bsize zero or misaligned", cb);
681         break;
682       }
683       if ( cb + bs > es_addr ) {
684         v.note("tlsf: block extends past data region", cb);
685         break;
686       }
687       const bool flags_known = (cur->flags == __block_free || cur->flags == __block_alloc || cur->flags == __block_tombstone
688                                 || cur->flags == (__block_alloc | __block_temporal));
689       if ( !flags_known ) v.note("tlsf: block flags not a valid block_flags", cb);
690       if ( cur->prev_phys != prev ) {
691         v.note("tlsf: prev_phys back-link mismatch", cb);
692         if ( v.repair ) {
693           cur->prev_phys = prev;
694           v.did_repair("tlsf: fixed prev_phys back-link", cb);
695         }
696       }
697       tlsf_hdr *nx = next_phys(cur);
698       if ( cur->flags == __block_free && reinterpret_cast<byte *>(nx) < es_addr && nx->flags == __block_free )
699         v.note("tlsf: adjacent free blocks (missed coalescing)", cb);
700       prev = cur;
701       cur = nx;
702     }
703     if ( guard >= maxg )
704       v.note("tlsf: physical walk overran (corrupt bsize chain)", base);
705     else if ( es->prev_phys != prev )
706       v.note("tlsf: end sentinel prev_phys != last block", es_addr);
707 
708     for ( i32 fi = 0; fi < __fl_count; ++fi ) {
709       u32 want_sl = 0;
710       for ( i32 si = 0; si < __sl_count; ++si )
711         if ( heads[idx(fi, si)] != nullptr ) want_sl |= (1u << si);
712       if ( sl_bitmap[fi] != want_sl ) {
713         if ( v.repair ) {
714           sl_bitmap[fi] = want_sl;
715           v.did_repair("tlsf: recomputed sl_bitmap from heads", base);
716         } else
717           v.note("tlsf: sl_bitmap disagrees with heads[]", base);
718       }
719     }
720     u32 want_fl = 0;
721     for ( i32 fi = 0; fi < __fl_count; ++fi )
722       if ( sl_bitmap[fi] ) want_fl |= (1u << fi);
723     if ( fl_bitmap != want_fl ) {
724       if ( v.repair ) {
725         fl_bitmap = want_fl;
726         v.did_repair("tlsf: recomputed fl_bitmap from sl_bitmap", base);
727       } else
728         v.note("tlsf: fl_bitmap disagrees with sl_bitmap", base);
729     }
730 
731     for ( i32 i = 0; i < __list_count; ++i ) {
732       tlsf_hdr *n = heads[i];
733       tlsf_hdr *pr = nullptr;
734       usize gc = 0;
735       while ( n && gc++ < maxg ) {
736         ++v.freelist_nodes;
737         byte *nb = reinterpret_cast<byte *>(n);
738         if ( nb < data_lo || nb >= es_addr ) {
739           v.note("tlsf: free-list node out of bounds", n);
740           if ( v.repair ) {
741             if ( pr )
742               pr->next_free = nullptr;
743             else
744               heads[i] = nullptr;
745             v.did_repair("tlsf: truncated free-list at bad node", n);
746           }
747           break;
748         }
749         if ( n->flags != __block_free ) v.note("tlsf: free-list node not flagged free", n);
750         if ( n->prev_free != pr ) {
751           v.note("tlsf: free-list prev_free back-link mismatch", n);
752           if ( v.repair ) {
753             n->prev_free = pr;
754             v.did_repair("tlsf: fixed prev_free back-link", n);
755           }
756         }
757         pr = n;
758         n = n->next_free;
759       }
760       if ( gc >= maxg ) v.note("tlsf: free-list cycle / overrun", base);
761     }
762   }
763 #endif
764 };
