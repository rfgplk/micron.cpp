33 template<u32 Slots> struct alignas(64) __tier_tcache {
34   static constexpr u32 __cache_slots = Slots;
35 
36   u32 _size[Slots];
37   byte *_ptr[Slots];
38   u32 _count;
39 
40   constexpr __tier_tcache() noexcept : _size{}, _ptr{}, _count(0) { }
41 
42   [[nodiscard, gnu::always_inline]] inline i32
43   probe(u32 sz) const noexcept
44   {
45     if ( _count == 0 ) [[unlikely]]
46       return -1;
47     __micron_gcc_unroll_8 for ( u32 i = _count; i > 0; --i )
48     {
49       if ( _size[i - 1] == sz ) [[likely]]
50         return static_cast<i32>(i - 1);
51     }
52     return -1;
53   }
54 
55   [[nodiscard, gnu::always_inline]] inline i32
56   probe_ge(u32 sz, u32 maximum = ~u32{ 0 }) const noexcept
57   {
58     if ( _count == 0 ) [[unlikely]]
59       return -1;
60     __micron_gcc_unroll_8 for ( u32 i = _count; i > 0; --i )
61     {
62       if ( _size[i - 1] >= sz && _size[i - 1] <= maximum ) [[likely]]
63         return static_cast<i32>(i - 1);
64     }
65     return -1;
66   }
67 
68   [[gnu::always_inline]] inline bool
69   push(byte *p, u32 sz) noexcept
70   {
71     if ( _count >= Slots ) return false;
72     _ptr[_count] = p;
73     _size[_count] = sz;
74     ++_count;
75     return true;
76   }
77 
78   // true iff p is already parked in this cache
79   [[nodiscard, gnu::always_inline]] inline bool
80   contains(const byte *p) const noexcept
81   {
82     for ( u32 i = 0; i < _count; ++i )
83       if ( _ptr[i] == p ) return true;
84     return false;
85   }
86 
87   [[gnu::always_inline]] inline __tcache_chunk
88   pop_at(u32 idx) noexcept
89   {
90     __tcache_chunk out = { _ptr[idx], _size[idx] };
91     --_count;
92     if ( idx != _count ) {
93       _ptr[idx] = _ptr[_count];
94       _size[idx] = _size[_count];
95     }
96     return out;
97   }
98 
99   [[gnu::always_inline]] inline void
100   invalidate_range(const byte *lo, const byte *hi) noexcept
101   {
102     u32 w = 0;
103     for ( u32 r = 0; r < _count; ++r ) {
104       if ( _ptr[r] < lo || _ptr[r] >= hi ) {
105         if ( w != r ) {
106           _ptr[w] = _ptr[r];
107           _size[w] = _size[r];
108         }
109         ++w;
110       }
111     }
112     _count = w;
113   }
114 
115   template<typename Fn>
116   [[gnu::always_inline]] inline void
117   drain(Fn &&fn) noexcept
118   {
119     for ( u32 i = 0; i < _count; ++i ) fn(_ptr[i], _size[i]);
120     _count = 0;
121   }
122 };

124 // emulating arena design but for the precise class, helps maintain 0% bmiss on rapid repeated allocs
125 struct alignas(64) __slab_tcache {
126   static constexpr u32 __num_classes = 16;
127   static constexpr u32 __slots_per = 8;
128   static constexpr u32 __class_step = 32;
129   static constexpr u32 __class_min_payload = 64;                                                            // class 0
130   static constexpr u32 __class_max_payload = __class_min_payload + (__num_classes - 1) * __class_step;      // 544
131   static constexpr u32 __class_max_user = __num_classes * __class_step;                                     // 512
132   static constexpr u32 __cache_slots = __num_classes * __slots_per;                                         // 128
133 
134   byte *_buckets[__num_classes][__slots_per];
135   u8 _counts[__num_classes];
136   u32 _occupied;
137 
138   constexpr __slab_tcache() noexcept : _buckets{}, _counts{}, _occupied(0) { }
139 
140   [[nodiscard, gnu::always_inline]] static inline i32
141   class_for_user(u32 user_n) noexcept
142   {
143     if ( user_n == 0 or user_n > __class_max_user ) return -1;
144     return static_cast<i32>((user_n - 1) >> 5);      // /32
145   }
146 
147   [[nodiscard, gnu::always_inline]] static inline i32
148   class_for_payload(u32 payload) noexcept
149   {
150     if ( payload < __class_min_payload or payload > __class_max_payload ) return -1;
151     if ( (payload & 31u) != 0 ) return -1;
152     return static_cast<i32>((payload - __class_min_payload) >> 5);
153   }
154 
155   [[gnu::always_inline]] static inline i32
156   encode_hit(u32 c, u32 s) noexcept
157   {
158     return static_cast<i32>((c << 16) | (s & 0xffffu));
159   }
160 
161   // non redzone probe
162   [[nodiscard, gnu::always_inline]] inline i32
163   probe_ge(u32 user_n) const noexcept
164   {
165     const i32 c = class_for_user(user_n);
166     if ( c < 0 ) [[unlikely]]
167       return -1;
168     if ( _counts[c] != 0 ) [[likely]]
169       return encode_hit(static_cast<u32>(c), static_cast<u32>(_counts[c] - 1));
170     const u32 m = _occupied & (~0u << static_cast<u32>(c));
171     if ( m == 0 ) return -1;
172     const u32 hit = static_cast<u32>(__builtin_ctz(m));
173     return encode_hit(hit, static_cast<u32>(_counts[hit] - 1));
174   }
175 
176   // redzoned probe
177   [[nodiscard, gnu::always_inline]] inline i32
178   probe(u32 user_n) const noexcept
179   {
180     const i32 c = class_for_user(user_n);
181     if ( c < 0 or _counts[c] == 0 ) return -1;
182     return encode_hit(static_cast<u32>(c), static_cast<u32>(_counts[c] - 1));
183   }
184 
185   [[gnu::always_inline]] inline bool
186   push(byte *p, u32 sz) noexcept
187   {
188     const i32 c = class_for_payload(sz);
189     if ( c < 0 ) return false;
190     const u32 cu = static_cast<u32>(c);
191     if ( _counts[cu] >= __slots_per ) return false;
192     _buckets[cu][_counts[cu]] = p;
193     if ( _counts[cu] == 0 ) _occupied |= (1u << cu);
194     ++_counts[cu];
195     return true;
196   }
197 
198   [[gnu::always_inline]] inline __tcache_chunk
199   pop_at(u32 enc) noexcept
200   {
201     const u32 c = enc >> 16;
202     const u32 s = enc & 0xffffu;
203     __tcache_chunk out = { _buckets[c][s], __class_min_payload + c * __class_step };
204     --_counts[c];
205     if ( s != _counts[c] ) _buckets[c][s] = _buckets[c][_counts[c]];
206     if ( _counts[c] == 0 ) _occupied &= ~(1u << c);
207     return out;
208   }
209 
210   // true iff p is parked in any size-class bucket
211   [[nodiscard, gnu::always_inline]] inline bool
212   contains(const byte *p) const noexcept
213   {
214     for ( u32 c = 0; c < __num_classes; ++c )
215       for ( u32 i = 0; i < _counts[c]; ++i )
216         if ( _buckets[c][i] == p ) return true;
217     return false;
218   }
219 
220   [[gnu::always_inline]] inline void
221   invalidate_range(const byte *lo, const byte *hi) noexcept
222   {
223     for ( u32 c = 0; c < __num_classes; ++c ) {
224       u32 w = 0;
225       for ( u32 r = 0; r < _counts[c]; ++r ) {
226         if ( _buckets[c][r] < lo or _buckets[c][r] >= hi ) {
227           if ( w != r ) _buckets[c][w] = _buckets[c][r];
228           ++w;
229         }
230       }
231       _counts[c] = static_cast<u8>(w);
232       if ( w == 0 ) _occupied &= ~(1u << c);
233     }
234   }
235 
236   template<typename Fn>
237   [[gnu::always_inline]] inline void
238   drain(Fn &&fn) noexcept
239   {
240     for ( u32 c = 0; c < __num_classes; ++c ) {
241       const u32 sz = __class_min_payload + c * __class_step;
242       for ( u32 i = 0; i < _counts[c]; ++i ) fn(_buckets[c][i], sz);
243       _counts[c] = 0;
244     }
245     _occupied = 0;
246   }
247 };
