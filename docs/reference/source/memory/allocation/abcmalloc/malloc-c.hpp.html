31 // NOTE: these should be noexcept so we avoid conflicting declarations
32 // Keep every C allocator entry point exported while coalescing the header definitions across TUs.
33 #define __MICRON_C_ALLOC_INTERPOSE __attribute__((weak, used, retain))
34 
35 extern "C" __MICRON_C_ALLOC_INTERPOSE __attribute__((malloc, alloc_size(1))) void *
36 malloc(usize size) noexcept      // alloc memory of size 'size', prefer using alloc
37 {
38   return reinterpret_cast<void *>(abc::alloc(size));
39 }
40 
41 extern "C" __MICRON_C_ALLOC_INTERPOSE void *
42 calloc(usize num, usize size) noexcept      // alloc's zero'd out memory, prefer using salloc()
43 {
44   if ( size != 0 && (size * num) / size != num ) return nullptr;
45 
46   byte *mem = abc::alloc(size * num);
47   if ( !mem ) return nullptr;
48   micron::zero(mem, size * num);
49   return mem;
50 }
51 
52 extern "C" __MICRON_C_ALLOC_INTERPOSE void *
53 realloc(void *ptr, usize size) noexcept      // reallocates memory
54 {
55   if ( !ptr ) return reinterpret_cast<void *>(abc::alloc(size));
56   if ( size == 0 ) {
57     abc::free(ptr);
58     return nullptr;
59   }
60 
61   // NOTE: this always gets the full size of the allocated memory, not what was
62   // requested;
63   // for posix_memalign() the block starts before ptr
64   byte *const old = reinterpret_cast<byte *>(ptr);
65   byte *const base = abc::__aligned_base_of(old);
66   const usize block = abc::query_size(micron::ptr_cast<addr_t *>(base != nullptr ? base : old));
67   const usize displacement = base != nullptr ? static_cast<usize>(old - base) : 0u;
68   const usize old_size = block > displacement ? block - displacement : 0u;
69 
70   byte *new_block = abc::alloc(size);
71   if ( !new_block ) return nullptr;      // allocation failed
72 
73   const usize copy_size = old_size < size ? old_size : size;
74   if ( copy_size != 0 ) micron::memcpy(new_block, old, copy_size);
75 
76   if ( base != nullptr )
77     abc::dealloc(base);
78   else
79     abc::dealloc(old);
80 
81   return new_block;
82 }

84 extern "C" __MICRON_C_ALLOC_INTERPOSE void
85 free(void *ptr) noexcept      // frees memory, prefer abc::dealloc always
86 {
87   abc::free(ptr);

91 // POSIX ALIGNED FUNCTIONS
92 //
93 // WARNING: ITS ABSOLUTELY CRITICAL that we declare all aligned_* posix compliant fns;
94 // posix_memalign/memalign/valloc/pvalloc; if any external lib/code calls those fns they
95 // will INTERPOSE LIBCs malloc and handing back those ptrs to us on a regular free()
96 //
97 // NOTE: abc::aligned_balloc is used rather than abc::aligned_alloc
98 
99 extern "C" __MICRON_C_ALLOC_INTERPOSE void *
100 aligned_alloc(usize alignment, usize size) noexcept
101 {
102   return reinterpret_cast<void *>(abc::aligned_balloc(alignment, size).ptr);
103 }
104 
105 extern "C" __MICRON_C_ALLOC_INTERPOSE int
106 posix_memalign(void **out, usize alignment, usize size) noexcept
107 {
108   if ( out == nullptr ) return 22;      // EINVAL
109   // POSIX: alignment is a power of two AND a multiple of sizeof(void *)
110   if ( alignment < sizeof(void *) || (alignment & (alignment - 1)) != 0 ) return 22;
111   if ( size == 0 ) {
112     *out = nullptr;
113     return 0;
114   }
115   void *mem = reinterpret_cast<void *>(abc::aligned_balloc(alignment, size).ptr);
116   if ( mem == nullptr ) return 12;      // ENOMEM
117   *out = mem;
118   return 0;
119 }

127 extern "C" __MICRON_C_ALLOC_INTERPOSE void *
128 valloc(usize size) noexcept
129 {
130   return reinterpret_cast<void *>(abc::aligned_balloc(abc::__system_pagesize, size).ptr);
131 }

133 extern "C" __MICRON_C_ALLOC_INTERPOSE void *
134 pvalloc(usize size) noexcept
135 {
136   const usize rounded = (size + abc::__system_pagesize - 1) & ~(abc::__system_pagesize - 1);
137   return reinterpret_cast<void *>(abc::aligned_balloc(abc::__system_pagesize, rounded).ptr);
138 }
