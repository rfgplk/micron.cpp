33 inline bool
34 __validate_mmap_addr(addr_t *ptr)
35 {
36   return !micron::mmap_failed(ptr);
37 }

39 // base allocator which all others inherit from
40 template<typename T, usize Sz = alloc_auto_sz>
41   requires(micron::is_integral_v<T>)
42 struct sys_allocator {
43   static T *
44   alloc(usize n)      // allocate 'smartly'
45   {
46     // NOTE: from mm/mmap.c - do_mmap
47     // mmap will return either, -ENOMEM, -EOVERFLOW, -EINVAL, -EEXIST, -EPERM, -EAGAIN, -EOVERFLOW; depending on
48     // fail/error condition. not a raw -1 (this is a libc reconciliation).
49     addr_t *ptr = nullptr;
50     if constexpr ( Sz == page_size )
51       ptr = micron::map_normal(nullptr, n);
52     else if constexpr ( Sz == large_page_size )
53       ptr = micron::map_large(nullptr, n);
54     if ( !__validate_mmap_addr(ptr) || ptr == nullptr ) exc<except::critical_error>("sys_allocator::alloc(): mmap() refused by kernel");
55 
56     if ( micron::madvise(ptr, n, micron::madv_dontneed) != 0 ) {
57     }
58 
59     return reinterpret_cast<T *>(ptr);
60   };
61 
62   static void
63   dealloc(T *mem, usize len)
64   {      // deallocate at location N
65     if ( mem == nullptr ) return;
66     if ( micron::munmap(micron::ptr_cast<addr_t *>(mem), len) == -1 )
67       exc<except::critical_error>("sys_allocator::dealloc(): munmap() failed");
68     mem = nullptr;
69   }
70 };
71 };      // namespace micron
