269 // every quantity that decides where a byte is. two builds agreeing here agree on the layout
270 constexpr u64
271 config_fp(void)
272 {
273   u64 h = 14695981039346656037ull;
274   h = fnv(h, sizeof(__arena));
275   h = fnv(h, __max_arenas);
276   h = fnv(h, __va_reservation_size);
277   h = fnv(h, __sheet_align_log2);
278   h = fnv(h, __hdr_offset);
279   h = fnv(h, flags());
280   h = fnv(h, __ins_probe::off_precise);
281   h = fnv(h, __ins_probe::off_huge);
282   h = fnv(h, sizeof(buddy_book));
283   h = fnv(h, sizeof(tlsf_book));
284   h = fnv(h, __arena::__ins_sizeof_range);
285   h = fnv(h, arch_id());
286   h = fnv(h, sizeof(void *));
287   return h;
288 }

290 constexpr u8
291 shift_of(u64 v)
292 {
293   u8 r = 0;
294   while ( v > 1 ) {
295     v >>= 1;
296     r++;
297   }
298   return r;
299 }

301 template<typename T>
302 constexpr abc_tier_desc
303 tier(u64 off, u64 cls, u32 sheets, u32 slots, u8 kind, u8 user)
304 {
305   abc_tier_desc t{};
306   t.off_idx = static_cast<u32>(__arena::__ins_tier_off_idx<T>);
307   t.off_count = static_cast<u32>(__arena::__ins_tier_off_count<T>);
308   t.off_mask = static_cast<u32>(__arena::__ins_tier_off_mask<T>);
309   t.off_cache = static_cast<u32>(__arena::__ins_tier_off_cache<T>);
310   t.cache_off_size = static_cast<u32>(__arena::__ins_cache_off_size<typename T::__ins_cache_t>());
311   t.cache_off_ptr = static_cast<u32>(__arena::__ins_cache_off_ptr<typename T::__ins_cache_t>());
312   t.cache_off_count = static_cast<u32>(__arena::__ins_cache_off_count<typename T::__ins_cache_t>());
313   t.off = off;
314   t.class_size = cls;
315   t.max_sheets = sheets;
316   t.cache_slots = slots;
317   t.kind = kind;
318   t.user = user;
319   t.min_shift = shift_of(cls);
320   return t;
321 }

514 constexpr bool
515 abc_inspect_valid(const abc_inspect_desc *d)
516 {
517   if !consteval {
518     if ( d == nullptr ) return false;
519   }
520   if ( d->magic != abc_inspect_magic ) return false;
521   if ( d->version != abc_inspect_version ) return false;
522   if ( d->abi != static_cast<u16>(MICRON_ABI) ) return false;
523   if ( d->size < __builtin_offsetof(abc_inspect_desc, details) ) return false;
524   if ( d->ptr_width != sizeof(void *) ) return false;
525   if ( d->n_tiers != abc_n_tiers ) return false;
526   return true;
527 }

546 static_assert(__ins::__self.tiers[2].min_shift != __ins::__self.tiers[4].min_shift);

548 // the reader walks tier_base + off_idx as __range[max_sheets] and reads __count immediately after it;
549 // that is the one relation it cannot get wrong. checking it per tier also proves the single published
550 // sizeof_range describes all six, and that no offset truncated into its u32 field
551 //
552 // NOTE: deliberately NOT an inequality between two tiers. config_embed gives every tier 64 sheets, so
553 // tiers[0].off_count == tiers[3].off_count there is correct, not a defect
554 constexpr bool
555 __ins_tiers_consistent(void)
556 {
557   const auto &d = __ins::__self;
558   for ( u32 i = 0; i < abc_n_tiers; ++i ) {
559     if ( d.tiers[i].off_idx != d.tiers[0].off_idx ) return false;
560     if ( d.tiers[i].off_count != static_cast<u64>(d.tiers[i].off_idx) + static_cast<u64>(d.tiers[i].max_sheets) * d.sizeof_range )
561       return false;
562   }
563   return true;
564 }
