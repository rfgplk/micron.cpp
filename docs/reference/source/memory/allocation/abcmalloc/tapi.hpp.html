58 // grow-only, lock-free, single-owner stack used when all __max_arenas primary slots are concurrently allocated
59 struct __arena_node {

61   micron::atomic_token<i32> owner{ 0 };

74 // safety net for a thread that died WITHOUT running the exit hook
75 [[gnu::cold]] static inline bool
76 __owner_alive(i32 tid) noexcept
77 {
78   if ( tid == 0 ) return false;
79   const i32 pid = static_cast<i32>(micron::syscall(SYS_getpid));
80   return micron::syscall(SYS_tgkill, pid, tid, 0) == 0;
81 }

83 // the initial thread is the one whose tid == pid; it only ever "exits" at process teardown
84 [[gnu::cold]] static inline bool
85 __is_initial_thread(void) noexcept
86 {
87   return __this_tid() == static_cast<i32>(micron::syscall(SYS_getpid));

90 // runs on the exiting thread (via thread_kernel) and returns its arena slot to the pool
91 static void
92 __release_tls_arena(void) noexcept
93 {
94   __arena *a = __tls_arena;
95   if ( !a ) return;
96   if ( __is_initial_thread() ) {
97     a->__maybe_drain();
98     return;
99   }
100   const i32 tid = __this_tid();
101   for ( u32 i = 0; i < __max_arenas; ++i ) {
102     if ( __arena_pool[i] == a ) {
103       if ( __arena_owner[i].get(micron::memory_order_acquire) == tid ) {
104         a->__maybe_drain();      // flush pending cross-thread frees while we still own it
105         __arena_owner[i].store(__arena_slot_free, micron::memory_order_release);      // recyclable (ABC-10)
106       }
107       __tls_arena = nullptr;
108       return;
109     }
110   }
111   // overflow arena: same release protocol on the node's own owner field
112   for ( __arena_node *nd = __overflow_head.get(micron::memory_order_acquire); nd != nullptr; nd = nd->next ) {
113     if ( &nd->arena == a ) {
114       if ( nd->owner.get(micron::memory_order_acquire) == tid ) {
115         a->__maybe_drain();
116         nd->owner.store(__arena_slot_free, micron::memory_order_release);
117       }
118       break;
119     }
120   }
121   __tls_arena = nullptr;
122 }

132 // cold init; called only when __tls_arena is nullptr (first hit on this thread)
133 [[gnu::cold, gnu::noinline]] inline __arena *
134 __claim_arena_slow(void) noexcept
135 {
136   micron::__thread_exit_hook = &__release_tls_arena;      // micron::thread exit path
137   (void)&__arena_releaser_tls;                            // force-instantiate the TLS-dtor releaser
138 
139   const i32 tid = __this_tid();
140 
141   {
142     const u32 n = __arena_pool_next.get(micron::memory_order_acquire);
143     const u32 lim = n > __max_arenas ? __max_arenas : n;
144     for ( u32 i = 0; i < lim; ++i ) {
145       i32 expect = __arena_slot_free;
146       if ( __arena_owner[i].compare_exchange_strong(expect, tid, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
147         __arena *a = __arena_pool[i];
148         a->__maybe_drain();
149         __tls_arena = a;
150         return a;
151       }
152     }
153   }
154 
155   // WARNING: reclaim a slot whose owner thread has died without releasing it;
156   // a joined threads slot can lag the next threads reclaim, under fast churn stale slots can accumulate rapidly
157   {
158     const u32 n = __arena_pool_next.get(micron::memory_order_acquire);
159     const u32 lim = n > __max_arenas ? __max_arenas : n;
160     for ( u32 i = 0; i < lim; ++i ) {
161       __arena *a = __arena_pool[i];
162       if ( !a ) continue;
163       i32 owner = __arena_owner[i].get(micron::memory_order_acquire);
164       if ( owner <= 0 || owner == tid ) continue;
165       if ( __owner_alive(owner) ) continue;
166       if ( __arena_owner[i].compare_exchange_strong(owner, tid, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
167         a->__maybe_drain();
168         __tls_arena = a;
169         return a;
170       }
171     }
172   }
173 
174   u32 cur = __arena_pool_next.get(micron::memory_order_acquire);
175   while ( cur < __max_arenas ) {
176     if ( __arena_pool_next.compare_exchange_strong(cur, cur + 1, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
177       __arena *a = new (&__arena_pool_storage[cur * sizeof(__arena)]) __arena();
178       __arena_pool[cur] = a;
179       __arena_owner[cur].store(tid, micron::memory_order_release);
180       __tls_arena = a;
181       return a;
182     }
183   }
184 
185   for ( __arena_node *nd = __overflow_head.get(micron::memory_order_acquire); nd != nullptr; nd = nd->next ) {
186     i32 owner = nd->owner.get(micron::memory_order_acquire);
187     const bool reclaimable = (owner == __arena_slot_free) || (owner > 0 && owner != tid && !__owner_alive(owner));
188     if ( !reclaimable ) continue;
189     i32 expect = owner;      // free or a dead owner; claim it
190     if ( nd->owner.compare_exchange_strong(expect, tid, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
191       nd->arena.__maybe_drain();
192       __tls_arena = &nd->arena;
193       return &nd->arena;
194     }
195   }
196 
197   byte *mem = micron::sys_allocator<byte>::alloc(sizeof(__arena_node));
198   __arena_node *node = new (mem) __arena_node();
199   node->owner.store(tid, micron::memory_order_release);
200   node->next = __overflow_head.get(micron::memory_order_acquire);
201   while ( !__overflow_head.compare_exchange_weak(node->next, node, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
202   }
203   __tls_arena = &node->arena;
204   return &node->arena;
205 }

207 // hot path init; taken when arena already live
208 [[gnu::always_inline]] static inline __arena *
209 __current_arena(void) noexcept
210 {
211   __arena *a = __tls_arena;
212   if ( a ) [[likely]] {
213     a->__maybe_drain();
214     return a;
215   }
216   return __claim_arena_slow();
217 }

219 [[gnu::always_inline]] static inline bool
220 __route_dealloc(byte *p, usize sz) noexcept
221 {
222   __arena *me = __current_arena();
223   if constexpr ( !__default_multithread_safe ) {
224     // single-thread
225     return sz ? me->pop(micron::__chunk<byte>{ p, sz }) : me->pop(p);
226   }
227   __arena *owner = __owner_of(p);
228   if ( !owner || owner == me ) [[likely]] {
229     return sz ? me->pop(micron::__chunk<byte>{ p, sz }) : me->pop(p);
230   }
231   // wait-free: the owner's ring, falling back to the embedded-node overflow LIFO when the ring is full; never spin here
232   ABC_DOCTOR(doctor::record_remote_free(p, sz);)
233   (void)owner->__remote_push(p, sz);
234   return true;
235 }

237 [[gnu::always_inline]] static inline bool
238 __route_retire(byte *p) noexcept
239 {
240   __arena *me = __current_arena();
241   if constexpr ( !__default_multithread_safe ) return me->ts_pop(p);
242   __arena *owner = __owner_of(p);
243   if ( !owner || owner == me ) [[likely]]
244     return me->ts_pop(p);
245   ABC_DOCTOR(doctor::record_remote_tombstone(p);)
246   (void)owner->__remote_push(p, __remote_tombstone_size);
247   return true;
248 }

250 [[gnu::always_inline]] static inline __arena *
251 __query_arena(const void *p) noexcept
252 {
253   if constexpr ( !__default_multithread_safe ) {
254     (void)p;
255     return __current_arena();
256   } else {
257     __arena *owner = __owner_of(p);
258     return owner ? owner : __current_arena();
259   }
260 }

262 template<typename Fn>
263 [[gnu::always_inline]] static inline void
264 __for_each_live_arena(Fn &&fn) noexcept
265 {
266   if constexpr ( !__default_multithread_safe ) {
267     if ( auto *a = __tls_arena; a ) fn(*a);
268     return;
269   }
270   const u32 n = __arena_pool_next.get(micron::memory_order_acquire);
271   const u32 lim = n > __max_arenas ? __max_arenas : n;
272   for ( u32 i = 0; i < lim; ++i ) {
273     if ( auto *a = __arena_pool[i]; a ) fn(*a);
274   }
275   for ( __arena_node *nd = __overflow_head.get(micron::memory_order_acquire); nd != nullptr; nd = nd->next ) fn(nd->arena);
276 }
