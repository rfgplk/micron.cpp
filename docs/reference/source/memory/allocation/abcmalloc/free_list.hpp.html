59   static constexpr int __log2_min = []() constexpr {
60     int r = 0;
61     i64 v = Min;
62     while ( v > 1 ) {

212   [[gnu::cold, gnu::noinline]] bool
213   __freelist_remove_slow(byte *buddy, i32 o) noexcept
214   {
215     free_block *prev = nullptr;
216     free_block *cur = free_lists[o];
217     usize guard = 0;
218     while ( cur && guard++ <= tag_count ) {
219       free_block *nx = cur->next;
220       if ( !__link_valid(nx) ) nx = nullptr;
221       if ( (byte *)cur == buddy ) {
222         if ( prev )
223           prev->next = nx;
224         else
225           free_lists[o] = nx;
226         if ( prev && nx ) nx->prev = prev;
227         mask_clear_if_empty(o);
228         return true;
229       }
230       prev = cur;
231       cur = nx;
232     }
233     return false;
234   }

236   __attribute__((always_inline)) inline bool
237   freelist_remove(byte *buddy, i32 o) noexcept
238   {
239     free_block *node = __free_block_at(buddy);
240     free_block *next = node->next;
241     free_block *head = free_lists[o];
242 
243     if ( head == node ) {
244       if ( !__link_valid(next) ) [[unlikely]]
245         return __freelist_remove_slow(buddy, o);
246       free_lists[o] = next;
247       if ( next == nullptr ) free_mask &= ~(u64(1) << o);
248       return true;
249     }
250 
251     free_block *prev = node->prev;
252     if ( !__link_valid(next) ) [[unlikely]]
253       return __freelist_remove_slow(buddy, o);
254     if ( !__link_valid(prev) || prev == nullptr || prev->next != node || (next && next->prev != node) ) [[unlikely]]
255       return __freelist_remove_slow(buddy, o);
256 
257     prev->next = next;
258     if ( next ) next->prev = prev;
259     return true;
260   }

262   __attribute__((always_inline)) inline void
263   freelist_push_off(byte *addr, i32 o, usize off) noexcept
264   {
265     free_block *nb = __free_block_at(addr);
266     free_block *head = free_lists[o];
267     nb->next = head;
268     if ( head )
269       head->prev = nb;

271       mask_set(o);

273     tag_set_free_at(off >> __log2_min, o);

276   __attribute__((always_inline)) inline free_block *
277   __take_and_split(i32 from, i32 target) noexcept
278   {
279     free_block *blk = free_lists[from];
280     free_block *next = blk->next;
281     u64 mask = free_mask;
282     const u64 old_mask = mask;
283 
284     if ( !__link_valid(next) ) [[unlikely]]
285       next = nullptr;      // hard drop a corrupted tail
286     free_lists[from] = next;
287     if ( next == nullptr ) mask &= ~(u64(1) << from);
288 
289     usize split_size = order_size(from);
290     while ( from > target ) {
291       --from;
292       split_size >>= 1;
293       byte *right = reinterpret_cast<byte *>(blk) + split_size;
294       free_block *node = __free_block_at(right);
295       free_block *head = free_lists[from];
296       node->next = head;
297       if ( head )
298         head->prev = node;
299       else
300         mask |= (u64(1) << from);
301       free_lists[from] = node;
302       block_tags[tag_index(right)] = static_cast<u8>(from | __tag_free);
303     }
304 
305     if ( mask != old_mask ) free_mask = mask;
306     return blk;
307   }

309   __attribute__((always_inline)) inline free_block *
310   tcache_pop(i32 o) noexcept
311   {
312     if ( tcache_count[o] <= 0 ) return nullptr;
313     free_block *blk = tcache[o];
314     free_block *__nx = blk->next;
315     if ( __link_valid(__nx) ) {
316       tcache[o] = __nx;
317       --tcache_count[o];
318     } else {
319       tcache[o] = nullptr;
320       tcache_count[o] = 0;
321     }      // hard drop corrupted cache tail
322     return blk;
323   }

399   void
400   __merge_and_free(byte *addr, i32 o) noexcept
401   {
402     usize off = (usize)(addr - base);
403 
404     while ( o < max_order - 1 ) {
405       usize blk_sz = order_size(o);
406       usize buddy_off = off ^ blk_sz;
407 
408       if ( !tag_is_free_at_off(buddy_off, o) ) break;
409 
410       if ( !freelist_remove(base + buddy_off, o) ) [[unlikely]]
411         break;
412       const usize interior_off = (buddy_off > off) ? buddy_off : off;
413       block_tags[interior_off >> __log2_min] = __tag_none;
414 
415       off = (buddy_off < off) ? buddy_off : off;
416       ++o;
417     }
418 
419     freelist_push_off(base + off, o, off);
420   }

437   void
438   __impl_init_memory(byte *_ptr, usize _len, u8 *ext_tags = nullptr)
439   {
440     uintptr_t ptr = (uintptr_t)_ptr;
441     uintptr_t a = alignof(void *);
442     uintptr_t r = (ptr + (a - 1)) & ~(a - 1);
443     usize adjust = r - ptr;
444     if ( _len <= adjust ) {
445       base = nullptr;
446       total = 0;
447       max_order = 0;
448       tag_count = 0;
449       block_tags = nullptr;
450       tags_external = false;
451       return;
452     }
453 
454     byte *aligned = (byte *)r;
455     usize usable = _len - adjust;
456 
457     if ( ext_tags ) {
458 
459       tags_external = true;
460       block_tags = ext_tags;
461       base = aligned;
462       usize data_usable = (usable / Min) * Min;
463       if ( data_usable < Min ) {
464         base = nullptr;
465         total = 0;
466         max_order = 0;
467         tag_count = 0;
468         block_tags = nullptr;
469         tags_external = false;
470         return;
471       }
472       total = data_usable;
473     } else {
474 
475       tags_external = false;
476       constexpr usize min_sz = static_cast<usize>(Min);
477       usize approx_tags = (usable + min_sz) / (min_sz + 1);
478       usize tag_area = (approx_tags + min_sz - 1) & ~(min_sz - 1);
479       if ( tag_area >= usable ) {
480         base = nullptr;
481         total = 0;
482         max_order = 0;
483         tag_count = 0;
484         block_tags = nullptr;
485         return;
486       }
487       block_tags = aligned;
488       base = aligned + tag_area;
489       usize data_usable = usable - tag_area;
490       data_usable = (data_usable / Min) * Min;
491       if ( data_usable < Min ) {
492         base = nullptr;
493         total = 0;
494         max_order = 0;
495         tag_count = 0;
496         block_tags = nullptr;
497         return;
498       }
499       total = data_usable;
500     }
501 
502     tag_count = total >> __log2_min;
503 
504     const usize units = total >> __log2_min;
505     max_order = static_cast<i32>(63 - __builtin_clzll(units)) + 1;
506     if ( max_order > Mx ) max_order = Mx;
507 
508     __impl_zero_arrays();
509 
510     micron::memset(block_tags, __tag_none, tag_count);
511 
512     free_lists[max_order - 1] = __free_block_at(base);
513     free_lists[max_order - 1]->next = nullptr;
514     mask_set(max_order - 1);
515     tag_set_free(base, max_order - 1);
516   }

626   T
627   allocate(usize n) noexcept
628   {
629     if ( n > micron::numeric_limits<usize>::max() - __hdr_offset ) return { nullptr, 0 };
630     n += __hdr_offset;
631     if ( !base ) return { nullptr, 0 };
632     i32 o = order_for_size(n);
633     if ( o >= max_order ) return { nullptr, 0 };
634     const usize target_size = order_size(o);
635 
636     if constexpr ( __cold_cap > 0 ) {
637       if ( free_block *cold = cold_pop(o) ) {
638         block_header *hdr = hdr_of((byte *)cold, o);
639         hdr->order = static_cast<i32>(o);
640         hdr->flags = __block_alloc;
641         tag_set_alloc((byte *)cold, o);
642         allocated_bytes += target_size;
643         return { (byte *)cold, target_size - __hdr_offset };
644       }
645     }
646 
647     i32 i = find_free_order(o);
648     if ( i >= max_order ) {
649       if constexpr ( __cold_cap > 0 ) {
650         cold_flush_all();
651         i = find_free_order(o);
652       }
653       if ( i >= max_order ) return { nullptr, 0 };
654     }
655 
656     free_block *blk = __take_and_split(i, o);
657 
658     // write header at the tail of the block
659     block_header *hdr = hdr_of((byte *)blk, o);
660     hdr->order = static_cast<i32>(o);
661     hdr->flags = __block_alloc;
662     tag_set_alloc((byte *)blk, o);
663     allocated_bytes += target_size;
664 
665     return { (byte *)blk, target_size - __hdr_offset };
666   }

668   T
669   temporal_allocate(usize n) noexcept
670   {
671     if ( n > micron::numeric_limits<usize>::max() - __hdr_offset ) return { nullptr, 0 };
672     n += __hdr_offset;
673     if ( !base ) return { nullptr, 0 };
674 
675     i32 o = order_for_size(n);
676     if ( o >= max_order ) return { nullptr, 0 };
677 
678     usize target_size = order_size(o);
679 
680     u8 r0 = active_rotor[o] & (__active_ring - 1);
681     free_block *active_block = active[o][r0];
682     if ( !active_block ) {
683       r0 ^= 1u;
684       active_block = active[o][r0];
685     }
686     if ( active_block ) {
687       active_rotor[o] = r0 ^ 1u;
688       return { reinterpret_cast<byte *>(active_block), target_size - __hdr_offset };
689     }
690 
691     free_block *cached = tcache_pop(o);
692     if ( cached ) {
693       block_header *hdr = hdr_of((byte *)cached, o);
694       hdr->order = static_cast<i32>(o);
695       hdr->flags = __block_alloc | __block_temporal;
696       tag_set_alloc((byte *)cached, o);
697       allocated_bytes += target_size;
698       active[o][r0] = cached;
699       active_rotor[o] = r0 ^ 1u;
700       return { (byte *)cached, target_size - __hdr_offset };
701     }
702 
703     i32 i = find_free_order(o);
704     if ( i >= max_order ) return { nullptr, 0 };
705 
706     free_block *blk = __take_and_split(i, o);
707 
708     block_header *hdr = hdr_of((byte *)blk, o);
709     hdr->order = static_cast<i32>(o);
710     hdr->flags = __block_alloc | __block_temporal;
711     tag_set_alloc((byte *)blk, o);
712     allocated_bytes += target_size;
713 
714     active[o][r0] = blk;
715     active_rotor[o] = r0 ^ 1u;
716 
717     return { (byte *)blk, target_size - __hdr_offset };
718   }

720   T
721   allocate_exact(usize n) noexcept
722   {
723     if ( !base ) return { nullptr, 0 };
724     if ( n < Min ) return { nullptr, 0 };
725     if ( (n & (n - 1)) != 0 ) return { nullptr, 0 };
726     i32 o = order_for_size(n);
727     if ( o >= max_order ) return { nullptr, 0 };
728     if ( !free_lists[o] ) return { nullptr, 0 };
729 
730     free_block *blk = __take_and_split(o, o);
731     const usize target_size = order_size(o);
732 
733     block_header *hdr = hdr_of((byte *)blk, o);
734     hdr->order = static_cast<i32>(o);
735     hdr->flags = __block_alloc;
736     tag_set_alloc((byte *)blk, o);
737     allocated_bytes += target_size;
738 
739     return { (byte *)blk, target_size - __hdr_offset };
740   }

742   ret_flag
743   tombstone(byte *ptr) noexcept
744   {
745     if ( !is_allocated(ptr) ) return { __flag_invalid };
746     block_header *hdr = hdr_of_tagged(ptr);
747     i32 o = hdr->order;
748     if ( o < 0 || o >= max_order ) return { __flag_invalid };
749     if ( !(hdr->flags & __block_alloc) ) return { __flag_invalid };
750 
751     hdr->flags = __block_tombstone;
752     const usize target_size = order_size(o);
753     allocated_bytes -= target_size;
754     tombstoned_bytes += target_size;
755 
756     return __flag_tombstoned;
757   }

780   ret_flag
781   deallocate(byte *ptr) noexcept
782   {
783     // WARNING: validate a min-block-aligned, authoritatively-allocated block START before any header access
784     // protects against forged/out-of-range ptrs
785     if ( !is_allocated(ptr) ) return { __flag_invalid };
786     byte *addr = ptr;
787     block_header *hdr = hdr_of_tagged(addr);
788     const i32 original_o = hdr->order;
789     if ( original_o < 0 || original_o >= max_order ) return { __flag_invalid };
790     if ( !base || !addr ) return __flag_failure;
791     const i32 flags = hdr->flags;
792     if ( flags == __block_free ) return { __flag_invalid };
793 
794     const usize target_size = order_size(original_o);
795     allocated_bytes -= target_size;
796     if ( flags & __block_tombstone ) tombstoned_bytes -= target_size;
797 
798     bool was_temporal = (flags & __block_temporal) != 0;
799     if ( was_temporal ) {
800       free_block *free_addr = __free_block_at(addr);
801       for ( i32 r = 0; r < __active_ring; ++r ) {
802         if ( active[original_o][r] == free_addr ) {
803           active[original_o][r] = nullptr;
804           break;
805         }
806       }
807       if ( tcache_push(free_addr, original_o) ) {
808         block_tags[tag_index(addr)] = __tag_none;
809         return { __flag_ok };
810       }
811     }
812 
813     if constexpr ( __cold_cap > 0 ) {
814       if ( !was_temporal && cold_push(__free_block_at(addr), original_o) ) {
815         block_tags[tag_index(addr)] = __tag_none;
816         return { __flag_ok };
817       }
818     }
819 
820     __merge_and_free(addr, original_o);
821     return { __flag_ok };
822   }

831   T
832   reallocate(T node, usize new_size) noexcept
833   {
834     if ( !base ) return { nullptr, 0 };
835     if ( !node.ptr ) return allocate(new_size);
836     if ( new_size == 0 ) {
837       deallocate(node);
838       return { nullptr, 0 };
839     }
840 
841     if ( node.len >= new_size && new_size > (node.len >> 1) ) return node;
842 
843     T nnode = allocate(new_size);
844     if ( !nnode.ptr ) return { nullptr, 0 };
845 
846     usize to_copy = (node.len < nnode.len) ? node.len : nnode.len;
847     micron::memcpy(nnode.ptr, node.ptr, to_copy);
848     deallocate(node);
849     return nnode;
850   }

877   usize
878   block_size(byte *ptr) const noexcept
879   {
880     if ( !base || !ptr ) return 0;
881     if ( ptr < base || ptr >= base + total ) return 0;
882     if ( ((usize)(ptr - base) & (Min - 1)) != 0 ) return 0;
883     u8 tag = block_tags[tag_index(ptr)];
884     i32 o = static_cast<i32>(tag & ~__tag_free);
885     if ( (u64)o >= (u64)max_order ) return 0;
886     return order_size(o);
887   }

893     if ( ((usize)(ptr - base) & (Min - 1)) != 0 ) return false;
894     u8 tag = block_tags[tag_index(ptr)];
895     return (tag & __tag_free) == 0 && tag < (u8)max_order;
896   }

905 #if defined(ABCMALLOC_DOCTOR_HELP)
906   // deep corruption walk
907   template<class V>
908   void
909   __doctor_walk(V &v)
910   {
911     if ( !base || !block_tags ) return;
912 
913     usize idx = 0;
914     usize guard = 0;
915     const usize maxg = tag_count + 4;
916     while ( idx < tag_count && guard++ < maxg ) {
917       u8 tag = block_tags[idx];
918       if ( tag == __tag_none ) {      // interior/padding at a boundary; skip one min-block
919         ++idx;
920         continue;
921       }
922       i32 o = static_cast<i32>(tag & ~__tag_free);
923       byte *blk = base + (idx << __log2_min);
924       ++v.blocks;
925       if ( o < 0 || o >= max_order ) {
926         v.note("buddy: block tag order out of range", blk);
927         ++idx;
928         continue;
929       }
930       usize osz = order_size(o);
931       if ( (usize)(blk - base) + osz > total ) {
932         v.note("buddy: block extends past sheet bounds", blk);
933         break;
934       }
935       if ( ((usize)(blk - base) & (osz - 1)) != 0 ) v.note("buddy: block not aligned to its order size", blk);
936 
937       const bool tag_free = (tag & __tag_free) != 0;
938       // the tail block_header is a maintained redundant copy
939       // only for allocated blocks
940       if ( !tag_free ) {
941         block_header *h = hdr_of(blk, o);
942         if ( h->order != static_cast<i32>(o) ) {
943           v.note("buddy: allocated block tail header order != block_tags order", blk);
944           if ( v.repair ) {
945             h->order = static_cast<i32>(o);
946             v.did_repair("buddy: rewrote tail header order from tag", blk);
947           }
948         }
949         const bool flags_alloc
950             = (h->flags == __block_alloc || h->flags == (__block_alloc | __block_temporal) || h->flags == __block_tombstone);
951         if ( !flags_alloc ) {
952           v.note("buddy: allocated block tail header flags not allocated", blk);
953           if ( v.repair ) {
954             h->flags = __block_alloc | (h->flags & __block_temporal);
955             v.did_repair("buddy: reset tail header flags to allocated form", blk);
956           }
957         }
958       }
959       idx += (osz >> __log2_min);
960     }
961     if ( guard >= maxg ) v.note("buddy: tiling walk overran (corrupt tag stream)", base);
962 
963     // per order free lists
964     for ( i32 o = 0; o < max_order; ++o ) {
965       const bool has = (free_lists[o] != nullptr);
966       const bool bit = ((free_mask >> o) & 1u) != 0;
967       if ( has != bit ) {
968         v.note("buddy: free_mask bit disagrees with free_lists[o]", base);
969         if ( v.repair ) {
970           if ( has )
971             free_mask |= (u64(1) << o);
972           else
973             free_mask &= ~(u64(1) << o);
974           v.did_repair("buddy: fixed free_mask bit", base);
975         }
976       }
977       free_block *n = free_lists[o];
978       free_block *prev = nullptr;
979       usize gc = 0;
980       while ( n && gc++ < tag_count + 4 ) {
981         ++v.freelist_nodes;
982         if ( !__link_valid(n) ) {
983           v.note("buddy: free-list node out of bounds / misaligned", n);
984           if ( v.repair ) {
985             if ( prev )
986               prev->next = nullptr;
987             else
988               free_lists[o] = nullptr;
989             mask_clear_if_empty(o);
990             v.did_repair("buddy: truncated free-list at bad node", n);
991           }
992           break;
993         }
994         if ( !tag_is_free_at(reinterpret_cast<byte *>(n), o) ) v.note("buddy: free-list node not tagged free at its order", n);
995         if ( prev && n->prev != prev ) {
996           v.note("buddy: free-list prev back-link mismatch", n);
997           if ( v.repair ) {
998             n->prev = prev;
999             v.did_repair("buddy: fixed free-list prev back-link", n);
1000           }
1001         }
1002         if ( n->next && !__link_valid(n->next) ) {
1003           v.note("buddy: forged free-list next link", n);
1004           if ( v.repair ) {
1005             n->next = nullptr;
1006             v.did_repair("buddy: nulled forged free-list link", n);
1007           }
1008           break;
1009         }
1010         prev = n;
1011         n = n->next;
1012       }
1013       if ( gc >= tag_count + 4 ) v.note("buddy: free-list cycle / overrun at order", base);
1014     }
1015   }
