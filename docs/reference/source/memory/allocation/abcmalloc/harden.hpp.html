25 __micron_diagnostic_push __micron_diagnostic_suggest_noreturn inline bool
26 fail_state(void)
27 {
28   if constexpr ( __default_fail_result == 0 ) {
29     // hard abort, no diagnostics
30     abort_state();
31   } else if constexpr ( __default_fail_result == 1 ) {
32     // diagnostic print then abort
33     __debug_print("fail_state(): allocator integrity violation (mode=1), aborting", 0);
34     abort_state();
35   } else if constexpr ( __default_fail_result == 2 ) {
36     // silent failure, return false
37     __debug_print("fail_state(): allocator integrity violation (mode=2), returning false", 0);
38     return false;
39   }
40   return true;
41 }

43 __micron_diagnostic_pop __micron_diagnostic_push __micron_diagnostic_suggest_noreturn inline __attribute__((always_inline)) bool
44 handle_double_free([[maybe_unused]] byte *addr)
45 {
46   ABC_DOCTOR(if ( doctor::on_double_free(addr, __FILE__, __LINE__) ) return true;)
47   if constexpr ( __default_double_free_action == 0 ) {
48     // ignore silently
49     return false;
50   } else if constexpr ( __default_double_free_action == 1 ) {
51     __debug_print_addr("handle_double_free(): double free / unrecognised block at: ", addr);
52     return false;
53   } else if constexpr ( __default_double_free_action == 2 ) {
54     __debug_print_addr("handle_double_free(): double free / unrecognised block, aborting at: ", addr);
55     abort_state();
56   }
57   return false;
58 }

66 __micron_diagnostic_pop inline __attribute__((always_inline)) auto
67 check_constraint(const usize sz) -> bool
68 {
69   if constexpr ( __is_constrained ) {
70     if ( sz > __alloc_limit and __alloc_limit != 0 ) {
71       __debug_print("check_constraint(): allocation exceeds limit, requested: ", sz);
72       __debug_print("check_constraint(): configured limit: ", __alloc_limit);
73       return true;
74     }
75   }
76   // catch obvious corruption
77   if ( sz > (micron::numeric_limits<usize>::max() >> 1) ) [[unlikely]] {
78     __debug_print("check_constraint(): allocation size exceeds half address space, likely overflow: ", sz);
79     return true;
80   }
81   return false;
82 }

187 // start canary: __default_redzone_size bytes starting at user_ptr - __default_redzone_size
188 // end canary: __default_redzone_size bytes starting at user_ptr + user_sz
189 inline __attribute__((always_inline)) void
190 write_redzone([[maybe_unused]] byte *user_ptr, [[maybe_unused]] usize user_sz)
191 {
192   if constexpr ( __default_redzone ) {
193     byte *lo = user_ptr - __default_redzone_size;
194     micron::memset(lo, __default_redzone_byte, __default_redzone_size);
195     byte *hi = user_ptr + user_sz;
196     micron::memset(hi, __default_redzone_byte, __default_redzone_size);
197   } else {
198     // nothing
199   }
200 }

202 inline __attribute__((always_inline)) bool
203 verify_redzone_leading([[maybe_unused]] const byte *user_ptr)
204 {
205   if constexpr ( __default_redzone ) {
206     const byte *lo = user_ptr - __default_redzone_size;
207     for ( usize i = 0; i < __default_redzone_size; ++i ) {
208       if ( lo[i] != __default_redzone_byte ) {
209         __debug_print("verify_redzone_leading(): leading canary corrupted at byte: ", i);
210         return false;
211       }
212     }
213     return true;
214   }
215   // nothing
216   return true;
217 }

219 inline __attribute__((always_inline)) bool
220 verify_redzone([[maybe_unused]] const byte *user_ptr, [[maybe_unused]] usize user_sz)
221 {
222   if constexpr ( __default_redzone ) {
223     if ( !verify_redzone_leading(user_ptr) ) return false;
224     const byte *hi = user_ptr + user_sz;
225     for ( usize i = 0; i < __default_redzone_size; ++i ) {
226       if ( hi[i] != __default_redzone_byte ) {
227         __debug_print("verify_redzone(): trailing canary corrupted at byte: ", i);
228         return false;
229       }
230     }
231     return true;
232   }
233   return true;
234 }

254 inline __attribute__((always_inline)) bool
255 check_chunk_valid(const byte *ptr, usize len)
256 {
257   if ( !check_ptr_valid(ptr) ) return false;
258   if ( len == 0 ) [[unlikely]] {
259     __debug_print("check_chunk_valid(): zero-length chunk", 0);
260     return false;
261   }
262   if ( reinterpret_cast<uintptr_t>(ptr) + len < reinterpret_cast<uintptr_t>(ptr) ) [[unlikely]] {
263     __debug_print("check_chunk_valid(): ptr + len wraps address space", 0);
264     return false;
265   }
266   return true;
267 }
