35 // carve granule
36 //
37 // width-32 uses a 64 KiB granule, NOT 2 MiB
38 #if defined(__micron_arch_width_64)
39 constexpr static const usize __sheet_align_log2 = 21;      // 2 MiB
40 #else
41 constexpr static const usize __sheet_align_log2 = 16;      // 64 KiB == the minimum sheet size
42 #endif
43 constexpr static const usize __sheet_align = 1ULL << __sheet_align_log2;
44 constexpr static const usize __sheet_align_mask = __sheet_align - 1;
45 
46 #ifndef MICRON_ABC_VA_RESERVE_SIZE
47 #if defined(__micron_arch_width_64)
48 // 256 GiB
49 #define MICRON_ABC_VA_RESERVE_SIZE (256ULL << 30)
50 #else
51 // 1 GiB. 256 MiB was not enough to hold the coroutine engine
52 #define MICRON_ABC_VA_RESERVE_SIZE (1024U << 20)
53 #endif
54 #endif
55 constexpr static const usize __va_reservation_size = MICRON_ABC_VA_RESERVE_SIZE;
56 static_assert(__va_reservation_size >= __sheet_align, "abcmalloc: MICRON_ABC_VA_RESERVE_SIZE must be at least one sheet granule.");
57 static_assert((__va_reservation_size & __sheet_align_mask) == 0,
58               "abcmalloc: MICRON_ABC_VA_RESERVE_SIZE must be a whole multiple of the sheet granule.");

62 inline micron::atomic_token<addr_t *> __va_base{ nullptr };      // PROT_NONE base, or nullptr if not yet reserved
63 inline micron::atomic_token<u64> __va_offset{ 0 };               // bump cursor in bytes
64 inline micron::atomic_flag __va_init_lock{};                     // one-shot init guard

66 struct __va_free_run {
67   u64 off;           // byte offset from __va_base
68   u32 granules;      // run length in __sheet_align granules
69 };

71 constexpr static const usize __va_free_granules = __va_reservation_size >> __sheet_align_log2;
72 #if defined(__micron_arch_width_64)
73 constexpr static const usize __va_free_cap = __va_free_granules;
74 #else
75 constexpr static const usize __va_free_cap = __va_free_granules > 1024 ? usize{ 1024 } : __va_free_granules;
76 #endif
77 inline __va_free_run __va_free_runs[__va_free_cap]{};
78 inline usize __va_free_count{ 0 };
79 inline micron::atomic_flag __va_free_lock{};

81 [[gnu::cold, gnu::noinline]] inline addr_t *
82 __va_reserve_once() noexcept
83 {
84   addr_t *base = __va_base.get(micron::memory_order_acquire);
85   if ( base ) return base;
86 
87   // serialize the actual mmap; double-checked under the lock so contenders just observe the publish
88   micron::free_guard<> guard{ &__va_init_lock };
89   base = __va_base.get(micron::memory_order_relaxed);
90   if ( base ) return base;
91 
92   base = micron::mmap(nullptr, __va_reservation_size, micron::prot_none, micron::map_private | micron::map_anonymous | __map_noreserve_flag,
93                       -1, 0);
94   if ( micron::mmap_failed(base) || !base ) {
95     return nullptr;
96   }
97   __va_base.store(base, micron::memory_order_release);
98   return base;
99 }

101 // commit a carved run: replace its PROT_NONE backing with PROT_READ|WRITE anonymous pages
102 [[gnu::always_inline]] inline addr_t *
103 __va_commit(addr_t *slot, usize rounded) noexcept
104 {
105   addr_t *got = micron::mmap(slot, rounded, micron::prot_read | micron::prot_write,
106                              micron::map_private | micron::map_anonymous | micron::map_fixed, -1, 0);
107   if ( micron::mmap_failed(got) || got != slot ) [[unlikely]]
108     return nullptr;
109   return slot;
110 }

112 constexpr static const u64 __va_bump_fail = ~static_cast<u64>(0);
113 
114 [[gnu::always_inline]] inline u64
115 __va_bump(usize rounded) noexcept
116 {
117   u64 off = __va_offset.get(micron::memory_order_acquire);
118   for ( ;; ) {
119     if ( off + rounded > __va_reservation_size ) [[unlikely]]
120       return __va_bump_fail;
121     if ( __va_offset.compare_exchange_weak(off, off + rounded, micron::memory_order_acq_rel, micron::memory_order_acquire) ) return off;
122   }
123 }

125 inline u64
126 __va_reuse(u32 want) noexcept
127 {
128   constexpr usize __none = ~static_cast<usize>(0);
129   micron::free_guard<> guard{ &__va_free_lock };
130   usize best = __none;
131   for ( usize i = __va_free_count; i-- > 0; ) {
132     const u32 g = __va_free_runs[i].granules;
133     if ( g < want ) continue;
134     if ( best == __none || g < __va_free_runs[best].granules ) best = i;
135     if ( g == want ) break;      // exact fit, stop looking
136   }
137   if ( best == __none ) return __va_bump_fail;
138   const u64 off = __va_free_runs[best].off;
139   const u32 g = __va_free_runs[best].granules;
140   if ( g > want ) {      // hand back the head, keep the shrunk tail in place
141     __va_free_runs[best].off = off + (static_cast<u64>(want) << __sheet_align_log2);
142     __va_free_runs[best].granules = g - want;
143   } else {
144     __va_free_runs[best] = __va_free_runs[--__va_free_count];      // swap-remove
145   }
146   return off;
147 }

149 inline addr_t *
150 __va_carve(usize bytes) noexcept
151 {
152   addr_t *base = __va_base.get(micron::memory_order_acquire);
153   if ( !base ) [[unlikely]] {
154     base = __va_reserve_once();
155     if ( !base ) return nullptr;
156   }
157 
158   const usize rounded = (bytes + __sheet_align_mask) & ~__sheet_align_mask;
159   const u32 want = static_cast<u32>(rounded >> __sheet_align_log2);
160 
161   const u64 reuse_off = __va_reuse(want);
162   if ( reuse_off != __va_bump_fail ) {
163     addr_t *slot = reinterpret_cast<addr_t *>(reinterpret_cast<uintptr_t>(base) + reuse_off);
164     if ( addr_t *got = __va_commit(slot, rounded); got ) [[likely]]
165       return got;
166     // remap failed: the run is now dropped from the list (effectively leaked); fall through to a fresh carve
167   }
168 
169   const u64 off = __va_bump(rounded);
170   if ( off == __va_bump_fail ) [[unlikely]]
171     return nullptr;      // reservation exhausted
172 
173   return __va_commit(reinterpret_cast<addr_t *>(reinterpret_cast<uintptr_t>(base) + off), rounded);
174 }

176 inline addr_t *
177 __va_carve_reserved(usize bytes) noexcept
178 {
179   addr_t *base = __va_base.get(micron::memory_order_acquire);
180   if ( !base ) [[unlikely]] {
181     base = __va_reserve_once();
182     if ( !base ) return nullptr;
183   }
184 
185   const usize rounded = (bytes + __sheet_align_mask) & ~__sheet_align_mask;
186   const u32 want = static_cast<u32>(rounded >> __sheet_align_log2);
187 
188   const u64 reuse_off = __va_reuse(want);
189   if ( reuse_off != __va_bump_fail ) return reinterpret_cast<addr_t *>(reinterpret_cast<uintptr_t>(base) + reuse_off);
190 
191   const u64 off = __va_bump(rounded);
192   if ( off == __va_bump_fail ) [[unlikely]]
193     return nullptr;      // reservation exhausted
194   return reinterpret_cast<addr_t *>(reinterpret_cast<uintptr_t>(base) + off);
195 }

197 inline void
198 __va_release(addr_t *slot, usize bytes) noexcept
199 {
200   if ( !slot ) return;
201   addr_t *base = __va_base.get(micron::memory_order_relaxed);
202   if ( !base ) return;
203   const uintptr_t si = reinterpret_cast<uintptr_t>(slot);
204   const uintptr_t bi = reinterpret_cast<uintptr_t>(base);
205   if ( si < bi || si >= bi + __va_reservation_size ) [[unlikely]]
206     return;      // not a carved VA slot; nothing to reclaim
207   const usize rounded = (bytes + __sheet_align_mask) & ~__sheet_align_mask;
208   // replace the populated region with PROT_NONE again to release physical pages (keeps the VA reserved)
209   (void)micron::mmap(slot, rounded, micron::prot_none,
210                      micron::map_private | micron::map_anonymous | micron::map_fixed | __map_noreserve_flag, -1, 0);
211   const u64 off = static_cast<u64>(si - bi);
212   const u32 granules = static_cast<u32>(rounded >> __sheet_align_log2);
213   micron::free_guard<> guard{ &__va_free_lock };
214   if ( __va_free_count < __va_free_cap ) __va_free_runs[__va_free_count++] = __va_free_run{ off, granules };
215 }

217 [[gnu::always_inline]] inline bool
218 __va_contains(const void *p) noexcept
219 {
220   addr_t *base = __va_base.get(micron::memory_order_relaxed);
221   if ( !base ) return false;
222   const uintptr_t pi = reinterpret_cast<uintptr_t>(p);
223   const uintptr_t bi = reinterpret_cast<uintptr_t>(base);
224   return pi >= bi && pi < bi + __va_reservation_size;
225 }
