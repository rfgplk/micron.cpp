54 // precomputed weights
55 // weight(shift) = round(2^((63 - shift) * alpha)),  alpha = 0.33
56 constexpr static f64 __prealloc_alpha = 0.33;

58 constexpr static u64
59 __compute_weight(u64 shift)
60 {
61   // powerf is constexpr
62   return static_cast<u64>(micron::math::powerf(2.0, (63.0 - (f64)shift) * __prealloc_alpha) + 0.5);
63 }

65 constexpr static u64 __class_weight_precise = __compute_weight(__class_precise_shift);
66 constexpr static u64 __class_weight_small = __compute_weight(__class_small_shift);
67 constexpr static u64 __class_weight_medium = __compute_weight(__class_medium_shift);
68 constexpr static u64 __class_weight_large = __compute_weight(__class_large_shift);

70 constexpr static u64 __class_weight_denom
71     = __class_weight_precise + __class_weight_small + __class_weight_medium + __class_weight_large + __class_weight_huge;

81 template<u64 Class>
82 static inline u64
83 __prealloc_share(u64 total)
84 {
85   if ( total == 0 ) [[unlikely]]
86     return __default_magic_size;
87 
88   constexpr u64 w = __weight_for_class<Class>();
89 
90   if constexpr ( __wordsize == 64 ) {
91     u64 frac = static_cast<u64>((static_cast<u128>(total) * w) / __class_weight_denom);
92     if ( frac % __system_pagesize != 0 ) frac += __system_pagesize - (frac % __system_pagesize);
93     return frac;
94   } else {
95     u64 scaled = total / __class_weight_denom;
96     u64 rem = total % __class_weight_denom;
97     u64 frac = scaled * w + (rem * w) / __class_weight_denom;
98     if ( frac % __system_pagesize != 0 ) frac += __system_pagesize - (frac % __system_pagesize);
99     return frac;
100   }
101 }

211     inline __attribute__((always_inline)) void
212     __mask_insert(u32 pos, bool value) noexcept
213     {
214       const u32 word = pos >> 6;
215       const u32 bit = pos & 63;
216       const u64 w = __space_mask[word];
217       const u64 carry = (w >> 63) & 1ULL;      // bit 63 carries into next-higher word
218 
219       const u64 lo_mask = (bit == 0) ? 0ULL : ((1ULL << bit) - 1);
220       const u64 lo = w & lo_mask;
221       const u64 hi_keep_mask = (bit == 63) ? 0ULL : ~((1ULL << (bit + 1)) - 1);      // covers [bit+1..63]
222       const u64 hi = (w << 1) & hi_keep_mask;
223       __space_mask[word] = lo | hi | ((value ? 1ULL : 0ULL) << bit);
224 
225       u64 c = carry;
226       for ( u32 i = word + 1; i < __detail_words; ++i ) {
227         const u64 nc = (__space_mask[i] >> 63) & 1ULL;
228         __space_mask[i] = (__space_mask[i] << 1) | c;
229         c = nc;
230       }
231     }

233     inline __attribute__((always_inline)) void
234     __mask_remove(u32 pos) noexcept
235     {
236       const u32 word = pos >> 6;
237       const u32 bit = pos & 63;
238 
239       const u64 carry_in = (word + 1 < __detail_words) ? (__space_mask[word + 1] & 1ULL) : 0ULL;
240       const u64 w = __space_mask[word];
241 
242       const u64 lo_mask = (bit == 0) ? 0ULL : ((1ULL << bit) - 1);
243       const u64 lo = w & lo_mask;
244       const u64 hi_keep_mask = (bit == 0) ? ((1ULL << 63) - 1) : (((1ULL << 63) - 1) & ~((1ULL << bit) - 1));
245       const u64 hi = (w >> 1) & hi_keep_mask;
246       __space_mask[word] = lo | hi | (carry_in << 63);
247 
248       for ( u32 i = word + 1; i < __detail_words; ++i ) {
249         const u64 next_carry = (i + 1 < __detail_words) ? (__space_mask[i + 1] & 1ULL) : 0ULL;
250         __space_mask[i] = (__space_mask[i] >> 1) | (next_carry << 63);
251       }
252     }

284     u32
285     register_sheet(node<sheet_type> *nd)
286     {
287       if ( __count >= __max_sheets ) [[unlikely]]
288         return __max_sheets;      // sentinel
289 
290       addr_t *lo = nd->nd->addr();
291       addr_t *hi = nd->nd->addr_end();
292 
293       u32 pos = 0;
294       while ( pos < __count and __idx[pos].lo < lo ) ++pos;
295 
296       // shift entries right
297       for ( u32 i = __count; i > pos; --i ) __idx[i] = __idx[i - 1];
298 
299       __idx[pos] = { lo, hi, nd };
300 
301       // insert availability bit at pos, shifting bits >= pos up by one
302       __mask_insert(pos, true);
303 
304       if ( __last_hit != __no_hit and pos <= __last_hit and __last_hit + 1 < __max_sheets ) ++__last_hit;
305 
306       ++__count;
307       return pos;
308     }

310     void
311     unregister(u32 pos)
312     {
313       if ( pos >= __count ) return;
314 
315       // shift entries left
316       for ( u32 i = pos; i + 1 < __count; ++i ) __idx[i] = __idx[i + 1];
317 
318       // remove bit at pos, shift upper bits down
319       __mask_remove(pos);
320 
321       if ( __last_hit != __no_hit ) {
322         if ( pos == __last_hit )
323           __last_hit = __no_hit;
324         else if ( pos < __last_hit )
325           --__last_hit;
326       }
327 
328       --__count;
329     }

331     i32
332     find_range(addr_t *addr) const
333     {
334       i32 lo = 0, hi = static_cast<i32>(__count) - 1;
335       while ( lo <= hi ) {
336         i32 mid = lo + ((hi - lo) >> 1);
337         if ( addr < __idx[mid].lo )
338           hi = mid - 1;
339         else if ( addr >= __idx[mid].hi )
340           lo = mid + 1;
341         else
342           return mid;
343       }
344       return -1;
345     }

503   __expand_arena_tier(usize sz)
504   {
505     // WARNING: no __struct_guard(), will deadlock otherwise under arena metadata exhaustion
506     __debug_print("__expand_arena_tier(): requested expansion size: ", sz);
507     using Nd = node<sheet<__class_arena_internal>>;
508     using Sh = sheet<__class_arena_internal>;
509     usize pair_sz = sizeof(Nd) + sizeof(Sh);
510     micron::__chunk<byte> buf = _arena_memory.try_mark(pair_sz);
511     if ( buf.failed_allocation() ) [[unlikely]] {
512       __debug_print("__expand_arena_tier()!!!: arena metadata exhausted, req: ", sz);
513       __debug_print("__expand_arena_tier(): arena metadata buf remaining: ", _arena_memory.available());
514       abort_state();
515     }
516     byte *p = buf.ptr;
517     auto *nd = new (p) Nd();
518     p += sizeof(Nd);
519     if constexpr ( __default_guard_arena_metadata ) {
520       __debug_print("__expand_arena_tier(): inserting guard page for arena tier expansion", 0);
521       auto chnk = __get_guarded_kernel_chunk(sz);
522       nd->nd = new (p) Sh(this, chnk, __system_pagesize);
523     } else {
524       auto chnk = __get_kernel_chunk<micron::__chunk<byte>>(sz);
525       if ( !__kernel_chunk_valid(chnk) ) [[unlikely]] {
526         __debug_print("__expand_arena_tier()!!!: mmap failed for arena tier expansion, req: ", sz);
527         abort_state();
528       }
529       nd->nd = new (p) Sh(this, chnk);
530     }
531     if ( nd->nd->empty() ) [[unlikely]] {
532       __debug_print("__expand_arena_tier()!!!: sheet construction failed, kernel chunk invalid", 0);
533       abort_state();
534     }
535     nd->nxt = nullptr;
536     _arena_tier.link_at_tail(nd);
537     if ( _arena_tier.register_sheet(nd) == _arena_tier.__max_sheets ) [[unlikely]] {
538       __debug_print("__expand_arena_tier()!!!: arena tier index full", 0);
539       abort_state();

541     __debug_print("__expand_arena_tier(): new arena node allocated, size: ", sz);
542   }

569   template<u64 Sz, typename TierT>
570   inline __attribute__((always_inline)) bool
571   __expand_tlsf(TierT &tier, usize sz)
572   {
573     auto __g = __struct_guard();
574     __debug_print("__expand_tlsf(): class size: ", Sz);
575     __debug_print("__expand_tlsf(): requested backing region: ", sz);
576     using Nd = node<tlsf_sheet<Sz>>;
577     using Sh = tlsf_sheet<Sz>;
578     usize pair_sz = sizeof(Nd) + sizeof(Sh);
579     micron::__chunk<byte> buf = __mark_arena(pair_sz);
580     byte *p = buf.ptr;
581     usize aligned_sz = __page_round(sz);
582     auto chnk = __get_kernel_chunk<micron::__chunk<byte>>(aligned_sz);
583     if ( !__kernel_chunk_valid(chnk) ) [[unlikely]] {
584       __debug_print("__expand_tlsf(): mmap failed for tlsf expansion, class: ", Sz);
585       __debug_print("__expand_tlsf(): requested size: ", aligned_sz);
586       __unmark_from_arena(buf.ptr, pair_sz);
587       return false;
588     }
589     auto *nd = new (p) Nd();
590     p += sizeof(Nd);
591     nd->nd = new (p) Sh(this, chnk);

647   inline __attribute__((always_inline)) bool
648   __expand_buddy(TierT &tier, usize sz)
649   {

651     __debug_print("__expand_buddy(): class size: ", Sz);
652     __debug_print("__expand_buddy(): requested expansion size: ", sz);
653     using Nd = node<sheet<Sz>>;
654     using Sh = sheet<Sz>;
655     usize pair_sz = sizeof(Nd) + sizeof(Sh);
656     micron::__chunk<byte> buf = __mark_arena(pair_sz);
657     byte *p = buf.ptr;
658 
659     micron::__chunk<byte> chnk;
660     if constexpr ( __default_insert_guard_pages ) {
661       __debug_print("__expand_buddy(): inserting guard page for class: ", Sz);
662       chnk = __get_kernel_chunk<micron::__chunk<byte>>(sz + __system_pagesize);
663       if ( !__kernel_chunk_valid(chnk) ) [[unlikely]] {
664         __debug_print("__expand_buddy(): mmap failed for buddy expansion, class: ", Sz);
665         __unmark_from_arena(buf.ptr, pair_sz);
666         return false;
667       }
668       __make_guard(chnk);
669     } else {
670       chnk = __get_kernel_chunk<micron::__chunk<byte>>(sz);

674         return false;
675       }
676     }

679     p += sizeof(Nd);
680     if constexpr ( __default_insert_guard_pages ) {
681       nd->nd = new (p) Sh(this, chnk, __system_pagesize);
682     } else {
683       nd->nd = new (p) Sh(this, chnk);
684     }
685     if ( nd->nd->empty() ) [[unlikely]] {
686       __debug_print("__expand_buddy(): sheet construction failed (empty), class: ", Sz);
687       nd->nd->release();
688       __unmark_from_arena(buf.ptr, pair_sz);

696       tier.unlink_node(nd);
697       nd->nd->release();
698       __unmark_from_arena(buf.ptr, pair_sz);
699       return false;
700     }
701     __debug_print("__expand_buddy(): new buddy node ready for class: ", Sz);
702     return true;
703   }

778   template<typename TierT>
779   inline __attribute__((always_inline)) micron::__chunk<byte>
780   __bucket_insert(TierT &tier, const usize sz)
781   {
782     // mru cache opt
783     u32 lh = tier.__last_hit;
784     if ( lh < tier.__count and tier.__mask_get(lh) ) {
785       auto &sh = *tier.__idx[lh].nd->nd;
786       micron::__chunk<byte> mem;
787       if constexpr ( __default_launder ) {
788         mem = sh.temporal_mark(sz);
789       } else {
790         mem = sh.mark(sz);
791       }
792       if ( !mem.zero() ) return mem;
793       tier.mark_exhausted(lh);
794     }
795 
796     // bitmap scan fallback: walk each detail word, pop bits via ctz
797     for ( u32 w = 0; w < TierT::__detail_words; ++w ) {
798       u64 mask = tier.__space_mask[w];
799       while ( mask ) {
800         const u32 bit = static_cast<u32>(__builtin_ctzll(mask));
801         const u32 pos = (w << 6) | bit;
802         if ( pos >= tier.__count ) break;      // shouldn't trip
803         auto &sh = *tier.__idx[pos].nd->nd;
804         micron::__chunk<byte> mem;
805         if constexpr ( __default_launder ) {
806           mem = sh.temporal_mark(sz);
807         } else {
808           mem = sh.mark(sz);
809         }
810         if ( !mem.zero() ) {
811           tier.__last_hit = pos;
812           return mem;
813         }
814         // sheet exhausted for this size, clear bit
815         tier.mark_exhausted(pos);
816         mask &= mask - 1;      // clear lowest set bit in the working copy
817       }
818     }
819     return { nullptr, 0 };
820   }

822   template<typename TierT>
823   inline __attribute__((always_inline)) micron::__chunk<byte>
824   __bucket_insert_temporal(TierT &tier, const usize sz)
825   {
826     // mru cache opt
827     u32 lh = tier.__last_hit;
828     if ( lh < tier.__count and tier.__mask_get(lh) ) {
829       micron::__chunk<byte> mem = tier.__idx[lh].nd->nd->temporal_mark(sz);
830       if ( !mem.zero() ) return mem;
831       tier.mark_exhausted(lh);
832     }
833 
834     for ( u32 w = 0; w < TierT::__detail_words; ++w ) {
835       u64 mask = tier.__space_mask[w];
836       while ( mask ) {
837         const u32 bit = static_cast<u32>(__builtin_ctzll(mask));
838         const u32 pos = (w << 6) | bit;
839         if ( pos >= tier.__count ) break;
840         auto &sh = *tier.__idx[pos].nd->nd;
841         micron::__chunk<byte> mem = sh.temporal_mark(sz);
842         if ( !mem.zero() ) {
843           tier.__last_hit = pos;
844           return mem;
845         }
846         tier.mark_exhausted(pos);
847         mask &= mask - 1;
848       }
849     }
850     return { nullptr, 0 };
851   }

886   __cache_pop_or_insert(TierT &tier, const usize sz)
887   {
888     // if caching is disabled behavior is identical to without it, comped out
889     if constexpr ( __default_per_class_free_cache && TierT::__cache_slots > 0 && !__default_launder ) {
890       i32 hit;

898         hit = tier.__cache.probe_ge(static_cast<u32>(sz), maximum);
899       } else {
900         hit = tier.__cache.probe_ge(static_cast<u32>(sz));
901       }
902       if ( hit >= 0 ) [[likely]] {
903         __tcache_chunk c = tier.__cache.pop_at(static_cast<u32>(hit));
904         return { c.ptr, static_cast<usize>(c.size) };
905       }
906     }
907     return __bucket_insert(tier, sz);

969   __sweep_tier_tombstones(TierT &tier)
970   {

972     auto __g = __struct_guard();

974     tier.__dealloc_count = 0;

1013   template<typename TierT>
1014   inline __attribute__((always_inline)) void
1015   __tombstone_accounting(TierT &tier, i32 range_idx, node<typename TierT::sheet_t> *nd)
1016   {
1017     if constexpr ( __default_tombstone_sweep_interval == 0 ) {
1018       auto &sh = *nd->nd;
1019       usize ts = sh.tombstoned();
1020       usize ft = sh.ftotal();
1021       __debug_print("__tombstone_accounting(): tombstoned: ", ts);
1022       __debug_print("__tombstone_accounting(): ftotal: ", ft);
1023       if ( (ts > (ft >> 1)) and sh.used() == 0 and nd != &tier.head ) {
1024         if constexpr ( !__default_persistent_mode ) {
1025           __debug_print("__tombstone_accounting(): threshold crossed, compacting sheet", 0);
1026           sh.reset();
1027           tier.unlink_node(nd);
1028           tier.unregister(range_idx);
1029           __unmark_from_arena(reinterpret_cast<byte *>(nd), sizeof(node<typename TierT::sheet_t>) + sizeof(typename TierT::sheet_t));
1030         }
1031       }
1032     } else {
1033       // immediate reclaim: if this specific sheet is fully drained (used == 0)
1034       // AND tombstoned bytes occupy >= 50% of the sheet's total capacity,
1035       // reclaim now rather than waiting for the batch sweep
1036       // prevents sheet accumulation under monotonically-growing size patterns while preserving
1037       // mostly-pristine sheets that still have reusable free space
1038       if constexpr ( !__default_persistent_mode ) {
1039         if ( nd != &tier.head and nd->nd->used() == 0 ) {
1040           usize ts = nd->nd->tombstoned();
1041           usize ft = nd->nd->ftotal();
1042           if ( ts > (ft >> 1) ) {
1043             __debug_print("__tombstone_accounting(): drained + ratio met, immediate reclaim", 0);
1044             __debug_print("__tombstone_accounting(): tombstoned: ", ts);
1045             __debug_print("__tombstone_accounting(): ftotal: ", ft);
1046             nd->nd->reset();
1047             tier.unlink_node(nd);
1048             tier.unregister(static_cast<u32>(range_idx));
1049             __unmark_from_arena(reinterpret_cast<byte *>(nd), sizeof(node<typename TierT::sheet_t>) + sizeof(typename TierT::sheet_t));
1050             return;
1051           }
1052         }
1053       }
1054       if ( tier.bump_dealloc() ) __sweep_tier_tombstones(tier);
1055     }
1056   }

1058   // unified __tier_remove
1059   template<bool HasSize, bool ForceTombstone, typename TierT>
1060   inline bool
1061   __tier_remove_impl(TierT &tier, i32 range_idx, byte *addr, const micron::__chunk<byte> &memory)
1062   {
1063     auto *nd = tier.__idx[range_idx].nd;
1064     auto &sh = *nd->nd;
1065     __debug_print_addr("__tier_remove_impl(): found in sheet at addr: ", addr);
1066 
1067     if constexpr ( ForceTombstone ) {
1068       if constexpr ( HasSize )
1069         sh.try_tombstone(memory);
1070       else
1071         sh.try_tombstone_no_size(addr);
1072       tier.mark_available(static_cast<u32>(range_idx));
1073       __debug_print("__tier_remove_impl(): force tombstone set", 0);
1074       __tombstone_accounting(tier, range_idx, nd);
1075       return true;
1076     } else {
1077       // hot tiers default to unmark+reclaim
1078       // cold tiers default to tombstoning
1079       using sheet_t = typename TierT::sheet_t;
1080       constexpr bool tomb = tomb_for<sheet_t::__size_class>();
1081       bool ok;
1082       if constexpr ( !tomb ) {
1083         if constexpr ( HasSize )
1084           ok = sh.try_unmark(memory);
1085         else
1086           ok = sh.try_unmark_no_size(addr);
1087         if ( ok ) {
1088           tier.mark_available(static_cast<u32>(range_idx));
1089           __debug_print("__tier_remove_impl(): unmark succeeded, sheet used: ", sh.used());
1090           __try_reclaim_empty(tier, range_idx, nd);
1091           return true;
1092         }
1093       } else {
1094         if constexpr ( HasSize )
1095           ok = sh.try_tombstone(memory);
1096         else
1097           ok = sh.try_tombstone_no_size(addr);
1098         if ( ok ) {
1099           tier.mark_available(static_cast<u32>(range_idx));
1100           __debug_print("__tier_remove_impl(): tombstone set", 0);
1101           __tombstone_accounting(tier, range_idx, nd);
1102           return true;
1103         }
1104       }
1105       __debug_print_addr("__tier_remove_impl(): try failed (possible double free): ", addr);
1106       return handle_double_free(addr);
1107     }
1108   }

1117   template<typename TierT>
1118   inline bool
1119   __tier_remove_at(TierT &tier, i32 range_idx, byte *addr)
1120   {

1122     // NOTE: block-validity guard on EVERY tier
1123     if constexpr ( !__default_redzone ) {
1124       if ( !sh.is_block_allocated(addr) ) [[unlikely]]

1665   hot_fn(micron::__chunk<byte>) push(const usize sz)
1666   {
1667     __debug_print("push(): requested size: ", sz);
1668     collect_stats<stat_type::alloc>();
1669     collect_stats<stat_type::total_memory_req>(sz);
1670 
1671     if ( check_constraint(sz) ) [[unlikely]] {
1672       __debug_print("push()!!!: size exceeds constraint limit: ", sz);
1673       abort_state();
1674     }
1675     if ( check_oom() ) [[unlikely]] {
1676       __debug_print("push()!!!: OOM check triggered at size: ", sz);
1677       abort_state();
1678     }
1679 
1680     usize alloc_sz = __rz_inflate(sz);
1681     bool rz_active = __rz_active(sz);
1682 
1683     micron::__chunk<byte> memory;
1684 
1685     // __default_max_retries + 1 not __default_max_retries
1686     // if __default_max_retries is == 0 or 1 this meant that this loop either never fired or did not reattempt to populate memory after the
1687     // first sheet expansion
1688     for ( u64 i = 0; i <= __default_max_retries; ++i ) {
1689       if ( memory = __vmap_alloc(alloc_sz); !memory.zero() ) [[likely]] {
1690         if ( rz_active ) __rz_apply(memory, sz);
1691         __debug_print("push(): allocated bytes: ", memory.len);
1692         zero_on_alloc(memory.ptr, memory.len);
1693         sanitize_on_alloc(memory.ptr, memory.len);
1694         collect_stats<stat_type::total_memory_throughput>(__stats_release_size(memory.ptr, memory.len));
1695         ABC_DOCTOR(doctor::record_alloc(memory.ptr, sz);)
1696         return memory;
1697       }
1698       if ( i == __default_max_retries ) break;
1699 
1700       __debug_print("push(): alloc failed, retry: ", i);
1701       __debug_print("push(): expanding for alloc_sz: ", alloc_sz);
1702 
1703       // one ladder for every profile
1704       bool expanded = false;
1705 
1706       if ( alloc_sz <= __class_small ) {
1707         usize __next_sz = __calculate_space_cache(__default_cache_step);
1708         __debug_print("push(): precise/small path, expanding by: ", __next_sz);
1709         expanded = __buf_expand_exact(alloc_sz, __next_sz);
1710       } else if ( alloc_sz < __class_medium ) {
1711         // small tier
1712         usize __next_sz = __calculate_space_small(alloc_sz) * __default_overcommit;
1713         __predict += __next_sz;
1714         usize predicted = __predict.predict_size(__next_sz);
1715         __debug_print("push(): small path, next_sz: ", __next_sz);
1716         __debug_print("push(): predictor suggested: ", predicted);
1717         expanded = __buf_expand_exact(alloc_sz, predicted);
1718       } else if ( alloc_sz <= __class_large ) {
1719         // medium tier
1720         usize __next_sz = __calculate_space_medium(alloc_sz) * __default_overcommit;
1721         __predict += __next_sz;
1722         usize predicted = __predict.predict_size(__next_sz);
1723         __debug_print("push(): medium path, next_sz: ", __next_sz);
1724         __debug_print("push(): predictor suggested: ", predicted);
1725         expanded = __buf_expand_exact(alloc_sz, predicted);
1726       } else if ( alloc_sz <= __class_huge ) {
1727         // large tier; gets 2x medium now
1728         usize __next_sz = __calculate_space_large(alloc_sz) * __default_overcommit;
1729         __predict += __next_sz;
1730         usize predicted = __predict.predict_size(__next_sz);
1731         __debug_print("push(): large path, next_sz: ", __next_sz);
1732         __debug_print("push(): predictor suggested: ", predicted);
1733         expanded = __buf_expand_exact(alloc_sz, predicted);
1734       } else if ( alloc_sz < __class_gb ) {
1735         // huge tier; new growth fn, more aggressive low allocs with tapered high allocs
1736         // now multivariate, grows more rapidly the more sheets are in use
1737         usize __base = __calculate_space_huge(alloc_sz) * __default_overcommit;
1738         usize __mult = (alloc_sz >= __class_1mb) ? 1ULL : (1ULL + static_cast<usize>(_huge.__count));
1739         usize __next_sz = __base * __mult;
1740         // WARNING: off-by-one safeguard; our buddy requires that the chunk to strictly exceed 2 * alloc_sz due to __hdr_offsets
1741         usize __min_huge = (alloc_sz << 1) + __class_huge;
1742         if ( __next_sz < __min_huge ) __next_sz = __min_huge;
1743         __predict += __next_sz;
1744         usize predicted = __predict.predict_size(__next_sz);
1745         __debug_print("push(): huge path, next_sz: ", __next_sz);
1746         __debug_print("push(): predictor suggested: ", predicted);
1747         expanded = __buf_expand_exact(alloc_sz, predicted);
1748       } else {
1749         // bulk; same off-by-one safeguard
1750         usize bulk = __calculate_space_bulk(alloc_sz);
1751         usize __min_bulk = (alloc_sz << 1) + __class_huge;
1752         if ( bulk < __min_bulk ) bulk = __min_bulk;
1753         __debug_print("push(): bulk (>=gb) path, bulk_sz: ", bulk);
1754         expanded = __buf_expand_exact(alloc_sz, bulk);
1755       }
1756 
1757       if ( !expanded ) [[unlikely]] {
1758         __debug_print("push(): expansion failed (mmap OOM or tier full), giving up", 0);
1759         break;
1760       }
1761     }
1762 
1763     __debug_print("push()!!!: all retries exhausted for size: ", sz);
1764     return { (byte *)-1, micron::numeric_limits<usize>::max() };
1765   }

2061   byte *
2062   resize(byte *ptr, usize new_sz, usize preserve_sz)
2063   {
2064     if ( __is_cached(ptr) ) [[unlikely]]
2065       return nullptr;
2066     const usize old_size = __size_of_alloc(micron::ptr_cast<addr_t *>(ptr));
2067     if ( old_size == 0 ) [[unlikely]]
2068       return nullptr;
2069     // in-place reuse
2070     if ( old_size >= new_sz and new_sz > (old_size >> 1) ) {
2071       // the trailing canary still sits at ptr + the ORIGINAL request, i.e. inside the region the
2072       // caller may now legally write, so re-lay it at the new bound. new_sz <= old_size ==
2073       // bs - hdr - 2 * rz guarantees ptr + new_sz + rz <= block end
2074       if constexpr ( __default_redzone ) {

2078       ABC_DOCTOR(doctor::record_realloc(ptr, new_sz);)

2082     micron::__chunk<byte> fresh = push(new_sz);

2087     if ( copy_size > preserve_sz ) copy_size = preserve_sz;
2088     micron::memcpy(fresh.ptr, ptr, copy_size);
2089     (void)pop(ptr);
2090     return fresh.ptr;

2114       byte *tlsf_user = reinterpret_cast<byte *>(addr);
2115       usize tlsf_overhead = __hdr_offset;
2116       if constexpr ( __default_redzone ) {
2117         tlsf_user -= static_cast<usize>(__default_redzone_size);
2118         tlsf_overhead = __hdr_offset + 2 * static_cast<usize>(__default_redzone_size);
2119       }
2120       if ( (idx = _precise.find_range(addr)) >= 0 ) {
2121         const usize bs = _precise.__idx[idx].nd->nd->block_size_of(tlsf_user);
2122         const usize recovered = (bs > tlsf_overhead) ? bs - tlsf_overhead : 0;

2124         return recovered;
2125       }
2126       if ( (idx = _small.find_range(addr)) >= 0 ) {
2127         const usize bs = _small.__idx[idx].nd->nd->block_size_of(tlsf_user);
2128         const usize recovered = (bs > tlsf_overhead) ? bs - tlsf_overhead : 0;
2129         __debug_print("__size_of_alloc(): small tlsf user size: ", recovered);
2130         return recovered;
2131       }
2132     }
2133 
2134     // buddy classes: the AUTHORITATIVE order lives in the buddy's block_tags
2135     // (one tag per min-block, rewritten on every (de)allocation, split and merge)
2136     if ( (idx = _medium.find_range(addr)) >= 0 ) {
2137       const usize bs = _medium.__idx[idx].nd->nd->block_size_of(reinterpret_cast<byte *>(addr));
