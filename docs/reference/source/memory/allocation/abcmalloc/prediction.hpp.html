22 #pragma once
23 
24 #include "../../../array/arrays.hpp"
25 #include "config.hpp"
26 
27 namespace abc
28 {
29 constexpr static const u64 __max_cached = 32;
30 
31 class alloc_predictor
32 {
33   // Keep sum and div maintained incrementally instead of recomputing each time
34   micron::carray<usize, __max_cached> allocs;
35   u64 running_sum;
36   u64 running_div;      // count of non-zero entries
37   u32 index;
38 
39   static constexpr usize
40   align_to_page(const usize n) noexcept
41   {
42     return (n + __system_pagesize - 1) & ~(usize(__system_pagesize) - 1);
43   }
44 
45   [[nodiscard]] inline u64
46   of_class(const usize sz) const noexcept
47   {
48     return (sz < __class_small)    ? __class_precise
49            : (sz < __class_medium) ? __class_small
50            : (sz < __class_large)  ? __class_medium
51            : (sz < __class_1mb)    ? __class_large
52            : (sz < __class_gb)     ? __class_1mb
53                                    : __class_gb;
54   }
55 
56 public:
57   ~alloc_predictor() = default;
58 
59   alloc_predictor() noexcept : allocs{}, running_sum(0), running_div(0), index(0) { }
60 
61   alloc_predictor(const alloc_predictor &) = default;
62   alloc_predictor(alloc_predictor &&) = default;
63   alloc_predictor &operator=(const alloc_predictor &) = default;
64   alloc_predictor &operator=(alloc_predictor &&) = default;
65 
66   inline alloc_predictor &
67   operator+=(const usize n) noexcept
68   {
69     const u32 i = index;
70 
71     const usize old = allocs[i];
72     running_sum -= old;
73     running_div -= (old != 0);
74 
75     allocs[i] = n;
76     running_sum += n;
77     running_div += (n != 0);
78 
79     index = (i + 1) & (__max_cached - 1);
80 
81     return *this;
82   }
83 
84   [[nodiscard]] inline usize
85   predict_size(const usize n) const noexcept
86   {
87     if ( __builtin_expect(running_div == 0, 0) ) return align_to_page(n);
88 
89     const usize mean_alloc = static_cast<usize>(running_sum / running_div);
90 
91     if ( mean_alloc == 0 || n > mean_alloc * 3 || of_class(n) != of_class(mean_alloc) ) {
92       return align_to_page(n);
93     }
94 
95     return align_to_page(mean_alloc > n ? mean_alloc : n);
96   }
97 };
98 };      // namespace abc
