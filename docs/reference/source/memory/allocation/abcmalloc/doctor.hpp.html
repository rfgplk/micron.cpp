199 [[gnu::noinline]] inline usize
200 __capture_backtrace(void **out, usize maxn) noexcept
201 {
202   usize n = 0;
203   if constexpr ( __default_doctor_backtrace ) {
204     void **fp = reinterpret_cast<void **>(__builtin_frame_address(0));
205     const uintptr_t lo = reinterpret_cast<uintptr_t>(fp);
206     const uintptr_t hi = lo + (16u << 20);      // 16 MB window above the current frame
207     usize guard = 0;
208     while ( fp && n < maxn && guard++ < 128 ) {
209       const uintptr_t a = reinterpret_cast<uintptr_t>(fp);
210       if ( a < lo || a >= hi || (a & (sizeof(void *) - 1)) ) break;      // not a plausible frame pointer
211       void *ret = fp[1];
212       void **next = reinterpret_cast<void **>(fp[0]);
213       if ( !ret ) break;
214       out[n++] = ret;
215       if ( reinterpret_cast<uintptr_t>(next) <= a ) break;      // fp chain must strictly ascend
216       fp = next;
217     }
218   }
219   for ( usize i = n; i < maxn; ++i ) out[i] = nullptr;
220   return n;
221 }

223 inline void
224 __thread_name(i32 tid, char *out, usize cap) noexcept
225 {
226   if ( cap ) out[0] = 0;
227   if ( cap < 2 || tid <= 0 ) return;
228   char path[48];
229   usize n = 0;
230   for ( const char *p = "/proc/self/task/"; *p; ++p ) path[n++] = *p;
231   char tmp[12];
232   usize tn = 0;
233   u32 v = static_cast<u32>(tid);
234   do {
235     tmp[tn++] = static_cast<char>('0' + v % 10);
236     v /= 10;
237   } while ( v );
238   while ( tn ) path[n++] = tmp[--tn];
239   for ( const char *p = "/comm"; *p; ++p ) path[n++] = *p;
240   path[n] = 0;
241   const int fd = static_cast<int>(micron::syscall(SYS_open, path, 0 /*O_RDONLY*/, 0));
242   if ( fd < 0 ) return;
243   const long r = static_cast<long>(micron::syscall(SYS_read, fd, out, cap - 1));
244   micron::syscall(SYS_close, fd);
245   if ( r <= 0 ) {
246     out[0] = 0;
247     return;
248   }
249   usize len = static_cast<usize>(r);
250   while ( len && (out[len - 1] == '\n' || out[len - 1] == '\r') ) --len;      // strip trailing newline
251   out[len] = 0;
252 }

351 inline void
352 __doctor_fault_handler(int sig, micron::posix::siginfo_t *info, void *ucontext) noexcept
353 {
354   const void *addr = info ? info->_sifields._sigfault.si_addr : nullptr;
355   if ( __fault_armed ) {
356     __fault_armed = false;
357     __fault_addr = addr;
358 #if defined(__micron_arch_amd64)
359     __dr_longjmp(__fault_jmp, 1);      // noreturn -> back to the armed __dr_setjmp
360 #endif
361   }
362 
363   __report_hw_fault(sig, addr, ucontext);
364   micron::posix::sigaction(sig, (sig == micron::posix::sig_bus) ? __prev_bus : __prev_segv, nullptr);
365 }

367 inline void
368 __install_fault_handler(void) noexcept
369 {
370   if constexpr ( __crash_safe_on ) {
371     if ( __sig_install_once.test_and_set(micron::memory_order_acquire) ) return;      // exactly once
372     micron::posix::sigaction_t sa{};
373     sa.sigaction_handler.sa_sigaction = &__doctor_fault_handler;
374     micron::posix::sigemptyset(sa.sa_mask);
375     // WARNING: SA_NODEFER is required we longjmp out instead of returning via sigreturn
376     // the kernel never unblocks the signal and without NODEFER the second fault arrives blocked
377     sa.sa_flags = micron::posix::sa_siginfo | micron::posix::sa_nodefer;
378     micron::posix::sigaction(micron::posix::sig_segv, sa, &__prev_segv);
379     micron::posix::sigaction(micron::posix::sig_bus, sa, &__prev_bus);
380   }

411 {
412   const uintptr_t a = reinterpret_cast<uintptr_t>(base);
413   byte buf[16];
414   const auto emit = [&](uintptr_t at, usize n) {

428   for ( usize off = 0; off < len; off += 16 ) emit(a + off, (len - off < 16) ? (len - off) : 16);

552   void
553   __rehash(usize newcap) noexcept
554   {
555     if ( newcap == 0 ) return;
556     __rec *old = slots;
557     usize oldcap = cap;
558     usize bytes = newcap * sizeof(__rec);
559     __rec *fresh = reinterpret_cast<__rec *>(__mmap_bytes(bytes));
560     if ( !fresh ) {
561       // cannot grow: leave the old table in place; doctor becomes best-effort
562       __banner("ledger: mmap failed to grow ledger; tracking is now best-effort\n");
563       return;
564     }
565     slots = fresh;
566     cap = newcap;
567     mask = newcap - 1;
568     occupied = 0;
569     if ( old ) {
570       for ( usize i = 0; i < oldcap; ++i ) {
571         if ( old[i].state != __rec_state::empty && old[i].key != 0 ) {
572           __rec *dst = __slot_for(old[i].key);
573           if ( dst ) {
574             *dst = old[i];
575             ++occupied;
576           }
577         }
578       }
579       __munmap_bytes(old, oldcap * sizeof(__rec));
580     }
581   }

589   void
590   __grow_if_needed(void) noexcept
591   {
592     if ( cap == 0 ) return;      // ledger not initialized (mmap failed); nothing to grow
593     // grow at 70% load
594     if ( occupied * 10 < cap * 7 ) return;
595     usize newcap = cap << 1;
596     if ( newcap > __default_doctor_max_records ) {
597       if ( __drop_freed() && occupied * 10 < cap * 7 ) return;
598     }
599     __rehash(newcap);

602   __rec *
603   __slot_for(uintptr_t key) noexcept
604   {
605     if ( !slots ) return nullptr;
606     usize i = static_cast<usize>(__hash(key)) & mask;
607     for ( usize probes = 0; probes <= cap; ++probes ) {
608       __rec &s = slots[i];
609       if ( s.state == __rec_state::empty || s.key == key ) return &s;
610       i = (i + 1) & mask;
611     }
612     return nullptr;
613   }

615   __rec *
616   __find(uintptr_t key) noexcept
617   {
618     if ( !slots ) return nullptr;
619     usize i = static_cast<usize>(__hash(key)) & mask;
620     for ( usize probes = 0; probes <= cap; ++probes ) {
621       __rec &s = slots[i];
622       if ( s.state == __rec_state::empty ) return nullptr;
623       if ( s.key == key ) return &s;
624       i = (i + 1) & mask;
625     }
626     return nullptr;
627   }

629   bool
630   __drop_freed(void) noexcept
631   {
632     if ( !slots || occupied <= n_live ) return false;      // nothing non-live to reclaim
633     __rec *old = slots;
634     usize oldcap = cap;
635     const usize before = occupied;
636     __rec *fresh = reinterpret_cast<__rec *>(__mmap_bytes(oldcap * sizeof(__rec)));
637     if ( !fresh ) return false;
638     slots = fresh;
639     mask = oldcap - 1;      // cap unchanged
640     occupied = 0;
641     for ( usize i = 0; i < oldcap; ++i ) {
642       if ( old[i].state == __rec_state::live ) {

646           ++occupied;
647         }
648       }
649     }
650     n_freed = 0;
651     __munmap_bytes(old, oldcap * sizeof(__rec));
652     __banner("ledger: evicted ");
653     __d_u(before - occupied);
654     __d(" non-live records (freed/tombstoned/quarantined) to stay under the record cap\n");
655     return true;
656   }

749 inline void
750 __doctor_arm_canaries(byte *ptr, usize req, const void *owner, __rec &s) noexcept
751 {
752   s.hdr_shadow = 0;
753   s.blk_size = 0;
754   __arena *o = static_cast<__arena *>(const_cast<void *>(owner));
755   if ( !o || o != __tls_arena ) return;      // foreign / cross-arena: nothing here is safe to read
756   const int kind = o->__doctor_tier_kind(reinterpret_cast<addr_t *>(ptr));
757   const usize real = o->__size_of_alloc(reinterpret_cast<addr_t *>(ptr));
758   // record it unconditionally
759   s.blk_size = real;
760   if constexpr ( __default_doctor_canary ) {
761     if ( kind == 1 ) {      // TLSF: bsize (u32) at user - __hdr_offset
762       u32 bs = 0;
763       __builtin_memcpy(&bs, ptr - __hdr_offset, sizeof(bs));
764       s.hdr_shadow = bs;
765     } else if ( kind == 2 ) {      // buddy: order (i32) at the tail = user + real
766       i32 ord = 0;
767       __builtin_memcpy(&ord, ptr + real, sizeof(ord));
768       // OR in the captured marker so order 0 is distinct from the 0 "not captured" sentinel
769       s.hdr_shadow = __hdr_shadow_buddy_captured | static_cast<u32>(ord);
770     }
771     if constexpr ( !ABC_EFF_REDZONE ) {      // the allocator's own redzone owns that region if enabled
772       if ( real > req ) __builtin_memset(ptr + req, __default_doctor_canary_byte, real - req);
773     }
774   }

777 inline void
778 record_alloc(byte *ptr, usize req_size) noexcept
779 {
780   if ( !ptr ) return;
781   __reentry re;
782   if ( !re ) return;
783   __scoped_lock g;
784   __dr.__ensure();
785   if ( !__dr.slots ) {
786 #if defined(MICRON_ABC_INSPECT_HISTORY)
787     ++__dr.record_losses;
788 #endif
789     return;
790   }
791   __dr.__grow_if_needed();
792   __rec *s = __dr.__slot_for(reinterpret_cast<uintptr_t>(ptr));
793   if ( !s ) {
794 #if defined(MICRON_ABC_INSPECT_HISTORY)
795     ++__dr.record_losses;
796 #endif
797     return;
798   }
799 #if defined(MICRON_ABC_INSPECT_HISTORY)
800   for ( usize i = 0; i < __doctor_bt_depth; ++i ) s->free_bt[i] = nullptr;
801 #endif
802   if ( s->state == __rec_state::empty )
803     ++__dr.occupied;
804   else if ( s->state == __rec_state::freed || s->state == __rec_state::tombstoned ) {
805     if ( __dr.n_freed ) --__dr.n_freed;
806   }
807   if ( s->state != __rec_state::live ) ++__dr.n_live;
808   s->key = reinterpret_cast<uintptr_t>(ptr);
809   s->req_size = req_size;
810   s->alloc_op = __dr.next_op();
811   s->free_op = 0;
812   s->alloc_tid = __gettid();
813   s->free_tid = 0;
814   s->free_len = 0;
815   s->flags = __exe_init_done ? 0u : __rec_flag_pre_exe_init;
816   s->owner = static_cast<const void *>(__owner_of(ptr));
817   s->state = __rec_state::live;
818   __doctor_arm_canaries(ptr, req_size, s->owner, *s);
819   __capture_backtrace(s->alloc_bt, __doctor_bt_depth);
820   __dr.__push_event(0, s->key, req_size, s->alloc_op, s->alloc_tid);
821 }

848 // mark a tracked pointer freed
849 inline void
850 __mark_free_locked(byte *ptr, __rec_state to, usize len) noexcept
851 {
852   __rec *s = __dr.__find(reinterpret_cast<uintptr_t>(ptr));
853   if ( !s ) return;      // untracked (foreign)
854   if ( s->state == __rec_state::freed || s->state == __rec_state::tombstoned || s->state == __rec_state::quarantined ) {
855     // already freed/tombstoned (double free), or quarantined by a rescue handler
856     return;
857   }
858   if ( s->state == __rec_state::live && __dr.n_live ) --__dr.n_live;
859   ++__dr.n_freed;
860   s->state = to;
861   s->free_op = __dr.next_op();
862   s->free_tid = __gettid();
863   s->free_len = len;
864 #if defined(MICRON_ABC_INSPECT_HISTORY)
865   __capture_backtrace(s->free_bt, __doctor_bt_depth);
866 #endif
867   __dr.__push_event(to == __rec_state::tombstoned ? 2 : 1, s->key, len, s->free_op, s->free_tid);
868 }

1073 leaks(void)
1074 {
1075   __reentry re;
1076   if ( !re ) return;
1077   __scoped_lock g;
1078   __install_fault_handler();
1079   __banner("leaks: live tracked pointers (classified by tier)\n");
1080   __arena *const self = __tls_arena;
1081   usize shown = 0, total_bytes = 0;
1082   usize pre_cnt = 0, pre_bytes = 0;      // recorded before the executable's own init: DSO / bootstrap
1083   // classes: 0 unresolved/internal, 1 TLSF, 2 buddy, 3 cross-thread (owned by another arena)
1084   usize cls_cnt[4] = { 0, 0, 0, 0 };
1085   usize cls_bytes[4] = { 0, 0, 0, 0 };
1086   usize cls_max[4] = { 0, 0, 0, 0 };
1087   if ( __dr.slots ) {

1094         ++pre_cnt;
1095         pre_bytes += s.req_size;
1096       }
1097       byte *u = reinterpret_cast<byte *>(s.key);
1098       int cls = 0;
1099       __arena *o = __va_contains(u) ? __owner_of(u) : nullptr;
1100       if ( o && o != self )
1101         cls = 3;
1102       else if ( o == self ) {
1103         int kind = 0;
1104         __guard_read([&] { kind = o->__doctor_tier_kind(reinterpret_cast<addr_t *>(u)); });
1105         cls = (kind == 1) ? 1 : (kind == 2) ? 2 : 0;
1106       }
1107       ++cls_cnt[cls];
1108       cls_bytes[cls] += s.req_size;
1109       if ( s.req_size > cls_max[cls] ) cls_max[cls] = s.req_size;

1111       __d_ptr(u);
1112       __d("  ");
1113       __d_u(s.req_size);

1118       __d_nl();
1119     }
1120   }
1121   __d("  total live: ");
1122   __d_u(shown);
1123   __d(" (");
1124   __d_u(total_bytes);
1125   __d(" B)\n");
1126   static const char *const names[4] = { "unresolved/internal", "TLSF (user)", "buddy (user)", "cross-thread" };

1135       __d(" B (largest ");
1136       __d_u(cls_max[c]);

1143     __d_u(pre_bytes);
1144     __d(" B allocated before this executable's init (DSO .init_array / runtime bootstrap).\n");

1233 // decode && splat the block metadata header for a live/known user pointer
1234 inline void
1235 __decode_header(byte *user, usize user_size, int kind, bool expect_live, const __rec *rec) noexcept
1236 {
1237   if ( kind != 1 && kind != 2 ) {
1238     __d("  header       (allocator kind unresolved; block not in any tier)\n");
1239     return;
1240   }
1241   byte *h = (kind == 1) ? user - __hdr_offset : user + user_size;      // buddy header lives at the block tail
1242   const usize hdr_n = (kind == 1) ? __hdr_offset : 2 * sizeof(i32);
1243   byte c[__hdr_offset] = {};
1244   const bool readable = __guard_read([&] {
1245     for ( usize i = 0; i < hdr_n; ++i ) c[i] = h[i];      // copy under guard; decode outside
1246   });

1712 {

1714   __install_fault_handler();

1716   __d(cause);
1717   __d(")\n");
1718 
1719   __arena *const self = __tls_arena;
1720   __for_each_live_arena([&](__arena &a) {
1721     if ( &a != self ) {
1722       ++ctx.arenas;
1723       ++ctx.skipped_arenas;
1724       return;
1725     }
1726     ctx.repair = __default_doctor_rescue && !__default_doctor_rescue_conservative;
1727     if ( !__guard_read([&] {
1728            a.__doctor_check_struct(ctx);
1729            a.__doctor_walk_blocks(ctx);
1730          }) )
1731       ctx.note("deep walk FAULTED (corrupt sheet metadata); aborted this arena's walk", self);
1732   });
1733   ctx.repair = false;
1734 
1735   if ( __dr.slots ) {
1736     for ( usize i = 0; i < __dr.cap; ++i ) {
1737       __rec &r = __dr.slots[i];
1738       if ( r.state != __rec_state::live ) continue;
1739       byte *u = reinterpret_cast<byte *>(r.key);
1740       ++ctx.live_checked;
1741       if ( !__va_contains(u) ) {
1742         ctx.note("live record: pointer not in abcmalloc VA region", u);
1743         continue;
1744       }
1745       __arena *o = __owner_of(u);
1746       if ( !o ) {
1747         ctx.note("live record: granule has no owner (sheet released underneath?)", u);
1748         continue;
1749       }
1750       if ( o != self ) {
1751         ++ctx.cross_thread_live;
1752         continue;
1753       }
1754       const bool ok = __guard_read([&] {
1755         if ( !o->has_provenance(reinterpret_cast<addr_t *>(u)) ) {
1756           ctx.note("live record: arena lost provenance", u);
1757           return;
1758         }
1759         const usize sz = o->__size_of_alloc(reinterpret_cast<addr_t *>(u));
1760         const int kind = o->__doctor_tier_kind(reinterpret_cast<addr_t *>(u));
1761         if ( !o->is_valid_block(reinterpret_cast<addr_t *>(u)) && !o->__is_cached(u) )
1762           ctx.note("live record: allocator reports block NOT live (use-after-free / silent free)", u);
1763         const bool hdr_struct_ok = !(kind && sz) || __check_header_ok(u, sz, kind);
1764         if ( kind && sz && !hdr_struct_ok ) ctx.note("live record: block metadata header INVALID (corruption)", u);
1765         if ( r.owner && o != r.owner )
1766           ctx.note("live record: granule owner changed since alloc (sheet released + re-registered under a live block)", u);
1767         if ( sz && r.req_size > sz )
1768           ctx.note("live record: recorded request exceeds the real block size (ledger/allocator disagree -- corruption)", u);
1769         if constexpr ( __default_doctor_canary ) {
1770           bool hdr_shadow_ok = true;
1771           if ( kind && r.hdr_shadow ) {
1772             u32 cur = 0;
1773             if ( kind == 1 )
1774               __builtin_memcpy(&cur, u - __hdr_offset, sizeof(cur));
1775             else {
1776               __builtin_memcpy(&cur, u + sz, sizeof(cur));
1777               cur = __hdr_shadow_buddy_captured | cur;      // match the captured marker armed for buddy blocks
1778             }
1779             if ( cur != r.hdr_shadow ) {
1780               hdr_shadow_ok = false;
1781               ctx.note("live record: header structural field changed since alloc (metadata overwrite)", u);
1782               __note_overflow_source(u);
1783             }
1784           }
1785           if constexpr ( !ABC_EFF_REDZONE ) {
1786             // skip chunk-API blocks: [req_size, blk_size) is the caller's to write
1787             if ( hdr_struct_ok && hdr_shadow_ok && sz > r.req_size && !(r.flags & __rec_flag_chunk) ) {

2337   __d(" -- or use the size-less abc::free(ptr)\n");
2338   if ( real && claimed > real )
2339     __note("claimed > block size: the size-based free scrub (poison/zero-on-free) would overrun the block -> heap corruption");
2340   if constexpr ( __default_doctor_rescue ) {

2342     __d("  RESCUE       clamped free size to ");
2343     __d_u(safe);
2344     __d("; continuing\n\n");
2345     return safe;
2346   }
2347   const usize safe = (real && claimed > real) ? real : claimed;
2348   __d("  action       wrong-sized free -> fail_state() per __default_fail_result (size clamped to ");
2349   __d_u(safe);
2350   __d(")\n\n");
2351   abc::fail_state();      // aborts under fail_result 0/1; returns under mode 2
2352   return safe;            // fail_result==2: proceed with the clamped-safe size so the scrub cannot overrun

2405 // A .fini_array entry, NOT a static destructor. glibc registers _dl_fini first, so it runs last, and
2406 // _dl_fini walks the executable's .fini_array after every __cxa_atexit handler -- so this sees the
2407 // blocks that ordinary static destructors free, which an inline-object destructor in the executable
2408 // could not (its registration is the most recent, so LIFO fires it first). Freestanding matches:
2409 // __run_exit_sequence() settles exit callbacks and TLS destructors before walking __fini_array_* in reverse.
2410 // What still cannot be seen is DSO teardown, which _dl_fini reaches after us -- hence the pre-exe
2411 // bucket in leaks(), and the note below.
2412 [[maybe_unused]] __attribute__((destructor(101))) static void
2413 __doctor_leak_report_fini(void) noexcept
2414 {
2415   if constexpr ( __default_doctor_leak_report_at_exit ) {
2416     u32 expect = 0;      // one report per process, however many TUs pulled this header in
2417     if ( !__leak_reported.compare_exchange_strong(expect, 1, micron::memory_order_acq_rel, micron::memory_order_acquire) ) return;
2418     __banner("process-exit leak report (runs before DSO teardown; a nonzero floor can be structural)\n");
2419     leaks();
2420     stats();
2421   }
