104 mark_at(byte *ptr, usize size)
105 {
106   if ( !ptr or size == 0 ) [[unlikely]]
107     return nullptr;
108 
109   if ( __query_arena(ptr)->has_provenance(micron::ptr_cast<addr_t *>(ptr)) || !register_external(ptr, size) ) [[unlikely]] {

111     micron::exc<micron::except::memory_error_abc_mark>("mark_at(): region already tracked by allocator");
112     return nullptr;
113   }
114 
115   return ptr;

213 salloc(usize size)
214 {
215   if ( size == 0 ) [[unlikely]]
216     return nullptr;
217 
218   micron::__chunk<byte> mem = __current_arena()->push(size);
219   if ( __is_sentinel(mem.ptr) ) [[unlikely]] {

229 inline void
230 dealloc(byte *ptr)
231 {
232   if ( !ptr ) [[unlikely]]
233     return;
234 
235   // WARNING:: due to poor interop with missing the STL and micron, there are edge cases where global objects constructed by glibc might
236   // be invoked through *this* dealloc, instead of the libc free() one
237   if ( !__route_dealloc(ptr, 0) ) [[unlikely]] {
238     // REGARDING FOREIGN POINTERS
239     // there are certain cases (aka bad code) where we genuinely get hit by a rouge pointer not owned by us;
240     // there are two possible ways to handle it, a) pass it off to the real allocator which owns it (impossible) b) throw an error
241     // since in testing this only occured during atexit destructor calls, we will simple fall out silently and ignore dead pointers
242     (void)ptr;
243     // this effectively means that dealloc() NO LONGER has free() like semantics, since you can pass any pointer you want and it won't
244     // crash/alert you, be careful!
245   }
246 }

248 inline void
249 dealloc(byte *ptr, usize len)
250 {
251   if ( !ptr ) [[unlikely]]
252     return;
253   if ( len == 0 ) [[unlikely]] {
254     ABC_DOCTOR(if ( doctor::on_bad_free(ptr, 0, "dealloc(): zero-length free is invalid", __FILE__, __LINE__) ) return;)
255     micron::exc<micron::except::memory_error_abc_dealloc_zero>("dealloc(): zero-length free is invalid");
256     return;
257   }
258 
259   ABC_DOCTOR(len = doctor::check_free_size(ptr, len, __FILE__, __LINE__);)
260   // dealloc(ptr len) is always explicit us, no fall throughs; treat it as a hard error, we went wrong somewhere
261   if ( !__route_dealloc(ptr, len) ) [[unlikely]] {
262     ABC_DOCTOR(if ( doctor::on_bad_free(ptr, len, "dealloc(ptr,len): no arena owns this pointer", __FILE__, __LINE__) ) return;)
263     micron::exc<micron::except::memory_error_abc_dealloc_size>(
264         "dealloc(ptr,len): no arena owns this pointer (foreign provenance, double free, or interior pointer)");
265   }
266 }

322 inline void
323 which(void)
324 {
325   // walk every tier's index and report live (non-tombstoned) sheets
326   // NOTE: this only reports sheet-level information (base addr + allocated bytes).
327   // per-block enumeration would require walking each sheet's internal book for now, report total usage per tier class
328   usize tot_precise = 0, tot_small = 0, tot_medium = 0, tot_large = 0, tot_huge = 0;
329   usize tot_all = 0, tot_meta_avail = 0;
330   __for_each_live_arena([&](__arena &a) {
331     tot_precise += a.total_usage_of_class<__class_precise>();
332     tot_small += a.total_usage_of_class<__class_small>();
333     tot_medium += a.total_usage_of_class<__class_medium>();
334     tot_large += a.total_usage_of_class<__class_large>();
335     tot_huge += a.total_usage_of_class<__class_huge>();
336     tot_all += a.total_usage();
337     tot_meta_avail += a.__available_buffer();
338   });
339   __debug_print("which(): precise tier allocated: ", tot_precise);
340   __debug_print("which(): small tier allocated: ", tot_small);
341   __debug_print("which(): medium tier allocated: ", tot_medium);
342   __debug_print("which(): large tier allocated: ", tot_large);
343   __debug_print("which(): huge tier allocated: ", tot_huge);
344   __debug_print("which(): total: ", tot_all);
345   __debug_print("which(): arena metadata remaining: ", tot_meta_avail);
346 }

394 [[nodiscard, gnu::always_inline]] inline byte *
395 __aligned_base_of(byte *ptr) noexcept
396 {
397   if ( !__aligned_prefix_readable(ptr) ) [[unlikely]]
398     return nullptr;
399   const __aligned_prefix *prefix = micron::ptr_cast<const __aligned_prefix *>(ptr - sizeof(__aligned_prefix));
400   byte *const raw = prefix->raw;
401   if ( raw >= ptr || prefix->guard != __aligned_guard(ptr, raw) ) [[likely]]
402     return nullptr;
403   return __aligned_verify(raw);
404 }

406 [[nodiscard]] inline micron::__chunk<byte>
407 aligned_balloc(usize alignment, usize size)
408 {
409   if ( alignment == 0 || (alignment & (alignment - 1)) != 0 || size == 0 ) return { nullptr, 0 };
410   if ( alignment <= native_block_alignment ) return balloc(size);
411 
412   usize overhead;
413   usize total;
414   if ( check_add_overflow(alignment - 1, sizeof(__aligned_prefix), overhead) || check_add_overflow(size, overhead, total) )
415     return { nullptr, 0 };
416   micron::__chunk<byte> raw = balloc(total);
417   if ( raw.ptr == nullptr ) return { nullptr, 0 };
418   const uintptr_t first = reinterpret_cast<uintptr_t>(raw.ptr) + sizeof(__aligned_prefix);
419   const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
420   auto *prefix = reinterpret_cast<__aligned_prefix *>(aligned - sizeof(__aligned_prefix));
421   prefix->raw = raw.ptr;
422   prefix->guard = __aligned_guard(reinterpret_cast<const byte *>(aligned), raw.ptr);
423   const usize displacement = static_cast<usize>(aligned - reinterpret_cast<uintptr_t>(raw.ptr));
424   return { reinterpret_cast<byte *>(aligned), raw.len - displacement };
425 }

427 [[nodiscard]] inline micron::__chunk<byte>
428 aligned_launder(usize alignment, usize size)
429 {
430   if ( alignment == 0 || (alignment & (alignment - 1)) != 0 || size == 0 ) return { nullptr, 0 };
431   if ( alignment <= native_block_alignment ) return launder_chunk(size);
432 
433   usize overhead;
434   usize total;
435   if ( check_add_overflow(alignment - 1, sizeof(__aligned_prefix), overhead) || check_add_overflow(size, overhead, total) )
436     return { nullptr, 0 };
437   micron::__chunk<byte> raw = launder_chunk(total);
438   if ( raw.ptr == nullptr ) return { nullptr, 0 };
439   const uintptr_t first = reinterpret_cast<uintptr_t>(raw.ptr) + sizeof(__aligned_prefix);
440   const uintptr_t aligned = (first + alignment - 1) & ~(static_cast<uintptr_t>(alignment) - 1);
441   auto *prefix = reinterpret_cast<__aligned_prefix *>(aligned - sizeof(__aligned_prefix));
442   prefix->raw = raw.ptr;
443   prefix->guard = __aligned_guard(reinterpret_cast<const byte *>(aligned), raw.ptr);
444   const usize displacement = static_cast<usize>(aligned - reinterpret_cast<uintptr_t>(raw.ptr));
445   return { reinterpret_cast<byte *>(aligned), raw.len - displacement };
446 }

484 [[nodiscard]] inline micron::__chunk<byte>
485 resize_chunk(micron::__chunk<byte> old, usize size, usize preserve_bytes)
486 {
487   if ( old.ptr == nullptr ) return balloc(size);
488   if ( size == 0 ) {
489     dealloc(old.ptr);
490     return { nullptr, 0 };
491   }
492   __arena *const owner = __query_arena(old.ptr);
493   if constexpr ( __default_multithread_safe ) {
494     if ( owner != __current_arena() ) {
495       micron::__chunk<byte> next = balloc(size);
496       if ( next.ptr == nullptr ) return { nullptr, 0 };
497       const usize retained = micron::min(preserve_bytes, old.len, next.len);
498       if ( retained != 0 ) micron::memcpy(next.ptr, old.ptr, retained);
499       dealloc(old.ptr, old.len);
500       return next;
501     }
502   }
503   byte *ptr = owner->resize(old.ptr, size, preserve_bytes);
504   if ( ptr == nullptr ) return { nullptr, 0 };
505   return { ptr, __query_arena(ptr)->__size_of_alloc(micron::ptr_cast<addr_t *>(ptr)) };
506 }

508 [[nodiscard]] inline micron::__chunk<byte>
509 aligned_resize(micron::__chunk<byte> old, usize size, usize preserve_bytes, usize alignment)
510 {
511   if ( old.ptr == nullptr ) return aligned_balloc(alignment, size);
512   if ( size == 0 ) {
513     aligned_dealloc(old, alignment);
514     return { nullptr, 0 };
515   }
516   if ( alignment <= native_block_alignment ) return resize_chunk(old, size, preserve_bytes);
517 
518   micron::__chunk<byte> next = aligned_balloc(alignment, size);
519   if ( next.ptr == nullptr ) return { nullptr, 0 };
520   usize copied = old.len < next.len ? old.len : next.len;
521   if ( copied > preserve_bytes ) copied = preserve_bytes;
522   if ( copied ) micron::memcpy(next.ptr, old.ptr, copied);
523   aligned_dealloc(old, alignment);
524   return next;
525 }

558 usize
559 query_size(T *ptr)
560 {
561   if ( !ptr ) [[unlikely]]
562     return 0;
563 
564   const usize external = external_query_size(reinterpret_cast<const byte *>(ptr));
565   if ( external ) return external;
566   __arena *const owner = __query_arena(ptr);
567   if constexpr ( __default_multithread_safe ) {

569   }
570   return owner->__size_of_alloc(micron::ptr_cast<addr_t *>(ptr));

602 inline void *
603 calloc(usize num, usize size)      // alloc's zero'd out memory, prefer using salloc()
604 {
605   if ( num == 0 or size == 0 ) return nullptr;
606 
607   usize total;
608   if ( check_mul_overflow(num, size, total) ) [[unlikely]]
609     return nullptr;
610 
611   byte *mem = abc::alloc(total);
612   if ( !mem ) return nullptr;
613   micron::zero(mem, total);
614   return mem;
615 }

617 inline void *
618 realloc(void *ptr, usize size)      // reallocates memory
619 {
620   if ( !ptr ) return reinterpret_cast<void *>(abc::alloc(size));
621 
622   if ( size == 0 ) {
623     abc::dealloc(reinterpret_cast<byte *>(ptr));
624     return nullptr;
625   }
626 
627   byte *const old = reinterpret_cast<byte *>(ptr);
628 
629   // WARNING: C spec requires realloc() to accept a posix_memalign() pointer;
630   // alignment is not preserved
631   if ( byte *const raw = __aligned_base_of(old) ) [[unlikely]] {
632     const usize block = query_size(raw);
633     const usize displacement = static_cast<usize>(old - raw);
634     const usize usable = block > displacement ? block - displacement : 0;
635     byte *const next = abc::alloc(size);
636     if ( next == nullptr ) [[unlikely]]
637       return nullptr;
638     const usize kept = usable < size ? usable : size;
639     if ( kept != 0 ) micron::memcpy(next, old, kept);
640     abc::dealloc(raw);
641     return reinterpret_cast<void *>(next);
642   }
643 
644   const usize old_size = query_size(old);
645   micron::__chunk<byte> result = resize_chunk({ old, old_size }, size, old_size);
646   if ( !result.ptr ) [[unlikely]] {
647     // rescue: report, then signal failure the C-standard way (return nullptr, original block untouched)
648     ABC_DOCTOR(if ( doctor::on_bad_free(reinterpret_cast<byte *>(ptr), size, "realloc(): pointer not recognised or allocation OOM",
649                                         __FILE__, __LINE__) ) return nullptr;)
650     micron::exc<micron::except::memory_error_abc_realloc_unknown>("realloc(): resize failed (pointer not recognised or allocation OOM)");
651     return nullptr;
652   }
653   return reinterpret_cast<void *>(result.ptr);
654 }

656 inline void
657 free(void *ptr)      // frees memory, prefer abc::dealloc always
658 {
659   if ( !ptr ) [[unlikely]]
660     return;
661   byte *const p = reinterpret_cast<byte *>(ptr);
662   if ( byte *const raw = __aligned_base_of(p) ) [[unlikely]] {
663     abc::dealloc(raw);
664     return;
665   }
666   abc::dealloc(p);
667 }

670 
671 // NOTE: when alignment > __hdr_offset (32) the returned pointer differs from the allocator's block start
672 
673 inline void *
674 aligned_alloc(usize alignment, usize size)
675 {
676   // !alignment must be power of two
677   if ( alignment == 0 or (alignment & (alignment - 1)) != 0 ) [[unlikely]]
678     return nullptr;
679 
680   // !size must be a multiple of alignment
681   if ( size % alignment != 0 ) [[unlikely]]
682     return nullptr;
683 
684   if ( size == 0 ) [[unlikely]]
685     return nullptr;
686 
687   return reinterpret_cast<void *>(aligned_balloc(alignment, size).ptr);
688 }
689 
690 inline void
691 aligned_free(void *ptr)
692 {
693   if ( !ptr ) [[unlikely]]
694     return;
695 
696   byte *aligned = reinterpret_cast<byte *>(ptr);
697   if ( is_present(aligned) ) {
698     abc::dealloc(aligned);
699     return;
700   }
701 
702   if ( byte *const raw = __aligned_base_of(aligned) ) {
703     abc::dealloc(raw);
704     return;
705   }
706 
707   ABC_DOCTOR(
708       if ( doctor::on_bad_free(aligned, 0, "aligned_free(): not an aligned_alloc pointer (no valid prefix)", __FILE__, __LINE__) ) return;)
709   micron::exc<micron::except::memory_error_abc_aligned_free_bad>(
710       "aligned_free(): stashed raw pointer is invalid, this pointer was not allocated by aligned_alloc");
711 }

713 inline void
714 aligned_free(void *ptr, usize alignment)
715 {
716   aligned_dealloc(reinterpret_cast<byte *>(ptr), alignment);
717 }
