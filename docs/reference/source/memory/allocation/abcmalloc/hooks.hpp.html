33 // NOTE: u64 and usize are the same on amd64 but NOT on 32-bit targets (armv7 has u64 defined but usize is __UINTPTR__ (32b))
34 #if defined(__micron_arch_width_32)
35 // ceiling on speculative growth
36 constexpr static const u64 __width32_sheet_cap = 64ULL << 20;
37 #endif
38 
39 static inline usize
40 __saturate_pages_to_bytes(u64 pages) noexcept
41 {
42   const u64 ps = static_cast<u64>(__system_pagesize);
43   const u64 max_usize = static_cast<u64>(micron::numeric_limits<usize>::max());
44 #if defined(__micron_arch_width_32)
45   if ( pages > __width32_sheet_cap / ps ) return static_cast<usize>(__width32_sheet_cap);
46 #endif
47   if ( pages > max_usize / ps ) return static_cast<usize>(max_usize & ~(ps - 1));
48   return static_cast<usize>(pages * ps);
49 }
50 
51 // same contract as the pages variant above, for a count that is already in bytes: a byte count wider
52 // than usize can never be mapped on this target, so clamp rather than let the narrowing wrap
53 inline usize
54 __saturate_bytes(u64 bytes) noexcept
55 {
56   const u64 max_usize = static_cast<u64>(micron::numeric_limits<usize>::max());
57   return bytes > max_usize ? static_cast<usize>(max_usize) : static_cast<usize>(bytes);
58 }

60 inline usize
61 __calculate_space_cache(usize sz)
62 {
63   // x^2 * ln(x * sqrt(x))
64   // WARNING: square in double; usize sz*sz wraps at sz >= 2^16 on width-32
65   u64 t = static_cast<u64>(static_cast<flong>((double)sz * (double)sz)
66                            * micron::math::logf128(static_cast<flong>((double)sz * micron::math::hw::sqrt_sd((double)sz))));
67   float t_2 = (float)t / __system_pagesize;
68   t_2 = micron::math::ceil(t_2);
69   u64 pages = static_cast<u64>(t_2);
70   if ( pages < __default_minimum_page_mul ) pages = __default_minimum_page_mul;
71   pages = micron::math::nearest_pow2ll(pages);
72   return __saturate_pages_to_bytes(pages);
73 }

75 inline usize
76 __calculate_space_small(usize sz)
77 {
78   // new equation: x * sqrt(x) * 400 smooth sublinear-in-class growth; 4 MiB at sz=513, ~128 MiB at sz=4095
79   u64 t = static_cast<u64>(static_cast<double>(sz) * micron::math::hw::sqrt_sd(static_cast<double>(sz)) * 400.0);
80   float t_2 = (float)t / __system_pagesize;
81   t_2 = micron::math::ceil(t_2);
82   u64 pages = static_cast<u64>(t_2);
83   if ( pages < __default_minimum_page_mul ) pages = __default_minimum_page_mul;
84   pages = micron::math::nearest_pow2ll(pages);
85   return __saturate_pages_to_bytes(pages);
86 }

88 inline usize
89 __calculate_space_medium(usize sz)
90 {
91   // new equation: x * ln(x) * 150;  4 MiB at sz=4096, ~64 MiB at sz=32768; hot tier so heavy floor
92   flong f_sz = static_cast<flong>(sz);
93   u64 t = static_cast<u64>(f_sz * micron::math::logf128(f_sz) * 150.0L);
94   float t_2 = (float)t / __system_pagesize;
95   t_2 = micron::math::ceil(t_2);
96   u64 pages = static_cast<u64>(t_2);
97   if ( pages < __default_minimum_page_mul ) pages = __default_minimum_page_mul;
98   pages = micron::math::nearest_pow2ll(pages);
99   return __saturate_pages_to_bytes(pages);
100 }

102 inline usize
103 __calculate_space_large(usize sz)
104 {
105   // 2 * x * ln(x)^2; the old medium equation scaled by a factor of 2
106   flong f_sz = static_cast<flong>(sz);
107   flong lg = micron::math::logf128(f_sz);
108   u64 t = static_cast<u64>(2.0L * f_sz * lg * lg);
109   float t_2 = (float)t / __system_pagesize;
110   t_2 = micron::math::ceil(t_2);
111   u64 pages = static_cast<u64>(t_2);
112   if ( pages < __default_minimum_page_mul ) pages = __default_minimum_page_mul;
113   pages = micron::math::nearest_pow2ll(pages);
114   return __saturate_pages_to_bytes(pages);
115 }

117 inline usize
118 __calculate_space_huge(usize sz)
119 {
120   // sqrt(x) * ln(x)^2 * 125; aggressive at the floor, tapers at the top
121   double f_sz = static_cast<double>(sz);
122   double lg = static_cast<double>(micron::math::logf128(static_cast<flong>(sz)));
123   u64 t = static_cast<u64>(micron::math::hw::sqrt_sd(f_sz) * lg * lg * 125.0);
124   float t_2 = (float)t / __system_pagesize;
125   t_2 = micron::math::ceil(t_2);
126   u64 pages = static_cast<u64>(t_2);
127   if ( pages < __default_minimum_page_mul ) pages = __default_minimum_page_mul;
128   pages = micron::math::nearest_pow2ll(pages);
129   return __saturate_pages_to_bytes(pages);
130 }

132 inline usize
133 __calculate_space_bulk(usize sz)
134 {
135   // logarithmic taper
136   long double factor = 1.0L + 0.1L * micron::math::logf128(static_cast<flong>(static_cast<double>(sz) / (1024 * 1024 * 1024)));
137   if ( factor < 1.0L ) factor = 1.0L;      // never shrink
138 
139   usize t = static_cast<usize>(sz * factor);
140 
141   usize pow2_sz = 1;
142   while ( pow2_sz < t ) {
143     if ( pow2_sz > (micron::numeric_limits<usize>::max() >> 1) ) break;      // next shift would wrap to 0 (32-bit)
144     pow2_sz <<= 1;
145   }
146 #if defined(__micron_arch_width_32)
147   if ( pow2_sz > static_cast<usize>(__width32_sheet_cap) ) pow2_sz = static_cast<usize>(__width32_sheet_cap);
148 #endif
149   return pow2_sz;
150 }

152 inline usize
153 __calculate_space_saturated(usize sz)
154 {
155   // x * ln(x) * (ln(ln(x)))
156   flong f_sz = static_cast<flong>(sz);
157   u64 t = static_cast<u64>(f_sz * micron::math::logf128(f_sz) * (micron::math::logf128(micron::math::logf128(f_sz))));
158   float t_2 = (float)t / __system_pagesize;
159   t_2 = micron::math::ceil(t_2);
160   // route through the saturating pages->bytes conversion: the raw multiply wraps on width-32
161   sz = __saturate_pages_to_bytes(
162       micron::math::nearest_pow2ll(((usize)t_2) < __default_minimum_page_mul ? __default_minimum_page_mul : (usize)t_2));
163   return sz;
164 }

172 template<typename T>
173 inline T
174 __get_kernel_chunk(u64 sz)
175 {
176   if ( auto *p = __va_carve(static_cast<usize>(sz)); p ) [[likely]] {
177     const usize rounded = (static_cast<usize>(sz) + __sheet_align_mask) & ~__sheet_align_mask;
178     return { reinterpret_cast<byte *>(p), rounded };
179   }
180   return { micron::sys_allocator<byte>::alloc(static_cast<usize>(sz)), static_cast<usize>(sz) };
181 }

183 template<typename T>
184 inline void
185 __release_kernel_chunk(const T &mem)
186 {
187   if ( __va_contains(mem.ptr) ) {
188     __va_release(micron::ptr_cast<addr_t *>(mem.ptr), mem.len);
189     return;
190   }
191   micron::sys_allocator<byte>::dealloc(mem.ptr, mem.len);
192 }
