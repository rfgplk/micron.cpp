64 extern "C" [[gnu::used]] inline int
65 posix_memalign(void **out, usize alignment, usize size) noexcept
66 {
67   if ( out == nullptr || alignment < sizeof(void *) || !bb::__bits::__is_pow2(alignment) ) return 22;
68   void *p = static_cast<void *>(bb::aligned_balloc(alignment, size).ptr);
69   if ( p == nullptr && size != 0 ) return 12;
70   *out = p;
71   return 0;
72 }

80 extern "C" [[gnu::used]] inline void *
81 pvalloc(usize size) noexcept
82 {
83   const usize n = bb::__bits::__round_up_sat(size, micron::page_size);
84   if ( n == bb::__bits::__usize_max ) return nullptr;
85   return static_cast<void *>(bb::aligned_balloc(micron::page_size, n).ptr);
86 }
