38 inline void
39 __write(const char *s, usize n) noexcept

49 inline usize
50 __strlen(const char *s) noexcept
51 {
52   usize n = 0;
53   while ( s[n] != '\0' ) ++n;
54   return n;
55 }
56 
57 inline void
58 __write(const char *s) noexcept
59 {
60   __write(s, __strlen(s));
61 }

63 inline void
64 __print_unsigned(u64 v) noexcept
65 {
66   char buf[24];
67   usize i = sizeof(buf);
68   do {
69     buf[--i] = static_cast<char>('0' + (v % 10));
70     v /= 10;
71   } while ( v != 0 );
72   __write(buf + i, sizeof(buf) - i);
73 }
74 
75 inline void
76 __print_ptr(const void *p) noexcept
77 {
78   char buf[2 + sizeof(void *) * 2];
79   buf[0] = '0';
80   buf[1] = 'x';
81   uintptr_t v = reinterpret_cast<uintptr_t>(p);
82   for ( usize i = sizeof(buf); i > 2; --i ) {
83     const unsigned d = static_cast<unsigned>(v & 0xF);
84     buf[i - 1] = static_cast<char>(d < 10 ? '0' + d : 'a' + d - 10);
85     v >>= 4;
86   }
87   __write(buf, sizeof(buf));
88 }

90 __bb_cold void
91 __report(const char *msg, const void *p) noexcept
92 {
93   __write("bbmalloc: ");
94   __write(msg);
95   __write(" at ");
96   __print_ptr(p);
97   __write("\n");
98 }

100 inline void
101 __notice(const char *msg, u64 v) noexcept
102 {
103   if constexpr ( __default_debug_notices ) {
104     __write("bbmalloc: ");
105     __write(msg);
106     __write(" ");
107     __print_unsigned(v);
108     __write("\n");
109   } else {
110     (void)msg;
111     (void)v;
112   }
113 }
