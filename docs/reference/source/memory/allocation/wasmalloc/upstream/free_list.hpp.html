180   __bb_hot void
181   __link_front(byte *addr, i32 o) noexcept
182   {
183     free_block *nb = __at(addr);
184     free_block *head = free_lists[o];
185     nb->next = head;
186     nb->prev = nullptr;
187     free_block *t = head != nullptr ? head : &__scratch;
188     t->prev = nb;
189     free_lists[o] = nb;
190     block_tags[tag_index(addr)] = static_cast<u8>(__tag_free | o);
191   }

200   __bb_hot bool
201   __unlink_check(const free_block *node, i32 o) const noexcept
202   {
203     if constexpr ( __check_links ) {
204       const free_block *next = node->next;
205       const bool is_head = free_lists[o] == node;
206       const free_block *prev = is_head ? nullptr : node->prev;
207       if ( !(__link_valid(next) & __link_valid(prev)) ) [[unlikely]]
208         return false;
209       const free_block *pv = prev != nullptr ? prev : &__scratch;
210       const free_block *nv = next != nullptr ? next : &__scratch;
211       const bool prev_ok = is_head | ((prev != nullptr) & (pv->next == node));
212       const bool next_ok = (next == nullptr) | (nv->prev == node);
213       return prev_ok & next_ok;
214     } else {
215       (void)node;
216       (void)o;
217       return true;
218     }
219   }

221   __bb_hot void
222   __unlink_do(free_block *node, i32 o) noexcept
223   {
224     free_block *next = node->next;
225     const bool is_head = free_lists[o] == node;
226     free_block *prev = is_head ? nullptr : node->prev;
227     free_block *pv = prev != nullptr ? prev : &__scratch;
228     free_block **slot = is_head ? &free_lists[o] : &pv->next;
229     *slot = next;
230     free_block *t = ((next != nullptr) & !is_head) ? next : &__scratch;
231     t->prev = prev;
232     free_mask &= ~(__bit(o) & __all_if(free_lists[o] == nullptr));
233   }

235   __bb_hot bool
236   __unlink(byte *addr, i32 o) noexcept
237   {
238     free_block *node = __at(addr);
239     if ( !__unlink_check(node, o) ) [[unlikely]]
240       return false;
241     __unlink_do(node, o);
242     return true;

245   __bb_hot free_block *
246   __take_and_split(i32 from, i32 target) noexcept
247   {
248     free_block *blk = free_lists[from];
249     if ( blk == nullptr ) [[unlikely]] {
250       free_mask &= ~__bit(from);
251       return nullptr;
252     }
253     free_block *next = blk->next;
254     if constexpr ( __check_links ) {
255       if ( !__link_valid(next) ) [[unlikely]]
256         next = nullptr;
257     }
258     free_lists[from] = next;
259     mask_t m = free_mask & ~(__bit(from) & __all_if(next == nullptr));
260     usize split = order_size(from);
261     byte *b = reinterpret_cast<byte *>(blk);
262     for ( i32 o = from; o > target; ) {
263       --o;
264       split >>= 1;
265       __link_front(b + split, o);
266     }
267     m |= (__bit(from) - 1) & ~(__bit(target) - 1);
268     free_mask = m;
269     return blk;
270   }

272   __bb_hot void
273   __merge_and_free(byte *addr, i32 o) noexcept
274   {
275     byte *const b = base;
276     u8 *const tags = block_tags;
277     const usize tot = total;
278     const i32 top = max_order - 1;
279     usize off = static_cast<usize>(addr - b);
280     while ( o < top ) {
281       const usize sz = order_size(o);
282       const usize buddy_off = off ^ sz;
283       if ( buddy_off >= tot ) break;
284       if ( tags[buddy_off >> __log2_min] != static_cast<u8>(__tag_free | o) ) break;
285       if ( !__unlink(b + buddy_off, o) ) [[unlikely]]
286         break;
287       tags[(off | sz) >> __log2_min] = __tag_none;
288       off &= ~sz;
289       ++o;
290     }
291     __push_free(b + off, o);
292   }

294   inline void
295   __zero_arrays() noexcept
296   {
297     free_mask = 0;
298     __scratch.next = nullptr;
299     __scratch.prev = nullptr;
300     for ( i32 i = 0; i < Mx; ++i ) free_lists[i] = nullptr;
301     if constexpr ( __cache_cap > 0 ) {
302       for ( i32 i = 0; i < Mx; ++i ) {
303         cache.head[i] = nullptr;
304         cache.count[i] = 0;
305       }
306     }
307   }

309   inline void
310   __seed() noexcept
311   {
312     const usize units = tag_count;
313 #if defined(BB_SEED_TOP_ONLY)
314     __push_free(base, max_order - 1);
315     (void)units;
316 #else
317     usize off = 0;
318     for ( i32 o = max_order - 1; o >= 0; --o ) {
319       if ( units & (static_cast<usize>(1) << o) ) {
320         __push_free(base + off, o);
321         off += order_size(o);
322       }
323     }
324 #endif
325   }

327   bool
328   __setup(bool zeroed) noexcept
329   {
330     base = nullptr;
331     total = 0;
332     max_order = 0;
333     allocated_bytes = 0;
334     base_align = 0;
335     block_tags = nullptr;
336     tag_count = 0;
337     __zero_arrays();
338     if ( raw_mem == nullptr || raw_len < 4 * Min ) return false;
339 
340     usize align = want_align < Min ? Min : want_align;
341     if ( !__bits::__is_pow2(align) ) align = static_cast<usize>(1) << __bits::__fls(align);
342     if ( align < Min ) align = Min;
343     const uintptr_t end = reinterpret_cast<uintptr_t>(raw_mem) + raw_len;
344     usize tag_area = 0;
345     u8 *tags_at = ext_tags;
346     if ( ext_tags == nullptr ) {
347       tag_area = (raw_len + Min) / (Min + 1);
348       tags_at = raw_mem;
349     }
350     const uintptr_t data = reinterpret_cast<uintptr_t>(raw_mem) + tag_area;
351     uintptr_t b = 0;
352     for ( ;; ) {
353       b = __bits::__align_up_ptr(data, align);
354       const usize need = align > 2 * Min ? align : 2 * Min;
355       if ( b >= data && b <= end && static_cast<usize>(end - b) >= need ) break;
356       if ( align <= Min ) return false;
357       align >>= 1;
358     }
359     base = reinterpret_cast<byte *>(b);
360     base_align = align;
361     usize t = static_cast<usize>(end - b);
362     if constexpr ( Mx < static_cast<i32>(sizeof(usize) * 8) ) {
363       const usize cap = ((static_cast<usize>(1) << Mx) - 1) * Min;
364       if ( t > cap ) t = cap;
365     }
366     if ( t > __max_total ) t = __max_total;
367     t = (t / Min) * Min;
368     usize count = t / Min;
369     if ( ext_tags != nullptr && count > ext_cap ) {
370       count = ext_cap;
371       t = count * Min;
372     }
373     if ( count == 0 ) return false;
374     total = t;
375     tag_count = count;
376     block_tags = tags_at;
377     max_order = __bits::__fls(count) + 1;
378     if ( !zeroed ) __builtin_memset(block_tags, __tag_none, tag_count);
379     __seed();
380     return true;
381   }

400   void
401   flush_cache() noexcept
402   {
403     if constexpr ( __cache_cap > 0 ) {
404       for ( i32 o = 0; o < max_order; ++o ) {
405         while ( cache.head[o] != nullptr ) {
406           free_block *blk = cache.head[o];
407           cache.head[o] = __link_valid(blk->next) ? blk->next : nullptr;
408           __merge_and_free(reinterpret_cast<byte *>(blk), o);
409         }
410         cache.count[o] = 0;
411       }
412     }
413   }

425   __bb_hot chunk_t
426   __allocate_order(i32 o) noexcept
427   {
428     if ( o >= max_order ) [[unlikely]]
429       return { nullptr, 0 };
430     if constexpr ( __cache_cap > 0 ) {
431       if ( o <= __cache_max_order && cache.head[o] != nullptr ) {
432         free_block *blk = cache.head[o];
433         free_block *nxt = __link_valid(blk->next) ? blk->next : nullptr;
434         cache.head[o] = nxt;
435         cache.count[o] = nxt == nullptr ? 0 : static_cast<u8>(cache.count[o] - 1);
436         return __grant(blk, o);
437       }
438     }
439     i32 i = find_free_order(o);
440     if ( i >= max_order ) [[unlikely]] {
441       if constexpr ( __cache_cap > 0 ) {
442         flush_cache();
443         i = find_free_order(o);
444         if ( i >= max_order ) return { nullptr, 0 };
445       } else {
446         return { nullptr, 0 };
447       }
448     }
449     free_block *blk = __take_and_split(i, o);
450     if ( blk == nullptr ) [[unlikely]]
451       return { nullptr, 0 };
452     return __grant(blk, o);
453   }

481   __order_at(const byte *p) const noexcept
482   {
483     const usize off = static_cast<usize>(reinterpret_cast<uintptr_t>(p) - reinterpret_cast<uintptr_t>(base));
484     if ( (off >= total) | ((off & (Min - 1)) != 0) ) [[unlikely]]
485       return -1;

487     const i32 o = t & __tag_order;

520   __bb_hot void
521   deallocate_order(byte *p, i32 o) noexcept
522   {
523     allocated_bytes -= order_size(o);
524     if constexpr ( __cache_cap > 0 ) {
525       if ( o <= __cache_max_order && cache.count[o] < __cache_cap ) {
526         free_block *blk = __at(p);
527         blk->next = cache.head[o];
528         blk->prev = nullptr;
529         cache.head[o] = blk;
530         ++cache.count[o];
531         block_tags[tag_index(p)] = __tag_none;
532         return;
533       }
534     }
535     __merge_and_free(p, o);
536   }

548   usize
549   grow(byte *p, usize n) noexcept
550   {
551     const i32 o = __order_at(p);
552     if ( o < 0 || n == 0 ) return 0;
553     const i32 want = order_for_size(n);
554     if ( want <= o ) return order_size(o);
555     if ( want >= max_order ) return 0;
556     const usize off = static_cast<usize>(p - base);
557     if ( (off & (order_size(want) - 1)) != 0 ) return 0;
558     for ( i32 j = o; j < want; ++j ) {
559       const usize boff = off + order_size(j);
560       if ( boff >= total || block_tags[boff >> __log2_min] != static_cast<u8>(__tag_free | j) ) return 0;
561       if ( !__unlink_check(__at(base + boff), j) ) return 0;
562     }
563     for ( i32 j = o; j < want; ++j ) {
564       const usize boff = off + order_size(j);
565       __unlink_do(__at(base + boff), j);
566       block_tags[boff >> __log2_min] = __tag_none;
567     }
568     block_tags[off >> __log2_min] = static_cast<u8>(__tag_alloc | want);
569     allocated_bytes += order_size(want) - order_size(o);
570     return order_size(want);
571   }
