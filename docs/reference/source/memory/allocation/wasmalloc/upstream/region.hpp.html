36 struct alignas(64) __region {
37   using buddy_t = __buddy_list<__default_min_block, __max_orders>;
38   using tlsf_t = __tlsf_list<__default_tlsf_sheet, __sl_log2, __max_tlsf_sheets, __class_small>;
39   using chunk_t = micron::__chunk<byte>;
40 
41   static constexpr usize __sheet = __default_tlsf_sheet;
42 
43   static_assert(__class_small + header_size + (__default_redzone ? __default_redzone_size : 0) <= tlsf_t::__data_bytes,
44                 "bb: the small class must fit inside one sheet");
45 
46   buddy_t buddy;
47   tlsf_t tlsf;
48   u32 spare;
49   bool live;
50 
51   bool
52   attach(byte *mem, usize len, u8 *tags, usize tag_cap, bool zeroed = false) noexcept
53   {
54     live = buddy.init(mem, len, tags, tag_cap, __default_max_alignment, zeroed);
55     spare = __null_off;
56     if ( live ) tlsf.init(buddy.base);
57     return live;
58   }
59 
60   void
61   reset() noexcept
62   {
63     if ( !live ) return;
64     buddy.reset();
65     tlsf.init(buddy.base);
66     spare = __null_off;
67   }
68 
69   __bb_slow void
70   flush_spare() noexcept
71   {
72     if ( spare == __null_off ) return;
73     const u32 s = spare;
74     spare = __null_off;
75     if ( buddy.is_sheet_at(s) && tlsf.sheet_empty(s) ) release_sheet(s);
76   }
77 
78   __bb_hot bool
79   contains(const void *p) const noexcept
80   {
81     return buddy.__in_range(p);
82   }
83 
84   inline bool
85   overlaps(const void *lo, usize len) const noexcept
86   {
87     if ( !live ) return false;
88     const uintptr_t a = reinterpret_cast<uintptr_t>(lo);
89     const uintptr_t b = a + len;
90     const uintptr_t r = reinterpret_cast<uintptr_t>(buddy.raw_mem);
91     const uintptr_t s = r + buddy.raw_len;
92     return a < s && r < b;
93   }
94 
95   inline u32
96   off_of(const void *p) const noexcept
97   {
98     return static_cast<u32>(reinterpret_cast<uintptr_t>(p) - reinterpret_cast<uintptr_t>(buddy.base));
99   }
100 
101   __bb_slow void
102   release_sheet(u32 sheet) noexcept
103   {
104     if ( !live || sheet == __null_off ) return;
105     byte *p = buddy.base + sheet;
106     if ( !buddy.__in_range(p) || !buddy.is_sheet_at(sheet) ) return;
107     if ( !tlsf.remove_sheet(sheet) ) return;
108     buddy.unmark_sheet(p);
109     if ( buddy.deallocate(p) != ret_flag::ok ) {
110       buddy.mark_sheet(p);
111       (void)tlsf.add_sheet(sheet);
112       return;
113     }
114     if ( spare == sheet ) spare = __null_off;
115     collect_stats<stat_type::sheet_released>();
116   }
117 
118   __bb_slow chunk_t
119   small_alloc_slow(usize n) noexcept
120   {
121     if ( tlsf.count >= __max_tlsf_sheets ) return { nullptr, 0 };
122     chunk_t s = buddy.allocate(__sheet);
123     if ( s.ptr == nullptr ) return { nullptr, 0 };
124     const u32 fresh = off_of(s.ptr);
125     if ( !tlsf.add_sheet(fresh) ) {
126       (void)buddy.deallocate(s.ptr);
127       return { nullptr, 0 };
128     }
129     buddy.mark_sheet(s.ptr);
130     collect_stats<stat_type::sheet_acquired>();
131     chunk_t c = tlsf.allocate(n);
132     if ( c.ptr == nullptr ) {
133       release_sheet(fresh);
134       return { nullptr, 0 };
135     }
136     return c;
137   }
138 
139   __bb_hot chunk_t
140   small_alloc(usize n) noexcept
141   {
142     const chunk_t c = tlsf.allocate(n);
143     if ( c.ptr == nullptr ) [[unlikely]]
144       return small_alloc_slow(n);
145     return c;
146   }
147 
148   __bb_slow void
149   __sheet_emptied(u32 sheet) noexcept
150   {
151     if ( spare == __null_off || !tlsf.sheet_empty(spare) ) {
152       spare = sheet;
153       return;
154     }
155     release_sheet(sheet);
156   }
157 
158   __bb_hot void
159   small_free(u32 hdr_off) noexcept
160   {
161     const u32 s = tlsf.deallocate(hdr_off);
162     if ( s != __null_off && s != spare ) __sheet_emptied(s);
163   }
164 
165   __bb_slow chunk_t
166   large_alloc_slow(usize n) noexcept
167   {
168     if ( spare == __null_off || !buddy.servable(n) ) return { nullptr, 0 };
169     flush_spare();
170     return buddy.allocate(n);
171   }
172 
173   __bb_hot chunk_t
174   large_alloc(usize n) noexcept
175   {
176     const chunk_t c = buddy.allocate(n);
177     if ( c.ptr == nullptr ) [[unlikely]]
178       return large_alloc_slow(n);
179     return c;
180   }
181 
182   chunk_t
183   large_alloc_aligned(usize n, usize alignment) noexcept
184   {
185     chunk_t c = buddy.allocate_aligned(n, alignment);
186     if ( c.ptr == nullptr && spare != __null_off && buddy.servable_aligned(n, alignment) ) {
187       flush_spare();
188       c = buddy.allocate_aligned(n, alignment);
189     }
190     return c;
191   }
192 
193   usize
194   available() const noexcept
195   {
196     if ( !live ) return 0;
197     const usize t = buddy.__total();
198     const usize u = used();
199     return u >= t ? 0 : t - u;
200   }
201 
202   usize
203   used() const noexcept
204   {
205     const usize b = buddy.used();
206     usize credit = static_cast<usize>(tlsf.count) * tlsf_t::__data_bytes;
207     if ( spare != __null_off && buddy.is_sheet_at(spare) && tlsf.sheet_empty(spare) ) credit += __sheet - tlsf_t::__data_bytes;
208     return (b < credit ? 0 : b - credit) + tlsf.allocated_bytes;
209   }
210 
211   usize
212   largest_free() noexcept
213   {
214     flush_spare();
215     const usize b = buddy.largest_free();
216     usize t = tlsf.largest_free();
217     if ( t > __class_small ) t = __class_small;
218     return b > t ? b : t;
219   }
220 };

222 };      // namespace bb
