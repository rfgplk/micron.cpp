35 struct __pool_desc {
36   byte *mem;
37   usize len;
38   u8 *tags;
39   usize tag_cap;
40   bool zeroed;
41 };
42 
43 #if defined(MICRON_BB_STATIC_POOL)
44 template<usize Id> struct __pool_store {
45   alignas(__default_max_alignment) static inline byte mem[MICRON_BB_STATIC_POOL];
46   static inline u8 tags[MICRON_BB_STATIC_POOL / __default_min_block];
47 };
48 
49 static_assert(MICRON_BB_STATIC_POOL >= 4 * __default_tlsf_sheet, "bb: static pool too small");
50 #endif
51 
52 #if defined(MICRON_BB_LINKER_POOL)
53 extern "C" byte __heap_start[];
54 extern "C" byte __heap_end[];
55 #endif
56 
57 inline __pool_desc
58 __configured_pool() noexcept
59 {
60 #if defined(MICRON_BB_STATIC_POOL)
61   using store = __pool_store<__abi_id>;
62   return { store::mem, sizeof(store::mem), store::tags, sizeof(store::tags), true };
63 #elif defined(MICRON_BB_LINKER_POOL)
64   const uintptr_t lo = reinterpret_cast<uintptr_t>(&__heap_start[0]);
65   const uintptr_t hi = reinterpret_cast<uintptr_t>(&__heap_end[0]);
66   if ( hi <= lo ) return { nullptr, 0, nullptr, 0, false };
67   return { __heap_start, static_cast<usize>(hi - lo), nullptr, 0, false };
68 #elif defined(MICRON_BB_PORT_POOL)
69   auto s = micron::port::page_alloc(static_cast<usize>(MICRON_BB_PORT_POOL));
70   if ( s.failed() ) return { nullptr, 0, nullptr, 0, false };
71   return { reinterpret_cast<byte *>(s.ptr), s.len, nullptr, 0, false };
72 #else
73   return { nullptr, 0, nullptr, 0, false };
74 #endif
75 }
76 
77 #if defined(MICRON_BB_PORT_GROW)
78 
79 constexpr bool __growable = true;
80 constexpr usize __grow_bytes = static_cast<usize>(MICRON_BB_PORT_GROW);
81 
82 static_assert(__grow_bytes >= 4 * __default_tlsf_sheet, "bb: MICRON_BB_PORT_GROW is below four sheets");
83 static_assert(__grow_bytes <= __default_max_region_bytes, "bb: MICRON_BB_PORT_GROW exceeds the region cap");
84 
85 inline __pool_desc
86 __grow_pool(usize bytes) noexcept
87 {
88   usize n = bytes < __grow_bytes ? __grow_bytes : bytes;
89   if ( n > __default_max_region_bytes ) n = __default_max_region_bytes;
90   if ( n < bytes ) return { nullptr, 0, nullptr, 0, false };
91   auto s = micron::port::page_alloc(n);
92   if ( s.failed() ) return { nullptr, 0, nullptr, 0, false };
93   return { reinterpret_cast<byte *>(s.ptr), s.len, nullptr, 0, false };
94 }
95 
96 #else
97 
98 constexpr bool __growable = false;
99 
100 inline __pool_desc
101 __grow_pool(usize) noexcept
102 {
103   return { nullptr, 0, nullptr, 0, false };
104 }
