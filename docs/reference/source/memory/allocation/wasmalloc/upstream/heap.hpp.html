46 struct __heap {
47   using chunk_t = micron::__chunk<byte>;
48   using buddy_t = __region::buddy_t;

51   u32 count;

54   __bb_cold bool
55   __clear_of_all(const void *lo, usize len) const noexcept
56   {
57     for ( u32 i = 0; i < count; ++i )
58       if ( regions[i].overlaps(lo, len) ) return false;
59     return true;
60   }

62   __bb_cold bool
63   __attach(byte *mem, usize len, u8 *tags, usize tag_cap, bool zeroed) noexcept
64   {
65     if ( count >= __max_regions || mem == nullptr || len == 0 ) return false;
66     const uintptr_t a = reinterpret_cast<uintptr_t>(mem);
67     if ( a + len < a ) return false;
68     if ( !__clear_of_all(mem, len) ) return false;
69     if ( tags != nullptr ) {
70       if ( tag_cap == 0 ) return false;
71       const uintptr_t t = reinterpret_cast<uintptr_t>(tags);
72       if ( t + tag_cap < t ) return false;
73       if ( t < a + len && a < t + tag_cap ) return false;
74       if ( !__clear_of_all(tags, tag_cap) ) return false;
75     }
76     if ( !regions[count].attach(mem, len, tags, tag_cap, zeroed) ) return false;
77     ++count;
78     inited = true;
79     return true;
80   }

82   bool
83   attach(byte *mem, usize len, u8 *tags, usize tag_cap) noexcept
84   {
85     return __attach(mem, len, tags, tag_cap, false);
86   }

88   __bb_cold bool
89   init() noexcept
90   {
91     if ( inited ) return count > 0;
92     inited = true;
93     const __pool_desc d = __configured_pool();
94     if ( d.mem != nullptr ) __attach(d.mem, d.len, d.tags, d.tag_cap, d.zeroed);
95     return count > 0;
96   }

98   inline bool
99   __ensure() noexcept
100   {
101     return inited ? count > 0 : init();
102   }

106   {

111   __bb_slow __region *
112   __region_slow(const void *p) noexcept
113   {
114     if ( !__ensure() ) return nullptr;
115     for ( u32 i = 0; i < count; ++i )
116       if ( regions[i].contains(p) ) return &regions[i];
117     return nullptr;

121   region_of(const void *p) noexcept
122   {
123     if ( regions[0].contains(p) ) [[likely]]
124       return &regions[0];

128   __bb_hot __located
129   locate(const void *p) noexcept
130   {
131     __region *r = region_of(p);
132     if ( r == nullptr ) [[unlikely]]
133       return { nullptr, 0, __null_off, tier::none };
134     const u32 off = r->off_of(p);
135     if ( r->buddy.is_sheet_at(off & ~static_cast<u32>(__region::__sheet - 1)) ) {
136       const auto o = r->tlsf.owns_len(p);
137       if ( o.hdr == __null_off ) [[unlikely]]
138         return { r, 0, __null_off, tier::none };
139       return { r, o.payload, o.hdr, tier::tlsf };
140     }
141     const i32 o = r->buddy.__order_at(static_cast<const byte *>(p));
142     if ( o < 0 ) [[unlikely]]
143       return { r, 0, __null_off, tier::none };
144     return { r, buddy_t::order_size(o), static_cast<u32>(o), tier::buddy };
145   }

147   __bb_cold bool
148   __grow(usize bytes) noexcept
149   {
150     if constexpr ( !__growable ) {
151       (void)bytes;
152       return false;
153     } else {
154       if ( count >= __max_regions ) return false;
155       usize want = __bits::__next_pow2(bytes);
156       if ( want <= __bits::__usize_max / 2 ) want *= 2;
157       want = __bits::__round_up_sat(want, __default_tlsf_sheet);
158       const __pool_desc d = __grow_pool(want);
159       if ( d.mem == nullptr ) return false;
160       return __attach(d.mem, d.len, d.tags, d.tag_cap, d.zeroed);
161     }
162   }

164   __bb_slow chunk_t
165   __alloc_scan(usize n) noexcept
166   {
167     chunk_t c{ nullptr, 0 };
168     if ( n <= __class_small ) {
169       c = regions[0].small_alloc_slow(n);
170       if constexpr ( __max_regions > 1 ) {
171         for ( u32 i = 1; i < count && c.ptr == nullptr; ++i ) c = regions[i].small_alloc(n);
172       }
173       if ( c.ptr != nullptr ) return c;
174       collect_stats<stat_type::fallback>();
175       c = regions[0].large_alloc(n);
176       if constexpr ( __max_regions > 1 ) {
177         for ( u32 i = 1; i < count && c.ptr == nullptr; ++i ) c = regions[i].large_alloc(n);
178       }
179       return c;
180     }
181     c = regions[0].large_alloc_slow(n);
182     if constexpr ( __max_regions > 1 ) {
183       for ( u32 i = 1; i < count && c.ptr == nullptr; ++i ) c = regions[i].large_alloc(n);
184     }
185     return c;
186   }

188   __bb_slow chunk_t
189   __alloc_slow(usize n) noexcept
190   {
191     if ( n == 0 || (!inited && !init()) ) return { nullptr, 0 };
192     chunk_t c = __alloc_scan(n);
193     if constexpr ( __growable ) {
194       if ( c.ptr == nullptr && __grow(n) ) c = __alloc_scan(n);
195     }
196     return c;
197   }

199   __bb_hot chunk_t
200   alloc(usize n) noexcept
201   {
202     if ( count == 0 ) [[unlikely]]
203       return __alloc_slow(n);
204     if ( n <= __class_small ) {
205       const chunk_t c = regions[0].tlsf.allocate(n);
206       if ( c.ptr != nullptr ) [[likely]]
207         return c;
208       return __alloc_slow(n);
209     }
210     const chunk_t c = regions[0].buddy.allocate(n);
211     if ( c.ptr != nullptr ) [[likely]]
212       return c;
213     return __alloc_slow(n);
214   }

216   __bb_slow chunk_t
217   __alloc_aligned_scan(usize n, usize alignment) noexcept
218   {
219     chunk_t c{ nullptr, 0 };
220     for ( u32 i = 0; i < count && c.ptr == nullptr; ++i ) c = regions[i].large_alloc_aligned(n, alignment);
221     return c;
222   }

224   __bb_slow chunk_t
225   __alloc_aligned_slow(usize n, usize alignment) noexcept
226   {
227     if ( !inited && !init() ) return { nullptr, 0 };
228     chunk_t c = __alloc_aligned_scan(n, alignment);
229     if constexpr ( __growable ) {
230       if ( c.ptr == nullptr && __grow(n > alignment ? n + alignment : alignment * 2) ) c = __alloc_aligned_scan(n, alignment);
231     }
232     return c;
233   }

235   __bb_hot chunk_t
236   alloc_aligned(usize n, usize alignment) noexcept
237   {
238     if ( n == 0 || !__bits::__is_pow2(alignment) ) [[unlikely]]
239       return { nullptr, 0 };
240     if ( alignment <= native_alignment ) return alloc(n);
241     if ( count == 0 ) [[unlikely]]
242       return __alloc_aligned_slow(n, alignment);
243     const chunk_t c = regions[0].buddy.allocate_aligned(n, alignment);
244     if ( c.ptr != nullptr ) [[likely]]
245       return c;
246     return __alloc_aligned_slow(n, alignment);
247   }

249   __bb_hot bool
250   release(const __located &l, void *p) noexcept
251   {
252     if ( l.t == tier::tlsf ) {
253       l.r->small_free(l.off);
254       return true;
255     }
256     if ( l.t == tier::buddy ) {
257       l.r->buddy.deallocate_order(static_cast<byte *>(p), static_cast<i32>(l.off));
258       return true;
259     }
260     return false;
261   }

287   usize
288   shrink_at(const __located &l, usize n) noexcept
289   {
290     if ( l.t == tier::tlsf ) return l.r->tlsf.shrink(l.off, n) ? l.r->tlsf.payload(l.off) : 0;
291     if ( l.t == tier::buddy ) return n <= l.len ? l.len : 0;
292     return 0;
293   }

295   usize
296   grow_at(const __located &l, void *p, usize n) noexcept
297   {
298     if ( l.t == tier::tlsf ) {
299       if ( n > __class_small ) return 0;
300       return l.r->tlsf.grow(l.off, n) ? l.r->tlsf.payload(l.off) : 0;
301     }
302     if ( l.t == tier::buddy ) return l.r->buddy.grow(static_cast<byte *>(p), n);
303     return 0;
304   }
