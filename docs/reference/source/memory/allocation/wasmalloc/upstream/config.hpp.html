112 constexpr static const i32 __max_orders = []() constexpr {
113   i32 r = 0;
114   usize v = __default_max_region_bytes / __default_min_block;
115   while ( v > 0 ) {
116     v >>= 1;
117     ++r;
118   }
119   return r;
120 }();

122 constexpr static const usize __abi_id = []() constexpr {
123   usize h = static_cast<usize>(0x9E3779B97F4A7C15ull);
124   const usize k[] = { __default_min_block,
125                       __default_max_region_bytes,
126                       static_cast<usize>(__max_regions),
127                       __class_small,
128                       __default_tlsf_sheet,
129                       static_cast<usize>(__max_tlsf_sheets),
130                       static_cast<usize>(__sl_log2),
131                       __default_max_alignment,
132                       static_cast<usize>(__default_order_cache_cap),
133                       static_cast<usize>(__max_orders),
134                       __default_redzone ? __default_redzone_size : 0 };
135   for ( usize i = 0; i < sizeof(k) / sizeof(k[0]); ++i ) {
136     h ^= k[i] + static_cast<usize>(0x165667B19E3779F9ull) + (h << 6) + (h >> 2);
137   }
138   return h;
139 }();
