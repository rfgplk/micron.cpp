36 template<typename T>
37   requires(micron::is_integral_v<T>)
38 struct __bb_allocator {
39   static auto
40   calloc(usize n) -> micron::__chunk<byte>
41   {
42     auto mem = bb::zalloc(n);
43     if ( mem.ptr == nullptr ) micron::exc<micron::except::critical_error>("bb_allocator::calloc(): pool failed to satisfy request");
44     return mem;
45   }
46 
47   static T *
48   alloc(usize n)
49   {
50     T *ptr = reinterpret_cast<T *>(bb::alloc(n));
51     if ( ptr == nullptr ) micron::exc<micron::except::critical_error>("bb_allocator::alloc(): pool failed to satisfy request");
52     return ptr;
53   }
54 
55   static auto
56   allocate_aligned(usize n, usize alignment) -> micron::__chunk<byte>
57   {
58     auto mem = bb::aligned_balloc(alignment, n);
59     if ( mem.ptr == nullptr )
60       micron::exc<micron::except::critical_error>("bb_allocator::allocate_aligned(): pool failed to satisfy request");
61     return mem;
62   }
63 
64   static void
65   dealloc(T *mem, usize)
66   {
67     if ( mem == nullptr ) micron::exc<micron::except::critical_error>("bb_allocator::dealloc(): nullptr was provided");
68     bb::dealloc(reinterpret_cast<byte *>(mem));
69   }
70 
71   static void
72   dealloc(T *mem)
73   {
74     if ( mem == nullptr ) return;
75     bb::dealloc(reinterpret_cast<byte *>(mem));
76   }
77 
78   static void
79   dealloc_aligned(T *mem, usize alignment)
80   {
81     if ( mem == nullptr ) return;
82     bb::aligned_dealloc(reinterpret_cast<byte *>(mem), alignment);
83   }

86 class bb_allocator
87 {
88 public:
89   static constexpr bool allocator_trusted = true;
90   using chunk_t = micron::__chunk<byte>;
91 
92   static constexpr usize
93   auto_size() noexcept
94   {
95     return 64;
96   }
97 
98   static constexpr usize
99   allocation_extent(usize bytes, usize alignment) noexcept
100   {
101     if ( bytes == 0 ) return 0;
102     if ( __bits::__usize_max - __rz < bytes ) return __bits::__usize_max;
103     const usize want = bytes + __rz;
104     if ( alignment <= native_alignment && want <= __class_small ) {
105       const usize r = __bits::__round_up_sat(want, 16);
106       return (r < 16 ? 16 : r) - __rz;
107     }
108     usize v = want < __default_min_block ? __default_min_block : want;
109     if ( v <= 1 ) return alignment > 1 ? alignment : 1;
110     i32 top = 0;
111     usize t = v - 1;
112     while ( t > 0 ) {
113       t >>= 1;
114       ++top;
115     }
116     if ( top >= static_cast<i32>(sizeof(usize) * 8) ) return __bits::__usize_max;
117     const usize p = static_cast<usize>(1) << top;
118     return (p < alignment ? alignment : p) - __rz;
119   }
120 
121   static constexpr usize
122   allocation_extent(usize bytes) noexcept
123   {
124     return allocation_extent(bytes, native_alignment);
125   }
126 
127   static chunk_t
128   create(usize bytes, usize alignment)
129   {
130     if ( bytes == 0 ) return { nullptr, 0 };
131     chunk_t c = alignment <= native_alignment ? bb::balloc(bytes) : bb::aligned_balloc(alignment, bytes);
132     if ( c.ptr == nullptr ) micron::exc<micron::except::memory_error>("bb_allocator::create(): pool exhausted");
133     return { c.ptr, allocation_extent(bytes, alignment) };
134   }
135 
136   template<usize Alignment>
137   static chunk_t
138   create(usize bytes)
139   {
140     return create(bytes, Alignment);
141   }
142 
143   static chunk_t
144   create(usize bytes)
145   {
146     return create(bytes, native_alignment);
147   }
148 
149   template<usize Alignment>
150   static chunk_t
151   resize(chunk_t old, usize bytes, usize preserve)
152   {
153     if ( bytes == 0 ) {
154       destroy(old, Alignment);
155       return { nullptr, 0 };
156     }
157     chunk_t c = bb::resize(old, bytes, preserve, Alignment);
158     if ( c.ptr == nullptr ) micron::exc<micron::except::memory_error>("bb_allocator::resize(): pool exhausted");
159     return { c.ptr, allocation_extent(bytes, Alignment) };
160   }
161 
162   static chunk_t
163   resize(chunk_t old, usize bytes, usize preserve, usize alignment)
164   {
165     if ( bytes == 0 ) {
166       destroy(old, alignment);
167       return { nullptr, 0 };
168     }
169     chunk_t c = bb::resize(old, bytes, preserve, alignment);
170     if ( c.ptr == nullptr ) micron::exc<micron::except::memory_error>("bb_allocator::resize(): pool exhausted");
171     return { c.ptr, allocation_extent(bytes, alignment) };
172   }
173 
174   static void
175   destroy(chunk_t memory, usize) noexcept
176   {
177     if ( memory.ptr != nullptr ) bb::dealloc(static_cast<void *>(memory.ptr));
178   }
179 
180   static void
181   destroy(chunk_t memory) noexcept
182   {
183     if ( memory.ptr != nullptr ) bb::dealloc(static_cast<void *>(memory.ptr));
184   }
185 
186   static void
187   destroy(byte *ptr, usize) noexcept
188   {
189     if ( ptr != nullptr ) bb::dealloc(static_cast<void *>(ptr));
190   }
191 
192   template<usize Alignment>
193   static void
194   destroy(byte *ptr) noexcept
195   {
196     if ( ptr != nullptr ) bb::dealloc(static_cast<void *>(ptr));
197   }
198 
199   static constexpr usize
200   recommend(usize current, usize minimum) noexcept
201   {
202     usize next = current > __bits::__usize_max / 2 ? __bits::__usize_max : current * 2;
203     if ( next < minimum ) next = minimum;
204     if ( next < auto_size() ) next = auto_size();
205     return next;
206   }

208   static void
209   reset() noexcept
210   {
211     bb::reset();
212   }
