79 constexpr inline i32
80 __log2(usize v) noexcept
81 {
82   i32 r = 0;
83   while ( v > 1 ) {
84     v >>= 1;
85     ++r;
86   }
87   return r;
88 }

90 [[gnu::always_inline]] inline i32
91 __fls(usize v) noexcept
92 {
93   if constexpr ( sizeof(usize) == 8 )
94     return 63 - __builtin_clzll(static_cast<unsigned long long>(v));
95   else
96     return 31 - __builtin_clz(static_cast<unsigned int>(v));
97 }
98 
99 [[gnu::always_inline]] inline i32
100 __fls32(u32 v) noexcept
101 {
102   return 31 - __builtin_clz(v);
103 }

105 [[gnu::always_inline]] inline i32
106 __ctz32(u32 v) noexcept

109 }

111 [[gnu::always_inline]] inline i32

117 [[gnu::always_inline]] inline usize
118 __next_pow2(usize v) noexcept
119 {
120   if ( v <= 1 ) return 1;
121   const i32 top = __fls(v - 1) + 1;
122   if ( top >= static_cast<i32>(sizeof(usize) * 8) ) return __usize_max;
123   return static_cast<usize>(1) << top;
124 }

126 [[gnu::always_inline]] inline usize
127 __round_up_sat(usize v, usize a) noexcept
128 {
129   const usize r = (v + a - 1) & ~(a - 1);
130   return r < v ? __usize_max : r;
131 }
