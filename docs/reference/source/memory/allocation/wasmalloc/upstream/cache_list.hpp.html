40   static constexpr usize __block_align = 16;
41   static constexpr usize __min_block = 32;
42   static constexpr i32 __fl_shift = 5;
43   static constexpr i32 __sl_count = 1 << SlLog2;
44   static constexpr usize __bitmap_bytes = ((Sheet / 128) + 15) & ~static_cast<usize>(15);
45   static constexpr usize __data_off = header_size + __bitmap_bytes;
46   static constexpr usize __data_bytes = Sheet - __data_off - header_size;
47   static constexpr i32 __fl_count = __bits::__log2(__data_bytes) - __fl_shift + 1;
48   static constexpr i32 __list_count = __fl_count * __sl_count;
49   static constexpr usize __max_payload = __data_bytes - header_size;
50   static constexpr usize __table_max = ((SmallMax + 31) & ~static_cast<usize>(15)) + 16;

53   static_assert(__fl_count >= 1 && __fl_count <= 27, "bb: derived first-level count out of range");

73   static constexpr __search_table
74   __make_table() noexcept
75   {
76     __search_table t{};
77     for ( usize i = 0; i < __table_n; ++i ) {
78       const u32 size = static_cast<u32>(i << 4);
79       if ( size < __min_block ) {
80         t.fl[i] = 0;
81         t.sl[i] = 0;
82         continue;
83       }
84       const i32 f = __bits::__log2(size);
85       const u32 r = size + (1u << (f - SlLog2)) - 1;
86       const i32 f2 = __bits::__log2(r);
87       t.fl[i] = static_cast<u8>(f2 - __fl_shift);
88       t.sl[i] = static_cast<u8>((r >> (f2 - SlLog2)) & static_cast<u32>(__sl_count - 1));
89     }
90     return t;
91   }

160   static inline void
161   mapping_insert(u32 size, i32 &fl, i32 &sl) noexcept
162   {
163     fl = __bits::__fls32(size) - __fl_shift;
164     sl = static_cast<i32>((size >> (fl + __fl_shift - SlLog2)) & static_cast<u32>(__sl_count - 1));
165   }

167   static inline void
168   mapping_search(u32 size, i32 &fl, i32 &sl) noexcept
169   {
170     const i32 f = __bits::__fls32(size);
171     const u32 r = size + (1u << (f - SlLog2)) - 1;
172     const i32 f2 = f + static_cast<i32>(r >> (f + 1));
173     fl = f2 - __fl_shift;
174     sl = static_cast<i32>((r >> (f2 - SlLog2)) & static_cast<u32>(__sl_count - 1));
175   }

177   __hit
178   __find_exact(u32 size) const noexcept
179   {
180     i32 fl = 0;
181     i32 sl = 0;
182     mapping_insert(size, fl, sl);
183     if ( fl >= __fl_count ) return { __null_off, 0 };
184     const u32 idx = static_cast<u32>(fl) * __sl_count + static_cast<u32>(sl);
185     const u32 off = heads[idx];
186     if ( off != __null_off && bsize(hdr(off)) >= size ) return { off, idx };
187     return { __null_off, 0 };
188   }

190   __bb_hot __hit
191   find_free(u32 size) const noexcept
192   {
193     i32 fl = 0;
194     i32 sl = 0;
195     if ( size <= __table_max ) [[likely]] {
196       fl = __search.fl[size >> 4];
197       sl = __search.sl[size >> 4];
198     } else {
199       mapping_search(size, fl, sl);
200       if ( fl >= __fl_count ) [[unlikely]]
201         return __find_exact(size);
202     }
203     const u32 m = sl_bitmap[fl] & (~0u << sl);
204     if ( m != 0 ) [[likely]] {
205       const u32 idx = static_cast<u32>(fl) * __sl_count + static_cast<u32>(__bits::__ctz32(m));
206       return { heads[idx], idx };
207     }
208     const u32 fm = fl_bitmap & (~0u << (fl + 1));
209     if ( fm != 0 ) {
210       const i32 hf = __bits::__ctz32(fm);
211       const u32 idx = static_cast<u32>(hf) * __sl_count + static_cast<u32>(__bits::__ctz32(sl_bitmap[hf]));
212       return { heads[idx], idx };
213     }
214     return __find_exact(size);
215   }

217   __bb_hot void
218   __clear_if_empty(u32 idx, u32 head_now) noexcept
219   {
220     const i32 fl = static_cast<i32>(idx >> SlLog2);
221     const i32 sl = static_cast<i32>(idx & static_cast<u32>(__sl_count - 1));
222     const u32 drop = head_now == __null_off ? (1u << sl) : 0u;
223     const u32 s = sl_bitmap[fl] & ~drop;
224     sl_bitmap[fl] = s;
225     const u32 fdrop = s == 0 ? (1u << fl) : 0u;
226     fl_bitmap &= ~fdrop;
227   }

229   __bb_hot void
230   insert_free(u32 off, u32 bs) noexcept
231   {
232     byte *const rb = rbase;
233     i32 fl = 0;
234     i32 sl = 0;
235     mapping_insert(bs, fl, sl);
236     const u32 idx = static_cast<u32>(fl) * __sl_count + static_cast<u32>(sl);
237     tlsf_hdr *h = hdr_at(rb, off);
238     const u32 head = heads[idx];
239     h->next_free = head;
240     const u32 hs = head != __null_off ? head : 0;
241     tlsf_hdr *t = head != __null_off ? hdr_at(rb, hs) : &__scratch;
242     t->prev_free = off;
243     heads[idx] = off;
244     sl_bitmap[fl] |= 1u << sl;
245     fl_bitmap |= 1u << fl;
246   }

248   __bb_hot void
249   remove_free(u32 off, u32 bs) noexcept
250   {
251     byte *const rb = rbase;
252     i32 fl = 0;
253     i32 sl = 0;
254     mapping_insert(bs, fl, sl);
255     const u32 idx = static_cast<u32>(fl) * __sl_count + static_cast<u32>(sl);
256     tlsf_hdr *h = hdr_at(rb, off);
257     const u32 nx = h->next_free;
258     const u32 pv = h->prev_free;
259     const u32 head = heads[idx];
260     const bool is_head = head == off;
261     const u32 pvs = is_head ? 0 : pv;
262     u32 *slot = is_head ? &heads[idx] : &hdr_at(rb, pvs)->next_free;
263     *slot = nx;
264     const bool relink = (nx != __null_off) & !is_head;
265     const u32 nxs = relink ? nx : 0;
266     tlsf_hdr *t = relink ? hdr_at(rb, nxs) : &__scratch;
267     t->prev_free = pv;
268     __clear_if_empty(idx, is_head ? nx : head);
269   }

272   adjusted(usize n) noexcept

279   __bb_hot u8 *
280   __mark_byte(u32 off, u32 &bit) const noexcept
281   {
282     const u32 rel = off & static_cast<u32>(Sheet - 1);
283     const u32 bi = (rel >> 4) + 1;
284     bit = 1u << (bi & 7);
285     return rbase + (off - rel) + header_size + (bi >> 3);

311   __bb_hot chunk_t
312   allocate(usize n) noexcept
313   {
314     const usize adj = adjusted(n);
315     if ( adj == 0 ) [[unlikely]]
316       return { nullptr, 0 };
317     byte *const rb = rbase;
318     const __hit f = find_free(static_cast<u32>(adj));
319     if ( f.off == __null_off ) [[unlikely]]
320       return { nullptr, 0 };
321     tlsf_hdr *h = hdr_at(rb, f.off);
322     const u32 bs = bsize(h);
323     const u32 nx = h->next_free;
324     heads[f.idx] = nx;
325     __clear_if_empty(f.idx, nx);
326     u32 take = static_cast<u32>(adj);
327     if ( bs - take >= __min_block ) {
328       const u32 rem = f.off + take;
329       tlsf_hdr *rh = hdr_at(rb, rem);
330       rh->bsize_flags = bs - take;
331       rh->prev_phys = f.off;
332       hdr_at(rb, f.off + bs)->prev_phys = rem;
333       insert_free(rem, bs - take);
334     } else {
335       take = bs;
336     }
337     h->bsize_flags = take | __tlsf_alloc;
338     __mark(f.off);
339     allocated_bytes += take;
340     return { rb + f.off + header_size, take - header_size };
341   }

343   __bb_hot __own
344   owns_len(const void *p) const noexcept
345   {
346     const u32 up = static_cast<u32>(reinterpret_cast<uintptr_t>(p) - reinterpret_cast<uintptr_t>(rbase));
347     const u32 sheet = sheet_of(up);
348     const u32 rel = up - sheet;
349     constexpr u32 lo = static_cast<u32>(__data_off + header_size);
350     constexpr u32 span = static_cast<u32>(Sheet - header_size) - lo;
351     if ( ((rel - lo) >= span) | ((rel & (__block_align - 1)) != 0) ) [[unlikely]]
352       return { __null_off, 0 };
353     const tlsf_hdr *h = hdr(up - static_cast<u32>(header_size));
354     const u32 f = h->bsize_flags;
355     const u32 bs = f & ~__tlsf_flags;
356     const bool ok = __marked(sheet, rel) & ((f & (__tlsf_alloc | __tlsf_sentinel)) == __tlsf_alloc) & (bs >= __min_block)
357                     & (rel - static_cast<u32>(header_size) + bs <= Sheet - header_size);
358     if ( !ok ) [[unlikely]]
359       return { __null_off, 0 };
360     return { up - static_cast<u32>(header_size), bs - static_cast<u32>(header_size) };
361   }

375   __bb_hot u32
376   deallocate(u32 off) noexcept
377   {
378     byte *const rb = rbase;
379     tlsf_hdr *h = hdr_at(rb, off);
380     u32 bs = bsize(h);
381     __unmark(off);
382     allocated_bytes -= bs;
383     const u32 next = off + bs;
384     tlsf_hdr *nh = hdr_at(rb, next);
385     if ( is_free(nh) ) {
386       const u32 nb = bsize(nh);
387       remove_free(next, nb);
388       bs += nb;
389     }
390     const u32 prev = h->prev_phys;
391     tlsf_hdr *ph = hdr_at(rb, prev);
392     if ( is_free(ph) ) {
393       const u32 pb = bsize(ph);
394       remove_free(prev, pb);
395       bs += pb;
396       off = prev;
397       h = ph;
398     }
399     h->bsize_flags = bs;
400     hdr_at(rb, off + bs)->prev_phys = off;
401     insert_free(off, bs);
402     return bs == static_cast<u32>(__data_bytes) ? sheet_of(off) : __null_off;
403   }

405   bool
406   shrink(u32 off, usize n) noexcept
407   {
408     const usize adj = adjusted(n);
409     if ( adj == 0 ) return false;
410     byte *const rb = rbase;
411     tlsf_hdr *h = hdr_at(rb, off);
412     const u32 bs = bsize(h);
413     if ( adj > bs ) return false;
414     if ( bs - adj < __min_block ) return true;
415     const u32 take = static_cast<u32>(adj);
416     const u32 rem = off + take;
417     tlsf_hdr *rh = hdr_at(rb, rem);
418     u32 rest = bs - take;
419     rh->prev_phys = off;
420     h->bsize_flags = take | __tlsf_alloc;
421     allocated_bytes -= rest;
422     const u32 next = off + bs;
423     tlsf_hdr *nh = hdr_at(rb, next);
424     if ( is_free(nh) ) {
425       const u32 nb = bsize(nh);
426       remove_free(next, nb);
427       rest += nb;
428     }
429     rh->bsize_flags = rest;
430     hdr_at(rb, rem + rest)->prev_phys = rem;
431     insert_free(rem, rest);
432     return true;
433   }

435   bool
436   grow(u32 off, usize n) noexcept
437   {
438     const usize adj = adjusted(n);
439     if ( adj == 0 ) return false;
440     byte *const rb = rbase;
441     tlsf_hdr *h = hdr_at(rb, off);
442     const u32 bs = bsize(h);
443     if ( adj <= bs ) return true;
444     const u32 next = off + bs;
445     tlsf_hdr *nh = hdr_at(rb, next);
446     if ( !is_free(nh) ) return false;
447     const u32 nb = bsize(nh);
448     const u32 span = bs + nb;
449     if ( span < adj ) return false;
450     remove_free(next, nb);
451     const u32 take = static_cast<u32>(adj);
452     const u32 rest = span - take;
453     if ( rest >= __min_block ) {
454       const u32 rem = off + take;
455       tlsf_hdr *rh = hdr_at(rb, rem);
456       rh->bsize_flags = rest;
457       rh->prev_phys = off;
458       hdr_at(rb, off + span)->prev_phys = rem;
459       h->bsize_flags = take | __tlsf_alloc;
460       allocated_bytes += take - bs;
461       insert_free(rem, rest);
462     } else {
463       hdr_at(rb, off + span)->prev_phys = off;
464       h->bsize_flags = span | __tlsf_alloc;
465       allocated_bytes += span - bs;
466     }
467     return true;
468   }

477   bool
478   add_sheet(u32 sheet) noexcept
479   {
480     if ( count >= MaxSheets || rbase == nullptr ) return false;
481     tlsf_hdr *start = hdr(sheet);
482     start->bsize_flags = static_cast<u32>(__data_off) | __tlsf_alloc | __tlsf_sentinel;
483     start->prev_phys = __null_off;
484     start->next_free = 0;
485     start->prev_free = 0;
486     __builtin_memset(bitmap(sheet), 0, __bitmap_bytes);
487     tlsf_hdr *end = hdr(sheet + static_cast<u32>(Sheet - header_size));
488     end->bsize_flags = static_cast<u32>(header_size) | __tlsf_alloc | __tlsf_sentinel;
489     end->prev_phys = sheet + static_cast<u32>(__data_off);
490     end->next_free = __null_off;
491     end->prev_free = __null_off;
492     tlsf_hdr *data = hdr(sheet + static_cast<u32>(__data_off));
493     data->bsize_flags = static_cast<u32>(__data_bytes);
494     data->prev_phys = sheet;
495     ++count;
496     insert_free(sheet + static_cast<u32>(__data_off), static_cast<u32>(__data_bytes));
497     return true;
498   }
499 
500   bool
501   remove_sheet(u32 sheet) noexcept
502   {
503     if ( count == 0 || !sheet_empty(sheet) ) return false;
504     remove_free(sheet + static_cast<u32>(__data_off), static_cast<u32>(__data_bytes));
505     --count;
506     return true;
507   }

509   usize
510   largest_free() const noexcept
511   {
512     if ( fl_bitmap == 0 ) return 0;
513     const i32 fl = __bits::__fls32(fl_bitmap);
514     if ( fl >= __fl_count || sl_bitmap[fl] == 0 ) return 0;
515     const i32 sl = __bits::__fls32(sl_bitmap[fl]);
516     const u32 off = heads[fl * __sl_count + sl];
517     if ( off == __null_off ) return 0;
518     const u32 bs = bsize(hdr(off));
519     return bs <= header_size ? 0 : bs - header_size;
520   }
521 
522   u32
523   free_block_count() const noexcept
524   {
525     u32 c = 0;
526     for ( i32 i = 0; i < __list_count; ++i )
527       for ( u32 o = heads[i]; o != __null_off && c <= 0xFFFFFF; o = hdr(o)->next_free ) ++c;
528     return c;
529   }
