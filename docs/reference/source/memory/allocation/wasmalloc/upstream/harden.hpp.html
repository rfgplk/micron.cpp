52 [[gnu::always_inline]] inline void
53 zero_on_alloc(byte *p, usize n) noexcept
54 {
55   if constexpr ( __default_zero_on_alloc )
56     __builtin_memset(p, 0, n);
57   else {
58     (void)p;
59     (void)n;
60   }
61 }

63 [[gnu::always_inline]] inline void
64 scrub_on_free(byte *p, usize n) noexcept
65 {
66   if constexpr ( __default_zero_on_free )
67     __builtin_memset(p, 0, n);
68   else if constexpr ( __default_poison_on_free )
69     __builtin_memset(p, __default_poison_byte, n);
70   else {
71     (void)p;
72     (void)n;
73   }
74 }

76 constexpr static const usize __rz = __default_redzone ? __default_redzone_size : 0;

78 [[gnu::always_inline]] inline void
79 write_redzone(byte *p, usize user) noexcept
80 {
81   if constexpr ( __default_redzone )
82     __builtin_memset(p + user, __default_redzone_byte, __rz);
83   else {
84     (void)p;
85     (void)user;
86   }
87 }
88 
89 [[gnu::always_inline]] inline bool
90 verify_redzone(const byte *p, usize user) noexcept
91 {
92   if constexpr ( __default_redzone ) {
93     for ( usize i = 0; i < __rz; ++i )
94       if ( p[user + i] != __default_redzone_byte ) return false;
95     return true;
96   } else {
97     (void)p;
98     (void)user;
99     return true;
100   }
101 }

103 __bb_cold bool
104 handle_bad_free(const void *p) noexcept
105 {
106   if constexpr ( __default_double_free_action == 0 ) {
107     (void)p;
108     return false;
109   } else if constexpr ( __default_double_free_action == 1 ) {
110     __bits::__report("bad or double free", p);
111     return false;
112   } else {
113     __bits::__report("bad or double free", p);
114     __bits::__halt(__critical_exit);
115   }
116 }
117 
118 __bb_cold bool
119 handle_redzone_trip(const void *p) noexcept
120 {
121   if constexpr ( __default_redzone_action == 0 ) {
122     (void)p;
123     return false;
124   } else if constexpr ( __default_redzone_action == 1 ) {
125     __bits::__report("redzone overrun", p);
126     return false;
127   } else {
128     __bits::__report("redzone overrun", p);
129     __bits::__halt(__critical_exit);
130   }
131 }
