49 [[nodiscard]] inline bool
50 __grow(usize bytes, usize alignment) noexcept
51 {
52   if ( __region_count >= bb::__max_regions ) return false;
53   if ( alignment > bb::__default_max_alignment || bytes > static_cast<usize>(-1) - bb::__rz ) return false;
54   bytes += bb::__rz;
55   usize want = bytes > alignment ? bytes : alignment;
56   if ( want > bb::__default_max_region_bytes ) return false;
57   usize region = __next_region_bytes;
58   while ( region < want ) {
59     if ( region > bb::__default_max_region_bytes / 2 ) return false;
60     region *= 2;
61   }
62   usize tags = 0;
63   byte *base = nullptr;
64   for ( ;; ) {
65     tags = region / bb::__default_min_block;
66     if ( tags > static_cast<usize>(-1) - region ) return false;
67     const usize base_alignment = region < bb::__default_max_alignment ? region : bb::__default_max_alignment;
68     const usize place = alignment > base_alignment ? alignment : base_alignment;
69     base = static_cast<byte *>(wasm::linear_reserve(region + tags, place));
70     if ( base != nullptr ) break;
71     if ( region / 2 < want || region / 2 < 4 * bb::__default_tlsf_sheet ) return false;
72     region /= 2;
73   }
74   if ( !bb::attach(base, region, reinterpret_cast<u8 *>(base + region), tags) ) __builtin_trap();
75   ++__region_count;
76   __next_region_bytes = region <= bb::__default_max_region_bytes / 2 ? region * 2 : region;
77   return true;
78 }

80 [[nodiscard]] inline chunk<byte>
81 __allocate(usize bytes, usize alignment) noexcept
82 {
83   if ( bytes == 0 || alignment == 0 || (alignment & (alignment - 1)) != 0 ) return { nullptr, 0 };
84   auto result = bb::aligned_balloc(alignment, bytes);
85   if ( result.ptr == nullptr && __grow(bytes, alignment) ) result = bb::aligned_balloc(alignment, bytes);
86   return result;
87 }

226 [[nodiscard]] inline chunk<byte>
227 resize(chunk<byte> old, usize bytes, usize preserve, usize alignment) noexcept
228 {
229   if ( alignment == 0 || (alignment & (alignment - 1)) != 0 ) return { nullptr, 0 };
230   wasm::__lock_guard lock{ __heap_mutex };
231   auto result = bb::resize(old, bytes, preserve, alignment);
232   if ( bytes != 0 && result.ptr == nullptr && __grow(bytes, alignment) ) result = bb::resize(old, bytes, preserve, alignment);
233   return result;
234 }
