22 template<typename T = byte> struct alignas(16) __chunk {      // total memory allocated
23   T *ptr;
24   usize len;
25 
26   __chunk &
27   operator=(nullptr_t t [[maybe_unused]])
28   {
29     ptr = nullptr;
30     len = 0;
31     return *this;
32   }
33 
34   bool
35   zero(void) const noexcept
36   {
37     if ( ptr == nullptr or len == 0 ) return true;
38     return false;
39   }
40 
41   bool
42   failed_allocation(void) const noexcept
43   {
44     return ((ptr == micron::numeric_limits<byte *>::max()) and len == 0xFF);
45   }
46 
47   bool
48   invalid(void) const noexcept
49   {
50     return (ptr == (byte *)-1);
51   }
52 };
53 
54 template<typename T> using chunk = micron::__chunk<T>;
55 
56 template<typename T>
57 chunk<byte>
58 to_chunk(T *ptr, usize len)
59 {
60   return chunk<byte>(reinterpret_cast<byte *>(ptr), len);
61 }
