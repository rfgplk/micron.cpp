22 [[gnu::noinline]] static __micron_optimize_no_tree_loop_distribute byte *
23 __memmove_large(byte *d, const byte *s, const u64 n) noexcept
24 {
25   if ( d < s || d >= s + n ) {
26 #if defined(__micron_arch_x86_any)
27     if ( n < __mem_rep_min ) return simd::__memmove_bulk_fwd(d, s, n);
28     const __mem_tunables &t = __mem_tun_get();
29     const bool disjoint = (d + n <= s) || (s + n <= d);
30     if ( disjoint && n >= t.nt_copy_threshold ) return simd::__memcpy_bulk_nt(d, s, n);
31     if ( n >= t.rep_movsb_threshold && n < t.nt_copy_threshold ) {
32       simd::__bits::__rep_movsb(d, s, static_cast<usize>(n));
33       return d;
34     }
35 #elif defined(__micron_arch_arm64)
36     if ( ((d + n <= s) || (s + n <= d)) && n >= __mem_nt_threshold_arm64 ) return simd::__memcpy_bulk_nt(d, s, n);
37 #endif
38     return simd::__memmove_bulk_fwd(d, s, n);
39   }
40   return simd::__memmove_bulk_bwd(d, s, n);
41 }

43 [[gnu::always_inline]] static inline byte *
44 __memmove_bytes(byte *d, const byte *s, const u64 bytes) noexcept
45 {
46   if ( d == s || bytes == 0 ) return d;
47   if ( bytes <= 32 ) {
48     __ml::__copy_le32(d, s, bytes);
49     return d;
50   }
51   if ( bytes <= 64 ) {
52     __ml::__copy_33_64(d, s, bytes);
53     return d;
54   }
55   if ( bytes <= 128 ) {
56     __ml::__copy_65_128(d, s, bytes);
57     return d;
58   }
59 #if defined(__micron_x86_avx2)
60   if ( bytes <= 256 ) {
61     __ml::__copy_129_256(d, s, bytes);
62     return d;
63   }
64 #endif
65   return __memmove_large(d, s, bytes);
66 }

188 template<typename F, typename D>
189 F *
190 memmove(F *dest, D *src, const u64 cnt) noexcept
191 {
192   if constexpr ( sizeof(F) == sizeof(D) && micron::is_trivially_copyable_v<F> && micron::is_trivially_copyable_v<D> ) {
193     __memmove_bytes(reinterpret_cast<byte *>(dest), reinterpret_cast<const byte *>(src), cnt * sizeof(D));
194     return dest;
195   } else {
196     if ( dest < src ) {
197       for ( u64 i = 0; i < cnt; i++ ) dest[i] = static_cast<F>(src[i]);
198     } else if ( dest > src ) {
199       for ( u64 i = cnt; i > 0; --i ) dest[i - 1] = static_cast<F>(src[i - 1]);
200     }
201     return dest;
202   }
203 };
