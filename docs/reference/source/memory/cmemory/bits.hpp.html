94 #if defined(__micron_arch_x86_any) && !defined(MICRON_MEM_NO_PROBE)
95 [[gnu::cold, gnu::noinline]] inline void
96 __mem_probe() noexcept
97 {
98   u32 a, b, c, dx;
99   __asm__ __volatile__("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(dx) : "0"(0u), "2"(0u));
100   const u32 max_leaf = a;
101   bool erms = false, fsrm = false;
102   if ( max_leaf >= 7 ) {
103     __asm__ __volatile__("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(dx) : "0"(7u), "2"(0u));
104     erms = (b >> 9) & 1u;
105     fsrm = (dx >> 4) & 1u;
106   }
107   usize l3 = 0;
108   for ( u32 pass = 0; pass < 2 && !l3; ++pass ) {
109     const u32 leaf = pass == 0 ? 4u : 0x8000001Du;
110     if ( pass == 1 ) {
111       __asm__ __volatile__("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(dx) : "0"(0x80000000u), "2"(0u));
112       if ( a < 0x8000001Du ) break;
113     }
114     for ( u32 sub = 0; sub < 8; ++sub ) {
115       __asm__ __volatile__("cpuid" : "=a"(a), "=b"(b), "=c"(c), "=d"(dx) : "0"(leaf), "2"(sub));
116       if ( (a & 0x1Fu) == 0 ) break;
117       const u32 level = (a >> 5) & 0x7u;
118       const usize ways = ((b >> 22) & 0x3FFu) + 1;
119       const usize parts = ((b >> 12) & 0x3FFu) + 1;
120       const usize line = (b & 0xFFFu) + 1;
121       const usize sets = static_cast<usize>(c) + 1;
122       if ( level == 3 ) l3 = ways * parts * line * sets;
123     }
124   }
125   const usize floor_1m = 1u << 20;
126   const usize nt_copy = l3 ? (((l3 / 4) < floor_1m) ? floor_1m : (l3 / 4)) : static_cast<usize>(__mem_nt_threshold_default);
127   const usize nt_set = l3 ? ((((l3 / 4) * 3) < floor_1m) ? floor_1m : ((l3 / 4) * 3)) : static_cast<usize>(__mem_nt_threshold_default);
128   usize movsb = __mem_tier_disabled;
129   if ( fsrm )
130     movsb = 2112;
131   else if ( erms )
132     movsb = 8192;
133   usize stosb = __mem_tier_disabled;
134   if ( fsrm )
135     stosb = 2048;
136   else if ( erms )
137     stosb = 16384;
138   __mem_tun.nt_copy_threshold = nt_copy;
139   __mem_tun.nt_set_threshold = nt_set;
140   __mem_tun.rep_movsb_threshold = movsb < nt_copy ? movsb : nt_copy;
141   __mem_tun.rep_stosb_threshold = stosb < nt_set ? stosb : nt_set;
142   __asm__ __volatile__("" ::: "memory");
143 }
144 #endif

146 [[gnu::always_inline]] static inline const __mem_tunables &
147 __mem_tun_get() noexcept
148 {
149   // if we're in freestanding mode tunables get primed in _start
150 #if defined(__micron_arch_x86_any) && !defined(MICRON_MEM_NO_PROBE) && !defined(__micron_freestanding)
151   if ( __atomic_load_n(&__mem_tun_state, __ATOMIC_ACQUIRE) != 2u ) [[unlikely]] {
152     u32 __exp = 0u;
153     if ( __atomic_compare_exchange_n(&__mem_tun_state, &__exp, 1u, false, __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE) ) {
154       __mem_probe();
155       __atomic_store_n(&__mem_tun_state, 2u, __ATOMIC_RELEASE);
156     } else {
157       while ( __atomic_load_n(&__mem_tun_state, __ATOMIC_ACQUIRE) != 2u ) __asm__ __volatile__("pause");      // wait for the winner
158     }
159   }
160 #endif
161   return __mem_tun;
162 }

402 // pattern fill for wordset/typeset
403 [[gnu::always_inline]] static inline void
404 __wset_le64(byte *d, const u64 w, const u64 n) noexcept
405 {
406   // n in [1, 64]
407   if ( n >= 16 ) {
408 #if defined(__micron_arch_wasm32)
409     const __v16 v = simd::__wasm_splat<__v16>(w);
410 #else
411     const __v16 v = { w, w };
412 #endif
413     __st16(d, v);
414     if ( n > 16 ) __st16(d + n - 16, v);
415     if ( n > 32 ) __st16(d + 16, v);
416     if ( n > 48 ) __st16(d + 32, v);
417   } else if ( n >= 8 ) {
418     __stu64(d, w);
419     if ( n > 8 ) __stu64(d + n - 8, w);
420   } else {
421     const byte *wb = reinterpret_cast<const byte *>(&w);
422     for ( u64 i = 0; i < n; i++ ) d[i] = wb[i & 7];
423   }
424 }

491 template<typename F>
492 bool
493 __is_at_stack(const F *ptr, const u64 size) noexcept
494 {
495   if ( ptr == nullptr ) return false;
496   const uintptr_t a = reinterpret_cast<uintptr_t>(ptr);
497   stack_t st = get_stack();
498   const uintptr_t lo = reinterpret_cast<uintptr_t>(st.start);
499   const uintptr_t hi = lo + st.size;
500   if ( a < lo or a > hi or size > static_cast<uintptr_t>(hi - a) ) return false;
501   return true;
502 }
