13 // Counts are bytes. Empty/null ranges never overlap; no pointed-to byte is read.
14 inline bool
15 byte_ranges_overlap(const void *left, usize left_size, const void *right, usize right_size) noexcept
16 {
17   if ( !left || !right || left_size == 0 || right_size == 0 ) return false;
18   const uintptr_t a = reinterpret_cast<uintptr_t>(left);
19   const uintptr_t b = reinterpret_cast<uintptr_t>(right);
20   return a <= b ? b - a < left_size : a - b < right_size;
21 }
22 
23 template<typename Left, typename Right>
24   requires(sizeof(Left) == 1 && sizeof(Right) == 1)
25 constexpr bool
26 byte_ranges_overlap(const Left *left, usize left_size, const Right *right, usize right_size) noexcept
27 {
28   if consteval {
29     if ( !left || !right || left_size == 0 || right_size == 0 ) return false;
30     // Ordering pointers from unrelated arrays is not a constant expression.
31     for ( usize index = 0; index < left_size; ++index )
32       if ( static_cast<const void *>(left + index) == static_cast<const void *>(right) ) return true;
33     for ( usize index = 0; index < right_size; ++index )
34       if ( static_cast<const void *>(right + index) == static_cast<const void *>(left) ) return true;
35     return false;
36   } else {
37     return byte_ranges_overlap(static_cast<const void *>(left), left_size, static_cast<const void *>(right), right_size);
38   }
39 }
