13 // A unique pointer, a general replacement of unique_ptr with a few differences
14 // unique_ptr is meant to always create a new instance of the object on creation and this is intentional
15 // it's impossible to create a unique_pointer by passing a raw pointer into it without moving it
16 // prevents two instances of the same pointer existing simult.
17 template<class Type> class unique_pointer: private __internal_pointer_alloc<Type>
18 {
19   Type *internal_pointer;
20 
21 public:
22   using pointer_type = owning_pointer_tag;
23   using category_type = pointer_tag;
24   using mutability_type = mutable_tag;
25   using value_type = Type;
26   using element_type = Type;
27 
28   using __alloc = __internal_pointer_alloc<Type>;
29 
30   ~unique_pointer() noexcept { __alloc::__impl_dealloc(internal_pointer); };

33   template<typename V>
34     requires micron::is_null_pointer_v<V>
35   constexpr unique_pointer(V) noexcept : internal_pointer(nullptr){};

38   template<class... Args>
39     requires(sizeof...(Args) > 0)
40   unique_pointer(Args &&...args) : internal_pointer(__alloc::__impl_alloc(micron::forward<Args>(args)...)){};

55   };
