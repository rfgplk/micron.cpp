11 namespace micron
12 {
13 template<class Type> class empty_pointer
14 {
15   Type *internal_pointer;
16 
17 public:
18   ~empty_pointer()
19   {
20     if ( internal_pointer != nullptr ) exc<except::memory_error>("empty_pointer ~(): pointer is not empty");
21   }
22 
23   empty_pointer(void) : internal_pointer(occupied) { }
24 
25   empty_pointer(const empty_pointer &) = delete;
26 
27   empty_pointer(empty_pointer &&o) : internal_pointer(o.internal_pointer) { o.internal_pointer = nullptr; }
28 
29   empty_pointer &operator=(const empty_pointer &) = delete;
30 
31   empty_pointer &
32   operator=(empty_pointer &&o)
33   {
34     if ( internal_pointer != nullptr ) exc<except::memory_error>("empty_pointer operator=(): pointer is not empty");
35     internal_pointer = o.internal_pointer;
36     o.internal_pointer = nullptr;
37     return *this;
38   }
39 
40   inline Type *
41   release(void) noexcept
42   {
43     Type *ptr = internal_pointer;
44     internal_pointer = nullptr;
45     return ptr;
46   }
47 
48   constexpr explicit
49   operator bool() const noexcept
50   {
51     return internal_pointer != nullptr;
52   }
53 };
