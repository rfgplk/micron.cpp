15 // TO ALL READERS
16 // this set of functions is ever so slightly different from the standard string.h
17 // set of memory manip. functions
18 // generally if you'd like to set memory you should use
19 // bset     = set bytes, in bytes
20 // memset   = set memory, in bytes or type provided (ie size of the type you provided)
21 // cmemset  = set memory with known length at compile time
22 // wordset  = set word, in words
23 // typeset  = set memory, as if it were provided type
24 // functions suffixed with a number represent the width or alignment of the function
25 // functions prefixed with a 'c' all take in certain parameters at compile time
