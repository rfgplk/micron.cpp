26 constexpr u32
27 ver(u32 maj, u32 min = 0, u32 patch = 0) noexcept
28 {
29   if ( maj > 255 ) maj = 255;
30   if ( min > 255 ) min = 255;
31   if ( patch > 255 ) patch = 255;
32   return (maj << 16) | (min << 8) | patch;
33 }

57 // NOTE: redeclared here on purpose so kernel.hpp doesn't pull in sys/system.hpp
58 struct __utsname {
59   char sysname[65];
60   char nodename[65];
61   char release[65];
62   char version[65];
63   char machine[65];
64   char domainname[65];
65 };

67 constexpr u32
68 __parse_release(const char *s) noexcept
69 {
70   u32 part[3]{ 0, 0, 0 };
71   usize i = 0;
72   for ( usize p = 0; p < 3; p++ ) {
73     if ( s[i] < '0' || s[i] > '9' ) break;
74     u32 acc = 0;
75     for ( ; s[i] >= '0' && s[i] <= '9'; i++ ) {
76       acc = acc * 10 + static_cast<u32>(s[i] - '0');
77       if ( acc > 255 ) acc = 255;
78     }
79     part[p] = acc;
80     if ( s[i] != '.' ) break;
81     i++;
82   }
83   if ( part[0] == 0 ) return 0;
84   return ver(part[0], part[1], part[2]);
85 }

87 // pre-5.0 and parse failures
88 constexpr u32
89 __bucket(u32 v) noexcept
90 {
91   return v < floor_version ? floor_version : v;
92 }

94 inline u32 __version_slot = 0;
95 
96 [[gnu::cold]] [[gnu::noinline]] inline u32
97 __detect(void) noexcept
98 {
99   __utsname u{};
100   long r = micron::syscall(SYS_uname, &u);
101   u32 v = __bucket(micron::syscall_failed(r) ? 0 : __parse_release(u.release));
102   atom::store(&__version_slot, v, micron::atomic_relaxed);
103   return v;
104 }
105 };      // namespace __impl

107 [[gnu::always_inline]] inline u32
108 version(void) noexcept
109 {
110   u32 v = atom::load(&__impl::__version_slot, micron::atomic_relaxed);
111   if ( v != 0 ) [[likely]]
112     return v;
113   return __impl::__detect();
114 }

116 [[gnu::always_inline]] inline bool
117 since(u32 v) noexcept
118 {
119   if ( v <= floor_version ) return true;
120   return version() >= v;
121 }

123 [[gnu::always_inline]] inline bool
124 since(u32 maj, u32 min, u32 patch = 0) noexcept

197 // NOTE: a kernel can report >= X yet still refuse the syscall (seccomp/docker filters, qemu-user, CONFIG_ off, vendor kernels).
198 // WARNING: demote on ENOSYS only; EPERM can be a legitimate result and must never demote
199 struct probe_gate {
200   u32 __demoted = 0;
201 
202   [[gnu::always_inline]] bool
203   open(u32 feature_ver) noexcept
204   {
205     return since(feature_ver) && atom::load(&__demoted, micron::atomic_relaxed) == 0;
206   }
207 
208   [[gnu::always_inline]] void
209   demote(void) noexcept
210   {
211     atom::store(&__demoted, 1u, micron::atomic_relaxed);
212   }
213 
214   void
215   restore(void) noexcept
216   {
217     atom::store(&__demoted, 0u, micron::atomic_relaxed);
218   }
219 };
