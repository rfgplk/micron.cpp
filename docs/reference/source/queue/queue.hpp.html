22 template<typename T, usize N = micron::alloc_auto_sz, class Alloc = micron::allocator_serial<>>
23   requires micron::is_copy_constructible_v<T> and micron::is_move_constructible_v<T> and micron::is_copy_assignable_v<T>
24            and micron::is_destructible_v<T>

30   template<typename I>
31   static constexpr usize
32   __checked_elements(I count)
33   {
34     if ( static_cast<umax_t>(count) > static_cast<umax_t>(static_cast<usize>(-1) / sizeof(T)) ) [[unlikely]]
35       exc<except::library_error>("micron::queue capacity overflow");
36     return static_cast<usize>(count);
37   }
38 
39   inline void
40   __relocate(const usize requested)
41   {
42     __checked_elements(requested);
43     const usize count = __mem::length;
44     chunk<byte> block = __allocator_create<Alloc, alignof(T)>(allocation_multiply_or_throw(requested, sizeof(T)));
45     __mem replacement(micron::move(block));
46     T *fresh = replacement.memory;
47 
48     if constexpr ( micron::is_trivially_copyable_v<T> ) {
49       if ( count ) micron::memcpy(fresh, micron::addressof(__mem::memory[head]), count);
50     } else {
51       usize made = 0;
52 #if !defined(__micron_freestanding) || defined(__micron_eh)
53       try {
54 #endif
55         for ( ; made < count; ++made ) {
56           if constexpr ( micron::is_nothrow_move_constructible_v<T> or !micron::is_copy_constructible_v<T> )
57             new (micron::addr(fresh[made])) T(micron::move(__mem::memory[head + made]));
58           else
59             new (micron::addr(fresh[made])) T(__mem::memory[head + made]);
60         }
61 #if !defined(__micron_freestanding) || defined(__micron_eh)
62       } catch ( ... ) {
63         __impl_container::destroy(fresh, made);
64         throw;
65       }
66 #endif
67       if ( count ) __impl_container::destroy(micron::addr(__mem::memory[head]), count);
68     }
69 
70     __mem::swap(replacement);
71     __mem::length = count;
72     head = 0;
73   }

75   [[gnu::noinline, gnu::cold]] void
76   __make_tail_space()
77   {
78     if ( head ) {
79       if constexpr ( micron::is_trivially_copyable_v<T> ) {
80         micron::memmove(__mem::memory, micron::addressof(__mem::memory[head]), __mem::length);
81         head = 0;
82       } else {
83         __relocate(__mem::capacity);
84       }
85       if ( __mem::length < __mem::capacity ) return;
86     }
87     const usize max_elements = static_cast<usize>(-1) / sizeof(T);
88     if ( __mem::capacity == max_elements ) [[unlikely]]
89       exc<except::library_error>("micron::queue capacity overflow");
90     __relocate(__mem::recommended_capacity(__mem::capacity, __mem::capacity + 1));
91   }
92 
93   [[gnu::always_inline]] inline void
94   __ensure_tail_space()
95   {
96     if ( head <= __mem::capacity && __mem::length == __mem::capacity - head ) [[unlikely]]
97       __make_tail_space();
98   }

122   queue(const std::initializer_list<T> &lst) : __mem(lst.size()), head(0)
123   {
124     if constexpr ( !micron::is_trivially_copyable_v<T> ) {
125       usize i = 0;
126 #if !defined(__micron_freestanding) || defined(__micron_eh)
127       try {
128 #endif
129         for ( const T &value : lst ) {
130           new (micron::addr(__mem::memory[i])) T(value);
131           ++i;
132         }
133 #if !defined(__micron_freestanding) || defined(__micron_eh)
134       } catch ( ... ) {
135         __impl_container::destroy(__mem::memory, i);
136         throw;
137       }
138 #endif
139       __mem::length = lst.size();
140     } else {
141       usize i = 0;
142       for ( T value : lst ) {
143         __mem::memory[i++] = value;
144       }
145       __mem::length = lst.size();
146     }
147   };

185   queue &
186   operator=(queue &&o)
187   {
188     if ( this == &o ) return *this;
189     // release destination first: clear() runs element dtors (as ~queue does) and
190     // __mem::free() releases the old buffer; the base move-assign does not free.
191     if ( __mem::memory ) {
192       clear();
193       __mem::free();
194     }
195     __mem::operator=(micron::move(o));
196     head = o.head;
197     o.head = 0;
198     return *this;
199   }

201   inline void
202   clear()
203   {
204     if ( !__mem::length ) return;
205     __impl_container::destroy(micron::addr(__mem::memory[head]), __mem::length);
206     head = 0;
207     __mem::length = 0;
208   }

305   inline queue &
306   push(void)
307   {
308     __ensure_tail_space();
309     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_constructible_v<T> ) {
310       new (micron::addr(__mem::memory[head + __mem::length])) T{};
311     } else {
312       __mem::memory[head + __mem::length] = T{};
313     }
314     __mem::length++;
315     return *this;
316   }

344   inline queue &
345   pop(void)
346   {
347     if ( __mem::length == 0 ) return *this;
348     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
349       (__mem::memory)[head].~T();
350     }
351     head++;
352     __mem::length--;
353     if ( __mem::length == 0 ) head = 0;
354     return *this;
355   }
