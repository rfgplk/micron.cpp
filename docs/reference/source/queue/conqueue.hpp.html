49   __checked_elements(I count)
50   {
51     if ( static_cast<umax_t>(count) > static_cast<umax_t>(static_cast<usize>(-1) / sizeof(T)) ) [[unlikely]]

56   static inline void
57   __lock_ordered(micron::fast_mutex &a, __defer &la, micron::fast_mutex &b, __defer &lb)
58   {
59     const uintptr_t aa = reinterpret_cast<uintptr_t>(micron::addr(a));
60     const uintptr_t bb = reinterpret_cast<uintptr_t>(micron::addr(b));
61     if ( aa == bb ) {
62       la.lock();
63     } else if ( aa < bb ) {
64       la.lock();
65       lb.lock();
66     } else {
67       lb.lock();
68       la.lock();
69     }
70   }

72   inline void
73   __reserve_unsafe(const usize n)
74   {
75     if ( n <= __mem::capacity && head == 0 ) return;
76     const usize count = __mem::length;
77     const usize requested = n < count ? count : n;
78     const usize elements = __checked_elements(requested ? requested : 1);
79     chunk<byte> block = __allocator_create<Alloc, alignof(T)>(allocation_multiply_or_throw(elements, sizeof(T)));
80     __mem replacement(micron::move(block));
81     T *fresh = replacement.memory;
82 
83     if constexpr ( micron::is_trivially_copyable_v<T> ) {
84       if ( count ) micron::memcpy(fresh, micron::addressof(__mem::memory[head]), count);
85     } else {
86       usize made = 0;
87 #if !defined(__micron_freestanding) || defined(__micron_eh)
88       try {
89 #endif
90         for ( ; made < count; ++made ) {
91           if constexpr ( micron::is_nothrow_move_constructible_v<T> or !micron::is_copy_constructible_v<T> )
92             new (micron::addr(fresh[made])) T(micron::move(__mem::memory[head + made]));
93           else
94             new (micron::addr(fresh[made])) T(__mem::memory[head + made]);
95         }
96 #if !defined(__micron_freestanding) || defined(__micron_eh)
97       } catch ( ... ) {
98         __impl_container::destroy(fresh, made);
99         throw;
100       }
101 #endif
102       if ( count ) __impl_container::destroy(micron::addr(__mem::memory[head]), count);
103     }
104 
105     __mem::swap(replacement);
106     __mem::length = count;
107     head = 0;
108   }

110   [[gnu::always_inline]] inline void
111   __ensure_one_unsafe()
112   {
113     if ( head <= __mem::capacity && __mem::length < __mem::capacity - head ) return;
114     if constexpr ( micron::is_trivially_copyable_v<T> ) {
115       if ( head ) {
116         micron::memmove(__mem::memory, micron::addressof(__mem::memory[head]), __mem::length);
117         head = 0;
118         return;
119       }
120     }
121     const usize max_elements = static_cast<usize>(-1) / sizeof(T);
122     if ( __mem::length == __mem::capacity && __mem::capacity == max_elements ) [[unlikely]]
123       exc<except::library_error>("micron::conqueue capacity overflow");
124     const usize wanted
125         = __mem::length < __mem::capacity ? __mem::capacity : __mem::recommended_capacity(__mem::capacity, __mem::capacity + 1);
126     __reserve_unsafe(wanted);
127   }

130   __clear_unsafe() noexcept

133     __mem::length = 0;
134     head = 0;

137   template<typename U>
138   inline void
139   __push_unsafe(U &&val)
140   {
141     __ensure_one_unsafe();
142     if constexpr ( micron::is_trivially_copyable_v<T> )
143       __mem::memory[head + __mem::length] = micron::forward<U>(val);
144     else
145       new (micron::addr(__mem::memory[head + __mem::length])) T(micron::forward<U>(val));
146     ++__mem::length;
147   }

149   inline bool
150   __pop_unsafe(T *out)
151   {
152     if ( __mem::length == 0 ) return false;
153     T &v = __mem::memory[head];
154     if ( out ) {
155       if constexpr ( micron::is_move_assignable_v<T> )
156         *out = micron::move(v);
157       else
158         *out = v;
159     }
160     if constexpr ( !micron::is_trivially_destructible_v<T> ) v.~T();
161     ++head;
162     --__mem::length;
163     if ( __mem::length == 0 ) head = 0;
164     return true;
165   }

455   inline usize
456   push_batch(const T *items, usize count)
457   {
458     __hold __lock(__mtx);
459     if ( count == 0 ) return 0;
460     if ( count > static_cast<usize>(-1) - __mem::length ) [[unlikely]]
461       exc<except::library_error>("micron::conqueue capacity overflow");
462     const usize needed = __mem::length + count;
463     if ( head > __mem::capacity || needed > __mem::capacity - head ) {
464       if constexpr ( micron::is_trivially_copyable_v<T> ) {
465         if ( needed <= __mem::capacity ) {
466           micron::memmove(__mem::memory, micron::addressof(__mem::memory[head]), __mem::length);
467           head = 0;
468         } else {
469           __reserve_unsafe(__mem::recommended_capacity(__mem::capacity, needed));
470         }
471       } else {
472         __reserve_unsafe(needed <= __mem::capacity ? __mem::capacity : __mem::recommended_capacity(__mem::capacity, needed));
473       }
474     }
475     if constexpr ( micron::is_trivially_copyable_v<T> ) {
476       micron::memcpy(micron::addressof(__mem::memory[head + __mem::length]), items, count);
477       __mem::length += count;
478       return count;
479     } else {
480       usize pushed = 0;
481       for ( ; pushed < count; ++pushed ) {
482         new (micron::addr(__mem::memory[head + __mem::length])) T(items[pushed]);
483         ++__mem::length;
484       }
485       return pushed;
486     }
487   }

489   inline usize
490   pop_batch(T *items, usize count)
491   {
492     __hold __lock(__mtx);
493     const usize n = count < __mem::length ? count : __mem::length;
494     usize popped = 0;
495     for ( ; popped < n; ++popped ) __pop_unsafe(micron::addressof(items[popped]));
496     return popped;
