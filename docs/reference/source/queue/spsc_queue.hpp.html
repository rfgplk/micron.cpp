33   constexpr static usize
34   __next_pow2(usize n)
35   {
36     n--;
37     n |= n >> 1;
38     n |= n >> 2;
39     n |= n >> 4;
40     n |= n >> 8;
41     n |= n >> 16;
42     if constexpr ( sizeof(usize) > 4 ) {
43       constexpr usize __hi_shift = sizeof(usize) * 8u / 2u;
44       n |= n >> __hi_shift;
45     }
46     return n + 1;
47   }

112   inline void
113   clear()
114   {
115     const usize h = __consumer.head.get(memory_order_relaxed);
116     const usize t = __producer.tail.get(memory_order_relaxed);
117 
118     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
119       for ( usize i = h; i != t; ++i ) {
120         (__mem::memory)[i & __mask].~T();
121       }
122     }
123 
124     __consumer.head.store(0, memory_order_relaxed);
125     __producer.tail.store(0, memory_order_relaxed);
126     __producer.cached_head = 0;
127     __consumer.cached_tail = 0;
128   }

158   __attribute__((always_inline)) inline bool
159   push(void)
160   {
161     const usize t = __producer.tail.get(memory_order_relaxed);
162     const usize __next = t + 1;
163 
164     usize avail = __spsc_capacity - (t - __producer.cached_head);
165 
166     if ( avail == 0 ) [[unlikely]] {
167       __producer.cached_head = __consumer.head.get(memory_order_acquire);
168       avail = __spsc_capacity - (t - __producer.cached_head);
169 
170       if ( avail == 0 ) return false;
171     }
172 
173     const usize idx = t & __mask;
174 
175     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_constructible_v<T> ) {
176       new (micron::addr(__mem::memory[idx])) T{};
177     } else {
178       __mem::memory[idx] = T{};
179     }
180 
181     __producer.tail.store(__next, memory_order_release);
182     return true;
183   }

263   __attribute__((always_inline)) inline bool
264   pop(T &out)
265   {
266     const usize h = __consumer.head.get(memory_order_relaxed);
267 
268     usize avail = __consumer.cached_tail - h;
269 
270     if ( avail == 0 ) [[unlikely]] {
271       __consumer.cached_tail = __producer.tail.get(memory_order_acquire);
272       avail = __consumer.cached_tail - h;
273 
274       if ( avail == 0 ) return false;
275     }
276 
277     const usize idx = h & __mask;
278 
279     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_copyable_v<T> ) {
280       __assign(out, __mem::memory[idx]);
281       __mem::memory[idx].~T();
282     } else {
283       out = __mem::memory[idx];
284     }
285 
286     __consumer.head.store(h + 1, memory_order_release);
287     return true;

351   __attribute__((always_inline)) inline usize
352   push_batch(const T *items, usize count)
353   {
354     const usize t = __producer.tail.get(memory_order_relaxed);
355     usize avail = __spsc_capacity - (t - __producer.cached_head);
356 
357     if ( avail < count ) [[unlikely]] {
358       __producer.cached_head = __consumer.head.get(memory_order_acquire);
359       avail = __spsc_capacity - (t - __producer.cached_head);
360     }
361 
362     const usize to_push = (avail < count) ? avail : count;
363     if ( to_push == 0 ) return 0;
364     const usize idx = t & __mask;
365     const usize first = to_push < (__spsc_capacity - idx) ? to_push : (__spsc_capacity - idx);
366 
367     if constexpr ( micron::is_trivially_copyable_v<T> ) {
368       micron::memcpy(micron::addressof(__mem::memory[idx]), items, first);
369       if ( first != to_push ) micron::memcpy(__mem::memory, items + first, to_push - first);
370     } else {
371       usize made = 0;
372 #if !defined(__micron_freestanding) || defined(__micron_eh)
373       try {
374 #endif
375         for ( ; made < to_push; ++made ) new (micron::addr(__mem::memory[(t + made) & __mask])) T(items[made]);
376 #if !defined(__micron_freestanding) || defined(__micron_eh)
377       } catch ( ... ) {
378         for ( usize i = 0; i < made; ++i ) __mem::memory[(t + i) & __mask].~T();
379         throw;
380       }
381 #endif
382     }
383 
384     __producer.tail.store(t + to_push, memory_order_release);
385     return to_push;
386   }

388   __attribute__((always_inline)) inline usize
389   pop_batch(T *items, usize count)
390   {
391     const usize h = __consumer.head.get(memory_order_relaxed);
392     usize avail = __consumer.cached_tail - h;
393 
394     if ( avail < count ) [[unlikely]] {
395       __consumer.cached_tail = __producer.tail.get(memory_order_acquire);
396       avail = __consumer.cached_tail - h;
397     }
398 
399     const usize to_pop = (avail < count) ? avail : count;
400     if ( to_pop == 0 ) return 0;
401     const usize idx = h & __mask;
402     const usize first = to_pop < (__spsc_capacity - idx) ? to_pop : (__spsc_capacity - idx);
403 
404     if constexpr ( micron::is_trivially_copyable_v<T> ) {
405       micron::memcpy(items, micron::addressof(__mem::memory[idx]), first);
406       if ( first != to_pop ) micron::memcpy(items + first, __mem::memory, to_pop - first);
407       __consumer.head.store(h + to_pop, memory_order_release);
408     } else if constexpr ( __assign_nothrow ) {
409       for ( usize i = 0; i < to_pop; ++i ) {
410         T &v = __mem::memory[(h + i) & __mask];
411         __assign(items[i], v);
412         v.~T();
413       }
414       __consumer.head.store(h + to_pop, memory_order_release);
415     } else {
416       for ( usize i = 0; i < to_pop; ++i ) {
417         T &v = __mem::memory[(h + i) & __mask];
418         __assign(items[i], v);
419         v.~T();
420         __consumer.head.store(h + i + 1, memory_order_release);
421       }
422     }
423     return to_pop;
424   }
