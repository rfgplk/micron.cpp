29 constexpr usize
30 __next_pow2(usize n) noexcept
31 {
32   n--;
33   n |= n >> 1;
34   n |= n >> 2;
35   n |= n >> 4;
36   n |= n >> 8;
37   n |= n >> 16;
38   if constexpr ( sizeof(usize) > 4 ) {
39     constexpr usize __hi = sizeof(usize) * 8u / 2u;
40     n |= n >> __hi;
41   }
42   return n + 1;
43 }
44 
45 constexpr usize
46 __log2_pow2(usize n) noexcept
47 {
48   usize s = 0;
49   while ( (static_cast<usize>(1) << s) < n ) ++s;
50   return s;
51 }

71 [[gnu::always_inline]] inline __idx
72 __cl_dec_bottom_load_top(micron::atomic_token<__idx> &__bottom, micron::atomic_token<__idx> &__top, __idx __b) noexcept
73 {
74 #if defined(__micron_arch_arm64)
75   __bottom.store(__b, memory_order_seq_cst);      // stlr
76   return __top.get(memory_order_seq_cst);         // ldar
77 #else
78   __bottom.store(__b, memory_order_relaxed);
79   micron::atom::thread_fence(atomic_seq_cst);
80   return __top.get(memory_order_relaxed);
81 #endif
82 }

98 template<is_atomic_type T, usize N>
99   requires(N > 0 and N <= __cl_impl::__max_pow2_capacity)
100 class chase_lev
101 {
102   static constexpr u64 __cache_line = cache_line_size();
103   static constexpr usize __cap = __cl_impl::__next_pow2(N);
104   static constexpr __idx __mask = static_cast<__idx>(__cap) - 1;
105 
106   using __slot = micron::atomic_token<T>;
107   static_assert(__cap <= static_cast<usize>(-1) / sizeof(__slot), "chase_lev: slot allocation size is not representable");
108 
109   // cache line 0
110   alignas(__cache_line) micron::atomic_token<__idx> __bottom;
111   __slot *__slots = nullptr;
112   __idx __top_cache = 0;
113   // no explicit padding
114 
115   // cache line 1, touched on last elem
116   alignas(__cache_line) micron::atomic_token<__idx> __top;
117   [[no_unique_address]] __cache_pad<__cache_line - sizeof(__idx)> __pad1;      // keep a neighbour off this line
118 
119   // must always be inlined; runs significantly worse without (even when taking into account inflated inst pressure)
120   [[gnu::always_inline]] inline bool
121   __pop_cold(T &__out, __idx __b, __idx __t) noexcept
122   {
123     bool __got = false;
124     if ( __t == __b ) {
125       const T __x = __slots[__b & __mask].get(memory_order_relaxed);
126       __idx __e = __t;
127       if ( __top.compare_exchange_strong(__e, __t + 1, memory_order_seq_cst, memory_order_relaxed) ) {
128         __out = __x;
129         __got = true;
130       }
131     }
132     __bottom.store(__b + 1, memory_order_relaxed);
133     return __got;
134   }
135 
136   // cold tail
137   [[gnu::noinline, gnu::cold]] bool
138   __push_full(__idx __b) noexcept
139   {
140     __top_cache = __top.get(memory_order_relaxed);
141     return __sdiff(__b, __top_cache) <= static_cast<__sidx>(__mask);
142   }
143 
144   [[gnu::noinline]] steal_result<T>
145   __take_top(__idx __t, __idx __b) noexcept
146   {
147     const T __x = __slots[__t & __mask].get(memory_order_relaxed);
148     __idx __e = __t;
149     // weak is better, keep it
150     if ( !__top.compare_exchange_weak(__e, __t + 1, memory_order_seq_cst, memory_order_relaxed) ) [[unlikely]]
151       return { __empty, steal_status::lost, true };
152     return { __x, steal_status::got, __sdiff(__b, __t) > 1 };
153   }
154 
155 public:
156   typedef T value_type;
157   static constexpr T __empty = T{};
158 
159   chase_lev() : __bottom(0), __top(0)
160   {
161     __slots = static_cast<__slot *>(::operator new(sizeof(__slot) * __cap, static_cast<std::align_val_t>(__cache_line)));
162     for ( usize i = 0; i < __cap; ++i ) new (&__slots[i]) __slot(T{});
163   }
164 
165   ~chase_lev()
166   {
167     if ( __slots ) {
168       for ( usize i = 0; i < __cap; ++i ) __slots[i].~__slot();
169       ::operator delete(__slots, static_cast<std::align_val_t>(__cache_line));
170       __slots = nullptr;
171     }
172   }
173 
174   chase_lev(const chase_lev &) = delete;
175   chase_lev(chase_lev &&) = delete;
176   chase_lev &operator=(const chase_lev &) = delete;
177   chase_lev &operator=(chase_lev &&) = delete;
178 
179   static constexpr usize
180   capacity() noexcept
181   {
182     return __cap;
183   }
184 
185   inline usize
186   size() const noexcept
187   {
188     const __idx t = __top.get(memory_order_acquire);
189     const __idx b = __bottom.get(memory_order_acquire);
190     const __sidx d = __sdiff(b, t);
191     if ( d <= 0 ) return 0;
192     const usize n = static_cast<usize>(d);
193     return n < __cap ? n : __cap;
194   }
195 
196   inline bool
197   empty() const noexcept
198   {
199     return __sdiff(__bottom.get(memory_order_acquire), __top.get(memory_order_acquire)) <= 0;
200   }
201 
202   // OWNER ONLY
203   [[gnu::always_inline]] inline bool
204   push_bottom(T x) noexcept
205   {
206     const __idx b = __bottom.get(memory_order_relaxed);
207     if ( __sdiff(b, __top_cache) > static_cast<__sidx>(__mask) ) [[unlikely]] {
208       if ( !__push_full(b) ) return false;      // genuinely full
209     }
210     __slots[b & __mask].store(x, memory_order_relaxed);
211     // release store, not a release fence
212     __bottom.store(b + 1, memory_order_release);
213     return true;
214   }
215 
216   // OWNER ONLY (no CAS nor restore)
217   [[gnu::always_inline]] inline bool
218   pop_bottom(T &out) noexcept
219   {
220     const __idx b = __bottom.get(memory_order_relaxed) - 1;
221     const __idx t = __cl_impl::__cl_dec_bottom_load_top(__bottom, __top, b);
222     if ( __sdiff(b, t) > 0 ) [[likely]] {
223       out = __slots[b & __mask].get(memory_order_relaxed);
224       return true;
225     }
226     return __pop_cold(out, b, t);
227   }
228 
229   [[gnu::always_inline]] inline T
230   pop_bottom() noexcept
231   {
232     T out = __empty;
233     return pop_bottom(out) ? out : __empty;
234   }
235 
236   [[gnu::always_inline]] inline steal_result<T>
237   try_steal() noexcept
238   {
239     // don't use a fence
240     const __idx t = __top.get(memory_order_acquire);
241     const __idx b = __bottom.get(memory_order_seq_cst);      // seq_cst, not acquire
242     if ( __sdiff(b, t) <= 0 ) [[likely]]
243       return { __empty, steal_status::empty, false };
244     return __take_top(t, b);
245   }
246 
247   // ANY NON OWNER
248   [[gnu::always_inline]] inline T
249   steal_top() noexcept
250   {
251     return try_steal().__v;
252   }
253 
254   // ANY NON OWNER
255   [[gnu::always_inline]] inline T
256   steal_top(steal_status &__st) noexcept
257   {
258     const steal_result<T> r = try_steal();
259     __st = r.__st;
260     return r.__v;
261   }
262 };

269 template<is_atomic_type T, usize InitN>
270   requires(InitN > 0 and InitN <= __cl_impl::__max_pow2_capacity)
271 class chase_lev_grow
272 {
273   static constexpr u64 __cache_line = cache_line_size();
274   static constexpr uintptr_t __tag_mask = static_cast<uintptr_t>(63);      // low 6 bits of a 64b aligned pointer
275   static constexpr usize __pointer_bits = sizeof(uintptr_t) * 8u;
276 
277   using __slot = micron::atomic_token<T>;
278   static constexpr usize __init_cap = __cl_impl::__next_pow2(InitN);
279   static_assert(__init_cap <= (static_cast<usize>(-1) - __cache_line) / sizeof(__slot),
280                 "chase_lev_grow: initial allocation size is not representable");
281 
282   struct __hdr {
283     usize __cap;
284     __hdr *__prev;
285   };
286 
287   [[gnu::always_inline]] static __hdr *
288   __hdr_of(uintptr_t __g) noexcept
289   {
290     return reinterpret_cast<__hdr *>(__g & ~__tag_mask);
291   }
292 
293   [[gnu::always_inline]] static __slot *
294   __slots_of(uintptr_t __g) noexcept
295   {
296     return reinterpret_cast<__slot *>((__g & ~__tag_mask) + __cache_line);
297   }
298 
299   [[gnu::always_inline]] static __idx
300   __mask_of(uintptr_t __g) noexcept
301   {
302     return (static_cast<__idx>(1) << static_cast<usize>(__g & __tag_mask)) - 1;
303   }
304 
305   // returns the tagged base word
306   static uintptr_t
307   __make_seg(usize __shift, __hdr *__prev)
308   {
309     const usize __cap = static_cast<usize>(1) << __shift;
310     void *__raw = ::operator new(__cache_line + sizeof(__slot) * __cap, static_cast<std::align_val_t>(__cache_line));
311     __hdr *__h = static_cast<__hdr *>(__raw);
312     __h->__cap = __cap;
313     __h->__prev = __prev;
314     __slot *__s = micron::ptr_cast<__slot *>(reinterpret_cast<byte *>(__raw) + __cache_line);
315     for ( usize __i = 0; __i < __cap; ++__i ) new (&__s[__i]) __slot(T{});
316     return reinterpret_cast<uintptr_t>(__raw) | static_cast<uintptr_t>(__shift);
317   }
318 
319   static void
320   __free_seg(__hdr *__h) noexcept
321   {
322     __slot *__s = micron::ptr_cast<__slot *>(reinterpret_cast<byte *>(__h) + __cache_line);
323     for ( usize __i = 0; __i < __h->__cap; ++__i ) __s[__i].~__slot();
324     ::operator delete(static_cast<void *>(__h), static_cast<std::align_val_t>(__cache_line));
325   }
326 
327   alignas(__cache_line) micron::atomic_token<__idx> __bottom;
328   micron::atomic_token<uintptr_t> __base;
329   uintptr_t __owner_base = 0;
330   __idx __owner_mask = 0;
331   __idx __top_cache = 0;
332   // no padding
333 
334   alignas(__cache_line) micron::atomic_token<__idx> __top;
335   [[no_unique_address]] __cache_pad<__cache_line - sizeof(__idx)> __pad1;      // keep a neighbour off this line
336 
337   [[gnu::noinline, gnu::cold]] uintptr_t
338   __grow(__idx __b, __idx __t, uintptr_t __g)
339   {
340     __hdr *__oh = __hdr_of(__g);
341     const usize __old_shift = static_cast<usize>(__g & __tag_mask);
342     if ( __old_shift + 1 >= __pointer_bits ) return 0;
343     const usize __ns = __old_shift + 1;
344     const usize __new_cap = static_cast<usize>(1) << __ns;
345     if ( __new_cap > (static_cast<usize>(-1) - __cache_line) / sizeof(__slot) ) return 0;
346     const uintptr_t __ng = __make_seg(__ns, __oh);
347     __slot *__new = __slots_of(__ng);
348     __slot *__old = __slots_of(__g);
349     const __idx __nm = __mask_of(__ng);
350     const __idx __om = __mask_of(__g);
351     for ( __idx __i = __t; __sdiff(__b, __i) > 0; ++__i )
352       __new[__i & __nm].store(__old[__i & __om].get(memory_order_relaxed), memory_order_relaxed);
353     __owner_base = __ng;
354     __owner_mask = __nm;
355     __base.store(__ng, memory_order_release);      // pairs with the thief's acquire load
356     return __ng;
357   }
358 
359   [[gnu::noinline, gnu::cold]] uintptr_t
360   __push_full(__idx __b, uintptr_t __g)
361   {
362     __top_cache = __top.get(memory_order_relaxed);
363     if ( __sdiff(__b, __top_cache) <= static_cast<__sidx>(__owner_mask) ) return __g;      // the cache was just stale
364     return __grow(__b, __top_cache, __g);
365   }
366 
367   [[gnu::always_inline]] inline bool
368   __pop_cold(T &__out, __idx __b, __idx __t, uintptr_t __g, __idx __mask) noexcept
369   {
370     bool __got = false;
371     if ( __t == __b ) {
372       const T __x = __slots_of(__g)[__b & __mask].get(memory_order_relaxed);
373       __idx __e = __t;      // STRONG: see the fixed variant
374       if ( __top.compare_exchange_strong(__e, __t + 1, memory_order_seq_cst, memory_order_relaxed) ) {
375         __out = __x;
376         __got = true;
377       }
378     }
379     __bottom.store(__b + 1, memory_order_relaxed);
380     return __got;
381   }
382 
383   [[gnu::noinline]] steal_result<T>
384   __take_top(__idx __t, __idx __b) noexcept
385   {
386     const uintptr_t __g = __base.get(memory_order_acquire);      // pairs with __grow's release
387     const T __x = __slots_of(__g)[__t & __mask_of(__g)].get(memory_order_relaxed);
388     __idx __e = __t;
389     if ( !__top.compare_exchange_weak(__e, __t + 1, memory_order_seq_cst, memory_order_relaxed) ) [[unlikely]]
390       return { __empty, steal_status::lost, true };
391     return { __x, steal_status::got, __sdiff(__b, __t) > 1 };
392   }
393 
394 public:
395   typedef T value_type;
396   static constexpr T __empty = T{};
397 
398   chase_lev_grow() : __bottom(0), __base(0), __top(0)
399   {
400     __owner_base = __make_seg(__cl_impl::__log2_pow2(__cl_impl::__next_pow2(InitN)), nullptr);
401     __owner_mask = __mask_of(__owner_base);
402     __base.store(__owner_base, memory_order_relaxed);
403   }
404 
405   ~chase_lev_grow()
406   {
407     __hdr *__h = __hdr_of(__base.get(memory_order_relaxed));
408     while ( __h != nullptr ) {
409       __hdr *__p = __h->__prev;
410       __free_seg(__h);
411       __h = __p;
412     }
413   }

415   chase_lev_grow(const chase_lev_grow &) = delete;
416   chase_lev_grow(chase_lev_grow &&) = delete;
417   chase_lev_grow &operator=(const chase_lev_grow &) = delete;
418   chase_lev_grow &operator=(chase_lev_grow &&) = delete;
419 
420   inline usize
421   capacity() const noexcept
422   {
423     return static_cast<usize>(__mask_of(__base.get(memory_order_acquire))) + 1u;
424   }
425 
426   inline usize
427   size() const noexcept
428   {
429     const __idx t = __top.get(memory_order_acquire);
430     const __idx b = __bottom.get(memory_order_acquire);
431     const __sidx d = __sdiff(b, t);
432     if ( d <= 0 ) return 0;
433     const usize cap = static_cast<usize>(__mask_of(__base.get(memory_order_acquire))) + 1u;
434     const usize n = static_cast<usize>(d);
435     return n < cap ? n : cap;
436   }
437 
438   inline bool
439   empty() const noexcept
440   {
441     return __sdiff(__bottom.get(memory_order_acquire), __top.get(memory_order_acquire)) <= 0;
442   }
443 
444   [[gnu::always_inline]] inline bool
445   push_bottom(T x) noexcept
446   {
447     const __idx b = __bottom.get(memory_order_relaxed);
448     uintptr_t g = __owner_base;
449     if ( __sdiff(b, __top_cache) > static_cast<__sidx>(__owner_mask) ) [[unlikely]]
450       g = __push_full(b, g);
451     if ( g == 0 ) return false;
452     __slots_of(g)[b & __owner_mask].store(x, memory_order_relaxed);
453     __bottom.store(b + 1, memory_order_release);      // also publishes __grow's copy above
454     return true;
455   }
456 
457   // OWNER ONLY
458   [[gnu::always_inline]] inline bool
459   pop_bottom(T &out) noexcept
460   {
461     const __idx b = __bottom.get(memory_order_relaxed) - 1;
462     const uintptr_t g = __owner_base;
463     const __idx mask = __owner_mask;
464     const __idx t = __cl_impl::__cl_dec_bottom_load_top(__bottom, __top, b);
465     if ( __sdiff(b, t) > 0 ) [[likely]] {
466       out = __slots_of(g)[b & mask].get(memory_order_relaxed);
467       return true;
468     }
469     return __pop_cold(out, b, t, g, mask);
470   }
471 
472   [[gnu::always_inline]] inline T
473   pop_bottom() noexcept
474   {
475     T out = __empty;
476     return pop_bottom(out) ? out : __empty;
477   }
478 
479   // ANY NON OWNER
480   [[gnu::always_inline]] inline steal_result<T>
481   try_steal() noexcept
482   {
483     const __idx t = __top.get(memory_order_acquire);
484     const __idx b = __bottom.get(memory_order_seq_cst);
485     if ( __sdiff(b, t) <= 0 ) [[likely]]
486       return { __empty, steal_status::empty, false };
487     return __take_top(t, b);
488   }
489 
490   [[gnu::always_inline]] inline T
491   steal_top() noexcept
492   {
493     return try_steal().__v;
494   }
495 
496   [[gnu::always_inline]] inline T
497   steal_top(steal_status &__st) noexcept
498   {
499     const steal_result<T> r = try_steal();
500     __st = r.__st;
501     return r.__v;
502   }
