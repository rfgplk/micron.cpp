23 inline u32 __secure_exec_state = 0;      // 0 = unprobed, 1 = ordinary, 2 = secure

25 [[gnu::cold, gnu::noinline]] inline bool
26 __secure_exec_probe() noexcept
27 {
28   if ( micron::auxval_has(micron::at_secure) ) return micron::getauxval(micron::at_secure) != 0;
29   // auxv unreadable (no /proc)
30   return posix::getuid() != posix::geteuid() || posix::getgid() != posix::getegid();
31 }

33 inline bool
34 is_secure_exec() noexcept
35 {
36   u32 st = __atomic_load_n(&__secure_exec_state, __ATOMIC_ACQUIRE);
37   if ( st == 0 ) [[unlikely]] {
38     st = __secure_exec_probe() ? 2u : 1u;
39     __atomic_store_n(&__secure_exec_state, st, __ATOMIC_RELEASE);
40   }
41   return st == 2u;
42 }

44 inline bool
45 __segment_is_trusted(const char *seg, usize len) noexcept
46 {
47   if ( len == 0 || seg[0] != '/' ) return false;
48   for ( usize i = 0; i < len; ++i )
49     if ( seg[i] == '$' ) return false;
50   return true;
51 }

53 inline path_str_t
54 __dirname_of(const char *path) noexcept
55 {
56   path_str_t out;
57   if ( !path ) return out;
58   const usize n = micron::strlen(path);
59   usize cut = 0;
60   for ( usize i = n; i-- > 0; ) {
61     if ( path[i] == '/' ) {
62       cut = i;
63       break;
64     }
65   }
66   if ( cut == 0 ) return out;
67   for ( usize i = 0; i < cut && out.size() + 1 < out.max_size(); ++i ) out += path[i];
68   out.null_term();
69   return out;
70 }

72 inline path_str_t
73 __expand_origin(const char *spec, usize len, const char *origin) noexcept
74 {
75   path_str_t out;
76   for ( usize i = 0; i < len && out.size() + 1 < out.max_size(); ) {
77     if ( spec[i] != '$' ) {
78       out += spec[i++];
79       continue;
80     }
81     const bool braced = (i + 1 < len && spec[i + 1] == '{');
82     const usize name_at = i + (braced ? 2 : 1);
83     auto starts = [&](const char *w) {
84       const usize wl = micron::strlen(w);
85       if ( name_at + wl > len ) return false;
86       for ( usize k = 0; k < wl; ++k )
87         if ( spec[name_at + k] != w[k] ) return false;
88       return true;
89     };
90     if ( starts("ORIGIN") ) {
91       if ( origin )
92         for ( usize k = 0; origin[k] && out.size() + 1 < out.max_size(); ++k ) out += origin[k];
93       i = name_at + 6 + (braced ? 1 : 0);
94     } else if ( starts("LIB") ) {
95 
96       i = name_at + 3 + (braced ? 1 : 0);
97     } else if ( starts("PLATFORM") ) {
98       i = name_at + 8 + (braced ? 1 : 0);
99     } else {
100       out += spec[i++];
101     }
102   }
103   out.null_term();
104   return out;
105 }

107 inline path_str_t
108 __try_path_list(const char *list, const char *leaf, const char *origin, bool secure = is_secure_exec()) noexcept
109 {
110   path_str_t hit;
111   if ( !list || !*list ) return hit;
112 
113   const usize n = micron::strlen(list);
114   usize start = 0;
115   while ( start <= n ) {
116     const char *seg_end = micron::format::find(list + start, n - start, ':');
117     const usize end = seg_end ? static_cast<usize>(seg_end - list) : n;
118     if ( end > start && (!secure || __segment_is_trusted(list + start, end - start)) ) {
119       path_str_t dir = __expand_origin(list + start, end - start, origin);
120       if ( !dir.empty() ) {
121         path_str_t cand = __join_path(dir.c_str(), leaf);
122         if ( !cand.empty() && __file_is_native_elf(cand.c_str()) ) return cand;
123       }
124     }
125     if ( !seg_end ) break;
126     start = end + 1;
127   }
128   return hit;
129 }

131 inline path_str_t
132 resolve_dependency(const char *soname, const char *rpath, const char *runpath, const char *origin) noexcept
133 {
134   path_str_t out;
135   if ( !soname || !soname[0] ) return out;
136 
137   const bool secure = is_secure_exec();
138 
139   if ( micron::format::find(soname, micron::strlen(soname), '/') ) {
140     if ( secure && soname[0] != '/' ) return out;      // relative path, attacker-chosen cwd
141     const usize sn = micron::strlen(soname);
142     if ( sn >= out.max_size() ) return path_str_t{};
143     for ( usize i = 0; i < sn; ++i ) out += soname[i];
144     out.null_term();
145     return __file_is_native_elf(out.c_str()) ? out : path_str_t{};
146   }
147 
148   if ( path_str_t h = __try_path_list(rpath, soname, origin, secure); !h.empty() ) return h;
149   if ( !secure )
150     if ( path_str_t h = __try_path_list(env_get("LD_LIBRARY_PATH"), soname, origin, secure); !h.empty() ) return h;
151   if ( path_str_t h = __try_path_list(runpath, soname, origin, secure); !h.empty() ) return h;
152 
153   for ( usize i = 0; i < default_search_path_count; ++i ) {
154     path_str_t cand = __join_path(default_search_paths[i], soname);
155     if ( !cand.empty() && __file_is_native_elf(cand.c_str()) ) return cand;
156   }
157   return path_str_t{};
158 }
