19 inline constexpr usize ver_chain_max = 4096;
20 
21 inline const char *
22 verneed_name(const dyn_info_t &d, half vidx) noexcept
23 {
24   if ( vidx == ver_ndx_local || vidx == ver_ndx_global ) return nullptr;
25   if ( !d.verneed || !d.strtab ) return nullptr;
26 
27   const u8 *vn = reinterpret_cast<const u8 *>(d.verneed);
28   for ( usize guard = 0; guard < ver_chain_max; ++guard ) {
29     const verneed_t *n = micron::ptr_cast<const verneed_t *>(vn);
30     const u8 *va = vn + n->aux;
31     for ( half k = 0; k < n->cnt && k < ver_chain_max; ++k ) {
32       const vernaux_t *a = micron::ptr_cast<const vernaux_t *>(va);
33       if ( elf_ver_ndx(a->other) == vidx ) return d.strtab + a->name;
34       if ( a->next == 0 ) break;
35       va += a->next;
36     }
37     if ( n->next == 0 ) break;
38     vn += n->next;
39   }
40   return nullptr;
41 }

43 inline const char *
44 verdef_name(const dyn_info_t &d, half vidx) noexcept
45 {
46   if ( !d.verdef || !d.strtab ) return nullptr;
47 
48   const u8 *vd = reinterpret_cast<const u8 *>(d.verdef);
49   for ( usize guard = 0; guard < ver_chain_max; ++guard ) {
50     const verdef_t *v = micron::ptr_cast<const verdef_t *>(vd);
51     if ( elf_ver_ndx(v->ndx) == vidx && v->cnt != 0 && v->aux != 0 ) {
52       const verdaux_t *a = micron::ptr_cast<const verdaux_t *>(vd + v->aux);
53       return d.strtab + a->name;
54     }
55     if ( v->next == 0 ) break;
56     vd += v->next;
57   }
58   return nullptr;
59 }

61 inline bool
62 version_matches(const dyn_info_t &def, usize di, const char *want) noexcept
63 {
64   if ( !def.versym ) return true;
65   if ( def.symcount && di >= def.symcount ) return false;
66 
67   const half raw = def.versym[di];
68   const half vidx = elf_ver_ndx(raw);
69   const bool hidden = (raw & ver_ndx_hidden) != 0;
70 
71   if ( vidx == ver_ndx_local ) return false;
72 
73   if ( !want ) return !hidden;
74 
75   if ( vidx == ver_ndx_global ) {
76 
77     return true;
78   }
79 
80   const char *have = verdef_name(def, vidx);
81   if ( !have ) return false;
82   return micron::strcmp(have, want) == 0;
83 }

85 inline const char *
86 wanted_version(const dyn_info_t &ref, u32 si) noexcept
87 {
88   if ( !ref.versym || !ref.symtab ) return nullptr;
89   if ( ref.symcount && si >= ref.symcount ) return nullptr;
90   return verneed_name(ref, elf_ver_ndx(ref.versym[si]));

93 inline const nsym_t *
94 gnu_lookup_versioned(const dyn_info_t &d, const char *name, const char *want) noexcept
95 {
96   using tr = native_traits;
97   using bloom_t = tr::uword;
98   constexpr u32 bb = static_cast<u32>(tr::bloom_bits);
99 
100   if ( !d.gnu_hash || !d.symtab || !d.strtab ) return nullptr;
101   const word *gh = d.gnu_hash;
102   const word nbuckets = gh[0];
103   const word symbias = gh[1];
104   const word bloom_size = gh[2];
105   const word bloom_shift = gh[3];
106   if ( !nbuckets || !bloom_size ) return nullptr;
107 
108   const bloom_t *bloom = reinterpret_cast<const bloom_t *>(gh + 4);
109   const word *buckets = reinterpret_cast<const word *>(bloom + bloom_size);
110   const word *chain = buckets + nbuckets;
111 
112   const u32 h = gnu_hash(name);
113   const bloom_t bw = bloom[(h / bb) % bloom_size];
114   const bloom_t bit_a = static_cast<bloom_t>(1) << (h % bb);
115   // mask by the bloom word width, word size varies on arch spec
116   const bloom_t bit_b = static_cast<bloom_t>(1) << ((static_cast<bloom_t>(h) >> (bloom_shift & (bb - 1))) % bb);
117   if ( (bw & (bit_a | bit_b)) != (bit_a | bit_b) ) return nullptr;
118 
119   word idx = buckets[h % nbuckets];
120   if ( idx < symbias ) return nullptr;
121 
122   const word maxidx = d.symcount ? static_cast<word>(d.symcount) : (idx + (static_cast<word>(1) << 24));
123   for ( ; idx < maxidx; ++idx ) {
124     const word chain_h = chain[idx - symbias];
125     if ( ((chain_h ^ h) >> 1) == 0 ) {
126       const nsym_t &s = d.symtab[idx];
127       if ( __streq(d.strtab + s.name, name) && version_matches(d, idx, want) ) return &s;
128     }
129     if ( chain_h & 1 ) break;
130   }
131   return nullptr;
132 }

134 inline const nsym_t *
135 sysv_lookup_versioned(const dyn_info_t &d, const char *name, const char *want) noexcept
136 {
137   if ( !d.hash || !d.symtab || !d.strtab ) return nullptr;
138   const word nbuckets = d.hash[0];
139   const word nchain = d.hash[1];
140   if ( !nbuckets ) return nullptr;
141   const word *buckets = d.hash + 2;
142   const word *chain = buckets + nbuckets;
143 
144   const u32 h = sysv_hash(name);
145   word i = buckets[h % nbuckets];
146   for ( word steps = 0; i != 0 && i < nchain && steps < nchain; i = chain[i], ++steps ) {
147     const nsym_t &s = d.symtab[i];
148     if ( __streq(d.strtab + s.name, name) && version_matches(d, i, want) ) return &s;
149   }
150   return nullptr;
151 }

153 inline const nsym_t *
154 lookup_versioned(const dyn_info_t &d, const char *name, const char *want) noexcept
155 {
156   if ( const nsym_t *s = gnu_lookup_versioned(d, name, want) ) return s;
157   return sysv_lookup_versioned(d, name, want);
158 }
