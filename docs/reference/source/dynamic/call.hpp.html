21 struct dynamic_t {
22   u32 index = 0;
23   u32 generation = 0;
24 
25   explicit constexpr
26   operator bool() const noexcept
27   {
28     return generation != 0;
29   }
30 
31   constexpr bool
32   operator==(const dynamic_t &o) const noexcept
33   {
34     return index == o.index && generation == o.generation;
35   }
36 };

47 inline dynamic_t
48 dynamic_open(const char *soname_or_path, rtld flags = rtld::now | rtld::local, dynamic_options options = {})
49 {
50   using namespace micron::elf::dl;
51   __err_clear();
52 
53   if ( !soname_or_path || !*soname_or_path ) exc<except::library_error>("dynamic_open: empty name");
54 
55   micron::lock_guard __lk(__registry_mtx);
56 
57   if ( dl_module *hit = __find_by_path(soname_or_path) ) {
58     if ( hit->options.resolve != options.resolve || hit->options.validate != options.validate || hit->options.context != options.context )
59       exc<except::library_error>("dynamic_open: incompatible host resolver for existing module");
60     ++hit->refcount;
61     if ( has(flags, rtld::global) ) hit->global = true;
62     if ( has(flags, rtld::nodelete) ) hit->nodelete = true;
63     return dynamic_t{ hit->index, hit->generation };
64   }
65 
66   const elf::path_str_t p = resolve_dependency(soname_or_path, nullptr, nullptr, nullptr);
67   if ( p.empty() ) {
68     __err_set_once("dynamic_open: not found", soname_or_path);
69     if ( has(flags, rtld::noload) ) return dynamic_t{};
70     exc<except::library_error>("dynamic_open: library not found in any search path");
71   }
72 
73   if ( has(flags, rtld::noload) ) {
74     u64 dev = 0, ino = 0;
75     if ( !__file_ident(p.c_str(), dev, ino) ) return dynamic_t{};
76     dl_module *hit = __find_by_ident(dev, ino);
77     if ( !hit ) return dynamic_t{};
78     ++hit->refcount;
79     return dynamic_t{ hit->index, hit->generation };
80   }
81 
82   load_state st;
83   st.options = options;
84   dl_module *root = __map_recursive(p.c_str(), flags, st, 0);
85   if ( !root || st.failed ) {
86     __unwind(st);
87     exc<except::library_error>("dynamic_open: failed to map the dependency closure");
88   }
89 
90   if ( !__relocate_closure(root) ) {
91     __unwind(st);
92     exc<except::library_error>("dynamic_open: relocation failed");
93   }
94 
95   elf::dl::__debug_publish(elf::dl::rt_add);
96   elf::dl::__debug_rebuild_chain();
97   elf::dl::__debug_publish(elf::dl::rt_consistent);
98 
99   __init_closure(root);
100 
101   return dynamic_t{ root->index, root->generation };
102 }

111 // Run finalizers once, retaining mappings for callbacks used during host shutdown.
112 inline bool
113 dynamic_finalize(dynamic_t d) noexcept
114 {
115   using namespace micron::elf::dl;
116   micron::lock_guard __lk(__registry_mtx);
117   auto *m = __slot_at(d.index, d.generation);
118   if ( !m ) return false;
119   __finalize_closure(m);
120   return true;
121 }

123 inline bool
124 dynamic_close(dynamic_t d) noexcept
125 {
126   using namespace micron::elf::dl;
127   __err_clear();
128   if ( d.generation == 0xffffffffu ) return false;
129 
130   micron::lock_guard __lk(__registry_mtx);
131   dl_module *m = __slot_at(d.index, d.generation);
132   if ( !m ) {
133     __err_set_once("dynamic_close: stale or invalid handle");
134     return false;
135   }
136   const bool ok = __close_module(m);
137 
138   __debug_publish(rt_delete);
139   __debug_rebuild_chain();
140   __debug_publish(rt_consistent);
141   return ok;
142 }

144 inline void *
145 dynamic_sym(dynamic_t d, const char *name) noexcept
146 {
147   using namespace micron::elf::dl;
148   __err_clear();
149   if ( !name || !*name ) return nullptr;
150 
151   micron::lock_guard __lk(__registry_mtx);
152 
153   if ( d == dynamic_default ) {
154     if ( found_t f = __lookup_global(name, nullptr) ) return f.addr;
155     return __host_resolve(name);
156   }
157 
158   if ( d == dynamic_next ) {
159 
160     dl_module *caller = __find_by_address(__builtin_return_address(0));
161     if ( !caller ) {
162       __err_set_once("dynamic_sym: RTLD_NEXT called from outside any loaded module");
163       return nullptr;
164     }
165     if ( found_t f = __lookup_global(name, nullptr, caller->load_order) ) return f.addr;
166     __err_set_once("dynamic_sym: no next definition", name);
167     return nullptr;
168   }
169 
170   dl_module *m = __slot_at(d.index, d.generation);
171   if ( !m ) {
172     __err_set_once("dynamic_sym: stale or invalid handle");
173     return nullptr;
174   }
175 
176   if ( found_t f = __lookup_closure(m, name, nullptr) ) return f.addr;
177   __err_set_once("dynamic_sym: symbol not found", name);
178   return nullptr;
179 }

231 template<class R = void, class... Args>
232 inline R
233 dynamic_call(dynamic_t d, const char *name, Args... args)
234 {
235   using fn_t = R (*)(Args...);
236   fn_t fn = dynamic_sym_as<fn_t>(d, name);
237   if ( !fn ) exc<except::library_error>("dynamic_call: symbol not found");
238   if constexpr ( micron::is_void_v<R> ) {
239     fn(static_cast<Args>(args)...);
240     return;
241   } else {
242     return fn(static_cast<Args>(args)...);
243   }
244 }
