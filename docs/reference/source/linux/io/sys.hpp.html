227 inline pipe_pair
228 make_pipe(i32 flags = 0)
229 {
230   i32 fds[2];
231   i32 r = static_cast<i32>(micron::syscall(SYS_pipe2, fds, flags));

245 inline i32
246 dup2(i32 old, i32 newfd)
247 {
248 #if defined(__micron_syscall_generic)
249   // WARNING: arm64 has no dup2; dup3 EINVALs when old==newfd, whereas dup2 returns newfd if old is a valid fd
250   if ( old == newfd ) {
251     long __r = micron::syscall(SYS_fcntl, old, f_getfd, 0);
252     return static_cast<i32>(__r < 0 ? __r : static_cast<long>(newfd));
253   }
254   return static_cast<i32>(micron::syscall(SYS_dup3, old, newfd, 0));
255 #else
256   return static_cast<i32>(micron::syscall(SYS_dup2, old, newfd));
257 #endif
258 }

640 inline posix::off64_t
641 lseek(i32 fd, posix::off64_t offset, i32 whence)
642 {
643 #if defined(__micron_arch_width_32)
644   posix::off64_t result = 0;
645   long r = micron::syscall(SYS__llseek, fd, static_cast<u32>(static_cast<u64>(offset) >> 32),
646                            static_cast<u32>(static_cast<u64>(offset) & 0xffffffffu), &result, whence);
647   if ( micron::syscall_failed(r) ) return static_cast<posix::off64_t>(r);      // raw -errno
648   return result;
649 #else
650   return micron::syscall(SYS_lseek, fd, offset, whence);
651 #endif
652 }
