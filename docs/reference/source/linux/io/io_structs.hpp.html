104 inline u16
105 __dirent64_validate(const void *buf, usize pos, usize nread, const __linux_kernel_dirent64 *&out) noexcept
106 {
107   constexpr usize hdr = __builtin_offsetof(__linux_kernel_dirent64, d_name);
108   if ( pos >= nread || pos + hdr > nread ) return 0;
109   out = micron::ptr_cast<const __linux_kernel_dirent64 *>(static_cast<const char *>(buf) + pos);
110   const u16 reclen = out->d_reclen;
111   if ( reclen <= hdr ) return 0;
112   if ( (reclen % alignof(__linux_kernel_dirent64)) != 0 ) return 0;
113   if ( pos + reclen > nread ) return 0;
114   const usize navail = reclen - hdr;
115   const usize scan = navail < (name_max + 1) ? navail : (name_max + 1);
116   for ( usize i = 0; i < scan; ++i )
117     if ( out->d_name[i] == '\0' ) return reclen;
118   return 0;
119 }
