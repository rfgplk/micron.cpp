38   auto_fd(auto_fd &&o) noexcept : fd(o.fd) { o.fd.reset(); }

40   auto_fd &
41   operator=(auto_fd &&o) noexcept
42   {
43     if ( this == &o ) return *this;
44     if ( fd.open() ) posix::close(fd.fd);

46     o.fd.reset();

189 inline __impl_dir
190 readdir(const fd_t &_f)
191 {
192   alignas(__linux_kernel_dirent64) static char __readdir_buf[8192];
193   static usize __readdir_bufpos = 0;
194   static usize __readdir_nread = 0;
195   static i32 __readdir_last_fd = -1;
196 
197   if ( _f.fd != __readdir_last_fd ) {
198     __readdir_bufpos = 0;
199     __readdir_nread = 0;
200     __readdir_last_fd = _f.fd;
201   }
202 
203   if ( __readdir_bufpos >= __readdir_nread ) {
204     max_t __nr = posix::getdents64(_f.fd, &__readdir_buf, sizeof(__readdir_buf));
205     __readdir_bufpos = 0;
206     if ( __nr <= 0 ) {
207       __readdir_nread = 0;
208       return __impl_dir{ "", dt_end, 0 };
209     }
210     __readdir_nread = static_cast<usize>(__nr);
211   }
212 
213   const __linux_kernel_dirent64 *p = nullptr;
214   const u16 reclen = __dirent64_validate(__readdir_buf, __readdir_bufpos, __readdir_nread, p);
215   if ( reclen == 0 ) {      // malformed/hostile record: stop iterating
216     __readdir_bufpos = __readdir_nread;
217     return __impl_dir{ "", dt_end, 0 };
218   }
219   __readdir_bufpos += reclen;
220   return { p->d_name, p->d_type, p->d_ino };
221 }

229 inline __impl_dir
230 readdir_r(const fd_t &_f, readdir_ctx &ctx)
231 {
232   if ( ctx.bufpos >= ctx.nread ) {
233     max_t __nr = posix::getdents64(_f.fd, ctx.buf, sizeof(ctx.buf));
234     ctx.bufpos = 0;
235     if ( __nr <= 0 ) {
236       ctx.nread = 0;
237       return { "", dt_end, 0 };
238     }
239     ctx.nread = static_cast<usize>(__nr);
240   }
241 
242   const __linux_kernel_dirent64 *p = nullptr;
243   const u16 reclen = __dirent64_validate(ctx.buf, ctx.bufpos, ctx.nread, p);
244   if ( reclen == 0 ) {      // malformed/hostile record: stop iterating
245     ctx.bufpos = ctx.nread;
246     return { "", dt_end, 0 };
247   }
248   ctx.bufpos += reclen;
249   return { p->d_name, p->d_type, p->d_ino };
250 }

264 // mkdir -p equiv, make all parents
265 inline i32
266 mkdir_p(const char *path, u32 mode = mode_dir)
267 {
268   char buf[path_max];
269   usize i = 0;
270   while ( path[i] && i < path_max - 1 ) {
271     buf[i] = path[i];
272     ++i;
273   }
274   buf[i] = '\0';
275 
276   for ( usize j = 1; j <= i; ++j ) {
277     if ( buf[j] == '/' || buf[j] == '\0' ) {
278       char saved = buf[j];
279       buf[j] = '\0';
280       i32 r = posix::mkdir(buf, mode);
281       // eexist, not loading in errno
282       if ( r < 0 && static_cast<u32>(-r) != 17u ) return r;
283       buf[j] = saved;
284     }
285   }
286   return 0;
287 }

303 template<typename T>
304   requires(!micron::is_pointer_v<T>)
305 inline max_t
306 write_all(fd_t fd, const T &buf, usize len)
307 {
308   usize written = 0;
309   const byte *p = real_addr_as<const byte>(buf);
310   while ( written < len ) {
311     max_t r = posix::write(fd.fd, p + written, len - written);
312     if ( r < 0 ) {
313       if ( r == -error::interrupted ) continue;      // retry on EINTR rather than truncating
314       return r;
315     }
316     if ( r == 0 ) break;
317     written += static_cast<usize>(r);
318   }
319   return static_cast<max_t>(written);
320 }

400 inline max_t
401 copy_fd(fd_t out_fd, fd_t in_fd, usize count)
402 {
403   usize done = 0;
404   while ( done < count ) {
405     max_t r = posix::sendfile(out_fd.fd, in_fd.fd, nullptr, count - done);
406     if ( r < 0 ) {
407       if ( r == -error::interrupted ) continue;      // retry on EINTR
408       return r;
409     }
410     if ( r == 0 ) break;      // source exhausted
411     done += static_cast<usize>(r);
412   }
413   return static_cast<max_t>(done);
414 }

416 inline i32
417 copy_file(const char *src, const char *dst, u32 mode = mode_file)
418 {
419   fd_t in = open_read(src);
420   if ( !in ) return in.fd;
421 
422   fd_t out = open_write(dst, mode);
423   if ( !out ) {
424     close_fd(in);
425     return out.fd;
426   }
427 
428   posix::off64_t sz = file_size(in);
429   i32 r = 0;
430   if ( sz < 0 ) {
431     r = static_cast<i32>(sz);
432   } else if ( sz > 0 ) {
433     max_t sent = copy_fd(out, in, static_cast<usize>(sz));
434     r = (sent < 0) ? static_cast<i32>(sent) : 0;
435   }
436   close_fd(in);
437   close_fd(out);
438   return r;
439 }

486 template<typename Fn>
487 inline void
488 for_each_entry(const char *path, Fn &&fn)
489 {
490   dir_handle dh(path);
491   if ( !dh ) return;
492 
493   readdir_ctx ctx{};
494   for ( ;; ) {
495     __impl_dir e = readdir_r(dh.fd, ctx);
496     if ( e.at_end() ) break;
497     // Skip self and parent entries
498     if ( e.d_name[0] == '.' && (e.d_name[1] == '\0' || (e.d_name[1] == '.' && e.d_name[2] == '\0')) ) continue;
499     fn(e);
500   }
501 }
