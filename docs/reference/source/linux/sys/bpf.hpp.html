73 struct insn_t {
74   u16 code;
75   u8 jt;      // instructions to skip on true branch
76   u8 jf;      // instructions to skip on false branch
77   u32 k;      // immediate / offset operand
78 };
79 
80 // BPF_STMT equivalent: instruction with no jump
81 [[nodiscard]] constexpr insn_t
82 stmt(u16 code, u32 kval) noexcept
83 {
84   return insn_t{ code, 0, 0, kval };
85 }
86 
87 // BPF_JUMP equivalent: conditional jump
88 [[nodiscard]] constexpr insn_t
89 jump(u16 code, u32 kval, u8 jt, u8 jf) noexcept
90 {
91   return insn_t{ code, jt, jf, kval };
92 }
