52 inline usize
53 __read_node(const char *path, char *buf, usize cap)
54 {
55   buf[0] = '\0';
56   i32 fd = __open_rd(path);
57   if ( fd < 0 ) return 0;
58   usize off = 0;
59   for ( ;; ) {
60     if ( off >= cap - 1 ) break;
61     max_t n = posix::read(fd, buf + off, cap - 1 - off);
62     if ( n < 0 ) {
63       if ( n == -error::interrupted ) continue;
64       break;
65     }
66     if ( n == 0 ) break;      // EOF
67     off += static_cast<usize>(n);
68   }
69   __close(fd);
70   buf[off] = '\0';
71   // strip trailing newline
72   if ( off > 0 && buf[off - 1] == '\n' ) {
73     buf[off - 1] = '\0';
74     --off;
75   }
76   return off;
77 }

79 inline bool
80 __write_node(const char *path, const char *data, usize len)
81 {
82   i32 fd = __open_wr(path);
83   if ( fd < 0 ) return false;
84   usize off = 0;
85   while ( off < len ) {
86     max_t n = posix::write(fd, const_cast<char *>(data) + off, len - off);
87     if ( n < 0 ) {
88       if ( n == -error::interrupted ) continue;
89       __close(fd);
90       return false;      // real write error (the old code reported success regardless)
91     }
92     if ( n == 0 ) break;
93     off += static_cast<usize>(n);
94   }
95   __close(fd);
96   return off == len;
97 }

99 inline u64
100 __parse_u64(const char *buf)
101 {
102   u64 val = 0;
103   const char *p = buf;
104   while ( *p == ' ' || *p == '\t' || *p == '\n' ) ++p;
105   while ( *p >= '0' && *p <= '9' ) val = val * 10 + static_cast<u64>(*p++ - '0');
106   return val;
107 }

109 inline u32
110 __parse_range_count(const char *buf)
111 {
112   const char *p = buf;
113   u32 total = 0;
114   while ( *p ) {
115     while ( *p == ' ' || *p == '\t' || *p == '\n' ) ++p;
116     if ( *p < '0' || *p > '9' ) break;
117     u32 lo = 0;
118     while ( *p >= '0' && *p <= '9' ) lo = lo * 10 + static_cast<u32>(*p++ - '0');
119     u32 hi = lo;
120     if ( *p == '-' ) {
121       ++p;
122       hi = 0;
123       while ( *p >= '0' && *p <= '9' ) hi = hi * 10 + static_cast<u32>(*p++ - '0');
124     }
125     if ( hi >= lo ) total += hi - lo + 1;
126     if ( *p == ',' )
127       ++p;
128     else
129       break;
130   }
131   return total;
132 }

134 inline u64
135 __parse_size(const char *buf)
136 {
137   u64 val = 0;
138   const char *p = buf;
139   while ( *p >= '0' && *p <= '9' ) val = val * 10 + static_cast<u64>(*p++ - '0');
140   if ( *p == 'K' || *p == 'k' )
141     val *= 1024ULL;
142   else if ( *p == 'M' || *p == 'm' )
143     val *= 1024ULL * 1024ULL;
144   else if ( *p == 'G' || *p == 'g' )
145     val *= 1024ULL * 1024ULL * 1024ULL;
146   return val;
147 }

196 inline i64
197 read_i64(const char *path)
198 {
199   char buf[64];
200   usize n = __impl::__read_node(path, buf, 64);
201   if ( n == 0 ) return 0;
202   const char *p = buf;
203   bool neg = false;
204   if ( *p == '-' ) {
205     neg = true;
206     ++p;
207   }
208   u64 val = __impl::__parse_u64(p);
209   return neg ? -static_cast<i64>(val) : static_cast<i64>(val);
210 }

544 inline u16
545 __parse_type(const char *s)
546 {
547   if ( s[0] == 'D' ) return data_cache;
548   if ( s[0] == 'I' ) return instruction_cache;
549   if ( s[0] == 'U' ) return unified_cache;
550   return no_cache;
551 }

553 inline info_t
554 read_info(u32 cpu_id, u32 index)
555 {
556   info_t c;
557   auto ts = type_str(cpu_id, index);
558   c.type = __parse_type(&ts[0]);
559 
560   if ( c.type == no_cache ) {
561     c.level = 0;
562     c.line_size = 0;
563     c.associativity = 0;
564     c.sets = 0;
565     c.size = 0;
566     return c;
567   }
568 
569   c.level = static_cast<u16>(level(cpu_id, index));
570   c.line_size = coherency_line_size(cpu_id, index);
571   c.associativity = ways_of_associativity(cpu_id, index);
572   c.sets = number_of_sets(cpu_id, index);
573   c.size = size(cpu_id, index);
574 
575   if ( c.size == 0 && c.line_size > 0 ) c.size = static_cast<u64>(c.line_size) * c.associativity * c.sets;
576 
577   return c;
578 }
