60 inline usize
61 getpagesizelive(void) noexcept
62 {
63   static usize cached = 0;
64   if ( cached ) return cached;
65   usize ps = static_cast<usize>(getpagesize());      // fallback if the probe fails
66   long fd = micron::syscall(SYS_openat, -100 /* AT_FDCWD */, "/proc/self/auxv", 0 /* O_RDONLY */, 0);
67   if ( !micron::syscall_failed(fd) ) {
68     unsigned long e[2];      // each auxv entry is { a_type, a_val }, both unsigned long
69     for ( ;; ) {
70       long n = micron::syscall(SYS_read, fd, &e[0], sizeof(e));
71       if ( n != static_cast<long>(sizeof(e)) ) break;
72       if ( e[0] == 6UL ) {      // AT_PAGESZ
73         if ( e[1] ) ps = static_cast<usize>(e[1]);
74         break;
75       }
76       if ( e[0] == 0UL ) break;      // AT_NULL terminates the auxv
77     }
78     micron::syscall(SYS_close, fd);
79   }
80   cached = ps;
81   return ps;
82 }

84 inline long int
85 get_phys_pages(void)
86 {
87   sysinfo sys;
88   unsigned long int _page_size = getpagesizelive();
89   while ( sys.mem_unit > 1 && _page_size > 1 ) {
90     sys.mem_unit >>= 1;
91     _page_size >>= 1;
92   }
93   sys.totalram *= sys.mem_unit;
94   while ( _page_size > 1 ) {
95     _page_size >>= 1;
96     sys.totalram >>= 1;
97   }
98   return static_cast<long int>(sys.totalram);
99 }
