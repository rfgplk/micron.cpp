26 template<typename T>
27 concept __syscall_arg = (micron::is_integral_v<T> || micron::is_pointer_v<T> || micron::is_enum_v<T> || micron::is_null_pointer_v<T>)
28                         && sizeof(T) <= sizeof(long int);
29 
30 template<__syscall_arg T>
31 inline __attribute__((always_inline)) long int
32 __coerce(T __v) noexcept
33 {
34   if constexpr ( micron::is_null_pointer_v<T> )
35     return 0L;
36   else if constexpr ( micron::is_pointer_v<T> )
37     return static_cast<long int>(reinterpret_cast<__tt_size_t>(__v));
38   else
39     return static_cast<long int>(__v);
40 }

44 inline __attribute__((always_inline)) long int
45 __do_syscall(long int __n) noexcept
46 {
47   register long int __x8 asm("x8") = __n;
48   register long int __x0 asm("x0");
49   asm volatile("svc #0" : "=r"(__x0) : "r"(__x8) : "memory");
50   return __x0;
51 }
