77 inline int
78 poll(pollfd &pfd, nfds_t nfds, int timeout)
79 {
80 #if defined(__micron_syscall_generic)
81   // arm64 has no poll
82   if ( timeout < 0 ) return static_cast<int>(micron::syscall(SYS_ppoll, &pfd, nfds, nullptr, nullptr, __sig_syscall_size));
83   timespec_t __ts{};
84   __ts.tv_sec = timeout / 1000;
85   __ts.tv_nsec = static_cast<decltype(__ts.tv_nsec)>(timeout % 1000) * 1000000;
86   return static_cast<int>(micron::syscall(SYS_ppoll, &pfd, nfds, &__ts, nullptr, __sig_syscall_size));
87 #else
88   return static_cast<int>(micron::syscall(SYS_poll, &pfd, nfds, timeout));
89 #endif
90 }

171 inline int
172 epoll_pwait2(int epfd, epoll_event *events, int maxevents, const timespec_t *timeout, const sigset_t *sigmask)
173 {
174   if ( __epoll_pwait2_gate.open(kernel::feature::epoll_pwait2) ) [[likely]] {
175     int r = static_cast<int>(micron::syscall(SYS_epoll_pwait2, epfd, events, maxevents, timeout, sigmask, __sig_syscall_size));
176     if ( r != -ENOSYS ) [[likely]]
177       return r;
178     __epoll_pwait2_gate.demote();
179   }
180   int ms = -1;      // nullptr -> block indefinitely
181   if ( timeout != nullptr ) {
182     if ( timeout->tv_sec < 0 || static_cast<u64>(timeout->tv_nsec) >= 1000000000ull ) return -error::invalid_arg;
183     constexpr i64 __ms_max = 0x7fffffff;
184     const i64 __sec = static_cast<i64>(timeout->tv_sec);
185     // nonzero wait must not spin as 0
186     const i64 v = __sec > __ms_max / 1000 ? __ms_max : __sec * 1000 + (static_cast<i64>(timeout->tv_nsec) + 999999) / 1000000;
187     ms = static_cast<int>(v > __ms_max ? __ms_max : v);
188   }
189   return epoll_pwait(epfd, events, maxevents, ms, sigmask);
190 }
