119 struct stack_region {
120   addr_t *low;       // lowest usable byte -> clone_args.stack
121   usize usable;      // bytes the kernel may use (it computes the top)
122 };
123 
124 template<typename T>
125 inline stack_region
126 guard_stack(T *region, usize size) noexcept
127 {
128   constexpr usize guard = __micron_page_size_default;
129   addr_t *base = reinterpret_cast<addr_t *>(region);
130   usize usable = size;
131   // mprotect needs a page-aligned addr; if the region isn't aligned (e.g. an in-object stack), skip
132   if ( size > guard && micron::mprotect(reinterpret_cast<addr_t *>(region), guard, micron::prot_none) == 0 ) {
133     base = micron::ptr_cast<addr_t *>(reinterpret_cast<char *>(region) + guard);
134     usable = size - guard;
135   }
136   return { base, usable };
137 }

139 template<auto KFn, typename... KArgs>
140 inline tid_t
141 spawn(addr_t *stack_low, usize usable, unsigned long tls, int *ctid, KArgs &&...kargs)
142 {
143   using payload_t = posix::__clone_payload<KFn, micron::decay_t<KArgs>...>;
144   auto *p = new payload_t{ { reinterpret_cast<void *>(new micron::decay_t<KArgs>(static_cast<KArgs &&>(kargs)))... } };
145 
146   pid_t tid = posix::clone3_spawn(&payload_t::thunk, p, stack_low, usable, tls, nullptr, ctid);
147   if ( micron::syscall_failed(tid) ) {
148     payload_t::destroy(p);      // no child ran; reclaim the payload + arg copies
149     errno = static_cast<int>(micron::syscall_errno(tid));
150     return thread_failed;
151   }
152   return static_cast<tid_t>(tid);
153 }

163 [[nodiscard]] inline tier
164 level(void) noexcept
165 {
166   if ( !micron::__tls_ensure_template() ) return tier::none;
167   return posix::__clone3_gate.open(kernel::feature::clone3) ? tier::full : tier::legacy;
168 }

170 [[nodiscard]] inline bool
171 available(void) noexcept
172 {
173   return level() != tier::none;

176 inline int
177 join(int *ctid) noexcept
178 {
179   for ( ;; ) {
180     int v = atom::load(ctid, static_cast<int>(memory_order_acquire));
181     if ( v == 0 ) return 0;
182     // NOTE: deliberately NON-private FUTEX_WAIT (no futex_private_flag): the kernel's CHILD_CLEARTID
183     // wake on thread exit is a shared-class wake, so a private waiter would miss it.
184     auto r = micron::__futex(reinterpret_cast<u32 *>(ctid), futex_wait, static_cast<u32>(v), nullptr, nullptr, 0);
185     (void)r;      // -EAGAIN (value changed) / -EINTR are benign; loop re-checks
186   }
187 }

189 // non-blocking: returns 0 if the thread has exited (ctid cleared), error::busy if still running
190 inline int
191 try_join(int *ctid) noexcept
192 {
193   return atom::load(ctid, static_cast<int>(memory_order_acquire)) == 0 ? 0 : static_cast<int>(error::busy);
194 }

196 // we name threads via /proc
197 inline void
198 set_name(tid_t tid, const char *name) noexcept
199 {
200   // simple but dirty
201   char path[64];
202   char *q = path;
203   for ( const char *s = "/proc/self/task/"; *s; ++s ) *q++ = *s;
204   // itoa tid
205   char num[16];
206   int n = 0;
207   unsigned t = static_cast<unsigned>(tid);
208   if ( t == 0 ) num[n++] = '0';
209   while ( t ) {
210     num[n++] = static_cast<char>('0' + (t % 10));
211     t /= 10;
212   }
213   while ( n-- ) *q++ = num[n];
214   for ( const char *s = "/comm"; *s; ++s ) *q++ = *s;
215   *q = '\0';
216 
217   i32 fd = posix::open(path, posix::o_wronly);
218   if ( fd < 0 ) return;
219   usize len = 0;
220   while ( name[len] && len < 15 ) ++len;
221   posix::write(fd, name, len);
222   posix::close(fd);
223 }

225 inline auto
226 get_name(tid_t, char *buf, usize sz) noexcept
227 {
228   char tmp[16] = {};
229   micron::prctl(micron::PR_GET_NAME, reinterpret_cast<unsigned long>(tmp));
230   usize i = 0;
231   for ( ; i + 1 < sz && tmp[i]; ++i ) buf[i] = tmp[i];
232   if ( sz ) buf[i] = '\0';
233   return 0;
234 }
