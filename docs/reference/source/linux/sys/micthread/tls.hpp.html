27 // main thread TLS block
28 // (tp -> clone_args.tls) installed via CLONE_SETTLS before thread execs
29 struct __tls_frame {
30   byte *base = nullptr;      // page-aligned
31   usize size = 0;            // page-rounded in bytes
32   byte *tp = nullptr;        // thread pointer (TCB start) [%fs/%gs/TPIDR]
33   u64 image_size = 0;        // PT_TLS memsz rounded to p_align
34 };
35 
36 struct __tls_template_t {
37   const byte *image = nullptr;      // .tdata source
38   u64 filesz = 0;                   // initialised bytes to copy
39   u64 memsz = 0;                    // total tls size (filesz + tbss)
40   u64 align = 0;                    // PT_TLS p_align
41   usize pagesz = 4096;
42   bool valid = false;
43 };

61 inline __attribute__((always_inline)) constexpr u64
62 __tls_round_up(u64 v, u64 a) noexcept
63 {
64   return (v + (a - 1)) & ~(a - 1);
65 }

74 inline bool
75 __tls_capture_from_proc_auxv() noexcept
76 {
77 #if defined(MICRON_TLS_NO_PROC)
78   return false;
79 #else
80   // WARNING: auxv is usually small, but this _might_ overflow, be careful!
81   alignas(16) unsigned long buf[512];
82   i32 fd = micron::posix::open("/proc/self/auxv", micron::posix::o_rdonly);
83   if ( fd < 0 ) return false;
84   usize total = 0;
85   for ( ;; ) {
86     max_t n = micron::posix::read(fd, reinterpret_cast<char *>(buf) + total, sizeof(buf) - total);
87     if ( n <= 0 ) break;
88     total += static_cast<usize>(n);
89     if ( total >= sizeof(buf) ) break;
90   }
91   micron::posix::close(fd);
92 
93   const usize pairs = total / (2 * sizeof(unsigned long));
94   unsigned long phdr_addr = 0, phent = 0, phnum = 0, pagesz = 0;
95   for ( usize i = 0; i < pairs; ++i ) {
96     unsigned long t = buf[2 * i], v = buf[2 * i + 1];
97     if ( t == at_null ) break;
98     if ( t == at_phdr )
99       phdr_addr = v;
100     else if ( t == at_phent )
101       phent = v;
102     else if ( t == at_phnum )
103       phnum = v;
104     else if ( t == at_pagesz )
105       pagesz = v;
106   }
107   if ( phdr_addr == 0 || phent == 0 || phnum == 0 ) return false;
108 
109   const tls_image img = find_tls_in_phdrs(phdr_addr, phent, phnum);
110   __micron_tls_template
111       = __tls_template_t{ img.image, img.filesz, img.memsz, img.align ? img.align : __micron_tls_min_align, pagesz ? pagesz : 4096, true };
112   return true;
113 #endif
114 }

116 inline bool
117 __tls_capture_from_ehdr() noexcept
118 {
119   if ( !__ehdr_usable() ) return false;
120   const tls_image img = find_tls_from_ehdr();
121   __micron_tls_template = __tls_template_t{ img.image, img.filesz, img.memsz, img.align ? img.align : __micron_tls_min_align, 4096, true };
122   return true;

125 // called by start/__tls in freestanding mode)
126 inline void
127 __tls_capture_template(const byte *image, u64 filesz, u64 memsz, u64 align, usize pagesz) noexcept
128 {
129   __micron_tls_template = __tls_template_t{ image, filesz, memsz, align ? align : __micron_tls_min_align, pagesz ? pagesz : 4096, true };
130   atom::store(__micron_tls_template_state.ptr(), __tls_tmpl_ready, static_cast<int>(memory_order_release));
131 }

133 inline bool
134 __tls_ensure_template() noexcept
135 {
136   if ( atom::load(__micron_tls_template_state.ptr(), static_cast<int>(memory_order_acquire)) == __tls_tmpl_ready ) return true;
137   u32 expected = __tls_tmpl_uninit;
138   if ( __micron_tls_template_state.compare_exchange_strong(expected, __tls_tmpl_capturing, memory_order_acq_rel, memory_order_acquire) ) {
139 #if defined(MICRON_ATTACH_MODULE)
140     const bool ok = __micron_tls_template.valid ? true : (__tls_capture_from_proc_auxv() or __tls_capture_from_ehdr());
141 #else
142     const bool ok = __micron_tls_template.valid ? true : (__tls_capture_from_ehdr() or __tls_capture_from_proc_auxv());
143 #endif
144     atom::store(__micron_tls_template_state.ptr(), ok ? __tls_tmpl_ready : __tls_tmpl_uninit, static_cast<int>(memory_order_release));
145     return ok;
146   }
147   while ( atom::load(__micron_tls_template_state.ptr(), static_cast<int>(memory_order_acquire)) == __tls_tmpl_capturing ) {
148   }
149   return atom::load(__micron_tls_template_state.ptr(), static_cast<int>(memory_order_acquire)) == __tls_tmpl_ready;
150 }

152 #if defined(MICRON_ENABLE_ATTACH)
153 inline bool
154 __attach_host_tls_init() noexcept
155 {
156   if constexpr ( __micron_tls_surplus == 0 ) {
157     return false;
158   } else {
159     if ( __attach_host_tls_ready() ) return true;
160     if ( !__tls_ensure_template() ) return false;
161     byte *tp = __attach_read_tp();
162     if ( tp == nullptr ) return false;
163     const i64 surplus_tpoff = static_cast<i64>(&__micron_attach_surplus[0] - tp);
164     const __tls_template_t &t = __micron_tls_template;
165     const u64 p_align = t.align ? t.align : __micron_tls_min_align;
166     return __attach_host_tls_record(surplus_tpoff, __tls_round_up(t.memsz, p_align));
167   }
168 }
169 #endif

171 // this doesnt install the TP (CLONE_SETTLS in clone3 will)
172 inline __tls_frame
173 __tls_make_frame(const byte *image, u64 filesz, u64 memsz, u64 align, usize page_sz) noexcept
174 {
175   __tls_frame f{};
176   if ( page_sz == 0 ) page_sz = 4096;
177   const u64 p_align = align ? align : __micron_tls_min_align;
178   if ( p_align > page_sz ) return f;      // bad align
179   const u64 block = __tls_round_up(memsz, p_align);
180 
181   // NOTE: no surplus region
182 #if defined(__micron_arch_amd64) || defined(__micron_arch_x86)
183   // Variant II: [ image_block | TCB ], tp = base + block
184   const u64 alloc = __tls_round_up(block + __micron_tcb_sz, page_sz);
185   byte *base = __tls_raw_mmap(static_cast<usize>(alloc));
186   if ( !base ) return f;
187   byte *tp = base + block;
188   byte *image_dst = base;
189   for ( u64 i = 0; i < filesz; ++i ) image_dst[i] = image[i];
190   *reinterpret_cast<void **>(tp) = tp;      // TCB self-pointer (%fs:0 == tp)
191   f = __tls_frame{ base, static_cast<usize>(alloc), tp, block };
192 #elif defined(__micron_arch_arm64) || defined(__micron_arch_arm32)
193   // Variant I: [ TCB | pad | image_block ], tp = base; no self-pointer
194   // WARNING: armv7/armv8+ exec model resolves a thread_local at PT_TLS offset to read_tp + round_up(tcbhead, p_align) + o,
195   // the image __MUST__ start at p_align past the TCB, not at tcbhead
196   const u64 head = (sizeof(void *) == 8) ? __arm64_tcbhead_sz : __arm_tcbhead_sz;
197   const u64 head_aligned = __tls_round_up(head, p_align);
198   const u64 alloc = __tls_round_up(head_aligned + block, page_sz);
199   byte *base = __tls_raw_mmap(static_cast<usize>(alloc));
200   if ( !base ) return f;
201   byte *image_dst = base + head_aligned;
202   for ( u64 i = 0; i < filesz; ++i ) image_dst[i] = image[i];
203   f = __tls_frame{ base, static_cast<usize>(alloc), base, block };
204 #endif
205   return f;
206 }

208 inline __tls_frame
209 __tls_make_frame_cached() noexcept
210 {
211   if ( !__tls_ensure_template() ) return __tls_frame{};
212   const __tls_template_t &t = __micron_tls_template;
213   return __tls_make_frame(t.image, t.filesz, t.memsz, t.align, t.pagesz);
214 }

216 // NOTE: if under -fstack-protector-strong the child needs a valid stack canary at %fs:0x28
217 // (and pointer guard at 0x30) or it faults on the first guarded epilogue
218 inline void
219 __tls_seed_tcb_from_current([[maybe_unused]] const __tls_frame &f) noexcept
220 {
221 #if defined(__micron_arch_amd64)
222   if ( !f.tp ) return;
223   unsigned long guard = 0, pguard = 0;
224   asm volatile("movq %%fs:0x28, %0" : "=r"(guard));
225   asm volatile("movq %%fs:0x30, %0" : "=r"(pguard));
226   *reinterpret_cast<unsigned long *>(f.tp + 0x28) = guard;
227   *reinterpret_cast<unsigned long *>(f.tp + 0x30) = pguard;
228 #elif defined(__micron_arch_x86)
229   if ( !f.tp ) return;
230   unsigned long guard = 0, pguard = 0;
231   asm volatile("movl %%gs:0x14, %0" : "=r"(guard));      // i386 glibc stack guard
232   asm volatile("movl %%gs:0x18, %0" : "=r"(pguard));
233   *reinterpret_cast<unsigned long *>(f.tp + 0x14) = guard;
234   *reinterpret_cast<unsigned long *>(f.tp + 0x18) = pguard;
235 #endif
236 }

238 inline __tls_frame
239 __tls_make_child_frame() noexcept
240 {
241 #if defined(MICRON_ENABLE_ATTACH)
242   (void)__attach_host_tls_init();
243 #endif
244   __tls_frame f = __tls_make_frame_cached();
245   if ( f.base ) {
246     __tls_seed_tcb_from_current(f);
247 #if defined(__micron_attach_capable)
248     // WARNING: join the registry before seeding, never after
249     __attach_frame_register(f.base);
250     // seed every attached guest's thread_locals into this frame's surplus
251     __attach_copy_registered_tdata(f.base);
252 #endif
253   }
254   return f;
255 }

257 inline void
258 __tls_free_frame(const __tls_frame &f) noexcept
259 {
260   if ( !f.base ) return;
261 #if defined(__micron_attach_capable)
262   // leave the registry before the pages go away, so no assign() walk can write into them
263   __attach_frame_unregister(f.base);
264 #endif
265   micron::munmap(micron::ptr_cast<addr_t *>(f.base), f.size);
266 }

268 #if defined(__micron_attach_capable)
269 inline int
270 __attach_make_child_frame(__tls_frame *out) noexcept
271 {
272   __tls_frame frame = __tls_make_child_frame();
273   if ( frame.base == nullptr ) return -1;
274   *out = frame;
275   return 0;
276 }
