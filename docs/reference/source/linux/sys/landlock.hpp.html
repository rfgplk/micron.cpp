52 // fs masks
53 constexpr static const u64 landlock_access_fs_abi1 = 0x1fffuLL;
54 constexpr static const u64 landlock_access_fs_abi2 = 0x3fffuLL;
55 constexpr static const u64 landlock_access_fs_abi3 = 0x7fffuLL;
56 constexpr static const u64 landlock_access_fs_abi5 = 0xffffuLL;
57 constexpr static const u64 landlock_access_fs_abi9 = 0x1ffffuLL;
58 
59 [[nodiscard]] constexpr u64
60 landlock_fs_mask_for(i32 abi) noexcept
61 {
62   if ( abi <= 1 ) return landlock_access_fs_abi1;
63   if ( abi == 2 ) return landlock_access_fs_abi2;
64   if ( abi == 3 or abi == 4 ) return landlock_access_fs_abi3;
65   if ( abi >= 5 and abi <= 8 ) return landlock_access_fs_abi5;
66   return landlock_access_fs_abi9;
67 }
68 
69 [[nodiscard]] constexpr u64
70 landlock_net_mask_for(i32 abi) noexcept
71 {
72   return abi >= 4 ? (landlock_access_net_bind_tcp | landlock_access_net_connect_tcp) : 0uLL;
73 }
74 
75 [[nodiscard]] constexpr u64
76 landlock_scope_mask_for(i32 abi) noexcept
77 {
78   return abi >= 6 ? (landlock_scope_abstract_unix_socket | landlock_scope_signal) : 0uLL;
79 }

96 // records

98 struct landlock_ruleset_attr_t {
99   u64 handled_access_fs;
100   u64 handled_access_net;      // abi 4
101   u64 scoped;                  // abi 6
102 };
103 
104 constexpr static const usize landlock_ruleset_size_ver0 = 8;       // abi 1-3
105 constexpr static const usize landlock_ruleset_size_ver1 = 16;      // abi 4-5
106 constexpr static const usize landlock_ruleset_size_ver2 = 24;      // abi 6+
107 
108 [[nodiscard]] constexpr usize
109 landlock_ruleset_size_for(i32 abi) noexcept
110 {
111   if ( abi <= 3 ) return landlock_ruleset_size_ver0;
112   if ( abi <= 5 ) return landlock_ruleset_size_ver1;
113   return landlock_ruleset_size_ver2;
114 }
115 
116 // must be packed
117 struct __attribute__((packed)) landlock_path_beneath_attr_t {
118   u64 allowed_access;
119   i32 parent_fd;      // preferably opened O_PATH
120 };
121 
122 struct landlock_net_port_attr_t {
123   u64 allowed_access;
124   u64 port;      // host endianness, NOT network order
125 };
126 
127 static_assert(sizeof(landlock_ruleset_attr_t) == 24, "landlock_ruleset_attr ABI");
128 static_assert(sizeof(landlock_path_beneath_attr_t) == 12, "landlock_path_beneath_attr ABI (must be packed)");
129 static_assert(sizeof(landlock_net_port_attr_t) == 16, "landlock_net_port_attr ABI");
130 static_assert(__builtin_offsetof(landlock_ruleset_attr_t, scoped) == 16, "landlock_ruleset_attr.scoped ABI");
131 static_assert(__builtin_offsetof(landlock_path_beneath_attr_t, parent_fd) == 8, "landlock_path_beneath_attr.parent_fd ABI");
