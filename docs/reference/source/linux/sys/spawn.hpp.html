71 int
72 spawnattr_init(spawnattr_t &attr)
73 {
74   attr.__flags = 0;
75   attr.pgrp = 0;
76 
77   sigemptyset(attr.sd);
78   sigemptyset(attr.ss);
79 
80   attr.__sp.sched_priority = 0;
81   attr.policy = sched_other;
82 
83   attr.cgroup = -1;
84 
85   for ( int i = 0; i < 15; ++i ) attr.pad[i] = 0;
86 
87   return 0;
88 }

90 int
91 __apply_file_actions(const spawn_file_actions_t &fa)
92 {
93   for ( int i = 0; i < fa.used; ++i ) {
94     const spawn_action &a = fa.__actions[i];
95 
96     switch ( a.type ) {
97     case SPAWN_ACTION_OPEN: {
98       auto fd = micron::openat(at_fdcwd, a.path, a.oflag, a.mode);
99       if ( fd < 0 ) return static_cast<int>(fd);
100       if ( fd != a.fd ) {
101         micron::dup3(fd, a.fd, 0);
102         micron::close(fd);
103       }
104       break;
105     }
106     case SPAWN_ACTION_CLOSE:
107       micron::close(a.fd);
108       break;
109 
110     case SPAWN_ACTION_DUP2:
111       micron::dup2(a.fd, a.newfd);
112       break;
113     }
114   }
115   return 0;
116 }

118 int
119 __apply_spawnattr(const spawnattr_t &attr)
120 {
121   // TODO: push as separate file
122   if ( attr.__flags & posix_spawn_setsid ) {
123     long r = micron::syscall(SYS_setsid);      // new session + process group leader; do this before setpgid
124     if ( micron::syscall_failed(r) ) return static_cast<int>(r);
125   }
126   if ( attr.__flags & posix_spawn_setpgroup ) {
127     long r = micron::syscall(SYS_setpgid, 0, attr.pgrp);
128     if ( micron::syscall_failed(r) ) return static_cast<int>(r);
129   }
130   if ( attr.__flags & posix_spawn_setsigmask ) {
131     long r = micron::syscall(SYS_rt_sigprocmask, sig_setmask, &attr.ss, nullptr, __sig_syscall_size);
132     if ( micron::syscall_failed(r) ) return static_cast<int>(r);
133   }
134   if ( attr.__flags & posix_spawn_setsigdef ) {
135     for ( int s = 1; s <= 64; ++s ) {
136       if ( !sigismember(attr.sd, s) ) continue;
137       __syscall_sigaction_t act{};
138       act.k_sa_handler = reinterpret_cast<sighandler_t>(0);      // SIG_DFL
139       long r = micron::syscall(SYS_rt_sigaction, s, &act, nullptr, __sig_syscall_size);
140       if ( micron::syscall_failed(r) ) return static_cast<int>(r);
141     }
142   }
143   if ( attr.__flags & posix_spawn_setscheduler ) {
144     long r = micron::syscall(SYS_sched_setscheduler, 0, attr.policy, &attr.__sp);
145     if ( micron::syscall_failed(r) ) return static_cast<int>(r);
146   }
147   if ( attr.__flags & posix_spawn_setschedparam ) {
148     long r = micron::syscall(SYS_sched_setparam, 0, &attr.__sp);
149     if ( micron::syscall_failed(r) ) return static_cast<int>(r);
150   }
151 
152   // if ( attr.cgroup >= 0 ) {
153   //   long r = micron::syscall(SYS_setns, attr.cgroup, CLONE_NEWCGROUP);
154   //   if ( micron::syscall_failed(r) ) return static_cast<int>(r);
155   // }
156 
157   return 0;
158 }

160 __attribute__((noreturn)) int
161 spawn_process(spawn_ctx *ctx)
162 {
163   long err = 0;
164   if ( ctx->fa ) {
165     err = __apply_file_actions(*ctx->fa);
166     if ( err < 0 ) goto fail;
167   }
168   if ( ctx->attr ) {
169     err = __apply_spawnattr(*ctx->attr);
170     if ( err < 0 ) goto fail;
171   }
172   err = posix::execve(ctx->path, ctx->argv, ctx->envp);      // only returns on failure (raw -errno)
173 
174 fail:
175   int e = micron::syscall_errno(err);      // positive errno for the parent
176   if ( ctx->errfd > 0 ) micron::write(ctx->errfd, &e, sizeof(e));
177   micron::posix::exit(127);
178   __builtin_unreachable();
179 }
