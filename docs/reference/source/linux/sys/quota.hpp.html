16 // disk-quota control
17 constexpr int quota_subcmdshift = 8;
18 constexpr int quota_subcmdmask = 0x00ff;
19 
20 constexpr int
21 qcmd(int subcmd, int type)
22 {
23   return (subcmd << quota_subcmdshift) | (type & quota_subcmdmask);
24 }

49 // dqblk.dqb_valid bits (QIF_*)
50 constexpr u32 qif_blimits = 1;
51 constexpr u32 qif_space = 2;
52 constexpr u32 qif_ilimits = 4;
53 constexpr u32 qif_inodes = 8;
54 constexpr u32 qif_btime = 16;
55 constexpr u32 qif_itime = 32;
56 constexpr u32 qif_limits = (qif_blimits | qif_ilimits);
57 constexpr u32 qif_usage = (qif_space | qif_inodes);
58 constexpr u32 qif_times = (qif_btime | qif_itime);
59 constexpr u32 qif_all = (qif_limits | qif_usage | qif_times);
60 
61 // dqinfo.dqi_valid bits (IIF_*)
62 constexpr u32 iif_bgrace = 1;
63 constexpr u32 iif_igrace = 2;
64 constexpr u32 iif_flags = 4;
65 constexpr u32 iif_all = (iif_bgrace | iif_igrace | iif_flags);
66 
67 // dqinfo.dqi_flags (DQF_*)

69 constexpr u32 dqf_sys_file = 0x10000;

71 struct dqblk_t {
72   u64 dqb_bhardlimit;      // absolute limit on disk quota blocks
73   u64 dqb_bsoftlimit;      // preferred limit on disk quota blocks
74   u64 dqb_curspace;        // current quota block count (bytes)
75   u64 dqb_ihardlimit;      // maximum number of allocated inodes
76   u64 dqb_isoftlimit;      // preferred inode limit
77   u64 dqb_curinodes;       // current number of allocated inodes
78   u64 dqb_btime;           // time limit for excessive disk use
79   u64 dqb_itime;           // time limit for excessive files
80   u32 dqb_valid;           // bitmask of QIF_* fields that are valid
81 };
82 
83 struct nextdqblk_t {
84   u64 dqb_bhardlimit;
85   u64 dqb_bsoftlimit;
86   u64 dqb_curspace;
87   u64 dqb_ihardlimit;
88   u64 dqb_isoftlimit;
89   u64 dqb_curinodes;
90   u64 dqb_btime;
91   u64 dqb_itime;
92   u32 dqb_valid;
93   u32 dqb_id;      // id of the quota actually returned
94 };
95 
96 struct dqinfo_t {
97   u64 dqi_bgrace;      // seconds before the block soft limit becomes a hard limit
98   u64 dqi_igrace;      // seconds before the inode soft limit becomes a hard limit
99   u32 dqi_flags;       // DQF_* flags
100   u32 dqi_valid;       // bitmask of IIF_* fields that are valid
101 };

103 inline auto
104 quotactl(int cmd, const char *special, int id, void *addr) -> i32
105 {
106   return static_cast<i32>(micron::syscall(SYS_quotactl, cmd, special, id, addr));
107 }
