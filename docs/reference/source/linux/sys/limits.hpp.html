98 inline auto
99 getrlimit(limits lm, rlimit_t &rl)
100 {
101 #if defined(__micron_arch_amd64) || defined(__micron_arch_arm64)
102   return micron::syscall(SYS_getrlimit, static_cast<i32>(lm), &rl);
103 #elif defined(__micron_arch_arm32) || defined(__micron_arch_x86)
104   // NOTE: 32-bit (arm32 / i386): getrlimit is the legacy variant, use ugetrlimit
105   return micron::syscall(SYS_ugetrlimit, static_cast<i32>(lm), &rl);
106 #endif
107 }

170 // helper class
171 struct limits_t {
172   rlimit64_t lim[rlimit_nlimits] = {};      // zero-init: a failed get_process_limits leaves the entry 0, not garbage
173   ~limits_t() = default;
174 
175   limits_t(const pid_t proc = 0)      // for us by default
176   {
177     for ( rlim_t i = 0; i < rlimit_nlimits; i++ ) get_process_limits(proc, i, lim[i]);
178   }
179 
180   limits_t(const limits_t &o) { micron::voidcpy(lim, o.lim, sizeof(rlimit64_t) * rlimit_nlimits); }
181 
182   limits_t(limits_t &&o)
183   {
184     micron::voidcpy(lim, o.lim, sizeof(rlimit64_t) * rlimit_nlimits);
185     micron::memset(o.lim, 0x0, sizeof(rlimit64_t) * rlimit_nlimits);
186   }
187 
188   limits_t &
189   operator=(const limits_t &o)
190   {
191     micron::voidcpy(lim, o.lim, sizeof(rlimit64_t) * rlimit_nlimits);
192     return *this;
193   }
194 
195   limits_t &
196   operator=(limits_t &&o)
197   {
198     micron::voidcpy(lim, o.lim, sizeof(rlimit64_t) * rlimit_nlimits);
199     micron::memset(o.lim, 0x0, sizeof(rlimit64_t) * rlimit_nlimits);
200     return *this;
201   }
202 };
