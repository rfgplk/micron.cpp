1572 constexpr u32
1573 setup_mask_for_kernel(u32 maj, u32 min) noexcept
1574 {
1575   const u32 v = (maj << 8) | min;
1576   u32 mask = setup_iopoll | setup_sqpoll | setup_sq_aff | setup_cqsize | setup_clamp | setup_attach_wq;      // 5.1-5.6 era
