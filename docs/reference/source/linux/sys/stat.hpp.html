24 // stat dispatchx:
25 //   amd64     -> SYS_stat / SYS_lstat / SYS_fstat / SYS_newfstatat    -> 64-bit native stat_t
26 //   arm64     -> SYS_fstat / SYS_newfstatat (no stat/lstat in asm-gen)-> 64-bit stat_t
27 //   arm32/x86 -> statx (arch-uniform) translated into stat64_t
28 #if defined(__micron_arch_arm32) || defined(__micron_arch_x86)
29 constexpr dev_t
30 __statx_makedev(u32 maj, u32 mnr)
31 {
32   return static_cast<dev_t>((static_cast<u64>(mnr) & 0xffu) | ((static_cast<u64>(maj) & 0xfffu) << 8)
33                             | ((static_cast<u64>(mnr) & ~0xffull) << 12) | ((static_cast<u64>(maj) & ~0xfffull) << 32));
34 }

36 inline void
37 __statx_to_stat64(const statx_t &sx, stat64_t &st)
38 {
39   st.st_dev = __statx_makedev(sx.stx_dev_major, sx.stx_dev_minor);
40   st.__pad1 = 0;
41   st.__st_ino = static_cast<ino_t>(sx.stx_ino);
42   st.st_mode = static_cast<mode_t>(sx.stx_mode);
43   st.st_nlink = static_cast<nlink_t>(sx.stx_nlink);
44   st.st_uid = static_cast<uid_t>(sx.stx_uid);
45   st.st_gid = static_cast<gid_t>(sx.stx_gid);
46   st.st_rdev = __statx_makedev(sx.stx_rdev_major, sx.stx_rdev_minor);
47   st.__pad2 = 0;
48   st.st_size = static_cast<off64_t>(sx.stx_size);
49   st.st_blksize = static_cast<blksize_t>(sx.stx_blksize);
50   st.st_blocks = static_cast<__blkcnt64_t_type>(sx.stx_blocks);
51   st.st_atim.tv_sec = sx.stx_atime.tv_sec;
52   st.st_atim.tv_nsec = sx.stx_atime.tv_nsec;
53   st.st_mtim.tv_sec = sx.stx_mtime.tv_sec;
54   st.st_mtim.tv_nsec = sx.stx_mtime.tv_nsec;
55   st.st_ctim.tv_sec = sx.stx_ctime.tv_sec;
56   st.st_ctim.tv_nsec = sx.stx_ctime.tv_nsec;
57   st.st_ino = static_cast<ino64_t>(sx.stx_ino);
58 }

60 inline i32
61 __stat64_via_statx(int dirfd, const char *path, int flags, stat64_t &buf)
62 {
63   statx_t sx{};
64   i32 r = static_cast<i32>(micron::syscall(SYS_statx, dirfd, path, flags, statx_basic_stats, &sx));
65   if ( r == 0 ) __statx_to_stat64(sx, buf);
66   return r;
67 }
