247 // returns a sigset_t word, so it is typed as one -- the shift is already reduced modulo the word

249 inline constexpr unsigned long int
250 sigmask(const u64 mask)
251 {
252   return (1UL << (((mask)-1) % (8 * sizeof(unsigned long))));

337 inline int
338 sigaction(int sig, const sigaction_t &action, sigaction_t *old)
339 {
340   __syscall_sigaction_t __system_action = {};
341   __syscall_sigaction_t __system_oldaction = {};
342 
343   __system_action.k_sa_handler = action.sigaction_handler.sa_handler;
344   micron::voidcpy(&__system_action.sa_mask, &action.sa_mask, sizeof(posix::sigset_t));
345   __system_action.sa_flags = (u32)action.sa_flags;
346 #if !defined(__micron_syscall_generic) && !defined(__micron_arch_arm32)
347   // WARNING: amd64/i386 have no kernel-provided default restorer, so a userspace SA_RESTORER
348   // trampoline is mandatory; arm32 (and arm64) GET a kernel restorer via the sigpage/vDSO;
349   // installing a user Thumb restorer there is BROKEN
350   __system_action.sa_flags |= 0x04000000;      // SA_RESTORER
351   __system_action.sa_restorer = &restore_rt;
352 #endif
353   // restorer
354   int result
355       = static_cast<int>(micron::syscall(SYS_rt_sigaction, sig, &__system_action, old ? &__system_oldaction : nullptr, __sig_syscall_size));
356   if ( old && result >= 0 ) {
357     old->sigaction_handler.sa_handler = __system_oldaction.k_sa_handler;
358     micron::voidcpy(&old->sa_mask, &__system_oldaction.sa_mask, sizeof(posix::sigset_t));
359     old->sa_flags = static_cast<u32>(__system_oldaction.sa_flags);
360 #if !defined(__micron_syscall_generic)
361     old->sa_restorer = __system_action.sa_restorer;
362 #else
363     old->sa_restorer = nullptr;
364 #endif
365   }
366   return result;
367 }

369 inline int
370 sigwait(const posix::sigset_t &set, int &sig)
371 {
372   siginfo_t info = {};
373   long ret = 0;
374   do {
375     ret = micron::syscall(SYS_rt_sigtimedwait, &set, &info, nullptr, __sig_syscall_size);
376   } while ( micron::syscall_failed(ret) && micron::syscall_errno(ret) == EINTR );
377   if ( micron::syscall_failed(ret) ) return micron::syscall_errno(ret);      // POSIX sigwait: positive errno
378   sig = info.si_signo;
379   return 0;
380 }

410 inline int
411 sigqueue(posix::pid_t pid, int sig, const sigval_t &value)
412 {
413   siginfo_t info{};
414   info.si_signo = sig;
415   info.si_code = -1;      // SI_QUEUE
416   info._sifields._rt.si_pid = static_cast<posix::pid_t>(micron::syscall(SYS_getpid));
417   info._sifields._rt.si_uid = static_cast<posix::uid_t>(micron::syscall(SYS_getuid));
418   info._sifields._rt.si_sigval = value;
419   return static_cast<int>(micron::syscall(SYS_rt_sigqueueinfo, pid, sig, &info));
