98 struct stat_t {
99   posix::dev_t st_dev;     /* Device.  */
100   posix::ino64_t st_ino;   /* File serial number.  */
101   posix::nlink_t st_nlink; /* Link count.  */
102 
103   posix::mode_t st_mode; /* File mode.  */
104   posix::uid_t st_uid;   /* User ID of the file’s owner.  */
105   posix::gid_t st_gid;   /* Group ID of the file’s group. */
106   i32 __pad0;
107   posix::dev_t st_rdev;        /* Device number, if device.  */
108   posix::off64_t st_size;      /* Size of file, in bytes.  */
109   posix::blksize_t st_blksize; /* Optimal block size for I/O.  */
110 
111   posix::blkcnt_t st_blocks; /* Number of 512-byte blocks allocated. */
112 
113   // NOTE: was using pthread linked timespec instead of time.hpp defined timespec_t
114   struct timespec_t st_atim;    /* Time of last access.  */
115   struct timespec_t st_mtim;    /* Time of last modification.  */
116   struct timespec_t st_ctim;    /* Time of last status change.  */
117 #define st_atime st_atim.tv_sec /* Backward compatibility.  */
118 #define st_mtime st_mtim.tv_sec
119 #define st_ctime st_ctim.tv_sec
120 
121   __syscall_slong_type __glibc_reserved[3];      // micron's typedef (not glibc's __syscall_slong_t) for freestanding
122 
123   bool
124   operator!=(const stat_t &o) const
125   {
126     return micron::memcmp<byte>(this, &o,
127                                 static_cast<u64>(reinterpret_cast<const u8 *>(&st_blksize) - reinterpret_cast<const u8 *>(&st_dev)));
128   }
129 
130   bool
131   operator==(const stat_t &o) const
132   {
133     return !micron::memcmp<byte>(this, &o,
134                                  static_cast<u64>(reinterpret_cast<const u8 *>(&st_blksize) - reinterpret_cast<const u8 *>(&st_dev)));
135   }
136 };
