30     [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp
31     long __micron_clone_entry(int (*)(void *), void *, unsigned long, void *, int *, void *, int *)
32 {
33   // entry: rdi=fn rsi=stack rdx=flags rcx=arg r8=ptid r9=tls 8(%rsp)=ctid
34   asm volatile("xorl %eax, %eax\n\t"
35                "movb $56, %al\n\t"           // SYS_clone
36                "movq %rdi, %r11\n\t"         // save fn
37                "movq %rdx, %rdi\n\t"         // flags -> a1
38                "movq %r8, %rdx\n\t"          // ptid  -> a3
39                "movq %r9, %r8\n\t"           // tls   -> a5
40                "movq 8(%rsp), %r10\n\t"      // ctid  -> a4
41                "movq %r11, %r9\n\t"          // fn    -> r9 (preserved across syscall)
42                "andq $-16, %rsi\n\t"         // align child stack top
43                "subq $8, %rsi\n\t"
44                "movq %rcx, (%rsi)\n\t"      // seed arg on child stack
45                "syscall\n\t"
46                "testq %rax, %rax\n\t"
47                "jnz 1f\n\t"               // parent -> return pid in rax
48                "xorl %ebp, %ebp\n\t"      // child: terminate unwind chain
49                "popq %rdi\n\t"            // arg -> a1 of fn
50                "callq *%r9\n\t"           // fn(arg)
51                "movl %eax, %edi\n\t"      // status
52                "xorl %eax, %eax\n\t"
53                "movb $60, %al\n\t"      // SYS_exit
54                "syscall\n\t"
55                "hlt\n\t"
56                "1:\n\t"
57                "ret\n\t");
58 }
59 #elif defined(__micron_arch_x86)
60     [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp
61     long __micron_clone_entry(int (*)(void *), void *, unsigned long, void *, int *, void *, int *)
62 {
63   // all args on stack; +16 after pushing the 4 callee-saved regs
64   asm volatile("pushl %ebp\n\t"
65                "pushl %edi\n\t"
66                "pushl %esi\n\t"
67                "pushl %ebx\n\t"
68                "movl 20(%esp), %eax\n\t"      // fn
69                "movl 24(%esp), %ecx\n\t"      // stack top
70                "movl 28(%esp), %ebx\n\t"      // flags -> a1
71                "movl 36(%esp), %edx\n\t"      // ptid  -> a3
72                "movl 40(%esp), %esi\n\t"      // tls   -> a4 (CLONE_BACKWARDS)
73                "movl 44(%esp), %edi\n\t"      // ctid  -> a5
74                "andl $-16, %ecx\n\t"          // align top
75                "subl $16, %ecx\n\t"           // reserve seed area below newsp
76                "movl %eax, (%ecx)\n\t"        // fn  at newsp+0
77                "movl 32(%esp), %eax\n\t"      // arg
78                "movl %eax, 4(%ecx)\n\t"       // arg at newsp+4
79                "movl $120, %eax\n\t"          // SYS_clone
80                "int $0x80\n\t"
81                "testl %eax, %eax\n\t"
82                "jnz 1f\n\t"                  // parent
83                "movl (%esp), %eax\n\t"       // child: fn
84                "movl 4(%esp), %edx\n\t"      // arg
85                "xorl %ebp, %ebp\n\t"
86                "subl $16, %esp\n\t"         // align outgoing frame
87                "movl %edx, (%esp)\n\t"      // arg -> outgoing slot
88                "calll *%eax\n\t"            // fn(arg)
89                "movl %eax, %ebx\n\t"        // status
90                "movl $1, %eax\n\t"          // SYS_exit
91                "int $0x80\n\t"
92                "hlt\n\t"
93                "1:\n\t"
94                "popl %ebx\n\t"
95                "popl %esi\n\t"
96                "popl %edi\n\t"
97                "popl %ebp\n\t"
98                "ret\n\t");
99 }

104 asm(".text\n\t"
105     ".weak __micron_clone_entry\n\t"
106     ".type __micron_clone_entry, %function\n\t"
107     ".p2align 2\n"

109     // entry: x0=fn x1=stack x2=flags x3=arg x4=ptid x5=tls x6=ctid
110     "and x1, x1, #-16\n\t"             // align child stack top
111     "stp x0, x3, [x1, #-16]!\n\t"      // seed {fn, arg}; x1 becomes newsp
112     "mov x0, x2\n\t"                   // flags -> a0
113     "mov x2, x4\n\t"                   // ptid  -> a2
114     "mov x3, x5\n\t"                   // tls   -> a3
115     "mov x4, x6\n\t"                   // ctid  -> a4
116     "mov x8, #220\n\t"                 // SYS_clone
117     "svc #0\n\t"
118     "cbz x0, 1f\n\t"      // child -> 1
119     "ret\n\t"             // parent -> return pid in x0
120     "1:\n\t"
121     "ldp x1, x0, [sp], #16\n\t"      // x1=fn, x0=arg

126     [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp
127     long __micron_clone_entry(int (*)(void *), void *, unsigned long, void *, int *, void *, int *)
128 {
129   // entry: r0=fn r1=stack r2=flags r3=arg; ptid@[sp,#0] tls@[sp,#4] ctid@[sp,#8] (+16 after push)
130   asm volatile("stmfd sp!, {r4, r5, r6, r7}\n\t"
131                "mov r7, #120\n\t"           // SYS_clone
132                "mov r5, r0\n\t"             // fn  -> r5 (callee-saved, survives into child)
133                "mov r6, r3\n\t"             // arg -> r6 (callee-saved, survives into child)
134                "mov r0, r2\n\t"             // flags -> a0
135                "and r1, r1, #-16\n\t"       // align newsp (a1)
136                "ldr r2, [sp, #16]\n\t"      // ptid -> a2
137                "ldr r3, [sp, #20]\n\t"      // tls  -> a3 (CLONE_BACKWARDS)
138                "ldr r4, [sp, #24]\n\t"      // ctid -> a4
139                "svc #0\n\t"
140                "tst r0, r0\n\t"
141                "bne 1f\n\t"          // parent
142                "mov r0, r6\n\t"      // child: arg -> a0
143                "blx r5\n\t"          // fn(arg)
144                "mov r7, #1\n\t"      // SYS_exit
145                "svc #0\n\t"
146                "1:\n\t"
147                "ldmfd sp!, {r4, r5, r6, r7}\n\t"
148                "bx lr\n\t");
149 }

153 #if defined(__micron_arch_amd64)
154 [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp long
155 __micron_clone3_entry(int (*)(void *), void *, void *, unsigned long)
156 {
157   // rdi=thunk rsi=payload rdx=&clone_args rcx=args_size
158   asm volatile("pushq %r12\n\t"      // preserve callee-saved (parent path restores)
159                "pushq %r13\n\t"
160                "movq %rdi, %r12\n\t"      // thunk -> r12 (survives into child)
161                "movq %rsi, %r13\n\t"      // payload -> r13
162                "movq %rdx, %rdi\n\t"      // &clone_args -> a1
163                "movq %rcx, %rsi\n\t"      // args_size  -> a2
164                "movl $435, %eax\n\t"      // SYS_clone3
165                "syscall\n\t"
166                "testq %rax, %rax\n\t"
167                "jnz 1f\n\t"               // parent -> return tid in rax
168                "xorl %ebp, %ebp\n\t"      // child: terminate unwind chain
169                "andq $-16, %rsp\n\t"      // 16-byte align for the call ABI
170                "movq %r13, %rdi\n\t"      // payload -> a1
171                "callq *%r12\n\t"          // thunk(payload)
172                "movl %eax, %edi\n\t"      // status
173                "movl $60, %eax\n\t"       // SYS_exit
174                "syscall\n\t"
175                "hlt\n\t"
176                "1:\n\t"
177                "popq %r13\n\t"
178                "popq %r12\n\t"
179                "ret\n\t");
180 }

362   inline pid_t
363   clone3_spawn(int (*thunk)(void *), void *payload, void *stack_low, usize stack_size, unsigned long tls, pid_t *parent_tid, int *child_tid,
364                unsigned long flags = micthread_flags)
365   {
366     if ( !thunk || !stack_low || stack_size == 0 ) return -EINVAL;
367     if ( child_tid ) *child_tid = __micthread_ctid_alive;
368     unsigned long tls_arg = tls;
369 #if defined(__micron_arch_x86)
370     // CLONE_SETTLS replaces the inherited %gs slot; unlike set_thread_area it cannot allocate one.
371     struct __i386_clone_user_desc {
372       unsigned int entry_number, base_addr, limit, flags;
373     };
374 
375     unsigned short selector;
376     asm volatile("movw %%gs, %0" : "=r"(selector));
377     __i386_clone_user_desc __i386_tls_desc{ static_cast<unsigned int>(selector >> 3), static_cast<unsigned int>(tls), 0xfffffu, 0x51u };
378     tls_arg = reinterpret_cast<unsigned long>(&__i386_tls_desc);
379 #endif
380     long raw;
381     if ( __clone3_gate.open(kernel::feature::clone3) ) [[likely]] {
382       clone_args args{};
383       args.flags = flags;
384       args.exit_signal = 0;
385       args.stack = static_cast<u64>(reinterpret_cast<uintptr_t>(stack_low));
386       args.stack_size = stack_size;
387       args.tls = tls_arg;
388       args.parent_tid = static_cast<u64>(reinterpret_cast<uintptr_t>(parent_tid));
389       args.child_tid = static_cast<u64>(reinterpret_cast<uintptr_t>(child_tid));
390       raw = ::__micron_clone3_entry(thunk, payload, &args, sizeof(args));
391       if ( !(micron::syscall_failed(raw) && micron::syscall_errno(raw) == ENOSYS) ) [[likely]] {
392         if ( micron::syscall_failed(raw) && child_tid ) *child_tid = 0;      // no child ran
393         return static_cast<pid_t>(raw);
394       }
395       __clone3_gate.demote();
396     }
397     byte *top = reinterpret_cast<byte *>(stack_low) + stack_size;
398     raw = ::__micron_clone_entry(thunk, top, flags, payload, reinterpret_cast<int *>(parent_tid), reinterpret_cast<void *>(tls_arg),
399                                  child_tid);
400     if ( micron::syscall_failed(raw) && child_tid ) *child_tid = 0;      // no child ran
401     return static_cast<pid_t>(raw);
402   }

415   inline __attribute__((always_inline)) unsigned long

421   template<auto Fn, typename... Args> struct __clone_payload {
422     void *args[sizeof...(Args)];
423 
424     static int
425     thunk(void *p)
426     {
427       auto *self = static_cast<__clone_payload *>(p);
428       int rc = __invoke(self, micron::make_index_sequence<sizeof...(Args)>{});
429       destroy(self);
430       return rc;
431     }
432 
433     static void
434     destroy(__clone_payload *self)
435     {
436       __cleanup(self, micron::make_index_sequence<sizeof...(Args)>{});
437       delete self;
438     }
439 
440   private:
441     template<usize... I>
442     static int
443     __invoke(__clone_payload *self, micron::index_sequence<I...>)
444     {
445       using ret_t = decltype(Fn(static_cast<Args &&>(*static_cast<Args *>(self->args[I]))...));
446       if constexpr ( micron::is_same_v<ret_t, int> )
447         return Fn(static_cast<Args &&>(*static_cast<Args *>(self->args[I]))...);
448       else {
449         Fn(static_cast<Args &&>(*static_cast<Args *>(self->args[I]))...);
450         return 0;
451       }
452     }
453 
454     template<usize... I>
455     static void
456     __cleanup([[maybe_unused]] __clone_payload *self, micron::index_sequence<I...>)
457     {
458       (delete static_cast<Args *>(self->args[I]), ...);
459     }
460   };

462   template<usize Sz, auto Fn, typename... Args>
463   pid_t
464   clone(addr_t *stack, int flags, Args &&...args)
465   {
466     if ( !stack ) return -EINVAL;
467 
468     using payload_t = __clone_payload<Fn, micron::decay_t<Args>...>;
469     auto *payload = new payload_t{ { reinterpret_cast<void *>(new micron::decay_t<Args>(static_cast<Args &&>(args)))... } };
470 
471     unsigned long f = static_cast<unsigned long>(static_cast<unsigned>(flags));
472     int exit_signal = sig_chld;
473     if ( (f & clone_parent) or ((f & clone_sighand) and (f & clone_thread)) ) exit_signal = 0;      // kernel constraint otherwise -22
474     f = __micron_fold_exit_signal(f, exit_signal);
475 
476     uintptr_t top = (reinterpret_cast<uintptr_t>(stack) + Sz) & ~uintptr_t(0xF);
477 
478     long raw = ::__micron_clone_entry(&payload_t::thunk, reinterpret_cast<void *>(top), f, reinterpret_cast<void *>(payload), nullptr,
479                                       nullptr, nullptr);
480 
481     if ( micron::syscall_failed(raw) ) {
482       payload_t::destroy(payload);      // clone failed, no child ran
483       return static_cast<pid_t>(raw);
484     }
485     if ( !(f & clone_vm) ) payload_t::destroy(payload);      // parent owns its independent (COW) copy
486     return static_cast<pid_t>(raw);
487   }
488 
489   // special clone func
490   pid_t
491   __fork_clone(int exit_signal)
492   {
493     // these should be exactly like this
494     posix::clone_args clone_args{};
495     clone_args.flags = 0;
496     clone_args.exit_signal = static_cast<u64>(exit_signal);      // should be sig_chld
497     clone_args.stack = 0;
498     clone_args.stack_size = 0;
499     clone_args.parent_tid = 0;
500     clone_args.child_tid = 0;
501     clone_args.tls = 0;
502 
503     long raw = -ENOSYS;
504     if ( kernel::has(kernel::feature::clone3) ) [[likely]]
505       raw = posix::clone3_kernel(clone_args);
506     if ( micron::syscall_failed(raw) && micron::syscall_errno(raw) == ENOSYS ) {
507       // clone3 unavailable (pre-5.3 kernel, or under qemu-user)
508       raw = posix::fork_kernel();
509     }
510     return static_cast<pid_t>(raw);
511   }
512 
513   // c-style entry
514   pid_t
515   clone(int (*fn)(void *), void *arg, unsigned long flags, void *stack, usize stack_size, int *parent_tid = nullptr,
516         int *child_tid = nullptr, unsigned long tls = 0, int exit_signal = sig_chld)
517   {
518     if ( !stack or !fn or stack_size == 0 ) return -EINVAL;
519 
520     if ( (flags & clone_parent) or ((flags & clone_sighand) and (flags & clone_thread)) )
521       exit_signal = 0;      // kernel constraint otherwise -22
522     unsigned long f = __micron_fold_exit_signal(flags, exit_signal);
523 
524     uintptr_t top = (reinterpret_cast<uintptr_t>(stack) + stack_size) & ~uintptr_t(0xF);
525 
526     long raw = ::__micron_clone_entry(fn, reinterpret_cast<void *>(top), f, arg, parent_tid, reinterpret_cast<void *>(tls), child_tid);
527     return static_cast<pid_t>(raw);
528   }
