15 template<typename T>
16 constexpr auto
17 __ioctl_arg(T v)
18 {
19   if constexpr ( micron::is_pointer_v<T> )
20     return v;
21   else
22     return static_cast<unsigned long>(v);
23 }

25 template<typename... Args>
26 int
27 ioctl(i32 fd, Args... ops)
28 {
29   return static_cast<int>(micron::syscall(SYS_ioctl, fd, __ioctl_arg(ops)...));

59 consteval u64
60 io_request(u64 __dir, u64 type, u64 nr, u64 size)
61 {
62   return (((__dir) << __ioc_dirshift) | ((type) << __ioc_typeshift) | ((nr) << __ioc_nrshift) | ((size) << __ioc_sizeshift));
63 }
