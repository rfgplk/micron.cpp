29 template<typename T>
30 concept __syscall_arg
31     = (micron::is_integral_v<T> || micron::is_pointer_v<T> || micron::is_enum_v<T> || micron::is_null_pointer_v<T>) && sizeof(T) <= 8;
32 
33 template<__syscall_arg T>
34 inline __attribute__((always_inline)) long int
35 __coerce(T __v) noexcept
36 {
37   if constexpr ( micron::is_null_pointer_v<T> )
38     return 0L;
39   else if constexpr ( micron::is_pointer_v<T> )
40     return static_cast<long int>(reinterpret_cast<__tt_size_t>(__v));
41   else
42     return static_cast<long int>(__v);
43 }

45 template<__tt_size_t Slot>
46 consteval __tt_size_t
47 __count_slots() noexcept
48 {
49   return Slot;
50 }

52 template<__tt_size_t Slot, typename A0, typename... Rest>
53 consteval __tt_size_t
54 __count_slots() noexcept
55 {
56   if constexpr ( sizeof(A0) <= sizeof(long int) ) {
57     return __count_slots<Slot + 1, Rest...>();
58   } else {
59     return __count_slots<Slot + 2, Rest...>();
60   }
61 }

63 template<__tt_size_t Slot>
64 constexpr void
65 __fill(long int *) noexcept
66 {
67 }
68 
69 template<__tt_size_t Slot, typename A0, typename... Rest>
70 constexpr void
71 __fill(long int *__slots, A0 __a0, Rest... __rest) noexcept
72 {
73   if constexpr ( sizeof(A0) <= sizeof(long int) ) {
74     __slots[Slot] = __coerce(__a0);
75     __fill<Slot + 1>(__slots, __rest...);
76   } else {
77     unsigned long long __v = static_cast<unsigned long long>(__a0);
78     __slots[Slot] = static_cast<long int>(__v & 0xFFFFFFFFull);
79     __slots[Slot + 1] = static_cast<long int>(__v >> 32);
80     __fill<Slot + 2>(__slots, __rest...);
81   }
82 }

86 inline __attribute__((always_inline)) long int
87 __do_syscall(long int __n) noexcept
88 {
89   long int __r;
90   asm volatile("int $0x80" : "=a"(__r) : "0"(__n) : "memory", "cc");
91   return __r;

169 inline __attribute__((always_inline)) long int
170 __do_syscall(long int __n, long int __a1, long int __a2, long int __a3, long int __a4, long int __a5, long int __a6) noexcept
171 {
172   struct __args_t {
173     long int a1, a6;
174   };
175 
176   __args_t __s = { __a1, __a6 };
177   long int __r;
178   register long int __r2 asm("ecx") = __a2;
179   register long int __r3 asm("edx") = __a3;
180   register long int __r4 asm("esi") = __a4;
181   register long int __r5 asm("edi") = __a5;
182   asm volatile("pushl %%ebp\n\t"
183                "pushl %%ebx\n\t"
184                "pushl  (%[s])\n\t"      // push a1
185                "pushl 4(%[s])\n\t"      // push a6
186                "popl  %%ebp\n\t"        // ebp = a6
187                "popl  %%ebx\n\t"        // ebx = a1
188                "int   $0x80\n\t"
189                "popl  %%ebx\n\t"
190                "popl  %%ebp"
191                : "=a"(__r)
192                : "0"(__n), [s] "r"(&__s), "r"(__r2), "r"(__r3), "r"(__r4), "r"(__r5)
193                : "memory", "cc");
194   return __r;
195 }

216 template<__impl::__syscall_arg... Args>
217   requires(__impl::__count_slots<0, Args...>() <= 6)
218 inline __attribute__((always_inline)) long int
219 syscall(long int __n, Args... __args) noexcept
220 {
221   long int __slots[6] = { 0, 0, 0, 0, 0, 0 };
222   __impl::__fill<0>(__slots, __args...);
223   return __impl::__dispatch<__impl::__count_slots<0, Args...>()>(__n, __slots);
224 }
