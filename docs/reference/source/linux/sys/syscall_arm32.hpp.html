25 template<typename T>
26 concept __syscall_arg
27     = (micron::is_integral_v<T> || micron::is_pointer_v<T> || micron::is_enum_v<T> || micron::is_null_pointer_v<T>) && sizeof(T) <= 8;
28 
29 template<__syscall_arg T>
30 inline __attribute__((always_inline)) long int
31 __coerce(T __v) noexcept
32 {
33   if constexpr ( micron::is_null_pointer_v<T> )
34     return 0L;
35   else if constexpr ( micron::is_pointer_v<T> )
36     return static_cast<long int>(reinterpret_cast<__tt_size_t>(__v));
37   else
38     return static_cast<long int>(__v);
39 }

41 template<__tt_size_t Slot>
42 consteval __tt_size_t
43 __count_slots() noexcept
44 {
45   return Slot;
46 }

48 template<__tt_size_t Slot, typename A0, typename... Rest>
49 consteval __tt_size_t
50 __count_slots() noexcept
51 {
52   if constexpr ( sizeof(A0) <= sizeof(long int) ) {
53     return __count_slots<Slot + 1, Rest...>();
54   } else {
55     constexpr __tt_size_t __aligned = (Slot + 1) & ~__tt_size_t(1);
56     return __count_slots<__aligned + 2, Rest...>();
57   }
58 }

60 template<__tt_size_t Slot>
61 constexpr void
62 __fill(long int *) noexcept
63 {
64 }
65 
66 template<__tt_size_t Slot, typename A0, typename... Rest>
67 constexpr void
68 __fill(long int *__slots, A0 __a0, Rest... __rest) noexcept
69 {
70   if constexpr ( sizeof(A0) <= sizeof(long int) ) {
71     __slots[Slot] = __coerce(__a0);
72     __fill<Slot + 1>(__slots, __rest...);
73   } else {
74     constexpr __tt_size_t __aligned = (Slot + 1) & ~__tt_size_t(1);
75     unsigned long long __v = static_cast<unsigned long long>(__a0);
76     __slots[__aligned] = static_cast<long int>(__v & 0xFFFFFFFFull);
77     __slots[__aligned + 1] = static_cast<long int>(__v >> 32);
78     __fill<__aligned + 2>(__slots, __rest...);
79   }
80 }

84 // WARNING:
85 // on ARMv7-A Linux EABI, r7 is the syscall number register
86 // the obvious implementation is register long int __r7 asm("r7") = __n;
87 // plus a matching "r"(__r7) operand. That works ONLY when -fomit-frame-pointer is in effect
88 //
89 // However, in Thumb-2 mode, r7 is the architectural frame pointer;
90 // under -fno-omit-frame-pointer, GCC reserves r7 for the function prologue/epilogue and refuses any register T x asm("r7")
91 //   "error: r7 cannot be used in 'asm' here"
92 
93 inline __attribute__((always_inline)) long int
94 __do_syscall(long int __n) noexcept
95 {
96   register long int __r0 asm("r0");
97   asm volatile("push {r7}\n\t"
98                "mov r7, %[n]\n\t"
99                "svc #0\n\t"
100                "pop {r7}"
101                : "=r"(__r0)
102                : [n] "r"(__n)
103                : "memory");
104   return __r0;
105 }

225 template<__impl::__syscall_arg... Args>
226   requires(__impl::__count_slots<0, Args...>() <= 6)
227 inline __attribute__((always_inline)) long int
228 syscall(long int __n, Args... __args) noexcept
229 {
230   long int __slots[6] = { 0, 0, 0, 0, 0, 0 };
231   __impl::__fill<0>(__slots, __args...);
232   return __impl::__dispatch<__impl::__count_slots<0, Args...>()>(__n, __slots);
233 }
