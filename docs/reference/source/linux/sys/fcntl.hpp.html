28 fadvise(i32 fd, posix::off_t offset, posix::off_t len, i32 advice) -> i32
29 {
30 #if defined(__micron_arch_amd64) || defined(__micron_arch_arm64)
31   return static_cast<i32>(micron::syscall(SYS_fadvise64, fd, offset, len, advice));
32 #elif defined(__micron_arch_arm32)
33   return static_cast<i32>(micron::syscall(SYS_arm_fadvise64_64, fd, advice, static_cast<i64>(offset), static_cast<i64>(len)));

35   return static_cast<i32>(micron::syscall(SYS_fadvise64_64, fd, static_cast<i64>(offset), static_cast<i64>(len), advice));
36 #else

38 #endif
