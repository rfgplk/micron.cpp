14 template<typename... Args>
15 inline i32
16 prctl(i32 option, Args &&...args)
17 {
18   static_assert(sizeof...(Args) <= 4, "prctl takes at most 4 option arguments");
19   unsigned long a[4] = { 0, 0, 0, 0 };
20   if constexpr ( sizeof...(Args) > 0 ) {
21     unsigned long src[] = { static_cast<unsigned long>(args)... };
22     for ( usize i = 0; i < sizeof...(Args); ++i ) a[i] = src[i];
23   }
24   return static_cast<i32>(micron::syscall(SYS_prctl, option, a[0], a[1], a[2], a[3]));
25 }
