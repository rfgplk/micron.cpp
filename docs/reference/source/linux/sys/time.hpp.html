187 inline clock_t
188 clock(void)
189 {
190   timespec_t tm;
191   if ( clock_gettime(clock_process_cputime_id, tm) != 0 ) return -1;
192   // clock_t is width-32 on a width-32 target, so this wraps at ~36 min of cpu time -- the same
193   // contract every libc has here; the cast says the narrowing is meant
194   return static_cast<clock_t>(tm.tv_sec * clocks_per_sec + tm.tv_nsec / (1000000000 / clocks_per_sec));
195 }

197 inline time64_t
198 time(void)
199 {
200 #if defined(__micron_arch_amd64)
201   return micron::syscall(SYS_time, nullptr);      // amd64 SYS_time returns a 64-bit time_t
202 #else
203   timespec_t __ts{};
204   if ( clock_gettime(clock_realtime, __ts) != 0 ) return static_cast<time64_t>(-1);
205   return static_cast<time64_t>(__ts.tv_sec);
206 #endif
207 }

303 // NOTE: SYS_alarm is x86 only
304 inline u32
305 alarm(u32 seconds)
306 {
307 #if defined(__micron_arch_amd64) || defined(__micron_arch_x86)
308   return static_cast<u32>(micron::syscall(SYS_alarm, seconds));
309 #else
310   itimerval_t nv{};
311   itimerval_t ov{};
312   nv.it_value.tv_sec = static_cast<decltype(nv.it_value.tv_sec)>(seconds);
313   nv.it_value.tv_usec = 0;
314   nv.it_interval.tv_sec = 0;
315   nv.it_interval.tv_usec = 0;
316   if ( setitimer(itimer_real, nv, &ov) < 0 ) return 0;
317   u32 rem = static_cast<u32>(ov.it_value.tv_sec);
318   if ( ov.it_value.tv_usec != 0 ) ++rem;
319   return rem;
320 #endif
321 }

323 inline i32
324 clock_getcpuclockid(posix::pid_t pid, clockid_t &clc)
325 {
326   clc = static_cast<clockid_t>(((~static_cast<u32>(pid)) << 3) | 2u);
327   timespec_t probe;
328   long r = micron::syscall(__sys_clock_gettime, clc, &probe);
329   if ( r != 0 ) return static_cast<i32>(-r); /* return positive errno */
330   return 0;
331 }
