45 inline bool
46 __file_exists(const char *path) noexcept
47 {
48   i32 fd = posix::open(path, posix::o_rdonly);
49   if ( fd < 0 ) return false;
50   posix::close(fd);
51   return true;
52 }

54 inline bool
55 __file_is_native_elf(const char *path) noexcept
56 {
57   i32 fd = posix::open(path, posix::o_rdonly);
58   if ( fd < 0 ) return false;
59 
60   // ident_size + e_type(2) + e_machine(2)
61   u8 probe[ident_size + 4];
62   const max_t n = posix::pread(fd, probe, sizeof(probe), 0);
63   posix::close(fd);
64   if ( n != static_cast<max_t>(sizeof(probe)) ) return false;
65 
66   if ( probe[ei_mag0] != mag0 || probe[ei_mag1] != static_cast<u8>(mag1) || probe[ei_mag2] != static_cast<u8>(mag2)
67        || probe[ei_mag3] != static_cast<u8>(mag3) )
68     return false;
69   if ( probe[ei_class] != native_traits::ident_class ) return false;
70   if ( probe[ei_data] != (native_data == fmt_data::msb ? elfdata2msb : elfdata2lsb) ) return false;
71 
72   const half machine = static_cast<half>(static_cast<half>(probe[ident_size + 2]) | (static_cast<half>(probe[ident_size + 3]) << 8));
73   return expected_machine == 0 || machine == expected_machine;
74 }

76 inline path_str_t
77 __join_path(const char *dir, const char *leaf) noexcept
78 {
79   path_str_t out;
80   const usize dn = micron::strlen(dir);
81   const usize ln = micron::strlen(leaf);
82   if ( dn == 0 || dn + 1 + ln + 1 >= out.max_size() ) return out;
83   for ( usize i = 0; i < dn; ++i ) out += dir[i];
84   if ( out[out.size() - 1] != '/' ) out += '/';
85   for ( usize i = 0; i < ln; ++i ) out += leaf[i];
86   out.null_term();
87   return out;
88 }

90 inline path_str_t
91 __try_runpath(const char *runpath, const char *soname) noexcept
92 {
93   path_str_t hit;
94   if ( !runpath ) return hit;
95 
96   const usize rp_len = micron::strlen(runpath);
97   usize start = 0;
98   while ( start < rp_len ) {
99     const char *seg_end = micron::format::find(runpath + start, rp_len - start, ':');
100     const usize end = seg_end ? static_cast<usize>(seg_end - runpath) : rp_len;
101     if ( end > start ) {
102       if ( end - start >= 512u ) {
103         start = end + 1;
104         continue;
105       }
106       micron::sstring<512> dir;
107       for ( usize i = start; i < end; ++i ) dir += runpath[i];
108       dir.null_term();
109       path_str_t cand = __join_path(dir.c_str(), soname);
110       if ( !cand.empty() && __file_is_native_elf(cand.c_str()) ) return cand;
111     }
112     start = end + 1;
113   }
114   return hit;
115 }

117 inline path_str_t
118 resolve_soname(const char *soname, const char *runpath = nullptr) noexcept
119 {
120   path_str_t out;
121   if ( !soname || !soname[0] ) return out;
122 
123   const usize sn_len = micron::strlen(soname);
124 
125   if ( micron::format::find(soname, sn_len, '/') ) {
126     if ( sn_len >= out.max_size() ) return path_str_t{};
127     for ( usize i = 0; i < sn_len; ++i ) out += soname[i];
128     out.null_term();
129     return __file_exists(out.c_str()) ? out : path_str_t{};
130   }
131 
132   if ( path_str_t hit = __try_runpath(runpath, soname); !hit.empty() ) return hit;
133 
134   for ( usize i = 0; i < default_search_path_count; ++i ) {
135     path_str_t cand = __join_path(default_search_paths[i], soname);
136     if ( !cand.empty() && __file_is_native_elf(cand.c_str()) ) return cand;
137   }
138   return path_str_t{};
139 }
