45 inline usize
46 __page_floor(usize v) noexcept
47 {
48   const usize ps = __runtime_page_size();
49   return v & ~(ps - 1);
50 }

52 inline usize
53 __page_ceil(usize v) noexcept
54 {
55   const usize ps = __runtime_page_size();
56   return (v + ps - 1) & ~(ps - 1);

152 inline void *
153 __resolve_across_loaded(void *user, const char *name, u32 sym_index) noexcept
154 {
155   (void)user;
156   (void)sym_index;
157   for ( module_t *m = __loaded_modules; m; m = m->next ) {
158     const nsym_t *s = lookup_sym(m->dyn, name);
159     if ( s && s->shndx != shn_undef ) {
160       void *a = reinterpret_cast<void *>(m->load_base + s->value);
161       if ( elf_st_type(s->info) == stt_gnu_ifunc ) {
162         using ifn = void *(*)();
163         a = reinterpret_cast<ifn>(a)();
164       }
165       return a;
166     }
167   }
168   return host_resolve_sym(name);
169 }

174   for ( const ndyn_t *d = dyn; d->tag != dt_null; ++d ) {

323 inline bool
324 __reloc_is_relative(u32 t) noexcept
325 {
326 #if defined(__micron_arch_amd64)
327   return t == r_x86_64_relative || t == r_x86_64_irelative;
328 #elif defined(__micron_arch_arm64)
329   return t == r_aarch64_relative || t == r_aarch64_irelative;
330 #elif defined(__micron_arch_x86)
331   return t == r_386_relative || t == r_386_irelative;
332 #elif defined(__micron_arch_arm32)
333   return t == r_arm_relative || t == r_arm_irelative;
334 #else
335   return false;
336 #endif

352 template<class Rec>
353 inline bool
354 __apply_reloc_table(module_t &m, const Rec *tbl, usize count_bytes, reloc_mode_t mode, resolve_fn resolve = nullptr, void *user = nullptr)
355 {
356   if ( !tbl || count_bytes == 0 ) return true;
357   const usize n = count_bytes / sizeof(Rec);
358   const reloc_ctx_t ctx = __reloc_ctx_for(m, resolve, user);
359   for ( usize i = 0; i < n; ++i ) {
360     const reloc_view v = reloc_view_of<native_class>(tbl[i]);
361     if ( mode == reloc_mode_t::relative_only && !__reloc_is_relative(v.type) ) continue;
362     const reloc_result r = apply_reloc(ctx, v);
363     if ( r == reloc_result::unsupported ) return false;
364     if ( r == reloc_result::unresolved && mode == reloc_mode_t::strict ) return false;
365   }
366   return true;
367 }

369 inline bool
370 __apply_plt_table(module_t &m, reloc_mode_t mode, resolve_fn resolve = nullptr, void *user = nullptr)
371 {
372   if ( !m.dyn.jmprel || m.dyn.pltrelsz == 0 ) return true;
373   if ( m.dyn.pltrel == dt_rela )
374     return __apply_reloc_table(m, reinterpret_cast<const nrela_t *>(m.dyn.jmprel), static_cast<usize>(m.dyn.pltrelsz), mode, resolve, user);
375   if ( m.dyn.pltrel == dt_rel )
376     return __apply_reloc_table(m, reinterpret_cast<const nrel_t *>(m.dyn.jmprel), static_cast<usize>(m.dyn.pltrelsz), mode, resolve, user);
377   if constexpr ( arch_uses_rel )
378     return __apply_reloc_table(m, reinterpret_cast<const nrel_t *>(m.dyn.jmprel), static_cast<usize>(m.dyn.pltrelsz), mode, resolve, user);
379   else
380     return __apply_reloc_table(m, reinterpret_cast<const nrela_t *>(m.dyn.jmprel), m.dyn.pltrelsz, mode, resolve, user);
381 }

383 inline bool
384 __apply_all_relocs(module_t &m, reloc_mode_t mode, resolve_fn resolve = nullptr, void *user = nullptr)
385 {
386   if ( mode == reloc_mode_t::defer ) return true;

389   return __apply_plt_table(m, mode, resolve, user);

392 inline void
393 __apply_relr(module_t &m) noexcept
394 {
395   using tr = native_traits;
396   using rw = tr::uword;
397 
398   if ( !m.dyn.relr || m.dyn.relrsz == 0 ) return;
399   const uintptr_t l_addr = reinterpret_cast<uintptr_t>(m.load_base);
400   const usize n = static_cast<usize>(m.dyn.relrsz / sizeof(rw));
401   uintptr_t *where = nullptr;
402   for ( usize k = 0; k < n; ++k ) {
403     rw entry = m.dyn.relr[k];
404     if ( (entry & 1) == 0 ) {
405       where = reinterpret_cast<uintptr_t *>(l_addr + entry);
406       *where += l_addr;
407       ++where;
408     } else {
409       for ( usize i = 0; (entry >>= 1) != 0; ++i )
410         if ( (entry & 1) != 0 ) where[i] += l_addr;
411       where += tr::relr_bits;
412     }
413   }
414 }

416 inline void
417 __run_initializers(module_t &m) noexcept
418 {
419   // DT_PREINIT_ARRAY is legal exclusively in actual executables
420   if ( m.dyn.preinit_array && m.dyn.preinit_arraysz ) {
421     const usize n = static_cast<usize>(m.dyn.preinit_arraysz / sizeof(void *));
422     for ( usize i = 0; i < n; ++i ) {
423       if ( m.dyn.preinit_array[i] ) m.dyn.preinit_array[i]();
424     }
425   }
426   if ( m.dyn.init ) m.dyn.init();
427   if ( m.dyn.init_array && m.dyn.init_arraysz ) {
428     const usize n = static_cast<usize>(m.dyn.init_arraysz / sizeof(void *));
429     for ( usize i = 0; i < n; ++i ) {
430       if ( m.dyn.init_array[i] ) m.dyn.init_array[i]();
431     }
432   }
433 }

435 inline void
436 __run_finalizers(module_t &m) noexcept
437 {
438   if ( m.dyn.fini_array && m.dyn.fini_arraysz ) {
439     usize n = static_cast<usize>(m.dyn.fini_arraysz / sizeof(void *));

447 inline void
448 __record_relro(module_t &m, const nphdr_t *phdrs, half phnum) noexcept
449 {
450   for ( half i = 0; i < phnum; ++i ) {
451     if ( phdrs[i].type != pt_gnu_relro ) continue;
452     const usize start = __page_floor(phdrs[i].vaddr);
453     const usize end = __page_floor(phdrs[i].vaddr + phdrs[i].memsz);
454     if ( end <= start ) continue;
455     m.relro_start = start;
456     m.relro_len = end - start;
457     return;
458   }
459 }

487 // load a single .so by absolute path
488 // DOES NOT recursively load DT_NEEDED: host libc/libpthread/etc. are expected to be already mapped by the runtime
489 // if running in freestanding mode nothing will be loaded, so you need to load it manually (if required)
490 // TODO: should extend via proper /proc/self/maps walking and loading all depend separately
491 inline module_t
492 __load_module_from_path(const char *path, const load_opts_t &opts = {})
493 {
494   i32 fd = posix::open(path, posix::o_rdonly);
495   if ( fd < 0 ) exc<except::library_error>("elf: open failed");
496 
497   nehdr_t eh;
498   if ( posix::pread(fd, &eh, sizeof(eh), 0) != static_cast<max_t>(sizeof(eh)) ) {
499     posix::close(fd);
500     exc<except::library_error>("elf: short ehdr read");
501   }
502   if ( eh.ident[ei_mag0] != mag0 || eh.ident[ei_mag1] != mag1 || eh.ident[ei_mag2] != mag2 || eh.ident[ei_mag3] != mag3 ) {
503     posix::close(fd);
504     exc<except::library_error>("elf: bad magic");
505   }
506   if ( eh.ident[ei_class] != native_traits::ident_class ) {
507     posix::close(fd);
508     exc<except::library_error>("elf: wrong ELF class for this target (32/64-bit mismatch)");
509   }
510   if ( eh.ident[ei_data] != (native_data == fmt_data::msb ? elfdata2msb : elfdata2lsb) ) {
511     posix::close(fd);
512     exc<except::library_error>("elf: wrong byte order");
513   }
514   if ( eh.machine != expected_machine ) {
515     posix::close(fd);
516     exc<except::library_error>("elf: wrong machine");
517   }
518   if ( eh.type != et_dyn ) {
519     posix::close(fd);
520     exc<except::library_error>("elf: not a shared object");
521   }
522 
523   if ( eh.phentsize != sizeof(nphdr_t) || eh.phnum == 0 ) {
524     posix::close(fd);
525     exc<except::library_error>("elf: bad phdr table");
526   }
527 
528   nphdr_t phdrs[max_phdrs];
529   if ( eh.phnum > max_phdrs ) {
530     posix::close(fd);
531     exc<except::library_error>("elf: too many phdrs");
532   }
533   if ( posix::pread(fd, phdrs, sizeof(nphdr_t) * eh.phnum, eh.phoff) != static_cast<max_t>(sizeof(nphdr_t) * eh.phnum) ) {
534     posix::close(fd);
535     exc<except::library_error>("elf: short phdr read");
536   }
537 
538   usize vaddr_min = ~usize(0);
539   usize vaddr_max = 0;
540   const nphdr_t *dyn_ph = nullptr;
541   const nphdr_t *tls_ph = nullptr;
542   for ( half i = 0; i < eh.phnum; ++i ) {
543     if ( phdrs[i].type == pt_load ) {
544       if ( phdrs[i].vaddr < vaddr_min ) vaddr_min = phdrs[i].vaddr;
545       const usize end = phdrs[i].vaddr + phdrs[i].memsz;
546       if ( end < phdrs[i].vaddr ) {      // vaddr + memsz overflow on a crafted/corrupt phdr
547         posix::close(fd);
548         exc<except::library_error>("elf: PT_LOAD vaddr+memsz overflow");
549       }
550       if ( end > vaddr_max ) vaddr_max = end;
551     } else if ( phdrs[i].type == pt_dynamic ) {
552       dyn_ph = &phdrs[i];
553     } else if ( phdrs[i].type == pt_tls ) {
554       tls_ph = &phdrs[i];
555     }
556   }
557   if ( vaddr_min == ~usize(0) || !dyn_ph ) {
558     posix::close(fd);
559     exc<except::library_error>("elf: no PT_LOAD / PT_DYNAMIC");
560   }
561   vaddr_min = __page_floor(vaddr_min);
562   vaddr_max = __page_ceil(vaddr_max);
563   const usize span = vaddr_max - vaddr_min;
564 
565   u8 *base = reinterpret_cast<u8 *>(micron::mmap(nullptr, span, prot_none, map_private | map_anonymous, -1, 0));
566   if ( mmap_failed(base) ) {
567     posix::close(fd);
568     exc<except::library_error>("elf: reserve mmap failed");
569   }
570 
571   for ( half i = 0; i < eh.phnum; ++i ) {
572     if ( phdrs[i].type != pt_load ) continue;
573     const usize vstart = __page_floor(phdrs[i].vaddr);
574     const usize vend = __page_ceil(phdrs[i].vaddr + phdrs[i].memsz);
575     const usize fstart = __page_floor(phdrs[i].offset);
576 
577     if constexpr ( native_class == fmt_class::elf32 ) {
578       if ( (fstart & 0xfffu) != 0 ) {
579         micron::munmap(micron::ptr_cast<addr_t *>(base), span);
580         posix::close(fd);
581         exc<except::library_error>("elf: PT_LOAD file offset is not a multiple of 4096 (mmap2 cannot express it)");
582       }
583     }
584     const usize segoff = phdrs[i].vaddr - vstart;
585     u8 *target = base + (vstart - vaddr_min);
586 
587     const i32 prot = __prot_from_phdr(phdrs[i].flags);
588 
589     if ( ((phdrs[i].vaddr - phdrs[i].offset) & (__runtime_page_size() - 1)) != 0 ) {
590       micron::munmap(micron::ptr_cast<addr_t *>(base), span);
591       posix::close(fd);
592       exc<except::library_error>("elf: PT_LOAD vaddr/offset not page-congruent (rebuild with max-page-size)");
593     }
594 
595     if ( phdrs[i].filesz ) {
596       const usize file_end = __page_ceil(phdrs[i].filesz + segoff);
597       const usize mem_end = __page_ceil(phdrs[i].memsz + segoff);
598       u8 *got = reinterpret_cast<u8 *>(
599           micron::mmap(micron::ptr_cast<addr_t *>(target), file_end, prot_read | prot_write, map_private | map_fixed, fd, fstart));
600       if ( mmap_failed(got) ) {
601         micron::munmap(micron::ptr_cast<addr_t *>(base), span);
602         posix::close(fd);
603         exc<except::library_error>("elf: segment mmap failed");
604       }
605       if ( mem_end > file_end ) {
606         u8 *bgot = reinterpret_cast<u8 *>(micron::mmap(micron::ptr_cast<addr_t *>(target + file_end), mem_end - file_end,
607                                                        prot_read | prot_write, map_private | map_anonymous | map_fixed, -1, 0));
608         if ( mmap_failed(bgot) ) {
609           micron::munmap(micron::ptr_cast<addr_t *>(base), span);
610           posix::close(fd);
611           exc<except::library_error>("elf: bss tail mmap failed");
612         }
613       }
614       const usize bss_start = segoff + phdrs[i].filesz;
615       const usize bss_end = segoff + phdrs[i].memsz;
616       if ( bss_end > bss_start ) {
617         micron::memset(target + bss_start, byte{ 0 }, bss_end - bss_start);
618       }
619       micron::mprotect(micron::ptr_cast<addr_t *>(target), mem_end, prot);
620     } else {
621       u8 *got = reinterpret_cast<u8 *>(
622           micron::mmap(micron::ptr_cast<addr_t *>(target), vend - vstart, prot, map_private | map_anonymous | map_fixed, -1, 0));
623       if ( mmap_failed(got) ) {
624         micron::munmap(micron::ptr_cast<addr_t *>(base), span);
625         posix::close(fd);
626         exc<except::library_error>("elf: bss mmap failed");
627       }
628     }
629   }
630 
631   posix::close(fd);
632 
633   module_t m;
634   m.load_base = base - vaddr_min;      // bias so load_base + vaddr == actual VA
635   m.load_span = span;
636   for ( usize i = 0; path[i] && m.path.size() + 1 < m.path.max_size(); ++i ) m.path += path[i];
637   m.path.null_term();
638 
639   const ndyn_t *dyn = micron::ptr_cast<const ndyn_t *>(m.load_base + dyn_ph->vaddr);
640   __build_dyn_info(m.dyn, m.load_base, dyn);
641 
642   if ( tls_ph ) {
643     m.tls_modid = tls_register(m.load_base + tls_ph->vaddr, tls_ph->filesz, tls_ph->memsz, tls_ph->align);
644   }
645 
646   __record_relro(m, phdrs, eh.phnum);
647 
648   // point at the phdr table inside the mapping itself
649   {
650     const usize ph_end = static_cast<usize>(eh.phoff) + static_cast<usize>(eh.phnum) * sizeof(nphdr_t);
651     for ( half i = 0; i < eh.phnum; ++i ) {
652       if ( phdrs[i].type != pt_load ) continue;
653       if ( eh.phoff >= phdrs[i].offset && ph_end <= phdrs[i].offset + phdrs[i].filesz ) {
654         m.phdrs = micron::ptr_cast<const nphdr_t *>(m.load_base + phdrs[i].vaddr + (eh.phoff - phdrs[i].offset));
655         m.phnum = eh.phnum;
656         break;
657       }
658     }
659   }
660 
661   if ( opts.reloc == reloc_mode_t::defer ) return m;
662 
663   micron::lock_guard __ll(__loaded_modules_mtx);      // held through reloc + unlink (line below) until return
664   m.next = __loaded_modules;
665   __loaded_modules = &m;
666 
667   if ( !__apply_all_relocs(m, opts.reloc, opts.resolve, opts.resolve_user) ) {
668     __loaded_modules = m.next;      // unlink the partially-relocated module before unwinding
669     micron::munmap(micron::ptr_cast<addr_t *>(base), span);
670     exc<except::library_error>("elf: unsupported relocation (static TLS / TLSDESC / COPY) — refusing to load module with corrupt TLS");
671   }
672 
673   __apply_relr(m);
674 
675   if ( opts.apply_relro ) __apply_relro_now(m);
676   if ( opts.run_init ) __run_initializers(m);
677 
678   __loaded_modules = m.next;
679   m.next = nullptr;
680   return m;
681 }
682 
683 inline module_t
684 load(const char *soname, const char *runpath = nullptr, const load_opts_t &opts = {})
685 {
686   path_str_t resolved = resolve_soname(soname, runpath);
687   if ( resolved.empty() ) {
688     exc<except::library_error>("elf: soname not found in search paths");
689   }
690   return __load_module_from_path(resolved.c_str(), opts);
691 }
