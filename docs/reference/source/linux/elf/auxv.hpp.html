106 inline __attribute__((always_inline)) unsigned long
107 __auxv_lookup(const auxv_t *av, unsigned long type) noexcept
108 {
109   for ( ; av->a_type != at_null; ++av ) {
110     if ( av->a_type == type ) return av->a_val;
111   }
112   return 0;
113 }

131 struct tls_image {
132   const byte *image;      // base of the .tdata template (ph->p_vaddr + bias)
133   u64 filesz;             // bytes of initialised tdata
134   u64 memsz;              // total tls block size (filesz + tbss)
135   u64 align;              // p_align from PT_TLS (0 if no PT_TLS)

138 // computes the PIE / static-pie load bias from PT_PHDR (0 for ET_EXEC) so the returned image pointer is the runtime address of the .tdata
139 // template under -pie/-static-pie the template would be read from the link-time vaddr
140 //
141 // WARNING: this cannot compute a bias when the image has no PT_PHDR, and the bias is not recoverable from these three arguments alone
142 // ld emits PT_PHDR for a dynamic PIE and for anything with a PT_INTERP, but not for static ET_EXECs (bias is genuinely 0)
143 // but _not_ for --static-pie (NOT HARMLESS: bias stays 0 yet out.image is the link-time vaddr)
144 inline __attribute__((always_inline)) tls_image
145 find_tls_in_phdrs(unsigned long phdr_addr, unsigned long phent, unsigned long phnum) noexcept
146 {
147   tls_image out{ nullptr, 0, 0, 0 };
148   if ( phdr_addr == 0 || phent == 0 || phnum == 0 ) return out;
149 
150   const byte *p = reinterpret_cast<const byte *>(phdr_addr);
151 
152   u64 bias = 0;
153   for ( unsigned long i = 0; i < phnum; ++i ) {
154     const phdr_t *ph = micron::ptr_cast<const phdr_t *>(p + i * phent);
155     if ( ph->p_type == elf::pt_phdr ) {
156       bias = static_cast<u64>(phdr_addr) - static_cast<u64>(ph->p_vaddr);
157       break;
158     }
159   }
160 
161   for ( unsigned long i = 0; i < phnum; ++i ) {
162     const phdr_t *ph = micron::ptr_cast<const phdr_t *>(p + i * phent);
163     if ( ph->p_type == elf::pt_tls ) {
164       out.image = reinterpret_cast<const byte *>(static_cast<u64>(ph->p_vaddr) + bias);
165       out.filesz = ph->p_filesz;
166       out.memsz = ph->p_memsz;
167       out.align = ph->p_align;
168       return out;
169     }
170   }
171   return out;
172 }

174 inline __attribute__((always_inline)) tls_image
175 __auxv_find_tls(const auxv_t *av) noexcept
176 {
177   return find_tls_in_phdrs(__auxv_lookup(av, at_phdr), __auxv_lookup(av, at_phent), __auxv_lookup(av, at_phnum));

216 // true when the linker actually gave us a usable ELF header for the running image
217 inline bool
218 __ehdr_usable(void) noexcept
219 {
220   if ( __ehdr_start == nullptr ) return false;
221   const ehdr_t *eh = micron::ptr_cast<const ehdr_t *>(__ehdr_start);
222   if ( eh->e_ident[0] != 0x7f or eh->e_ident[1] != 'E' or eh->e_ident[2] != 'L' or eh->e_ident[3] != 'F' ) return false;
223   return eh->e_phoff != 0 and eh->e_phentsize != 0 and eh->e_phnum != 0;
224 }

226 // WARNING: the load bias is not derived from PT_PHDR alone
227 inline tls_image
228 find_tls_from_ehdr(void) noexcept
229 {
230   tls_image out{ nullptr, 0, 0, 0 };
231   if ( !__ehdr_usable() ) return out;
232 
233   const ehdr_t *eh = micron::ptr_cast<const ehdr_t *>(__ehdr_start);
234   const byte *p = __ehdr_start + eh->e_phoff;
235 
236   u64 bias = 0;
237   bool have_bias = false;
238   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
239     const phdr_t *ph = micron::ptr_cast<const phdr_t *>(p + static_cast<usize>(i) * eh->e_phentsize);
240     if ( ph->p_type == elf::pt_phdr ) {
241       bias = reinterpret_cast<u64>(p) - static_cast<u64>(ph->p_vaddr);
242       have_bias = true;
243       break;
244     }
245   }
246   if ( !have_bias ) {
247     for ( u16 i = 0; i < eh->e_phnum; ++i ) {
248       const phdr_t *ph = micron::ptr_cast<const phdr_t *>(p + static_cast<usize>(i) * eh->e_phentsize);
249       if ( ph->p_type == elf::pt_load and ph->p_offset == 0 ) {
250         bias = reinterpret_cast<u64>(__ehdr_start) - static_cast<u64>(ph->p_vaddr);
251         break;
252       }
253     }
254   }
255 
256   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
257     const phdr_t *ph = micron::ptr_cast<const phdr_t *>(p + static_cast<usize>(i) * eh->e_phentsize);
258     if ( ph->p_type == elf::pt_tls ) {
259       out.image = reinterpret_cast<const byte *>(static_cast<u64>(ph->p_vaddr) + bias);
260       out.filesz = ph->p_filesz;
261       out.memsz = ph->p_memsz;
262       out.align = ph->p_align;
263       return out;
264     }
265   }
266   return out;
267 }
