17 inline constexpr u32
18 sysv_hash(const char *name) noexcept
19 {
20   u32 h = 0;
21   u32 g = 0;
22   while ( *name ) {
23     h = (h << 4) + static_cast<u8>(*name++);
24     g = h & 0xf0000000;
25     if ( g ) h ^= g >> 24;
26     h &= ~g;
27   }
28   return h;
29 }

31 // modern preferred
32 inline constexpr u32
33 gnu_hash(const char *name) noexcept
34 {
35   u32 h = 5381;
36   while ( *name ) h = h * 33 + static_cast<u8>(*name++);
37   return h;
38 }

40 inline constexpr bool
41 __streq(const char *n, const char *m) noexcept
42 {
43   while ( *n && *n == *m ) {
44     ++n;
45     ++m;
46   }
47   return *n == 0 && *m == 0;
48 }

50 // spec sysv layout
51 // word nbuckets
52 // word nchain
53 // word bucket[nbuckets]
54 // word chain[nchain]
55 //
56 // DT_HASH entries are 32-bit in both classes
57 template<fmt_class C>
58 inline const typename elf_traits<C>::sym *
59 sysv_lookup(const dyn_info<C> &d, const char *name) noexcept
60 {
61   if ( !d.hash || !d.symtab || !d.strtab ) return nullptr;
62   const word nbuckets = d.hash[0];
63   const word nchain = d.hash[1];      // chain[] length == dynsym count
64   if ( !nbuckets ) return nullptr;
65   const word *buckets = d.hash + 2;
66   const word *chain = buckets + nbuckets;
67 
68   const u32 h = sysv_hash(name);
69   word i = buckets[h % nbuckets];
70   for ( word steps = 0; i != 0 && i < nchain && steps < nchain; i = chain[i], ++steps ) {
71     const auto &s = d.symtab[i];
72     if ( __streq(d.strtab + s.name, name) ) return &s;
73   }
74   return nullptr;
75 }

77 // gnu hash layout:
78 // word  nbuckets
79 // word  symbias
80 // word  bloom_size
81 // word  bloom_shift
82 // BLOOM bloom[bloom_size]        <- Elf32_Word on ELF32, Elf64_Xword on ELF64
83 // word  bucket[nbuckets]
84 // word  chain[]
85 //
86 template<fmt_class C>
87 inline const typename elf_traits<C>::sym *
88 gnu_lookup(const dyn_info<C> &d, const char *name) noexcept
89 {
90   using tr = elf_traits<C>;
91   using bloom_t = typename tr::uword;
92   constexpr u32 bb = static_cast<u32>(tr::bloom_bits);
93 
94   if ( !d.gnu_hash || !d.symtab || !d.strtab ) return nullptr;
95   const word *gh = d.gnu_hash;
96   const word nbuckets = gh[0];
97   const word symbias = gh[1];
98   const word bloom_size = gh[2];
99   const word bloom_shift = gh[3];
100   if ( !nbuckets || !bloom_size ) return nullptr;      // bloom_size==0 would divide-by-zero below
101 
102   const bloom_t *bloom = reinterpret_cast<const bloom_t *>(gh + 4);
103   const word *buckets = reinterpret_cast<const word *>(bloom + bloom_size);
104   const word *chain = buckets + nbuckets;
105 
106   const u32 h = gnu_hash(name);
107   const bloom_t bw = bloom[(h / bb) % bloom_size];
108   const bloom_t bit_a = static_cast<bloom_t>(1) << (h % bb);
109   // mask by the bloom word width
110   const bloom_t bit_b = static_cast<bloom_t>(1) << ((static_cast<bloom_t>(h) >> (bloom_shift & (bb - 1))) % bb);
111   if ( (bw & (bit_a | bit_b)) != (bit_a | bit_b) ) return nullptr;
112 
113   word idx = buckets[h % nbuckets];
114   if ( idx < symbias ) return nullptr;
115 
116   const word maxidx = d.symcount ? static_cast<word>(d.symcount) : (idx + (static_cast<word>(1) << 24));
117   for ( ; idx < maxidx; ++idx ) {
118     const word chain_h = chain[idx - symbias];
119     if ( ((chain_h ^ h) >> 1) == 0 ) {
120       const auto &s = d.symtab[idx];
121       if ( __streq(d.strtab + s.name, name) ) return &s;
122     }
123     if ( chain_h & 1 ) break;
124   }
125   return nullptr;
126 }

130 inline const typename elf_traits<C>::sym *
131 lookup_sym(const dyn_info<C> &d, const char *name) noexcept
132 {
133   if ( const auto *s = gnu_lookup(d, name) ) return s;
134   return sysv_lookup(d, name);

137 template<fmt_class C>
138 inline half
139 sym_version_index(const dyn_info<C> &d, const typename elf_traits<C>::sym *s) noexcept
140 {
141   if ( !d.versym || !s || !d.symtab ) return ver_ndx_global;
142   const usize i = static_cast<usize>(s - d.symtab);
143   if ( d.symcount && i >= d.symcount ) return ver_ndx_global;
144   return elf_ver_ndx(d.versym[i]);
145 }

148 inline bool
149 sym_version_hidden(const dyn_info<C> &d, const typename elf_traits<C>::sym *s) noexcept
150 {
151   if ( !d.versym || !s || !d.symtab ) return false;
152   const usize i = static_cast<usize>(s - d.symtab);
153   if ( d.symcount && i >= d.symcount ) return false;

157 // libLLVM.so.22.1 is the largest .dynsym on a stock box at ~42k entries
158 inline constexpr xword max_dynsyms = xword(1) << 24;
159 
160 // WARNING: do NOT simplify this to (DT_STRTAB - DT_SYMTAB); wrong for lld
161 // which lays out .dynsym, .gnu.version, .gnu.version_r, .gnu.hash, .dynstr
162 template<fmt_class C>
163 inline xword
164 __dynsym_upper_bound(const dyn_info<C> &d) noexcept
165 {
166   using tr = elf_traits<C>;
167 
168   const u8 *const base = reinterpret_cast<const u8 *>(d.symtab);
169   if ( !base ) return 0;
170 
171   const void *const cands[] = {
172     reinterpret_cast<const void *>(d.strtab),     reinterpret_cast<const void *>(d.versym),
173     reinterpret_cast<const void *>(d.verdef),     reinterpret_cast<const void *>(d.verneed),
174     reinterpret_cast<const void *>(d.gnu_hash),   reinterpret_cast<const void *>(d.hash),
175     reinterpret_cast<const void *>(d.rela),       reinterpret_cast<const void *>(d.rel),
176     reinterpret_cast<const void *>(d.relr),       reinterpret_cast<const void *>(d.jmprel),
177     reinterpret_cast<const void *>(d.pltgot),     reinterpret_cast<const void *>(d.init_array),
178     reinterpret_cast<const void *>(d.fini_array), reinterpret_cast<const void *>(d.preinit_array),
179   };
180 
181   const u8 *first = nullptr;
182   for ( const void *c : cands ) {
183     const u8 *p = reinterpret_cast<const u8 *>(c);
184     if ( !p || p <= base ) continue;
185     if ( !first || p < first ) first = p;
186   }
187   if ( !first ) return 0;
188 
189   const usize ent = d.syment ? static_cast<usize>(d.syment) : sizeof(typename tr::sym);
190   if ( ent < sizeof(typename tr::sym) ) return 0;      // a bogus DT_SYMENT would mis-stride symtab[]
191 
192   return static_cast<xword>(static_cast<usize>(first - base) / ent);
193 }

195 // WARNING: the GNU hash chains cover only the exported definitions, [symoffset, symcount)
196 template<fmt_class C>
197 inline xword
198 __gnu_hash_high_water(const dyn_info<C> &d, xword limit) noexcept
199 {
200   using tr = elf_traits<C>;
201   using bloom_t = typename tr::uword;
202 
203   if ( !d.gnu_hash ) return 0;
204   const word *gh = d.gnu_hash;
205   const word nbuckets = gh[0];
206   const word symbias = gh[1];
207   const word bloom_size = gh[2];
208   if ( !nbuckets ) return symbias;
209 
210   const bloom_t *bloom = reinterpret_cast<const bloom_t *>(gh + 4);
211   const word *buckets = reinterpret_cast<const word *>(bloom + bloom_size);
212   const word *chain = buckets + nbuckets;
213 
214   const word cap = (limit && limit < max_dynsyms) ? static_cast<word>(limit) : static_cast<word>(max_dynsyms);
215 
216   word high = symbias;
217   for ( word b = 0; b < nbuckets; ++b ) {
218     word idx = buckets[b];
219     if ( idx < symbias ) continue;
220     for ( ; idx < cap; ++idx ) {
221       if ( idx >= high ) high = idx + 1;
222       if ( chain[idx - symbias] & 1 ) break;
223     }
224   }
225   return high;
226 }

228 template<fmt_class C>
229 inline xword
230 count_dynsyms(const dyn_info<C> &d) noexcept
231 {
232   if ( !d.symtab ) return 0;      // no table to index; every consumer bails on the null pointer
233 
234   const xword bound = __dynsym_upper_bound(d);
235 
236   if ( d.hash ) {
237     const xword n = static_cast<xword>(d.hash[1]);
238     if ( n && n <= max_dynsyms && (bound == 0 || n <= bound) ) return n;
239   }
240 
241   if ( bound ) {
242     const xword floor_hi = __gnu_hash_high_water(d, bound);
243     const xword n = floor_hi > bound ? floor_hi : bound;
244     return n <= max_dynsyms ? n : max_dynsyms;
245   }
246 
247   if ( d.gnu_hash ) {
248     const xword n = __gnu_hash_high_water(d, 0);
249     return n ? n : 1;
250   }
251 
252   return 1;
253 }
