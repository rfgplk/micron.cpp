129 template<> struct elf_traits<fmt_class::elf32> {
130   static constexpr fmt_class cls = fmt_class::elf32;
131   static constexpr u8 ident_class = elfclass32;
132 
133   using addr = addr32;
134   using off = off32;
135   using uword = word;
136   using sxwd = sword;
137   using ehdr = ehdr32_t;
138   using phdr = phdr32_t;
139   using shdr = shdr32_t;
140   using dyn = dyn32_t;
141   using sym = sym32_t;
142   using rel = rel32_t;
143   using rela = rela32_t;
144 
145   static constexpr usize bloom_bits = 32;      // DT_GNU_HASH bloom words are 32-bit here
146   static constexpr usize relr_bits = 31;       // slots named by one DT_RELR bitmap word
147 
148   static constexpr u32
149   r_sym(uword info) noexcept
150   {
151     return info >> 8;
152   }
153 
154   static constexpr u32
155   r_type(uword info) noexcept
156   {
157     return info & 0xffu;
158   }
159 };
