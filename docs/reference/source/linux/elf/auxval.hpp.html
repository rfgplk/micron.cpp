16 namespace micron

19 inline constexpr usize __auxv_cache_max = 48;
20 
21 struct __auxv_cache_t {
22   auxv_t e[__auxv_cache_max];
23   usize n;
24   u32 state;      // 0 = unread, 1 = published
25 };
26 
27 inline __auxv_cache_t __auxv_cache{};
28 
29 [[gnu::cold, gnu::noinline]] inline void
30 __auxv_load(void) noexcept
31 {
32   usize count = 0;
33   // NOTE: avoid pulling in io spaghetti
34   const long fd = micron::syscall(SYS_openat, -100 /* AT_FDCWD */, "/proc/self/auxv", 0 /* O_RDONLY */, 0);
35   if ( !micron::syscall_failed(fd) ) {
36     for ( ; count < __auxv_cache_max; ) {
37       auxv_t ent{};
38       const long r = micron::syscall(SYS_read, fd, &ent, sizeof(ent));
39       if ( r != static_cast<long>(sizeof(ent)) ) break;
40       if ( ent.a_type == at_null ) break;
41       __auxv_cache.e[count++] = ent;
42     }
43     micron::syscall(SYS_close, fd);
44   }
45   __auxv_cache.n = count;
46   // NOTE: avoid pulling in the whole atomic chain, spaghetti
47   __atomic_store_n(&__auxv_cache.state, 1u, __ATOMIC_RELEASE);
48 }
49 
50 inline unsigned long
51 getauxval(unsigned long type) noexcept
52 {
53   if ( __atomic_load_n(&__auxv_cache.state, __ATOMIC_ACQUIRE) == 0u ) [[unlikely]]
54     __auxv_load();
55   for ( usize i = 0; i < __auxv_cache.n; ++i ) {
56     if ( __auxv_cache.e[i].a_type == type ) return __auxv_cache.e[i].a_val;
57   }
58   return 0;
59 }
60 
61 inline bool
62 auxval_has(unsigned long type) noexcept
63 {
64   if ( __atomic_load_n(&__auxv_cache.state, __ATOMIC_ACQUIRE) == 0u ) [[unlikely]]
65     __auxv_load();
66   for ( usize i = 0; i < __auxv_cache.n; ++i ) {
67     if ( __auxv_cache.e[i].a_type == type ) return true;
68   }
69   return false;
70 }

72 template<typename T = void>
73 inline T *
74 getauxptr(unsigned long type) noexcept
75 {
76   return reinterpret_cast<T *>(getauxval(type));
77 }
