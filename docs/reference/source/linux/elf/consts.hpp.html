213 // ELF32 is (sym << 8 | type) in a word
214 // ELF64 is (sym << 32 | type) in an xword
215 inline constexpr u32
216 elf32_r_sym(word info) noexcept
217 {
218   return info >> 8;
219 }
220 
221 inline constexpr u32
222 elf32_r_type(word info) noexcept
223 {
224   return info & 0xffu;
225 }
226 
227 inline constexpr word
228 elf32_r_info(u32 sym, u32 type) noexcept
229 {
230   return static_cast<word>((sym << 8) | (type & 0xffu));
231 }
232 
233 inline constexpr u32
234 elf64_r_sym(xword info) noexcept
235 {
236   return static_cast<u32>(info >> 32);
237 }
238 
239 inline constexpr u32
240 elf64_r_type(xword info) noexcept
241 {
242   return static_cast<u32>(info & 0xffffffff);
243 }
244 
245 inline constexpr xword
246 elf64_r_info(u32 sym, u32 type) noexcept
247 {
248   return (static_cast<xword>(sym) << 32) | static_cast<xword>(type);
249 }
250 
251 inline constexpr u32
252 elf_r_sym(xword info) noexcept
253 {
254   if constexpr ( native_class == fmt_class::elf32 )
255     return elf32_r_sym(static_cast<word>(info));
256   else
257     return elf64_r_sym(info);
258 }
