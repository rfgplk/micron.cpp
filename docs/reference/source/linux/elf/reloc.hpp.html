27 struct reloc_view {
28   u64 offset = 0;      // r_offset, a vaddr relative to load_base
29   u32 sym = 0;         // index into .dynsym; stn_undef means "no symbol"
30   u32 type = 0;
31   i64 addend = 0;      // authoritative only when rela is true
32   bool rela = false;
33 };
34 
35 using resolve_fn = void *(*)(void *user, const char *name, u32 sym_index);
36 
37 struct reloc_ctx_t {
38   u8 *load_base = nullptr;
39   const dyn_info_t *d = nullptr;
40   resolve_fn resolve = nullptr;
41   void *user = nullptr;
42   u64 tls_modid = 0;
43   i64 tls_offset = 0;
44 };
45 
46 // we'll distinguish a tolerable miss from a hard failure (thread_local et al)
47 enum class reloc_result : u8 {
48   applied,           // written successfully
49   skipped_weak,      // symbol unresolved but weak
50   unresolved,        // non-weak symbol could not be resolved
51   unsupported,       // relocation type not implemented
52 };

57 template<fmt_class C>
58 inline reloc_view
59 reloc_view_of(const typename elf_traits<C>::rela &r) noexcept
60 {
61   using tr = elf_traits<C>;
62   reloc_view v;
63   v.offset = static_cast<u64>(r.offset);
64   v.sym = tr::r_sym(r.info);
65   v.type = tr::r_type(r.info);
66   v.addend = static_cast<i64>(static_cast<typename tr::sxwd>(r.addend));
67   v.rela = true;
68   return v;
69 }
70 
71 template<fmt_class C>
72 inline reloc_view
73 reloc_view_of(const typename elf_traits<C>::rel &r) noexcept
74 {
75   using tr = elf_traits<C>;
76   reloc_view v;
77   v.offset = static_cast<u64>(r.offset);
78   v.sym = tr::r_sym(r.info);
79   v.type = tr::r_type(r.info);
80   v.addend = 0;      // implicit; the backend reads it out of the slot
81   v.rela = false;
82   return v;
83 }

105 inline i64
106 __implicit_addend(const u8 *p) noexcept
107 {
108   if constexpr ( native_class == fmt_class::elf32 )
109     return static_cast<i64>(static_cast<i32>(*micron::ptr_cast<const u32 *>(p)));
110   else
111     return static_cast<i64>(*micron::ptr_cast<const u64 *>(p));
112 }
113 
114 inline void *
115 __resolve_for(const reloc_ctx_t &ctx, u32 si, bool &weak) noexcept
116 {
117   weak = false;
118   if ( !ctx.d || !ctx.d->symtab || !ctx.d->strtab ) return nullptr;
119   const char *name = ctx.d->strtab + ctx.d->symtab[si].name;
120   weak = elf_st_bind(ctx.d->symtab[si].info) == stb_weak;
121   return ctx.resolve ? ctx.resolve(ctx.user, name, si) : nullptr;
122 }

126 // amd64 reloc
127 inline reloc_result
128 apply_reloc(const reloc_ctx_t &ctx, const reloc_view &r) noexcept
129 {
130   const u32 type = r.type;
131   const u32 si = r.sym;
132   if ( !ctx.d ) return reloc_result::unresolved;
133   if ( si != stn_undef && (!ctx.d->symtab || si >= ctx.d->symcount) ) return reloc_result::unresolved;
134   if ( si != stn_undef && elf_st_type(ctx.d->symtab[si].info) == stt_tls && ctx.d->symtab[si].shndx == shn_undef )
135     return reloc_result::unsupported;
136   u8 *const p = ctx.load_base + r.offset;
137   const i64 a = r.rela ? r.addend : __implicit_addend(p);
138   const u8 *const b = ctx.load_base;
139 
140   switch ( type ) {
141   case r_x86_64_none:
142     return reloc_result::applied;
143   case r_x86_64_relative:
144     *reinterpret_cast<u64 *>(p) = reinterpret_cast<u64>(b) + static_cast<u64>(a);
145     return reloc_result::applied;
146   case r_x86_64_irelative: {
147     using ifn = u64 (*)();
148     ifn fn = reinterpret_cast<ifn>(reinterpret_cast<u64>(b) + static_cast<u64>(a));
149     *reinterpret_cast<u64 *>(p) = fn();
150     return reloc_result::applied;
151   }
152   case r_x86_64_glob_dat:
153   case r_x86_64_jump_slot:
154   case r_x86_64_64: {
155     bool weak = false;
156     void *s = __resolve_for(ctx, si, weak);
157     if ( !s ) return weak ? reloc_result::skipped_weak : reloc_result::unresolved;
158     *reinterpret_cast<u64 *>(p) = reinterpret_cast<u64>(s) + (type == r_x86_64_64 ? static_cast<u64>(a) : 0);
159     return reloc_result::applied;
160   }
161   case r_x86_64_pc32:
162   case r_x86_64_plt32: {
163     bool weak = false;
164     void *s = __resolve_for(ctx, si, weak);
165     if ( !s ) return weak ? reloc_result::skipped_weak : reloc_result::unresolved;
166     i64 v = static_cast<i64>(reinterpret_cast<u64>(s)) + a - static_cast<i64>(reinterpret_cast<u64>(p));
167     *reinterpret_cast<i32 *>(p) = static_cast<i32>(v);
168     return reloc_result::applied;
169   }
170   case r_x86_64_dtpmod64:
171     *reinterpret_cast<u64 *>(p) = ctx.tls_modid;
172     return reloc_result::applied;
173   case r_x86_64_dtpoff64:
174     // NOTE: resolves the offset within THIS module's TLS block only
175     *reinterpret_cast<u64 *>(p) = ctx.d->symtab[si].value + static_cast<u64>(a);
176     return reloc_result::applied;
177   case r_x86_64_tpoff64:
178     // WARNING: static (initial-exec/local-exec) TLS is __NOT__ implemented
179     return reloc_result::unsupported;
180   case r_x86_64_copy:
181     return reloc_result::unsupported;
182   default:
183     return reloc_result::unsupported;
184   }
185 }

236   case r_aarch64_tls_tprel:

240   case r_aarch64_copy:

249 // i386
250 inline reloc_result
251 apply_reloc(const reloc_ctx_t &ctx, const reloc_view &r) noexcept
252 {
253   const u32 type = r.type;
254   const u32 si = r.sym;
255   if ( !ctx.d ) return reloc_result::unresolved;
256   if ( si != stn_undef && (!ctx.d->symtab || si >= ctx.d->symcount) ) return reloc_result::unresolved;
257   if ( si != stn_undef && elf_st_type(ctx.d->symtab[si].info) == stt_tls && ctx.d->symtab[si].shndx == shn_undef )
258     return reloc_result::unsupported;
259   u8 *const p = ctx.load_base + r.offset;
260   const i64 a = r.rela ? r.addend : __implicit_addend(p);
261   const u32 b = static_cast<u32>(reinterpret_cast<uintptr_t>(ctx.load_base));
262 
263   switch ( type ) {
264   case r_386_none:
265     return reloc_result::applied;
266   case r_386_relative:
267     *reinterpret_cast<u32 *>(p) = b + static_cast<u32>(a);
268     return reloc_result::applied;
269   case r_386_irelative: {
270     using ifn = u32 (*)();
271     ifn fn = reinterpret_cast<ifn>(static_cast<uintptr_t>(b + static_cast<u32>(a)));
272     *reinterpret_cast<u32 *>(p) = fn();
273     return reloc_result::applied;
274   }
275   case r_386_glob_dat:
276   case r_386_jmp_slot: {
277     // the addend is ignored for these two even in REL form
278     bool weak = false;
279     void *s = __resolve_for(ctx, si, weak);
280     if ( !s ) return weak ? reloc_result::skipped_weak : reloc_result::unresolved;
281     *reinterpret_cast<u32 *>(p) = static_cast<u32>(reinterpret_cast<uintptr_t>(s));
282     return reloc_result::applied;
283   }
284   case r_386_32: {
285     bool weak = false;
286     void *s = __resolve_for(ctx, si, weak);
287     if ( !s ) return weak ? reloc_result::skipped_weak : reloc_result::unresolved;
288     *reinterpret_cast<u32 *>(p) = static_cast<u32>(reinterpret_cast<uintptr_t>(s)) + static_cast<u32>(a);
289     return reloc_result::applied;
290   }
291   case r_386_pc32:
292   case r_386_plt32: {
293     bool weak = false;
294     void *s = __resolve_for(ctx, si, weak);
295     if ( !s ) return weak ? reloc_result::skipped_weak : reloc_result::unresolved;
296     const u32 sv = static_cast<u32>(reinterpret_cast<uintptr_t>(s));
297     *reinterpret_cast<u32 *>(p) = sv + static_cast<u32>(a) - static_cast<u32>(reinterpret_cast<uintptr_t>(p));
298     return reloc_result::applied;
299   }
300   case r_386_tls_dtpmod32:
301     *reinterpret_cast<u32 *>(p) = static_cast<u32>(ctx.tls_modid);
302     return reloc_result::applied;
303   case r_386_tls_dtpoff32:
304     *reinterpret_cast<u32 *>(p) = static_cast<u32>(ctx.d->symtab[si].value) + static_cast<u32>(a);
305     return reloc_result::applied;
306   case r_386_tls_tpoff:
307   case r_386_tls_ie:
308   case r_386_tls_gotie:
309   case r_386_tls_le:
310   case r_386_tls_tpoff32:
311   case r_386_tls_ie_32:
312   case r_386_tls_le_32:
313   case r_386_tls_desc:
314     return reloc_result::unsupported;
315   case r_386_copy:
316     return reloc_result::unsupported;
317   default:
318     return reloc_result::unsupported;
319   }
320 }
