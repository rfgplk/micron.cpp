295 // the processor range, which only e_machine can disambiguate. falls back to the generic table for
296 // every tag below DT_LOPROC.
297 constexpr const char *
298 dt_tag_name(half machine, i64 tag) noexcept
299 {
300   if ( tag < dt_loproc || tag > dt_hiproc ) return dt_tag_name(tag);
301   switch ( machine ) {
302   case em_arm:
303     switch ( tag ) {
304     case dt_arm_exidx:
305       return "DT_ARM_EXIDX";
306     case dt_arm_exidxsz:
307       return "DT_ARM_EXIDXSZ";
308     default:
309       return nullptr;
310     }
311   case em_x86_64:
312     switch ( tag ) {
313     case 0x70000000:
314       return "DT_X86_64_PLT";
315     case 0x70000001:
316       return "DT_X86_64_PLTSZ";
317     case 0x70000003:
318       return "DT_X86_64_PLTENT";
319     default:
320       return nullptr;
321     }
322   default:
323     return nullptr;
324   }
325 }
