24 struct dyn_row {
25   i64 tag = 0;
26   u64 val = 0;

29 struct dynamic_info {
30   micron::vector<dyn_row> entries{};
31 
32   micron::vector<micron::string> needed{};
33   micron::string soname{};
34   micron::string rpath{};
35   micron::string runpath{};
36   bool present = false;
37   bool from_section = false;      // PT_DYNAMIC was absent and the .dynamic SECTION answered instead
38 };

40 constexpr bool
41 dt_is_strtab_offset(i64 tag)
42 {
43   return tag == dt_needed || tag == dt_soname || tag == dt_rpath || tag == dt_runpath;
44 }

46 inline dynamic_info
47 read_dynamic(const image &img, const micron::vector<segment_row> &segs, const micron::vector<section_row> &secs)
48 {
49   dynamic_info out{};
50   if ( !img.ok() || img.src == nullptr ) return out;
51 
52   u64 base = 0, span = 0;
53   const segment_row *dynseg = find_segment(segs, pt_dynamic);
54   if ( dynseg != nullptr ) {
55     base = dynseg->offset;
56     span = dynseg->filesz;
57   } else {
58     // a relocatable object or a stripped-phdr file still carries .dynamic as a section
59     const section_row *dynsec = find_section_by_type(secs, sht_dynamic);
60     if ( dynsec == nullptr || dynsec->type == sht_nobits ) return out;
61     base = dynsec->offset;
62     span = dynsec->size;
63     out.from_section = true;
64   }
65   out.present = true;
66 
67   const u64 entsize = img.is64() ? sizeof(dyn_t) : sizeof(dyn32_t);
68   if ( span < entsize ) return out;
69   const u64 count = clamp_records(img, base, entsize, span / entsize);
70 
71   micron::vector<dyn_row> entries{};
72   entries.reserve(static_cast<usize>(count));
73   for ( u64 i = 0; i < count; i++ ) {
74     const u64 off = base + i * entsize;
75     dyn_row e{};
76     if ( img.is64() ) {
77       const span_t raw = img.at(off, sizeof(dyn_t));
78       if ( raw.len < sizeof(dyn_t) ) break;
79       e.tag = static_cast<i64>(rd<u64>(raw.ptr, __builtin_offsetof(dyn_t, tag), img.hdr.data));
80       e.val = rd<u64>(raw.ptr, __builtin_offsetof(dyn_t, un), img.hdr.data);
81     } else {
82       const span_t raw = img.at(off, sizeof(dyn32_t));
83       if ( raw.len < sizeof(dyn32_t) ) break;
84       // d_tag is a SIGNED word on ELF32; the OS/proc-range tags are negative when read unsigned
85       e.tag = static_cast<i64>(static_cast<i32>(rd<u32>(raw.ptr, __builtin_offsetof(dyn32_t, tag), img.hdr.data)));
86       e.val = rd<u32>(raw.ptr, __builtin_offsetof(dyn32_t, un), img.hdr.data);
87     }
88     entries.push_back(e);
89     if ( e.tag == dt_null ) break;
90   }
91 
92   out.entries = micron::move(entries);
93 
94   u64 strtab_vaddr = 0;
95   bool have_strtab = false;
96   u64 strsz = 0;
97   micron::vector<u64> needed_offs{};
98   u64 soname_off = ~0ull, rpath_off = ~0ull, runpath_off = ~0ull;
99 
100   for ( const auto &e : out.entries ) {
101     if ( e.tag == dt_strtab ) {
102       strtab_vaddr = e.val;
103       have_strtab = true;
104     } else if ( e.tag == dt_strsz )
105       strsz = e.val;
106     else if ( e.tag == dt_needed )
107       needed_offs.push_back(e.val);
108     else if ( e.tag == dt_soname )
109       soname_off = e.val;
110     else if ( e.tag == dt_rpath )
111       rpath_off = e.val;
112     else if ( e.tag == dt_runpath )
113       runpath_off = e.val;
114   }
115 
116   if ( !have_strtab ) return out;
117 
118   // DT_STRTAB is a VADDR; turn it into a file offset, falling back to .dynstr when no PT_LOAD
119   // covers it (which happens in an unlinked or section-only file)
120   u64 strtab_off = 0;
121   const auto off_r = vaddr_to_offset(segs, strtab_vaddr);
122   if ( off_r.is_second() ) {
123     const section_row *ds = find_section(secs, ".dynstr");
124     if ( ds == nullptr ) return out;
125     strtab_off = ds->offset;
126     if ( strsz == 0 ) strsz = ds->size;
127   } else {
128     strtab_off = off_r.cast<u64>();
129   }
130   const u64 strtab_limit = strsz != 0 ? strsz : (64ull << 10);
131 
132   const auto resolve = [&](u64 name_off) {
133     const u64 lim = strtab_limit > name_off ? strtab_limit - name_off : 0;
134     return read_cstr_at(*img.src, strtab_off + name_off, lim);
135   };
136 
137   for ( u64 noff : needed_offs ) out.needed.push_back(resolve(noff));
138   if ( soname_off != ~0ull ) out.soname = resolve(soname_off);
139   if ( rpath_off != ~0ull ) out.rpath = resolve(rpath_off);
140   if ( runpath_off != ~0ull ) out.runpath = resolve(runpath_off);
141 
142   return out;
143 }
