42 struct image {
43   const source *src = nullptr;
44   header hdr{};
45 
46   bool
47   ok() const noexcept
48   {
49     return src != nullptr && (hdr.cls == fmt_class::elf32 || hdr.cls == fmt_class::elf64);
50   }
51 
52   bool
53   is64() const noexcept
54   {
55     return hdr.cls == fmt_class::elf64;
56   }
57 
58   bool
59   is_native() const noexcept
60   {
61     return hdr.cls == native_class && hdr.data == native_data && (expected_machine == 0 || hdr.machine == expected_machine);
62   }

64   span_t

67     return src ? src->at(off, want) : span_t{};

71 inline u64
72 records_available(const image &img, u64 off, u64 esz) noexcept
73 {
74   if ( esz == 0 || img.src == nullptr ) return 0;
75   const u64 len = static_cast<u64>(img.src->size());
76   if ( off >= len ) return 0;
77   return (len - off) / esz;
78 }
79 
80 inline u64
81 clamp_records(const image &img, u64 off, u64 esz, u64 want) noexcept
82 {
83   const u64 room = records_available(img, off, esz);
84   return want < room ? want : room;
85 }

87 inline micron::option<image, const char *>
88 open_image(const source &src) noexcept
89 {
90   using res = micron::option<image, const char *>;
91 
92   const span_t ident = src.at(0, ident_size);
93   if ( ident.len < ident_size ) return res{ "too short to be an elf file" };
94 
95   const u8 *id = ident.ptr;
96   if ( id[ei_mag0] != mag0 || id[ei_mag1] != static_cast<u8>(mag1) || id[ei_mag2] != static_cast<u8>(mag2)
97        || id[ei_mag3] != static_cast<u8>(mag3) )
98     return res{ "bad magic" };
99 
100   fmt_class cls = fmt_class::invalid;
101   if ( id[ei_class] == elfclass32 )
102     cls = fmt_class::elf32;
103   else if ( id[ei_class] == elfclass64 )
104     cls = fmt_class::elf64;
105   else
106     return res{ "unknown elf class" };
107 
108   fmt_data data = fmt_data::invalid;
109   if ( id[ei_data] == elfdata2lsb )
110     data = fmt_data::lsb;
111   else if ( id[ei_data] == elfdata2msb )
112     data = fmt_data::msb;
113   else
114     return res{ "unknown elf data encoding" };
115 
116   if ( id[ei_version] != static_cast<u8>(ev_current) ) return res{ "unsupported elf version" };
117 
118   header hdr{};
119   hdr.cls = cls;
120   hdr.data = data;
121   hdr.osabi = id[ei_osabi];
122   hdr.abiversion = id[ei_abiversion];
123 
124   if ( cls == fmt_class::elf64 ) {
125     const span_t raw = src.at(0, sizeof(ehdr_t));
126     if ( raw.len < sizeof(ehdr_t) ) return res{ "truncated elf64 header" };
127     hdr.type = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, type), data);
128     hdr.machine = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, machine), data);
129     hdr.version = rd<u32>(raw.ptr, __builtin_offsetof(ehdr_t, version), data);
130     hdr.entry = rd<u64>(raw.ptr, __builtin_offsetof(ehdr_t, entry), data);
131     hdr.phoff = rd<u64>(raw.ptr, __builtin_offsetof(ehdr_t, phoff), data);
132     hdr.shoff = rd<u64>(raw.ptr, __builtin_offsetof(ehdr_t, shoff), data);
133     hdr.flags = rd<u32>(raw.ptr, __builtin_offsetof(ehdr_t, flags), data);
134     hdr.ehsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, ehsize), data);
135     hdr.phentsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, phentsize), data);
136     hdr.phnum = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, phnum), data);
137     hdr.shentsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, shentsize), data);
138     hdr.shnum = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, shnum), data);
139     hdr.shstrndx = rd<u16>(raw.ptr, __builtin_offsetof(ehdr_t, shstrndx), data);
140   } else {
141     const span_t raw = src.at(0, sizeof(ehdr32_t));
142     if ( raw.len < sizeof(ehdr32_t) ) return res{ "truncated elf32 header" };
143     hdr.type = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, type), data);
144     hdr.machine = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, machine), data);
145     hdr.version = rd<u32>(raw.ptr, __builtin_offsetof(ehdr32_t, version), data);
146     hdr.entry = rd<u32>(raw.ptr, __builtin_offsetof(ehdr32_t, entry), data);
147     hdr.phoff = rd<u32>(raw.ptr, __builtin_offsetof(ehdr32_t, phoff), data);
148     hdr.shoff = rd<u32>(raw.ptr, __builtin_offsetof(ehdr32_t, shoff), data);
149     hdr.flags = rd<u32>(raw.ptr, __builtin_offsetof(ehdr32_t, flags), data);
150     hdr.ehsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, ehsize), data);
151     hdr.phentsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, phentsize), data);
152     hdr.phnum = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, phnum), data);
153     hdr.shentsize = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, shentsize), data);
154     hdr.shnum = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, shnum), data);
155     hdr.shstrndx = rd<u16>(raw.ptr, __builtin_offsetof(ehdr32_t, shstrndx), data);
156   }
157 
158   image img{};
159   img.src = &src;
160   img.hdr = hdr;
161   return res{ micron::move(img) };
162 }
