41 inline void
42 read_one_symbol(const image &img, const u8 *raw, symbol_row &row, u64 &name_off)
43 {
44   u8 info = 0, other = 0;
45   if ( img.is64() ) {
46     name_off = rd<u32>(raw, __builtin_offsetof(sym_t, name), img.hdr.data);
47     row.value = rd<u64>(raw, __builtin_offsetof(sym_t, value), img.hdr.data);
48     row.size = rd<u64>(raw, __builtin_offsetof(sym_t, size), img.hdr.data);
49     info = raw[__builtin_offsetof(sym_t, info)];
50     other = raw[__builtin_offsetof(sym_t, other)];
51     row.shndx = rd<u16>(raw, __builtin_offsetof(sym_t, shndx), img.hdr.data);
52   } else {
53     name_off = rd<u32>(raw, __builtin_offsetof(sym32_t, name), img.hdr.data);
54     row.value = rd<u32>(raw, __builtin_offsetof(sym32_t, value), img.hdr.data);
55     row.size = rd<u32>(raw, __builtin_offsetof(sym32_t, size), img.hdr.data);
56     info = raw[__builtin_offsetof(sym32_t, info)];
57     other = raw[__builtin_offsetof(sym32_t, other)];
58     row.shndx = rd<u16>(raw, __builtin_offsetof(sym32_t, shndx), img.hdr.data);
59   }
60   row.bind = elf_st_bind(info);
61   row.type = elf_st_type(info);
62   row.visibility = elf_st_visibility(other);
63   row.defined = row.shndx != shn_undef;
64 }

66 inline void
67 walk_one_table(const image &img, const section_row *symtab_sec, const micron::vector<section_row> &secs, sym_table_kind kind,
68                micron::vector<symbol_row> &out)
69 {
70   if ( symtab_sec == nullptr || img.src == nullptr ) return;
71   const u64 entsize = img.is64() ? sizeof(sym_t) : sizeof(sym32_t);
72   if ( symtab_sec->size < entsize ) return;
73   const u64 count = clamp_records(img, symtab_sec->offset, entsize, symtab_sec->size / entsize);
74 
75   const section_row *strtab_sec = symtab_sec->link < secs.size() ? &secs[symtab_sec->link] : nullptr;
76 
77   out.reserve(out.size() + static_cast<usize>(count));
78   for ( u64 i = 0; i < count; i++ ) {
79     const u64 off = symtab_sec->offset + i * entsize;
80     const span_t raw = img.at(off, entsize);
81     if ( raw.len < entsize ) break;
82 
83     symbol_row row{};
84     row.table = kind;
85     u64 name_off = 0;
86     read_one_symbol(img, raw.ptr, row, name_off);
87     if ( strtab_sec != nullptr && name_off != 0 ) {
88       const u64 lim = strtab_sec->size > name_off ? strtab_sec->size - name_off : 0;
89       row.name = read_cstr_at(*img.src, strtab_sec->offset + name_off, lim);
90     }
91     out.push_back(micron::move(row));
92   }
93 }

97 inline micron::vector<symbol_row>
98 walk_symbols(const image &img, const micron::vector<section_row> &secs)
99 {
100   micron::vector<symbol_row> out{};
101   if ( !img.ok() ) return out;
102   __impl::walk_one_table(img, find_section_by_type(secs, sht_symtab), secs, sym_table_kind::static_symtab, out);
103   __impl::walk_one_table(img, find_section_by_type(secs, sht_dynsym), secs, sym_table_kind::dynamic_dynsym, out);
104   return out;
105 }
