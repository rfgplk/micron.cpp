22 struct segment_row {
23   u32 type = 0;
24   u32 flags = 0;
25   u64 offset = 0;
26   u64 vaddr = 0;
27   u64 paddr = 0;
28   u64 filesz = 0;
29   u64 memsz = 0;
30   u64 align = 0;
31 };

35 inline micron::vector<segment_row>
36 walk_segments(const image &img)
37 {
38   micron::vector<segment_row> out{};
39   if ( !img.ok() || img.hdr.phoff == 0 || img.hdr.phnum == 0 ) return out;
40 
41   out.reserve(static_cast<usize>(clamp_records(img, img.hdr.phoff, img.hdr.phentsize, img.hdr.phnum)));
42   for ( u16 i = 0; i < img.hdr.phnum; i++ ) {
43     const u64 off = img.hdr.phoff + static_cast<u64>(i) * static_cast<u64>(img.hdr.phentsize);
44     segment_row row{};
45 
46     if ( img.is64() ) {
47       const span_t raw = img.at(off, sizeof(phdr_t));
48       if ( raw.len < sizeof(phdr_t) ) break;
49       row.type = rd<u32>(raw.ptr, __builtin_offsetof(phdr_t, type), img.hdr.data);
50       row.flags = rd<u32>(raw.ptr, __builtin_offsetof(phdr_t, flags), img.hdr.data);
51       row.offset = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, offset), img.hdr.data);
52       row.vaddr = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, vaddr), img.hdr.data);
53       row.paddr = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, paddr), img.hdr.data);
54       row.filesz = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, filesz), img.hdr.data);
55       row.memsz = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, memsz), img.hdr.data);
56       row.align = rd<u64>(raw.ptr, __builtin_offsetof(phdr_t, align), img.hdr.data);
57     } else {
58       // WARNING: phdr32_t moves p_flags from field 2 to field 7
59       const span_t raw = img.at(off, sizeof(phdr32_t));
60       if ( raw.len < sizeof(phdr32_t) ) break;
61       row.type = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, type), img.hdr.data);
62       row.flags = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, flags), img.hdr.data);
63       row.offset = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, offset), img.hdr.data);
64       row.vaddr = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, vaddr), img.hdr.data);
65       row.paddr = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, paddr), img.hdr.data);
66       row.filesz = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, filesz), img.hdr.data);
67       row.memsz = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, memsz), img.hdr.data);
68       row.align = rd<u32>(raw.ptr, __builtin_offsetof(phdr32_t, align), img.hdr.data);
69     }
70     out.push_back(row);
71   }
72   return out;
73 }

75 inline const segment_row *
76 find_segment(const micron::vector<segment_row> &segs, u32 pt)
77 {
78   for ( const auto &s : segs )
79     if ( s.type == pt ) return &s;
80   return nullptr;
81 }

83 inline micron::option<u64, const char *>
84 vaddr_to_offset(const micron::vector<segment_row> &segs, u64 vaddr)
85 {
86   using res = micron::option<u64, const char *>;
87   for ( const auto &s : segs ) {
88     if ( s.type != pt_load ) continue;
89     if ( vaddr >= s.vaddr && vaddr < s.vaddr + s.filesz ) return res{ s.offset + (vaddr - s.vaddr) };
90   }
91   return res{ "vaddr not backed by a PT_LOAD segment" };
92 }
93 
94 inline micron::option<micron::string, const char *>
95 read_interp(const image &img, const micron::vector<segment_row> &segs)
96 {
97   using res = micron::option<micron::string, const char *>;
98   const segment_row *seg = find_segment(segs, pt_interp);
99   if ( seg == nullptr || img.src == nullptr ) return res{ "no PT_INTERP segment" };
100   return res{ read_cstr_at(*img.src, seg->offset, seg->filesz) };
101 }
102 
103 inline link_kind
104 classify_link(const micron::vector<segment_row> &segs)
105 {
106   if ( find_segment(segs, pt_interp) != nullptr ) return link_kind::dynamic;
107   if ( find_segment(segs, pt_dynamic) != nullptr ) return link_kind::static_pie;
108   return link_kind::static_exec;
109 }
