78 inline runtime_t
79 load_stack_heap(void)
80 {
81   micron::string dt;
82   io::open_file("/proc/self/maps") >> dt;
83   micron::string::iterator stck = micron::format::find(dt, "[stack]");
84   micron::string::iterator heap = micron::format::find(dt, "[heap]");
85   // NOTE: the reason this is here is because we might not always have a heap, in which case heap will be nullptr
86   if ( !stck and !heap ) {
87     runtime_t rt = { .stack = nullptr, .heap = nullptr };
88     return rt;
89   } else if ( heap ) {
90     micron::string::iterator stck_nl = micron::format::find_reverse(dt, stck, "\n") + 1;
91     micron::string::iterator heap_nl = micron::format::find_reverse(dt, heap, "\n") + 1;
92     micron::string::iterator stck_ptr = micron::format::find(dt, stck_nl, "-");
93     micron::string::iterator heap_ptr = micron::format::find(dt, heap_nl, "-");
94     byte *sptr = micron::format::to_pointer_addr<byte, micron::string>(stck_nl, stck_ptr);
95     byte *hptr = micron::format::to_pointer_addr<byte, micron::string>(heap_nl, heap_ptr);
96     runtime_t rt = { .stack = sptr, .heap = hptr };
97     return rt;
98   } else {
99     micron::string::iterator stck_nl = micron::format::find_reverse(dt, stck, "\n") + 1;
100     micron::string::iterator stck_ptr = micron::format::find(dt, stck_nl, "-");
101     byte *sptr = micron::format::to_pointer_addr<byte, micron::string>(stck_nl, stck_ptr);
102     runtime_t rt = { .stack = sptr, .heap = nullptr };
103     return rt;
104   }
105   runtime_t rt = { .stack = nullptr, .heap = nullptr };
106   return rt;
107 }

109 inline runtime_t
110 load_stack_heap_pid(pid_t pid)
111 {
112   micron::string dt;
113   io::open_file(__impl::proc_path(pid, "maps").c_str()) >> dt;
114 
115   runtime_t rt{ nullptr, nullptr };
116   if ( dt.empty() ) return rt;
117 
118   micron::string::iterator stck = micron::format::find(dt, "[stack]");
119   micron::string::iterator heap = micron::format::find(dt, "[heap]");
120 
121   if ( stck ) {
122     micron::string::iterator nl = micron::format::find_reverse(dt, stck, "\n") + 1;
123     micron::string::iterator dash = micron::format::find(dt, nl, "-");
124     rt.stack = micron::format::to_pointer_addr<byte, micron::string>(nl, dash);
125   }
126   if ( heap ) {
127     micron::string::iterator nl = micron::format::find_reverse(dt, heap, "\n") + 1;
128     micron::string::iterator dash = micron::format::find(dt, nl, "-");
129     rt.heap = micron::format::to_pointer_addr<byte, micron::string>(nl, dash);
130   }
131   return rt;
132 }

145 // uses containers rather than PODs, redesigned with spawn_ctx as info
146 struct uprocess_t {
147   upid_t pids;      // pid of process
148   micron::sstring<posix::path_max + 1> path;
149   micron::svector<micron::string> argv;      // can be variable, up to cca 2 MiB
150   micron::svector<micron::string> envp;
151   posix::limits_t lims;      // limits of the process, slower to get, but makes handling effortless
152   posix::cpu_set_t affinity;
153   int status;
154   ~uprocess_t() = default;
155 
156   uprocess_t(void)
157       : pids{ posix::getuid(), posix::geteuid(), posix::getgid(), posix::getpid(), posix::getppid() }, path(), argv{}, envp{}, lims(0),
158         affinity(), status(0)
159   {
160     // programatically get argv and environ
161     micron::string str;
162     io::open_file("/proc/self/cmdline") >> str;
163     umax_t k = 0;
164     umax_t t = 0;
165     // stored as null term strings, so we're iterating according to what we read
166     for ( usize i = 0; i < str.size(); ++i ) {
167       if ( str[i] == 0x0 ) {
168         t = i;
169         argv.emplace_back(str.begin() + k, str.begin() + t);
170         k = t + 1;
171       }
172     }
173     // argv[0] is always path, not absolute but works without reparsing
174     if ( !argv.empty() ) path = argv[0];
175     str.clear();
176     k = 0;
177     t = 0;
178     io::open_file("/proc/self/environ") >> str;
179     for ( usize i = 0; i < str.size(); ++i ) {
180       if ( str[i] == 0x0 ) {
181         t = i;
182         envp.emplace_back(str.begin() + k, str.begin() + t);
183         // our vector can only hold 64 elements, and as it turns out environ could easily exceed that
184         if ( envp.full_or_overflowed() ) break;
185         k = t + 1;
186       }
187     }
188     posix::get_affinity(pids.pid, affinity);
189   }
190 
191   template<typename... Args>
192   uprocess_t(const char *str, Args &&...args)
193       : pids{}, path(str), argv{ str, micron::forward<Args>(args)... }, envp{}, lims(0), affinity(), status(0)
194   {
195   }
196 
197   template<typename... Args>
198   uprocess_t(micron::sstring<posix::path_max + 1> &&o, Args &&...args)
199       : pids{}, path(micron::move(o)), argv{ path.c_str(), micron::forward<Args>(args)... }, envp{}, lims(0), affinity(), status(0)
200   {
201     // path is initialized before argv (declaration order), so path.c_str() is the moved-in name
202   }
203 
204   uprocess_t(const uprocess_t &o)
205       : pids(o.pids), path(o.path), argv(o.argv), envp(o.envp), lims(o.lims), affinity(o.affinity), status(o.status)
206   {
207   }
208 
209   uprocess_t(uprocess_t &&o)
210       : pids(micron::move(o.pids)), path(micron::move(o.path)), argv(micron::move(o.argv)), envp(micron::move(o.envp)),
211         lims(micron::move(o.lims)), affinity(micron::move(o.affinity)), status(o.status)
212   {
213     o.status = 0;
214   }
215 
216   template<typename... Args>
217   uprocess_t &
218   operator=(uprocess_t &&o)
219   {
220 
221     pids = micron::move(o.pids);
222     path = micron::move(o.path);
223     argv = micron::move(o.argv);

282 // implementation of daemon
283 template<int Stack = default_stack_size, typename F, typename... Args>
284   requires micron::is_function_v<F> or micron::is_function_v<micron::remove_pointer_t<F>>
285 int
286 daemon(F f, Args &&...args)
287 {
288   int pid = micron::fork();
289   if ( pid < 0 ) exc<except::runtime_error>("micron process daemon: first fork failed");
290   if ( pid > 0 ) micron::posix::exit(0);      // parent exits
291 
292   if ( posix::setsid() < 0 ) exc<except::runtime_error>("micron process daemon failed to create new session");
293 
294   // double-fork: the second child is NOT a session leader, so it can never reacquire a tty
295   int pid2 = micron::fork();
296   if ( pid2 < 0 ) exc<except::runtime_error>("micron process daemon: second fork failed");
297   if ( pid2 > 0 ) micron::posix::exit(0);
298 
299   // don't change dir
300   micron::umask(0);
301   micron::close(stdin_fileno);
302   micron::close(stdout_fileno);
303   micron::close(stderr_fileno);
304 
305   int nfd = micron::open("/dev/null", posix::o_rdwr);      // lowest free fd == 0 (just closed)
306   if ( nfd == 0 ) {
307     micron::dup(0);      // -> 1
308     micron::dup(0);      // -> 2
309   }
310   f(micron::forward<Args>(args)...);      // forward, don't decay to lvalues
311   return 0;
312 }

338 inline void
339 read_exe(pid_t pid, char *buf, usize bufsz)
340 {
341   if ( bufsz == 0 ) return;
342   buf[0] = '\0';
343   auto path = __impl::proc_path(pid, "exe");
344   // readlink syscall
345   max_t n = micron::readlink(path.c_str(), buf, bufsz - 1);
346   if ( n > 0 ) buf[n] = '\0';
347 }

351 inline proc_info_t
352 get_proc_info(pid_t pid)
353 {
354   proc_info_t info{};
355   info.pid = pid;
356   info.stat = read_proc_stat(pid);
357   info.status = read_proc_status(pid);
358   info.ppid = info.stat.ppid;
359   info.resources.stat = info.stat;
360   info.resources.status = info.status;
361   info.resources.limits = posix::limits_t(pid);
362   info.resources.usage = get_rusage_proc(pid);
363 
364   info.caps = ucap_set_t(info.status.cap_eff, info.status.cap_prm, info.status.cap_inh, info.status.cap_bnd, info.status.cap_amb);
365 
366   info.runtime = load_stack_heap_pid(pid);
367   __impl::read_exe(pid, info.exe, sizeof(info.exe));
368   return info;
369 }

371 inline proc_info_t
372 this_proc_info()

387   info.runtime = load_stack_heap();
388   __impl::read_exe(0, info.exe, sizeof(info.exe));
389   return info;

408 inline child_result_t
409 wait_and_collect(pid_t pid)
410 {
411   child_result_t r{};
412   r.pid = pid;
413 
414   // read /proc/PID resources while the child is still a (possibly zombie) process, THEN reap
415   r.resources = get_process_resources(pid);
416 
417   int wstatus = 0;
418   micron::wait4(pid, &wstatus, 0, &r.usage);      // blocking reap (EINTR-retried inside wait4)
419 
420   if ( wifexited(wstatus) ) {
421     r.exited_normally = true;
422     r.exit_code = wexitstatus(wstatus);
423   } else if ( wifsignaled(wstatus) ) {
424     r.signaled = true;
425     r.term_signal = wtermsig(wstatus);
426     r.exit_code = -r.term_signal;
427   }
428   return r;
429 }

645 rexecute(uprocess_t &t, const Caps &caps)
646 {
647   apply_caps_child(caps);
648   micron::svector<char *> argv;
649   if ( t.argv.empty() ) argv.push_back(&t.path[0]);      // t.argv already carries argv[0]; only seed if empty (was duplicated)

651   argv.push_back(nullptr);
