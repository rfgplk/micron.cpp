35 // fork and run a function in the child
36 // NOTE: flags=0 fork is copy-on-write: the kernel does NOT switch the stack pointer,
37 // so the child resumes in *this* frame on a COW copy of the parent stack
38 template<auto Fn = __default_callback, int Stack = default_stack_size, typename... Args>
39 int
40 __base_fork(Args &&...args)
41 {
42   int pid = micron::posix::__fork_clone(posix::sig_chld);
43   if ( pid < 0 ) exc<except::system_error>("micron process failed to fork()");
44   if ( pid == 0 ) {
45     // child: COW address space, same stack pointer -> forwarded args are live here
46     using ret_t = decltype(Fn(micron::forward<Args>(args)...));
47     if constexpr ( micron::is_same_v<ret_t, int> ) {
48       int rc = Fn(micron::forward<Args>(args)...);
49       micron::syscall(SYS_exit_group, rc);
50     } else {
51       Fn(micron::forward<Args>(args)...);
52       micron::syscall(SYS_exit_group, 0);
53     }
54     __builtin_unreachable();
55   }
56   return pid;      // parent
57 }

66 // implementation of ::fork() but automatically wait for child proc to end
67 template<int Stack = default_stack_size>
68 int
69 wfork()
70 {
71   int pid = __base_fork();
72   if ( pid == 0 ) return pid;
73   if ( pid < 1 ) {
74     return pid;
75   }
76   int status = 0;
77   micron::waitpid(pid, &status, 0);
78   if ( micron::wifexited(status) ) return micron::wexitstatus(status);
79   return status;
80 }
