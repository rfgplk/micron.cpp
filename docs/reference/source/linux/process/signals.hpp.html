175 template<typename Fn>
176   requires(micron::is_function_v<Fn> or micron::is_invocable_v<Fn, int>)
177 auto
178 create_handler(Fn &&fn, signal sig)
179 {
180   posix::sigaction_t sa = {};
181   sa.sigaction_handler.sa_handler = fn;
182   micron::posix::sigemptyset(sa.sa_mask);
183   sa.sa_flags = posix::sa_restart;
184   micron::posix::sigaction(static_cast<int>(sig), sa, nullptr);
185   return sa;

211 inline int
212 ignore(signal sig)
213 {
214   posix::sigaction_t sa = {};
215   sa.sigaction_handler.sa_handler = reinterpret_cast<posix::sighandler_t>(1);      // SIG_IGN == (void*)1

231 inline posix::sigaction_t
232 query_handler(signal sig)
233 {
234   posix::sigaction_t old = {};
235   posix::sigaction_t dummy = {};
236   dummy.sigaction_handler.sa_handler = reinterpret_cast<posix::sighandler_t>(1);
237   micron::posix::sigemptyset(dummy.sa_mask);
238   dummy.sa_flags = 0;
239   micron::posix::sigaction(static_cast<int>(sig), dummy, &old);
240   micron::posix::sigaction(static_cast<int>(sig), old, nullptr);
241   return old;
242 }

244 // similar to critical_section, blocks signal for lifetime duration
245 struct signal_mask_guard {
246   posix::sigset_t saved = {};
247   bool active = false;
248 
249   ~signal_mask_guard()
250   {
251     if ( active ) micron::posix::sigprocmask(posix::sig_setmask, saved, nullptr);
252   }
253 
254   signal_mask_guard() = default;
255 
256   explicit signal_mask_guard(const posix::sigset_t &mask) { block(mask); }
257 
258   template<typename... Sigs>
259     requires(... && micron::is_same_v<Sigs, signal>)
260   explicit signal_mask_guard(Sigs... sigs)
261   {
262     posix::sigset_t m = {};
263     micron::posix::sigemptyset(m);
264     (micron::posix::sigaddset(m, static_cast<int>(sigs)), ...);
265     block(m);
266   }
267 
268   // not copyable
269   signal_mask_guard(const signal_mask_guard &) = delete;
270   signal_mask_guard &operator=(const signal_mask_guard &) = delete;
271 
272   signal_mask_guard(signal_mask_guard &&o) noexcept : saved(o.saved), active(o.active) { o.active = false; }
273 
274   void
275   block(const posix::sigset_t &mask)
276   {
277     micron::posix::sigprocmask(posix::sig_block, mask, &saved);
278     active = true;
279   }
280 
281   void
282   release()
283   {
284     if ( active ) {
285       micron::posix::sigprocmask(posix::sig_setmask, saved, nullptr);
286       active = false;
287     }
288   }
289 };

295   micron::posix::sigemptyset(m);

400 // mark procmask on constructor if true
401 template<bool M = false> class signal_t
402 {
403   alignas(16) posix::sigset_t __signal;
404   micron::array<posix::sigaction_t, 65> __acts;
405   mutable int __sig;
406   mutable posix::sigset_t __saved_mask{};      // the process mask before this object blocked anything
407   mutable bool __mask_saved = false;
408 
409 public:
410   ~signal_t()
411   {
412     // restore ONLY the mask this object actually changed
413     if ( __mask_saved ) posix::sigprocmask(posix::sig_setmask, __saved_mask, nullptr);
414   }
415 
416   signal_t(void) : __acts(), __sig(0) { posix::sigemptyset(__signal); }
417 
418   template<typename... Sigs> signal_t(const Sigs... sig) : __acts{}
419   {
420     posix::sigemptyset(__signal);
421     (posix::sigaddset(__signal, (i32)sig), ...);
422     if constexpr ( M ) {
423       posix::sigprocmask(posix::sig_block, __signal, &__saved_mask);      // save old mask for the dtor to restore
424       __mask_saved = true;
425     }
426     __sig = 0;
427   }
428 
429   signal_t(const signal_t &) = default;
430   signal_t(signal_t &&) = default;
431   signal_t &operator=(const signal_t &) = default;
432   signal_t &operator=(signal_t &&) = default;
433 
434   // mask masks the signals, to be caugh via operator()
435   int
436   mask(void) const
437   {
438     if ( !__mask_saved ) {      // remember the pre-block mask the first time so the dtor can restore it
439       __mask_saved = true;
440       return posix::sigprocmask(posix::sig_block, __signal, &__saved_mask);
441     }
442     return posix::sigprocmask(posix::sig_block, __signal, nullptr);
443   }
444 
445   // on signal sets a universal function callback, when a signal is received call the function
446   int
447   on_signal(const micron::signal s, void (*fhandler)(int))
448   {
449     if ( (u64)s == 0 || (u64)s >= __acts.size() ) return -1;
450     __acts[(i32)s].sigaction_handler.sa_handler = fhandler;
451     return posix::sigaction((i32)s, __acts[(i32)s], nullptr);
452   }
453 
454   posix::sigset_t &
455   get_signal()
456   {
457     return __signal;
458   };
459 
460   int
461   wait(void) const
462   {
463     return posix::sigwait(__signal, __sig);
464   }
465 
466   int
467   operator()(void) const
468   {
469     return posix::sigwait(__signal, __sig);
470   }
471 };
