25 template<is_limits_set Lims>
26 inline void
27 child_apply_limits(const Lims &lims)
28 {
29   for ( rlim_t i = 0; i < posix::rlimit_nlimits; ++i ) {
30     posix::rlimit64_t rl = lims.lim[i];       // local copy (64-bit fields)
31     posix::set_process_limits(0, i, rl);      // prlimit64 on self (64-bit correct on every arch)
32   }
33 }

51 template<bool Lim = false, bool Cap = false, typename Lims = posix::limits_t, typename Caps = ucap_set_t>
52 inline int
53 __spawn_caps(pid_t &pid, const char *path, char *const *argv, char *const *envp, [[maybe_unused]] const Lims *lims = nullptr,
54              [[maybe_unused]] const Caps *caps = nullptr, const posix::spawn_file_actions_t *fa = nullptr,
55              const posix::spawnattr_t *attr = nullptr, const char *cwd = nullptr)
56 {
57   int pipefd[2];
58   int pr = micron::pipe2(pipefd, posix::o_cloexec);
59   if ( pr < 0 ) return micron::syscall_errno(pr);
60 
61   pid = micron::try_fork();
62   if ( pid < 0 ) {
63     micron::close(pipefd[0]);
64     micron::close(pipefd[1]);
65     return micron::syscall_errno(pid);
66   }
67   if ( pid == 0 ) {
68     micron::close(pipefd[0]);
69 
70     if ( cwd ) {
71       long cr = micron::posix::chdir(cwd);
72       if ( micron::syscall_failed(cr) ) {
73         int e = micron::syscall_errno(cr);
74         micron::write(pipefd[1], &e, sizeof(e));
75         micron::posix::exit(127);
76         __builtin_unreachable();
77       }
78     }
79 
80     // NOTE: child_apply_limits/apply_caps_child are void
81     if constexpr ( Lim ) {
82       if ( lims ) child_apply_limits(*lims);
83     }
84 
85     if constexpr ( Cap ) {
86       if ( caps ) apply_caps_child(*caps);
87     }
88 
89     posix::spawn_ctx ctx{ path, argv, envp, fa, attr, pipefd[1] };
90     micron::posix::spawn_process(ctx);
91     micron::posix::exit(6);
92     __builtin_unreachable();
93   }
94 
95   micron::close(pipefd[1]);
96   int err = 0;
97   max_t n = micron::read(pipefd[0], &err, sizeof(err));
98   micron::close(pipefd[0]);
99   if ( n == static_cast<max_t>(sizeof(err)) ) {
100     micron::wait4(pid, nullptr, 0, nullptr);
101     return err;
102   }
103   return 0;
104 }

144 static int
145 __spawn_child_thunk(void *p)
146 {
147   auto *a = static_cast<__spawn_clone_arg *>(p);
148   micron::close(a->read_fd);
149   micron::posix::spawn_process(&a->ctx);
150   return 127;

153 int
154 __spawn(pid_t &pid, const char *__restrict path, const posix::spawn_file_actions_t *__restrict file_actions,
155         const posix::spawnattr_t *__restrict attrp, char *const *argv, char *const *envp)
156 {
157   int pipefd[2];
158   int pr = micron::pipe2(pipefd, posix::o_cloexec);
159   if ( pr < 0 ) return micron::syscall_errno(pr);
160 
161   constexpr usize stack_size = 1 << 20;
162   void *stack
163       = reinterpret_cast<void *>(micron::mmap(nullptr, stack_size + 4096, prot_read | prot_write, map_private | map_anonymous, -1, 0));
164   if ( mmap_failed(stack) ) return micron::syscall_errno(reinterpret_cast<long>(stack));
165 
166   micron::mprotect(stack, 4096, prot_none);
167   constexpr u32 red_zone = 128;
168   uintptr_t sp = reinterpret_cast<uintptr_t>(stack) + stack_size + 4096 - red_zone;
169   sp &= ~uintptr_t(0xF);
170 
171   auto *arg = reinterpret_cast<__spawn_clone_arg *>(
172       micron::mmap(nullptr, sizeof(__spawn_clone_arg), prot_read | prot_write, map_private | map_anonymous, -1, 0));
173   if ( mmap_failed(arg) ) return micron::syscall_errno(reinterpret_cast<long>(arg));
174 
175   arg->ctx = { path, argv, envp, file_actions, attrp, pipefd[1] };
176   arg->read_fd = pipefd[0];
177 
178   // child now runs __spawn_child_thunk on the fresh stack via the asm trampoline
179   unsigned long f = micron::posix::__micron_fold_exit_signal(0, static_cast<int>(posix::sig_chld));
180   pid_t child
181       = static_cast<pid_t>(::__micron_clone_entry(&__spawn_child_thunk, reinterpret_cast<void *>(sp), f, arg, nullptr, nullptr, nullptr));
182 
183   micron::close(pipefd[1]);
184 
185   int err;
186   max_t n = micron::read(pipefd[0], &err, sizeof(err));
187   micron::close(pipefd[0]);
188 
189   if ( n == sizeof(err) ) {
190     micron::wait4(child, nullptr, 0, nullptr);
191     return err;
192   }
193 
194   if ( pid ) pid = child;
195 
196   return 0;
197 }
