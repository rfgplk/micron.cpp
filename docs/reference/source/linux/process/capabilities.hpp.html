62 constexpr u64 cap_all_mask = (u64(1) << static_cast<u32>(cap::__count)) - 1u;

64 constexpr u64
65 cap_bit(cap c) noexcept
66 {
67   return u64(1) << static_cast<u32>(c);

70 template<typename T>
71 concept is_cap_set = requires(const T &t) {
72   { t.effective } -> micron::convertible_to<u64>;
73   { t.permitted } -> micron::convertible_to<u64>;
74   { t.inheritable } -> micron::convertible_to<u64>;
75 };

249 // NOTE: bounding and ambient sets are only accessible via prctl for the calling thread
250 // for foreign pids use ucap_set_t::from_masks()
251 inline ucap_set_t
252 get_caps(pid_t pid = 0)
253 {
254   posix::cap_data_t data[2]{};
255   if ( posix::capget_pid(pid, data) < 0 ) return ucap_set_t{};
256 
257   ucap_set_t cs{};
258   posix::data_to_caps(data, cs.effective, cs.permitted, cs.inheritable);
259 
260   if ( pid == 0 || pid == posix::getpid() ) {
261     cs.bounding = posix::cap_bset_read_all();
262     cs.ambient = posix::cap_ambient_read_all();
263   } else {
264     cs.bounding = cap_all_mask;
265     cs.ambient = 0;
266   }
267   return cs;
268 }

270 inline ucap_set_t

276 inline bool
277 has_cap(cap c)
278 {
279   posix::cap_data_t data[2]{};
280   posix::capget_pid(0, data);
281   u64 eff = u64(data[0].effective) | (u64(data[1].effective) << 32);
282   return (eff & cap_bit(c)) != 0;
283 }

285 template<bool Effective = true, bool Permitted = false, bool Inheritable = false>
286 inline bool
287 has_cap(cap c)
288 {
289   posix::cap_data_t data[2]{};
290   posix::capget_pid(0, data);
291   u64 eff, prm, inh;
292   posix::data_to_caps(data, eff, prm, inh);
293   if constexpr ( Effective ) return (eff & cap_bit(c)) != 0;
294   if constexpr ( Permitted ) return (prm & cap_bit(c)) != 0;
295   if constexpr ( Inheritable ) return (inh & cap_bit(c)) != 0;
296   return false;
297 }

351 template<is_cap_set Caps>
352 inline int
353 apply_caps_child(const Caps &cs) noexcept
354 {
355   if ( const int r = posix::cap_set_keepcaps(1); r < 0 ) return r;
356 
357   u64 want_bnd;
358   if constexpr ( requires { cs.bounding; } )
359     want_bnd = static_cast<u64>(cs.bounding);
360   else
361     want_bnd = cap_all_mask;
362 
363   u64 drop_bnd = ~want_bnd & cap_all_mask;
364   int err = 0;
365   for ( u32 i = 0; i <= posix::cap_last_cap; ++i ) {
366     if ( !(drop_bnd & (u64(1) << i)) ) continue;
367     const int d = posix::cap_bset_drop(i);
368     if ( d < 0 && d != -error::invalid_arg && err == 0 ) err = d;
369   }
370   if ( err < 0 ) return err;
371 
372   posix::cap_data_t data[2]{};
373   posix::caps_to_data(cs.effective, cs.permitted, cs.inheritable, data);
374   int r = posix::capset_self(data);
375   if ( r < 0 ) return r;
376 
377   if constexpr ( requires { cs.ambient; } ) {
378     u64 amb = static_cast<u64>(cs.ambient) & static_cast<u64>(cs.inheritable);
379     for ( u32 i = 0; i <= posix::cap_last_cap; ++i ) {
380       if ( !(amb & (u64(1) << i)) ) continue;
381       const int a = posix::cap_ambient_raise(i);
382       if ( a < 0 && err == 0 ) err = a;
383     }
384   }
385 
386   return err;
387 }
