22 // NOTE: the errno convention is literal, not -errno, do NOT flip it
23 int
24 __spawn(pid_t &pid, const char *__restrict path, char *const *argv, char *const *envp)
25 {
26   int pipefd[2];
27   int pr = micron::pipe2(pipefd, posix::o_cloexec);
28   if ( pr < 0 ) return micron::syscall_errno(pr);      // positive errno
29   micron::posix::spawn_ctx ctx = { path, argv, envp, nullptr, nullptr, pipefd[1] };
30   // NOTE: try_fork, fork exc<>s out
31   pid = micron::try_fork();
32   if ( pid < 0 ) {
33     micron::close(pipefd[0]);
34     micron::close(pipefd[1]);
35     return micron::syscall_errno(pid);
36   }
37   if ( pid == 0 ) {
38     micron::close(pipefd[0]);
39     micron::posix::spawn_process(ctx);
40   }
41 
42   micron::close(pipefd[1]);
43 
44   int err;
45   max_t n = micron::read(pipefd[0], &err, sizeof(err));
46   micron::close(pipefd[0]);
47 
48   // the write only happens when execve failed
49   if ( n == sizeof(err) ) {
50     micron::wait4(pid, nullptr, 0, nullptr);
51     return err;
52   }
53 
54   return 0;
55 }

57 int
58 spawn(pid_t &pid, const char *__restrict path, char *const *argv, char *const *envp)
59 {
60   return __spawn(pid, path, argv, envp);
61 }
