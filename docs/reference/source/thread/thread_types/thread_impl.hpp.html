47 // give a thread's TLS frame back to whoever built it.
48 // WARNING: when attached, the frame came from the host's info->make_child_frame() and must go back
49 // through info->free_frame() -- that callback exists in the ABI precisely so the host can own frame
50 // lifetime (a recycling pool, landlord accounting). A guest that munmaps it itself hands back memory
51 // the host still believes it owns, and the next attached thread handed that frame faults on its
52 // first thread_local access. Every teardown path routes here, not just the spawn-failure one.
53 inline void
54 __tls_release_frame(const micron::__tls_frame &f) noexcept
55 {
56 #if defined(__micron_attach_capable)
57   if ( micron::__micron_attach_info != nullptr ) {
58     micron::__micron_attach_info->free_frame(&f);
59     return;
60   }
61 #endif
62   micron::__tls_free_frame(f);
63 }

82 struct __thread_payload {
83   atomic_token<bool> alive;
84   // the child sets this true the instant it reaches __thread_kernel
85   atomic_token<bool> started{ false };
86   // native park/sleep/awake/terminate control word
87   atomic_token<u32> park{ __park_run };
88   // NOTE:
89   // the pointer is used as the operand of a static_cast (7.6.1.8), except when the conversion is to pointer
90   // to cv void, or to pointer to cv void and subsequently to pointer to cv char, cv unsigned char, or
91   // cv std::byte (17.2.1)....
92   // If a program attempts to access (3.1) the stored value of an object through a glvalue whose type is not
93   // similar (7.3.5) to one of the following types the behavior is undefined:51
94   //(11.1) — the dynamic type of the object,
95   //(11.2) — a type that is the signed or unsigned type corresponding to the dynamic type of the object, or
96   //(11.3) — a char, unsigned char, or std::byte type.
97   atomic_ptr<byte *> ret_val;      // stop gap measure for returning arbitrary return types
98   enum class tag : u8 { none = 0, literal, heap, __end };
99   atomic_token<u8> tag_val{ (u8)tag::none };
100   void (*deleter)(byte *) = nullptr;
101 
102   posix::rusage_t usage;      // non atomic
103 
104   void
105   clear(void)
106   {
107     alive.store(false);
108     started.store(false, memory_order_release);
109     park.store(__park_run, memory_order_release);
110     tag_val.store((u8)tag::none, memory_order_release);
111     ret_val.store(nullptr, memory_order_release);
112     deleter = nullptr;
113     usage = posix::rusage_t{};
114   }
115 };

117 // the running thread kernel self-exit routine
118 [[noreturn]] inline void
119 __micron_thread_die_impl() noexcept
120 {
121   // run this thread's C++ thread_local dtors (guest modules) before the arena hook
122   micron::__run_thread_dtors();
123   if ( micron::__dynamic_tls_exit_hook ) micron::__dynamic_tls_exit_hook();
124   if ( micron::__thread_exit_hook ) micron::__thread_exit_hook();
125   if ( micron::__micron_thread_alive_word )
126     static_cast<atomic_token<bool> *>(micron::__micron_thread_alive_word)->store(false, memory_order_seq_cst);
127   micthread::__exit_thread(0);
128 }

138 // The kernel of the thread, encapsulating the req'd function
139 // status flags if the thread is running or not
140 // NOTE: args is always instantiated with decayed (value) types by __as_*_thread_attached
141 // taking args by value gives the kernel local storage; invoking fn(args...) passes lvalues
142 template<typename Fn, typename... Args>
143   requires(micron::is_invocable_v<Fn, Args &...>)
144 i32
145 __thread_kernel(__thread_payload *payload, Fn fn, Args... args)
146 {
147   // prologue
148   using ret_t = micron::invoke_result_t<Fn, Args &...>;
149   __thread_handler();
150   payload->alive.store(true, memory_order_seq_cst);
151   // confirm to the spawner that we actually reached our kernel
152   payload->started.store(true, memory_order_seq_cst);
153   // publish this thread's native control state: alive flag + park word + self-exit routine, consumed by
154   micron::__micron_thread_alive_word = static_cast<void *>(&payload->alive);
155   micron::__micron_thread_park = payload->park.ptr();
156   micron::__micron_thread_die = &__micron_thread_die_impl;
157 
158   // work
159 
160   if constexpr ( micron::is_void_v<ret_t> ) {
161     fn(args...);
162     // function has _no_ return value, signal that it returned successfully (1)
163     // NOTE: this may cause erronous behavior if 1 happens to be a valid pointer on your platform. be careful!
164     payload->deleter = nullptr;
165     payload->tag_val.store((u8)__thread_payload::tag::none, memory_order_release);
166     payload->ret_val.store(reinterpret_cast<byte *>(1), memory_order_release);
167   } else if constexpr ( micron::is_class_v<ret_t> and !micron::is_trivially_constructible_v<ret_t> and !micron::is_literal_type_v<ret_t> ) {
168     // NOTE: IMPORTANT
169     // make sure this is freed either by the calling thread, or whatever handles returns
170     ret_t *ret = new ret_t(fn(args...));
171     payload->deleter = +[](byte *p) { delete reinterpret_cast<ret_t *>(p); };
172     payload->tag_val.store((u8)__thread_payload::tag::heap, memory_order_release);
173     payload->ret_val.store(reinterpret_cast<byte *>(ret), memory_order_release);
174   } else if constexpr ( sizeof(ret_t) > sizeof(byte *) ) {
175     // bigger than a pointer (8B on 64-bit, 4B on 32-bit): cannot be packed into ret_val, use new
176     // NOTE: the bound is sizeof(byte*), NOT a hard-coded 8
177     ret_t *ret = new ret_t(fn(args...));
178     payload->deleter = +[](byte *p) { delete reinterpret_cast<ret_t *>(p); };
179     payload->tag_val.store((u8)__thread_payload::tag::heap, memory_order_release);
180     payload->ret_val.store(reinterpret_cast<byte *>(ret), memory_order_release);
181   } else {
182     // type is literal, can be stored.
183     ret_t __ret_val_storage = fn(args...);
184     u64 __ret_raw = 0;
185     micron::cbytecpy<sizeof(ret_t)>(reinterpret_cast<byte *>(&__ret_raw), reinterpret_cast<const byte *>(&__ret_val_storage));
186     payload->deleter = nullptr;
187     payload->tag_val.store((u8)__thread_payload::tag::literal, memory_order_release);
188     payload->ret_val.store(reinterpret_cast<byte *>(__ret_raw), memory_order_release);
189   }
190   // end work
191 
192   // epilogue
193   // NOTE: run any registered per-thread cleanup (abcmalloc releasing this thread's arena slot) on the exiting thread while its TLS is still
194   // valid. thread_local C++ dtors (guest modules) run first, while the arena is still live
195   micron::__run_thread_dtors();
196 #if defined(__micron_freestanding)
197   if ( micron::__dynamic_tls_exit_hook ) micron::__dynamic_tls_exit_hook();
198 #endif
199   if ( micron::__thread_exit_hook ) micron::__thread_exit_hook();
200   posix::getrusage(posix::rusage_thread, payload->usage);
201   payload->alive.store(false, memory_order_seq_cst);
202   return return_success;
203 }

205 // worker kernel, for concurrent arenas and parallelism
206 // main difference compared to a regular kernel is that this function NEVER returns, instead constantly loops checking
207 // for new work

209 inline i32
210 __worker_loop(__worker_payload *payload)
211 {
212   for ( ;; ) {
213     while ( payload->queue.empty() ) {
214       if ( payload->should_die.get(memory_order_acquire) ) return return_force;
215       // WARNING: consume a pending wake token via CAS 1->0 instead of an unconditional store(0)
216       // an unconditional clear races the producer/stop() path: if our store(0) lands __AFTER__
217       // the producer's store(1) and the wake fired before we reach wait_futex, the wakeup is
218       // lost and we sleep on 0 forever
219       u32 __hw_token = 1;
220       payload->has_work.compare_exchange_strong(__hw_token, 0u, memory_order_acq_rel, memory_order_acquire);
221       if ( !payload->queue.empty() ) break;      // a producer raced the clear; don't sleep
222       if ( payload->should_die.get(memory_order_acquire) ) return return_force;
223       wait_futex(payload->has_work.ptr(), 0);
224     }
225     if ( payload->should_die.get(memory_order_acquire) ) return return_force;
226     while ( !payload->queue.empty() ) {
227       payload->queue.execute();
228       if ( payload->should_die.get(memory_order_acquire) ) return return_force;
229     }
230     // epilogue (seqlock-protected usage snapshot)
231     payload->usage_seq.add_fetch(1, memory_order_acq_rel);
232     posix::getrusage(posix::rusage_thread, payload->usage);
233     payload->usage_seq.add_fetch(1, memory_order_acq_rel);
234   }
235   return return_success;
236 }

238 i32
239 __worker_kernel(__worker_payload *payload)
240 {
241   // prologue
242   __thread_handler();
243   const i32 rc = __worker_loop(payload);
244   // epilogue
245   // NOTE: a worker exits through here, not through __thread_kernel, so it needs the same teardown:
246   // thread_local C++ dtors (guest modules) then the arena hook, while this thread's TLS is still
247   // mapped. Without it a worker's thread_local RAII objects (a uring ring, a buffered file, a socket)
248   // are simply munmap'd with the frame and their fds leak for the life of the process
249   micron::__run_thread_dtors();
250 #if defined(__micron_freestanding)
251   if ( micron::__dynamic_tls_exit_hook ) micron::__dynamic_tls_exit_hook();
252 #endif
253   if ( micron::__thread_exit_hook ) micron::__thread_exit_hook();
254   return rc;
255 }

258 #if !defined(__micron_freestanding)
259 template<auto KFn, typename... Args> struct __pth_box {
260   micron::tuple<Args...> args;
261 
262   template<typename... U>
263     requires(sizeof...(U) == sizeof...(Args))
264   explicit __pth_box(U &&...values) : args(micron::forward<U>(values)...)
265   {
266   }
267 
268   static void *
269   trampoline(void *p)
270   {
271     auto *self = static_cast<__pth_box *>(p);
272     __invoke(self, micron::make_index_sequence<sizeof...(Args)>{});
273     delete self;
274     return nullptr;
275   }
276 
277   static void
278   destroy(__pth_box *self)
279   {
280     delete self;
281   }
282 
283 private:
284   template<usize... I>
285   static void
286   __invoke(__pth_box *self, micron::index_sequence<I...>)
287   {
288     KFn(static_cast<Args &&>(micron::get<I>(self->args))...);
289   }
290 };

322 template<int Stack_Size, auto KFn, typename P, typename... KArgs>
323 void
324 __link_launch([[maybe_unused]] pid_t &tid, [[maybe_unused]] int &ctid, [[maybe_unused]] micron::__tls_frame &tls,
325               [[maybe_unused]] unsigned long &pth, P *payload, addr_t *fstack, KArgs &&...kargs)
326 {
327   if ( payload == nullptr || fstack == nullptr ) micron::exc<except::thread_error>("micron thread::__link_launch(): invalid arguments");
328 #if defined(__micron_freestanding)
329   micron::__tls_frame frame{};
330 #if defined(__micron_attach_capable)
331   // a guest (attached) thread's frame must be built by the host
332   if ( micron::__micron_attach_info != nullptr ) {
333     if ( micron::__micron_attach_info->make_child_frame(&frame) != 0 || frame.base == nullptr )
334       micron::exc<except::thread_error>("micron thread::__link_launch(): host failed to build attached TLS");
335   } else
336 #endif
337   {
338     frame = micron::__tls_make_child_frame();
339     if ( frame.base == nullptr ) micron::exc<except::thread_error>("micron thread::__link_launch(): failed to build thread TLS");
340   }
341   auto reg = micthread::guard_stack(fstack, static_cast<usize>(Stack_Size));
342   pid_t t = micthread::spawn<KFn>(reg.low, reg.usable, reinterpret_cast<unsigned long>(frame.tp), &ctid, payload,
343                                   micron::forward<KArgs>(kargs)...);
344   if ( t == micthread::thread_failed ) {
345     micron::__tls_release_frame(frame);
346     micron::exc<except::thread_error>("micron thread::__link_launch(): thread failed to spawn");
347   }
348   tls = frame;
349   tid = t;
350 #else
351   using box_t = __pth_box<KFn, P *, micron::decay_t<KArgs>...>;
352   auto *box = new box_t(payload, static_cast<KArgs &&>(kargs)...);
353   pthread_attr_t attr;
354   int err = pthread_attr_init(&attr);
355   if ( err != 0 ) {
356     box_t::destroy(box);
357     micron::exc<except::thread_error>("micron thread::__link_launch(): failed to initialize pthread attributes");
358   }
359   auto reg = micthread::guard_stack(fstack, static_cast<usize>(Stack_Size));
360   err = pthread_attr_setstack(&attr, reg.low, reg.usable);
361   if ( err != 0 ) {
362     pthread_attr_destroy(&attr);
363     if ( reg.low != reinterpret_cast<addr_t *>(fstack) )
364       micron::mprotect(reinterpret_cast<addr_t *>(fstack), __micron_page_size_default, micron::prot_read | micron::prot_write);
365     box_t::destroy(box);
366     micron::exc<except::thread_error>("micron thread::__link_launch(): invalid pthread stack");
367   }
368   pthread_t th{};
369   err = pthread_create(&th, &attr, &box_t::trampoline, box);
370   pthread_attr_destroy(&attr);
371   if ( err != 0 ) {
372     if ( reg.low != reinterpret_cast<addr_t *>(fstack) )
373       micron::mprotect(reinterpret_cast<addr_t *>(fstack), __micron_page_size_default, micron::prot_read | micron::prot_write);
374     box_t::destroy(box);
375     micron::exc<except::thread_error>("micron thread::__link_launch(): thread failed to spawn");
376   }
377   pth = static_cast<unsigned long>(th);
378 #endif
379 }

381 inline int
382 __link_join([[maybe_unused]] pid_t tid, [[maybe_unused]] int *ctid, [[maybe_unused]] unsigned long pth)
383 {
384   if ( !__link_valid(tid, pth) ) return 0;
385 #if defined(__micron_freestanding)
386   micthread::join(ctid);
387 #else
388   pthread_join(static_cast<pthread_t>(pth), nullptr);
389 #endif
390   return 0;

393 // non-blocking: 0 if exited/none, error::busy if still running
394 inline int
395 __link_try_join([[maybe_unused]] pid_t tid, [[maybe_unused]] int *ctid, [[maybe_unused]] unsigned long pth)
396 {
397   if ( !__link_valid(tid, pth) ) return 0;
398 #if defined(__micron_freestanding)
399   return micthread::try_join(ctid) == 0 ? 0 : static_cast<int>(error::busy);
400 #else
401   return pthread_tryjoin_np(static_cast<pthread_t>(pth), nullptr) == 0 ? 0 : static_cast<int>(error::busy);
402 #endif
