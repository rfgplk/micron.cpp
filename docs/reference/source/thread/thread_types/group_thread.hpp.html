40 // bounded start handshake budget
41 // regular time for a new thread to reach its kernel worker prologue is ~microseconds;
42 // this is a generous ceiling above which we declare it faulted and hard-error rather
43 inline constexpr f64 __thread_start_timeout_ms = 1000.0;
44 
45 // group_thread, a regular group_thread
46 template<usize Stack_Size = thread_stack_size> class group_thread
47 {
48   using thread_type = thread_tag;

50   void
51   thread_handler()
52   {
53     micron::create_handler(__thread_sigchld, signal::child);
54   }

56   template<typename F, typename... Args>
57     requires(micron::is_invocable_v<micron::decay_t<F>, micron::decay_t<Args> &...>)
58   inline __attribute__((always_inline)) void
59   __impl_preparethread(F f, Args &&...args)
60   {
61     if ( attributes.stack_addr == nullptr ) {
62       micron::exc<except::thread_error>("micron thread::__impl_preparethread(): a stack isn't allocated");
63     }
64     thread_handler();
65     payload.alive.store(true, memory_order_seq_cst);
66     payload.started.store(false, memory_order_seq_cst);
67     __link_launch<Stack_Size, &__thread_kernel<micron::decay_t<F>, micron::decay_t<Args>...>>(
68         attributes.pid, attributes.ctid, attributes.tls, attributes.pth, &payload, attributes.stack_addr, f,
69         micron::forward<Args>(args)...);
70 
71     // bounded start handshake
72     if ( !micron::until_timeout(true, __thread_start_timeout_ms, [this]() noexcept {
73            const bool s = payload.started.get(memory_order_acquire);
74            if ( !s ) micron::yield();
75            return s;
76          }) ) {
77       // WARNING: child was created (clone3 succeeded) but never published in time: scheduler-starved
78       // or wedged; NEVER throw and unwind out from under it, the live child WILL write &payload
79       // and the kernel still clears &attributes.ctid, so reclaiming this object's
80       // storage would be a use-after-free / cross-object corruption
81       payload.park.store(__park_dying, memory_order_release);
82       micron::wake_futex(payload.park.ptr(), 1);
83       if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
84       __release();
85       micron::exc<except::thread_error>(
86           "micron thread::__impl_preparethread(): spawned thread never reached its kernel (faulted on entry?)");
87     }
88   }
89 
90   void
91   __release(void) noexcept
92   {
93     if ( attributes.stack_addr != nullptr ) {
94       micron::try_unmap(attributes.stack_addr, Stack_Size);
95     }
96     micron::__tls_release_frame(attributes.tls);      // free this thread's from-scratch TLS block
97     attributes.clear();
98     payload.alive.store(false, memory_order_seq_cst);
99     if ( static_cast<__thread_payload::tag>(payload.tag_val.get(memory_order_acquire)) == __thread_payload::tag::heap ) {
100       byte *ptr = payload.ret_val.exchange(nullptr, memory_order_acq_rel);
101       auto del = payload.deleter;
102       payload.deleter = nullptr;
103       if ( ptr && del ) del(ptr);
104     }
105     payload.tag_val.store((u8)__thread_payload::tag::none, memory_order_release);
106   }
107 
108   void
109   __safe_release(void)
110   {
111     if ( alive() ) {
112       micron::exc<except::thread_error>("micron thread::__safe_release(): tried to release a running thread");
113     }
114     __release();
115   }

140   // prepared functions
141   template<typename Fn, typename... Args>
142     requires(micron::is_invocable_v<micron::decay_t<Fn>, micron::decay_t<Args> &...>)
143   group_thread(const thread_attr_t &_attrs, Fn &&fn, Args &&...args) : attributes(posix::getpid()), payload{}
144   {
145     // adopt the caller-allocated stack + scheduling intent (group owns the stack now)
146     attributes.stack_addr = _attrs.stack_addr;
147     attributes.stack_size = _attrs.stack_size;
148     attributes.sched_policy = _attrs.sched_policy;
149     attributes.sched_priority = _attrs.sched_priority;
150     __impl_preparethread(micron::forward<Fn>(fn), micron::forward<Args>(args)...);
151   }
152 
153   group_thread &
154   operator=(group_thread &&o)
155   {
156     if ( this == &o ) return *this;
157     micron::__thread_assert_movable_src(o.attributes.pid, o.attributes.pth);      // o must be pre-spawn/joined
158     // release *this's current thread+stack BEFORE adopting o; otherwise the running thread is orphaned
159     if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
160     __release();
161     attributes = micron::move(o.attributes);      // thread_attr_t move nulls o.pid / o.stack_addr
162     payload = micron::move(o.payload);
163     return *this;
164   }

188   auto
189   can_join(void) -> int
190   {
191     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;
192     int r = __link_try_join(attributes.pid, &attributes.ctid, attributes.pth);
193     if ( r == 0 ) {
194       return 1;
195     }
196     if ( r == error::busy or r == error::invalid_arg ) return r;
197     return 0;
198   }

200   auto
201   join(void) -> int      // group_thread
202   {
203     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;      // nothing to join (default/moved-from/already-joined)
204     auto r = __link_join(attributes.pid, &attributes.ctid, attributes.pth);
205     __safe_release();
206     return r;
207   }
208 
209   auto
210   dismiss(void) -> int      // group_thread
211   {
212     int r = try_join();
213     if ( r == error::busy ) {
214       auto rr = __link_join(attributes.pid, &attributes.ctid, attributes.pth);
215       __release();
216       return rr;
217     }
218     return r;
219   }
220 
221   auto
222   try_join(void) -> int
223   {
224     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;      // already joined
225     int r = __link_try_join(attributes.pid, &attributes.ctid, attributes.pth);
226     if ( r == 0 ) {
227       __release();
228       return 0;
229     }
230     return r;
231   }

246   signal(const micron::signal s)

248     if ( alive() ) {

261   // native per-thread suspend: PARK at the next checkpoint; resume with awaken(). no signals.
262   int
263   sleep(void)
264   {
265     if ( !alive() ) return -1;
266     payload.park.store(__park_parked, memory_order_release);
267     return 0;
268   }
269 
270   int
271   awaken(void)
272   {
273     if ( !alive() ) return -1;
274     payload.park.store(__park_run, memory_order_release);
275     micron::wake_futex(payload.park.ptr(), 1);
276     return 0;
277   }
278 
279   // cooperative stop-and-reap (kill): DYING at the next checkpoint
280   int
281   cancel(void)
282   {
283     if ( !alive() ) return -1;
284     payload.park.store(__park_dying, memory_order_release);
285     micron::wake_futex(payload.park.ptr(), 1);
286     if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
287     payload.alive.store(false, memory_order_seq_cst);
288     __safe_release();
289     return 0;
290   }
291 
292   // hard stop
293   int
294   terminate(void)
295   {
296     if ( !alive() ) return -1;
297     payload.park.store(__park_dying, memory_order_release);
298     micron::wake_futex(payload.park.ptr(), 1);
299     if ( micron::until_timeout(__thread_term_timeout_ms, [this]() noexcept { return !alive(); }) ) {
300       if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
301       __release();
302     } else {
303       micthread::thread_kill(attributes.parent, __link_tid(attributes.pid, attributes.pth), (int)signal::kill9);
304       __link_join(attributes.pid, &attributes.ctid, attributes.pth);
305       __release();
306     }
307     return 0;
308   }

310   void
311   wait_for(void) const
312   {
313     until(false, &group_thread<Stack_Size>::alive, this);
314   }

316   template<typename R>
317     requires(micron::is_trivially_copyable_v<R> && sizeof(R) <= sizeof(byte *))
318   R
319   result(void)
320   {
321     wait_for();
322     R val{};
323     if ( static_cast<__thread_payload::tag>(payload.tag_val.get(memory_order_acquire)) == __thread_payload::tag::literal ) {
324       byte *p = payload.ret_val.get(memory_order_acquire);
325       u64 raw = reinterpret_cast<u64>(p);
326       micron::cbytecpy<sizeof(R)>(reinterpret_cast<byte *>(&val), reinterpret_cast<const byte *>(&raw));
327     }
328     return val;
329   }
