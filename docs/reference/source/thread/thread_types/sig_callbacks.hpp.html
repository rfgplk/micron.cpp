35 // true per-thread suspend/resume
36 inline void
37 __thread_suspend_handler(int)
38 {
39   posix::sigset_t wait_mask = {};
40   micron::posix::sigfillset(wait_mask);
41   micron::posix::sigdelset(wait_mask, (int)signal::thread_resume);
42   micron::posix::sigdelset(wait_mask, (int)signal::terminate);
43   micron::posix::sigdelset(wait_mask, (int)signal::kill9);
44   micron::posix::sigdelset(wait_mask, (int)signal::segfault);
45   micron::posix::sigdelset(wait_mask, (int)signal::abort);
46   // sigsuspend is also a cancellation point
47   micron::posix::sigsuspend(wait_mask);
48 }

56 inline void
57 __thread_handler()
58 {
59   static atomic_token<bool> __installed{ false };
60   bool expected = false;
61   if ( !__installed.compare_exchange_strong(expected, true, memory_order_acq_rel, memory_order_acquire) ) return;
62   // SIGALRM -> cooperative yield: the only app-facing signal the core still installs
63   micron::create_handler(__thread_yield, signal::alarm);
64   // WARNING: NEVER install a handler for signal::terminate (SIGTERM)
65   // per thread termination is now controlled by the dying futex word
66   // NOTE: install with thread_resume blocked in its own mask so a resume racing the tiny window before sigsuspend stays pending (blocked)
67   // and is delivered the instant sigsuspend unblocks
68   {
69     micron::posix::sigaction_t ssa = {};
70     ssa.sigaction_handler.sa_handler = __thread_suspend_handler;
71     micron::posix::sigemptyset(ssa.sa_mask);
72     micron::posix::sigaddset(ssa.sa_mask, (int)signal::thread_resume);
73     ssa.sa_flags = micron::posix::sa_restart;
74     micron::posix::sigaction((int)signal::thread_suspend, ssa, nullptr);
75   }
76   micron::create_handler(__thread_resume_handler, signal::thread_resume);
77 }
