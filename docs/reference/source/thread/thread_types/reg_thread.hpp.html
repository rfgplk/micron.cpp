40 // thread, a regular thread
41 template<usize Stack_Size = thread_stack_size> class thread
42 {
43   using thread_type = thread_tag;
44 
45   void
46   thread_handler()
47   {
48     micron::create_handler(__thread_sigchld, signal::child);
49   }
50 
51   template<typename F, typename... Args>
52     requires(micron::is_invocable_v<micron::decay_t<F>, micron::decay_t<Args> &...>)
53   inline __attribute__((always_inline)) void
54   __impl_makethread(F f, Args &&...args)
55   {
56     if ( attributes.stack_addr ) {
57       micron::exc<except::thread_error>("micron thread::__impl_makethread(): a stack is already allocated");
58     }
59     attributes.stack_addr = micron::addrmap(Stack_Size);
60     if ( micron::mmap_failed(attributes.stack_addr) )
61       micron::exc<except::thread_error>("micron thread::__impl_makethread(): failed to allocate stack");
62 
63     thread_handler();
64     payload.alive.store(true, memory_order_seq_cst);
65     __link_launch<Stack_Size, &__thread_kernel<micron::decay_t<F>, micron::decay_t<Args>...>>(
66         attributes.pid, attributes.ctid, attributes.tls, attributes.pth, &payload, attributes.stack_addr, f,
67         micron::forward<Args>(args)...);
68   }
69 
70   void
71   __release(void) noexcept
72   {
73     if ( attributes.stack_addr != nullptr ) {
74       micron::try_unmap(attributes.stack_addr, Stack_Size);
75     }
76     micron::__tls_release_frame(attributes.tls);      // free this thread's from-scratch TLS block
77     attributes.clear();
78     payload.alive.store(false, memory_order_seq_cst);
79     if ( static_cast<__thread_payload::tag>(payload.tag_val.get(memory_order_acquire)) == __thread_payload::tag::heap ) {
80       byte *ptr = payload.ret_val.exchange(nullptr, memory_order_acq_rel);
81       auto del = payload.deleter;
82       payload.deleter = nullptr;
83       if ( ptr && del ) del(ptr);
84     }
85     payload.tag_val.store((u8)__thread_payload::tag::none, memory_order_release);
86   }
87 
88   void
89   __safe_release(void)
90   {
91     if ( alive() ) {
92       micron::exc<except::thread_error>("micron thread::__safe_release(): tried to release a running thread");
93     }
94     __release();
95   }

124   template<typename Fn, typename... Args>
125     requires(micron::is_invocable_v<micron::decay_t<Fn>, micron::decay_t<Args> &...>)
126   thread(Fn &&fn, Args &&...args) : attributes(posix::getpid()), payload{}
127   {
128     __impl_makethread(micron::forward<Fn>(fn), micron::forward<Args>(args)...);
129   }
130 
131   thread &
132   operator=(thread &&o)
133   {
134     if ( this == &o ) return *this;
135     micron::__thread_assert_movable_src(o.attributes.pid, o.attributes.pth);      // o must be pre-spawn/joined
136     // release *this's current thread+stack BEFORE adopting o; otherwise the running thread is orphaned
137     // (never joined)
138     if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
139     __release();
140     attributes = micron::move(o.attributes);      // thread_attr_t move nulls o.pid / o.stack_addr
141     payload = micron::move(o.payload);
142     return *this;
143   }

145   template<typename F, typename... Args>
146     requires(micron::is_invocable_v<micron::decay_t<F>, micron::decay_t<Args> &...>)
147   thread &
148   operator[](F f, Args &&...args)
149   {
150     join();
151     __impl_makethread(f, micron::forward<Args>(args)...);
152     return *this;

170   inline bool
171   active(void) const
172   {
173     // more reliable, although slower: probe with signal 0 (tgkill)
174     return (micthread::thread_kill(attributes.parent, __link_tid(attributes.pid, attributes.pth), 0) == 0 ? true : false);
175   }

177   auto
178   can_join(void) -> int
179   {
180     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;
181     return __link_try_join(attributes.pid, &attributes.ctid, attributes.pth) == 0 ? 1 : (int)error::busy;
182   }
183 
184   auto
185   join(void) -> int      // thread
186   {
187     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;      // nothing to join (default/moved-from/already-joined)
188     __link_join(attributes.pid, &attributes.ctid, attributes.pth);
189     __safe_release();
190     return 0;
191   }
192 
193   auto
194   dismiss(void) -> int      // thread
195   {
196     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;
197     __link_join(attributes.pid, &attributes.ctid, attributes.pth);
198     __release();
199     return 0;
200   }
201 
202   auto
203   try_join(void) -> int
204   {
205     if ( !__link_valid(attributes.pid, attributes.pth) ) return 0;
206     if ( __link_try_join(attributes.pid, &attributes.ctid, attributes.pth) == 0 ) {
207       __release();
208       return 0;
209     }
210     return error::busy;
211   }

225   long int
226   signal(const signal s)
227   {
228     if ( alive() ) {
229       return micthread::thread_kill(attributes.parent, __link_tid(attributes.pid, attributes.pth), (int)s);

241   // native per-thread suspend: PARK at the next checkpoint; resume with awaken(). no signals.
242   int
243   sleep(void)
244   {
245     if ( !alive() ) return -1;
246     payload.park.store(__park_parked, memory_order_release);
247     return 0;
248   }
249 
250   int
251   awaken(void)
252   {
253     if ( !alive() ) return -1;
254     payload.park.store(__park_run, memory_order_release);
255     micron::wake_futex(payload.park.ptr(), 1);
256     return 0;
257   }
258 
259   // cooperative stop-and-reap (kill): DYING at the next checkpoint
260   int
261   cancel(void)
262   {
263     if ( !alive() ) return -1;
264     payload.park.store(__park_dying, memory_order_release);
265     micron::wake_futex(payload.park.ptr(), 1);
266     if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
267     payload.alive.store(false, memory_order_seq_cst);
268     __safe_release();
269     return 0;
270   }
271 
272   // hard stop
273   int
274   terminate(void)
275   {
276     if ( !alive() ) return -1;
277     payload.park.store(__park_dying, memory_order_release);
278     micron::wake_futex(payload.park.ptr(), 1);
279     if ( micron::until_timeout(__thread_term_timeout_ms, [this]() noexcept { return !alive(); }) ) {
280       if ( __link_valid(attributes.pid, attributes.pth) ) __link_join(attributes.pid, &attributes.ctid, attributes.pth);
281       __release();
282     } else {
283       micthread::thread_kill(attributes.parent, __link_tid(attributes.pid, attributes.pth), (int)signal::kill9);
284       __link_join(attributes.pid, &attributes.ctid, attributes.pth);
285       __release();
286     }
287     return 0;
288   }

290   void
291   wait_for(void) const
292   {
293     until(false, &thread<Stack_Size>::alive, this);
294   }

296   template<typename R>
297     requires(micron::is_trivially_copyable_v<R> && sizeof(R) <= sizeof(byte *))
298   R
299   result(void)
300   {
301     wait_for();
302     R val{};
303     if ( static_cast<__thread_payload::tag>(payload.tag_val.get(memory_order_acquire)) == __thread_payload::tag::literal ) {
304       byte *p = payload.ret_val.get(memory_order_acquire);
305       u64 raw = reinterpret_cast<u64>(p);
306       micron::cbytecpy<sizeof(R)>(reinterpret_cast<byte *>(&val), reinterpret_cast<const byte *>(&raw));
307     }
308     return val;
309   }
