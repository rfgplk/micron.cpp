25 template<class E>
26 concept executor = requires(E &__e, __frame_base *__fb) {
27   { __e.submit(__fb) } -> micron::same_as<void>;
28   { __e.worker_count() } -> micron::convertible_to<u32>;
29 };
30 
31 struct work_stealing_executor {
32   using executor_tag = void;
33   engine *__e = nullptr;
34 
35   work_stealing_executor() noexcept = default;
36 
37   explicit work_stealing_executor(engine *__eng) noexcept : __e(__eng) { }
38 
39   void
40   submit(__frame_base *__fb) noexcept
41   {
42     if ( __e == nullptr ) [[unlikely]]
43       __builtin_trap();
44     __e->submit(__fb);
45   }
46 
47   u32
48   worker_count() const noexcept
49   {
50     return __e != nullptr ? __e->n : 0u;
51   }
52 };
53 
54 struct any_executor {
55   void *__self = nullptr;
56   void (*__submit_fn)(void *, __frame_base *) = nullptr;
57   u32 (*__wc_fn)(void *) = nullptr;
58 
59   any_executor() noexcept = default;
60 
61   template<executor E>
62   static any_executor
63   make(E &__ex) noexcept
64   {
65     any_executor __a;
66     __a.__self = &__ex;
67     __a.__submit_fn = [](void *__s, __frame_base *__fb) { static_cast<E *>(__s)->submit(__fb); };
68     __a.__wc_fn = [](void *__s) { return static_cast<E *>(__s)->worker_count(); };
69     return __a;
70   }
71 
72   void
73   submit(__frame_base *__fb) noexcept
74   {
75     if ( __submit_fn == nullptr ) [[unlikely]]
76       __builtin_trap();
77     __submit_fn(__self, __fb);
78   }
79 
80   u32
81   worker_count() const noexcept
82   {
83     return __wc_fn != nullptr ? __wc_fn(__self) : 0u;
84   }
85 };
86 
87 inline work_stealing_executor
88 default_executor() noexcept
89 {
90   start_coroutine_runtime();
91   return work_stealing_executor{ __global_engine };
92 }
