31 using standard_arena = arena::__default_arena<micron::thread_stack_size, micron::group_thread<>>;
32 using concurrent_arena = arena::__concurrent_arena<>;
33 
34 inline standard_arena *__global_threadpool = nullptr;
35 inline concurrent_arena *__global_parallelpool = nullptr;
36 
37 // hosted init via .init_array
38 gconstructor_ void
39 __make_threadarena(void)
40 {
41   // WARNING: intentionally never destroyed, prevents hangs in glibcs/freestanding via registering atexits
42   alignas(standard_arena) static unsigned char __pool_storage[sizeof(standard_arena)];
43   static standard_arena *local_pool = new (static_cast<void *>(__pool_storage)) standard_arena();
44   __global_threadpool = local_pool;
45 };
46 
47 // freestanding init via start
48 extern "C" void
49 __boot_threadpool(void)
50 {
51   __make_threadarena();
52 }
53 
54 gdestructor_ void __destroy_threadarena(void) {
55   // WARNING: do NOT null the pointer here; a still-running worker may dereference __global_threadpool during process
56   // teardown; nulling it concurrently is a use-after-free / null-deref race
57 };

59 #if defined(__micron_enable_concurrency_at_startup_var)
60 __attribute__((constructor)) void
61 #else
62 inline __attribute__((always_inline)) void
63 #endif
64 __make_parallelarena(void)
65 {
66   // WARNING: intentionally never destroyed, prevents hangs in glibcs/freestanding
67   alignas(concurrent_arena) static unsigned char __ppool_storage[sizeof(concurrent_arena)];
68   static concurrent_arena *local_pool = new (static_cast<void *>(__ppool_storage)) concurrent_arena();
69   static bool __inited = false;
70   __global_parallelpool = local_pool;
71   if ( __inited ) return;      // create() the workers exactly once, even if start_concurrent_pools() is called again
72   __inited = true;
73   // init here
74   umax_t c = cpu_count();
75   if ( c > concurrent_threads ) c = concurrent_threads;      // never create more workers than the fixed arena has slots
76   for ( umax_t i = 0; i < c; ++i ) local_pool->create();
77 };

85   // see __destroy_threadarena: do not null while workers may still read it (race)

88 void
89 start_concurrent_pools(void)
90 {
91   // hosted already started worker threads
92   if constexpr ( !micron::config::concurrency_at_startup ) __make_parallelarena();
93 }

95 void
96 end_concurrent_pools(void)
