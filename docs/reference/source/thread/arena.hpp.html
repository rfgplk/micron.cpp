35 // NOTE: maximum of 1024 threads, this is a conservative limit, but there's almost no legitimate instance where you would
36 // want to be running more threads than this due to a) few machines have this many cores, b) see a
37 // this is solely meant as a memory saving measure, if you need more threads increase the limit or define one of the
38 // macros below
39 #if defined(ABSURD_THREAD_COUNT)
40 constexpr static const u32 maximum_threads = (1 << 15);      // 32768
41 #elif defined(MICRON_HIGH_THREAD_COUNT)
42 constexpr static const u32 maximum_threads = (1 << 12);      // 4096

45 #else
46 constexpr static const u32 maximum_threads = (1 << 8);      // 256

215   // Create a new thread at a specific core that is currently more free than the rest
216   template<typename Func, typename... Args>
217     requires(micron::is_invocable_v<Func, Args...>)
218   auto &
219   create_burden(Func f, Args &&...args)
220   {
221     micron::lock_guard l(mtx);
222     if ( threads.full_or_overflowed() ) [[unlikely]]
223       micron::exc<except::thread_error>("micron arena::create(): Arena is full or corrupted");
224     fvector<i32> cores(cpu_count());
225 
226     cpu_t<true> c;
227     for ( u64 i = 0; i < threads.size(); ++i ) {
228       c.clear();
229       posix::sched_getaffinity(threads[i].thread.thread_id(), sizeof(c.get()), c.get());
230       for ( u64 k = 0; k < cores.size(); ++k )
231         if ( c[k] ) cores[k]++;
232     }
233 
234     fvector<i32>::const_iterator smallest = min_at(cores);
235     if ( smallest != cores.cbegin() ) {
236       c.clear();
237       c.set_core(smallest - cores.cbegin());
238       goto start_thread;
239     }
240     c.clear();
241     c.set_core(0);
242   start_thread:
243     addr_t *stack_ptr = __create_stack();
244     thread_attr_t attrs = micron::__thread_attr_with_stack(posix::getpid(), posix::sched_other, stack_ptr, Sz);
245     threads.emplace(c, micron::move(attrs), f, micron::forward<Args>(args)...);
246     sstring<16> thread_name = "arena/" + int_to_string_stack<u32, char, 12>(static_cast<u32>(threads.size()));
247     thread_name[15] = 0x0;
248     micthread::set_name(threads.top().thread.native_handle(), thread_name.c_str());
249     posix::sched_setaffinity(threads.top().thread.thread_id(), sizeof(c.get()), c.get());      // best-effort
250     return threads.top();
251   }

295   // cleanup functions
296   void
297   clean(void)
298   {
299     micron::lock_guard l(mtx);
300     int r = 0;
301     // NOTE: think about checking if thread is alive first before try_joining, although i believe the end result is the
302     // same
303     while ( !threads.empty() ) {
304       // the thread might be busy, in which case stall until it ends operation
305       if ( r = solo::try_join(threads.top().thread); r == error::busy ) {
306         __cpu_pause();
307         continue;
308       } else if ( r == error::invalid_arg ) {
309         micron::abort();
310       }
311       threads.pop();
312     }
313   }

315   void
316   force_clean(void)
317   {
318     micron::lock_guard l(mtx);
319     int r = 0;
320     while ( !threads.empty() ) {
321       // the thread might be busy, in which case stall until it ends operation
322       if ( r = solo::try_join(threads.top().thread); r == error::busy ) {
323         solo::terminate(threads.top().thread);
324       } else if ( r == error::invalid_arg ) {

326       } else if ( r == 0 or r == error::no_process ) {
327       }
328       threads.pop();
329     }

332   // NOTE: retries technically isn't forever, but it works
333   i32
334   join_all(u64 retries = numeric_limits<u64>::max())
335   {
336     micron::lock_guard l(mtx);
337     int r = 0;
338     while ( !threads.empty() ) {
339       if ( threads.top().thread.alive() ) {
340       retry_join:
341         if ( r = solo::try_join(threads.top().thread); r == error::busy ) {
342           cpu_pause<10000>();      // 10 us
343           if ( --retries != 0 )
344             goto retry_join;
345           else
346             goto force_join;
347         } else if ( r == error::invalid_arg ) {
348           micron::exc<except::thread_error>("micron arena::join_all(): invalid argument");
349         } else if ( r == error::deadlock ) {
350           micron::exc<except::thread_error>("micron arena::join_all(): deadlock occured");
351           return r;
352         } else if ( r == 0 or r == error::no_process ) {
353           threads.pop();
354         } else
355           micron::exc<except::thread_error>("micron arena::join_all(): something went wrong");
356 
357       } else {
358       force_join:
359         solo::join(threads.top().thread);
360         threads.pop();
361       }
362     }
363     return 0;
364   }

366   i32
367   join(pthread_t pid, u64 retries = numeric_limits<u64>::max())
368   {
369     // don't pop
370     micron::lock_guard l(mtx);
371     for ( u64 i = 0; i < threads.size(); ++i ) {
372       if ( threads[i].thread.native_handle() != pid ) continue;
373       int r = 0;
374       do {
375         r = solo::try_join(threads[i].thread);
376         if ( r == error::busy ) {
377           __cpu_pause();
378           if ( retries == 0 ) return r;
379           --retries;
380           continue;
381         }
382         if ( r == error::invalid_arg ) return r;
383         if ( r == 0 or r == error::no_process ) return 0;
384       } while ( r == error::busy );
385       return r;
386     }
387     return 0;
388   }

519 template<umax_t Sz = concurrent_thread_stack_size, typename Tr = void_thread<Sz>>
520   requires(micron::is_same_v<Tr, void_thread<Sz>>)
521 class __concurrent_arena
522 {
523   umax_t counter;
524   umax_t add_counter;
525   micron::iarray<thread_t<Tr>, concurrent_threads> threads;
526   addr_t *__stacks[concurrent_threads]{};      // arena owns the worker stacks. void_thread::__release() nulls
527                                                // its own copy of stack_addr, so the arena must remember them
528                                                // here to actually munmap at teardown (else every stack leaks).
529   mutable micron::mutex mtx;
530 
531   addr_t *
532   __create_stack(void) const
533   {
534     addr_t *fstack = micron::addrmap(Sz);
535     if ( micron::mmap_failed(fstack) )
536       micron::exc<except::thread_error>("micron concurrent_arena::__create_stack(): failed to allocate stack");
537     return fstack;
538   }
539 
540   void
541   __free_stack(umax_t idx)
542   {
543     if ( __stacks[idx] == nullptr ) return;
544     int r = micron::try_unmap(__stacks[idx], Sz);
545     __stacks[idx] = nullptr;
546     if ( r < 0 ) {
547       micron::exc<except::memory_error>("micron concurrent_arena::__free_stack(): failed to unmap thread stack");
548     }
549   }
550 
551   bool
552   __verify_domain(thread_t<Tr> *ptr)
553   {
554     if ( ptr < reinterpret_cast<thread_t<Tr> *>(&threads) or ptr > &threads.top() ) return false;
555     return true;
556   }

560   using thread_class = Tr;

562   ~__concurrent_arena()
563   {
564     stop_all();
565     join_all();
566     umax_t n = 0;
567     {
568       // NOTE: important so we don't race/lock accidentally
569       micron::lock_guard l(mtx);
570       n = counter;
571     }
572     for ( umax_t i = 0; i < n; ++i ) __free_stack(i);
573   }

582   // thread creation functions
583   template<typename Func, typename... Args>
584     requires(micron::is_invocable_v<Func, Args...>)
585   auto &
586   create(Func f, Args &&...args)
587   {
588     micron::lock_guard l(mtx);
589     if ( counter >= concurrent_threads ) micron::exc<except::thread_error>("micron concurrent_arena::create(): too many created threads");
590     addr_t *stack_ptr = __create_stack();
591     __stacks[counter] = stack_ptr;      // remember the stack so the arena can unmap it at teardown
592     thread_attr_t attrs = micron::__thread_attr_with_stack(posix::getpid(), posix::sched_other, stack_ptr, Sz);
593     // WARNING: important, make sure no temporaries are created
594     auto __cn = cpu_t<true>();
595     new (&threads.mut(counter)) thread_t<Tr>{ micron::move(__cn), micron::move(attrs), f, micron::forward<Args>(args)... };
596     sstring<16> thread_name = "c/" + int_to_string_stack<u32, char, 12>(static_cast<u32>(counter));
597     thread_name[15] = 0x0;
598     micthread::set_name(threads.mut(counter).thread.native_handle(), thread_name.c_str());
599     return threads.mut(counter++);
600   }

605     micron::lock_guard l(mtx);
606     if ( counter >= concurrent_threads ) micron::exc<except::thread_error>("micron concurrent_arena::create(): too many created threads");
607     addr_t *stack_ptr = __create_stack();
608     __stacks[counter] = stack_ptr;      // remember the stack so the arena can unmap it at teardown
609     thread_attr_t attrs = micron::__thread_attr_with_stack(posix::getpid(), posix::sched_other, stack_ptr, Sz);
610     // WARNING: important, make sure no temporaries are created
611     auto __cn = cpu_t<true>();
612     new (&threads.mut(static_cast<usize>(counter))) thread_t<Tr>{ micron::move(__cn), micron::move(attrs) };
613     sstring<16> thread_name = "c/" + int_to_string_stack<u32, char, 12>(static_cast<u32>(counter));
614     thread_name[15] = 0x0;
615     micthread::set_name(threads.mut(static_cast<usize>(counter)).thread.native_handle(), thread_name.c_str());
616     return threads.mut(static_cast<usize>(counter++));
617   }

622   template<typename Fn, typename... Args>
623     requires(micron::is_invocable_v<Fn, Args...>)
624   auto &
625   add(Fn &&fn, Args &&...args)
626   {
627     micron::lock_guard l(mtx);
628     if ( !counter ) micron::exc<except::thread_error>("micron concurrent_arena::add(): empty arena");
629     threads.mut(static_cast<usize>(add_counter)).thread[micron::forward<Fn &&>(fn), micron::forward<Args &&>(args)...];
630     umax_t __ot = add_counter;
631     if ( ++add_counter >= counter ) add_counter = 0;
632     return threads[static_cast<usize>(__ot)];
633   }

637   {
638     micron::lock_guard l(mtx);
639     for ( umax_t i = 0; i < counter; ++i ) {
640       threads.mut(i).thread.stop();
641     }
