13 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
14 // condition counter over a futex
15 
16 namespace micron
17 {
18 
19 class event_count
20 {
21   micron::atomic_token<u32> __epoch{ 0 };        // bumped on every notify; waiters sleep while it == their key
22   micron::atomic_token<u32> __waiters{ 0 };      // count of threads between prepare_wait and commit/cancel
23 
24 public:
25   event_count() noexcept = default;
26   event_count(const event_count &) = delete;
27   event_count &operator=(const event_count &) = delete;
28 
29   [[nodiscard]] u32
30   prepare_wait() noexcept
31   {
32     __waiters.fetch_add(1, memory_order::acq_rel);
33     return __epoch.get(memory_order::acquire);
34   }
35 
36   // abandon the wait
37   void
38   cancel_wait() noexcept
39   {
40     __waiters.sub_fetch(1, memory_order::acq_rel);
41   }
42 
43   void
44   commit_wait(u32 __key, const timespec_t *__timeout = nullptr) noexcept
45   {
46     while ( __epoch.get(memory_order::acquire) == __key ) {
47       auto __r = micron::__futex(__epoch.ptr(), futex_wait | futex_private_flag, __key, const_cast<timespec_t *>(__timeout), nullptr, 0);
48       if ( __timeout != nullptr && __r == -110 ) break;             // ETIMEDOUT: honor the finite timeout
49       if ( __r < 0 && __r != -11 && __r != -4 && __r != -110 )      // not EAGAIN/EINTR/ETIMEDOUT -> bail
50         break;
51     }
52     __waiters.sub_fetch(1, memory_order::acq_rel);
53   }
54 
55   // wake one parked waiter (if any)
56   void
57   notify_one() noexcept
58   {
59     __epoch.fetch_add(1, memory_order::acq_rel);
60     if ( __waiters.get(memory_order::acquire) != 0 ) micron::wake_futex(__epoch.ptr(), 1);
61   }
62 
63   // wake all parked waiters (if any)
64   void
65   notify_all() noexcept
66   {
67     __epoch.fetch_add(1, memory_order::acq_rel);
68     if ( __waiters.get(memory_order::acquire) != 0 ) micron::wake_futex(__epoch.ptr(), 0x7fffffff);
69   }
70 };
71 
72 };      // namespace micron
