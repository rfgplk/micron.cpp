22 // (when) is a framework for enabling asynchronous work. similar to 'async' in the STL and other languages, but with a
23 // twist the general usage pattern is you select some boolean condition (which must evaluate to true) with a
24 // function/workload attached. upon calling when a background thread is launched which periodically checks if the
25 // condition has been met, and if it has, launches the task
26 
27 namespace micron
28 {
29 
30 template<typename Fn, typename... Args>
31 void
32 when(atomic_token<bool> *result, Fn fn, Args... args)
33 {
34   go([result, fn = micron::move(fn), ... args = micron::move(args)]() mutable {
35     while ( result->get(memory_order::acquire) != true ) cpu_pause<500>();
36     fn(args...);
37   });
38 }
39 
40 template<typename Fn, typename... Flags>
41   requires(micron::is_invocable_v<Fn> && (micron::is_same_v<Flags, atomic_token<bool>> && ...))
42 void
43 when_any(Fn fn, Flags *...flags)
44 {
45   (go([f = flags, fn = fn]() mutable {
46      while ( f->get(memory_order::acquire) != true ) cpu_pause<500>();
47      fn();
48    }),
49    ...);
50 }
