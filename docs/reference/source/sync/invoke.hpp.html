22 template<class C, class Pointed, class Object, class... Args>
23 constexpr decltype(auto)
24 invoke_memptr(Pointed C::*member, Object &&object, Args &&...args)
25 {
26   using object_t = remove_cvref_t<Object>;
27   constexpr bool is_member_function = micron::is_function_v<Pointed>;
28   constexpr bool is_wrapped = is_reference_wrapper_v<object_t>;
29   constexpr bool is_derived_object = micron::is_same_v<C, object_t> || micron::is_base_of_v<C, object_t>;
30 
31   if constexpr ( is_member_function ) {
32     if constexpr ( is_derived_object )
33       return (micron::forward<Object>(object).*member)(micron::forward<Args>(args)...);
34     else if constexpr ( is_wrapped )
35       return (object.get().*member)(micron::forward<Args>(args)...);
36     else
37       return ((*micron::forward<Object>(object)).*member)(micron::forward<Args>(args)...);
38   } else {
39     static_assert(micron::is_object_v<Pointed> && sizeof...(args) == 0);
40     if constexpr ( is_derived_object )
41       return micron::forward<Object>(object).*member;
42     else if constexpr ( is_wrapped )
43       return object.get().*member;
44     else
45       return (*micron::forward<Object>(object)).*member;
46   }
47 }

49 template<class F, class... Args>
50 constexpr micron::invoke_result_t<F, Args...>
51 invoke(F &&f, Args &&...args) noexcept(micron::is_nothrow_invocable_v<F, Args...>)
52 {
53   if constexpr ( micron::is_member_pointer_v<micron::remove_cvref_t<F>> )
54     return micron::invoke_memptr(f, micron::forward<Args>(args)...);
55   else
56     return micron::forward<F>(f)(micron::forward<Args>(args)...);
57 }

63 {
64   if constexpr ( micron::is_void_v<R> )
