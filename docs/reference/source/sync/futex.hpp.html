61 inline auto
62 __futex(u32 *addr, int futex, u32 val, timespec_t *timeout, u32 *addr2, u32 val2)
63 {
64 #if defined(__micron_arch_width_32)
65   if ( timeout != nullptr ) {
66     if ( __futex_time64_gate.open(kernel::feature::futex_time64) ) [[likely]] {
67       const long r = micron::syscall(SYS_futex_time64, addr, futex, val, timeout, addr2, val2);
68       // demote on ENOSYS only
69       if ( r != -static_cast<long>(error::bad_syscall) ) return r;
70       __futex_time64_gate.demote();
71     }
72     // pre-5.1: narrow into the old layout rather than pass a struct the kernel will misread
73     const __futex_old_ts __old{ static_cast<i32>(timeout->tv_sec > 0x7FFF'FFFFll ? 0x7FFF'FFFFll : timeout->tv_sec),
74                                 static_cast<i32>(timeout->tv_nsec) };
75     return micron::syscall(SYS_futex, addr, futex, val, &__old, addr2, val2);
76   }
77 #endif
78   return micron::syscall(SYS_futex, addr, futex, val, timeout, addr2, val2);
79 }

83 // A single recoverable wait. Callers recheck their predicate after every
84 // return, including interruptions, value changes and spurious wakeups.
85 [[nodiscard]] inline int
86 try_wait_futex(u32 *ptr, u32 expected, timespec_t *timeout = nullptr) noexcept
87 {
88   if ( !ptr ) return -error::invalid_arg;
89   if ( atom::load(ptr, static_cast<int>(memory_order_acquire)) != expected ) return 0;
90   return static_cast<int>(__futex(ptr, futex_wait | futex_private_flag, expected, timeout, nullptr, 0));
91 }

93 template<typename T>
94   requires(sizeof(T) == 4)
95 void
96 wait_futex(T *ptr, decay_t<T> expected)
97 {
98   while ( atom::load(ptr, (int)memory_order_relaxed) == expected ) {
99     auto ret = __futex(reinterpret_cast<u32 *>(const_cast<decay_t<T> *>(ptr)), futex_wait | futex_private_flag, static_cast<u32>(expected),
100                        nullptr, nullptr, 0);
101     if ( ret < 0 and ret != -11 and ret != -4 ) {
102       micron::exc<except::thread_error>("futex wait failed");
103     }
104   }
105 }

125 // __D is the unlocked value
126 // __L is the locked value
127 template<typename T = u32, T __D = 0, T __L = 1>
128   requires(micron::is_integral_v<T> && __D != __L)
129 struct futex {
130   T __value;
131   ~futex() = default;
132   futex(void) : __value(__D) { };
133   futex(const futex &) = delete;
134   futex(futex &&) = delete;
135   futex &operator=(const futex &) = delete;
136   futex &operator=(futex &&) = delete;
137 
138   void
139   wait()
140   {
141     for ( ;; ) {
142       T expected = __D;
143       if ( atom::cmp_exchange_weak(&__value, &expected, __L) ) return;
144       auto ret
145           = __futex(reinterpret_cast<u32 *>(&__value), futex_wait | futex_private_flag, static_cast<u32>(expected), nullptr, nullptr, 0);
146       if ( ret < 0 && ret != -11 && ret != -4 ) {
147         micron::exc<except::thread_error>("futex wait failed");
148       }
149     }
150   }
151 
152   void
153   release()
154   {
155     atom::store(&__value, __D, atomic_seq_cst);
156     __futex(reinterpret_cast<u32 *>(&__value), futex_wake | futex_private_flag, 1, nullptr, nullptr, 0);
157   }
158 };
