18 class latch
19 {
20   atomic_token<int> counter;
21 
22 public:
23   explicit latch(int expected) : counter(expected) { }
24 
25   void
26   count_down(int n = 1) noexcept
27   {
28     if ( n <= 0 ) return;
29     int cur = counter.get(memory_order::relaxed);
30     for ( ;; ) {
31       if ( cur <= 0 ) return;
32       int step = (cur < n) ? cur : n;
33       if ( counter.compare_exchange_weak(cur, cur - step, memory_order::seq_cst, memory_order::relaxed) ) {
34         // used to issue no wake
35         if ( cur - step <= 0 ) micron::wake_futex(counter.ptr(), 0x7fffffff);
36         return;
37       }
38     }
39   }
40 
41   // spin briefly, then park
42   void
43   wait() const noexcept
44   {
45     default_backoff bo;
46     for ( ;; ) {
47       const int cur = counter.get(memory_order::acquire);
48       if ( cur <= 0 ) return;
49       if ( bo.next() != spin_step::park ) continue;
50       auto r = micron::__futex(const_cast<u32 *>(reinterpret_cast<const u32 *>(counter.ptr())), futex_wait | futex_private_flag,
51                                static_cast<u32>(cur), nullptr, nullptr, 0);
52       if ( r < 0 && r != -11 && r != -4 ) return;
53     }
54   }
55 
56   bool
57   try_wait() const noexcept
58   {
59     return counter.get(memory_order::seq_cst) <= 0;
60   }
61 
62   int
63   expected() const noexcept
64   {
65     return counter.get(memory_order::seq_cst);
66   }
67 };

69 class barrier
70 {
71   atomic_token<u64> state;
72   const u32 threshold;
73   mutable event_count gate;
74 
75   static constexpr u64
76   pack(u32 gen, u32 cnt) noexcept
77   {
78     return ((u64)gen << 32) | (u64)cnt;
79   }
80 
81 public:
82   explicit barrier(int num_threads) : state(pack(0, (u32)num_threads)), threshold((u32)num_threads) { }
83 
84   int
85   arrive_and_wait() noexcept
86   {
87     u64 cur = state.get(memory_order::acquire);
88     for ( ;; ) {
89       u32 gen = (u32)(cur >> 32);
90       u32 cnt = (u32)cur;
91       u64 desired;
92       if ( cnt == 1 ) {
93         desired = pack(gen + 1, threshold);
94       } else {
95         desired = pack(gen, cnt - 1);
96       }
97       if ( state.compare_exchange_weak(cur, desired, memory_order::acq_rel, memory_order::acquire) ) {
98         if ( cnt == 1 ) {
99           gate.notify_all();
100           return 0;
101         }
102         default_backoff bo;
103         for ( ;; ) {
104           if ( (u32)(state.get(memory_order::acquire) >> 32) != gen ) break;
105           if ( bo.next() != spin_step::park ) continue;
106           const u32 key = gate.prepare_wait();
107           if ( (u32)(state.get(memory_order::acquire) >> 32) != gen ) {
108             gate.cancel_wait();
109             break;
110           }
111           gate.commit_wait(key);
112         }
113         return (int)(cnt - 1);
114       }
115     }
116   }
117 
118   void
119   arrive_and_drop() noexcept
120   {
121     u64 cur = state.get(memory_order::acquire);
122     for ( ;; ) {
123       u32 gen = (u32)(cur >> 32);
124       u32 cnt = (u32)cur;
125       u64 desired = (cnt == 1) ? pack(gen + 1, threshold) : pack(gen, cnt - 1);
126       if ( state.compare_exchange_weak(cur, desired, memory_order::acq_rel, memory_order::acquire) ) {
127         if ( cnt == 1 ) gate.notify_all();
128         return;
129       }
130     }
131   }
132 
133   int
134   expected() const noexcept
135   {
136     return (int)((u32)state.get(memory_order::acquire));
137   }
138 };
