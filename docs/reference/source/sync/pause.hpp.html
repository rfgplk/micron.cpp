34 template<umax_t L>
35 void
36 cpu_pause()
37 {
38   timespec_t r = { L / 1000000000UL, L % 1000000000UL };
39 
40   for ( ;; ) {
41     micron::__testcancel();
42     if ( micron::nanosleep(r, r) != -error::interrupted ) break;
43     __cpu_pause();
44   }
45 }

47 template<typename F, typename... Args>
48 inline void
49 pause(F f, Args... args)
50 {
51   micron::__testcancel();
52   micron::posix::sigset_t old;
53   micron::posix::sigset_t signal;
54   micron::posix::sigemptyset(signal);
55   micron::posix::sigaddset(signal, posix::sig_cont);
56   micron::posix::sigprocmask(posix::sig_block, signal, &old);
57   int sig = 0;
58   auto s = micron::posix::sigwait(signal, sig);
59   micron::posix::sigprocmask(posix::sig_setmask, old, nullptr);
60   micron::__testcancel();
61   if ( s == 0 ) f(sig, args...);
62   return;
63 }

108 inline int
109 can_wait(int tid)
110 {
111   // int waitid(idtype_t idtype, id_t id, micron::siginfo_t *infop, int options);
112   micron::posix::siginfo_t i;
113   micron::waitid(static_cast<idtype_t>(P_PID), static_cast<posix::id_t>(tid), i, exited | nowait);
114   if ( i.si_code == posix::cld_exited or i.si_code == posix::cld_dumped or i.si_code == posix::cld_killed ) return true;
115   return false;
116 }

126 // wait for signal, equivalent to sigsuspend
127 template<typename... Args>
128 inline int
129 await(Args... args)
130 {
131   micron::__testcancel();
132   micron::posix::sigset_t old;
133   int save;
134   micron::posix::sigset_t signal;
135   micron::posix::sigemptyset(signal);
136   (micron::posix::sigaddset(signal, args), ...);
137   if ( micron::posix::sigprocmask(posix::sig_setmask, signal, &old) < 0 ) return -1;
138   micron::posix::sigwait(signal, save);
139   if ( micron::posix::sigprocmask(posix::sig_setmask, old, nullptr) < 0 ) return -1;
140   micron::__testcancel();
141   return 0;
142 }

168 inline void
169 sleep_duration(const fduration_t s)
170 {
171   timespec_t r, rmn;
172 
173   // clamp negatives to 0; a negative duration would otherwise yield a negative tv_nsec and EINVAL-spin.
174   const auto total_ms = s < static_cast<micron::fduration_t>(0) ? static_cast<micron::fduration_t>(0) : s;
175 
176   r.tv_sec = static_cast<time64_t>(total_ms / 1000.0);
177 
178   const auto frac_ms = total_ms - static_cast<micron::fduration_t>(r.tv_sec) * 1000.0;
179 
180   r.tv_nsec = static_cast<long>(frac_ms * 1'000'000.0);
181   for ( ;; ) {
182     micron::__testcancel();
183     if ( micron::nanosleep(r, rmn) != -error::interrupted ) break;
184     r = rmn;
185     __cpu_pause();
186   }
187 }
