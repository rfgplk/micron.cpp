25 // contract{} has nothing to do with cpp26 contracts other than borrowing it's name and gen. terminology (because it sounds nice).
26 // contracts are a high-level process synchronization system, by which you are able to (with near zero abstraction cost) make certain
27 // guarantees as to the state of your process. can be used either in multithreaded environments, or singlethreaded. on contract
28 // construction, achieved by passing in a function + args, a new contract is created, but it isn't yet signed off. once sign(contract_state)
29 // is called (either with void add. arg, or another function - 'the enforcing func') the contract is then being enforced. on destructor,
30 // or a valid trigger condition of the enforcing func, the contract runs, resulting in either having fulfilled your obligation, or a
31 // contract violation.
32 
33 enum class contract_state : i32 { lenient, enforcing, strict, __end };
34 
35 template<contract_state S, typename T> class contract;
36 
37 template<contract_state S, typename T, typename Fn>
38 void
39 violation(contract<S, T> *c, Fn *fn)
40 {
41   (void)c;
42   if constexpr ( S == contract_state::lenient ) {
43     (void)fn;
44   } else if constexpr ( S == contract_state::enforcing ) {
45     // enforcing: run the (optional) enforcing callback; if it returns true the breach is fatal -> abort; abort() is noreturn and safe to
46     // call from anywhere (incl. a destructor), unlike exc<>
47     if ( fn != nullptr && fn->call() ) micron::abort();
48   } else if constexpr ( S == contract_state::strict ) {
49     exc<except::future_error>("contract::violation(): strict enforcement, contract violation detected");
50   }
51 }

53 template<typename T> using requirement_t = micron::atomic_token<T>;
54 
55 template<contract_state S, typename T> class contract
56 {
57 
58   struct fn_base_t {
59     virtual bool call() = 0;
60     virtual ~fn_base_t() = default;
61   };
62 
63   template<typename Fn> struct fn_t: fn_base_t {
64     Fn fn;
65 
66     fn_t(Fn &&f) : fn(micron::move(f)) { }
67 
68     bool
69     call() override
70     {
71       if constexpr ( !micron::is_same_v<micron::invoke_result_t<Fn>, void> )
72         return fn();
73       else
74         static_assert(false, "contract function must return");
75     }
76   };
77 
78   micron::atomic_token<bool> is_signed;
79   micron::atomic_token<bool> __violated;
80   fduration_t __duration;
81   // the enforcing function runs within the lambda which is spawned off in a separate thread. will be called exclusively from violation(),
82   // essentially acts as a callback
83   fn_base_t *__enforcing_fn;
84   // the condition function determines whether the contract succeeds/is satisfied or not
85   fn_base_t *__condition_fn;
86   // may be added to the contract, all must evaluate to true for the contract to succeed
87   micron::queue<requirement_t<T> *> __requirements;
88 
89   // non-throwing predicate: are all requirements + the condition currently satisfied?  callable from the
90   // detached worker (records a violation flag, never throws / never calls the throwing violation())
91   bool
92   __all_satisfied(void)
93   {
94     for ( requirement_t<T> *const *it = __requirements.begin(); it != __requirements.end(); ++it ) {
95       requirement_t<T> *req = *it;
96       if ( req == nullptr || !req->get(memory_order_acquire) ) return false;
97     }
98     if ( __condition_fn != nullptr && !__condition_fn->call() ) return false;
99     return true;
100   }
101 
102   void
103   __spawn_worker(void)
104   {
105     go([this]() {
106       // __duration <= 0 => evaluate once, no waiting. otherwise treat __duration as the total budget (ms)
107       const fduration_t budget = this->__duration;
108       const fduration_t deadline = micron::steady_clock::now() + (budget > 0 ? budget : 0);
109       // poll slice: a fraction of the budget, clamped to a sane [1ms, 50ms] range
110       const fduration_t slice = budget > 0 ? (budget > 50.0 ? 50.0 : (budget < 1.0 ? budget : budget / 4.0)) : 0;
111       bool ok = this->__all_satisfied();
112       while ( !ok && budget > 0 && micron::steady_clock::now() < deadline ) {
113         sleep_duration(slice);
114         ok = this->__all_satisfied();
115       }
116       if ( !ok ) this->__violated.store(true, memory_order_release);
117       // release the contract last; ~contract/wait() observe this with acquire and then surface a breach
118       this->is_signed.store(false, memory_order_release);
119     });
120   }
121 
122 public:
123   constexpr static contract_state State = S;
124 
125   ~contract()
126   {
127     // if contract has been signed, the worker is in flight; it is deadline-bounded so this terminates;
128     // it clears is_signed on completion. we never re-enter the worker, only observe its result
129     while ( is_signed.get(memory_order::acquire) ) cpu_pause<5000>();
130     if ( __violated.get(memory_order::acquire) || (__condition_fn != nullptr && !__all_satisfied()) ) {
131       if constexpr ( S == contract_state::enforcing ) violation(this, __enforcing_fn);
132       // strict/lenient breach is recorded but not thrown from the destructor (use wait() to surface it)
133     }
134     if ( __enforcing_fn != nullptr ) {
135       delete __enforcing_fn;
136       __enforcing_fn = nullptr;
137     }
138     if ( __condition_fn != nullptr ) {
139       delete __condition_fn;
140       __condition_fn = nullptr;
141     }
142   }
143 
144   contract(void) = delete;
145 
146   template<typename Fn, typename... Args>
147   contract(Fn &&fn, Args &&...args)
148       : is_signed{ false }, __violated{ false }, __duration{ 0 }, __enforcing_fn(nullptr), __condition_fn(nullptr), __requirements{}
149   {
150     auto __fn = [f = micron::forward<Fn>(fn), ... a = micron::forward<Args>(args)] { return f(a...); };
151     __condition_fn = new fn_t<decltype(__fn)>(micron::move(__fn));
152   }

154   contract(const contract &o) = delete;

156   contract(contract &&o)

158         __enforcing_fn(o.__enforcing_fn), __condition_fn(o.__condition_fn), __requirements{ micron::move(o.__requirements) }
159   {
160     o.__condition_fn = nullptr;
161     o.__enforcing_fn = nullptr;
162   };

167   void
168   sign(void)
169   {
170     bool expected = false;
171     if ( !is_signed.compare_exchange_strong(expected, true, memory_order_acq_rel) ) [[unlikely]]
172       exc<except::future_error>("contract::sign(): was already signed");
173     __spawn_worker();
174   }
175 
176   template<typename Fn, typename... Args>
177     requires(micron::is_invocable_v<Fn, Args...>)
178   void
179   sign(Fn &&fn, Args &&...args)
180   {
181     if ( __enforcing_fn != nullptr ) exc<except::future_error>("contract::sign(): already has a stored enforcing function");
182     auto __fn = [f = micron::forward<Fn>(fn), ... a = micron::forward<Args>(args)] { return f(a...); };
183     fn_base_t *enf = new fn_t<decltype(__fn)>(micron::move(__fn));
184     bool expected = false;
185     if ( !is_signed.compare_exchange_strong(expected, true, memory_order_acq_rel) ) [[unlikely]] {
186       delete enf;
187       exc<except::future_error>("contract::sign(): was already signed");
188     }
189     __enforcing_fn = enf;
190     __spawn_worker();
191   }
192 
193   // TODO: think about making this private
194   void
195   satisfy(void)
196   {
197     is_signed.store(false, memory_order_release);
198   }
199 
200   void
201   wait(void)
202   {
203     while ( is_signed.get(memory_order::acquire) ) cpu_pause<5000>();
204     if ( __violated.get(memory_order::acquire) ) violation(this, __enforcing_fn);
205   }
206 
207   bool
208   violated(void) const
209   {
210     return __violated.get(memory_order::acquire);
211   }

213   void
214   duration(fduration_t tm)
215   {
216     __duration = tm;
217   }

219   template<typename Fn, typename... Args>
220   void
221   enforce(Fn &&fn, Args &&...args)
222   {
223     if ( __enforcing_fn != nullptr ) delete __enforcing_fn;
224     auto __fn = [f = micron::forward<Fn>(fn), ... a = micron::forward<Args>(args)] { return f(a...); };
225     __enforcing_fn = new fn_t<decltype(__fn)>(micron::move(__fn));
226   }

228   template<typename... Args>
229     requires((micron::is_same_v<micron::__remove_cvref_t<Args>, requirement_t<T>> && ...))
230   void
231   require(Args &...args)
232   {
233     (__requirements.push(micron::addressof(args)), ...);
234   }
