8 template<typename F> struct __defer_guard {
9   F f;
10   bool active;
11 
12   constexpr explicit __defer_guard(F &&fn) noexcept : f(static_cast<F &&>(fn)), active(true) { }
13 
14   constexpr __defer_guard(__defer_guard &&other) noexcept : f(static_cast<F &&>(other.f)), active(other.active) { other.active = false; }
15 
16   __defer_guard(const __defer_guard &) = delete;
17   __defer_guard &operator=(const __defer_guard &) = delete;
18   __defer_guard &operator=(__defer_guard &&) = delete;
19 
20   constexpr ~__defer_guard() noexcept(noexcept(f()))
21   {
22     if ( active ) f();
23   }
24 };
25 
26 template<typename F>
27 constexpr auto
28 __make_defer(F &&f) noexcept
29 {
30   return __defer_guard<F>(static_cast<F &&>(f));
31 }
