39 template<typename T> class shared_state
40 {
41   mutable mutex mtx;
42   atomic_token<usize> refs;
43   bool ready;
44   bool retrieved;
45   bool has_exception;
46   const char *exc_msg;
47 
48   union storage_t {
49     char dummy;
50     T value;
51 
52     storage_t() : dummy(0) { }
53 
54     ~storage_t() { }
55   } storage;
56 
57 public:
58   shared_state() : refs(1), ready(false), retrieved(false), has_exception(false), exc_msg(nullptr) { }
59 
60   ~shared_state()
61   {
62     if ( ready && !has_exception ) {
63       storage.value.~T();
64     }
65   }
66 
67   shared_state(const shared_state &) = delete;
68   shared_state &operator=(const shared_state &) = delete;
69 
70   void
71   __acquire() noexcept
72   {
73     refs.fetch_add(1, memory_order::relaxed);
74   }
75 
76   bool
77   __release() noexcept
78   {
79     return refs.sub_fetch(1, memory_order::acq_rel) == 0;
80   }
81 
82   void
83   wait() const
84   {
85     for ( ;; ) {
86       {
87         lock_guard<mutex> lck(mtx);
88         if ( ready ) return;
89       }
90       __cpu_pause();
91     }
92   }
93 
94   future_status
95   wait_for(fduration_t timeout) const
96   {
97     auto start = micron::system_clock<>::now();
98     future_status r;
99     for ( ;; ) {
100       {
101         lock_guard<mutex> lck(mtx);
102         if ( ready or (micron::system_clock<>::now() - start >= timeout) ) {
103           r = ready ? future_status::ready : future_status::timeout;
104           break;
105         }
106       }
107       __cpu_pause();
108     }
109     return r;
110   }
111 
112   future_status wait_until();
113 
114   template<typename... Args>
115   void
116   set_value(Args &&...args)
117   {
118     lock_guard<mutex> lck(mtx);
119     if ( ready ) {
120       exc<except::future_error>("");
121     }
122     new (__builtin_addressof(storage.value)) T(micron::forward<Args>(args)...);
123     ready = true;
124     has_exception = false;
125   }
126 
127   void
128   set_exception(const char *msg) noexcept
129   {
130     lock_guard<mutex> lck(mtx);
131     if ( ready ) return;
132     exc_msg = msg ? msg : "broken promise";
133     has_exception = true;
134     ready = true;
135   }
136 
137   T
138   get()
139   {
140     wait();
141 
142     lock_guard<mutex> lck(mtx);
143     if ( retrieved ) {
144       exc<except::future_error>("");
145     }
146     retrieved = true;
147 
148     if ( has_exception ) {
149 #if !defined(__micron_freestanding) || defined(__micron_eh)
150       throw micron::broken_promise{ exc_msg };
151 #else
152       // freestanding without exceptions (-k): cannot unwind; trap on a broken promise instead
153       micron::exc<except::future_error>(exc_msg);
154 #endif
155     }
156 
157     return micron::move(storage.value);
158   }
159 
160   bool
161   is_ready() const
162   {
163     lock_guard<mutex> lck(mtx);
164     return ready;
165   }
166 
167   bool
168   valid() const
169   {
170     return true;
171   }
172 };

407 template<typename T> class future
408 {
409   shared_state<T> *state;
410 
411   template<typename U> friend class promise;
412 
413   explicit future(shared_state<T> *s) : state(s) { }
414 
415 public:
416   future() noexcept : state(nullptr) { }
417 
418   future(const future &) = delete;
419   future &operator=(const future &) = delete;
420 
421   future(future &&other) noexcept : state(other.state) { other.state = nullptr; }
422 
423   future &
424   operator=(future &&other) noexcept
425   {
426     if ( this != &other ) {
427       if ( state && state->__release() ) {
428         delete state;
429       }
430       state = other.state;
431       other.state = nullptr;
432     }
433     return *this;
434   }
435 
436   ~future()
437   {
438     if ( state && state->__release() ) {
439       delete state;
440     }
441   }
442 
443   T
444   get()

446     if ( !state ) {
447       exc<except::future_error>("");
448     }

450   }

455     return state != nullptr;
456   }
