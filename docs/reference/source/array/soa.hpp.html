21 // standard struct-of-arrays container
22 // given a tuple of column types Ts..., allocate one heap array per column
23 
24 template<typename... Ts>
25   requires(sizeof...(Ts) > 0)
26 class soa
27 {
28   static constexpr usize __ncols = sizeof...(Ts);
29 
30   void *__cols[__ncols] = { nullptr };
31   usize __size = 0;
32   usize __capacity = 0;
33 
34   template<usize I> using __col_t = typename micron::tuple_element_t<I, micron::tuple<Ts...>>;
35 
36   template<usize I>
37   __col_t<I> *
38   __as(void *p) noexcept
39   {
40     return static_cast<__col_t<I> *>(p);
41   }
42 
43   template<usize I>
44   const __col_t<I> *
45   __as(const void *p) const noexcept
46   {
47     return static_cast<const __col_t<I> *>(p);
48   }
49 
50   template<typename T>
51   static void *
52   __raw_new(usize cap)
53   {
54     if constexpr ( alignof(T) > __STDCPP_DEFAULT_NEW_ALIGNMENT__ )
55       return ::operator new(sizeof(T) * cap, static_cast<std::align_val_t>(alignof(T)));
56     else
57       return ::operator new(sizeof(T) * cap);
58   }
59 
60   template<typename T>
61   static void
62   __raw_delete(void *p) noexcept
63   {
64     if ( !p ) return;
65     if constexpr ( alignof(T) > __STDCPP_DEFAULT_NEW_ALIGNMENT__ )
66       ::operator delete(p, static_cast<std::align_val_t>(alignof(T)));
67     else
68       ::operator delete(p);
69   }

71   template<usize I>
72   void
73   __alloc_column(usize cap)
74   {
75     __cols[I] = __raw_new<__col_t<I>>(cap);
76   }
77 
78   template<usize I>
79   void
80   __free_column() noexcept
81   {
82     using T = __col_t<I>;
83     if ( __cols[I] ) {
84       if constexpr ( !micron::is_trivially_destructible_v<T> ) {
85         T *p = __as<I>(__cols[I]);
86         for ( usize i = 0; i < __size; ++i ) p[i].~T();
87       }
88       __raw_delete<T>(__cols[I]);
89       __cols[I] = nullptr;
90     }
91   }
92 
93   template<usize I>
94   void
95   __move_column(void *src, void *dst, usize n)
96   {
97     using T = __col_t<I>;
98     T *s = static_cast<T *>(src);
99     T *d = static_cast<T *>(dst);
100     if constexpr ( micron::is_trivially_copyable_v<T> ) {
101       micron::bytecpy(reinterpret_cast<byte *>(d), reinterpret_cast<const byte *>(s), n * sizeof(T));
102     } else {
103       static_assert(micron::is_nothrow_move_constructible_v<T>,
104                     "soa: growable non-trivially-copyable column types must be nothrow-move-constructible");
105       for ( usize i = 0; i < n; ++i ) {
106         new (d + i) T(micron::move(s[i]));
107         s[i].~T();
108       }
109     }
110   }

112   template<usize... Is>
113   void
114   __alloc_all(usize cap, micron::index_sequence<Is...>)
115   {
116     (__alloc_column<Is>(cap), ...);
117   }
118 
119   template<usize... Is>
120   void
121   __free_all(micron::index_sequence<Is...>) noexcept
122   {
123     (__free_column<Is>(), ...);
124   }
125 
126   template<usize... Is>
127   void
128   __move_all(void **src_cols, void **dst_cols, usize n, micron::index_sequence<Is...>)
129   {
130     (__move_column<Is>(src_cols[Is], dst_cols[Is], n), ...);
131   }

133   // roll back a partial allocation
134   template<usize... Is>
135   static void
136   __free_raw_indexed(void **cols, usize upto, micron::index_sequence<Is...>) noexcept
137   {
138     (((void)(Is < upto ? (__raw_delete<__col_t<Is>>(cols[Is]), cols[Is] = nullptr, void()) : void())), ...);
139   }
140 
141   // free OLD column buffers after a successful relocate
142   template<usize... Is>
143   void
144   __free_old_raw(void **cols, micron::index_sequence<Is...>) noexcept
145   {
146     (((void)(cols[Is] ? (__raw_delete<__col_t<Is>>(cols[Is]), void()) : void())), ...);
147   }
148 
149   template<usize... Is>
150   void
151   __grow_to_impl(usize new_cap, void **new_cols, micron::index_sequence<Is...> seq)
152   {
153     usize done = 0;
154 #if !defined(__micron_freestanding) || defined(__micron_eh)
155     try {
156       (((new_cols[Is] = __raw_new<__col_t<Is>>(new_cap)), ++done), ...);
157     } catch ( ... ) {
158       __free_raw_indexed(new_cols, done, seq);      // free the columns already allocated, then rethrow
159       throw;
160     }
161 #else
162     (((new_cols[Is] = __raw_new<__col_t<Is>>(new_cap)), ++done), ...);
163 #endif
164   }
165 
166   void
167   __grow_to(usize new_cap)
168   {
169     if ( new_cap <= __capacity ) return;
170     auto seq = micron::make_index_sequence<__ncols>{};
171     void *new_cols[__ncols] = { nullptr };
172     __grow_to_impl(new_cap, new_cols, seq);
173     __move_all(__cols, new_cols, __size, seq);
174     __free_old_raw(__cols, seq);
175     for ( usize i = 0; i < __ncols; ++i ) __cols[i] = new_cols[i];
176     __capacity = new_cap;
177   }
178 
179   template<usize I>
180   void
181   __destroy_one_at(usize idx) noexcept
182   {
183     using T = __col_t<I>;
184     if constexpr ( !micron::is_trivially_destructible_v<T> ) __as<I>(__cols[I])[idx].~T();
185   }

187   template<usize... Is>
188   void
189   __destroy_row_indexed(usize idx, usize upto, micron::index_sequence<Is...>) noexcept
190   {
191     (((void)(Is < upto ? (__destroy_one_at<Is>(idx), void()) : void())), ...);
192   }

194   template<usize... Is>
195   void
196   __emplace_at(usize idx, micron::index_sequence<Is...> seq, Ts &&...vs)
197   {
198     usize done = 0;
199 #if !defined(__micron_freestanding) || defined(__micron_eh)
200     try {
201       (((void)(new (__as<Is>(__cols[Is]) + idx) __col_t<Is>(micron::forward<Ts>(vs))), ++done), ...);
202     } catch ( ... ) {
203       __destroy_row_indexed(idx, done, seq);      // unwind the already-built columns of this row
204       throw;
205     }
206 #else
207     (((void)(new (__as<Is>(__cols[Is]) + idx) __col_t<Is>(micron::forward<Ts>(vs))), ++done), ...);
208 #endif
209   }

211   template<usize I>
212   void
213   __destroy_column_elements() noexcept
214   {
215     using T = __col_t<I>;
216     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
217       T *p = __as<I>(__cols[I]);
218       for ( usize i = 0; i < __size; ++i ) p[i].~T();

220   }

229   template<usize I>
230   void
231   __pop_back_column() noexcept
232   {
233     using T = __col_t<I>;
234     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
235       __as<I>(__cols[I])[__size - 1].~T();
236     }
237   }

239   template<usize... Is>
240   void
241   __pop_back_all(micron::index_sequence<Is...>) noexcept
242   {
243     (__pop_back_column<Is>(), ...);
244   }

246 public:
247   using category_type = array_tag;

251   static constexpr usize column_count = __ncols;
252 
253   ~soa() { __free_all(micron::make_index_sequence<__ncols>{}); }

255   soa() = default;
256 
257   explicit soa(usize cap)
258   {
259     if ( cap > 0 ) {
260       __alloc_all(cap, micron::make_index_sequence<__ncols>{});
261       __capacity = cap;
262     }
263   }

265   soa(const soa &) = delete;
266   soa &operator=(const soa &) = delete;

268   soa(soa &&o) noexcept : __size(o.__size), __capacity(o.__capacity)
269   {
270     for ( usize i = 0; i < __ncols; ++i ) {
271       __cols[i] = o.__cols[i];
272       o.__cols[i] = nullptr;
273     }
274     o.__size = 0;
275     o.__capacity = 0;
276   }
277 
278   soa &
279   operator=(soa &&o) noexcept
280   {
281     if ( this == &o ) return *this;
282     __free_all(micron::make_index_sequence<__ncols>{});
283     for ( usize i = 0; i < __ncols; ++i ) {
284       __cols[i] = o.__cols[i];
285       o.__cols[i] = nullptr;
286     }
287     __size = o.__size;
288     __capacity = o.__capacity;
289     o.__size = 0;
290     o.__capacity = 0;
291     return *this;
292   }

312   void
313   reserve(usize cap)

318   void
319   emplace_back(Ts... vs)
320   {
321     if ( __size == __capacity ) __grow_to(__capacity == 0 ? 16 : __capacity * 2);
322     __emplace_at(__size, micron::make_index_sequence<__ncols>{}, micron::forward<Ts>(vs)...);
323     ++__size;
324   }

400   void
401   swap(soa &o) noexcept
402   {
403     for ( usize i = 0; i < __ncols; ++i ) {
404       void *t = __cols[i];
405       __cols[i] = o.__cols[i];
406       o.__cols[i] = t;
407     }
408     usize s = __size;
409     __size = o.__size;
410     o.__size = s;
411     usize c = __capacity;
412     __capacity = o.__capacity;
413     o.__capacity = c;
414   }

416   void
417   pop_back()
418   {
419     if ( __size == 0 ) [[unlikely]]
420       exc<except::library_error>("soa::pop_back: empty");
421     __pop_back_all(micron::make_index_sequence<__ncols>{});
422     --__size;
423   }
