51   static inline void
52   __lock_ordered(micron::fast_mutex &a, __defer &la, micron::fast_mutex &b, __defer &lb)
53   {
54     if ( micron::addr(a) == micron::addr(b) ) {
55       la.lock();
56     } else if ( static_cast<const void *>(micron::addr(a)) < static_cast<const void *>(micron::addr(b)) ) {
57       la.lock();
58       lb.lock();
59     } else {
60       lb.lock();
61       la.lock();
62     }
63   }
