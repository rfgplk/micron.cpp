30   static constexpr usize
31   __cpow(usize base, usize exp)
32   {
33     usize r = 1;
34     for ( usize i = 0; i < exp; ++i ) r *= base;
35     return r;
36   }

83   template<usize Lvl>
84   static inline void *
85   __retain(void *p)
86   {
87     if ( !p ) [[unlikely]]
88       return nullptr;
89     // atomic refcount: a persistent value may be shared (copied) across threads
90     if constexpr ( Lvl == 0 )
91       __atomic_fetch_add(&static_cast<__leaf *>(p)->refs, 1u, __ATOMIC_RELAXED);
92     else
93       __atomic_fetch_add(&static_cast<__node *>(p)->refs, 1u, __ATOMIC_RELAXED);
94     return p;
95   }

97   template<usize Lvl>
98   static void
99   __release(void *p)
100   {
101     if ( !p ) [[unlikely]]
102       return;
103 
104     if constexpr ( Lvl == 0 ) {
105       __leaf *l = static_cast<__leaf *>(p);
106       if ( __atomic_fetch_sub(&l->refs, 1u, __ATOMIC_ACQ_REL) == 1u ) [[unlikely]]
107         __dealloc_leaf(l);
108     } else {
109       __node *n = static_cast<__node *>(p);
110       if ( __atomic_fetch_sub(&n->refs, 1u, __ATOMIC_ACQ_REL) == 1u ) [[unlikely]] {
111         for ( usize i = 0; i < B; ++i ) __release<Lvl - 1>(n->children[i]);
112         abc::dealloc(reinterpret_cast<byte *>(n));
113       }
114     }
115   }

117   template<usize Lvl, typename Vf>
118   static void *
119   __set_impl(void *p, usize idx, Vf &&val)
120   {
121     if constexpr ( Lvl == 0 ) {
122       // leaf level
123       const usize slot = idx & __mask;
124       __leaf *fresh = __alloc_leaf();
125       fresh->refs = 1;
126 
127       if constexpr ( micron::is_trivially_copyable_v<T> ) {
128         if ( p )
129           micron::bytecpy(reinterpret_cast<byte *>(fresh->values), reinterpret_cast<const byte *>(static_cast<const __leaf *>(p)->values),
130                           B * sizeof(T));
131         else
132           for ( usize i = 0; i < B; ++i ) new (micron::addr(fresh->values[i])) T();
133         fresh->values[slot] = static_cast<Vf &&>(val);
134         return fresh;
135       } else {
136         usize built = 0;
137 #if !defined(__micron_freestanding) || defined(__micron_eh)
138         try {
139 #endif
140           if ( p ) {
141             const __leaf *old = static_cast<const __leaf *>(p);
142             for ( usize i = 0; i < B; ++i, ++built ) new (micron::addr(fresh->values[i])) T(old->values[i]);
143           } else {
144             for ( usize i = 0; i < B; ++i, ++built ) new (micron::addr(fresh->values[i])) T();
145           }
146           fresh->values[slot] = static_cast<Vf &&>(val);
147 #if !defined(__micron_freestanding) || defined(__micron_eh)
148         } catch ( ... ) {
149           for ( usize j = 0; j < built; ++j ) fresh->values[j].~T();
150           abc::dealloc(reinterpret_cast<byte *>(fresh));
151           throw;
152         }
153 #endif
154         return fresh;
155       }
156 
157     } else {
158       // internal level
159       const usize slot = (idx >> (Lvl * K)) & __mask;
160       __node *fresh = __alloc_internal();
161 #if !defined(__micron_freestanding) || defined(__micron_eh)
162       try {
163 #endif
164         if ( p ) {
165           const __node *old = static_cast<const __node *>(p);
166           for ( usize i = 0; i < B; ++i ) {
167             if ( i == slot )
168               fresh->children[i] = __set_impl<Lvl - 1>(old->children[i], idx, static_cast<Vf &&>(val));
169             else
170               fresh->children[i] = __retain<Lvl - 1>(old->children[i]);
171           }
172         } else {
173           fresh->children[slot] = __set_impl<Lvl - 1>(nullptr, idx, static_cast<Vf &&>(val));
174         }
175 #if !defined(__micron_freestanding) || defined(__micron_eh)
176       } catch ( ... ) {
177         __release<Lvl>(fresh);
178         throw;
179       }
180 #endif
181       return fresh;
182     }
183   }

207   template<usize Lvl>
208   static void *
209   __build_from(const T *data, usize count, usize base)
210   {
211     if ( base >= count ) [[unlikely]]
212       return nullptr;
213 
214     if constexpr ( Lvl == 0 ) {
215       __leaf *l = __alloc_leaf();
216       l->refs = 1;
217       usize built = 0;
218 #if !defined(__micron_freestanding) || defined(__micron_eh)
219       try {
220 #endif
221         for ( usize i = 0; i < B; ++i, ++built ) {
222           if ( base + i < count )
223             new (micron::addr(l->values[i])) T(data[base + i]);
224           else
225             new (micron::addr(l->values[i])) T();
226         }
227 #if !defined(__micron_freestanding) || defined(__micron_eh)
228       } catch ( ... ) {
229         for ( usize j = 0; j < built; ++j ) l->values[j].~T();
230         abc::dealloc(reinterpret_cast<byte *>(l));
231         throw;
232       }
233 #endif
234       return l;
235     } else {
236       static constexpr usize child_span = __cpow(B, Lvl);
237       __node *n = __alloc_internal();
238 #if !defined(__micron_freestanding) || defined(__micron_eh)
239       try {
240 #endif
241         for ( usize i = 0; i < B; ++i ) n->children[i] = __build_from<Lvl - 1>(data, count, base + i * child_span);
242 #if !defined(__micron_freestanding) || defined(__micron_eh)
243       } catch ( ... ) {
244         __release<Lvl>(n);
245         throw;
246       }
247 #endif
248       return n;
249     }
250   }

252   template<usize Lvl>
253   static void *
254   __build_filled(const T &val)
255   {
256     if constexpr ( Lvl == 0 ) {
257       __leaf *l = __alloc_leaf();
258       l->refs = 1;
259       usize built = 0;
260 #if !defined(__micron_freestanding) || defined(__micron_eh)
261       try {
262 #endif
263         for ( usize i = 0; i < B; ++i, ++built ) new (micron::addr(l->values[i])) T(val);
264 #if !defined(__micron_freestanding) || defined(__micron_eh)
265       } catch ( ... ) {
266         for ( usize j = 0; j < built; ++j ) l->values[j].~T();
267         abc::dealloc(reinterpret_cast<byte *>(l));
268         throw;
269       }
270 #endif
271       return l;
272     } else {
273       __node *n = __alloc_internal();
274       void *child;
275 #if !defined(__micron_freestanding) || defined(__micron_eh)
276       try {
277 #endif
278         child = __build_filled<Lvl - 1>(val);
279 #if !defined(__micron_freestanding) || defined(__micron_eh)
280       } catch ( ... ) {
281         __release<Lvl>(n);
282         throw;
283       }
284 #endif
285       // child is shared by all B slots
286       if constexpr ( Lvl - 1 == 0 )
287         static_cast<__leaf *>(child)->refs = B;
288       else
289         static_cast<__node *>(child)->refs = B;
290       for ( usize i = 0; i < B; ++i ) n->children[i] = child;
291       return n;
292     }

295   template<usize Lvl, typename Fn>
296   static void *
297   __update_impl(void *p, usize idx, Fn &&fn)
298   {
299     if constexpr ( Lvl == 0 ) {
300       const usize slot = idx & __mask;
301       __leaf *fresh = __alloc_leaf();
302       fresh->refs = 1;
303 
304       if constexpr ( micron::is_trivially_copyable_v<T> ) {
305         if ( p )
306           micron::bytecpy(reinterpret_cast<byte *>(fresh->values), reinterpret_cast<const byte *>(static_cast<const __leaf *>(p)->values),
307                           B * sizeof(T));
308         else
309           for ( usize i = 0; i < B; ++i ) new (micron::addr(fresh->values[i])) T();
310         fresh->values[slot] = fn(static_cast<const T &>(fresh->values[slot]));
311         return fresh;
312       } else {
313         usize built = 0;
314 #if !defined(__micron_freestanding) || defined(__micron_eh)
315         try {
316 #endif
317           if ( p ) {
318             const __leaf *old = static_cast<const __leaf *>(p);
319             for ( usize i = 0; i < B; ++i, ++built ) new (micron::addr(fresh->values[i])) T(old->values[i]);
320           } else {
321             for ( usize i = 0; i < B; ++i, ++built ) new (micron::addr(fresh->values[i])) T();
322           }
323           T result = fn(static_cast<const T &>(fresh->values[slot]));
324           fresh->values[slot] = micron::move(result);
325 #if !defined(__micron_freestanding) || defined(__micron_eh)
326         } catch ( ... ) {
327           for ( usize j = 0; j < built; ++j ) fresh->values[j].~T();
328           abc::dealloc(reinterpret_cast<byte *>(fresh));
329           throw;
330         }
331 #endif
332         return fresh;
333       }
334 
335     } else {
336       const usize slot = (idx >> (Lvl * K)) & __mask;
337       __node *fresh = __alloc_internal();
338 #if !defined(__micron_freestanding) || defined(__micron_eh)
339       try {
340 #endif
341         if ( p ) {
342           const __node *old = static_cast<const __node *>(p);
343           for ( usize i = 0; i < B; ++i ) {
344             if ( i == slot )
345               fresh->children[i] = __update_impl<Lvl - 1>(old->children[i], idx, static_cast<Fn &&>(fn));
346             else
347               fresh->children[i] = __retain<Lvl - 1>(old->children[i]);
348           }
349         } else {
350           fresh->children[slot] = __update_impl<Lvl - 1>(nullptr, idx, static_cast<Fn &&>(fn));
351         }
352 #if !defined(__micron_freestanding) || defined(__micron_eh)
353       } catch ( ... ) {
354         __release<Lvl>(fresh);
355         throw;
356       }
357 #endif
358       return fresh;
359     }

363   static void
364   __for_each_impl(const void *p, usize base, Fn &&fn)
365   {
366     if ( !p ) return;
367 
368     if constexpr ( Lvl == 0 ) {
369       const __leaf *l = static_cast<const __leaf *>(p);
370       for ( usize i = 0; i < B; ++i ) fn(base + i, static_cast<const T &>(l->values[i]));

373       const __node *n = static_cast<const __node *>(p);

375     }

378   template<usize Lvl>
379   static bool
380   __equal_impl(const void *a, const void *b)
381   {
382     if ( a == b ) [[unlikely]]
383       return true;
384 
385     // both null == equal
386     if ( !a && !b ) [[unlikely]]
387       return true;
388 
389     // one null, one valid
390     if ( !a || !b ) {
391       const void *live = a ? a : b;
392       if constexpr ( Lvl == 0 ) {
393         const __leaf *l = static_cast<const __leaf *>(live);
394         const T d{};
395         for ( usize i = 0; i < B; ++i )
396           if ( !(l->values[i] == d) ) return false;
397         return true;
398       } else {
399         const __node *n = static_cast<const __node *>(live);
400         for ( usize i = 0; i < B; ++i )
401           if ( !__equal_impl<Lvl - 1>(n->children[i], nullptr) ) return false;
402         return true;
403       }
404     }
405 
406     if constexpr ( Lvl == 0 ) {
407       const __leaf *la = static_cast<const __leaf *>(a);
408       const __leaf *lb = static_cast<const __leaf *>(b);
409       if constexpr ( micron::is_trivially_copyable_v<T> ) {
410         return micron::bytecmp(reinterpret_cast<const byte *>(la->values), reinterpret_cast<const byte *>(lb->values), B * sizeof(T)) == 0;
411       } else {
412         for ( usize i = 0; i < B; ++i )
413           if ( !(la->values[i] == lb->values[i]) ) return false;
414         return true;
415       }
416     } else {
417       const __node *na = static_cast<const __node *>(a);
418       const __node *nb = static_cast<const __node *>(b);
419       for ( usize i = 0; i < B; ++i )
420         if ( !__equal_impl<Lvl - 1>(na->children[i], nb->children[i]) ) return false;
421       return true;
422     }
423   }

433   template<usize Lvl, typename Fn>
434   static void *
435   __apply_scalar_impl(const void *p, const T &scalar, Fn &&fn)
436   {
437     if constexpr ( Lvl == 0 ) {
438       __leaf *fresh = __alloc_leaf();
439       fresh->refs = 1;
440       if ( p ) {
441         const __leaf *old = static_cast<const __leaf *>(p);
442         for ( usize i = 0; i < B; ++i ) new (micron::addr(fresh->values[i])) T(fn(old->values[i], scalar));
443       } else {
444         const T d{};
445         for ( usize i = 0; i < B; ++i ) new (micron::addr(fresh->values[i])) T(fn(d, scalar));
446       }
447       return fresh;
448     } else {
449       __node *fresh = reinterpret_cast<__node *>(abc::alloc(sizeof(__node)));
450       fresh->refs = 1;
451       if ( p ) {
452         const __node *old = static_cast<const __node *>(p);
453         for ( usize i = 0; i < B; ++i ) fresh->children[i] = __apply_scalar_impl<Lvl - 1>(old->children[i], scalar, fn);
454       } else {
455         for ( usize i = 0; i < B; ++i ) fresh->children[i] = __apply_scalar_impl<Lvl - 1>(nullptr, scalar, fn);
456       }
457       return fresh;
458     }
