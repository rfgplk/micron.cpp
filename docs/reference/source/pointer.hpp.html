15 // ptr aliases
16 // ptr<T>         : default single-object owning pointer; general unique ownership
17 // ptr_arr<T>     : owning pointer to a heap-allocated array of T
18 // atom_ptr<T>    : unique ownership with atomic load/store/exchange; use when the pointer itself is shared across threads
19 // atom_ptr_arr<T>: atomic owning pointer to a heap-allocated array of T
20 // uptr<T>        : alias for ptr<T>
21 // sptr<T>        : shared ownership across multiple owners; use when lifetime is non-deterministic
22 // fptr<T>        : manual-lifetime pointer handle, buts its destructor NEVER frees
23 // wptr<T>        : non-owning observer of a shared_pointer that does not extend lifetime; use to break cycles
24 // cptr<T>        : immutable owning pointer; value cannot be changed after construction
25 // cptr_arr<T>    : immutable owning pointer to a heap-allocated array of T
26 // gptr<T>        : global-scope owning pointer; lives for the entire binary execution, never frees on destruction
27 // gptr_arr<T>    : global-scope owning pointer to a heap-allocated array of T
28 // pointer<T>     : semantic alias for ptr<T>; use in APIs where "pointer" reads more naturally than "ptr"
29 // pointer_arr<T> : semantic alias for ptr_arr<T>
30 // shared<T>      : semantic alias for sptr<T>; use in APIs where "shared" reads more naturally

282 template<class T>
283 void *
284 voidify(const pointer<T> &pnt)
285 {
286   using erase_type = micron::remove_cv_t<decltype(pnt())>;
287   if ( pnt() != nullptr )
288     return const_cast<void *>(static_cast<const void *>(static_cast<erase_type>(pnt())));
289   else
290     return (void *)nullptr;
291 }
