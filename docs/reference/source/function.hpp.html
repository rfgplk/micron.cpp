73 // custom vtable for concrete callable type G
74 template<typename G, typename R, typename... Args>
75 const __fn_vtable<R, Args...> *
76 __make_vtable() noexcept
77 {
78   static const __fn_vtable<R, Args...> vt
79       = { // cast self back to G and invoke; through G*
80           [](void *self, Args &&...args) -> R { return (*static_cast<G *>(self))(micron::forward<Args>(args)...); },
81           // destroy
82           [](void *self) noexcept { static_cast<G *>(self)->~G(); },
83           // nullptr when G is not copy-constructible
84           __fn_copy_thunk<G>(),
85           // move_to
86           [](void *dst, void *src) noexcept {
87             new (dst) G(micron::move(*static_cast<G *>(src)));
88             static_cast<G *>(src)->~G();
89           },
90           // type_tag
91           __fn_type_tag<micron::decay_t<G>>(), sizeof(G), alignof(G),
92           // probed as G&
93           noexcept(micron::declval<G &>()(micron::declval<Args>()...))
94         };
95   return &vt;
96 }
