23 constexpr bool
24 is_scalar(u32 value) noexcept
25 {
26   return value <= 0x10ffff && !(value >= 0xd800 && value <= 0xdfff);
27 }

29 constexpr u32
30 encode_utf8(u32 value, char *output) noexcept
31 {
32   if ( !is_scalar(value) ) return 0;
33   if ( value < 0x80 ) {
34     output[0] = static_cast<char>(value);
35     return 1;
36   }
37   if ( value < 0x800 ) {
38     output[0] = static_cast<char>(0xc0 | (value >> 6));
39     output[1] = static_cast<char>(0x80 | (value & 63));
40     return 2;
41   }
42   if ( value < 0x10000 ) {
43     output[0] = static_cast<char>(0xe0 | (value >> 12));
44     output[1] = static_cast<char>(0x80 | ((value >> 6) & 63));
45     output[2] = static_cast<char>(0x80 | (value & 63));
46     return 3;
47   }
48   output[0] = static_cast<char>(0xf0 | (value >> 18));
49   output[1] = static_cast<char>(0x80 | ((value >> 12) & 63));
50   output[2] = static_cast<char>(0x80 | ((value >> 6) & 63));
51   output[3] = static_cast<char>(0x80 | (value & 63));
52   return 4;
53 }

55 template<typename Character>
56 constexpr bool
57 decode_utf8(const Character *input, usize size, usize &position, u32 &scalar) noexcept
58 {
59   static_assert(sizeof(Character) == 1);
60   if ( position >= size ) return false;
61   const auto head = static_cast<u8>(input[position++]);
62   if ( head < 0x80 ) {
63     scalar = head;
64     return true;
65   }
66   u32 count = 0, minimum = 0;
67   if ( head >= 0xc2 && head <= 0xdf ) {
68     scalar = head & 31;
69     count = 1;
70     minimum = 0x80;
71   } else if ( head >= 0xe0 && head <= 0xef ) {
72     scalar = head & 15;
73     count = 2;
74     minimum = 0x800;
75   } else if ( head >= 0xf0 && head <= 0xf4 ) {
76     scalar = head & 7;
77     count = 3;
78     minimum = 0x10000;
79   } else
80     return false;
81   if ( count > size - position ) return false;
82   while ( count-- ) {
83     const auto byte = static_cast<u8>(input[position++]);
84     if ( (byte & 0xc0) != 0x80 ) return false;
85     scalar = (scalar << 6) | (byte & 63);
86   }
87   return scalar >= minimum && is_scalar(scalar);
88 }

90 template<typename Character>
91 constexpr bool
92 validate_utf8(const Character *input, usize size) noexcept
93 {
94   usize position = 0;
95   while ( position < size ) {
96     u32 scalar = 0;
97     if ( !decode_utf8(input, size, position, scalar) ) return false;
98   }
99   return true;
100 }

102 // Caller retains only unconsumed bytes. The decoder owns at most four probe bytes
103 // and one partial scalar; scalar results can consume zero new bytes.
104 class stream_decoder
105 {
106   encoding __selected = encoding::automatic;
107   encoding __requested = encoding::automatic;
108   u8 __probe[4]{};
109   u8 __probe_size = 0;
110   u8 __probe_read = 0;
111   bool __detected = false;
112   bool __bom = false;
113   bool __done = false;
114   decode_status __failure = decode_status::need_input;
115   u64 __offset = 0;
116   u64 __scalar_offset = 0;
117   u32 __value = 0;
118   u32 __minimum = 0;
119   u16 __high = 0;
120   u8 __remaining = 0;
121   u8 __unit_byte = 0;
122   bool __half_unit = false;
123 
124   constexpr void
125   detect() noexcept
126   {
127     __detected = true;
128     __selected = __requested == encoding::automatic ? encoding::utf8 : __requested;
129     if ( __requested == encoding::automatic && __probe_size >= 4
130          && ((__probe[0] == 0 && __probe[1] == 0 && __probe[2] == 0xfe && __probe[3] == 0xff)
131              || (__probe[0] == 0xff && __probe[1] == 0xfe && __probe[2] == 0 && __probe[3] == 0)
132              || (__probe[0] == 0 && __probe[1] == 0 && __probe[2] == 0 && __probe[3] != 0)
133              || (__probe[0] != 0 && __probe[1] == 0 && __probe[2] == 0 && __probe[3] == 0)) ) {
134       __failure = decode_status::unsupported_encoding;
135       return;
136     }
137     if ( __probe_size >= 3 && __probe[0] == 0xef && __probe[1] == 0xbb && __probe[2] == 0xbf ) {
138       if ( __requested != encoding::automatic && __requested != encoding::utf8 ) {
139         __failure = decode_status::invalid;
140         return;
141       }
142       __selected = encoding::utf8;
143       __bom = true;
144       __probe_read = 3;
145     } else if ( __probe_size >= 2 && ((__probe[0] == 0xff && __probe[1] == 0xfe) || (__probe[0] == 0xfe && __probe[1] == 0xff)) ) {
146       const auto bom = __probe[0] == 0xff ? encoding::utf16_le : encoding::utf16_be;
147       if ( __requested != encoding::automatic && __requested != bom ) {
148         __failure = decode_status::invalid;
149         return;
150       }
151       __selected = bom;
152       __bom = true;
153       __probe_read = 2;
154     } else if ( __requested == encoding::automatic && __probe_size >= 4 ) {
155       // XML's signature also permits an encoding declaration before a BOM-less UTF-16 document.
156       if ( __probe[0] == 0 && __probe[1] == '<' && __probe[2] == 0 && __probe[3] == '?' )
157         __selected = encoding::utf16_be;
158       else if ( __probe[0] == '<' && __probe[1] == 0 && __probe[2] == '?' && __probe[3] == 0 )
159         __selected = encoding::utf16_le;
160     }
161   }

190   // Advance a separately validated ASCII run (UTF-8 only), for bounded SIMD readers.
191   constexpr bool
192   advance_ascii(usize bytes) noexcept
193   {
194     if ( __selected != encoding::utf8 || !at_scalar_boundary() || static_cast<u64>(bytes) > ~u64(0) - __offset ) return false;
195     __offset += bytes;
196     return true;
197   }

199   template<typename Character>
200     requires(sizeof(Character) == 1)
201   constexpr decode_result
202   next(const Character *input, usize size, bool final = false) noexcept
203   {
204     usize consumed = 0;
205     if ( __failure != decode_status::need_input ) return { __failure, 0, 0, __scalar_offset };
206     if ( __done ) {
207       if ( size == 0 ) return { decode_status::done, 0, 0, __offset };
208       __failure = decode_status::invalid;
209       __scalar_offset = __offset;
210       return { __failure, 0, 0, __offset };
211     }
212     if ( !__detected ) {
213       while ( __probe_size < 4 && consumed < size ) {
214         __probe[__probe_size++] = static_cast<u8>(input[consumed++]);
215         ++__offset;
216       }
217       if ( __probe_size < 4 && !final ) return { decode_status::need_input, consumed, 0, __offset };
218       detect();
219       if ( __failure != decode_status::need_input ) return { __failure, consumed, 0, 0 };
220     }
221     for ( ;; ) {
222       u8 byte = 0;
223       u64 position = 0;
224       if ( __probe_read < __probe_size ) {
225         position = __probe_read;
226         byte = __probe[__probe_read++];
227       } else if ( consumed < size ) {
228         position = __offset++;
229         byte = static_cast<u8>(input[consumed++]);
230       } else {
231         if ( !final ) return { decode_status::need_input, consumed, 0, __offset };
232         if ( __remaining || __half_unit || __high ) {
233           __failure = decode_status::truncated;
234           return { __failure, consumed, 0, __scalar_offset };
235         }
236         __done = true;
237         return { decode_status::done, consumed, 0, __offset };
238       }
239       if ( __selected == encoding::utf8 ) {
240         if ( __remaining == 0 ) {
241           __scalar_offset = position;
242           if ( byte < 0x80 ) return { decode_status::scalar, consumed, byte, __scalar_offset };
243           if ( byte >= 0xc2 && byte <= 0xdf ) {
244             __remaining = 1;
245             __value = byte & 31;
246             __minimum = 0x80;
247           } else if ( byte >= 0xe0 && byte <= 0xef ) {
248             __remaining = 2;
249             __value = byte & 15;
250             __minimum = 0x800;
251           } else if ( byte >= 0xf0 && byte <= 0xf4 ) {
252             __remaining = 3;
253             __value = byte & 7;
254             __minimum = 0x10000;
255           } else {
256             __failure = decode_status::invalid;
257             return { __failure, consumed, 0, __scalar_offset };
258           }
259         } else {
260           if ( (byte & 0xc0) != 0x80 ) {
261             __failure = decode_status::invalid;
262             return { __failure, consumed, 0, __scalar_offset };
263           }
264           __value = (__value << 6) | (byte & 63);
265           if ( --__remaining == 0 ) {
266             if ( __value < __minimum || !is_scalar(__value) ) {
267               __failure = decode_status::invalid;
268               return { __failure, consumed, 0, __scalar_offset };
269             }
270             return { decode_status::scalar, consumed, __value, __scalar_offset };
271           }
272         }
273       } else {
274         if ( !__half_unit ) {
275           if ( !__high ) __scalar_offset = position;
276           __unit_byte = byte;
277           __half_unit = true;
278           continue;
279         }
280         __half_unit = false;
281         const u16 unit = __selected == encoding::utf16_le ? static_cast<u16>(__unit_byte | (u16(byte) << 8))
282                                                           : static_cast<u16>((u16(__unit_byte) << 8) | byte);
283         if ( __high ) {
284           if ( unit < 0xdc00 || unit > 0xdfff ) {
285             __failure = decode_status::invalid;
286             return { __failure, consumed, 0, __scalar_offset };
287           }
288           const u32 value = 0x10000 + ((u32(__high) - 0xd800) << 10) + (u32(unit) - 0xdc00);
289           __high = 0;
290           return { decode_status::scalar, consumed, value, __scalar_offset };
291         }
292         if ( unit >= 0xd800 && unit <= 0xdbff )
293           __high = unit;
294         else if ( unit >= 0xdc00 && unit <= 0xdfff ) {
295           __failure = decode_status::invalid;
296           return { __failure, consumed, 0, __scalar_offset };
297         } else
298           return { decode_status::scalar, consumed, unit, __scalar_offset };
299       }
300     }
301   }
