47 template<is_string T>
48 inline void
49 invert(T &str)
50 {
51   if ( str.size() <= 1 ) return;
52   typename T::iterator start = str.begin();
53   typename T::iterator end = str.end() - 1;
54   while ( start < end ) {
55     typename T::value_type t = *start;
56     *start++ = *end;
57     *end-- = t;
58   }
59 }

64 template<typename I, usize N = 24>
65   requires(micron::is_integral_v<I> && N >= __impl::max_digits_v<I> + 1)
66 constexpr micron::sstring<N, char>
67 constexpr_int_to_string(I n)
68 {
69   micron::sstring<N, char> result;
70   char tmp[24];
71   usize pos = 0;
72   bool neg = false;
73   using U = micron::make_unsigned_t<I>;
74   U uval;
75   if constexpr ( micron::is_signed_v<I> ) {
76     if ( n < 0 ) {
77       neg = true;
78       uval = static_cast<U>(-(n + 1)) + 1u;
79     } else {
80       uval = static_cast<U>(n);
81     }
82   } else {
83     uval = static_cast<U>(n);
84   }
85   if ( uval == 0 ) {
86     tmp[pos++] = '0';
87   } else {
88     while ( uval > 0 ) {
89       tmp[pos++] = '0' + static_cast<char>(uval % 10);
90       uval /= 10;
91     }
92   }
93   char *out = &result[0];
94   usize out_pos = 0;
95   if ( neg ) out[out_pos++] = '-';
96   for ( usize i = pos; i > 0; --i ) out[out_pos++] = tmp[i - 1];
97   out[out_pos] = '\0';
98   result._buf_set_length(out_pos);
99   return result;
100 }

102 template<typename I, usize N = 24>
103   requires(micron::is_integral_v<I> && N >= 19)
104 constexpr micron::sstring<N, char>
105 constexpr_hex(I n, bool upper = false)
106 {
107   micron::sstring<N, char> result;
108   using U = micron::make_unsigned_t<I>;
109   U uval = static_cast<U>(n);
110   const char *digits = upper ? "0123456789ABCDEF" : "0123456789abcdef";
111   char tmp[24];
112   usize pos = 0;
113   if ( uval == 0 ) {
114     tmp[pos++] = '0';

116     while ( uval > 0 ) {
117       tmp[pos++] = digits[uval & 0xF];
118       uval >>= 4;
119     }
120   }
121   char *out = &result[0];
122   for ( usize i = pos; i > 0; --i ) out[pos - i] = tmp[i - 1];
123   out[pos] = '\0';
124   result._buf_set_length(pos);
125   return result;
126 }

156 constexpr const char *
157 u8_check(const char *str, usize n)
158 {
159   const char *end = str + n;
160   while ( str < end ) {
161     u8 c = static_cast<u8>(*str);
162     if ( c < 0x80 ) {
163       ++str;
164       continue;
165     }
166     u32 cp = 0;
167     usize seq = 0;
168     if ( (c & 0xE0) == 0xC0 ) {
169       cp = c & 0x1F;
170       seq = 2;
171     } else if ( (c & 0xF0) == 0xE0 ) {
172       cp = c & 0x0F;
173       seq = 3;
174     } else if ( (c & 0xF8) == 0xF0 ) {
175       cp = c & 0x07;
176       seq = 4;
177     } else
178       return nullptr;
179     ++str;
180     for ( usize j = 1; j < seq; ++j ) {
181       if ( str >= end ) return nullptr;
182       u8 cont = static_cast<u8>(*str);
183       if ( (cont & 0xC0) != 0x80 ) return nullptr;
184       cp = (cp << 6) | (cont & 0x3F);
185       ++str;
186     }
187     if ( seq == 2 && cp < 0x80 ) return nullptr;
188     if ( seq == 3 && cp < 0x800 ) return nullptr;
189     if ( seq == 4 && cp < 0x10000 ) return nullptr;
190     if ( cp >= 0xD800 && cp <= 0xDFFF ) return nullptr;
191     if ( cp > 0x10FFFF ) return nullptr;
192   }
193   return str;
194 }

196 constexpr const char16_t *
197 u16_check(const char16_t *str, usize n)
198 {
199   const char16_t *end = str + n;
200   while ( str < end ) {
201     u16 c = static_cast<u16>(*str);
202     ++str;
203     if ( c < 0xD800 || c > 0xDFFF ) continue;
204     if ( c >= 0xD800 && c <= 0xDBFF ) {
205       if ( str >= end ) return nullptr;
206       u16 low = static_cast<u16>(*str);
207       if ( low < 0xDC00 || low > 0xDFFF ) return nullptr;
208       ++str;
209     } else
210       return nullptr;
211   }
212   return str;
213 }

479 template<usize Sz = 80, typename C = char>
480 inline micron::sstring<Sz, C>
481 to_general_stack(f64 val, u32 precision = 6)
482 {
483   f64 abs_val = val < 0.0 ? -val : val;
484   if ( abs_val != 0.0 ) {
485     i32 exp = 0;
486     f64 tmp = abs_val;
487     while ( tmp >= 10.0 ) {
488       tmp /= 10.0;
489       ++exp;
490     }
491     while ( tmp < 1.0 ) {
492       tmp *= 10.0;
493       --exp;
494     }
495     if ( exp < -4 || exp >= static_cast<i32>(precision) ) return to_scientific_stack<Sz, C>(val, precision > 0 ? precision - 1 : 0);
496   }
497   return to_fixed_stack<Sz, C>(val, precision);
498 }

505 inline micron::sstring<Sz, C>
506 int_to_string_base_stack(I n, u32 base, bool upper = false)
507 {
508   // convert to tmp via backward-write, copy to stack string
509   char tmp[72];

511   using U = micron::make_unsigned_t<I>;

526   usize len = static_cast<usize>(tend - start);

532   result._buf_set_length(len);

642       uval = static_cast<U>(-(n + 1)) + 1u;
643     } else {
644       uval = static_cast<U>(n);
645     }
646   } else {
647     uval = static_cast<U>(n);
648   }
649   usize off = 0;
650   if ( neg ) tmp[off++] = '-';
651   char dtmp[24];
652   char *dend = dtmp + 24;
653   char *dstart = __impl::uint_to_buf_backward(dend, static_cast<u64>(uval));
654   usize dlen = static_cast<usize>(dend - dstart);
655   // raw length = off + dlen
656   usize raw_len = off + dlen;
657 
658   micron::sstring<Sz, C> result;
659   C *out = &result[0];
660   usize pos = 0;
661 
662   if ( raw_len >= width ) {
663     // no padding needed
664     if ( neg ) out[pos++] = '-';
665     for ( usize i = 0; i < dlen && pos < Sz - 1; ++i ) out[pos++] = static_cast<C>(dstart[i]);
666   } else {
667     usize pad = width - raw_len;
668     if ( neg ) out[pos++] = '-';
669     for ( usize i = 0; i < pad && pos < Sz - 1; ++i ) out[pos++] = '0';
670     for ( usize i = 0; i < dlen && pos < Sz - 1; ++i ) out[pos++] = static_cast<C>(dstart[i]);
671   }
672   out[pos] = '\0';
673   result._buf_set_length(pos);
674   return result;
675 }
