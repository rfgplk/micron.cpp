102 {
103

105   if ( ptr == nullptr ) {
106     buf[0] = '0';

123 inline constexpr usize
124 fmt_uint_to_buf(char *buf, usize buf_sz, u64 val, u32 base, bool upper)
125 {
126 
127   char tmp[72];
128   char *tend = tmp + 72;
129   char *start;
130   if ( base == 10 )
131     start = micron::__impl::uint_to_buf_backward(tend, val);
132   else
133     start = micron::__impl::uint_to_buf_base_backward(tend, val, base, upper);
134   const usize digits = static_cast<usize>(tend - start);
135   // A u64 has at most 64 digits in any supported radix (2..36).
136   if ( digits > 64 ) __builtin_unreachable();
137   const usize len = digits < buf_sz ? digits : buf_sz;
138   micron::memcpy(buf, tmp + sizeof(tmp) - digits, len);
139   return len;
140 }

190 // format spec parser: {[index][:[[fill]align][width][.precision][type]]}

192 struct fmt_spec {
193   char fill = ' ';
194   char align = '\0';      // '<' left, '>' right, '^' center
195   u32 width = 0;
196   u32 prec = 6;      // default float precision
197   bool has_prec = false;
198   char type = '\0';      // d, x, X, o, b, f, e, s, p
199   bool alt = false;      // '#' flag
200 };

202 constexpr u32 __fmt_spec_max = 1u << 24;
203 
204 constexpr inline u32
205 __spec_accum(u32 acc, char digit) noexcept
206 {
207   if ( acc > __fmt_spec_max ) return acc;
208   return acc * 10u + static_cast<u32>(digit - '0');
209 }

211 inline constexpr fmt_spec
212 parse_spec(const char *start, const char *end)
213 {
214   fmt_spec s{};
215   const char *p = start;
216   if ( p >= end ) return s;
217 
218   // fill + align
219   if ( (end - p) >= 2 && (p[1] == '<' || p[1] == '>' || p[1] == '^') ) {
220     s.fill = p[0];
221     s.align = p[1];
222     p += 2;
223   } else if ( *p == '<' || *p == '>' || *p == '^' ) {
224     s.align = *p;
225     ++p;
226   }
227 
228   if ( p < end && *p == '#' ) {
229     s.alt = true;
230     ++p;
231   }
232 
233   // width
234   while ( p < end && *p >= '0' && *p <= '9' ) {
235     s.width = __spec_accum(s.width, *p);
236     ++p;
237   }
238 
239   // precision
240   if ( p < end && *p == '.' ) {
241     ++p;
242     s.prec = 0;
243     s.has_prec = true;
244     while ( p < end && *p >= '0' && *p <= '9' ) {
245       s.prec = __spec_accum(s.prec, *p);
246       ++p;
247     }
248   }
249 
250   // type
251   if ( p < end ) {
252     s.type = *p;
253   }
254 
255   return s;
256 }

258 inline void
259 apply_padding(hstring<schar> &out, const char *content, usize content_len, const fmt_spec &spec)
260 {
261   if ( spec.width == 0 || content_len >= spec.width ) {
262     out.append(content, content_len);
263     return;
264   }
265   usize pad_total = spec.width - content_len;
266   char fill = spec.fill ? spec.fill : ' ';
267   char align = spec.align;
268 
269   // default alignment
270   if ( align == '\0' ) {
271     if ( spec.type == 's' || spec.type == '\0' )
272       align = '<';
273     else
274       align = '>';
275   }
276 
277   if ( align == '>' ) {
278     for ( usize i = 0; i < pad_total; ++i ) out += fill;
279     out.append(content, content_len);
280   } else if ( align == '<' ) {
281     out.append(content, content_len);
282     for ( usize i = 0; i < pad_total; ++i ) out += fill;
283   } else {
284     usize left = pad_total / 2;
285     usize right = pad_total - left;
286     for ( usize i = 0; i < left; ++i ) out += fill;
287     out.append(content, content_len);
288     for ( usize i = 0; i < right; ++i ) out += fill;
289   }
290 }

292 inline usize
293 fmt_u128_to_buf(char *buf, usize buf_sz, const u128 &val, const fmt_spec &spec)
294 {
295   u32 base = 10;
296   bool upper = false;
297   if ( spec.type == 'x' )
298     base = 16;
299   else if ( spec.type == 'X' ) {
300     base = 16;
301     upper = true;
302   } else if ( spec.type == 'o' )
303     base = 8;
304   else if ( spec.type == 'b' )
305     base = 2;
306 
307   usize off = 0;
308   if ( spec.alt ) {
309     if ( base == 16 && buf_sz >= 2 ) {
310       buf[0] = '0';
311       buf[1] = upper ? 'X' : 'x';
312       off = 2;
313     } else if ( base == 8 && buf_sz >= 1 ) {
314       buf[0] = '0';
315       off = 1;
316     } else if ( base == 2 && buf_sz >= 2 ) {
317       buf[0] = '0';
318       buf[1] = 'b';
319       off = 2;
320     }
321   }
322   const usize n = micron::to_chars(buf + off, buf_sz - off, val, base, upper);
323   // to_chars answers 0 for "did not fit", so `off + n` would report the base prefix alone as a
324   // complete rendering -- echof("{:#b}", ~u128(0)) printed "0b" and dropped all 128 digits.
325   // propagate the 0 instead, which is what every other writer in this layer means by failure
326   if ( n == 0 ) return 0;
327   return off + n;
328 }

1364 template<is_string T>
1365 micron::fvector<int>
1366 build_bad_char(const T &pattern)
1367 {
1368   micron::fvector<int> bad(256, -1);
1369   for ( usize i = 0; i < pattern.size(); ++i ) bad[static_cast<unsigned char>(pattern[i])] = static_cast<int>(i);
1370   return bad;
1371 }

1373 template<is_string T>
1374 micron::fvector<int>
1375 build_good_suffix(const T &pattern)
1376 {
1377   usize m = pattern.size();
1378   micron::fvector<int> shift(m + 1, 0);
1379   micron::fvector<int> border(m + 1, 0);
1380   for ( usize k = 0; k <= m; ++k ) shift[k] = static_cast<int>(m);
1381   int i = static_cast<int>(m), j = static_cast<int>(m + 1);
1382   border[static_cast<usize>(i)] = j;
1383   while ( i > 0 ) {
1384     while ( j <= static_cast<int>(m) && pattern[i - 1] != pattern[j - 1] ) {
1385       if ( shift[static_cast<usize>(j)] == static_cast<int>(m) ) shift[static_cast<usize>(j)] = j - i;
1386       j = border[static_cast<usize>(j)];
1387     }
1388     --i;
1389     --j;
1390     border[static_cast<usize>(i)] = j;
1391   }
1392   j = border[0];
1393   for ( i = 0; i <= static_cast<int>(m); ++i ) {
1394     if ( shift[static_cast<usize>(i)] == static_cast<int>(m) ) shift[static_cast<usize>(i)] = j;
1395     if ( i == j ) j = border[static_cast<usize>(j)];
1396   }
1397   return shift;
1398 }

1400 template<is_string T>
1401 micron::fvector<int>
1402 boyer_moore_search(const T &text, const T &pattern)
1403 {
1404   micron::fvector<int> result;
1405   usize n = text.size(), m = pattern.size();
1406   if ( m == 0 ) return result;
1407   result.reserve(n / (m ? m : 1));
1408   auto bad = build_bad_char(pattern);
1409   auto good = build_good_suffix(pattern);
1410   usize s = 0;
1411   while ( s <= n - m ) {
1412     int j = static_cast<int>(m - 1);
1413     while ( j >= 0 && pattern[j] == text[s + j] ) --j;
1414     if ( j < 0 ) {
1415       result.inline_push_back(static_cast<int>(s));
1416       s += good[0];
1417     } else {
1418       int bc_shift = j - bad[static_cast<unsigned char>(text[s + j])];
1419       s += micron::max(1, micron::max(bc_shift, good[j + 1]));
1420     }
1421   }
1422   return result;
1423 }

1425 template<usize Pt, usize N>
1426 constexpr micron::carray<usize, N>
1427 bm_find(const char (&text)[N], const char (&pattern)[Pt], usize &found_count)
1428 {
1429   micron::carray<int, 256> bad{};
1430   micron::carray<usize, Pt + 1> good{};
1431   micron::carray<usize, N> result{};
1432   found_count = 0;
1433   bad.fill(-1);
1434   for ( usize i = 0; i < Pt - 1; ++i ) bad[static_cast<unsigned char>(pattern[i])] = i;
1435   micron::carray<int, Pt + 1> border{};
1436   usize m = Pt - 1;
1437   int i = static_cast<int>(m), j = static_cast<int>(m + 1);
1438   border[i] = j;
1439   while ( i > 0 ) {
1440     while ( j <= static_cast<int>(m) && pattern[i - 1] != pattern[j - 1] ) j = border[j];
1441     --i;
1442     --j;
1443     border[i] = j;
1444   }
1445   j = border[0];
1446   for ( i = 0; i <= static_cast<int>(m); ++i ) {
1447     if ( good[i] == 0 ) good[i] = j;
1448     if ( i == j ) j = border[j];
1449   }
1450   usize s = 0;
1451   while ( s <= N - Pt ) {
1452     int k = Pt - 1;
1453     while ( k >= 0 && pattern[k] == text[s + k] ) --k;
1454     if ( k < 0 ) {
1455       result[found_count++] = s;
1456       s += good[0];
1457     } else {

3073 {
3074   if ( cp > 0x10FFFFu || (cp >= 0xD800u && cp <= 0xDFFFu) ) return 0;      // out of range / surrogate
3075   if ( cp < 0x80u ) {
3076     if ( buf_sz < 1 ) return 0;
3077     buf[0] = static_cast<char>(cp);
3078     return 1;
3079   }
3080   if ( cp < 0x800u ) {
3081     if ( buf_sz < 2 ) return 0;
3082     buf[0] = static_cast<char>(0xC0u | (cp >> 6));
3083     buf[1] = static_cast<char>(0x80u | (cp & 0x3Fu));

3087     if ( buf_sz < 3 ) return 0;

3090     buf[2] = static_cast<char>(0x80u | (cp & 0x3Fu));
3091     return 3;
3092   }
3093   if ( buf_sz < 4 ) return 0;
3094   buf[0] = static_cast<char>(0xF0u | (cp >> 18));
3095   buf[1] = static_cast<char>(0x80u | ((cp >> 12) & 0x3Fu));
3096   buf[2] = static_cast<char>(0x80u | ((cp >> 6) & 0x3Fu));

3407 template<typename... Args>
3408   requires(!micron::any_settling<Args...>)
3409 inline hstring<schar>
3410 format(const char *fmt, const Args &...args)
3411 {
3412   hstring<schar> out;
3413   if ( fmt == nullptr ) return out;
3414   usize auto_index = 0;
3415   const char *p = fmt;
3416   while ( *p ) {
3417     if ( *p == '{' ) {
3418       if ( *(p + 1) == '{' ) {
3419         out += '{';
3420         p += 2;
3421         continue;
3422       }
3423       ++p;
3424       const char *close = p;
3425       while ( *close && *close != '}' ) ++close;
3426       if ( *close != '}' ) break;
3427       const char *colon = p;
3428       while ( colon < close && *colon != ':' ) ++colon;
3429       usize index = auto_index;
3430       bool has_explicit_index = false;
3431       if ( colon > p ) {
3432         const char *num_p = p;
3433         bool all_digits = true;
3434         while ( num_p < colon ) {
3435           if ( *num_p < '0' || *num_p > '9' ) {
3436             all_digits = false;
3437             break;
3438           }
3439           ++num_p;
3440         }
3441         if ( all_digits ) {
3442           index = 0;
3443           num_p = p;
3444           while ( num_p < colon ) {
3445             index = index * 10 + static_cast<usize>(*num_p - '0');
3446             ++num_p;
3447           }
3448           has_explicit_index = true;
3449         }
3450       }
3451       const char *spec_start = (colon < close) ? colon + 1 : close;
3452       const char *spec_end = close;
3453       __impl::format_one(out, spec_start, spec_end, index, args...);
3454       if ( !has_explicit_index )
3455         ++auto_index;
3456       else
3457         auto_index = index + 1;
3458       p = close + 1;
3459     } else if ( *p == '}' && *(p + 1) == '}' ) {
3460       out += '}';
3461       p += 2;
3462     } else {
3463       out += *p;
3464       ++p;
3465     }
3466   }
3467   return out;
3468 }
