176   template<usize M, typename F> constexpr istring(const F (&str)[M]) : __mem(__alloc_size(M))
177   {
178     // WARNING: do NOT blind-bytecpy M*sizeof(F) bytes into an M*sizeof(T)-ELEMENT buffer:
179     // for sizeof(F) != sizeof(T) that is a cross-width OOB write

352   template<typename F>
353   usize
354   find(F ch, usize pos = 0) const
355   {
356     if ( pos >= __mem::length ) return npos;
357     const usize len = __mem::length - pos;
358     usize r;
359     if constexpr ( sizeof(T) == 1 ) {
360 #if defined(__micron_x86_avx2)
361       r = micron::simd::find_first_set_256(__mem::memory + pos, len, static_cast<char>(ch));
362 #else
363       r = micron::simd::find_first_set_128(__mem::memory + pos, len, static_cast<char>(ch));
364 #endif
365     } else {
366       r = micron::simd::find_first_elem<T>(__mem::memory + pos, len, static_cast<T>(ch));
367     }
368     return r == len ? npos : pos + r;
369   }

371   template<typename F>
372     requires(sizeof(F) == sizeof(T))
373   usize
374   find(const istring<F> &str, usize pos = 0) const
375   {
376     if ( str.empty() ) return pos;
377     if ( pos >= __mem::length ) return npos;
378     if ( str.size() > __mem::length - pos ) return npos;
379     if constexpr ( sizeof(T) != 1 ) {
380       const usize rr = micron::simd::find_substr_elem<T>(__mem::memory + pos, __mem::length - pos, reinterpret_cast<const T *>(str.cdata()),
381                                                          str.size());
382       return rr == (__mem::length - pos) ? npos : pos + rr;
383     }
384     auto *r = micron::memmem<byte>(reinterpret_cast<const byte *>(__mem::memory + pos), __mem::length - pos,
385                                    reinterpret_cast<const byte *>(str.cdata()), str.size());
386     return r == nullptr ? npos : static_cast<usize>(reinterpret_cast<const T *>(r) - __mem::memory);
387   }

740   template<typename R>
741     requires(__valid_needle<R>)
742   inline istring
743   operator-(const R &rhs) const
744   {
745     const __needle nd = __as_needle(rhs);
746     if ( nd.n == 0 || nd.n > __mem::length ) return __dup();
747     istring t(__mem::length + 1);
748     usize w = 0, r = 0;
749     while ( r < __mem::length ) {
750       if ( r + nd.n <= __mem::length
751            && micron::memcmp<byte>(reinterpret_cast<const byte *>(&__mem::memory[r]), reinterpret_cast<const byte *>(nd.p),
752                                    nd.n * sizeof(T))
753                   == 0 )
754         r += nd.n;
755       else
756         t.memory[w++] = __mem::memory[r++];
757     }
758     t.memory[w] = T{ 0 };
759     t.length = w;
760     return t;
761   };

771   template<typename I>
772     requires micron::integral<I> && (!micron::is_pointer_v<I>)
773   inline istring
774   operator*(I n) const
775   {
776     if ( n < 1 ) {
777       istring t(1);
778       t.memory[0] = T{ 0 };
779       t.length = 0;
780       return t;
781     }
782     const usize cnt = static_cast<usize>(n);
783     if ( __mem::length != 0 && cnt > static_cast<usize>(-1) / __mem::length )
784       exc<except::library_error>("micron::istring operator* size overflow");
785     const usize total = __mem::length * cnt;
786     istring t(total + 1);
787     for ( usize k = 0; k < cnt; ++k ) micron::memcpy(&t.memory[k * __mem::length], &__mem::memory[0], __mem::length);
788     t.memory[total] = T{ 0 };
789     t.length = total;
790     return t;
791   };
