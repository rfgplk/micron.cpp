52 // 32×64 to 96 bit multiply, portable split version
53 inline constexpr u32
54 mul_shift_32(u32 m, u64 factor, i32 shift)
55 {
56   u32 fLo = static_cast<u32>(factor);
57   u32 fHi = static_cast<u32>(factor >> 32);
58   u64 bits0 = static_cast<u64>(m) * fLo;
59   u64 bits1 = static_cast<u64>(m) * fHi;
60   u64 sum = (bits0 >> 32) + bits1;
61   return static_cast<u32>(sum >> (shift - 32));
62 }
63 
64 inline constexpr u32
65 mul_inv(u32 m, u32 q, i32 j)
66 {
67   return mul_shift_32(m, __pow5_inv[q], j);
68 }
69 
70 inline constexpr u32
71 mul_fwd(u32 m, u32 i, i32 j)
72 {
73   return mul_shift_32(m, __pow5[i], j);
74 }

97 inline constexpr u32
98 pow5fac(u32 v)
99 {
100   u32 c = 0;
101   while ( v > 0 ) {
102     u32 q = v / 5, r = v - 5 * q;
103     if ( r != 0 ) break;
104     v = q;
105     ++c;
106   }
107   return c;
108 }

110 inline constexpr bool
111 mul_of_5(u32 v, u32 p)
112 {
113   return pow5fac(v) >= p;

116 inline constexpr bool
117 mul_of_2(u32 v, u32 p)
118 {
119   return (v & ((1u << p) - 1)) == 0;

122 inline constexpr u32
123 dlen9(u32 v)
124 {
125   if ( v >= 100000000 ) return 9;
126   if ( v >= 10000000 ) return 8;
127   if ( v >= 1000000 ) return 7;
128   if ( v >= 100000 ) return 6;
129   if ( v >= 10000 ) return 5;
130   if ( v >= 1000 ) return 4;
131   if ( v >= 100 ) return 3;
132   if ( v >= 10 ) return 2;
133   return 1;
134 }

139 struct f2d_t {
140   u32 m;
141   i32 e;

144 inline constexpr f2d_t
145 f2d(u32 ieeeMant, u32 ieeeExp)
146 {
147   i32 e2;
148   u32 m2;
149   if ( ieeeExp == 0 ) {
150     e2 = 1 - __float_bias - static_cast<i32>(__float_mantissa_bits) - 2;
151     m2 = ieeeMant;
152   } else {
153     e2 = static_cast<i32>(ieeeExp) - __float_bias - static_cast<i32>(__float_mantissa_bits) - 2;
154     m2 = (1u << __float_mantissa_bits) | ieeeMant;
155   }
156   bool even = (m2 & 1) == 0;
157   bool acceptBounds = even;
158 
159   u32 mv = 4 * m2;
160   u32 mp = 4 * m2 + 2;
161   u32 mmShift = (ieeeMant != 0 || ieeeExp <= 1) ? 1 : 0;
162   u32 mm = 4 * m2 - 1 - mmShift;
163 
164   u32 vr, vp, vm;
165   i32 e10;
166   bool vmTZ = false, vrTZ = false;
167   u32 lastDig = 0;
168 
169   if ( e2 >= 0 ) {
170     u32 q = static_cast<u32>(log10p2(e2));
171     e10 = static_cast<i32>(q);
172     i32 k = __float_pow5_inv_bitcount + pow5bits(static_cast<i32>(q)) - 1;
173     i32 i = -e2 + static_cast<i32>(q) + k;
174     vr = mul_inv(mv, q, i);
175     vp = mul_inv(mp, q, i);
176     vm = mul_inv(mm, q, i);
177     // one removed digit must be known even when the loop below runs zero times
178     if ( q != 0 && (vp - 1) / 10 <= vm / 10 ) {
179       i32 l = __float_pow5_inv_bitcount + pow5bits(static_cast<i32>(q - 1)) - 1;
180       lastDig = mul_inv(mv, q - 1, -e2 + static_cast<i32>(q) - 1 + l) % 10;
181     }
182     if ( q <= 9 ) {
183       if ( mv % 5 == 0 ) {
184         vrTZ = mul_of_5(mv, q);
185       } else if ( acceptBounds ) {
186         vmTZ = mul_of_5(mm, q);
187       } else {
188         vp -= mul_of_5(mp, q) ? 1 : 0;
189       }
190     }
191   } else {
192     u32 q = static_cast<u32>(log10p5(-e2));
193     e10 = static_cast<i32>(q) + e2;
194     i32 i = -e2 - static_cast<i32>(q);
195     i32 k = pow5bits(i) - __float_pow5_bitcount;
196     i32 j = static_cast<i32>(q) - k;
197     vr = mul_fwd(mv, static_cast<u32>(i), j);
198     vp = mul_fwd(mp, static_cast<u32>(i), j);
199     vm = mul_fwd(mm, static_cast<u32>(i), j);
200     if ( q != 0 && (vp - 1) / 10 <= vm / 10 ) {
201       i32 j2 = static_cast<i32>(q) - 1 - (pow5bits(i + 1) - __float_pow5_bitcount);
202       lastDig = mul_fwd(mv, static_cast<u32>(i + 1), j2) % 10;
203     }
204     if ( q <= 1 ) {
205       vrTZ = true;
206       if ( acceptBounds )
207         vmTZ = (mmShift == 1);
208       else
209         --vp;
210     } else if ( q < 31 ) {
211       vrTZ = mul_of_2(mv, q - 1);
212     }
213   }
214 
215   // digit removal: find shortest representation
216   i32 removed = 0;
217   u32 output;
218 
219   if ( vmTZ || vrTZ ) {
220     while ( vp / 10 > vm / 10 ) {
221       vmTZ &= (vm % 10 == 0);
222       vrTZ &= (lastDig == 0);
223       lastDig = vr % 10;
224       vr /= 10;
225       vp /= 10;
226       vm /= 10;
227       ++removed;
228     }
229     if ( vmTZ ) {
230       while ( vm % 10 == 0 ) {
231         vrTZ &= (lastDig == 0);
232         lastDig = vr % 10;
233         vr /= 10;
234         vp /= 10;
235         vm /= 10;
236         ++removed;
237       }
238     }
239     if ( vrTZ && lastDig == 5 && vr % 2 == 0 ) lastDig = 4;
240     output = vr + ((vr == vm && !(acceptBounds && vmTZ)) || lastDig >= 5 ? 1 : 0);
241   } else {
242     bool roundUp = (lastDig >= 5);      // precomputed digit decides when no removal happens below
243     u32 vpD = vp / 100, vmD = vm / 100;
244     if ( vpD > vmD ) {
245       u32 vrD = vr / 100;
246       roundUp = (vr - 100 * vrD) >= 50;
247       vr = vrD;
248       vp = vpD;
249       vm = vmD;
250       removed += 2;
251     }
252     while ( vp / 10 > vm / 10 ) {
253       roundUp = (vr % 10) >= 5;
254       vr /= 10;
255       vp /= 10;
256       vm /= 10;
257       ++removed;
258     }
259     output = vr + (vr == vm || roundUp ? 1 : 0);
260   }
261 
262   return { output, e10 + removed };
263 }

266 // f2s_buffered
267 // buf must have at least 16 bytes returns length (not null-terminated)
268 
269 inline constexpr usize
270 f2s_buffered(f32 val, char *buf)
271 {
272   const u32 bits = micron::math::ieee::to_bits<f32>(val);
273 
274   bool sign = (bits >> 31) != 0;
275   u32 ieeeExp = (bits >> __float_mantissa_bits) & ((1u << __float_exponent_bits) - 1);
276   u32 ieeeMant = bits & ((1u << __float_mantissa_bits) - 1);
277 
278   usize idx = 0;
279 
280   // NaN
281   if ( ieeeExp == 0xFF && ieeeMant != 0 ) {
282     buf[0] = 'N';
283     buf[1] = 'a';
284     buf[2] = 'N';
285     return 3;
286   }
287   // Inf
288   if ( ieeeExp == 0xFF ) {
289     if ( sign ) buf[idx++] = '-';
290     buf[idx++] = 'I';
291     buf[idx++] = 'n';
292     buf[idx++] = 'f';
293     return idx;
294   }
295   // zero
296   if ( ieeeExp == 0 && ieeeMant == 0 ) {
297     if ( sign ) buf[idx++] = '-';
298     buf[idx++] = '0';
299     buf[idx++] = 'E';
300     buf[idx++] = '0';
301     return idx;
302   }
303 
304   if ( sign ) buf[idx++] = '-';
305 
306   f2d_t v = f2d(ieeeMant, ieeeExp);
307   u32 olength = dlen9(v.m);
308 
309   // write mantissa digits in reverse
310   char dtmp[10];
311   u32 dpos = 0;
312   u32 tmp = v.m;
313   while ( tmp > 0 ) {
314     dtmp[dpos++] = '0' + static_cast<char>(tmp % 10);
315     tmp /= 10;
316   }
317 
318   // format: first_digit.remaining E exponent
319   buf[idx++] = dtmp[dpos - 1];
320   if ( dpos > 1 ) {
321     buf[idx++] = '.';
322     for ( u32 j = dpos - 1; j-- > 0; ) buf[idx++] = dtmp[j];
323   }
324 
325   i32 exp10 = v.e + static_cast<i32>(olength) - 1;
326   buf[idx++] = 'E';
327   if ( exp10 < 0 ) {
328     buf[idx++] = '-';
329     exp10 = -exp10;
330   }
331   if ( exp10 >= 10 ) buf[idx++] = '0' + static_cast<char>(exp10 / 10);
332   buf[idx++] = '0' + static_cast<char>(exp10 % 10);
333 
334   return idx;
335 }

350 template<typename C, typename Fn>
351 inline micron::hstring<C>
352 __fp_to_hstring(usize need, Fn &&emit)
353 {
354   if ( need <= __impl::__fpconv::__fmt_fixed_max ) {
355     char buf[__impl::__fpconv::__fmt_fixed_max];
356     const usize n = emit(buf, sizeof(buf));
357     if ( n == 0 ) return micron::hstring<C>();
358     return micron::hstring<C>(buf, buf + n);
359   }
360   micron::hstring<C> s(need + 1);
361   const usize n = emit(reinterpret_cast<char *>(s.data()), need + 1);
362   s.set_size(n);
363   return s;
364 }

429 template<typename C = char>
430 inline micron::hstring<C>
431 to_general(f64 val, u32 precision = 6)
432 {
433   // derive sciExp directly from the selected shortest-decimal decomposition
434   const u64 __b = micron::math::ieee::to_bits<f64>(val);
435   u64 ieeeMantissa = __b & ((1ull << 52) - 1);
436   u32 ieeeExponent = static_cast<u32>((__b >> 52) & 0x7FF);
437 
438   // specials & zero pass through to_fixed
439   if ( ieeeExponent == 0x7FF || (ieeeExponent == 0 && ieeeMantissa == 0) ) return to_fixed<C>(val, precision);
440 
441 #if defined(MICRON_USE_RYU)
442   const __impl::__ryu::decimal64 dec = __impl::__ryu::d2d(ieeeMantissa, ieeeExponent);
443   const u32 olength = __impl::__ryu::decimalLength(dec.mantissa);
444   const i32 exp10 = dec.exponent + static_cast<i32>(olength) - 1;
445 #else
446   const __impl::__zmij::decimal_fp dec = __impl::__zmij::to_decimal(val);
447   const u32 olength = __impl::__ryu::decimalLength(dec.sig);
448   const i32 exp10 = dec.exp + static_cast<i32>(olength) - 1;
449 #endif
450 
451   if ( exp10 < -4 || exp10 >= static_cast<i32>(precision) ) return to_scientific<C>(val, precision > 0 ? precision - 1 : 0);
452   return to_fixed<C>(val, precision);
453 }
