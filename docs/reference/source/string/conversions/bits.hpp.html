25   inline constexpr __fmt_uint128_t
26   umul128(u64 a, u64 b)
27   {
28     unsigned __int128 r = static_cast<unsigned __int128>(a) * b;
29     return { static_cast<u64>(r), static_cast<u64>(r >> 64) };
30   }

32   // no 128-bit
33   inline constexpr __fmt_uint128_t
34   umul128(u64 a, u64 b)
35   {
36     u64 aHi = a >> 32, aLo = a & 0xFFFFFFFF;
37     u64 bHi = b >> 32, bLo = b & 0xFFFFFFFF;
38 
39     u64 p00 = aLo * bLo;
40     u64 p01 = aLo * bHi;
41     u64 p10 = aHi * bLo;
42     u64 p11 = aHi * bHi;
43 
44     u64 mid = (p00 >> 32) + (p01 & 0xFFFFFFFF) + (p10 & 0xFFFFFFFF);
45 
46     return { (p00 & 0xFFFFFFFF) | (mid << 32), p11 + (p01 >> 32) + (p10 >> 32) + (mid >> 32) };
47   }
48 #endif

50   [[gnu::always_inline]] inline constexpr u64
51   umul128_add_hi64(u64 a, u64 b, u64 c)
52   {
53 #if defined(__SIZEOF_INT128__)
54     if !consteval {
55       const unsigned __int128 p = static_cast<unsigned __int128>(a) * static_cast<unsigned __int128>(b) + c;
56       return static_cast<u64>(p >> 64);
57     }
58 #endif
59     const __fmt_uint128_t p = umul128(a, b);
60     return p.hi + (p.lo + c < p.lo);
61   }

63   inline constexpr u64
64   shiftright128(u64 lo, u64 hi, u32 dist)
65   {
66     if ( dist == 0 ) return lo;
67     if ( dist >= 64 ) return hi >> (dist - 64);
68     return (lo >> dist) | (hi << (64 - dist));
69   }

71   inline constexpr u64
72   shiftleft128(u64 lo, u64 hi, u32 dist)
73   {
74     if ( dist == 0 ) return hi;
75     if ( dist < 64 )
76       return (hi << dist) | (lo >> (64 - dist));
77     else
78       return lo << (dist - 64);
79   }

236   inline constexpr __attribute__((always_inline)) void
237   emit8_backward(char *buf, u32 v)
238   {
239     // v < 100000000
240     u32 hi4 = v / 10000;
241     u32 lo4 = v - hi4 * 10000;
242 
243     u32 d0 = lo4 / 100;
244     u32 d1 = lo4 - d0 * 100;
245     buf[7] = __digit_tbl[d1].d[1];
246     buf[6] = __digit_tbl[d1].d[0];
247     buf[5] = __digit_tbl[d0].d[1];
248     buf[4] = __digit_tbl[d0].d[0];
249 
250     u32 d2 = hi4 / 100;
251     u32 d3 = hi4 - d2 * 100;
252     buf[3] = __digit_tbl[d3].d[1];
253     buf[2] = __digit_tbl[d3].d[0];
254     buf[1] = __digit_tbl[d2].d[1];
255     buf[0] = __digit_tbl[d2].d[0];
256   }

258   inline constexpr __attribute__((always_inline)) char *
259   emit_block_backward(char *end, u32 v)
260   {
261     // not leading block emit exactly 8 digits
262     // leading block emit without leading zeros
263     if ( v == 0 ) {
264       *--end = '0';
265       return end;
266     }
267 
268     while ( v >= 100 ) {
269       u32 r = v % 100;
270       v /= 100;
271       *--end = __digit_tbl[r].d[1];
272       *--end = __digit_tbl[r].d[0];
273     }
274     if ( v >= 10 ) {
275       *--end = __digit_tbl[v].d[1];
276       *--end = __digit_tbl[v].d[0];
277     } else {
278       *--end = static_cast<char>('0' + v);
279     }
280     return end;
281   }

285   inline constexpr char *
286   uint_to_buf_backward(char *end, u64 val)
287   {
288     if ( val == 0 ) {
289       *--end = '0';
290       return end;
291     }
292 
293     if ( val < 100000000ull ) {
294       return emit_block_backward(end, static_cast<u32>(val));
295     }
296 
297     if ( val < 10000000000000000ull ) {
298       u64 q = fast_div1e8(val);
299       u32 lo = fast_mod1e8(val, q);
300 
301       emit8_backward(end - 8, lo);
302       end -= 8;
303 
304       return emit_block_backward(end, static_cast<u32>(q));
305     }
306 
307     u64 q1 = fast_div1e8(val);
308     u32 lo = fast_mod1e8(val, q1);
309 
310     u64 q2 = fast_div1e8(q1);
311     u32 mid = fast_mod1e8(q1, q2);
312 
313     emit8_backward(end - 8, lo);
314     end -= 8;
315 
316     emit8_backward(end - 8, mid);
317     end -= 8;
318 
319     return emit_block_backward(end, static_cast<u32>(q2));
320   }

322   inline constexpr char *
323   uint_to_buf_base_backward(char *end, u64 val, u32 base, bool upper)
324   {
325     if ( base < 2 || base > 36 ) base = 10;
326     if ( val == 0 ) {
327       *--end = '0';
328       return end;
329     }
330 
331     const char *digits = upper ? "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ" : "0123456789abcdefghijklmnopqrstuvwxyz";
332 
333     if ( base == 16 ) {
334       const char *hex = upper ? __hex_upper : __hex_lower;
335       while ( val > 0 ) {
336         *--end = hex[val & 0xF];
337         val >>= 4;
338       }
339       return end;
340     }
341 
342     if ( base == 8 ) {
343       while ( val > 0 ) {
344         *--end = static_cast<char>('0' + (val & 7));
345         val >>= 3;
346       }
347       return end;
348     }
349 
350     if ( base == 2 ) {
351       while ( val > 0 ) {
352         *--end = static_cast<char>('0' + (val & 1));
353         val >>= 1;
354       }
355       return end;
356     }
357 
358     if ( base == 4 ) {
359       while ( val > 0 ) {
360         *--end = digits[val & 3];
361         val >>= 2;
362       }
363       return end;
364     }
365 
366     if ( base == 32 ) {
367       while ( val > 0 ) {
368         *--end = digits[val & 31];
369         val >>= 5;
370       }
371       return end;
372     }
373 
374     while ( val > 0 ) {
375       *--end = digits[val % base];
376       val /= base;
377     }
378     return end;
379   }

414   inline constexpr u32
415   __div128_small(u32 (&l)[4], u32 d) noexcept
416   {
417     u64 rem = 0;
418     for ( i32 i = 3; i >= 0; --i ) {
419       const u64 cur = (rem << 32) | static_cast<u64>(l[i]);
420       l[i] = static_cast<u32>(cur / d);
421       rem = cur % d;
422     }
423     return static_cast<u32>(rem);
424   }

432   inline constexpr char *
433   uint128_to_buf_backward(char *end, u64 hi, u64 lo)
434   {
435     if ( hi == 0 ) return uint_to_buf_backward(end, lo);
436 
437     u32 l[4] = { static_cast<u32>(lo), static_cast<u32>(lo >> 32), static_cast<u32>(hi), static_cast<u32>(hi >> 32) };
438     char *p = end;
439     while ( __nz128(l) ) {
440       u32 r = __div128_small(l, 1000000000u);
441       const bool more = __nz128(l);
442       if ( more ) {
443         for ( u32 k = 0; k < 9; ++k ) {      // interior chunks keep their leading zeros
444           *--p = static_cast<char>('0' + r % 10u);
445           r /= 10u;
446         }
447       } else {
448         do {
449           *--p = static_cast<char>('0' + r % 10u);
450           r /= 10u;
451         } while ( r != 0 );
452       }
453     }
454     return p;
455   }

457   inline constexpr char *
458   uint128_to_buf_base_backward(char *end, u64 hi, u64 lo, u32 base, bool upper)
459   {
460     if ( base == 10 ) return uint128_to_buf_backward(end, hi, lo);
461     if ( hi == 0 ) return uint_to_buf_base_backward(end, lo, base, upper);
462     if ( base < 2 || base > 36 ) base = 10;
463 
464     const char *digits = upper ? "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ" : "0123456789abcdefghijklmnopqrstuvwxyz";
465     char *p = end;
466 
467     // a power-of-two base is pure shift and mask over the pair, no division at all
468     u32 sh = 0;
469     switch ( base ) {
470     case 2:
471       sh = 1;
472       break;
473     case 4:
474       sh = 2;
475       break;
476     case 8:
477       sh = 3;
478       break;
479     case 16:
480       sh = 4;
481       break;
482     case 32:
483       sh = 5;
484       break;
485     default:
486       sh = 0;
487       break;
488     }
489     if ( sh != 0 ) {
490       const u64 mask = (1ull << sh) - 1;
491       u64 h = hi, o = lo;
492       while ( h != 0 || o != 0 ) {
493         *--p = digits[o & mask];
494         o = (o >> sh) | (h << (64 - sh));
495         h >>= sh;
496       }
497       return p;
498     }
499 
500     u32 l[4] = { static_cast<u32>(lo), static_cast<u32>(lo >> 32), static_cast<u32>(hi), static_cast<u32>(hi >> 32) };
501     while ( __nz128(l) ) {
502       const u32 r = __div128_small(l, base);
503       *--p = digits[r];
504     }
505     return p;
506   }

601     return static_cast<i32>((static_cast<u64>(static_cast<u32>(e)) * 732923ull) >> 20);

604   // divisibility by 5 == v * 0xCCCCCCCCCCCCCCCD <= floor(2^64 / 5)
605   inline constexpr u32
606   pow5_factor(u64 v)
607   {
608     // lots of magic here
609     // 2^64 / 5 = 3689348814741910323
610     constexpr u64 inv5 = 0xCCCCCCCCCCCCCCCDull;
611     constexpr u64 thresh5 = 0x3333333333333333ull;      // floor(2^64 / 5)
612     u32 count = 0;
613     while ( true ) {
614       u64 q = v * inv5;
615       if ( q > thresh5 ) break;
616       v = q;
617       ++count;
618     }
619     return count;
620   }

634   inline constexpr void
635   pow5_compute(u32 i, u64 result[2])
636   {
637     const u32 base = i / 26;
638     const u32 base2 = base * 26;
639     const u64 *mul = __pow5_split2[base];
640 
641     if ( i == base2 ) {
642       result[0] = mul[0];
643       result[1] = mul[1];
644       return;
645     }
646 
647     u32 offset = i - base2;
648     u64 m = __pow5_table[offset];
649 
650     __fmt_uint128_t b0 = umul128(m, mul[0]);
651     __fmt_uint128_t b2 = umul128(m, mul[1]);
652 
653     u64 mid = b0.hi + b2.lo;
654     u64 high = b2.hi + (mid < b0.hi ? 1ull : 0ull);
655 
656     u32 delta = pow5bits(i) - pow5bits(base2);
657     u32 corr = (__pow5_offsets[i / 16] >> ((i % 16) << 1)) & 3;
658 
659     result[0] = shiftleft128(b0.lo, mid, 64 - delta) + corr;
660     result[1] = shiftleft128(mid, high, 64 - delta);
661   }
662 
663   inline constexpr void
664   pow5_compute_inv(u32 i, u64 result[2])
665   {
666     const u32 base = (i + 25) / 26;
667     const u32 base2 = base * 26;
668     const u64 *mul = __pow5_inv_split2[base];
669 
670     if ( i == base2 ) {
671       result[0] = mul[0];
672       result[1] = mul[1];
673       return;
674     }
675 
676     u32 offset = base2 - i;
677     u64 m = __pow5_table[offset];
678 
679     // anchors already store ceil()+1; derive from the floor form (mul[0]-1), the +1 is re-added below
680     __fmt_uint128_t b0 = umul128(m, mul[0] - 1);
681     __fmt_uint128_t b2 = umul128(m, mul[1]);
682 
683     u64 mid = b0.hi + b2.lo;
684     u64 high = b2.hi + (mid < b0.hi ? 1ull : 0ull);
685 
686     u32 delta = pow5bits(base2) - pow5bits(i);
687     u32 corr = (__pow5_inv_offsets[i / 16] >> ((i % 16) << 1)) & 3;
688 
689     result[0] = shiftleft128(b0.lo, mid, 64 - delta) + corr + 1;
690     result[1] = shiftleft128(mid, high, 64 - delta);
691   }

693   inline constexpr u64
694   shift_mul64(u64 m, const u64 mul[2], i32 j)
695   {
696     __fmt_uint128_t b0 = umul128(m, mul[0]);
697     __fmt_uint128_t b2 = umul128(m, mul[1]);
698 
699     u64 mid = b0.hi + b2.lo;
700     u64 hi = b2.hi + (mid < b0.hi ? 1ull : 0ull);
701 
702     return shiftright128(mid, hi, static_cast<u32>(j) - 64);
703   }

710   inline constexpr decimal64
711   d2d(u64 ieeeMantissa, u32 ieeeExponent)
712   {
713     i32 e2;
714     u64 m2;
715 
716     if ( ieeeExponent == 0 ) {
717       // -2 absorbs the factor of 4 in mv = 4*m2
718       e2 = 1 - __bias - static_cast<i32>(__mantissa_bits) - 2;
719       m2 = ieeeMantissa;
720     } else {
721       e2 = static_cast<i32>(ieeeExponent) - __bias - static_cast<i32>(__mantissa_bits) - 2;
722       m2 = (1ull << __mantissa_bits) | ieeeMantissa;
723     }
724 
725     bool even = (m2 & 1) == 0;
726     bool acceptBounds = even;
727 
728     u64 mv = m2 << 2;
729     u32 mmShift = (ieeeMantissa != 0 || ieeeExponent <= 1) ? 1 : 0;
730 
731     u64 vr, vp, vm;
732     i32 e10;
733     bool trailingVr = false, trailingVm = false;
734 
735     if ( e2 >= 0 ) {
736       // positive exponent
737       i32 q = log10_pow2(e2) - (e2 > 3 ? 1 : 0);
738       e10 = q;
739       i32 k = static_cast<i32>(__pow5_inv_bitcount + pow5bits(static_cast<u32>(q))) - 1;
740       i32 i = -e2 + q + k;
741 
742       u64 pow5[2];
743       pow5_compute_inv(static_cast<u32>(q), pow5);
744 
745       vr = shift_mul64(mv, pow5, i);
746       vp = shift_mul64(mv + 2, pow5, i);
747       vm = shift_mul64(mv - 1 - mmShift, pow5, i);
748 
749       if ( q <= 21 ) {
750         if ( mv % 5 == 0 )
751           trailingVr = pow5_multiple(mv, static_cast<u32>(q));
752         else if ( acceptBounds )
753           trailingVm = pow5_multiple(mv - 1 - mmShift, static_cast<u32>(q));
754         else
755           vp -= pow5_multiple(mv + 2, static_cast<u32>(q)) ? 1 : 0;
756       }
757     } else {
758       // negative exponent
759       i32 q = log10_pow5(-e2) - (-e2 > 1 ? 1 : 0);
760       e10 = q + e2;
761       i32 i = -e2 - q;
762       i32 k = static_cast<i32>(pow5bits(static_cast<u32>(i))) - static_cast<i32>(__pow5_bitcount);
763       i32 j = q - k;
764 
765       u64 pow5[2];
766       pow5_compute(static_cast<u32>(i), pow5);
767 
768       vr = shift_mul64(mv, pow5, j);
769       vp = shift_mul64(mv + 2, pow5, j);
770       vm = shift_mul64(mv - 1 - mmShift, pow5, j);
771 
772       if ( q <= 1 ) {
773         trailingVr = true;
774         if ( acceptBounds )
775           trailingVm = (mmShift == 1);
776         else
777           --vp;      // mp = mv + 2 always has a trailing zero bit
778       } else if ( q < 63 ) {
779         trailingVr = pow2_multiple(mv, static_cast<u32>(q));
780       }
781     }
782 
783     // 3. Remove trailing digits & round-to-even
784     i32 removed = 0;
785     u8 lastRemoved = 0;
786 
787     if ( trailingVm || trailingVr ) {
788       for ( ;; ) {
789         u64 vpD = vp / 10, vmD = vm / 10;
790         if ( vpD <= vmD ) break;
791         u64 vrD = vr / 10;
792         u32 vrM = static_cast<u32>(vr - 10 * vrD);
793         trailingVm &= (vm - 10 * vmD == 0);
794         trailingVr &= (lastRemoved == 0);
795         lastRemoved = static_cast<u8>(vrM);
796         vr = vrD;
797         vp = vpD;
798         vm = vmD;
799         ++removed;
800       }
801       if ( trailingVm ) {
802         for ( ;; ) {
803           u64 vmD = vm / 10;
804           if ( vm - 10 * vmD != 0 ) break;
805           u64 vpD = vp / 10, vrD = vr / 10;
806           u32 vrM = static_cast<u32>(vr - 10 * vrD);
807           trailingVr &= (lastRemoved == 0);
808           lastRemoved = static_cast<u8>(vrM);
809           vr = vrD;
810           vp = vpD;
811           vm = vmD;
812           ++removed;
813         }
814       }
815       if ( trailingVr && lastRemoved == 5 && (vr & 1) == 0 ) lastRemoved = 4;
816 
817       u64 out = vr + ((vr == vm && (!acceptBounds || !trailingVm)) || lastRemoved >= 5 ? 1ull : 0ull);
818       return { out, e10 + removed };
819     } else {
820       bool roundUp = false;
821       for ( ;; ) {
822         u64 vpD = vp / 10, vmD = vm / 10;
823         if ( vpD <= vmD ) break;
824         u64 vrD = vr / 10;
825         u32 vrM = static_cast<u32>(vr - 10 * vrD);
826         roundUp = (vrM >= 5);
827         vr = vrD;
828         vp = vpD;
829         vm = vmD;
830         ++removed;
831       }
832       u64 out = vr + ((vr == vm || roundUp) ? 1ull : 0ull);
833       return { out, e10 + removed };
834     }
835   }

837   inline constexpr u32
838   decimalLength(u64 v)
839   {
840     // floor(log10(2^(63-clz))) + correction
841     // powers of 10 for boundary checks
842     constexpr static const u64 __pow10[20] = { 1ull,
843                                                10ull,
844                                                100ull,
845                                                1000ull,
846                                                10000ull,
847                                                100000ull,
848                                                1000000ull,
849                                                10000000ull,
850                                                100000000ull,
851                                                1000000000ull,
852                                                10000000000ull,
853                                                100000000000ull,
854                                                1000000000000ull,
855                                                10000000000000ull,
856                                                100000000000000ull,
857                                                1000000000000000ull,
858                                                10000000000000000ull,
859                                                100000000000000000ull,
860                                                1000000000000000000ull,
861                                                10000000000000000000ull };
862     // log10 approximation from bit count floor(bits * 77/ 256) = floor(log10(2^bits))
863     // may be off by one
864     u32 bits = 64 - clz64(v);
865     u32 approx = (bits * 77) >> 8;
866     return approx + (v >= __pow10[approx] ? 1 : 0);
867   }

869   inline constexpr usize
870   d2s_buffered(f64 value, char *buf)
871   {
872     // NOT a union: reading the inactive member is a hard error in a constant expression, and
873     // ieee::to_bits is __builtin_bit_cast, which is both constexpr and exactly as free at run time
874     const u64 bits = micron::math::ieee::to_bits<f64>(value);
875 
876     bool sign = (bits >> 63) != 0;
877     u64 ieeeMantissa = bits & ((1ull << 52) - 1);
878     u32 ieeeExponent = static_cast<u32>((bits >> 52) & 0x7FF);
879 
880     usize pos = 0;
881 
882     if ( ieeeExponent == 0x7FF ) {
883       if ( ieeeMantissa ) {
884         buf[0] = 'N';
885         buf[1] = 'a';
886         buf[2] = 'N';
887         return 3;
888       }
889       if ( sign ) buf[pos++] = '-';
890       buf[pos] = 'I';
891       buf[pos + 1] = 'n';
892       buf[pos + 2] = 'f';
893       return pos + 3;
894     }
895 
896     if ( ieeeExponent == 0 && ieeeMantissa == 0 ) {
897       if ( sign ) buf[pos++] = '-';
898       buf[pos] = '0';
899       buf[pos + 1] = '.';
900       buf[pos + 2] = '0';
901       return pos + 3;
902     }
903 
904     if ( sign ) buf[pos++] = '-';
905 
906     decimal64 dec = d2d(ieeeMantissa, ieeeExponent);
907     u64 output = dec.mantissa;
908     i32 exp = dec.exponent;
909     u32 olength = decimalLength(output);
910 
911     i32 sciExp = exp + static_cast<i32>(olength) - 1;
912 
913     char digits[20];
914     char *dend = digits + 20;
915     char *dstart = uint_to_buf_backward(dend, output);
916     usize dlen = static_cast<usize>(dend - dstart);
917 
918     if ( sciExp >= -3 && sciExp <= 7 ) {
919       if ( exp >= 0 ) {
920         for ( usize i = 0; i < dlen; ++i ) buf[pos++] = dstart[i];
921         for ( i32 i = 0; i < exp; ++i ) buf[pos++] = '0';
922         buf[pos++] = '.';
923         buf[pos++] = '0';
924       } else if ( exp + static_cast<i32>(olength) > 0 ) {
925         i32 intDigits = static_cast<i32>(olength) + exp;
926         for ( i32 i = 0; i < intDigits; ++i ) buf[pos++] = dstart[i];
927         buf[pos++] = '.';
928         for ( usize i = static_cast<usize>(intDigits); i < dlen; ++i ) buf[pos++] = dstart[i];
929       } else {
930         buf[pos++] = '0';
931         buf[pos++] = '.';
932         i32 leadingZeros = -(exp + static_cast<i32>(olength));
933         for ( i32 i = 0; i < leadingZeros; ++i ) buf[pos++] = '0';
934         for ( usize i = 0; i < dlen; ++i ) buf[pos++] = dstart[i];
935       }
936     } else {
937       buf[pos++] = dstart[0];
938       if ( dlen > 1 ) {
939         buf[pos++] = '.';
940         for ( usize i = 1; i < dlen; ++i ) buf[pos++] = dstart[i];
941       }
942       buf[pos++] = 'e';
943       if ( sciExp >= 0 ) {
944         buf[pos++] = '+';
945       } else {
946         buf[pos++] = '-';
947         sciExp = -sciExp;
948       }
949       if ( sciExp >= 100 ) {
950         buf[pos++] = static_cast<char>('0' + sciExp / 100);
951         sciExp %= 100;
952         buf[pos++] = __digit_tbl[sciExp].d[0];
953         buf[pos++] = __digit_tbl[sciExp].d[1];
954       } else if ( sciExp >= 10 ) {
955         buf[pos++] = __digit_tbl[sciExp].d[0];
956         buf[pos++] = __digit_tbl[sciExp].d[1];
957       } else {
958         buf[pos++] = static_cast<char>('0' + sciExp);
959       }
960     }
961 
962     return pos;
963   }
