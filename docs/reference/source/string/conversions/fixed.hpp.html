27 struct parts {
28   u64 m2 = 0;      // full significand, implicit bit restored
29   i32 e2 = 0;      // value == m2 * 2^e2
30   bool neg = false;
31   bool is_zero = false;
32   bool is_inf = false;
33   bool is_nan = false;
34 };

36 inline constexpr parts
37 __decompose(f64 v) noexcept
38 {
39   const u64 bits = micron::math::ieee::to_bits<f64>(v);
40   const u32 be = static_cast<u32>((bits >> 52) & 0x7FFu);
41   const u64 mf = bits & ((1ull << 52) - 1);
42 
43   parts p{};
44   p.neg = (bits >> 63) != 0;
45   if ( be == 0x7FFu ) {
46     p.is_inf = (mf == 0);
47     p.is_nan = (mf != 0);
48     return p;
49   }
50   if ( be == 0 ) {
51     if ( mf == 0 ) {
52       p.is_zero = true;
53       return p;
54     }
55     p.m2 = mf;      // subnormal: no implicit bit
56     p.e2 = 1 - 1023 - 52;
57     return p;
58   }
59   p.m2 = mf | (1ull << 52);
60   p.e2 = static_cast<i32>(be) - 1023 - 52;
61   return p;
62 }

105 inline constexpr fixed64
106 __fixed_u64(u64 m2, i32 e2, u32 p) noexcept
107 {
108   fixed64 r{};
109   if ( p > __max_u64_prec ) return r;
110 
111   u64 ip = 0, frac = 0;
112   u32 t = 0;
113   if ( e2 >= 0 ) {
114     if ( e2 >= 64 ) return r;      // |v| >= 2^64
115     if ( e2 > 0 && (m2 >> (64 - static_cast<u32>(e2))) != 0 ) return r;
116     ip = m2 << e2;
117   } else {
118     t = static_cast<u32>(-e2);
119     if ( t >= 64 ) {
120       ip = 0;
121       frac = m2;      // the whole significand is fractional
122     } else {
123       ip = m2 >> t;
124       frac = m2 & ((1ull << t) - 1);
125     }
126   }
127 
128   u64 fp = 0;
129   bool up = false;
130   if ( frac != 0 && p > 0 ) {
131     if ( t <= p ) {
132       // fp = frac * 5^p * 2^(p-t)
133       fp = frac * __pow5_u64[p];
134       fp <<= (p - t);
135     } else {
136       const u32 u = t - p;
137       if ( u >= 99 ) {
138         fp = 0;      // product < 2^98 <= 2^(u-1): strictly below half, rounds down
139       } else {
140         const __fmt_uint128_t prod = umul128(frac, __pow5_u64[p]);
141         const u64 lo = prod.lo;      // a plain {lo,hi} struct on every arch, not a builtin
142         const u64 hi = prod.hi;
143         u64 rem_hi = 0, rem_lo = 0;
144         if ( u < 64 ) {
145           fp = (u == 0) ? lo : ((hi << (64 - u)) | (lo >> u));
146           rem_lo = (u == 0) ? 0ull : (lo & ((1ull << u) - 1));
147         } else {
148           const u32 s = u - 64;
149           fp = (s == 0) ? hi : (hi >> s);
150           rem_hi = (s == 0) ? 0ull : (hi & ((1ull << s) - 1));
151           rem_lo = lo;
152         }
153         // half == 2^(u-1); compare the remainder against it
154         if ( u < 64 ) {
155           const u64 half = 1ull << (u - 1);
156           up = (rem_lo > half) || (rem_lo == half && ((p == 0 ? ip : fp) & 1) != 0);
157         } else {
158           const u32 s = u - 64;
159           // half sits at bit (u-1): in rem_hi when s > 0, else the top bit of rem_lo
160           if ( s == 0 ) {
161             const u64 half = 1ull << 63;
162             up = (rem_lo > half) || (rem_lo == half && ((p == 0 ? ip : fp) & 1) != 0);
163           } else {
164             const u64 half_hi = 1ull << (s - 1);
165             up = (rem_hi > half_hi) || (rem_hi == half_hi && rem_lo != 0)
166                  || (rem_hi == half_hi && rem_lo == 0 && ((p == 0 ? ip : fp) & 1) != 0);
167           }
168         }
169       }
170     }
171   } else if ( frac != 0 && p == 0 ) {
172     // rounding the fraction away entirely: compare frac against half of 2^t
173     if ( t < 64 ) {
174       const u64 half = 1ull << (t - 1);
175       up = (frac > half) || (frac == half && (ip & 1) != 0);
176     } else {
177       // t >= 64 means |v| < 1 and frac == m2 < 2^53 < 2^(t-1): below half
178       up = false;
179     }
180   }
181 
182   if ( up ) ++fp;
183   if ( p > 0 && fp == __pow10_u64[p] ) {      // the fraction carried into the integer
184     fp = 0;
185     if ( ip == ~0ull ) return r;
186     ++ip;
187   } else if ( p == 0 && up ) {
188     if ( ip == ~0ull ) return r;
189     ++ip;
190   }
191 
192   r.ip = ip;
193   r.fp = fp;
194   r.ok = true;
195   return r;
196 }

198 inline constexpr usize
199 __emit_fixed64(char *buf, usize buf_sz, bool neg, u64 ip, u64 fp, u32 p) noexcept
200 {
201   char itmp[24];
202   char *iend = itmp + 24;
203   char *ist = uint_to_buf_backward(iend, ip);
204   const usize ilen = static_cast<usize>(iend - ist);
205 
206   const usize need = (neg ? 1u : 0u) + ilen + (p > 0 ? 1u + static_cast<usize>(p) : 0u);
207   if ( buf_sz < need ) return 0;
208 
209   usize pos = 0;
210   if ( neg ) buf[pos++] = '-';
211   for ( usize i = 0; i < ilen; ++i ) buf[pos++] = ist[i];
212   if ( p > 0 ) {
213     buf[pos++] = '.';
214     char ftmp[24];
215     char *fend = ftmp + 24;
216     char *fst = uint_to_buf_backward(fend, fp);
217     const usize flen = static_cast<usize>(fend - fst);
218     for ( usize i = flen; i < p; ++i ) buf[pos++] = '0';      // left-pad to exactly p places
219     for ( usize i = 0; i < flen; ++i ) buf[pos++] = fst[i];
220   }
221   return pos;
222 }

231 [[gnu::noinline, gnu::cold]] inline constexpr void
232 __exact_round(u64 m2, i32 e2, cut mode, i32 arg, char *dig, u32 &n, i32 &pt) noexcept
233 {
234   micron::__impl::__dec::__decimal v;
235   micron::__impl::__dec::__dec_load_u64(v, m2);
236   micron::__impl::__dec::__dec_shift(v, e2);
237 
238   pt = v.point;
239   n = 0;
240   if ( v.ndigits == 0 ) return;      // exact zero
241 
242   const i32 keep = (mode == cut::significant) ? arg : v.point + arg;
243   if ( keep < 0 ) return;      // every digit is past the cut: rounds to zero
244 
245   u32 k = static_cast<u32>(keep);
246   if ( k >= v.ndigits ) {      // nothing is cut, so nothing rounds
247     for ( u32 i = 0; i < v.ndigits; ++i ) dig[i] = static_cast<char>('0' + v.d[i]);
248     n = v.ndigits;
249     return;
250   }
251 
252   const bool up = micron::__impl::__dec::__dec_round_up(v, k);
253   for ( u32 i = 0; i < k; ++i ) dig[i] = static_cast<char>('0' + v.d[i]);
254   n = k;
255   if ( !up ) return;
256 
257   for ( u32 i = k; i-- > 0; ) {
258     if ( dig[i] != '9' ) {
259       ++dig[i];
260       return;
261     }
262     dig[i] = '0';
263   }
264   // the carry escaped the leading digit: 0.999.. -> 0.100.. * 10^(pt+1)
265   dig[0] = '1';
266   if ( n == 0 ) n = 1;
267   ++pt;
268 }

273 inline constexpr usize
274 __emit_special(const parts &p, char *buf, usize buf_sz) noexcept
275 {
276   if ( p.is_nan ) {
277     if ( buf_sz < 3 ) return 0;
278     buf[0] = 'N';
279     buf[1] = 'a';
280     buf[2] = 'N';
281     return 3;
282   }
283   // inf
284   const usize need = p.neg ? 4u : 3u;
285   if ( buf_sz < need ) return 0;
286   usize pos = 0;
287   if ( p.neg ) buf[pos++] = '-';
288   buf[pos++] = 'I';
289   buf[pos++] = 'n';
290   buf[pos++] = 'f';
291   return pos;
292 }

294 inline constexpr usize
295 __emit_fixed(char *buf, usize buf_sz, bool neg, const char *dig, u32 n, i32 pt, u32 prec) noexcept
296 {
297   const usize idig = (pt > 0) ? static_cast<usize>(pt) : 1u;
298   const usize need = (neg ? 1u : 0u) + idig + (prec > 0 ? 1u + static_cast<usize>(prec) : 0u);
299   if ( buf_sz < need ) return 0;
300 
301   usize pos = 0;
302   if ( neg ) buf[pos++] = '-';
303   if ( pt <= 0 )
304     buf[pos++] = '0';
305   else
306     for ( i32 i = 0; i < pt; ++i ) buf[pos++] = (static_cast<u32>(i) < n) ? dig[i] : '0';
307 
308   if ( prec > 0 ) {
309     buf[pos++] = '.';
310     for ( u32 j = 0; j < prec; ++j ) {
311       const i32 idx = pt + static_cast<i32>(j);
312       buf[pos++] = (idx >= 0 && static_cast<u32>(idx) < n) ? dig[idx] : '0';
313     }
314   }
315   return pos;
316 }

318 inline constexpr usize
319 __emit_sci(char *buf, usize buf_sz, bool neg, const char *dig, u32 n, i32 sci_exp, u32 prec) noexcept
320 {
321   u32 edig = 2;
322   u32 ae = static_cast<u32>(sci_exp < 0 ? -sci_exp : sci_exp);
323   if ( ae >= 100 ) edig = 3;
324   const usize need = (neg ? 1u : 0u) + 1u + (prec > 0 ? 1u + static_cast<usize>(prec) : 0u) + 2u + edig;
325   if ( buf_sz < need ) return 0;
326 
327   usize pos = 0;
328   if ( neg ) buf[pos++] = '-';
329   buf[pos++] = (n > 0) ? dig[0] : '0';
330   if ( prec > 0 ) {
331     buf[pos++] = '.';
332     for ( u32 j = 1; j <= prec; ++j ) buf[pos++] = (j < n) ? dig[j] : '0';
333   }
334   buf[pos++] = 'e';
335   buf[pos++] = (sci_exp < 0) ? '-' : '+';
336   if ( edig == 3 ) {
337     buf[pos++] = static_cast<char>('0' + ae / 100);
338     ae %= 100;
339   }
340   buf[pos++] = __digit_tbl[ae].d[0];
341   buf[pos++] = __digit_tbl[ae].d[1];
342   return pos;
343 }

350 inline constexpr usize
351 d2f_size(f64 val, u32 precision) noexcept
352 {
353   const __fx::parts p = __fx::__decompose(val);
354   if ( p.is_nan ) return 3;
355   if ( p.is_inf ) return p.neg ? 4u : 3u;
356 
357   // integer digits, from the values binary exponent
358   usize idig = 1;
359   if ( !p.is_zero ) {
360     const i32 bexp = p.e2 + static_cast<i32>(64u - micron::__impl::clz64(p.m2));
361     if ( bexp > 0 ) idig = static_cast<usize>(log10_pow2(bexp)) + 1u;
362   }
363   return (p.neg ? 1u : 0u) + idig + (precision > 0 ? 1u + static_cast<usize>(precision) : 0u);
364 }

375 inline constexpr usize
376 d2f_buffered(f64 val, char *buf, usize buf_sz, u32 precision)
377 {
378   const __fx::parts p = __fx::__decompose(val);
379   if ( p.is_nan || p.is_inf ) return __fx::__emit_special(p, buf, buf_sz);
380   if ( p.is_zero ) return __fx::__emit_fixed(buf, buf_sz, p.neg, nullptr, 0, 0, precision);
381 
382   const __fx::fixed64 k = __fx::__fixed_u64(p.m2, p.e2, precision);
383   if ( k.ok ) return __fx::__emit_fixed64(buf, buf_sz, p.neg, k.ip, k.fp, precision);
384 
385   char dig[__fx::__dig_cap];
386   u32 n = 0;
387   i32 pt = 0;
388   __fx::__exact_round(p.m2, p.e2, __fx::cut::fraction, static_cast<i32>(precision), dig, n, pt);
389   return __fx::__emit_fixed(buf, buf_sz, p.neg, dig, n, pt, precision);
390 }

392 inline constexpr usize
393 d2e_buffered(f64 val, char *buf, usize buf_sz, u32 precision)
394 {
395   const __fx::parts p = __fx::__decompose(val);
396   if ( p.is_nan || p.is_inf ) return __fx::__emit_special(p, buf, buf_sz);
397   if ( p.is_zero ) return __fx::__emit_sci(buf, buf_sz, p.neg, nullptr, 0, 0, precision);
398 
399   char dig[__fx::__dig_cap];
400   u32 n = 0;
401   i32 pt = 0;
402   // %e keeps precision+1 significant digits

404   return __fx::__emit_sci(buf, buf_sz, p.neg, dig, n, pt - 1, precision);

409 inline constexpr usize
410 __d2g_core(f64 val, char *buf, usize buf_sz, u32 precision, bool alt, bool upper)
411 {
412   const __fx::parts p = __fx::__decompose(val);
413   if ( p.is_nan || p.is_inf ) return __fx::__emit_special(p, buf, buf_sz);
414 
415   const u32 P = (precision == 0) ? 1u : precision;
416 
417   char dig[__fx::__dig_cap];
418   u32 n = 0;
419   i32 pt = 0;
420   if ( !p.is_zero ) __fx::__exact_round(p.m2, p.e2, __fx::cut::significant, static_cast<i32>(P), dig, n, pt);
421   const i32 X = p.is_zero ? 0 : (pt - 1);
422 
423   usize len = 0;
424   if ( X >= -4 && X < static_cast<i32>(P) )
425     len = __fx::__emit_fixed(buf, buf_sz, p.neg, dig, n, pt, static_cast<u32>(static_cast<i32>(P) - 1 - X));
426   else
427     len = __fx::__emit_sci(buf, buf_sz, p.neg, dig, n, X, P - 1u);
428   if ( len == 0 ) return 0;
429 
430   if ( !alt ) {
431     usize dot = len, epos = len;
432     for ( usize i = 0; i < len; ++i ) {
433       if ( buf[i] == '.' )
434         dot = i;
435       else if ( buf[i] == 'e' ) {
436         epos = i;
437         break;
438       }
439     }
440     if ( dot != len ) {
441       usize end = epos;
442       while ( end > dot + 1 && buf[end - 1] == '0' ) --end;
443       if ( end == dot + 1 ) end = dot;
444       if ( end != epos ) {
445         usize w = end;
446         for ( usize i = epos; i < len; ++i ) buf[w++] = buf[i];      // pull the exponent down
447         len = w;
448       }
449     }
450   }
451   if ( upper )
452     for ( usize i = 0; i < len; ++i )
453       if ( buf[i] == 'e' ) buf[i] = 'E';
454   return len;
455 }

474 // %%%%%%%%%%%%%%%%%%%
475 // %a
476 // (exact hex float)
477 //
478 // glibc-compatible
479 //   1.0 -> 0x1p+0    pi -> 0x1.921fb54442d11p+1    -0.0 -> -0x0p+0    0.0 -> 0x0p+0
480 inline constexpr usize
481 d2a_buffered(f64 val, char *buf, usize buf_sz, u32 precision, bool has_prec, bool upper)
482 {
483   const u64 bits = micron::math::ieee::to_bits<f64>(val);
484   const u32 be = static_cast<u32>((bits >> 52) & 0x7FFu);
485   const u64 mf = bits & ((1ull << 52) - 1);
486   const bool neg = (bits >> 63) != 0;
487 
488   if ( be == 0x7FFu ) {
489     __fx::parts sp{};
490     sp.neg = neg;
491     sp.is_nan = (mf != 0);
492     sp.is_inf = (mf == 0);
493     return __fx::__emit_special(sp, buf, buf_sz);
494   }
495 
496   u32 lead = 0;
497   i32 exp2 = 0;
498   if ( be == 0 ) {
499     lead = 0;
500     exp2 = (mf == 0) ? 0 : -1022;
501   } else {
502     lead = 1;
503     exp2 = static_cast<i32>(be) - 1023;
504   }
505 
506   u8 nib[13] = {};
507   for ( u32 i = 0; i < 13; ++i ) nib[i] = static_cast<u8>((mf >> (48u - 4u * i)) & 0xFull);
508 
509   u32 ndig = 13;
510   if ( has_prec ) {
511     if ( precision < 13u ) {
512       const u32 shift = 52u - 4u * precision;
513       const u64 disc = mf & ((1ull << shift) - 1);
514       const u64 half = 1ull << (shift - 1);
515       const u32 last = (precision == 0) ? lead : nib[precision - 1];
516       if ( disc > half || (disc == half && (last & 1u) != 0u) ) {
517         u32 i = precision;
518         bool carry = true;
519         while ( carry && i-- > 0 ) {
520           if ( nib[i] == 0xFu )
521             nib[i] = 0;
522           else {
523             ++nib[i];
524             carry = false;
525           }
526         }
527         if ( carry ) ++lead;      // 0x1.fff.. -> 0x2.000..
528       }
529     }
530     ndig = precision;
531   } else {
532     while ( ndig > 0 && nib[ndig - 1] == 0 ) --ndig;      // trim, and drop a bare '.'
533   }
534 
535   const char *tbl = upper ? __hex_upper : __hex_lower;
536   u32 ae = static_cast<u32>(exp2 < 0 ? -exp2 : exp2);
537   u32 edig = 1;
538   for ( u32 t = ae; t >= 10u; t /= 10u ) ++edig;
539 
540   const usize need = (neg ? 1u : 0u) + 2u + 1u + (ndig > 0 ? 1u + static_cast<usize>(ndig) : 0u) + 1u + 1u + edig;
541   if ( buf_sz < need ) return 0;
542 
543   usize pos = 0;
544   if ( neg ) buf[pos++] = '-';
545   buf[pos++] = '0';
546   buf[pos++] = upper ? 'X' : 'x';
547   buf[pos++] = static_cast<char>('0' + lead);
548   if ( ndig > 0 ) {
549     buf[pos++] = '.';
550     for ( u32 i = 0; i < ndig; ++i ) buf[pos++] = (i < 13u) ? tbl[nib[i]] : '0';
551   }
552   buf[pos++] = upper ? 'P' : 'p';
553   buf[pos++] = (exp2 < 0) ? '-' : '+';
554   {
555     char tmp[8];
556     u32 t = edig;
557     u32 v = ae;
558     while ( t-- > 0 ) {
559       tmp[t] = static_cast<char>('0' + v % 10u);
560       v /= 10u;
561     }
562     for ( u32 i = 0; i < edig; ++i ) buf[pos++] = tmp[i];
563   }
564   return pos;
565 }

568 template<typename F>
569 inline usize
570 x2a_buffered(F val, char *buf, usize buf_sz)
571 {
572   if constexpr ( sizeof(F) <= 8 ) {
573     return d2a_buffered(static_cast<f64>(val), buf, buf_sz, 0, false, false);
574   } else {
575     unsigned char raw[sizeof(F)] = {};
576     __builtin_memcpy(raw, &val, sizeof(F));      // every arch micron targets is little-endian
577 
578     u64 lo = 0, hi = 0;
579     for ( u32 i = 0; i < 8; ++i ) lo |= static_cast<u64>(raw[i]) << (8u * i);
580 
581     u32 se = 0;
582     i32 shift = 0;
583     bool nan_bit = false;
584 #if defined(__micron_ldbl_x87_80)
585     if constexpr ( micron::is_same_v<F, long double> ) {
586       se = static_cast<u32>(raw[8]) | (static_cast<u32>(raw[9]) << 8);
587       shift = 63;
588       nan_bit = (lo & 0x7FFFFFFFFFFFFFFFull) != 0;
589     } else
590 #endif
591     {
592       for ( u32 i = 8; i < 16 && i < sizeof(F); ++i ) hi |= static_cast<u64>(raw[i]) << (8u * (i - 8u));
593       se = static_cast<u32>(hi >> 48);
594       hi &= 0x0000FFFFFFFFFFFFull;
595       shift = 112;
596       nan_bit = (hi | lo) != 0;
597       if ( (se & 0x7FFFu) != 0 && (se & 0x7FFFu) != 0x7FFFu ) hi |= (1ull << 48);      // implicit bit
598     }
599 
600     const bool neg = (se & 0x8000u) != 0;
601     const u32 e = se & 0x7FFFu;
602 
603     if ( e == 0x7FFFu ) {
604       __fx::parts sp{};
605       sp.neg = neg;
606       sp.is_nan = nan_bit;
607       sp.is_inf = !nan_bit;
608       return __fx::__emit_special(sp, buf, buf_sz);
609     }
610 
611     i32 exp2 = (e == 0 ? 1 : static_cast<i32>(e)) - 16383 - shift;
612 
613     if ( (hi | lo) == 0 ) {
614       const usize need = (neg ? 1u : 0u) + 6u;
615       if ( buf_sz < need ) return 0;
616       usize z = 0;
617       if ( neg ) buf[z++] = '-';
618       buf[z++] = '0';
619       buf[z++] = 'x';
620       buf[z++] = '0';
621       buf[z++] = 'p';
622       buf[z++] = '+';
623       buf[z++] = '0';
624       return z;
625     }
626 
627     while ( (lo & 0xFull) == 0 ) {      // strip trailing zero nibbles, exact
628       lo = (lo >> 4) | (hi << 60);
629       hi >>= 4;
630       exp2 += 4;
631     }
632 
633     char tmp[40];
634     char *tend = tmp + 40;
635     char *start = micron::__impl::uint128_to_buf_base_backward(tend, hi, lo, 16u, false);
636     const usize hlen = static_cast<usize>(tend - start);
637 
638     u32 ae = static_cast<u32>(exp2 < 0 ? -exp2 : exp2);
639     u32 edig = 1;
640     for ( u32 t = ae; t >= 10u; t /= 10u ) ++edig;
641 
642     const usize need = (neg ? 1u : 0u) + 2u + hlen + 2u + edig;
643     if ( buf_sz < need ) return 0;
644 
645     usize pos = 0;
646     if ( neg ) buf[pos++] = '-';
647     buf[pos++] = '0';
648     buf[pos++] = 'x';
649     for ( char *q = start; q != tend; ++q ) buf[pos++] = *q;
650     buf[pos++] = 'p';
651     buf[pos++] = (exp2 < 0) ? '-' : '+';
652     {
653       char et[8];
654       u32 t = edig;
655       u32 v = ae;
656       while ( t-- > 0 ) {
657         et[t] = static_cast<char>('0' + v % 10u);
658         v /= 10u;
659       }
660       for ( u32 i = 0; i < edig; ++i ) buf[pos++] = et[i];
661     }
662     return pos;
663   }

667 inline constexpr usize
668 d2f_trim_buffered(f64 val, char *buf, usize buf_sz, u32 precision)
669 {
670   usize n = d2f_buffered(val, buf, buf_sz, precision);
671   usize dot = n;
672   for ( usize i = 0; i < n; ++i ) {
673     if ( buf[i] == '.' ) {
674       dot = i;
675       break;
676     }
677   }
678   usize end = n;
679   if ( dot != n ) {                                            // has a fractional part -> trim it
680     while ( end > dot + 1 && buf[end - 1] == '0' ) --end;      // strip trailing zeros
681     if ( end == dot + 1 ) end = dot;                           // drop a bare trailing '.'
682   }
683   if ( end >= 1 && buf[0] == '-' ) {
684     bool __all_zero = true;
685     for ( usize i = 1; i < end; ++i )
686       if ( buf[i] != '0' && buf[i] != '.' ) {
687         __all_zero = false;
688         break;
689       }
690     if ( __all_zero ) {
691       for ( usize i = 1; i < end; ++i ) buf[i - 1] = buf[i];      // shift left over the leading '-'
692       --end;
693     }
694   }
695   return end;
696 }
