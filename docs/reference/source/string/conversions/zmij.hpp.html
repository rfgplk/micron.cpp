47 inline constexpr i32
48 __compute_dec_exp(i32 bin_exp, bool regular = true) noexcept
49 {
50   constexpr i32 log10_3_over_4_sig = 131072;
51   constexpr i32 log10_2_sig = 315653;
52   constexpr i32 log10_2_exp = 20;
53   return (bin_exp * log10_2_sig - static_cast<i32>(!regular) * log10_3_over_4_sig) >> log10_2_exp;

112 struct pow10_significand_table {
113 #if defined(__OPTIMIZE_SIZE__)
114   static constexpr bool compressed = true;
115 #else
116   static constexpr bool compressed = false;
117 #endif
118   static constexpr i32 count = 649;
119   uint128 table[compressed ? 1 : count]{};
120 
121   static constexpr uint128
122   compute(u32 i) noexcept
123   {
124     constexpr u32 stride = 28;
125     const u64 m = __pow10_minor[(i + 24) % stride];
126     const uint128 h = __pow10_major[(i + 24) / stride];
127     const u64 h1 = umul128(h.lo, m).hi;
128     const u64 c0 = h.lo * m;
129     const u64 c1 = h1 + h.hi * m;
130     const u64 c2 = (c1 < h1) + umul128(h.hi, m).hi;
131     uint128 r = (c2 >> 63) != 0 ? uint128{ c2, c1 } : uint128{ c2 << 1 | c1 >> 63, c1 << 1 | c0 >> 63 };
132     r.lo -= (__pow10_fixups[i >> 5] >> (i & 31)) & 1u;
133     return r;
134   }
135 
136   constexpr pow10_significand_table() noexcept
137   {
138     if constexpr ( !compressed )
139       for ( i32 i = 0; i < count; ++i ) table[i] = compute(static_cast<u32>(i));
140   }
141 
142   constexpr uint128
143   operator[](i32 dec_exp) const noexcept
144   {
145     const u32 i = static_cast<u32>(dec_exp + 307);
146     if constexpr ( compressed ) return compute(i);
147     return table[i];
148   }

211 template<typename F, typename U>
212 inline constexpr shortest_decimal
213 __to_decimal(U bin_sig, i32 raw_exp, bool regular) noexcept
214 {
215   using T = traits<F>;
216   i32 bin_exp = raw_exp - T::exp_offset;
217   constexpr i32 extra_shift = exp_shift_table::extra_shift;
218 
219   if ( !regular ) {
220     const i32 dec_exp = __compute_dec_exp(bin_exp, false);
221     const u8 shift = static_cast<u8>(__compute_exp_shift(bin_exp, dec_exp + 1) + extra_shift);
222     const uint128 pow10 = __data.pow10_significands[-dec_exp - 1];
223     const uint128 p = __umul192_hi128(pow10.hi, pow10.lo, static_cast<u64>(bin_sig) << shift);
224     u64 integral = p.hi >> extra_shift;
225     const u64 fractional = p.hi << (64 - extra_shift) | p.lo >> extra_shift;
226     const u64 half_ulp = pow10.hi >> (extra_shift + 1 - shift);
227     const bool round_up = half_ulp > ~u64(0) - fractional;
228     const bool round_down = (half_ulp >> 1) > fractional;
229     integral += round_up;
230     i32 digit = static_cast<i32>(umul128_add_hi64(fractional, 10, (1ull << 63) - 1));
231     const i32 lo = static_cast<i32>(umul128_add_hi64(fractional - (half_ulp >> 1), 10, ~u64(0)));
232     if ( digit < lo ) digit = lo;
233     return { integral, dec_exp, digit, (round_up + round_down) == 0 };
234   }
235 
236   const i32 dec_exp = __compute_dec_exp(bin_exp);
237   u8 shift = exp_shift_table::enabled ? __data.exp_shifts.table[bin_exp + traits<f64>::exp_offset]
238                                       : static_cast<u8>(__compute_exp_shift(bin_exp, dec_exp + 1) + extra_shift);
239   const u64 even = 1 - (static_cast<u64>(bin_sig) & 1);
240 
241   if constexpr ( T::num_bits == 32 ) {
242     constexpr i32 float_shift = 34;
243     shift = static_cast<u8>(shift + float_shift - extra_shift);
244     const u64 pow10_hi = __data.pow10_significands[-dec_exp - 1].hi;
245     const u64 p = umul128(pow10_hi + 1, static_cast<u64>(bin_sig) << shift).hi;
246     u64 integral = p >> float_shift;
247     const u64 fractional = p & ((1ull << float_shift) - 1);
248     const u64 half_ulp = (pow10_hi >> (65 - shift)) + even;
249     const bool round_up = (fractional + half_ulp) >> float_shift;
250     const bool round_down = half_ulp > fractional;
251     integral += round_up;
252     i32 digit = static_cast<i32>((fractional * 10 + (1ull << (float_shift - 1))) >> float_shift);
253     if ( fractional == (1ull << (float_shift - 2)) ) digit = 2;
254     return { integral, dec_exp, digit, (round_up + round_down) == 0 };
255   } else {
256     const uint128 pow10 = __data.pow10_significands[-dec_exp - 1];
257     const uint128 p = __umul192_hi128(pow10.hi, pow10.lo, static_cast<u64>(bin_sig) << shift);
258     u64 integral = p.hi >> extra_shift;
259     const u64 fractional = p.hi << (64 - extra_shift) | p.lo >> extra_shift;
260     const u64 half_ulp = (pow10.hi >> (extra_shift + 1 - shift)) + even;
261     const bool round_up = fractional + half_ulp < fractional;
262     const bool round_down = half_ulp > fractional;
263     integral += round_up;
264     i32 digit = static_cast<i32>(umul128_add_hi64(fractional, 10, __data.biased_half));
265     if ( fractional == (1ull << 62) ) digit = 2;
266     return { integral, dec_exp, digit, (round_up + round_down) == 0 };
267   }

278 template<typename F, typename U>
279 inline constexpr decimal_fp
280 __to_decimal_untrimmed_bits(U bits, i32 &last_digit, bool &has_last_digit) noexcept
281 {
282   using T = traits<F>;
283   i32 bin_exp = static_cast<i32>((bits >> T::num_sig_bits) & static_cast<U>(T::exp_mask));
284   U bin_sig = bits & (T::implicit_bit - 1);
285   const bool negative = (bits >> (T::num_bits - 1)) != 0;
286   if ( bin_exp == 0 || bin_exp == T::exp_mask ) {
287     if ( bin_exp != 0 ) return { static_cast<u64>(bin_sig), 0, negative, true, bin_sig != 0 };
288     if ( bin_sig == 0 ) return { 0, 0, negative, false, false };
289     bin_exp = 1;
290     bin_sig |= T::implicit_bit;
291   }
292   const shortest_decimal dec = __to_decimal<F>(bin_sig ^ T::implicit_bit, bin_exp, bin_sig != 0);
293   last_digit = dec.last_digit;
294   has_last_digit = dec.has_last_digit;
295   return { dec.sig, dec.exp, negative, false, false };
296 }

332 inline __avx_uint128
333 __avx_umul128(__m256i x, __m256i y) noexcept
334 {
335   namespace av = micron::simd::avx2;
336   const __m256i mask32 = av::splat_i64(0xffffffffull);
337   const __m256i x_hi = av::shr_i64(x, 32);
338   const __m256i y_hi = av::shr_i64(y, 32);
339   const __m256i p00 = av::mul_2x32_to_2x64_u(x, y);
340   const __m256i p10 = av::mul_2x32_to_2x64_u(x_hi, y);
341   const __m256i p01 = av::mul_2x32_to_2x64_u(x, y_hi);
342   const __m256i p11 = av::mul_2x32_to_2x64_u(x_hi, y_hi);
343   const __m256i middle = av::add_i64(av::shr_i64(p00, 32), av::add_i64(av::and_i256(p10, mask32), av::and_i256(p01, mask32)));
344   const __m256i lo = av::or_i256(av::and_i256(p00, mask32), av::shl_i64(middle, 32));
345   const __m256i hi = av::add_i64(p11, av::add_i64(av::shr_i64(p10, 32), av::add_i64(av::shr_i64(p01, 32), av::shr_i64(middle, 32))));
346   return { hi, lo };

436 inline bool
437 __to_decimal4_avx2(const f32 *values, decimal_fp *dec, i32 *last_digit, bool *has_last_digit) noexcept
438 {
439   namespace av = micron::simd::avx2;
440   if constexpr ( !exp_shift_table::enabled ) return false;
441   const u32 bits0 = micron::math::ieee::to_bits<f32>(values[0]);
442   const u32 bits1 = micron::math::ieee::to_bits<f32>(values[1]);
443   const u32 bits2 = micron::math::ieee::to_bits<f32>(values[2]);
444   const u32 bits3 = micron::math::ieee::to_bits<f32>(values[3]);
445   const u32 raw0 = bits0 >> 23 & 0xffu;
446   const u32 raw1 = bits1 >> 23 & 0xffu;
447   const u32 raw2 = bits2 >> 23 & 0xffu;
448   const u32 raw3 = bits3 >> 23 & 0xffu;
449   constexpr u32 sig_mask = (1u << 23) - 1;
450   const u32 sig0 = bits0 & sig_mask;
451   const u32 sig1 = bits1 & sig_mask;
452   const u32 sig2 = bits2 & sig_mask;
453   const u32 sig3 = bits3 & sig_mask;
454   const bool exceptional = ((raw0 - 1u) >= 254u) | ((raw1 - 1u) >= 254u) | ((raw2 - 1u) >= 254u) | ((raw3 - 1u) >= 254u) | (sig0 == 0)
455                            | (sig1 == 0) | (sig2 == 0) | (sig3 == 0);
456   if ( exceptional ) return false;
457 
458   const i32 bin0 = static_cast<i32>(raw0) - 150;
459   const i32 bin1 = static_cast<i32>(raw1) - 150;
460   const i32 bin2 = static_cast<i32>(raw2) - 150;
461   const i32 bin3 = static_cast<i32>(raw3) - 150;
462   const i32 exp0 = __compute_dec_exp(bin0);
463   const i32 exp1 = __compute_dec_exp(bin1);
464   const i32 exp2 = __compute_dec_exp(bin2);
465   const i32 exp3 = __compute_dec_exp(bin3);
466   const u8 shift0 = static_cast<u8>(__data.exp_shifts.table[raw0 + 925] + 25);
467   const u8 shift1 = static_cast<u8>(__data.exp_shifts.table[raw1 + 925] + 25);
468   const u8 shift2 = static_cast<u8>(__data.exp_shifts.table[raw2 + 925] + 25);
469   const u8 shift3 = static_cast<u8>(__data.exp_shifts.table[raw3 + 925] + 25);
470   const u64 pow0 = __data.pow10_significands[-exp0 - 1].hi + 1;
471   const u64 pow1 = __data.pow10_significands[-exp1 - 1].hi + 1;
472   const u64 pow2 = __data.pow10_significands[-exp2 - 1].hi + 1;
473   const u64 pow3 = __data.pow10_significands[-exp3 - 1].hi + 1;
474 
475   const __m256i sig = __avx_set_u64(sig0 | 1u << 23, sig1 | 1u << 23, sig2 | 1u << 23, sig3 | 1u << 23);
476   const __m256i shifts = __avx_set_u64(shift0, shift1, shift2, shift3);
477   const __m256i product = __avx_umul128(__avx_set_u64(pow0, pow1, pow2, pow3), av::shl_per_i64(sig, shifts)).hi;
478   const __m256i fractional = av::and_i256(product, av::splat_i64((1ull << 34) - 1));
479   const __m256i even = av::sub_i64(av::splat_i64(1), av::and_i256(sig, av::splat_i64(1)));
480   const __m256i half
481       = av::add_i64(av::shr_per_i64(__avx_set_u64(pow0 - 1, pow1 - 1, pow2 - 1, pow3 - 1), av::sub_i64(av::splat_i64(65), shifts)), even);
482   const __m256i round_up = av::shr_i64(av::add_i64(fractional, half), 34);
483   const __m256i round_down_mask = __avx_lt_u64(fractional, half);
484   const __m256i integral = av::add_i64(av::shr_i64(product, 34), round_up);
485   __m256i digit = av::shr_i64(av::add_i64(__avx_umul128(fractional, av::splat_i64(10)).lo, av::splat_i64(1ull << 33)), 34);
486   digit = av::blendv_i8(digit, av::splat_i64(2), av::eq_i64(fractional, av::splat_i64(1ull << 32)));
487 
488   dec[0] = { __avx_extract_u64<0>(integral), exp0, (bits0 >> 31) != 0, false, false };
489   dec[1] = { __avx_extract_u64<1>(integral), exp1, (bits1 >> 31) != 0, false, false };
490   dec[2] = { __avx_extract_u64<2>(integral), exp2, (bits2 >> 31) != 0, false, false };
491   dec[3] = { __avx_extract_u64<3>(integral), exp3, (bits3 >> 31) != 0, false, false };
492   last_digit[0] = static_cast<i32>(__avx_extract_u64<0>(digit));
493   last_digit[1] = static_cast<i32>(__avx_extract_u64<1>(digit));
494   last_digit[2] = static_cast<i32>(__avx_extract_u64<2>(digit));
495   last_digit[3] = static_cast<i32>(__avx_extract_u64<3>(digit));
496   has_last_digit[0] = (__avx_extract_u64<0>(round_up) | __avx_extract_u64<0>(round_down_mask)) == 0;
497   has_last_digit[1] = (__avx_extract_u64<1>(round_up) | __avx_extract_u64<1>(round_down_mask)) == 0;
498   has_last_digit[2] = (__avx_extract_u64<2>(round_up) | __avx_extract_u64<2>(round_down_mask)) == 0;
499   has_last_digit[3] = (__avx_extract_u64<3>(round_up) | __avx_extract_u64<3>(round_down_mask)) == 0;
500   return true;
501 }

532 template<typename F>
533 inline constexpr decimal_fp
534 to_decimal(F value) noexcept
535 {
536   i32 last_digit = 0;
537   bool has_last_digit = false;
538   decimal_fp result = __to_decimal_untrimmed(value, last_digit, has_last_digit);
539   if ( result.nonfinite || (result.sig == 0 && !has_last_digit) ) return result;
540   u64 sig = result.sig * 10 + static_cast<u64>(has_last_digit ? last_digit : 0);
541   i32 exp = result.exp;
542   while ( sig % 10 == 0 ) {
543     sig /= 10;
544     ++exp;
545   }
546   result.sig = sig;
547   result.exp = exp;
548   return result;
549 }

578   } else if ( exp >= 10 ) {

585 inline constexpr usize
586 __emit_d2s(decimal_fp dec, i32 last_digit, bool has_last_digit, char *buf) noexcept
587 {
588   if ( dec.nonfinite ) return __emit_special(dec, buf);
589   usize pos = 0;
590   if ( dec.negative ) buf[pos++] = '-';
591   if ( dec.sig == 0 && !has_last_digit ) {
592     buf[pos++] = '0';
593     buf[pos++] = '.';
594     buf[pos++] = '0';
595     return pos;
596   }
597 
598   if ( has_last_digit )
599     dec.sig = dec.sig * 10 + static_cast<u64>(last_digit);
600   else
601     ++dec.exp;
602   char digits[18];
603   __digits18(dec.sig, digits);
604   usize len = __ryu::decimalLength(dec.sig);
605   char *start = digits + 18 - len;
606   while ( len > 1 && start[len - 1] == '0' ) {
607     --len;
608     ++dec.exp;
609   }
610   i32 exp = dec.exp;
611   i32 sci_exp = exp + static_cast<i32>(len) - 1;
612 
613   if ( sci_exp >= -3 && sci_exp <= 7 ) {
614     if ( exp >= 0 ) {
615       for ( usize i = 0; i < len; ++i ) buf[pos++] = start[i];
616       for ( i32 i = 0; i < exp; ++i ) buf[pos++] = '0';
617       buf[pos++] = '.';
618       buf[pos++] = '0';
619     } else if ( exp + static_cast<i32>(len) > 0 ) {
620       const i32 whole = static_cast<i32>(len) + exp;
621       for ( i32 i = 0; i < whole; ++i ) buf[pos++] = start[i];
622       buf[pos++] = '.';
623       for ( usize i = static_cast<usize>(whole); i < len; ++i ) buf[pos++] = start[i];
624     } else {
625       buf[pos++] = '0';
626       buf[pos++] = '.';
627       for ( i32 i = 0; i < -(exp + static_cast<i32>(len)); ++i ) buf[pos++] = '0';
628       for ( usize i = 0; i < len; ++i ) buf[pos++] = start[i];
629     }
630     return pos;
631   }
632 
633   buf[pos++] = start[0];
634   if ( len > 1 ) {
635     buf[pos++] = '.';
636     for ( usize i = 1; i < len; ++i ) buf[pos++] = start[i];
637   }
638   buf[pos++] = 'e';
639   if ( sci_exp >= 0 )
640     buf[pos++] = '+';
641   else {
642     buf[pos++] = '-';
643     sci_exp = -sci_exp;
644   }
645   return __emit_exp_digits(buf, pos, static_cast<u32>(sci_exp));
646 }

786 inline constexpr u64
787 __scale(u64 bin_sig, i32 bin_exp, i32 dec_exp) noexcept
788 {
789   constexpr i32 shift = 11;
790   const i32 point_shift = shift - __compute_exp_shift(bin_exp, dec_exp);
791   const uint128 pow10 = __data.pow10_significands[-dec_exp];
792   const uint128 p = __umul192_hi128(pow10.hi, pow10.lo + (dec_exp < -55 || dec_exp > 0), bin_sig << shift);
793   const u64 integral = p.hi >> point_shift;
794   const u64 half = p.hi >> (point_shift - 1) & 1;
795   const u64 tail = (p.hi & ((1ull << (point_shift - 1)) - 1)) | p.lo;
796   return integral << 2 | half << 1 | (tail != 0);
797 }

799 inline constexpr u64
800 __round_even(u64 x) noexcept
801 {
802   return (x + 1 + ((x >> 2) & 1)) >> 2;
803 }
804 
805 inline constexpr precision_decimal
806 __to_precision(u64 bin_sig, i32 bin_exp, i32 precision) noexcept
807 {
808   i32 dec_exp = __compute_dec_exp(bin_exp + 52) - (precision - 1);
809   const u64 scaled = __scale(bin_sig, bin_exp, dec_exp);
810   u64 dec_sig = __round_even(scaled);
811   if ( dec_sig >= __pow10s[precision] ) {
812     dec_sig = __round_even(scaled / 10 | (scaled & 1) | (scaled % 10 != 0));
813     ++dec_exp;
814   }
815   return { dec_sig * __pow10s[18 - precision], dec_exp + precision - 1 };
816 }

818 inline constexpr void
819 __digits18(u64 value, char (&digits)[18]) noexcept
820 {
821   if !consteval {

823     const u64 hi = value / 10000000000000000ull;
824     digits[0] = static_cast<char>('0' + hi / 10);
825     digits[1] = static_cast<char>('0' + hi % 10);
826     micron::simd::sse::decimal_digits_16(digits + 2, value - hi * 10000000000000000ull);
827     return;
828 #elif (defined(__micron_arch_arm32) || defined(__micron_arch_arm64)) && defined(__micron_arm_neon)
829     const u64 hi = value / 10000000000000000ull;
830     digits[0] = static_cast<char>('0' + hi / 10);
831     digits[1] = static_cast<char>('0' + hi % 10);
832     micron::simd::neon::decimal_digits_16(digits + 2, value - hi * 10000000000000000ull);
833     return;
834 #endif
835   }
836   for ( i32 i = 18; i-- > 0; ) {
837     digits[i] = static_cast<char>('0' + value % 10);
838     value /= 10;
839   }
840 }

842 inline constexpr char *
843 __digits_n(u64 value, u32 count, char (&digits)[18]) noexcept
844 {
845   char *const end = digits + 18;
846   if ( count <= 8 ) {
847     micron::__impl::emit8_backward(end - 8, static_cast<u32>(value));
848   } else if ( count <= 16 ) {
849     const u64 hi = micron::__impl::fast_div1e8(value);
850     micron::__impl::emit8_backward(end - 16, static_cast<u32>(hi));
851     micron::__impl::emit8_backward(end - 8, micron::__impl::fast_mod1e8(value, hi));
852   } else {
853     __digits18(value, digits);
854   }
855   return end - count;
856 }

878     limbs[0] = static_cast<u32>(value);
879     limbs[1] = static_cast<u32>(value >> 32);
880     size = limbs[1] != 0 ? 2 : limbs[0] != 0 ? 1 : 0;
881   }
882 
883   constexpr void
884   trim() noexcept
885   {
886     while ( size > 0 && limbs[size - 1] == 0 ) --size;
887   }
888 
889   constexpr void
890   shl(i32 bits) noexcept
891   {
892     if ( size == 0 ) return;
893     const i32 words = bits >> 5;
894     const i32 shift = bits & 31;
895     if ( shift == 0 ) {
896       for ( i32 i = size; i-- > 0; ) limbs[i + words] = limbs[i];
897       size += words;
898     } else {
899       limbs[size + words] = limbs[size - 1] >> (32 - shift);
900       for ( i32 i = size - 1; i > 0; --i ) limbs[i + words] = limbs[i] << shift | limbs[i - 1] >> (32 - shift);
901       limbs[words] = limbs[0] << shift;
902       size += words + 1;
903     }
904     for ( i32 i = 0; i < words; ++i ) limbs[i] = 0;
905     trim();
906   }
907 
908   constexpr void
909   mul(u32 factor) noexcept
910   {
911     u64 carry = 0;
912     for ( i32 i = 0; i < size; ++i ) {
913       const u64 product = static_cast<u64>(limbs[i]) * factor + carry;
914       limbs[i] = static_cast<u32>(product);
915       carry = product >> 32;
916     }
917     if ( carry != 0 ) limbs[size++] = static_cast<u32>(carry);
918   }
919 
920   constexpr void
921   mul_pow5(i32 exp) noexcept
922   {
923     constexpr u32 pow5[13] = { 1u, 5u, 25u, 125u, 625u, 3125u, 15625u, 78125u, 390625u, 1953125u, 9765625u, 48828125u, 244140625u };
924     while ( exp >= 13 ) {
925       mul(1220703125u);
926       exp -= 13;
927     }
928     if ( exp != 0 ) mul(pow5[exp]);
929   }
930 
931   constexpr u32
932   divmod_1e9() noexcept
933   {
934     u64 rem = 0;
935     for ( i32 i = size; i-- > 0; ) {
936       const u64 value = rem << 32 | limbs[i];
937       limbs[i] = static_cast<u32>(value / 1000000000u);
938       rem = value % 1000000000u;
939     }
940     trim();
941     return static_cast<u32>(rem);
942   }
943 };

945 [[gnu::noinline]] inline constexpr char *
946 __write_big_digits(bigint &value, char *end) noexcept
947 {
948   char *out = end;
949   u32 group = value.divmod_1e9();
950   while ( value.size != 0 ) {
951     out -= 9;
952     const u32 hi = group / 100000000u;
953     out[0] = static_cast<char>('0' + hi);
954     micron::__impl::emit8_backward(out + 1, group - hi * 100000000u);
955     group = value.divmod_1e9();
956   }
957   return micron::__impl::uint_to_buf_backward(out, group);

962 [[gnu::noinline]] inline constexpr void
963 __exact_round(u64 sig, i32 exp, cut mode, i32 argument, char *digits, u32 &count, i32 &point) noexcept
964 {
965   bigint value(sig);
966   i32 base_exp = 0;
967   if ( exp >= 0 )
968     value.shl(exp);
969   else {
970     value.mul_pow5(-exp);
971     base_exp = exp;
972   }
973 
974   char exact[__ryu::__fx::__dig_cap];
975   char *start = __write_big_digits(value, exact + sizeof(exact));
976   const u32 exact_count = static_cast<u32>(exact + sizeof(exact) - start);
977   point = static_cast<i32>(exact_count) + base_exp;
978   count = 0;
979 
980   const i32 keep = mode == cut::significant ? argument : point + argument;
981   if ( keep < 0 ) return;
982   const u32 kept = static_cast<u32>(keep);
983   if ( kept >= exact_count ) {
984     for ( u32 i = 0; i < exact_count; ++i ) digits[i] = start[i];
985     count = exact_count;
986     return;
987   }
988 
989   bool tail = false;
990   for ( u32 i = kept + 1; i < exact_count; ++i ) tail |= start[i] != '0';
991   const char dropped = start[kept];
992   const bool odd = kept != 0 && ((start[kept - 1] - '0') & 1) != 0;
993   const bool up = dropped > '5' || (dropped == '5' && (tail || odd));
994   for ( u32 i = 0; i < kept; ++i ) digits[i] = start[i];
995   count = kept;
996   if ( !up ) return;
997 
998   for ( u32 i = kept; i-- > 0; ) {
999     if ( digits[i] != '9' ) {
1000       ++digits[i];
1001       return;
1002     }
1003     digits[i] = '0';
1004   }
1005   digits[0] = '1';
1006   if ( count == 0 ) count = 1;
1007   ++point;
1008 }

1030 inline constexpr usize
1031 d2f_buffered(f64 value, char *buf, usize cap, u32 precision) noexcept
1032 {
1033   const __ryu::__fx::parts p = __ryu::__fx::__decompose(value);
1034   if ( p.is_nan || p.is_inf ) return __ryu::__fx::__emit_special(p, buf, cap);
1035   if ( p.is_zero ) return __ryu::__fx::__emit_fixed(buf, cap, p.neg, nullptr, 0, 0, precision);
1036   if ( precision > 18 ) return __d2f_exact(p, buf, cap, precision);
1037 
1038   const i32 dec_exp = __compute_dec_exp(p.e2 + 52) - 17;
1039   u64 scaled = __scale(p.m2, p.e2, dec_exp);
1040   const u64 integral = scaled >> 2;
1041   const i32 scaled_digits = 18 + static_cast<i32>(integral >= __pow10s[18]);
1042   i32 lead_exp = dec_exp + scaled_digits - 1;
1043   const i32 num_digits = lead_exp + 1 + static_cast<i32>(precision);
1044   if ( num_digits <= 0 ) {
1045     if ( num_digits < 0 ) return __ryu::__fx::__emit_fixed(buf, cap, p.neg, nullptr, 0, 0, precision);
1046     const u64 half = 5 * __pow10s[scaled_digits - 1];
1047     const bool up = integral > half || (integral == half && (scaled & 3) != 0);
1048     const char one = '1';
1049     return __ryu::__fx::__emit_fixed(buf, cap, p.neg, up ? &one : nullptr, static_cast<u32>(up), up ? 1 - static_cast<i32>(precision) : 0,
1050                                      precision);
1051   }
1052   if ( num_digits > 18 ) return __d2f_exact(p, buf, cap, precision);
1053   if ( num_digits < scaled_digits ) {
1054     const u64 pow = __pow10s[scaled_digits - num_digits];
1055     const u64 rem = integral % pow;
1056     const u64 half = rem >= pow / 2;
1057     const u64 sticky = (rem != pow / 2) || ((scaled & 3) != 0);
1058     scaled = (integral / pow) << 2 | half << 1 | sticky;
1059   }
1060   u64 dec_sig = __round_even(scaled);
1061   if ( dec_sig >= __pow10s[num_digits] ) {
1062     ++lead_exp;
1063     dec_sig /= 10;
1064   }
1065   char digits[18];
1066   char *const start = __digits_n(dec_sig, static_cast<u32>(num_digits), digits);
1067   return __ryu::__fx::__emit_fixed(buf, cap, p.neg, start, static_cast<u32>(num_digits), lead_exp + 1, precision);
1068 }

1087 inline constexpr usize
1088 __d2g_core(f64 value, char *buf, usize cap, u32 precision, bool alt, bool upper) noexcept
1089 {
1090   const __ryu::__fx::parts p = __ryu::__fx::__decompose(value);
1091   if ( p.is_nan || p.is_inf ) return __ryu::__fx::__emit_special(p, buf, cap);
1092   const u32 significant = precision == 0 ? 1u : precision;
1093   if ( significant > 18 ) {
1094     char digits[__ryu::__fx::__dig_cap];
1095     u32 count = 0;
1096     i32 point = 0;
1097     __exact_round(p.m2, p.e2, cut::significant, static_cast<i32>(significant), digits, count, point);
1098     const i32 lead_exp = point - 1;
1099     usize len = 0;
1100     if ( lead_exp >= -4 && lead_exp < static_cast<i32>(significant) )
1101       len = __ryu::__fx::__emit_fixed(buf, cap, p.neg, digits, count, point,
1102                                       static_cast<u32>(static_cast<i32>(significant) - 1 - lead_exp));
1103     else
1104       len = __ryu::__fx::__emit_sci(buf, cap, p.neg, digits, count, lead_exp, significant - 1);
1105     if ( len == 0 ) return 0;
1106     if ( !alt ) {
1107       usize dot = len;
1108       usize exponent = len;
1109       for ( usize i = 0; i < len; ++i ) {
1110         if ( buf[i] == '.' )
1111           dot = i;
1112         else if ( buf[i] == 'e' ) {
1113           exponent = i;
1114           break;
1115         }
1116       }
1117       if ( dot != len ) {
1118         usize end = exponent;
1119         while ( end > dot + 1 && buf[end - 1] == '0' ) --end;
1120         if ( end == dot + 1 ) end = dot;
1121         if ( end != exponent ) {
1122           usize out = end;
1123           for ( usize i = exponent; i < len; ++i ) buf[out++] = buf[i];
1124           len = out;
1125         }
1126       }
1127     }
1128     if ( upper )
1129       for ( usize i = 0; i < len; ++i )
1130         if ( buf[i] == 'e' ) buf[i] = 'E';
1131     return len;
1132   }
1133   char digits[18];
1134   for ( char &digit : digits ) digit = '0';
1135   i32 lead_exp = 0;
1136   if ( !p.is_zero ) {
1137     u64 sig = p.m2;
1138     i32 exp = p.e2;
1139     __normalize(sig, exp);
1140     const precision_decimal dec = __to_precision(sig, exp, static_cast<i32>(significant));
1141     __digits18(dec.sig, digits);
1142     lead_exp = dec.lead_exp;
1143   }
1144   usize len = 0;
1145   if ( lead_exp >= -4 && lead_exp < static_cast<i32>(significant) )
1146     len = __ryu::__fx::__emit_fixed(buf, cap, p.neg, digits, 18, lead_exp + 1,
1147                                     static_cast<u32>(static_cast<i32>(significant) - 1 - lead_exp));
1148   else
1149     len = __ryu::__fx::__emit_sci(buf, cap, p.neg, digits, 18, lead_exp, significant - 1);
1150   if ( len == 0 ) return 0;
1151   if ( !alt ) {
1152     usize dot = len;
1153     usize exponent = len;
1154     for ( usize i = 0; i < len; ++i ) {
1155       if ( buf[i] == '.' )
1156         dot = i;
1157       else if ( buf[i] == 'e' ) {
1158         exponent = i;
1159         break;
1160       }
1161     }
1162     if ( dot != len ) {
1163       usize end = exponent;
1164       while ( end > dot + 1 && buf[end - 1] == '0' ) --end;
1165       if ( end == dot + 1 ) end = dot;
1166       if ( end != exponent ) {
1167         usize out = end;
1168         for ( usize i = exponent; i < len; ++i ) buf[out++] = buf[i];
1169         len = out;
1170       }
1171     }
1172   }
1173   if ( upper )
1174     for ( usize i = 0; i < len; ++i )
1175       if ( buf[i] == 'e' ) buf[i] = 'E';
1176   return len;
1177 }
