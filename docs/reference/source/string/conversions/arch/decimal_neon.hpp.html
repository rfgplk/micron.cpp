19 inline bool
20 read16(const char *source, __prefix16 &value) noexcept
21 {
22 #if __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
23   return __scalar16(source, value);
24 #else
25   __bytes16 input;
26   __builtin_memcpy(&input, source, 16);
27   auto digits = input - __bytes16{ 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48 };
28   const auto dots = input == __bytes16{ 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46, 46 };
29   const auto invalid = (digits > __bytes16{ 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9, 9 }) & ~dots;
30   __block16 bad, mask;
31   __builtin_memcpy(&bad, &invalid, 16);
32   __builtin_memcpy(&mask, &dots, 16);
33   if ( bad.low || bad.high || (mask.low && mask.high) ) return false;
34   const u64 bits = mask.low ? mask.low : mask.high;
35   u32 point = 16;
36   if ( bits ) {
37     const u32 lane = static_cast<u32>(__builtin_ctzll(bits)) / 8;
38     if ( bits != (u64{ 255 } << (lane * 8)) ) return false;
39     point = lane + (mask.low ? 0 : 8);
40     __block16 block;
41     __builtin_memcpy(&block, &digits, 16);
42     block = __compact16(block, point);
43     __builtin_memcpy(&digits, &block, 16);
44   }
45   using __words8 = u16 __attribute__((vector_size(16)));
46   const auto words = __builtin_bit_cast(__words8, digits);
47   const auto pairs
48       = (words & __words8{ 255, 255, 255, 255, 255, 255, 255, 255 }) * __words8{ 10, 10, 10, 10, 10, 10, 10, 10 } + (words >> 8);
49   const auto packed = __builtin_bit_cast(__dwords4, pairs);
50   const auto groups = (packed & __dwords4{ 65535, 65535, 65535, 65535 }) * __dwords4{ 100, 100, 100, 100 } + (packed >> 16);
51   const u64 mantissa = static_cast<u64>(groups[0] * 10000 + groups[1]) * 100000000 + groups[2] * 10000 + groups[3];
52   value = { bits ? mantissa / 10 : mantissa, point };
53   return true;
54 #endif
55 }

57 inline bool
58 read8(const char *source, u32 &value) noexcept
59 {
60   __bytes8 bytes;
61   __builtin_memcpy(&bytes, source, 8);
62   const auto digits = bytes - __bytes8{ 48, 48, 48, 48, 48, 48, 48, 48 };
63   const auto invalid = digits > __bytes8{ 9, 9, 9, 9, 9, 9, 9, 9 };
64   if ( __builtin_bit_cast(u64, invalid) ) return false;
65   const auto words = __builtin_bit_cast(__words4, digits);
66 #if __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
67   const auto pairs = (words >> 8) * __words4{ 10, 10, 10, 10 } + (words & __words4{ 255, 255, 255, 255 });
68   const auto packed = __builtin_bit_cast(__dwords2, pairs);
69   const auto groups = (packed >> 16) * __dwords2{ 100, 100 } + (packed & __dwords2{ 65535, 65535 });
70 #else
71   const auto pairs = (words & __words4{ 255, 255, 255, 255 }) * __words4{ 10, 10, 10, 10 } + (words >> 8);
72   const auto packed = __builtin_bit_cast(__dwords2, pairs);
73   const auto groups = (packed & __dwords2{ 65535, 65535 }) * __dwords2{ 100, 100 } + (packed >> 16);
74 #endif
75   value = groups[0] * 10000 + groups[1];
76   return true;
77 }

79 inline void
80 write8(u8 *destination, u32 value) noexcept
81 {
82   const u32 high = value / 10000, low = value % 10000;
83   const __dwords4 pairs{ high / 100, high % 100, low / 100, low % 100 };
84   const auto tens = (pairs * __dwords4{ 52429, 52429, 52429, 52429 }) >> 19;
85   const auto ones = pairs - tens * __dwords4{ 10, 10, 10, 10 };
86   const auto q = __builtin_convertvector(tens, __words4), r = __builtin_convertvector(ones, __words4);
87 #if __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
88   const auto digits = (q << 8) | r;
89 #else
90   const auto digits = q | (r << 8);
91 #endif
92   __builtin_memcpy(destination, &digits, 8);
93 }

95 inline void
96 ascii(char *destination, const u8 *source, usize count) noexcept
97 {
98   usize i = 0;
99   const __bytes16 zero{ 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48, 48 };
100   for ( ; count - i >= 16; i += 16 ) {
101     __bytes16 digits;
102     __builtin_memcpy(&digits, source + i, 16);
103     const auto text = digits + zero;
104     __builtin_memcpy(destination + i, &text, 16);
105   }
106   for ( ; i < count; ++i ) destination[i] = static_cast<char>('0' + source[i]);
107 }
