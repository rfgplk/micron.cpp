16 inline bool
17 read16(const char *source, __prefix16 &value) noexcept
18 {
19   __m128i input;
20   __builtin_memcpy(&input, source, 16);
21   auto digits = sse::sub_i8(input, sse::splat_i8('0'));
22   const u32 valid = static_cast<u32>(sse::movemask_i8(sse::eq_i8(sse::min_u8(digits, sse::splat_i8(9)), digits)));
23   const u32 dots = static_cast<u32>(sse::movemask_i8(sse::eq_i8(input, sse::splat_i8('.'))));
24   if ( (valid | dots) != 65535 || (dots & (dots - 1)) ) return false;
25   const u32 point = dots ? static_cast<u32>(__builtin_ctz(dots)) : 16;
26   if ( dots ) {
27     __block16 block;
28     __builtin_memcpy(&block, &digits, 16);
29     block = __compact16(block, point);
30     __builtin_memcpy(&digits, &block, 16);
31   }
32 #if defined(__micron_x86_ssse3)
33   const auto pairs = sse::madd_u8s8(digits, sse::splat_i16(0x010a));
34 #else
35   const auto weights = sse::set_i16(1, 10, 1, 10, 1, 10, 1, 10);
36   const auto wide_low = sse::madd_i16(sse::unpack_lo_i8(digits, sse::zero_i128()), weights);
37   const auto wide_high = sse::madd_i16(sse::unpack_hi_i8(digits, sse::zero_i128()), weights);
38   const auto pairs = sse::pack_sat_i32(wide_low, wide_high);
39 #endif
40   const auto groups = sse::madd_i16(pairs, sse::splat_i32(0x00010064));
41   const u32 high
42       = static_cast<u32>(sse::extract_low_i32(groups)) * 10000 + static_cast<u32>(sse::extract_low_i32(sse::bsrl_i128<4>(groups)));
43   const u32 low = static_cast<u32>(sse::extract_low_i32(sse::bsrl_i128<8>(groups))) * 10000
44                   + static_cast<u32>(sse::extract_low_i32(sse::bsrl_i128<12>(groups)));
45   const u64 mantissa = static_cast<u64>(high) * 100000000 + low;
46   value = { dots ? mantissa / 10 : mantissa, point };
47   return true;
48 }

50 inline bool
51 read8(const char *source, u32 &value) noexcept
52 {
53   __m128i input = sse::zero_i128();
54   __builtin_memcpy(&input, source, 8);
55   const auto digits = sse::sub_i8(input, sse::splat_i8('0'));
56   const auto valid = sse::eq_i8(sse::min_u8(digits, sse::splat_i8(9)), digits);
57   if ( (sse::movemask_i8(valid) & 255) != 255 ) return false;
58 #if defined(__micron_x86_ssse3)
59   const auto pairs = sse::madd_u8s8(digits, sse::splat_i16(0x010a));
60 #else
61   const auto wide = sse::unpack_lo_i8(digits, sse::zero_i128());
62   const auto products = sse::madd_i16(wide, sse::set_i16(1, 10, 1, 10, 1, 10, 1, 10));
63   const auto pairs = sse::pack_sat_i32(products, sse::zero_i128());
64 #endif
65   const auto groups = sse::madd_i16(pairs, sse::splat_i32(0x00010064));
66   value = static_cast<u32>(sse::extract_low_i32(groups)) * 10000 + static_cast<u32>(sse::extract_low_i32(sse::bsrl_i128<4>(groups)));
67   return true;
68 }

70 inline void
71 write8(u8 *destination, u32 value) noexcept
72 {
73   const u32 high = value / 10000, low = value % 10000;
74   const auto pairs = sse::set_i16(0, 0, 0, 0, low % 100, low / 100, high % 100, high / 100);
75   const auto tens = sse::shr_i16(sse::mul_hi_u16(pairs, sse::splat_i16(static_cast<i16>(52429))), 3);
76   const auto ones = sse::sub_i16(pairs, sse::mul_lo_i16(tens, sse::splat_i16(10)));
77   const auto digits = sse::pack_satu_i16(sse::unpack_lo_i16(tens, ones), sse::zero_i128());
78   __builtin_memcpy(destination, &digits, 8);
79 }

81 inline void
82 ascii(char *destination, const u8 *source, usize count) noexcept
83 {
84   usize i = 0;
85   for ( ; count - i >= 16; i += 16 ) {
86     __m128i digits;
87     __builtin_memcpy(&digits, source + i, 16);
88     const auto text = sse::add_i8(digits, sse::splat_i8('0'));
89     __builtin_memcpy(destination + i, &text, 16);
90   }
91   for ( ; i < count; ++i ) destination[i] = static_cast<char>('0' + source[i]);
92 }
