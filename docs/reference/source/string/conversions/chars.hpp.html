26 inline constexpr usize
27 decimal_chars_length(u64 value) noexcept
28 {
29   if ( value < 10000000000ull ) {
30     if ( value < 100000ull ) {
31       if ( value < 100ull ) {
32         if ( value < 10ull ) return 1;
33         return 2;
34       }
35       if ( value < 10000ull ) {
36         if ( value < 1000ull ) return 3;
37         return 4;
38       }
39       return 5;
40     }
41     if ( value < 10000000ull ) {
42       if ( value < 1000000ull ) return 6;
43       return 7;
44     }
45     if ( value < 1000000000ull ) {
46       if ( value < 100000000ull ) return 8;
47       return 9;
48     }
49     return 10;
50   }
51   if ( value < 1000000000000000ull ) {
52     if ( value < 1000000000000ull ) {
53       if ( value < 100000000000ull ) return 11;
54       return 12;
55     }
56     if ( value < 10000000000000ull ) return 13;
57     if ( value < 100000000000000ull ) return 14;
58     return 15;
59   }
60   if ( value < 100000000000000000ull ) {
61     if ( value < 10000000000000000ull ) return 16;
62     return 17;
63   }
64   if ( value < 10000000000000000000ull ) {
65     if ( value < 1000000000000000000ull ) return 18;
66     return 19;
67   }
68   return 20;
69 }

71 inline constexpr usize
72 decimal_chars_length(i64 value) noexcept
73 {
74   const bool negative = value < 0;
75   const u64 magnitude = negative ? static_cast<u64>(0) - static_cast<u64>(value) : static_cast<u64>(value);
76   return decimal_chars_length(magnitude) + static_cast<usize>(negative);

79 template<__tc_int I>
80 constexpr usize
81 to_chars(char *buf, usize cap, I v, u32 base = 10u, bool upper = false)
82 {
83   if ( buf == nullptr || base < 2u || base > 36u ) return 0;
84 
85   using U = micron::make_unsigned_t<I>;
86   U mag;
87   bool neg = false;
88   if constexpr ( micron::is_signed_v<I> ) {
89     if ( v < 0 ) {
90       neg = true;
91       mag = static_cast<U>(0) - static_cast<U>(v);      // two's complement, safe at the minimum
92     } else {
93       mag = static_cast<U>(v);
94     }
95   } else {
96     mag = static_cast<U>(v);
97   }
98 
99   char tmp[__tc_int_max];
100   char *tend = tmp + __tc_int_max;
101   char *start = (base == 10u) ? __impl::uint_to_buf_backward(tend, static_cast<u64>(mag))
102                               : __impl::uint_to_buf_base_backward(tend, static_cast<u64>(mag), base, upper);
103 
104   const usize len = static_cast<usize>(tend - start) + (neg ? 1u : 0u);
105   if ( cap < len ) return 0;
106 
107   usize pos = 0;
108   if ( neg ) buf[pos++] = '-';
109   for ( char *q = start; q != tend; ++q ) buf[pos++] = *q;
110   return pos;
111 }

113 template<__tc_int I>
114 constexpr bool
115 from_chars(I &out, const char *p, usize n, u32 base = 10u)
116 {
117   out = static_cast<I>(0);
118   if ( p == nullptr || n == 0 || base < 2u || base > 36u ) return false;
119 
120   const char *ptr = p;
121   const char *const end = p + n;
122   bool neg = false;
123   if ( *ptr == '-' ) {
124     if constexpr ( !micron::is_signed_v<I> ) return false;
125     neg = true;
126     ++ptr;
127   } else if ( *ptr == '+' ) {
128     ++ptr;
129   }
130   if ( ptr == end ) return false;
131 
132   using U = micron::make_unsigned_t<I>;
133   constexpr u64 umax = static_cast<u64>(static_cast<U>(~static_cast<U>(0)));
134   u64 lim;
135   if constexpr ( micron::is_signed_v<I> )
136     lim = neg ? (umax >> 1) + 1ull : (umax >> 1);
137   else
138     lim = umax;
139 
140   u64 acc = 0;
141   for ( ; ptr != end; ++ptr ) {
142     const int d = __impl::digit_val(*ptr, base);
143     if ( d < 0 ) return false;
144     const u64 du = static_cast<u64>(d);
145     if ( du > lim || acc > (lim - du) / base ) return false;      // out of range for I
146     acc = acc * base + du;
147   }
148 
149   if constexpr ( micron::is_signed_v<I> ) {
150     if ( neg )
151       out = static_cast<I>(static_cast<U>(0ull - acc));
152     else
153       out = static_cast<I>(static_cast<U>(acc));
154   } else {
155     out = static_cast<I>(static_cast<U>(acc));
156   }
157   return true;
158 }

225 inline constexpr bool
226 from_chars(u128 &out, const char *p, usize n, u32 base = 10u)
227 {
228   out = static_cast<u128>(0);
229   if ( p == nullptr || n == 0 || base < 2u || base > 36u ) return false;
230   const char *ptr = p;
231   const char *const end = p + n;
232   if ( *ptr == '+' ) ++ptr;
233   if ( ptr == end ) return false;
234 
235   u128 acc = static_cast<u128>(0);
236   const u128 b = static_cast<u128>(base);
237   // ~0 / base, computed once: the overflow guard
238   const u128 umax = static_cast<u128>(0) - static_cast<u128>(1);
239   const u128 lim = umax / b;
240   const u128 limr = umax - lim * b;
241   for ( ; ptr != end; ++ptr ) {
242     const int d = __impl::digit_val(*ptr, base);
243     if ( d < 0 ) return false;
244     const u128 du = static_cast<u128>(static_cast<u64>(d));
245     if ( acc > lim || (acc == lim && du > limr) ) return false;
246     acc = acc * b + du;
247   }
248   out = acc;
249   return true;
250 }

252 inline constexpr bool
253 from_chars(i128 &out, const char *p, usize n, u32 base = 10u)
254 {
255   out = static_cast<i128>(0);
256   if ( p == nullptr || n == 0 || base < 2u || base > 36u ) return false;
257   bool neg = false;
258   const char *ptr = p;
259   if ( *ptr == '-' ) {
260     neg = true;
261     ++ptr;
262   } else if ( *ptr == '+' ) {
263     ++ptr;
264   }
265   if ( ptr == p + n ) return false;
266   if ( *ptr == '+' || *ptr == '-' ) return false;
267 
268   u128 mag = static_cast<u128>(0);
269   if ( !from_chars(mag, ptr, static_cast<usize>((p + n) - ptr), base) ) return false;
270 
271   // i128 spans [-2^127, 2^127-1]
272   const u128 pos_max = (static_cast<u128>(1) << 127) - static_cast<u128>(1);
273   const u128 neg_max = static_cast<u128>(1) << 127;
274   if ( neg ) {
275     if ( mag > neg_max ) return false;
276   } else {
277     if ( mag > pos_max ) return false;
278   }
279 #if defined(__micron_arch_width_64)
280   out = neg ? static_cast<i128>(static_cast<u128>(0) - mag) : static_cast<i128>(mag);
281 #else
282   out = neg ? i128(i128::__negate_u(mag)) : i128(mag);
283 #endif
284   return true;
285 }

311 inline constexpr usize
312 to_chars(char *buf, usize cap, f64 v, float_format fmt = float_format::shortest, i32 precision = -1)
313 {
314   if ( buf == nullptr ) return 0;
315   const bool has_prec = precision >= 0;
316   const u32 prec = has_prec ? static_cast<u32>(precision) : 6u;
317   switch ( fmt ) {
318   case float_format::fixed:
319     return __impl::__fpconv::d2f_buffered(v, buf, cap, prec);
320   case float_format::scientific:
321     return __impl::__fpconv::d2e_buffered(v, buf, cap, prec);
322   case float_format::general:
323     return __impl::__fpconv::d2g_buffered(v, buf, cap, prec, false, false);
324   case float_format::hex:
325     return __impl::__fpconv::d2a_buffered(v, buf, cap, prec, has_prec, false);
326   default:
327     break;
328   }
329   if ( cap < 26u ) {
330     char tmp[26];
331     const usize n = __impl::__fpconv::d2s_buffered(v, tmp);
332     if ( n > cap ) return 0;
333     for ( usize i = 0; i < n; ++i ) buf[i] = tmp[i];
334     return n;
335   }
336   return __impl::__fpconv::d2s_buffered(v, buf);
337 }

340 to_chars(char *buf, usize cap, f32 v, float_format fmt = float_format::shortest, i32 precision = -1)
341 {

349     return n;

351   return __impl::__fpconv::f2s_buffered(v, buf);

416 constexpr bool
417 from_chars(bool &out, const char *p, usize n)
418 {
419   out = false;
420   if ( p == nullptr ) return false;
421   if ( n == 1 ) {
422     if ( p[0] == '1' ) {
423       out = true;
424       return true;
425     }
426     return p[0] == '0';
427   }
428   if ( n == 4 && p[0] == 't' && p[1] == 'r' && p[2] == 'u' && p[3] == 'e' ) {
429     out = true;
430     return true;
431   }
432   if ( n == 5 && p[0] == 'f' && p[1] == 'a' && p[2] == 'l' && p[3] == 's' && p[4] == 'e' ) return true;
433   return false;
434 }

436 inline usize
437 to_chars(char *buf, usize cap, const void *v)
438 {
439   if ( buf == nullptr || cap < 3u ) return 0;
440   if ( v == nullptr ) {
441     buf[0] = '0';
442     buf[1] = 'x';
443     buf[2] = '0';
444     return 3;
445   }
446   char tmp[24];
447   char *tend = tmp + 24;
448   char *start = __impl::uint_to_buf_base_backward(tend, reinterpret_cast<u64>(v), 16u, false);
449   const usize len = static_cast<usize>(tend - start) + 2u;
450   if ( cap < len ) return 0;
451   buf[0] = '0';
452   buf[1] = 'x';
453   usize pos = 2;
454   for ( char *q = start; q != tend; ++q ) buf[pos++] = *q;
455   return pos;
456 }

458 constexpr usize
459 bytes_to_hex(char *buf, usize cap, const u8 *src, usize n, bool upper = false)
460 {
461   if ( buf == nullptr || src == nullptr ) return 0;
462   if ( cap < n * 2u ) return 0;
463   const char *tbl = upper ? __impl::__hex_upper : __impl::__hex_lower;
464   for ( usize i = 0; i < n; ++i ) {
465     buf[2 * i] = tbl[(src[i] >> 4) & 0xFu];
466     buf[2 * i + 1] = tbl[src[i] & 0xFu];
467   }
468   return n * 2u;
469 }
