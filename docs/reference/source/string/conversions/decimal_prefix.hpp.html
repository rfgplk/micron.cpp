12 struct __prefix16 {
13   u64 mantissa;
14   u32 point;
15 };
16 
17 inline constexpr bool
18 __scalar16(const char *source, __prefix16 &value) noexcept
19 {
20   u64 mantissa = 0;
21   u32 point = 16;
22   for ( u32 i = 0; i < 16; ++i ) {
23     if ( source[i] == '.' && point == 16 ) {
24       point = i;
25       continue;
26     }
27     const u32 digit = static_cast<u8>(source[i]) - static_cast<u32>('0');
28     if ( digit > 9 ) return false;
29     mantissa = mantissa * 10 + digit;
30   }
31   value = { mantissa, point };
32   return true;
33 }
34 
35 struct __block16 {
36   u64 low, high;
37 };
38 
39 // Little-endian byte deletion. The vacated last byte becomes a raw zero digit.
40 inline __block16
41 __compact16(__block16 value, u32 point) noexcept
42 {
43   if ( point < 8 ) {
44     const u64 mask = (u64{ 1 } << (point * 8)) - 1;
45     value.low = (value.low & mask) | ((value.low >> 8) & ~mask) | (value.high << 56);
46     value.high >>= 8;
47   } else {
48     const u64 mask = (u64{ 1 } << ((point - 8) * 8)) - 1;
49     value.high = (value.high & mask) | ((value.high >> 8) & ~mask);
50   }
51   return value;
52 }
