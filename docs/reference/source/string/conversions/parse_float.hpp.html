487   template<bool Bounded, typename T>
488   inline constexpr bool
489   __scan_hex(const T *&p, const T *e, __scan_t<T> &s) noexcept
490   {
491     const T *save = p;
492     p += 2;      // the caller matched "0x"
493     u64 mant = 0;
494     i64 bexp = 0;
495     bool any = false;
496     bool truncated = false;
497 
498     while ( __live<Bounded>(p, e) ) {
499       i32 d = __hex_val(*p);
500       if ( d < 0 ) break;
501       any = true;
502       if ( mant <= (~0ull >> 4) )
503         mant = (mant << 4) | static_cast<u64>(d);
504       else {
505         bexp += 4;
506         if ( d ) truncated = true;
507       }
508       ++p;
509     }
510     if ( __live<Bounded>(p, e) && *p == static_cast<T>('.') ) {
511       ++p;
512       while ( __live<Bounded>(p, e) ) {
513         i32 d = __hex_val(*p);
514         if ( d < 0 ) break;
515         any = true;
516         if ( mant <= (~0ull >> 4) ) {
517           mant = (mant << 4) | static_cast<u64>(d);
518           bexp -= 4;
519         } else if ( d )
520           truncated = true;
521         ++p;
522       }
523     }
524     if ( !any ) {      // "0x" with nothing after it is the integer 0 followed by an 'x'
525       p = save;
526       return false;
527     }
528     // the binary-exponent part is optional here, as it is for from_chars
529     if ( __live<Bounded>(p, e) && (*p == static_cast<T>('p') || *p == static_cast<T>('P')) ) {
530       const T *psave = p;
531       ++p;
532       bool eneg = false;
533       if ( __live<Bounded>(p, e) && (*p == static_cast<T>('+') || *p == static_cast<T>('-')) ) {
534         eneg = (*p == static_cast<T>('-'));
535         ++p;
536       }
537       if ( !__live<Bounded>(p, e) || !__is_digit(*p) )
538         p = psave;
539       else {
540         i64 v = 0;
541         while ( __live<Bounded>(p, e) && __is_digit(*p) ) {
542           if ( v < __exp_clamp ) v = v * 10 + static_cast<i64>(__digit_val(*p));
543           ++p;
544         }
545         bexp += eneg ? -v : v;
546       }
547     }
548     s.mant = mant;
549     s.exp10 = bexp;
550     s.truncated = truncated;
551     s.kind = __kind::hex;
552     return true;
553   }

558   __match_word(const T *&p, const T *e, const char *word) noexcept
559   {

569   template<bool Bounded, typename T>
570   inline constexpr __scan_t<T>
571   __scan(const T *p, const T *e) noexcept
572   {
573     __scan_t<T> s;
574     s.stop = p;
575 
576     while ( __live<Bounded>(p, e) && (*p == static_cast<T>(' ') || *p == static_cast<T>('\t')) ) ++p;
577 
578     if ( __live<Bounded>(p, e) ) {
579       if ( *p == static_cast<T>('-') ) {
580         s.neg = true;
581         ++p;
582       } else if ( *p == static_cast<T>('+') )
583         ++p;
584     }
585     if ( !__live<Bounded>(p, e) ) return s;
586 
587     // inf / infinity -- longest match wins
588     if ( __lower(*p) == 'i' ) {
589       const T *q = p;
590       if ( __match_word<Bounded>(q, e, "infinity") ) {
591         s.kind = __kind::inf;
592         s.stop = q;
593         return s;
594       }
595       q = p;
596       if ( __match_word<Bounded>(q, e, "inf") ) {
597         s.kind = __kind::inf;
598         s.stop = q;
599       }
600       return s;
601     }
602     // nan, optionally nan(payload)
603     if ( __lower(*p) == 'n' ) {
604       const T *q = p;
605       if ( !__match_word<Bounded>(q, e, "nan") ) return s;
606       s.kind = __kind::nan;
607       s.stop = q;
608       if ( __live<Bounded>(q, e) && *q == static_cast<T>('(') ) {
609         const T *r = q + 1;
610         const T *seq = r;
611         while ( __live<Bounded>(r, e) && (__is_digit(*r) || static_cast<u32>(__lower(*r) - 'a') <= 25u || *r == static_cast<T>('_')) ) ++r;
612         if ( __live<Bounded>(r, e) && *r == static_cast<T>(')') ) {
613           const T *d = seq;
614           if ( r - d >= 2 && *d == static_cast<T>('0') && (d[1] == static_cast<T>('x') || d[1] == static_cast<T>('X')) ) d += 2;
615           u64 pay = 0;
616           bool hex = (d != r);
617           for ( const T *k = d; k != r; ++k ) {
618             const i32 v = __hex_val(*k);
619             if ( v < 0 ) {
620               hex = false;
621               break;
622             }
623             pay = (pay << 4) | static_cast<u64>(v);
624           }
625           s.payload = hex ? pay : 0;
626           s.stop = r + 1;
627         }
628       }
629       return s;
630     }
631 
632     // hex float
633     if ( *p == static_cast<T>('0') ) {
634       const T *q = p + 1;
635       if ( __live<Bounded>(q, e) && (*q == static_cast<T>('x') || *q == static_cast<T>('X')) ) {
636         const T *h = p;
637         if ( __scan_hex<Bounded>(h, e, s) ) {
638           s.stop = h;
639           return s;
640         }
641       }
642     }
643 
644     u64 mant = 0;
645     i64 exp10 = 0;
646     bool any = false;
647 
648     s.int_first = p;
649     bool fraction_started = false;
650     if constexpr ( Bounded && __is_same(T, char) ) {
651       if !consteval {
652         micron::__impl::__digits::__prefix16 prefix;
653         if ( e - p >= 16 && micron::__impl::__digits::read16(p, prefix) ) {
654           mant = prefix.mantissa;
655           any = true;
656           if ( prefix.point < 16 ) {
657             s.int_last = p + prefix.point;
658             s.frac_first = s.int_last + 1;
659             exp10 = -static_cast<i64>(15 - prefix.point);
660             fraction_started = true;
661           }
662           p += 16;
663         }
664       }
665     }
666     if constexpr ( Bounded && __is_same(T, char) ) {
667       if !consteval {
668         while ( !fraction_started && e - p >= 8 && mant <= (~u64{ 0 } - 99999999) / 100000000 ) {
669           u32 group;
670           if ( !micron::__impl::__digits::read8(p, group) ) break;
671           mant = mant * 100000000 + group;
672           p += 8;
673           any = true;
674         }
675       }
676     }
677     while ( !fraction_started && __live<Bounded>(p, e) && __is_digit(*p) ) {
678       if constexpr ( Bounded && __is_same(T, char) ) {
679         if !consteval {
680           if ( mant > __mant_cap && e - p >= 8 ) {
681             u32 group;
682             if ( micron::__impl::__digits::read8(p, group) ) {
683               exp10 += 8;
684               s.truncated |= group != 0;
685               p += 8;
686               continue;
687             }
688           }
689         }
690       }
691       u32 d = __digit_val(*p);
692       if ( mant <= __mant_cap )
693         mant = mant * 10 + d;
694       else {
695         ++exp10;
696         if ( d ) s.truncated = true;
697       }
698       any = true;
699       ++p;
700     }
701     if ( !fraction_started ) {
702       s.int_last = p;
703       s.frac_first = p;
704     }
705     s.frac_last = p;
706     if ( fraction_started || (__live<Bounded>(p, e) && *p == static_cast<T>('.')) ) {
707       if ( !fraction_started ) {
708         ++p;
709         s.frac_first = p;
710       }
711       if constexpr ( Bounded && __is_same(T, char) ) {
712         if !consteval {
713           while ( e - p >= 8 && mant <= (~u64{ 0 } - 99999999) / 100000000 ) {
714             u32 group;
715             if ( !micron::__impl::__digits::read8(p, group) ) break;
716             mant = mant * 100000000 + group;
717             exp10 -= 8;
718             p += 8;
719             any = true;
720           }
721         }
722       }
723       while ( __live<Bounded>(p, e) && __is_digit(*p) ) {
724         if constexpr ( Bounded && __is_same(T, char) ) {
725           if !consteval {
726             if ( mant > __mant_cap && e - p >= 8 ) {
727               u32 group;
728               if ( micron::__impl::__digits::read8(p, group) ) {
729                 s.truncated |= group != 0;
730                 p += 8;
731                 continue;
732               }
733             }
734           }
735         }
736         u32 d = __digit_val(*p);
737         if ( mant <= __mant_cap ) {
738           mant = mant * 10 + d;
739           --exp10;
740         } else if ( d )
741           s.truncated = true;
742         any = true;
743         ++p;
744       }
745       s.frac_last = p;
746     }
747     if ( !any ) return s;
748 
749     if ( __live<Bounded>(p, e) && (*p == static_cast<T>('e') || *p == static_cast<T>('E')) ) {
750       const T *save = p;
751       ++p;
752       bool eneg = false;
753       if ( __live<Bounded>(p, e) && (*p == static_cast<T>('+') || *p == static_cast<T>('-')) ) {
754         eneg = (*p == static_cast<T>('-'));
755         ++p;
756       }
757       if ( !__live<Bounded>(p, e) || !__is_digit(*p) )
758         p = save;      // "1e" / "1e+"
759       else {
760         i64 v = 0;
761         while ( __live<Bounded>(p, e) && __is_digit(*p) ) {
762           if ( v < __exp_clamp ) v = v * 10 + static_cast<i64>(__digit_val(*p));      // clamp, never wrap
763           ++p;
764         }
765         s.exp_lit = eneg ? -v : v;
766         exp10 += s.exp_lit;
767       }
768     }
769 
770     if ( exp10 > __exp_clamp ) exp10 = __exp_clamp;
771     if ( exp10 < -__exp_clamp ) exp10 = -__exp_clamp;
772 
773     s.mant = mant;
774     s.exp10 = exp10;
775     s.kind = __kind::number;
776     s.stop = p;
777     return s;
778   }

818   template<typename F>
819   inline constexpr bool
820   __exact(u64 mant, i64 exp10, bool neg, F &out) noexcept
821   {
822     using TR = __fmt_traits<F>;
823     if constexpr ( !__fpu_evaluates_narrow )
824       return false;
825     else {
826       if ( mant > TR::fast_path_max_mantissa ) return false;
827       if ( exp10 < -TR::fast_path_max_exp ) return false;
828 
829       if ( exp10 > TR::fast_path_max_exp ) {
830         if ( exp10 > TR::fast_path_scale_max ) return false;
831         const i64 shift = exp10 - TR::fast_path_max_exp;
832         if ( mant > TR::fast_path_max_mantissa / __pow10_u64[shift] ) return false;
833         mant *= __pow10_u64[shift];
834         exp10 = TR::fast_path_max_exp;
835       }
836 
837       f64 v = static_cast<f64>(mant);
838       if ( exp10 > 0 )
839         v = v * __pow10_exact[exp10];
840       else if ( exp10 < 0 )
841         v = v / __pow10_exact[-exp10];
842       out = neg ? -static_cast<F>(v) : static_cast<F>(v);
843       return true;
844     }

857   // floor(w*T / 2^64) == w*T_hi + floor(w*T_lo / 2^64)
858   template<i32 BitPrecision>
859   inline constexpr __fmt_uint128_t
860   __product(i64 q, u64 w) noexcept
861   {
862     const i32 index = static_cast<i32>(q) - __pow5_128_min;
863     __fmt_uint128_t first = umul128(w, __pow5_128[index][1]);
864     constexpr u64 precision_mask = (BitPrecision < 64) ? (0xFFFFFFFFFFFFFFFFull >> BitPrecision) : 0xFFFFFFFFFFFFFFFFull;
865     if ( (first.hi & precision_mask) == precision_mask ) {
866       __fmt_uint128_t second = umul128(w, __pow5_128[index][0]);
867       u64 lo = first.lo + second.hi;
868       if ( second.hi > lo ) ++first.hi;
869       first.lo = lo;
870     }
871     return first;
872   }

885   template<typename F>
886   inline constexpr __am
887   __eisel_lemire(i64 q, u64 w) noexcept
888   {
889     using TR = __fmt_traits<F>;
890     __am a;
891 
892     if ( w == 0 || q < TR::smallest_power ) return a;      // +0
893     if ( q > TR::largest_power ) {
894       a.power2 = TR::infinite_power;
895       return a;
896     }
897 
898     const i32 lz = static_cast<i32>(clz64(w));
899     w <<= lz;
900 
901     const __fmt_uint128_t prod = __product<TR::mantissa_bits + 3>(q, w);
902     // provably unreachable for f32/f64 over the supported range
903     if ( prod.lo == 0xFFFFFFFFFFFFFFFFull ) {
904       a.power2 = -1;
905       return a;
906     }
907 
908     const i32 upperbit = static_cast<i32>(prod.hi >> 63);
909     u64 mantissa = prod.hi >> (upperbit + 64 - TR::mantissa_bits - 3);
910     i32 power2 = __log2_pow10(static_cast<i32>(q)) + 63 + upperbit - lz - TR::min_exponent;
911 
912     if ( power2 <= 0 ) {      // subnormal, or below it
913       if ( -power2 + 1 >= 64 ) return a;
914       mantissa >>= -power2 + 1;
915       mantissa += (mantissa & 1);
916       mantissa >>= 1;
917       a.mantissa = mantissa;
918       a.power2 = (mantissa < (1ull << TR::mantissa_bits)) ? 0 : 1;
919       return a;
920     }
921 
922     if ( prod.lo <= 1 && q >= -4 && q <= 23 && (mantissa & 3) == 1 ) {
923       if ( (mantissa << (upperbit + 64 - TR::mantissa_bits - 3)) == prod.hi ) mantissa &= ~1ull;
924     }
925 
926     mantissa += (mantissa & 1);
927     mantissa >>= 1;
928     if ( mantissa >= (2ull << TR::mantissa_bits) ) {
929       mantissa = (1ull << TR::mantissa_bits);
930       ++power2;
931     }
932     mantissa &= ~(1ull << TR::mantissa_bits);
933     if ( power2 >= TR::infinite_power ) {
934       a.mantissa = 0;
935       a.power2 = TR::infinite_power;
936       return a;
937     }
938     a.mantissa = mantissa;
939     a.power2 = power2;
940     return a;
941   }

947   inline constexpr u64
948   __dec_rounded(const __decimal &v) noexcept
949   {
950     if ( v.point < 0 || v.ndigits == 0 ) return 0;
951     if ( v.point > 20 ) return ~0ull;
952     const u32 np = static_cast<u32>(v.point);
953     u64 n = 0;
954     u32 i = 0;
955     for ( ; i < np && i < v.ndigits; ++i ) n = n * 10 + v.d[i];
956     for ( ; i < np; ++i ) n *= 10;
957     if ( __dec_round_up(v, np) ) ++n;
958     return n;
959   }

961   template<typename T>
962   inline constexpr void
963   __dec_load(__decimal &v, const __scan_t<T> &s) noexcept
964   {
965     u32 w = 0;
966     bool seen = false;
967     i32 point = 0;
968 
969     for ( const T *p = s.int_first; p != s.int_last; ++p ) {
970       u8 d = static_cast<u8>(__digit_val(*p));
971       if ( !seen && d == 0 ) continue;      // leading zeros carry nothing
972       seen = true;
973       if ( w < __dec_cap )
974         v.d[w++] = d;
975       else if ( d )
976         v.truncated = true;
977       ++point;
978     }
979     for ( const T *p = s.frac_first; p != s.frac_last; ++p ) {
980       u8 d = static_cast<u8>(__digit_val(*p));
981       if ( !seen && d == 0 ) {
982         --point;
983         continue;
984       }
985       seen = true;
986       if ( w < __dec_cap )
987         v.d[w++] = d;
988       else if ( d )
989         v.truncated = true;
990     }
991     v.ndigits = w;
992     v.point = point;
993     __dec_trim(v);
994   }

1002   template<typename F, typename T>
1003   [[gnu::noinline, gnu::cold]] inline constexpr F
1004   __decimal_path(const __scan_t<T> &s) noexcept
1005   {
1006     using TR = __fmt_traits<F>;
1007     __decimal v;
1008     __dec_load(v, s);
1009     if ( v.ndigits == 0 ) return __zero<F>(s.neg);
1010 
1011     const i64 pt = static_cast<i64>(v.point) + s.exp_lit;
1012     if ( pt > TR::largest_power + 2 ) return __inf<F>(s.neg);
1013     if ( pt < TR::smallest_power - 2 ) return __zero<F>(s.neg);
1014     v.point = static_cast<i32>(pt);
1015 
1016     i32 exp2 = 0;
1017     while ( v.ndigits != 0 && v.point > 0 ) {
1018       const i32 n = __dec_step(v.point);
1019       __dec_shift(v, -n);
1020       exp2 += n;
1021     }
1022     while ( v.ndigits != 0 && (v.point < 0 || (v.point == 0 && v.d[0] < 5)) ) {
1023       const i32 n = __dec_step(-v.point);
1024       __dec_shift(v, n);
1025       exp2 -= n;
1026     }
1027     if ( v.ndigits == 0 ) return __zero<F>(s.neg);
1028     --exp2;      // [0.5, 1) -> [1, 2)
1029 
1030     const i32 bias = TR::min_exponent;
1031     if ( exp2 < bias + 1 ) {
1032       const i32 n = bias + 1 - exp2;
1033       __dec_shift(v, -n);
1034       exp2 += n;
1035     }
1036     if ( exp2 - bias >= TR::infinite_power ) return __inf<F>(s.neg);
1037 
1038     __dec_shift(v, TR::mantissa_bits + 1);
1039     u64 mant = __dec_rounded(v);
1040     if ( mant == (2ull << TR::mantissa_bits) ) {      // the rounding carried into a new binade
1041       mant >>= 1;
1042       ++exp2;
1043       if ( exp2 - bias >= TR::infinite_power ) return __inf<F>(s.neg);
1044     }
1045     if ( (mant & (1ull << TR::mantissa_bits)) == 0 ) exp2 = bias;      // stayed subnormal
1046 
1047     return __assemble<F>(s.neg, static_cast<u64>(exp2 - bias), mant & ((1ull << TR::mantissa_bits) - 1));
1048   }

1053   template<typename F>
1054   inline constexpr F
1055   __hex_value(u64 mant, i64 bexp, bool neg, bool sticky) noexcept
1056   {
1057     using TR = __fmt_traits<F>;
1058     constexpr i32 mb = TR::mantissa_bits;
1059     if ( mant == 0 ) return __zero<F>(neg);
1060 
1061     const i32 lz = static_cast<i32>(clz64(mant));
1062     const u64 w = mant << lz;           // bit 63 set
1063     const i64 e2 = bexp + 63 - lz;      // value == 1.f * 2^e2
1064     i64 biased = e2 - TR::min_exponent;
1065 
1066     i32 drop = 63 - mb;
1067     if ( biased <= 0 ) {      // subnormal
1068       const i64 extra = 1 - biased;
1069       if ( extra >= 64 - drop ) return __zero<F>(neg);
1070       drop += static_cast<i32>(extra);
1071       biased = 0;
1072     }
1073 
1074     const u64 rest = w & ((1ull << drop) - 1);
1075     u64 top = w >> drop;
1076     const u64 half = 1ull << (drop - 1);
1077 
1078     bool up = false;
1079     if ( rest > half )
1080       up = true;
1081     else if ( rest == half )
1082       up = sticky || (top & 1) != 0;
1083     if ( up ) ++top;
1084 
1085     if ( top >= (2ull << mb) ) {
1086       top >>= 1;
1087       ++biased;
1088     }
1089     if ( biased >= TR::infinite_power ) return __inf<F>(neg);
1090     if ( biased >= 1 ) top &= ~(1ull << mb);      // at biased == 0 the implicit bit promotes instead
1091     return __assemble<F>(neg, static_cast<u64>(biased), top);
1092   }

1106   template<typename F, bool Bounded, typename T>
1107   inline constexpr __status
1108   __parse(const T *first, const T *last, F &out, const T *&stop) noexcept
1109   {
1110     const __scan_t<T> s = __scan<Bounded>(first, last);
1111     stop = s.stop;
1112 
1113     switch ( s.kind ) {
1114     case __kind::none:
1115       out = __zero<F>(false);      // a total failure is +0, as strtod has it
1116       return __status::invalid;
1117     case __kind::inf:
1118       out = __inf<F>(s.neg);
1119       return __status::ok;
1120     case __kind::nan:
1121       out = __nan<F>(s.neg, s.payload);
1122       return __status::ok;
1123     case __kind::hex:
1124       out = __hex_value<F>(s.mant, s.exp10, s.neg, s.truncated);
1125       return s.mant == 0 ? __status::ok : __range_of(out);
1126     default:
1127       break;
1128     }
1129 
1130     if ( s.mant == 0 ) {      // "0", "0.000", "0e999" and every signed form of them
1131       out = __zero<F>(s.neg);
1132       return __status::ok;
1133     }
1134     if ( !s.truncated && __exact<F>(s.mant, s.exp10, s.neg, out) ) return __status::ok;
1135 
1136     __am a = __eisel_lemire<F>(s.exp10, s.mant);
1137     bool punt = (a.power2 < 0);
1138     if ( !punt && s.truncated ) {
1139       punt = !(a == __eisel_lemire<F>(s.exp10, s.mant + 1));
1140     }
1141     if ( punt )
1142       out = __decimal_path<F>(s);
1143     else
1144       out = __assemble<F>(s.neg, static_cast<u64>(a.power2), a.mantissa);
1145     return __range_of(out);
1146   }
