152   inline __attribute__((always_inline)) int
153   __lexcmp(const T *a, usize alen, const T *b, usize blen) const noexcept
154   {
155     if constexpr ( sizeof(T) != 1 ) return micron::simd::lexcmp_elem<T>(a, alen, b, blen);      // byte memcmp mis-orders wide elems
156     const usize common = alen < blen ? alen : blen;
157     if ( common ) {
158       const i64 d = micron::memcmp<byte>(reinterpret_cast<const byte *>(a), reinterpret_cast<const byte *>(b), common);
159       if ( d != 0 ) return d < 0 ? -1 : 1;
160     }
161     if ( alen < blen ) return -1;
162     if ( alen > blen ) return 1;
163     return 0;
164   }

166   [[gnu::always_inline]] static inline usize
167   __simd_find_byte(const T *p, usize len, T ch) noexcept
168   {
169     if ( len == 0 ) return npos;
170     if constexpr ( sizeof(T) == 1 ) {
171 #if defined(__micron_x86_avx2)
172       const usize i = micron::simd::find_first_set_256(p, len, static_cast<char>(ch));
173 #else
174       const usize i = micron::simd::find_first_set_128(p, len, static_cast<char>(ch));
175 #endif
176       return i == len ? npos : i;
177     } else {
178       const usize i = micron::simd::find_first_elem<T>(p, len, ch);
179       return i == len ? npos : i;
180     }
181   }
