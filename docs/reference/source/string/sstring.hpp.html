242   template<size_type K>
243   [[gnu::always_inline]] static inline size_type
244   __simd_find_first_of_small(const T *p, size_type len, const T *chars, size_type pos) noexcept
245   {
246     static_assert(K >= 1 && K <= 8);
247     if constexpr ( sizeof(T) != 1 ) {
248       const size_type r = micron::simd::find_first_of_elem<T>(p, len, chars, K, pos);
249       return r == len ? npos : r;
250     }
251     size_type i = pos;
252 #if defined(__micron_x86_avx2)
253     __m256i cv[K];
254     for ( size_type k = 0; k < K; ++k ) cv[k] = _mm256_set1_epi8(static_cast<char>(chars[k]));
255     for ( ; i + 32 <= len; i += 32 ) {
256       __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(p + i));
257       __m256i any = _mm256_cmpeq_epi8(v, cv[0]);
258       for ( size_type k = 1; k < K; ++k ) any = _mm256_or_si256(any, _mm256_cmpeq_epi8(v, cv[k]));
259       u32 m = static_cast<u32>(_mm256_movemask_epi8(any));
260       if ( m ) return i + static_cast<size_type>(__builtin_ctz(static_cast<unsigned int>(m)));
261     }
262 #endif
263 #if defined(__micron_x86_sse2)
264     __m128i cv128[K];
265     for ( size_type k = 0; k < K; ++k ) cv128[k] = _mm_set1_epi8(static_cast<char>(chars[k]));
266     for ( ; i + 16 <= len; i += 16 ) {
267       __m128i v = _mm_loadu_si128(reinterpret_cast<const __m128i *>(p + i));
268       __m128i any = _mm_cmpeq_epi8(v, cv128[0]);
269       for ( size_type k = 1; k < K; ++k ) any = _mm_or_si128(any, _mm_cmpeq_epi8(v, cv128[k]));
270       u32 m = static_cast<u32>(_mm_movemask_epi8(any));
271       if ( m ) return i + static_cast<size_type>(__builtin_ctz(static_cast<unsigned int>(m)));
272     }
273 #elif defined(__micron_arm_neon)
274     micron::simd::__bits::uint8x16_t cv_n[K];
275     for ( size_type k = 0; k < K; ++k ) cv_n[k] = vdupq_n_u8(static_cast<u8>(chars[k]));
276     for ( ; i + 16 <= len; i += 16 ) {
277       auto v = vld1q_u8(reinterpret_cast<const u8 *>(p + i));
278       auto any = vceqq_u8(v, cv_n[0]);
279       for ( size_type k = 1; k < K; ++k ) any = vorrq_u8(any, vceqq_u8(v, cv_n[k]));
280       u32 m = micron::simd::__neon_movemask_u8(any);
281       if ( m ) return i + static_cast<size_type>(__builtin_ctz(static_cast<unsigned int>(m)));
282     }
283 #endif
284     for ( ; i < len; ++i )
285       for ( size_type k = 0; k < K; ++k )
286         if ( p[i] == chars[k] ) return i;
287     return npos;
288   }

291   template<size_type K>
292   [[gnu::always_inline]] static inline size_type
293   __simd_rfind_first_of_small(const T *p, size_type end, const T *chars) noexcept
294   {
295     static_assert(K >= 1 && K <= 8);
296     if constexpr ( sizeof(T) != 1 ) {
297       const size_type r = micron::simd::find_last_of_elem<T>(p, end, chars, K);
298       return r == end ? npos : r;
299     }
300     size_type i = end;
301 #if defined(__micron_x86_avx2)
302     __m256i cv[K];
303     for ( size_type k = 0; k < K; ++k ) cv[k] = _mm256_set1_epi8(static_cast<char>(chars[k]));
304     while ( i >= 32 ) {
305       __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(p + i - 32));
306       __m256i any = _mm256_cmpeq_epi8(v, cv[0]);
307       for ( size_type k = 1; k < K; ++k ) any = _mm256_or_si256(any, _mm256_cmpeq_epi8(v, cv[k]));
308       u32 m = static_cast<u32>(_mm256_movemask_epi8(any));
309       if ( m ) {
310         const u32 hi = static_cast<u32>(31 - __builtin_clz(static_cast<unsigned int>(m)));
311         return (i - 32) + hi;
312       }
313       i -= 32;
314     }
315 #endif
316 #if defined(__micron_x86_sse2)
317     __m128i cv128[K];
318     for ( size_type k = 0; k < K; ++k ) cv128[k] = _mm_set1_epi8(static_cast<char>(chars[k]));
319     while ( i >= 16 ) {
320       __m128i v = _mm_loadu_si128(reinterpret_cast<const __m128i *>(p + i - 16));
321       __m128i any = _mm_cmpeq_epi8(v, cv128[0]);
322       for ( size_type k = 1; k < K; ++k ) any = _mm_or_si128(any, _mm_cmpeq_epi8(v, cv128[k]));
323       u32 m = static_cast<u32>(_mm_movemask_epi8(any)) & 0xFFFFu;
324       if ( m ) {
325         const u32 hi = static_cast<u32>(31 - __builtin_clz(static_cast<unsigned int>(m)));
326         return (i - 16) + hi;
327       }
328       i -= 16;
329     }
330 #elif defined(__micron_arm_neon)
331     micron::simd::__bits::uint8x16_t cv_n[K];
332     for ( size_type k = 0; k < K; ++k ) cv_n[k] = vdupq_n_u8(static_cast<u8>(chars[k]));
333     while ( i >= 16 ) {
334       auto v = vld1q_u8(reinterpret_cast<const u8 *>(p + i - 16));
335       auto any = vceqq_u8(v, cv_n[0]);
336       for ( size_type k = 1; k < K; ++k ) any = vorrq_u8(any, vceqq_u8(v, cv_n[k]));
337       u32 m = micron::simd::__neon_movemask_u8(any);
338       if ( m ) {
339         const u32 hi = 31u - __builtin_clz(m & 0xFFFFu);
340         return (i - 16) + hi;
341       }
342       i -= 16;
343     }
344 #endif
345     while ( i > 0 ) {
346       --i;
347       for ( size_type k = 0; k < K; ++k )
348         if ( p[i] == chars[k] ) return i;
349     }
350     return npos;
351   }

353   [[gnu::always_inline]] static inline size_type
354   __simd_count_substr(const T *hay, size_type hlen, const T *needle, size_type nlen) noexcept
355   {
356     if ( nlen == 0 || nlen > hlen ) return 0;
357     if constexpr ( sizeof(T) != 1 ) {
358       size_type cnt = 0, p = 0;
359       while ( p + nlen <= hlen ) {
360         const size_type f = micron::simd::find_substr_elem<T>(hay + p, hlen - p, needle, nlen);
361         if ( f == hlen - p ) break;
362         ++cnt;
363         p += f + nlen;      // non-overlapping
364       }
365       return cnt;
366     }
367     if ( nlen == 1 ) {
368 #if defined(__micron_x86_avx2)
369       return micron::simd::count_set_256(hay, hlen, static_cast<char>(needle[0]));
370 #else
371       return micron::simd::count_set_128(hay, hlen, static_cast<char>(needle[0]));
372 #endif
373     }
374     size_type n = 0, pos = 0;
375     const size_type limit = hlen - nlen + 1;
376 #if defined(__micron_x86_avx2)
377     const __m256i first = _mm256_set1_epi8(static_cast<char>(needle[0]));
378     while ( pos + 32 <= limit ) {
379       __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(hay + pos));
380       u32 mask = static_cast<u32>(_mm256_movemask_epi8(_mm256_cmpeq_epi8(v, first)));
381       while ( mask ) {
382         const u32 bit = static_cast<u32>(__builtin_ctz(static_cast<unsigned int>(mask)));
383         const size_type cand = pos + static_cast<size_type>(bit);
384         bool match
385             = (nlen == 1)
386               || (micron::memcmp<byte>(reinterpret_cast<const byte *>(hay + cand + 1), reinterpret_cast<const byte *>(needle + 1), nlen - 1)
387                   == 0);
388         if ( match ) {
389           ++n;
390           const size_type next = cand + nlen;
391           if ( next < pos + 32 ) {
392             mask &= ~static_cast<u32>((1u << (next - pos)) - 1u);
393           } else {
394             pos = next;
395             goto sstring_count_substr_next_avx;
396           }
397         } else {
398           mask &= mask - 1u;
399         }
400       }
401       pos += 32;
402     sstring_count_substr_next_avx:;
403     }
404 #elif defined(__micron_arm_neon)
405     const micron::simd::__bits::uint8x16_t first = vdupq_n_u8(static_cast<u8>(needle[0]));
406     while ( pos + 16 <= limit ) {
407       auto v = vld1q_u8(reinterpret_cast<const u8 *>(hay + pos));
408       u32 mask = micron::simd::__neon_movemask_u8(vceqq_u8(v, first));
409       while ( mask ) {
410         const u32 bit = static_cast<u32>(__builtin_ctz(static_cast<unsigned int>(mask)));
411         const size_type cand = pos + bit;
412         bool match
413             = (nlen == 1)
414               || (micron::memcmp<byte>(reinterpret_cast<const byte *>(hay + cand + 1), reinterpret_cast<const byte *>(needle + 1), nlen - 1)
415                   == 0);
416         if ( match ) {
417           ++n;
418           const size_type next = cand + nlen;
419           if ( next < pos + 16 ) {
420             mask &= ~static_cast<u32>((1u << (next - pos)) - 1u);
421           } else {
422             pos = next;
423             goto sstring_count_substr_next_neon;
424           }
425         } else {
426           mask &= mask - 1u;
427         }
428       }
429       pos += 16;
430     sstring_count_substr_next_neon:;
431     }
432 #endif
433     while ( pos < limit ) {
434       bool match = true;
435       for ( size_type j = 0; j < nlen; ++j )
436         if ( hay[pos + j] != needle[j] ) {
437           match = false;
438           break;
439         }
440       if ( match ) {
441         ++n;
442         pos += nlen;
443       } else {
444         ++pos;
445       }
446     }
447     return n;
448   }
