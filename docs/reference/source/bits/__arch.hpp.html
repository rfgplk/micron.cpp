605 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
606 // __micron_no_ssp
607 //
608 // suppress the stack-protector prologue on a single function
609 //
610 // WARNING: __attribute__((naked)) does NOT imply "no compiler-generated prologue"
611 // under -fstack-protector-all gcc still prepends the canary spill (?!?!?!?!?):
612 //
613 //     ldr r3, [pc, #..]   ; &__stack_chk_guard
614 //     ldr r3, [r3]
615 //     str r3, [sp, #4]    ; <-- ABOVE sp: a naked fn reserved no frame
616 //     <the naked body>
617 //
618 // on x86 gcc emits no canary for naked fns and the store would land in the 128b red
619 // zone anyway, which is why amd64 never noticed
620 #if defined(__has_attribute)
621 #if __has_attribute(no_stack_protector)
622 #define __micron_no_ssp __attribute__((no_stack_protector))
623 #endif
624 #endif
625 #if !defined(__micron_no_ssp)
626 // gcc < 11 / clang < 12: the historical workaround. costs nothing here because every
627 // naked fn in the tree is also noinline, so losing the inline-into-caller opts is moot
628 #if defined(__micron_compiler_gcc)
629 #define __micron_no_ssp __attribute__((optimize("no-stack-protector")))
630 #else
631 #define __micron_no_ssp
632 #endif
633 #endif
