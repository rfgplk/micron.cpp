30 // shared container bounds/state predicates
31 // NOTE: this must remain an empty base
32 template<typename Derived, typename T, bool Sf> class __container_checks
33 {
34 protected:
35   [[gnu::always_inline]] inline const Derived *
36   __self(void) const noexcept
37   {
38     return static_cast<const Derived *>(this);
39   }
40 
41   // enforce this as noinline, saves I-cache and binary size
42   template<typename E>
43   [[gnu::noinline, gnu::cold, noreturn]] static void
44   __fail(const char *msg)
45   {
46     exc<E>(msg);
47   }
48 
49   [[gnu::always_inline]] inline bool
50   __empty_check(void) const
51   {
52     return __self()->__c_len() == 0 || __self()->__c_data() == nullptr;
53   }
54 
55   [[gnu::always_inline]] inline bool
56   __index_check(usize n) const
57   {
58     return n >= __self()->__c_len();
59   }
60 
61   [[gnu::always_inline]] inline bool
62   __capacity_check(usize n) const
63   {
64     return n >= __self()->__c_cap();
65   }
66 
67   // alias of __index_check
68   [[gnu::always_inline]] inline bool
69   __get_check(usize n) const
70   {
71     return n >= __self()->__c_len();
72   }
73 
74   [[gnu::always_inline]] inline bool
75   __iterator_check(const T *it) const
76   {
77     return it < __self()->__c_data() || it > __self()->__c_data() + __self()->__c_len();
78   }
79 
80   [[gnu::always_inline]] inline bool
81   __range_check(usize from, usize to) const
82   {
83     return from >= to || to > __self()->__c_len();
84   }
85 
86   // deliberately over capacity, not length
87   [[gnu::always_inline]] inline bool
88   __cap_range_check(usize from, usize to) const
89   {
90     return from >= to || to > __self()->__c_cap();
91   }
92 };

95 // shared heap vector core

97 // NOTE: this must remain an empty base
98 template<typename Derived, typename T> class __vector_core
99 {
100 protected:
101   [[gnu::always_inline]] inline Derived *
102   __vself(void) noexcept
103   {
104     return static_cast<Derived *>(this);
105   }
106 
107   [[gnu::always_inline]] inline const Derived *
108   __vself(void) const noexcept
109   {
110     return static_cast<const Derived *>(this);
111   }
112 
113   // must be noinline
114   [[gnu::noinline, gnu::cold]] void
115   __grow_one(void)
116   {
117     Derived *s = __vself();
118     if ( s->capacity == static_cast<usize>(-1) ) exc<except::length_error>("container: capacity overflow");
119     s->__core_reserve(s->recommended_capacity(s->capacity, s->capacity + 1));
120   }
121 
122   [[gnu::noinline, gnu::cold]] void
123   __grow_for(usize extra)
124   {
125     Derived *s = __vself();
126     if ( extra > static_cast<usize>(-1) - s->length ) exc<except::length_error>("container: size overflow");
127     const usize minimum = s->length + extra;
128     s->__core_reserve(s->recommended_capacity(s->capacity, minimum));
129   }
130 
131   template<typename U>
132   [[gnu::always_inline]] inline void
133   __emplace_at(usize i, U &&v)
134   {
135     Derived *s = __vself();
136     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_copyable_v<T> )
137       new (addr(s->memory[i])) T(micron::forward<U>(v));
138     else
139       s->memory[i] = micron::forward<U>(v);
140   }
141 
142 public:
143   [[gnu::always_inline]] inline void
144   push_back(const T &v)
145   {
146     Derived *s = __vself();
147     if ( s->length < s->capacity ) [[likely]] {
148       __emplace_at(s->length++, v);
149       return;
150     }
151     __grow_one();
152     __emplace_at(s->length++, v);
153   }
154 
155   [[gnu::always_inline]] inline void
156   push_back(T &&v)
157   {
158     Derived *s = __vself();
159     if ( s->length < s->capacity ) [[likely]] {
160       __emplace_at(s->length++, micron::move(v));
161       return;
162     }
163     __grow_one();
164     __emplace_at(s->length++, micron::move(v));
165   }
166 
167   template<typename... Args>
168   [[gnu::always_inline]] inline void
169   emplace_back(Args &&...v)
170   {
171     Derived *s = __vself();
172     if ( s->length < s->capacity ) [[likely]] {
173       new (addr(s->memory[s->length++])) T(micron::forward<Args>(v)...);
174       return;
175     }
176     __grow_one();
177     new (addr(s->memory[s->length++])) T(micron::forward<Args>(v)...);
178   }
179 };

211 inline void
212 deep_copy(T *__restrict dest, T *__restrict src, usize cnt)
213 {
214   // guard against copying into uninit memory
215   if constexpr ( noexcept(T(micron::declval<const T &>())) ) {
216     for ( usize i = 0; i < cnt; ++i ) new (addr(dest[i])) T(src[i]);
217   } else {
218     usize i = 0;
219 #if !defined(__micron_freestanding) || defined(__micron_eh)
220     try {
221       for ( ; i < cnt; ++i ) new (addr(dest[i])) T(src[i]);
222     } catch ( ... ) {
223       for ( usize j = 0; j < i; ++j ) dest[j].~T();
224       throw;
225     }
226 #else
227     for ( ; i < cnt; ++i ) new (addr(dest[i])) T(src[i]);
228 #endif
229   }

611 // linear scan
612 template<typename T>
613 [[gnu::always_inline]] inline usize
614 find_index(const T *p, usize len, const T &v) noexcept
615 {
616   using U = micron::remove_cv_t<T>;
617   if constexpr ( (micron::is_integral_v<U> or micron::is_enum_v<U> or micron::is_pointer_v<U>)
618                  and (sizeof(U) == 1 or sizeof(U) == 2 or sizeof(U) == 4 or sizeof(U) == 8) ) {
619     return micron::simd::find_first_elem(p, len, v);
620   } else {
621     for ( usize i = 0; i < len; ++i )
622       if ( p[i] == v ) return i;
623     return len;
624   }
625 }

627 template<typename T>
628 inline void
629 open_gap(T *base, usize len, usize p, usize cnt)
630 {
631   if ( cnt == 0 ) return;
632   if constexpr ( micron::is_trivially_copyable_v<micron::remove_cv_t<T>> ) {
633     micron::memmove(base + p + cnt, base + p, len - p);
634   } else {
635     const usize n = len - p;
636     for ( usize i = n; i-- > 0; ) {
637       if ( p + cnt + i >= len )
638         new (addr(base[p + cnt + i])) T(micron::move(base[p + i]));
639       else
640         base[p + cnt + i] = micron::move(base[p + i]);
641     }
642     const usize raw = (cnt < n) ? cnt : n;
643     for ( usize i = 0; i < raw; ++i ) base[p + i].~T();
644   }
645 }

647 // NOTE: assumes T's move-ctor does not throw during the tail slide
648 template<typename T, typename Fn>
649 inline void
650 fill_gap(T *base, usize length, usize p, usize cnt, Fn &&fn)
651 {
652   if ( cnt == 0 ) return;
653 #if !defined(__micron_freestanding) || defined(__micron_eh)
654   usize k = 0;
655   try {
656     for ( ; k < cnt; ++k ) fn(p + k);
657   } catch ( ... ) {
658     for ( usize j = 0; j < k; ++j ) base[p + j].~T();
659     const usize tail = length - p;
660     for ( usize j = 0; j < tail; ++j ) {
661       new (addr(base[p + j])) T(micron::move(base[p + cnt + j]));
662       base[p + cnt + j].~T();
663     }
664     throw;
665   }
666 #else
667   // surpress may be unused
668   (void)base;
669   (void)length;
670   for ( usize k = 0; k < cnt; ++k ) fn(p + k);
671 #endif
672 }

676 fill_gap_copy(T *base, usize length, usize p, usize cnt, const T &val)
677 {
678   if constexpr ( micron::is_trivially_copyable_v<micron::remove_cv_t<T>> and micron::is_trivially_constructible_v<T, const T &> ) {
679     if ( cnt == 0 ) return;
680     micron::typeset(base + p, val, cnt);
681     return;
682   } else {
683     fill_gap(base, length, p, cnt, [&](usize i) { new (addr(base[i])) T(val); });
684   }
685 }

687 template<typename T>
688 inline void
689 close_gap(T *base, usize len, usize p, usize cnt)
690 {
691   if ( cnt == 0 ) return;
692   if constexpr ( micron::is_trivially_copyable_v<micron::remove_cv_t<T>> ) {
693     micron::memmove(base + p, base + p + cnt, len - p - cnt);
694   } else {
695     const usize tail = len - p - cnt;
696     for ( usize i = 0; i < tail; ++i ) base[p + i] = micron::move(base[p + cnt + i]);
697     for ( usize i = len - cnt; i < len; ++i ) base[i].~T();
698     micron::byteset(reinterpret_cast<byte *>(base + (len - cnt)), 0x0, cnt * sizeof(T));
699   }
700 }

702 template<usize N, typename T>
703 inline void
704 destroy(T *src)
705 {
706   // avoid zero count UB
707   if constexpr ( N == 0 ) return;
708   if constexpr ( !micron::is_trivially_destructible_v<micron::remove_cv_t<T>> ) {
709     if constexpr ( N % 4 == 0 ) {
710       for ( usize i = 0; i < N; i += 4 ) {
711         src[i].~T();
712         src[i + 1].~T();
713         src[i + 2].~T();
714         src[i + 3].~T();
715       }
716     } else {
717       for ( usize i = 0; i < N; ++i ) src[i].~T();
718     }
719     // WARNING:after calling destructors, manually zero it out, wasteful but necessary for edge case correctness
720     // in case you're wondering why, there seem to be edge cases when applying aggressive optimization options where the compiler opts NOT
721     // to init certain memory (if allocing large ptr lists) causing the memory to be left in a garbled state. for OUR classes ~T() WILL
722     // destroy the objects BUT wont if they are in a garbled state (nothing to destroy). meaning destruction was technically valid, even
723     // though nothing was destroyed. ergo best to manually zero the memory afterwards
724     micron::cbyteset<N * sizeof(T)>(src, 0x0);
725   } else {
726     micron::cbyteset<N * sizeof(T)>(src, 0x0);
727   }
728 }

730 template<typename T>
731 inline void
732 destroy(T *__restrict src, usize cnt)
733 {
734   if ( cnt == 0 ) return;
735   if constexpr ( !micron::is_trivially_destructible_v<micron::remove_cv_t<T>> ) {
736     for ( usize i = 0; i < cnt; ++i ) src[i].~T();
737     // WARNING: after calling destructors, manually zero it out, wasteful but necessary for edge case correctness
738     micron::byteset(src, 0x0, cnt * sizeof(T));
739   } else {
740     micron::byteset(src, 0x0, cnt * sizeof(T));
741   }
742   // NOTE: the zeroing here is part of destroy()'s semantics, ie micron::array::clear() calls destroy and then skips reconstructing a
743   // trivially-constructible T; if you want the fast and cheap variant call destroy_fast()
744 }

818 template<usize N, typename T>
819 void
820 construct(T *__restrict src, const T &val)
821 {
822   if constexpr ( N == 0 ) return;
823   if constexpr ( !micron::is_trivially_constructible_v<T, const T &> ) {
824     if constexpr ( noexcept(T(micron::declval<const T &>())) ) {
825       if constexpr ( N % 4 == 0 ) {
826         for ( usize i = 0; i < N; i += 4 ) {
827           new (addr(src[i])) T(val);
828           new (addr(src[i + 1])) T(val);
829           new (addr(src[i + 2])) T(val);
830           new (addr(src[i + 3])) T(val);
831         }
832       } else {
833         for ( usize i = 0; i < N; ++i ) new (addr(src[i])) T(val);
834       }
835     } else {
836       // throwing element ctor: unwind the already-built prefix on throw
837       usize i = 0;
838 #if !defined(__micron_freestanding) || defined(__micron_eh)
839       try {
840         for ( ; i < N; ++i ) new (addr(src[i])) T(val);
841       } catch ( ... ) {
842         for ( usize j = 0; j < i; ++j ) src[j].~T();
843         throw;
844       }
845 #else
846       for ( ; i < N; ++i ) new (addr(src[i])) T(val);
847 #endif
848     }
849   } else {
850     micron::ctypeset<N>(src, val);
851   }
852 }
