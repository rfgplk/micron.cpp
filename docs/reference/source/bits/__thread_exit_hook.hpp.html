25 // per-thread C++ dtor list (attach only)
26 //
27 // guest modules don't have _start, so its thread_local objects with non-trivial
28 // dtors route their __cxa_thread_atexit registrations here
29 #if defined(MICRON_ATTACH_MODULE)
30 inline bool
31 __push_thread_dtor(void (*dtor)(void *), void *obj) noexcept
32 {
33   if ( __micron_attach_thread_atexit == nullptr ) return false;
34   return __micron_attach_thread_atexit(dtor, obj) == 0;
35 }

37 inline void
38 __run_thread_dtors() noexcept
39 {
40   if ( __micron_attach_run_thread_dtors ) __micron_attach_run_thread_dtors();
41 }
42 #elif defined(__micron_attach_capable) || defined(__micron_freestanding)
43 #define __micron_tdtor_real 1

45 struct __tdtor_node {
46   void (*dtor)(void *);
47   void *obj;
48 };
49 
50 #ifndef MICRON_TDTOR_CAP
51 #define MICRON_TDTOR_CAP 128
52 #endif
53 inline constexpr u32 __tdtor_cap = MICRON_TDTOR_CAP;
54 static_assert(__tdtor_cap >= 8, "micron: MICRON_TDTOR_CAP must leave room for the runtime's own thread_locals.");
55 
56 // WARNING: thread_local, so this costs __tdtor_cap * 2 * sizeof(void*) of PT_TLS p_memsz on every thread (2 KiB)
57 inline constinit thread_local __tdtor_node __tdtors[__tdtor_cap]{};
58 inline constinit thread_local u32 __tdtor_n = 0;
59 
60 // registrations lost to a full list; Itanium ABI doesn't provide for a way to see/recover dropped dtors
61 inline u32 __tdtor_dropped = 0;
62 
63 inline bool
64 __push_thread_dtor(void (*dtor)(void *), void *obj) noexcept
65 {
66   if ( __tdtor_n >= __tdtor_cap ) {
67     __atomic_fetch_add(&__tdtor_dropped, 1u, __ATOMIC_RELAXED);
68     return false;
69   }
70   __tdtors[__tdtor_n].dtor = dtor;
71   __tdtors[__tdtor_n].obj = obj;
72   ++__tdtor_n;
73   return true;
74 }
75 
76 // WARNING: do not rewrite this as a counted for-loop; that reintroduces the push during drain reg
77 inline void
78 __run_thread_dtors() noexcept
79 {
80   while ( __tdtor_n ) {
81     --__tdtor_n;
82     __tdtor_node e = __tdtors[__tdtor_n];
83     if ( e.dtor ) e.dtor(e.obj);
84   }
85 }

89 inline int
90 __attach_thread_atexit(void (*dtor)(void *), void *obj) noexcept

112 // native control word states
113 inline constexpr u32 __park_run = 0;         // running
114 inline constexpr u32 __park_parked = 1;      // sleep(): block at the next checkpoint until awaken()
115 inline constexpr u32 __park_dying = 2;       // terminate()/cancel(): exit at the next checkpoint
116 
117 // the running thread's control word (== &__thread_payload::park)
118 inline thread_local u32 *__micron_thread_park = nullptr;
119 #if defined(__micron_arch_wasm32)
120 inline thread_local void (*__micron_thread_die)() noexcept = nullptr;
121 #else
122 inline thread_local void (*__micron_thread_die)() = nullptr;

125 inline __attribute__((always_inline)) void
126 __micron_park_checkpoint() noexcept
127 {
128   u32 *p = __micron_thread_park;
129   if ( p == nullptr ) return;
130   u32 v = __atomic_load_n(p, __ATOMIC_ACQUIRE);
131   while ( v == __park_parked ) {
132     // NOTE: deliberately the RAW syscall, not micron::__futex, abcmalloc pulls this in
133 #if defined(__micron_arch_wasm32)
134     (void)wasm::wait32(p, __park_parked);
135 #else
136     micron::syscall(SYS_futex, p, 128 /*FUTEX_WAIT|FUTEX_PRIVATE_FLAG*/, __park_parked, nullptr, nullptr, 0);
137 #endif
138     v = __atomic_load_n(p, __ATOMIC_ACQUIRE);
139   }
140   if ( v == __park_dying && __micron_thread_die ) __micron_thread_die();
141 }
