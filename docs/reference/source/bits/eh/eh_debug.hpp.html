18 inline void
19 __dbg_s(const char *s) noexcept
20 {
21 #if defined(__MICRON_EH_DEBUG)
22   usize n = 0;
23   while ( s[n] ) ++n;
24   micron::syscall(SYS_write, 1, reinterpret_cast<const void *>(s), n);
25 #else
26   (void)s;
27 #endif
28 }
29 
30 inline void
31 __dbg_h(usize v) noexcept
32 {
33 #if defined(__MICRON_EH_DEBUG)
34   char buf[19];
35   buf[0] = '0';
36   buf[1] = 'x';
37   for ( int i = 0; i < 16; ++i ) {
38     const unsigned nyb = (v >> ((15 - i) * 4)) & 0xf;
39     buf[2 + i] = static_cast<char>(nyb < 10 ? '0' + nyb : 'a' + (nyb - 10));
40   }
41   buf[18] = '\n';
42   micron::syscall(SYS_write, 1, reinterpret_cast<const void *>(buf), 19);
43 #else
44   (void)v;
45 #endif
46 }

48 inline void
49 __dbg_kv(const char *k, usize v) noexcept
50 {
51 #if defined(__MICRON_EH_DEBUG)
52   __dbg_s(k);
53   __dbg_h(v);
54 #else
55   (void)k;
56   (void)v;
57 #endif
58 }
