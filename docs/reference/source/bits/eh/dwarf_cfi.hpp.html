53   DW_OP_deref = 0x06,
54   DW_OP_const1u = 0x08,
55   DW_OP_const1s = 0x09,

57   DW_OP_const2s = 0x0b,
58   DW_OP_const4u = 0x0c,
59   DW_OP_const4s = 0x0d,
60   DW_OP_const8u = 0x0e,

86 inline usize
87 eval_dwarf_expr(const byte *p, const byte *end, const reg_context &ctx, usize initial, bool push_initial) noexcept
88 {
89   usize stk[64];
90   int sp = 0;
91   if ( push_initial ) stk[sp++] = initial;
92 
93   while ( p < end ) {
94     const u8 op = *p++;
95     if ( op >= DW_OP_lit0 && op <= DW_OP_lit31 ) {
96       stk[sp++] = op - DW_OP_lit0;
97     } else if ( op >= DW_OP_breg0 && op <= DW_OP_breg31 ) {
98       const i64 off = read_sleb128(p);
99       const unsigned r = op - DW_OP_breg0;
100       stk[sp++] = ctx.reg[r < DWARF_MAX_REG ? r : 0] + static_cast<usize>(off);
101     } else if ( op >= DW_OP_reg0 && op <= DW_OP_reg31 ) {
102       const unsigned r = op - DW_OP_reg0;
103       stk[sp++] = ctx.reg[r < DWARF_MAX_REG ? r : 0];
104     } else {
105       switch ( op ) {
106       case DW_OP_addr:
107         stk[sp++] = read_unaligned<usize>(p);
108         p += sizeof(usize);
109         break;
110       case DW_OP_const1u:
111         stk[sp++] = *p++;
112         break;
113       case DW_OP_const1s:
114         stk[sp++] = static_cast<usize>(static_cast<intptr_t>(static_cast<i8>(*p++)));
115         break;
116       case DW_OP_const2u:
117         stk[sp++] = read_unaligned<u16>(p);
118         p += 2;
119         break;
120       case DW_OP_const2s:
121         stk[sp++] = static_cast<usize>(static_cast<intptr_t>(read_unaligned<i16>(p)));
122         p += 2;
123         break;
124       case DW_OP_const4u:
125         stk[sp++] = read_unaligned<u32>(p);
126         p += 4;
127         break;
128       case DW_OP_const4s:
129         stk[sp++] = static_cast<usize>(static_cast<intptr_t>(read_unaligned<i32>(p)));
130         p += 4;
131         break;
132       case DW_OP_const8u:
133       case DW_OP_const8s:
134         stk[sp++] = static_cast<usize>(read_unaligned<u64>(p));
135         p += 8;
136         break;
137       case DW_OP_constu:
138         stk[sp++] = static_cast<usize>(read_uleb128(p));
139         break;
140       case DW_OP_consts:
141         stk[sp++] = static_cast<usize>(read_sleb128(p));
142         break;
143       case DW_OP_dup:
144         stk[sp] = stk[sp - 1];
145         ++sp;
146         break;
147       case DW_OP_drop:
148         --sp;
149         break;
150       case DW_OP_over:
151         stk[sp] = stk[sp - 2];
152         ++sp;
153         break;
154       case DW_OP_swap: {
155         usize t = stk[sp - 1];
156         stk[sp - 1] = stk[sp - 2];
157         stk[sp - 2] = t;
158         break;
159       }
160       case DW_OP_deref:
161         stk[sp - 1] = read_unaligned<usize>(reinterpret_cast<const byte *>(stk[sp - 1]));
162         break;
163       case DW_OP_plus:
164         stk[sp - 2] += stk[sp - 1];
165         --sp;
166         break;
167       case DW_OP_minus:
168         stk[sp - 2] -= stk[sp - 1];
169         --sp;
170         break;
171       case DW_OP_and:
172         stk[sp - 2] &= stk[sp - 1];
173         --sp;
174         break;
175       case DW_OP_or:
176         stk[sp - 2] |= stk[sp - 1];
177         --sp;
178         break;
179       case DW_OP_shl:
180         stk[sp - 2] <<= stk[sp - 1];
181         --sp;
182         break;
183       case DW_OP_shr:
184         stk[sp - 2] >>= stk[sp - 1];
185         --sp;
186         break;
187       case DW_OP_plus_uconst:
188         stk[sp - 1] += static_cast<usize>(read_uleb128(p));
189         break;
190       case DW_OP_regx:
191         stk[sp++] = ctx.reg[read_uleb128(p) % DWARF_MAX_REG];
192         break;
193       case DW_OP_bregx: {
194         const unsigned r = static_cast<unsigned>(read_uleb128(p));
195         const i64 off = read_sleb128(p);
196         stk[sp++] = ctx.reg[r < DWARF_MAX_REG ? r : 0] + static_cast<usize>(off);
197         break;
198       }
199       case DW_OP_nop:
200       default:
201         break;
202       }
203     }
204     if ( sp <= 0 || sp >= 63 ) break;      // guard
205   }
206   return sp > 0 ? stk[sp - 1] : 0;
207 }

209 // CFI rule table
210 enum rule_kind : u8 {
211   rule_undefined = 0,      // unknown -> keep current value (callee preserved it)
212   rule_same,
213   rule_offset,             // value at *(CFA + off)
214   rule_val_offset,         // value is CFA + off
215   rule_register,           // value is reg[which]
216   rule_expression,         // value at *(eval(expr, initial=CFA))
217   rule_val_expression      // value is eval(expr, initial=CFA)
218 };

265 // run the CIE+FDE CFI program up to pc_target
266 inline void
267 run_cfi(const fde_info &fde, usize pc_target, cfi_row &row) noexcept
268 {
269   for ( int i = 0; i < DWARF_MAX_REG; ++i ) row.rules[i] = reg_rule{ rule_undefined, 0, 0, nullptr, nullptr };
270   row.cfa_is_expr = false;
271   row.cfa_reg = 0;
272   row.cfa_off = 0;
273   row.cfa_expr = nullptr;
274   row.cfa_expr_end = nullptr;
275 
276   const u64 caf = fde.cie.code_align;
277   const i64 daf = fde.cie.data_align;
278   const decode_ctx ctx{ 0, __eh_state().text_base, __eh_state().data_base, fde.func_start };
279 
280   // remember/restore stack
281   cfi_row stack[8];
282   int stack_sp = 0;
283   cfi_row initial;      // snapshot of rules after the CIE program (for DW_CFA_restore)
284   bool have_initial = false;
285 
286   auto run = [&](const byte *p, const byte *end, usize start_loc) {
287     usize loc = start_loc;
288     while ( p < end ) {
289       const u8 op = *p++;
290       const u8 hi = op & 0xc0;
291       if ( hi == 0x40 ) {      // DW_CFA_advance_loc
292         loc += static_cast<usize>(op & 0x3f) * caf;
293         if ( loc > pc_target ) return;
294         continue;
295       }
296       if ( hi == 0x80 ) {      // DW_CFA_offset
297         const u32 r = op & 0x3f;
298         const u64 o = read_uleb128(p);
299         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_offset, static_cast<i64>(o) * daf, 0, nullptr, nullptr };
300         continue;
301       }
302       if ( hi == 0xc0 ) {      // DW_CFA_restore
303         const u32 r = op & 0x3f;
304         if ( r < DWARF_MAX_REG ) row.rules[r] = have_initial ? initial.rules[r] : reg_rule{ rule_undefined, 0, 0, nullptr, nullptr };
305         continue;
306       }
307       switch ( op ) {
308       case DW_CFA_nop:
309         break;
310       case DW_CFA_set_loc: {
311         const byte *q = p;
312         loc = read_encoded(fde.cie.fde_enc, q, ctx);
313         p = q;
314         if ( loc > pc_target ) return;
315         break;
316       }
317       case DW_CFA_advance_loc1:
318         loc += static_cast<usize>(*p++) * caf;
319         if ( loc > pc_target ) return;
320         break;
321       case DW_CFA_advance_loc2:
322         loc += static_cast<usize>(read_unaligned<u16>(p)) * caf;
323         p += 2;
324         if ( loc > pc_target ) return;
325         break;
326       case DW_CFA_advance_loc4:
327         loc += static_cast<usize>(read_unaligned<u32>(p)) * caf;
328         p += 4;
329         if ( loc > pc_target ) return;
330         break;
331       case DW_CFA_offset_extended: {
332         const u32 r = static_cast<u32>(read_uleb128(p));
333         const u64 o = read_uleb128(p);
334         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_offset, static_cast<i64>(o) * daf, 0, nullptr, nullptr };
335         break;
336       }
337       case DW_CFA_offset_extended_sf: {
338         const u32 r = static_cast<u32>(read_uleb128(p));
339         const i64 o = read_sleb128(p);
340         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_offset, o * daf, 0, nullptr, nullptr };
341         break;
342       }
343       case DW_CFA_GNU_negative_offset_extended: {
344         const u32 r = static_cast<u32>(read_uleb128(p));
345         const u64 o = read_uleb128(p);
346         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_offset, -static_cast<i64>(o) * daf, 0, nullptr, nullptr };
347         break;
348       }
349       case DW_CFA_val_offset: {
350         const u32 r = static_cast<u32>(read_uleb128(p));
351         const u64 o = read_uleb128(p);
352         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_val_offset, static_cast<i64>(o) * daf, 0, nullptr, nullptr };
353         break;
354       }
355       case DW_CFA_val_offset_sf: {
356         const u32 r = static_cast<u32>(read_uleb128(p));
357         const i64 o = read_sleb128(p);
358         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_val_offset, o * daf, 0, nullptr, nullptr };
359         break;
360       }
361       case DW_CFA_restore_extended: {
362         const u32 r = static_cast<u32>(read_uleb128(p));
363         if ( r < DWARF_MAX_REG ) row.rules[r] = have_initial ? initial.rules[r] : reg_rule{ rule_undefined, 0, 0, nullptr, nullptr };
364         break;
365       }
366       case DW_CFA_undefined: {
367         const u32 r = static_cast<u32>(read_uleb128(p));
368         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_undefined, 0, 0, nullptr, nullptr };
369         break;
370       }
371       case DW_CFA_same_value: {
372         const u32 r = static_cast<u32>(read_uleb128(p));
373         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_same, 0, 0, nullptr, nullptr };
374         break;
375       }
376       case DW_CFA_register: {
377         const u32 r = static_cast<u32>(read_uleb128(p));
378         const u32 r2 = static_cast<u32>(read_uleb128(p));
379         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_register, 0, r2, nullptr, nullptr };
380         break;
381       }
382       case DW_CFA_remember_state:
383         if ( stack_sp < 8 ) stack[stack_sp++] = row;
384         break;
385       case DW_CFA_restore_state:
386         if ( stack_sp > 0 ) {
387           cfi_row saved = stack[--stack_sp];
388           // restore only the register rules + CFA
389           row.cfa_is_expr = saved.cfa_is_expr;
390           row.cfa_reg = saved.cfa_reg;
391           row.cfa_off = saved.cfa_off;
392           row.cfa_expr = saved.cfa_expr;
393           row.cfa_expr_end = saved.cfa_expr_end;
394           for ( int i = 0; i < DWARF_MAX_REG; ++i ) row.rules[i] = saved.rules[i];
395         }
396         break;
397       case DW_CFA_def_cfa: {
398         row.cfa_is_expr = false;
399         row.cfa_reg = static_cast<u32>(read_uleb128(p));
400         row.cfa_off = static_cast<i64>(read_uleb128(p));
401         break;
402       }
403       case DW_CFA_def_cfa_sf: {
404         row.cfa_is_expr = false;
405         row.cfa_reg = static_cast<u32>(read_uleb128(p));
406         row.cfa_off = read_sleb128(p) * daf;
407         break;
408       }
409       case DW_CFA_def_cfa_register:
410         row.cfa_reg = static_cast<u32>(read_uleb128(p));
411         row.cfa_is_expr = false;
412         break;
413       case DW_CFA_def_cfa_offset:
414         row.cfa_off = static_cast<i64>(read_uleb128(p));
415         row.cfa_is_expr = false;
416         break;
417       case DW_CFA_def_cfa_offset_sf:
418         row.cfa_off = read_sleb128(p) * daf;
419         row.cfa_is_expr = false;
420         break;
421       case DW_CFA_def_cfa_expression: {
422         const u64 len = read_uleb128(p);
423         row.cfa_is_expr = true;
424         row.cfa_expr = p;
425         row.cfa_expr_end = p + len;
426         p += len;
427         break;
428       }
429       case DW_CFA_expression: {
430         const u32 r = static_cast<u32>(read_uleb128(p));
431         const u64 len = read_uleb128(p);
432         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_expression, 0, 0, p, p + len };
433         p += len;
434         break;
435       }
436       case DW_CFA_val_expression: {
437         const u32 r = static_cast<u32>(read_uleb128(p));
438         const u64 len = read_uleb128(p);
439         if ( r < DWARF_MAX_REG ) row.rules[r] = reg_rule{ rule_val_expression, 0, 0, p, p + len };
440         p += len;
441         break;
442       }
443       case DW_CFA_GNU_args_size:
444         (void)read_uleb128(p);
445         break;
446       default:
447         return;
448       }
449     }
450   };
451 
452   run(fde.cie.cfi_begin, fde.cie.cfi_end, fde.func_start);
453   initial = row;
454   have_initial = true;
455   run(fde.cfi_begin, fde.cfi_end, fde.func_start);
456 }

458 inline usize
459 compute_cfa(const cfi_row &row, const reg_context &reg) noexcept
460 {
461   if ( row.cfa_is_expr ) return eval_dwarf_expr(row.cfa_expr, row.cfa_expr_end, reg, 0, false);
462   return reg.reg[row.cfa_reg < DWARF_MAX_REG ? row.cfa_reg : 0] + static_cast<usize>(row.cfa_off);
463 }
464 
465 inline usize
466 apply_rule(const reg_rule &r, const reg_context &reg, usize cfa, usize cur) noexcept
467 {
468   switch ( r.kind ) {
469   case rule_offset:
470     return read_unaligned<usize>(reinterpret_cast<const byte *>(cfa + static_cast<usize>(r.off)));
471   case rule_val_offset:
472     return cfa + static_cast<usize>(r.off);
473   case rule_register:
474     return reg.reg[r.which < DWARF_MAX_REG ? r.which : 0];
475   case rule_expression:
476     return read_unaligned<usize>(reinterpret_cast<const byte *>(eval_dwarf_expr(r.expr, r.expr_end, reg, cfa, true)));
477   case rule_val_expression:
478     return eval_dwarf_expr(r.expr, r.expr_end, reg, cfa, true);
479   case rule_same:
480   case rule_undefined:
481   default:
482     return cur;
483   }
484 }

486 inline bool
487 cfi_advance(reg_context &reg, const cfi_row &row, usize cfa, usize &pc) noexcept
488 {
489   reg_context nx = reg;
490   for ( int i = 0; i < DWARF_MAX_REG; ++i ) {
491     if ( i == DWARF_SP_COLUMN ) continue;
492     nx.reg[i] = apply_rule(row.rules[i], reg, cfa, reg.reg[i]);
493   }
494   const usize ra = apply_rule(row.rules[DWARF_RA_COLUMN], reg, cfa, 0);
495   nx.reg[DWARF_SP_COLUMN] = cfa;
496   nx.reg[DWARF_RA_COLUMN] = ra;
497 
498   reg = nx;
499   pc = ra;
500   return ra != 0;
501 }

503 inline bool
504 cfi_step(reg_context &reg, usize &pc, usize &out_cfa) noexcept
505 {
506   fde_info fde;
507   if ( !find_fde(pc - 1, fde) ) return false;
508   cfi_row row;
509   run_cfi(fde, pc - 1, row);
510   out_cfa = compute_cfa(row, reg);
511   return cfi_advance(reg, row, out_cfa, pc);
512 }
