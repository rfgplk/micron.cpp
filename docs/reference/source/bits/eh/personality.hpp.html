25 // fetch the index-th (1-based, backward) entry from the TType table
26 inline std::type_info *
27 get_ttype_entry(const byte *ttype_base, u8 ttype_enc, u64 index, const decode_ctx &ctx) noexcept
28 {
29   if ( ttype_enc == DW_EH_PE_omit ) return nullptr;
30   usize sz = encoded_size(ttype_enc);
31   if ( sz == 0 ) sz = sizeof(void *);      // leb-encoded ttype tables are not emitted by gcc
32   const byte *entry = ttype_base - index * sz;
33   const byte *q = entry;
34   const usize v = read_encoded(ttype_enc, q, ctx);
35   return reinterpret_cast<std::type_info *>(v);
36 }

40 extern "C" inline _Unwind_Reason_Code
41 __gxx_personality_v0(int version, _Unwind_Action actions, _Unwind_Exception_Class exception_class, _Unwind_Exception *ue,
42                      _Unwind_Context *context)
43 {
44   using namespace micron::eh;
45   using __cxxabiv1::__cxa_exception;
46 
47   if ( version != 1 || ue == nullptr || context == nullptr ) return _URC_FATAL_PHASE1_ERROR;
48 
49   __dbg_kv("[pers] actions=", static_cast<usize>(actions));
50 
51   const byte *lsda = reinterpret_cast<const byte *>(_Unwind_GetLanguageSpecificData(context));
52   if ( !lsda ) return _URC_CONTINUE_UNWIND;      // no actions in this frame
53 
54   const usize region_start = _Unwind_GetRegionStart(context);
55   int ip_before = 0;
56   usize ip = _Unwind_GetIPInfo(context, &ip_before);
57   if ( !ip_before && ip != 0 ) ip -= 1;      // call return address -> step back into the call
58   const usize pc_offset = ip - region_start;
59 
60   const decode_ctx dctx{ 0, _Unwind_GetTextRelBase(context), _Unwind_GetDataRelBase(context), region_start };
61 
62   // %%% parse the LSDA header %%%
63   const byte *p = lsda;
64   const u8 lpstart_enc = *p++;
65   usize lpstart = region_start;
66   if ( lpstart_enc != DW_EH_PE_omit ) lpstart = read_encoded(lpstart_enc, p, dctx);
67 
68   const u8 ttype_enc = *p++;
69   const byte *ttype_base = nullptr;
70   if ( ttype_enc != DW_EH_PE_omit ) {
71     const u64 ttype_off = read_uleb128(p);
72     ttype_base = p + ttype_off;
73   }
74 
75   const u8 cs_enc = *p++;
76   const u64 cs_table_len = read_uleb128(p);
77   const byte *cs_ptr = p;
78   const byte *cs_end = p + cs_table_len;
79   const byte *action_table = cs_end;
80 
81   // scan the call site table for the entry covering pc_offset
82   usize landing_pad = 0;
83   const byte *action_record = nullptr;
84   bool found_cs = false;
85 
86   while ( cs_ptr < cs_end ) {
87     const usize cs_start = read_encoded(cs_enc, cs_ptr, dctx);
88     const usize cs_len = read_encoded(cs_enc, cs_ptr, dctx);
89     const usize cs_lp = read_encoded(cs_enc, cs_ptr, dctx);
90     const u64 cs_action = read_uleb128(cs_ptr);
91 
92     if ( pc_offset < cs_start ) break;      // table is sorted; we have passed it
93     if ( pc_offset < cs_start + cs_len ) {
94       if ( cs_lp ) landing_pad = lpstart + cs_lp;
95       if ( cs_action ) action_record = action_table + (cs_action - 1);
96       found_cs = true;
97       break;
98     }
99   }
100 
101   __dbg_kv("[pers] pc_offset=", pc_offset);
102   __dbg_kv("  found_cs=", found_cs);
103   __dbg_kv("  landing_pad=", landing_pad);
104   if ( !found_cs ) return _URC_CONTINUE_UNWIND;             // PC has no EH region here
105   if ( landing_pad == 0 ) return _URC_CONTINUE_UNWIND;      // region with no landing pad
106 
107   // recover the thrown type / object from the exception header
108   __cxa_exception *xh = __cxxabiv1::__get_exception_header_from_ue(ue);
109   const bool native = (exception_class & 0xffffffffffffff00ULL) == (__cxxabiv1::__gxx_exception_class & 0xffffffffffffff00ULL);
110   std::type_info *throw_type = native ? xh->exceptionType : nullptr;
111   void *thrown_ptr = __cxxabiv1::__get_object_from_ue(ue);
112 
113   // no action chain -> cleanup-only landing pad
114   if ( action_record == nullptr ) {
115     if ( actions & _UA_SEARCH_PHASE ) return _URC_CONTINUE_UNWIND;      // cleanups don't stop phase 1
116     _Unwind_SetGR(context, __builtin_eh_return_data_regno(0), reinterpret_cast<_Unwind_Word>(ue));
117     _Unwind_SetGR(context, __builtin_eh_return_data_regno(1), 0);
118     _Unwind_SetIP(context, landing_pad);
119     return _URC_INSTALL_CONTEXT;
120   }
121 
122   // walk the action chain looking for a matching catch
123   int handler_switch = 0;
124   bool saw_cleanup = false;
125   bool saw_handler = false;
126   void *handler_ptr = thrown_ptr;
127 
128   const byte *a = action_record;
129   for ( ;; ) {
130     const byte *ap = a;
131     const i64 ar_filter = read_sleb128(ap);
132     const byte *next_field = ap;
133     const i64 ar_next = read_sleb128(ap);
134 
135     if ( ar_filter == 0 ) {
136       saw_cleanup = true;
137     } else if ( ar_filter > 0 ) {
138       std::type_info *catch_type = get_ttype_entry(ttype_base, ttype_enc, static_cast<u64>(ar_filter), dctx);
139       if ( catch_type == nullptr ) {
140         // catch(...) -- matches anything
141         saw_handler = true;
142         handler_switch = static_cast<int>(ar_filter);
143         handler_ptr = thrown_ptr;
144         break;
145       }
146       if ( throw_type != nullptr ) {
147         void *adj = thrown_ptr;
148         if ( catch_type->__do_catch(throw_type, &adj, 1) ) {
149           saw_handler = true;
150           handler_switch = static_cast<int>(ar_filter);
151           handler_ptr = adj;
152           break;
153         }
154       }
155     } else {
156       saw_handler = true;
157       handler_switch = static_cast<int>(ar_filter);
158       handler_ptr = thrown_ptr;
159       break;
160     }
161 
162     if ( ar_next == 0 ) break;
163     a = next_field + ar_next;      // ar_next is self-relative to its own field
164   }
165 
166   if ( actions & _UA_SEARCH_PHASE ) {
167     if ( saw_handler ) {
168       // stash for phase 2 / __cxa_begin_catch
169       xh->handlerSwitchValue = handler_switch;
170       xh->actionRecord = reinterpret_cast<const unsigned char *>(action_record);
171       xh->languageSpecificData = reinterpret_cast<const unsigned char *>(lsda);
172       xh->catchTemp = landing_pad;
173       xh->adjustedPtr = handler_ptr;
174       return _URC_HANDLER_FOUND;
175     }
176     return _URC_CONTINUE_UNWIND;      // only cleanups here
177   }
178 
179   // phase 2 / cleanup phase
180   const bool is_handler_frame = (actions & _UA_HANDLER_FRAME) != 0;
181   if ( is_handler_frame && saw_handler ) {
182     xh->handlerSwitchValue = handler_switch;
183     xh->adjustedPtr = handler_ptr;
184     _Unwind_SetGR(context, __builtin_eh_return_data_regno(0), reinterpret_cast<_Unwind_Word>(ue));
185     _Unwind_SetGR(context, __builtin_eh_return_data_regno(1), static_cast<_Unwind_Word>(static_cast<intptr_t>(handler_switch)));
186     _Unwind_SetIP(context, landing_pad);
187     return _URC_INSTALL_CONTEXT;
188   }
189   if ( saw_cleanup ) {
190     _Unwind_SetGR(context, __builtin_eh_return_data_regno(0), reinterpret_cast<_Unwind_Word>(ue));
191     _Unwind_SetGR(context, __builtin_eh_return_data_regno(1), 0);
192     _Unwind_SetIP(context, landing_pad);
193     return _URC_INSTALL_CONTEXT;
194   }
195   return _URC_CONTINUE_UNWIND;
196 }
