29 inline _Unwind_Reason_Code
30 unwind_phase1(_Unwind_Exception *exc, reg_context work) noexcept
31 {
32   usize pc = work.reg[DWARF_RA_COLUMN];
33   for ( ;; ) {
34     fde_info fde;
35     if ( !find_fde(pc - 1, fde) ) return _URC_END_OF_STACK;
36     cfi_row row;
37     run_cfi(fde, pc - 1, row);
38     const usize cfa = compute_cfa(row, work);
39 
40     if ( fde.cie.personality ) {
41       _Unwind_Context ctx{ &work, cfa, pc, &fde, fde.cie.signal_frame ? 1 : 0 };
42       const _Unwind_Reason_Code r = fde.cie.personality(1, _UA_SEARCH_PHASE, exc->exception_class, exc, &ctx);
43       if ( r == _URC_HANDLER_FOUND ) {
44         exc->private_2 = cfa;
45         return _URC_NO_REASON;
46       }
47       if ( r != _URC_CONTINUE_UNWIND ) return r;
48     }
49     if ( !cfi_advance(work, row, cfa, pc) ) return _URC_END_OF_STACK;
50   }
51 }
52 
53 [[noreturn]] inline void
54 unwind_phase2(_Unwind_Exception *exc, reg_context work) noexcept
55 {
56   usize pc = work.reg[DWARF_RA_COLUMN];
57   for ( ;; ) {
58     fde_info fde;
59     if ( !find_fde(pc - 1, fde) ) break;
60     cfi_row row;
61     run_cfi(fde, pc - 1, row);
62     const usize cfa = compute_cfa(row, work);
63 
64     if ( fde.cie.personality ) {
65       _Unwind_Action actions = _UA_CLEANUP_PHASE;
66       if ( cfa == exc->private_2 ) actions |= _UA_HANDLER_FRAME;
67       _Unwind_Context ctx{ &work, cfa, pc, &fde, fde.cie.signal_frame ? 1 : 0 };
68       const _Unwind_Reason_Code r = fde.cie.personality(1, actions, exc->exception_class, exc, &ctx);
69       if ( r == _URC_INSTALL_CONTEXT ) __micron_eh_install_context(&work);      // jumps to landing pad
70       if ( r != _URC_CONTINUE_UNWIND ) break;
71     }
72     if ( !cfi_advance(work, row, cfa, pc) ) break;
73   }
74   micron::abort(0xff);
75 }

81 inline _Unwind_Reason_Code
82 _Unwind_RaiseException(_Unwind_Exception *exc)
83 {
84   using namespace micron::eh;
85   reg_context start;
86   __micron_eh_capture(&start);      // capture this frame; phase loops walk copies
87 
88   const _Unwind_Reason_Code r = unwind_phase1(exc, start);
89   if ( r != _URC_NO_REASON ) return r;
90 
91   unwind_phase2(exc, start);      // installs (no return) on success
92   __builtin_unreachable();
93 }

95 inline void
96 _Unwind_Resume(_Unwind_Exception *exc)
97 {
98   using namespace micron::eh;
99   reg_context start;
100   __micron_eh_capture(&start);      // capture the cleanup frame; continue phase 2
101   unwind_phase2(exc, start);
102   __builtin_unreachable();
103 }

117 inline _Unwind_Reason_Code
118 _Unwind_ForcedUnwind(_Unwind_Exception *exc, _Unwind_Stop_Fn stop, void *stop_parameter)
119 {
120   using namespace micron::eh;
121   reg_context start;
122   __micron_eh_capture(&start);
123   reg_context work = start;
124   usize pc = work.reg[DWARF_RA_COLUMN];
125   for ( ;; ) {
126     fde_info fde;
127     const bool found = find_fde(pc - 1, fde);
128     cfi_row row;
129     usize cfa = 0;
130     if ( found ) {
131       run_cfi(fde, pc - 1, row);
132       cfa = compute_cfa(row, work);
133     }
134     _Unwind_Context ctx{ &work, cfa, pc, found ? &fde : nullptr, (found && fde.cie.signal_frame) ? 1 : 0 };
135     const _Unwind_Action act = _UA_FORCE_UNWIND | _UA_CLEANUP_PHASE | (found ? 0 : _UA_END_OF_STACK);
136     const _Unwind_Reason_Code sr = stop(1, act, exc->exception_class, exc, &ctx, stop_parameter);
137     if ( sr != _URC_NO_REASON ) return sr;
138     if ( found && fde.cie.personality ) {
139       const _Unwind_Reason_Code r = fde.cie.personality(1, act, exc->exception_class, exc, &ctx);
140       if ( r == _URC_INSTALL_CONTEXT ) __micron_eh_install_context(&work);
141     }
142     if ( !found || !cfi_advance(work, row, cfa, pc) ) return _URC_END_OF_STACK;
143   }
144 }
