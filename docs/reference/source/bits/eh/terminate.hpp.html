18 // std::terminate && terminate/unexpected entry points
19 
20 // special must be std namespace
21 namespace std
22 {
23 typedef void (*terminate_handler)();
24 typedef void (*unexpected_handler)();
25 };      // namespace std
26 
27 namespace micron::eh
28 {
29 
30 inline void
31 __eh_diag(const char *msg, usize len) noexcept
32 {
33   micron::syscall(SYS_write, 2, reinterpret_cast<const void *>(msg), len);
34 }
35 
36 inline std::terminate_handler __terminate_handler_slot = nullptr;
37 inline std::unexpected_handler __unexpected_handler_slot = nullptr;
38 
39 [[noreturn]] inline void
40 __default_terminate() noexcept
41 {
42   static const char m[] = "terminate called: uncaught exception\n";
43   __eh_diag(m, sizeof(m) - 1);
44   micron::abort(6);
45 }
46 
47 [[noreturn]] inline void
48 __terminate() noexcept
49 {
50   if ( __terminate_handler_slot ) {
51     __terminate_handler_slot();
52     // a terminate handler must not return; if it does, abort hard
53   }
54   __default_terminate();
55 }
56 
57 };      // namespace micron::eh
58 
59 namespace std
60 {
61 
62 [[noreturn]] inline void
63 terminate() noexcept
64 {
65   micron::eh::__terminate();
66 }
67 
68 inline terminate_handler
69 set_terminate(terminate_handler h) noexcept
70 {
71   terminate_handler old = micron::eh::__terminate_handler_slot;
72   micron::eh::__terminate_handler_slot = h;
73   return old;
74 }
75 
76 inline terminate_handler
77 get_terminate() noexcept
78 {
79   return micron::eh::__terminate_handler_slot;
80 }

86   micron::eh::__unexpected_handler_slot = h;
87   return old;
88 }

90 };      // namespace std
