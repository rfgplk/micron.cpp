129 inline void
130 register_eh_frame() noexcept
131 {
132   eh_frame_state &s = __eh_state();
133   if ( s.initialized ) return;
134   s.initialized = true;
135 
136   const elf_ehdr *eh = reinterpret_cast<const elf_ehdr *>(__ehdr_start);
137   const byte *phbase = __ehdr_start + eh->e_phoff;
138 
139   usize bias = 0;
140   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
141     const elf_phdr *ph = reinterpret_cast<const elf_phdr *>(phbase + static_cast<usize>(i) * eh->e_phentsize);
142     if ( ph->p_type == PT_PHDR ) {
143       bias = reinterpret_cast<usize>(phbase) - static_cast<usize>(ph->p_vaddr);
144       break;
145     }
146   }
147 
148   const elf_phdr *eh_frame_ph = nullptr;
149   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
150     const elf_phdr *ph = reinterpret_cast<const elf_phdr *>(phbase + static_cast<usize>(i) * eh->e_phentsize);
151     if ( ph->p_type == PT_GNU_EH_FRAME ) eh_frame_ph = ph;
152     if ( ph->p_type == PT_LOAD && (ph->p_flags & 0x1 /*X*/) && s.text_base == 0 ) s.text_base = static_cast<usize>(ph->p_vaddr) + bias;
153   }
154   if ( !eh_frame_ph ) return;
155 
156   const byte *hdr = reinterpret_cast<const byte *>(static_cast<usize>(eh_frame_ph->p_vaddr) + bias);
157   s.eh_frame_hdr = hdr;
158   s.hdr_base = reinterpret_cast<usize>(hdr);
159   s.data_base = bias;
160 
161   const byte *p = hdr;
162   const u8 version = *p++;
163   const u8 eh_frame_ptr_enc = *p++;
164   const u8 fde_count_enc = *p++;
165   const u8 table_enc = *p++;
166   if ( version != 1 ) return;
167 
168   decode_ctx ctx{ 0, s.text_base, s.hdr_base, 0 };
169   (void)read_encoded(eh_frame_ptr_enc, p, ctx);
170   const usize count = read_encoded(fde_count_enc, p, ctx);
171 
172   if ( fde_count_enc == DW_EH_PE_omit || count == 0 ) return;
173 
174   s.search_table = p;
175   s.fde_count = count;
176   s.table_enc = table_enc;
177   s.valid = true;
178 
179   __dbg_kv("[ehframe] hdr=", reinterpret_cast<usize>(hdr));
180   __dbg_kv("[ehframe] count=", s.fde_count);
181   __dbg_kv("[ehframe] table_enc=", s.table_enc);
182   __dbg_kv("[ehframe] text_base=", s.text_base);
183 }

185 inline bool
186 parse_cie(const byte *cie_record, const eh_frame_state &s, cie_info &out) noexcept
187 {
188   const byte *p = cie_record;
189   u32 length = read_unaligned<u32>(p);
190   p += 4;
191   if ( length == 0 ) return false;
192   if ( length == 0xffffffffu ) return false;      // 64-bit DWARF unsupported (not emitted by gcc here)
193   const byte *cie_end = p + length;
194 
195   p += 4;
196   const u8 ver = *p++;
197   const char *aug = reinterpret_cast<const char *>(p);
198   while ( *p ) ++p;
199   ++p;
200 
201   if ( ver >= 4 ) {
202     ++p;
203     ++p;
204   }
205 
206   out.code_align = read_uleb128(p);
207   out.data_align = read_sleb128(p);
208   out.ra_column = (ver == 1) ? static_cast<u32>(*p++) : static_cast<u32>(read_uleb128(p));
209 
210   out.fde_enc = DW_EH_PE_absptr;
211   out.lsda_enc = DW_EH_PE_omit;
212   out.has_aug_data = (aug[0] == 'z');
213   out.has_lsda = false;
214   out.signal_frame = false;
215   out.personality = nullptr;
216 
217   if ( aug[0] == 'z' ) {
218     const u64 aug_len = read_uleb128(p);
219     const byte *aug_end = p + aug_len;
220     decode_ctx ctx{ 0, s.text_base, s.data_base, 0 };
221     for ( const char *a = aug + 1; *a; ++a ) {
222       switch ( *a ) {
223       case 'L':
224         out.lsda_enc = *p++;
225         out.has_lsda = true;
226         break;
227       case 'P': {
228         const u8 penc = *p++;
229         const usize pv = read_encoded(penc, p, ctx);
230         out.personality = reinterpret_cast<_Unwind_Personality_Fn>(pv);
231         break;
232       }
233       case 'R':
234         out.fde_enc = *p++;
235         break;
236       case 'S':
237         out.signal_frame = true;
238         break;
239       default:
240         a = ""; /* unknown: stop, rely on aug_end */
241         p = aug_end;
242         break;
243       }
244     }
245     p = aug_end;
246   }
247 
248   out.cfi_begin = p;
249   out.cfi_end = cie_end;
250   return true;
251 }

253 inline bool
254 parse_fde(const byte *fde_record, const eh_frame_state &s, fde_info &out) noexcept
255 {
256   const byte *p = fde_record;
257   const u32 length = read_unaligned<u32>(p);
258   p += 4;
259   if ( length == 0 || length == 0xffffffffu ) return false;
260   const byte *fde_end = p + length;
261 
262   const byte *cie_ptr_field = p;
263   const u32 cie_off = read_unaligned<u32>(p);
264   p += 4;
265   if ( cie_off == 0 ) return false;      // this is a CIE, not an FDE
266   const byte *cie_record = cie_ptr_field - cie_off;
267 
268   if ( !parse_cie(cie_record, s, out.cie) ) return false;
269 
270   decode_ctx ctx{ 0, s.text_base, s.data_base, 0 };
271   const byte *pcfield = p;
272   const usize initial_loc = read_encoded(out.cie.fde_enc, p, ctx);
273   // address_range uses the same value format but is always absolute (no app base)
274   const u8 range_enc = static_cast<u8>(out.cie.fde_enc & 0x0f);
275   const usize range = read_encoded(range_enc, p, ctx);
276   (void)pcfield;
277 
278   out.func_start = initial_loc;
279   out.func_end = initial_loc + range;
280   out.lsda = nullptr;
281 
282   if ( out.cie.signal_frame ) { /* handled by the CFI step (ip_before_insn) */
283   }
284 
285   if ( out.cie.has_aug_data ) {
286     const byte *q = p;
287     const u64 aug_len = read_uleb128(q);
288     const byte *aug_end = q + aug_len;
289     if ( out.cie.has_lsda && out.cie.lsda_enc != DW_EH_PE_omit ) {
290       decode_ctx lctx{ 0, s.text_base, s.data_base, out.func_start };
291       const usize lsda = read_encoded(out.cie.lsda_enc, q, lctx);
292       out.lsda = lsda ? reinterpret_cast<const byte *>(lsda) : nullptr;
293     }
294     p = aug_end;
295   }
296 
297   out.cfi_begin = p;
298   out.cfi_end = fde_end;
299   return true;
300 }

302 inline bool
303 find_fde(usize pc, fde_info &out) noexcept
304 {
305   eh_frame_state &s = __eh_state();
306   if ( !s.initialized ) register_eh_frame();
307   if ( !s.valid ) return false;
308 
309   const usize entry_field_sz = encoded_size(s.table_enc);
310   if ( entry_field_sz == 0 ) return false;      // only fixed-width tables supported
311   const usize entry_sz = entry_field_sz * 2;
312 
313   usize lo = 0, hi = s.fde_count;
314   const byte *table = s.search_table;
315   decode_ctx ctx{ 0, s.text_base, s.hdr_base, 0 };
316 
317   auto entry_loc = [&](usize idx) -> usize {
318     const byte *e = table + idx * entry_sz;
319     return read_encoded(s.table_enc, e, ctx);
320   };
321 
322   if ( pc < entry_loc(0) ) return false;
323   while ( hi - lo > 1 ) {
324     const usize mid = lo + (hi - lo) / 2;
325     if ( entry_loc(mid) <= pc )
326       lo = mid;
327     else
328       hi = mid;
329   }
330 
331   const byte *e = table + lo * entry_sz;
332   (void)read_encoded(s.table_enc, e, ctx);      // skip initial_location
333   const usize fde_addr = read_encoded(s.table_enc, e, ctx);
334   if ( !fde_addr ) return false;
335 
336   if ( !parse_fde(reinterpret_cast<const byte *>(fde_addr), s, out) ) return false;
337   __dbg_kv("[find_fde] pc=", pc);
338   __dbg_kv("  func_start=", out.func_start);
339   __dbg_kv("  func_end=", out.func_end);
340   __dbg_kv("  personality=", reinterpret_cast<usize>(out.cie.personality));
341   __dbg_kv("  lsda=", reinterpret_cast<usize>(out.lsda));
342   if ( pc < out.func_start || pc >= out.func_end ) return false;
343   return true;
344 }
