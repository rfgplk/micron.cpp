22 // WARNING: arm32 does not use DWARF .eh_frame; instead it uses the ARM Exception Handling ABI
23 // (IHI0038) a sorted .ARM.exidx index that maps a PC to a .ARM.extab entry,

95 // decode a 31-bit place-relative (prel31) offset at *p into an absolute address
96 inline usize
97 prel31(const u32 *p) noexcept
98 {
99   i32 off = static_cast<i32>(*p << 1) >> 1;      // sign-extend bit30 -> 31-bit signed
100   return reinterpret_cast<usize>(p) + static_cast<usize>(off);
101 }
102 
103 inline void
104 register_exidx() noexcept
105 {
106   exidx_state &s = __exidx_state();
107   if ( s.initialized ) return;
108   s.initialized = true;
109 
110   const elf_ehdr *eh = reinterpret_cast<const elf_ehdr *>(__ehdr_start);
111   const byte *phbase = __ehdr_start + eh->e_phoff;
112   usize bias = 0;
113   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
114     const elf_phdr *ph = reinterpret_cast<const elf_phdr *>(phbase + static_cast<usize>(i) * eh->e_phentsize);
115     if ( ph->p_type == 6 /*PT_PHDR*/ ) {
116       bias = reinterpret_cast<usize>(phbase) - static_cast<usize>(ph->p_vaddr);
117       break;
118     }
119   }
120   for ( u16 i = 0; i < eh->e_phnum; ++i ) {
121     const elf_phdr *ph = reinterpret_cast<const elf_phdr *>(phbase + static_cast<usize>(i) * eh->e_phentsize);
122     if ( ph->p_type == PT_ARM_EXIDX ) {
123       s.table = reinterpret_cast<const u32 *>(static_cast<usize>(ph->p_vaddr) + bias);
124       s.count = static_cast<usize>(ph->p_memsz) / 8;      // 8 bytes per entry
125       s.valid = true;
126       return;
127     }
128   }
129 }

131 inline const u32 *
132 find_exidx(usize pc) noexcept
133 {
134   exidx_state &s = __exidx_state();
135   if ( !s.initialized ) register_exidx();
136   if ( !s.valid || s.count == 0 ) return nullptr;
137 
138   usize lo = 0, hi = s.count;
139   if ( pc < prel31(&s.table[0]) ) return nullptr;
140   while ( hi - lo > 1 ) {
141     const usize mid = lo + (hi - lo) / 2;
142     if ( prel31(&s.table[mid * 2]) <= pc )
143       lo = mid;
144     else
145       hi = mid;
146   }
147   return &s.table[lo * 2];
148 }

150 struct op_stream {
151   const u32 *words;
152   usize total;      // total opcode bytes
153   usize idx;
154   int first_n;      // opcode bytes carried in words[0] (3 or 2)
155 };
156 
157 inline byte
158 op_at(const op_stream &s, usize i) noexcept
159 {
160   if ( i < static_cast<usize>(s.first_n) ) return static_cast<byte>(s.words[0] >> (8 * (s.first_n - 1 - static_cast<int>(i))));
161   const usize j = i - static_cast<usize>(s.first_n);
162   const u32 w = s.words[1 + j / 4];
163   return static_cast<byte>(w >> (24 - 8 * (j % 4)));
164 }

166 inline bool
167 exec_opcodes(op_stream &st, _Unwind_Context *ctx) noexcept
168 {
169   usize vsp = ctx->core[13];
170   while ( st.idx < st.total ) {
171     const byte op = op_at(st, st.idx++);
172     if ( (op & 0xc0) == 0x00 ) {      // 00xxxxxx: vsp += (xxxxxx<<2)+4
173       vsp += (static_cast<usize>(op & 0x3f) << 2) + 4;
174     } else if ( (op & 0xc0) == 0x40 ) {      // 01xxxxxx: vsp -= (xxxxxx<<2)+4
175       vsp -= (static_cast<usize>(op & 0x3f) << 2) + 4;
176     } else if ( (op & 0xf0) == 0x80 ) {      // 1000iiii iiiiiiii: pop {r4-r15} by mask
177       const byte op2 = op_at(st, st.idx++);
178       const u32 mask = (static_cast<u32>(op & 0x0f) << 8) | op2;
179       if ( mask == 0 ) return false;      // refuse to unwind
180       for ( int r = 0; r < 12; ++r )
181         if ( mask & (1u << r) ) {
182           ctx->core[4 + r] = read_unaligned<usize>(reinterpret_cast<const byte *>(vsp));
183           vsp += 4;
184         }
185     } else if ( (op & 0xf0) == 0x90 && op != 0x9d && op != 0x9f ) {      // 1001nnnn: vsp = r[nnnn]
186       vsp = ctx->core[op & 0x0f];
187     } else if ( (op & 0xf8) == 0xa0 ) {      // 10100nnn: pop r4-r[4+n]
188       const int n = op & 0x07;
189       for ( int r = 0; r <= n; ++r ) {
190         ctx->core[4 + r] = read_unaligned<usize>(reinterpret_cast<const byte *>(vsp));
191         vsp += 4;
192       }
193     } else if ( (op & 0xf8) == 0xa8 ) {      // 10101nnn: pop r4-r[4+n] and r14
194       const int n = op & 0x07;
195       for ( int r = 0; r <= n; ++r ) {
196         ctx->core[4 + r] = read_unaligned<usize>(reinterpret_cast<const byte *>(vsp));
197         vsp += 4;
198       }
199       ctx->core[14] = read_unaligned<usize>(reinterpret_cast<const byte *>(vsp));
200       vsp += 4;
201     } else if ( op == 0xb0 ) {      // finish
202       ctx->core[13] = vsp;
203       if ( ctx->core[15] == 0 ) ctx->core[15] = ctx->core[14];      // pc not set -> use lr
204       return true;
205     } else if ( op == 0xb1 ) {      // 10110001 0000iiii: pop {r0-r3} by mask
206       const byte op2 = op_at(st, st.idx++);
207       if ( op2 == 0 || (op2 & 0xf0) ) { /* spare / reserved */
208       } else
209         for ( int r = 0; r < 4; ++r )
210           if ( op2 & (1u << r) ) {
211             ctx->core[r] = read_unaligned<usize>(reinterpret_cast<const byte *>(vsp));
212             vsp += 4;
213           }
214     } else if ( op == 0xb2 ) {      // vsp += 0x204 + (uleb<<2)
215       u64 shift = 0, v = 0;
216       byte b;
217       do {
218         b = op_at(st, st.idx++);
219         v |= static_cast<u64>(b & 0x7f) << shift;
220         shift += 7;
221       } while ( b & 0x80 );
222       vsp += 0x204 + (static_cast<usize>(v) << 2);
223     } else if ( op == 0xb3 ) {      // FSTMFDX: operand sssscccc -> (cccc+1) doubles + 1 alignment word
224       const byte d = op_at(st, st.idx++);
225       vsp += (static_cast<usize>(d & 0x0f) + 1) * 8 + 4;
226     } else if ( op == 0xc8 || op == 0xc9 || op == 0xc6 ) {      // VFP/iWMMXt pop: operand sssscccc -> (cccc+1) regs
227       const byte d = op_at(st, st.idx++);
228       vsp += (static_cast<usize>(d & 0x0f) + 1) * 8;
229     } else if ( (op & 0xf8) == 0xb8 ) {      // pop d8-d[8+nnn] (FSTMFDX): (nnn+1) doubles + 1 word
230       vsp += (static_cast<usize>(op & 0x07) + 1) * 8 + 4;
231     } else if ( (op & 0xf8) == 0xc0 ) {      // pop wR[10]-wR[10+nnn] / d-regs: (nnn+1) regs
232       vsp += (static_cast<usize>(op & 0x07) + 1) * 8;
233     } else if ( op == 0xc7 ) {      // iWMMXt wCGR pop: operand 0000iiii (no vsp change tracked here)
234       ++st.idx;
235     } else {
236       // NOTE: VFP/iWMMXt register vals are not restored
237       ctx->core[13] = vsp;
238       return true;
239     }
240   }
241   ctx->core[13] = vsp;
242   if ( ctx->core[15] == 0 ) ctx->core[15] = ctx->core[14];
243   return true;
244 }

302 // thread the exidx/extab lookup into an op_stream + (for C++ frames) the LSDA
303 struct frame_unwind {
304   int pr_kind;
305   op_stream ops;
306   const byte *lsda;      // nullptr if none
307   usize func_start;
308 };
309 
310 inline bool
311 decode_frame(usize pc, frame_unwind &out) noexcept
312 {
313   if ( pc == 0 ) return false;
314   const u32 *entry = find_exidx(pc - 1);
315   if ( !entry ) return false;
316   out.func_start = prel31(&entry[0]);
317   const u32 data = entry[1];
318   if ( data == EXIDX_CANTUNWIND ) return false;
319   out.lsda = nullptr;      // only the generic (C++) form carries an LSDA
320 
321   if ( data & 0x80000000u ) {
322     out.pr_kind = static_cast<int>((data >> 24) & 0x0f);
323     out.ops = op_stream{ &entry[1], 3, 0, 3 };
324     return true;
325   }
326   // long entry: prel31 -> .ARM.extab
327   const u32 *extab = reinterpret_cast<const u32 *>(prel31(&entry[1]));
328   const u32 w0 = extab[0];
329   if ( w0 & 0x80000000u ) {
330     // compact model in extab
331     out.pr_kind = static_cast<int>((w0 >> 24) & 0x0f);
332     if ( out.pr_kind == 0 ) {
333       out.ops = op_stream{ extab, 3, 0, 3 };
334     } else {
335       const u32 n = (w0 >> 16) & 0xff;      // pr1/pr2: byte 2 = extra-word count, bytes 1-0 = 2 opcodes
336       out.ops = op_stream{ extab, static_cast<usize>(2 + 4 * n), 0, 2 };
337     }
338     return true;
339   }
340   out.pr_kind = 3;
341   const u32 *ehtp = &extab[1];
342   const u32 count = ehtp[0] >> 24;
343   out.ops = op_stream{ ehtp, static_cast<usize>(3 + 4 * count), 0, 3 };
344   out.lsda = reinterpret_cast<const byte *>(&ehtp[1 + count]);
345   return true;
346 }

389 inline bool
390 ehabi_scan_lsda(const byte *lsda, usize ip, usize func_start, _Unwind_Exception *ue, bool search_phase, bool &is_handler,
391                 usize &landing_pad, int &selector, void *&adjusted) noexcept
392 {
393   using __cxxabiv1::__cxa_exception;
394   const decode_ctx dctx{ 0, 0, 0, func_start };
395 
396   const byte *p = lsda;
397   const u8 lpstart_enc = *p++;
398   usize lpstart = func_start;
399   if ( lpstart_enc != DW_EH_PE_omit ) lpstart = read_encoded(lpstart_enc, p, dctx);
400 
401   const u8 ttype_enc = *p++;
402   const byte *ttype_base = nullptr;
403   if ( ttype_enc != DW_EH_PE_omit ) {
404     const u64 off = read_uleb128(p);
405     ttype_base = p + off;
406   }
407   const u8 cs_enc = *p++;
408   const u64 cs_len = read_uleb128(p);
409   const byte *cs_ptr = p;
410   const byte *cs_end = p + cs_len;
411   const byte *action_table = cs_end;
412 
413   const usize pc_off = (ip - 1) - func_start;      // ip is a return address; step into the call site
414   usize lp = 0;
415   const byte *action_record = nullptr;
416   bool found = false;
417   while ( cs_ptr < cs_end ) {
418     const usize cs_start = read_encoded(cs_enc, cs_ptr, dctx);
419     const usize cs_clen = read_encoded(cs_enc, cs_ptr, dctx);
420     const usize cs_lp = read_encoded(cs_enc, cs_ptr, dctx);
421     const u64 cs_action = read_uleb128(cs_ptr);
422     if ( pc_off < cs_start ) break;
423     if ( pc_off < cs_start + cs_clen ) {
424       if ( cs_lp ) lp = lpstart + cs_lp;
425       if ( cs_action ) action_record = action_table + (cs_action - 1);
426       found = true;
427       break;
428     }
429   }
430   if ( !found || lp == 0 ) return false;
431 
432   __cxa_exception *xh = __cxxabiv1::__get_exception_header_from_ue(ue);
433   std::type_info *throw_type = xh->exceptionType;
434   void *thrown_ptr = __cxxabiv1::__get_object_from_ue(ue);
435 
436   if ( action_record == nullptr ) {      // cleanup-only
437     if ( search_phase ) return false;
438     is_handler = false;
439     landing_pad = lp;
440     selector = 0;
441     adjusted = thrown_ptr;
442     return true;
443   }
444 
445   const byte *a = action_record;
446   for ( ;; ) {
447     const byte *ap = a;
448     const i64 ar_filter = read_sleb128(ap);
449     const byte *next_field = ap;
450     const i64 ar_next = read_sleb128(ap);
451     if ( ar_filter == 0 ) {
452       // cleanup
453     } else if ( ar_filter > 0 ) {
454       usize sz = encoded_size(ttype_enc);
455       if ( sz == 0 ) sz = sizeof(void *);
456       const byte *ent = ttype_base - static_cast<u64>(ar_filter) * sz;
457       const byte *q = ent;
458       // ARM TARGET2 references are indirect place-relative offsets; Clang labels that table as absptr
459       const u8 pointer_enc = ttype_enc == DW_EH_PE_absptr ? (DW_EH_PE_pcrel | DW_EH_PE_sdata4 | DW_EH_PE_indirect) : ttype_enc;
460       std::type_info *catch_type = reinterpret_cast<std::type_info *>(read_encoded(pointer_enc, q, dctx));
461       if ( catch_type == nullptr ) {
462         is_handler = true;
463         landing_pad = lp;
464         selector = static_cast<int>(ar_filter);
465         adjusted = thrown_ptr;
466         return true;
467       }
468       void *adj = thrown_ptr;
469       if ( catch_type->__do_catch(throw_type, &adj, 1) ) {
470         is_handler = true;
471         landing_pad = lp;
472         selector = static_cast<int>(ar_filter);
473         adjusted = adj;
474         return true;
475       }
476     }
477     if ( ar_next == 0 ) break;
478     a = next_field + ar_next;
479   }
480 
481   if ( search_phase ) return false;      // only cleanups matched here
482   is_handler = false;
483   landing_pad = lp;
484   selector = 0;
485   adjusted = thrown_ptr;
486   return true;
487 }

493 inline _Unwind_Reason_Code
494 __gxx_personality_v0(_Unwind_State state, _Unwind_Exception *ucbp, _Unwind_Context *context)
495 {
496   using namespace micron::eh;
497   using __cxxabiv1::__cxa_exception;
498 
499   frame_unwind fu;
500   if ( !decode_frame(_Unwind_GetIP(context), fu) ) return _URC_FAILURE;
501 
502   const int action = state & _US_ACTION_MASK;
503   __cxa_exception *xh = __cxxabiv1::__get_exception_header_from_ue(ucbp);
504 
505   if ( action == _US_VIRTUAL_UNWIND_FRAME ) {
506     // phase 1: search for a handler in this frame
507     if ( fu.lsda ) {
508       bool is_handler = false;
509       usize lp = 0;
510       int sel = 0;
511       void *adj = nullptr;
512       if ( ehabi_scan_lsda(fu.lsda, _Unwind_GetIP(context), fu.func_start, ucbp, true, is_handler, lp, sel, adj) && is_handler ) {
513         // stash for phase 2 in the barrier cache
514         ucbp->barrier_cache.sp = context->core[13];
515         ucbp->barrier_cache.bitpattern[0] = static_cast<_Unwind_Word>(reinterpret_cast<usize>(adj));
516         ucbp->barrier_cache.bitpattern[1] = static_cast<_Unwind_Word>(static_cast<usize>(sel));
517         ucbp->barrier_cache.bitpattern[2] = static_cast<_Unwind_Word>(lp);
518         xh->adjustedPtr = adj;
519         xh->handlerSwitchValue = sel;
520         return _URC_HANDLER_FOUND;
521       }
522     }
523     // not a handler frame: unwind one frame and continue
524     context->core[15] = 0;
525     exec_opcodes(fu.ops, context);
526     return _URC_CONTINUE_UNWIND;
527   }
528 
529   const bool is_handler_frame = (state & _US_ACTION_MASK) == _US_UNWIND_FRAME_STARTING && context->core[13] == ucbp->barrier_cache.sp;
530 
531   if ( fu.lsda ) {
532     bool is_handler = false;
533     usize lp = 0;
534     int sel = 0;
535     void *adj = nullptr;
536     if ( ehabi_scan_lsda(fu.lsda, _Unwind_GetIP(context), fu.func_start, ucbp, false, is_handler, lp, sel, adj) ) {
537       if ( is_handler && !is_handler_frame ) {
538         // a handler match but not THE recorded handler frame -> treat as continue
539       } else {
540         if ( is_handler ) {
541           xh->adjustedPtr = adj;
542           xh->handlerSwitchValue = sel;
543         }
544         _Unwind_SetGR(context, 0, reinterpret_cast<_Unwind_Word>(ucbp));
545         _Unwind_SetGR(context, 1, static_cast<_Unwind_Word>(static_cast<intptr_t>(is_handler ? sel : 0)));
546         _Unwind_SetIP(context, lp);
547         return _URC_INSTALL_CONTEXT;
548       }
549     }
550   }
551   // nothing to do here: unwind and continue
552   context->core[15] = 0;
553   exec_opcodes(fu.ops, context);
554   return _URC_CONTINUE_UNWIND;
555 }

567 [[noreturn]] inline void
568 ehabi_phase2(_Unwind_Exception *ue, _Unwind_Context ctx) noexcept
569 {
570   for ( ;; ) {
571     frame_unwind fu;
572     if ( !decode_frame(_Unwind_GetIP(&ctx), fu) ) break;
573     _Unwind_Reason_Code r;
574     if ( fu.pr_kind == 3 )
575       r = __gxx_personality_v0(_US_UNWIND_FRAME_STARTING, ue, &ctx);
576     else {
577       // pure-unwind frame: just step
578       ctx.core[15] = 0;
579       exec_opcodes(fu.ops, &ctx);
580       continue;
581     }
582     if ( r == _URC_INSTALL_CONTEXT ) __micron_eh_install_context(&ctx);
583     if ( r != _URC_CONTINUE_UNWIND ) break;
584   }
585   micron::abort(0xff);
586 }

595 inline _Unwind_Reason_Code
596 _Unwind_RaiseException(_Unwind_Exception *ucbp)
597 {
598   using namespace micron::eh;
599   __cxa_get_globals()->propagatingExceptions = __cxxabiv1::__get_exception_header_from_ue(ucbp);
600   _Unwind_Context ctx;
601   __micron_eh_capture(&ctx);
602 
603   _Unwind_Context work = ctx;
604   for ( ;; ) {
605     frame_unwind fu;
606     if ( !decode_frame(_Unwind_GetIP(&work), fu) ) return _URC_END_OF_STACK;
607     if ( fu.pr_kind == 3 ) {
608       const _Unwind_Reason_Code r = __gxx_personality_v0(_US_VIRTUAL_UNWIND_FRAME, ucbp, &work);
609       if ( r == _URC_HANDLER_FOUND ) break;
610       if ( r != _URC_CONTINUE_UNWIND ) return r;
611     } else {
612       work.core[15] = 0;
613       exec_opcodes(fu.ops, &work);
614     }
615   }
616   ehabi_phase2(ucbp, ctx);
617   __builtin_unreachable();
618 }
