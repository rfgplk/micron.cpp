180 // NOTE: REQUIRES the walked code be built with -fno-omit-frame-pointer
181 [[gnu::always_inline]] inline bool
182 caller(record &r) noexcept
183 {
184   if ( r.fp == nullptr ) return false;
185   void **fp = static_cast<void **>(r.fp);
186   void *saved_fp = fp[0];      // caller's frame pointer
187   void *ret = fp[1];           // return address
188   // frames grow toward higher addresses up the chain; reject garbage / cycles
189   if ( saved_fp == nullptr || saved_fp <= r.fp ) return false;
190   r.sp = static_cast<void *>(fp + 2);
191   r.fp = saved_fp;
192   r.pc = ret;
193   return true;
194 }

213 inline __attribute__((noinline)) void
214 call_on_stack([[maybe_unused]] void *stack_top, [[maybe_unused]] void (*fn)(void *), [[maybe_unused]] void *arg) noexcept
215 {
216 #if defined(__micron_arch_amd64)
217   asm volatile("movq %%rsp, %%r15\n\t"      // save old sp (callee-saved, preserved by fn)
218                "andq $-16, %0\n\t"
219                "movq %0, %%rsp\n\t"         // pivot
220                "movq %2, %%rdi\n\t"         // arg -> a1
221                "callq *%1\n\t"              // fn(arg)
222                "movq %%r15, %%rsp\n\t"      // restore
223                : "+r"(stack_top)
224                : "r"(fn), "r"(arg)
225                : "r15", "rdi", "rax", "rcx", "rdx", "rsi", "r8", "r9", "r10", "r11", "memory", "cc");
226 #elif defined(__micron_arch_x86)
227   asm volatile("movl %%esp, %%edi\n\t"      // save old sp (edi callee-saved on i386)
228                "andl $-16, %0\n\t"
229                "movl %0, %%esp\n\t"
230                "subl $12, %%esp\n\t"      // keep 16-byte alignment after the arg push
231                "pushl %2\n\t"             // arg
232                "calll *%1\n\t"
233                "movl %%edi, %%esp\n\t"      // restore
234                : "+r"(stack_top)
235                : "r"(fn), "r"(arg)
236                : "edi", "eax", "ecx", "edx", "memory", "cc");
237 #elif defined(__micron_arch_arm64)
238   asm volatile("mov x19, sp\n\t"      // save old sp (callee-saved)
239                "and %0, %0, #-16\n\t"
240                "mov sp, %0\n\t"
241                "mov x0, %2\n\t"       // arg
242                "blr %1\n\t"           // fn(arg)
243                "mov sp, x19\n\t"      // restore
244                : "+r"(stack_top)
245                : "r"(fn), "r"(arg)
246                : "x19", "x0", "x1", "x2", "x3", "x4", "x5", "x6", "x7", "x8", "x9", "x10", "x11", "x12", "x13", "x14", "x15", "x16", "x17",
247                  "x18", "x30", "memory", "cc");
248 #elif defined(__micron_arch_arm32)
249   asm volatile("mov r4, sp\n\t"      // save old sp (callee-saved)
250                "and %0, %0, #-8\n\t"
251                "mov sp, %0\n\t"
252                "mov r0, %2\n\t"      // arg
253                "blx %1\n\t"          // fn(arg)
254                "mov sp, r4\n\t"      // restore
255                : "+r"(stack_top)
256                : "r"(fn), "r"(arg)
257                : "r4", "r0", "r1", "r2", "r3", "r12", "lr", "memory", "cc");
258 #endif
259 }

261 // fiber CPU contexts
262 //
263 // NOTE: by default our swap will _NOT_ save/restore the FP control state (measured 6.9 vs 8.4 cyc/switch hot)
264 // every fiber instead starts with the thread's current FP env
265 // code that changes rounding/exception modes must NEVER hold that change across a suspend point
266 // unless compiled with -DMICRON_FIBER_FPENV
267 //
268 // under FPENV only the control modes (rounding, exception masks) are per-fiber; the sticky MXCSR status bits are masked out at save
269 //
270 // ldmxcsr/fldcw serialize the pipeline (~50 cyc each) whenever the loaded word differs from the live one
271 // letting sticky status bits ride along makes every switch a real control word change (measured 113 cyc/switch)
272 
273 #if defined(__micron_arch_amd64)
274 
275 struct __fiber_ctx {
276   u64 rbx;        // 0x00
277   u64 rbp;        // 0x08
278   u64 r12;        // 0x10  carries fn-arg into the trampoline (make_context)
279   u64 r13;        // 0x18  carries fn-ptr into the trampoline (make_context)
280   u64 r14;        // 0x20
281   u64 r15;        // 0x28
282   u64 rsp;        // 0x30
283   u64 rip;        // 0x38  resume PC (trampoline for a fresh ctx; return addr for a suspended one)
284   u32 mxcsr;      // 0x40  SSE control/status (callee-saved bits)
285   u16 fcw;        // 0x44  x87 control word
286 };
287 
288 static_assert(__builtin_offsetof(__fiber_ctx, rsp) == 0x30, "amd64 __fiber_ctx rsp offset");
289 static_assert(__builtin_offsetof(__fiber_ctx, rip) == 0x38, "amd64 __fiber_ctx rip offset");
290 static_assert(__builtin_offsetof(__fiber_ctx, mxcsr) == 0x40, "amd64 __fiber_ctx mxcsr offset");
291 static_assert(__builtin_offsetof(__fiber_ctx, fcw) == 0x44, "amd64 __fiber_ctx fcw offset");

293 [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp void
294 __micron_swap_context(__fiber_ctx * /*rdi from*/, __fiber_ctx * /*rsi to*/) noexcept
295 {
296   // WARNING: xmm0 through xmm15 aren't saved explicitly
297   // red zones are _technically_ ignored (investigate)
298   // TODO: maybe do it via another macro? would suck up too many cycles unconditionally
299   asm volatile("movq %rbx, 0x00(%rdi)\n\t"
300                "movq %rbp, 0x08(%rdi)\n\t"
301                "movq %r12, 0x10(%rdi)\n\t"
302                "movq %r13, 0x18(%rdi)\n\t"
303                "movq %r14, 0x20(%rdi)\n\t"
304                "movq %r15, 0x28(%rdi)\n\t"
305                "movq (%rsp), %rax\n\t"      // return address -> from->rip
306                "movq %rax, 0x38(%rdi)\n\t"
307                "leaq 8(%rsp), %rax\n\t"
308                "movq %rax, 0x30(%rdi)\n\t"
309 #if defined(MICRON_FIBER_FPENV)
310                "stmxcsr 0x40(%rdi)\n\t"
311                "andl $-64, 0x40(%rdi)\n\t"      // drop sticky status bits (0-5); control modes only
312                "fnstcw 0x44(%rdi)\n\t"
313                "ldmxcsr 0x40(%rsi)\n\t"      // restore *to
314                "fldcw 0x44(%rsi)\n\t"
315 #endif
316                "movq 0x00(%rsi), %rbx\n\t"
317                "movq 0x08(%rsi), %rbp\n\t"
318                "movq 0x10(%rsi), %r12\n\t"
319                "movq 0x18(%rsi), %r13\n\t"
320                "movq 0x20(%rsi), %r14\n\t"
321                "movq 0x28(%rsi), %r15\n\t"
322                "movq 0x30(%rsi), %rsp\n\t"
323                "jmp *0x38(%rsi)\n\t");
324 }

326 [[maybe_unused]] static __attribute__((naked, noinline, noreturn)) __micron_no_ssp void
327 __micron_fiber_trampoline() noexcept
328 {
329   asm volatile("andq $-16, %rsp\n\t"      // defensive 16-align
330                "xorl %ebp, %ebp\n\t"      // terminate the unwind/backtrace chain
331                "movq %r12, %rdi\n\t"      // arg -> a0
332                "callq *%r13\n\t"          // fn(arg)
333                "ud2\n\t");                // fn must not return
334 }

336 [[gnu::always_inline]] inline void
337 make_context(__fiber_ctx *ctx, void *stack_top, void (*fn)(void *), void *arg) noexcept
338 {
339   // zero it first
340   *ctx = __fiber_ctx{};
341   ctx->rsp = reinterpret_cast<u64>(stack_top) & ~static_cast<u64>(15);
342   ctx->rip = reinterpret_cast<u64>(&__micron_fiber_trampoline);
343   ctx->r12 = reinterpret_cast<u64>(arg);
344   ctx->r13 = reinterpret_cast<u64>(fn);
345 #if defined(MICRON_FIBER_FPENV)
346   // seed from the live control state, status bits masked
347   u32 __mx;
348   u16 __cw;
349   asm volatile("stmxcsr %0" : "=m"(__mx));
350   asm volatile("fnstcw %0" : "=m"(__cw));
351   ctx->mxcsr = __mx & ~0x3Fu;
352   ctx->fcw = __cw;
353 #endif
354 }

374 [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp void
375 __micron_swap_context(__fiber_ctx *, __fiber_ctx *) noexcept
376 {
377   // cdecl: from @ 4(%esp), to @ 8(%esp)
378   asm volatile("movl 4(%esp), %eax\n\t"      // from
379                "movl %ebx, 0x00(%eax)\n\t"
380                "movl %esi, 0x04(%eax)\n\t"
381                "movl %edi, 0x08(%eax)\n\t"
382                "movl %ebp, 0x0c(%eax)\n\t"
383                "movl (%esp), %ecx\n\t"      // return addr -> from->eip
384                "movl %ecx, 0x14(%eax)\n\t"
385                "leal 4(%esp), %ecx\n\t"      // caller esp past ret -> from->esp
386                "movl %ecx, 0x10(%eax)\n\t"
387 #if defined(MICRON_FIBER_FPENV)
388 #if defined(__SSE__)
389                "stmxcsr 0x18(%eax)\n\t"
390                "andl $-64, 0x18(%eax)\n\t"      // drop sticky status bits (0-5); control modes only
391 #endif
392                "fnstcw 0x1c(%eax)\n\t"
393 #endif
394                "movl 8(%esp), %eax\n\t"      // to (esp still original here)
395 #if defined(MICRON_FIBER_FPENV)
396 #if defined(__SSE__)
397                "ldmxcsr 0x18(%eax)\n\t"
398 #endif
399                "fldcw 0x1c(%eax)\n\t"
400 #endif
401                "movl 0x00(%eax), %ebx\n\t"
402                "movl 0x04(%eax), %esi\n\t"
403                "movl 0x08(%eax), %edi\n\t"
404                "movl 0x0c(%eax), %ebp\n\t"
405                "movl 0x10(%eax), %esp\n\t"
406                "jmp *0x14(%eax)\n\t");
407 }

409 [[maybe_unused]] static __attribute__((naked, noinline, noreturn)) __micron_no_ssp void
410 __micron_fiber_trampoline() noexcept
411 {
412   asm volatile("andl $-16, %esp\n\t"
413                "xorl %ebp, %ebp\n\t"
414                "subl $12, %esp\n\t"      // keep 16-byte alignment after the arg push
415                "pushl %esi\n\t"          // arg
416                "calll *%edi\n\t"         // fn(arg)
417                "ud2\n\t");

420 [[gnu::always_inline]] inline void
421 make_context(__fiber_ctx *ctx, void *stack_top, void (*fn)(void *), void *arg) noexcept
422 {
423   *ctx = __fiber_ctx{};      // recycled FCBs: don't leak the previous fiber's callee-saved regs
424   ctx->esp = reinterpret_cast<u32>(stack_top) & ~static_cast<u32>(15);
425   ctx->eip = reinterpret_cast<u32>(&__micron_fiber_trampoline);
426   ctx->esi = reinterpret_cast<u32>(arg);
427   ctx->edi = reinterpret_cast<u32>(fn);

433 #if defined(__SSE__)
434   u32 __mx;
435   asm volatile("stmxcsr %0" : "=m"(__mx));
436   ctx->mxcsr = __mx & ~0x3Fu;
437 #endif
438 #endif

518 struct __fiber_ctx {
519   u32 r[8];      // 0x00  r4..r11 ; r[0]=r4 arg, r[1]=r5 fn (make_context)
520   u32 sp;        // 0x20
521   u32 lr;        // 0x24  resume pc
522   u64 d[8];      // 0x28  d8..d15 (guarded by __ARM_FP)
523 };
524 
525 static_assert(__builtin_offsetof(__fiber_ctx, sp) == 0x20, "arm32 __fiber_ctx sp offset");
526 static_assert(__builtin_offsetof(__fiber_ctx, lr) == 0x24, "arm32 __fiber_ctx lr offset");
527 static_assert(__builtin_offsetof(__fiber_ctx, d) == 0x28, "arm32 __fiber_ctx d offset");
528 static_assert(sizeof(__fiber_ctx) == 0x68, "arm32 __fiber_ctx size");
529 
530 [[maybe_unused]] static __attribute__((naked, noinline)) __micron_no_ssp void
531 __micron_swap_context(__fiber_ctx *, __fiber_ctx *) noexcept
532 {
533   // r0 = from, r1 = to
534   asm volatile("stm r0, {r4-r11}\n\t"      // r4..r11 at 0x00..0x1c
535                "str sp, [r0, #0x20]\n\t"
536                "str lr, [r0, #0x24]\n\t"
537 #if defined(__ARM_FP)
538                "add r2, r0, #0x28\n\t"
539                "vstm r2, {d8-d15}\n\t"      // callee-saved VFP regs
540                "add r2, r1, #0x28\n\t"
541                "vldm r2, {d8-d15}\n\t"
542 #endif
543                "ldr sp, [r1, #0x20]\n\t"
544                "ldr lr, [r1, #0x24]\n\t"
545                "ldm r1, {r4-r11}\n\t"
546                "bx lr\n\t");
547 }
548 
549 [[maybe_unused]] static __attribute__((naked, noinline, noreturn)) __micron_no_ssp void
550 __micron_fiber_trampoline() noexcept
551 {
552   // gcc's frame pointer is r11 in arm mode but r7 in thumb, and this
553   // toolchain defaults to thumb-2 (__thumb__), so zero both, we don't know which mode we're in
554   asm volatile("mov r11, #0\n\t"
555                "mov r7, #0\n\t"
556                "mov r0, r4\n\t"      // arg -> a0
557                "blx r5\n\t"          // fn(arg)
558                "udf #0\n\t");
559 }
560 
561 [[gnu::always_inline]] inline void
562 make_context(__fiber_ctx *ctx, void *stack_top, void (*fn)(void *), void *arg) noexcept
563 {
564   *ctx = __fiber_ctx{};      // recycled FCBs: don't leak the previous fiber's r6..r11 / d8..d15
565   ctx->sp = reinterpret_cast<u32>(stack_top) & ~static_cast<u32>(15);
566   ctx->lr = reinterpret_cast<u32>(&__micron_fiber_trampoline);
567   ctx->r[0] = reinterpret_cast<u32>(arg);      // r4
568   ctx->r[1] = reinterpret_cast<u32>(fn);       // r5
569 }
