64 struct __ct_type_checker {
65   static constexpr bool
66   check()
67   {
68     // WARNING: these typechecks are absolutely architecturally critical; the entire library essentially assumes that the following
69     // assertions are all true. if for some reason they aren't, everything breaks
70     static_assert(sizeof(byte) == 1, "byte must be exactly 1 byte");
71     static_assert(sizeof(byte) == sizeof(unsigned char), "byte must be exactly 1 unsigned char (check arch)");
72     static_assert(sizeof(int8_t) == 1, "int8_t must be 1 byte");
73     static_assert(sizeof(uint8_t) == 1, "uint8_t must be 1 byte");
74     static_assert(sizeof(int16_t) == 2, "int16_t must be 2 bytes");
75     static_assert(sizeof(uint16_t) == 2, "uint16_t must be 2 bytes");
76     static_assert(sizeof(int32_t) == 4, "int32_t must be 4 bytes");
77     static_assert(sizeof(uint32_t) == 4, "uint32_t must be 4 bytes");
78     static_assert(sizeof(int64_t) == 8, "int64_t must be 8 bytes");
79     static_assert(sizeof(uint64_t) == 8, "uint64_t must be 8 bytes");
80     // restore to spec
81     static_assert(sizeof(uintptr_t) == sizeof(usize), "uintptr_t should match usize");
82     static_assert(sizeof(uint64_t) == sizeof(umax_t), "uint64_t should match umax_t");
83 
84     static_assert(sizeof(int8_t) == sizeof(i8), "int8_t must be equaequal to i8");
85     static_assert(sizeof(uint8_t) == sizeof(u8), "uint8_t must be equal to u8");
86     static_assert(sizeof(int16_t) == sizeof(i16), "int16_t must be equal to i16");
87     static_assert(sizeof(uint16_t) == sizeof(u16), "uint16_t must be equal to u16");
88     static_assert(sizeof(int32_t) == sizeof(i32), "int32_t must be equal to i32");
89     static_assert(sizeof(uint32_t) == sizeof(u32), "uint32_t must be equal to u32");
90     static_assert(sizeof(int64_t) == sizeof(i64), "int64_t must be equal to i64");
91     static_assert(sizeof(uint64_t) == sizeof(u64), "uint64_t must be equal to u64");
92 
93     static_assert(sizeof(intptr_t) == sizeof(void *), "intptr_t must match pointer size");
94     static_assert(sizeof(uintptr_t) == sizeof(void *), "uintptr_t must match pointer size");
95     static_assert(sizeof(ptr_t) == sizeof(void *), "ptr_t must match pointer size");
96     static_assert(sizeof(addr_t) == sizeof(void *), "addr_t must match pointer size");
97     // NOTE: not true always
98     // static_assert(sizeof(usize) == sizeof(void *), "usize must match pointer size");
99     // static_assert(sizeof(max_t) == sizeof(ptrdiff_t), "max_t must match ptrdiff_t");
100 
101     // %%%%%%%%%%%%%%%%%%%%%%%%%%%%
102     // macro-poisoning
103     static_assert(__is_same(size_t, decltype(sizeof(0))), "size_t must be the type of sizeof");
104     static_assert(__is_same(usize, size_t), "usize must be size_t");
105     static_assert(__is_same(uintptr_t, size_t), "uintptr_t must be size_t on every micron target");
106     static_assert(__is_same(ptr_t, size_t), "ptr_t must be size_t");
107     static_assert(__is_same(addr_t, size_t), "addr_t must be size_t");
108     static_assert(__is_same(__SIZE_TYPE__, size_t), "__SIZE_TYPE__ must still name size_t");
109     static_assert(__is_same(ptrdiff_t, decltype(static_cast<char *>(nullptr) - static_cast<char *>(nullptr))),
110                   "ptrdiff_t must be the type of pointer subtraction");
111     static_assert(__is_same(intptr_t, ptrdiff_t), "intptr_t is ptrdiff_t on every micron target");
112     static_assert(__is_same(ssize_t, intptr_t), "ssize_t must be intptr_t");
113     static_assert(sizeof(ssize_t) == sizeof(size_t), "ssize_t must be size_t's width");
114     static_assert(__is_same(int32_t, int), "int32_t must be `int`");
115     static_assert(__is_same(uint32_t, unsigned int), "uint32_t must be `unsigned int`");
116     static_assert(__is_same(int8_t, signed char), "int8_t must be `signed char`");
117     static_assert(__is_same(uint8_t, unsigned char), "uint8_t must be `unsigned char`");
118 
119     // NOTE: each branch must use only macros defined on its arch; referring
120     // to a macro that isn't `#define`d as a bare C++ identifier fails on
121     // clang (which doesn't auto-substitute 0 outside of `#if` directives).
122 #if defined(__micron_arch_amd64)
123     static_assert(sizeof(long) == 8, "AMD64 long must be 64-bit (LP64)");
124 #elif defined(__micron_arch_x86)
125     static_assert(sizeof(long) == 4, "x86 long must be 32-bit (ILP32)");
126 #elif defined(__micron_arch_arm64)
127     static_assert(sizeof(long) == 8, "ARM64 long must be 64-bit");
128 #elif defined(__micron_arch_arm32)
129     static_assert(sizeof(long) == 4, "ARM32 long must be 32-bit");
130 #elif defined(__micron_arch_wasm32)
131     static_assert(sizeof(void *) == 4 && sizeof(long) == 4, "wasm32 uses ILP32");
132     static_assert(sizeof(long double) == 16 && alignof(long double) == 16, "wasm32 long double uses binary128");
133     static_assert(sizeof(__micron_guard_t) == 4, "wasm32 local-static guards are four bytes");
134 #endif
135 
136 #if defined(__GNUC__) && !defined(__clang__) && __cplusplus >= 202300L && defined(__micron_arch_amd64)
137     static_assert(sizeof(f16) == 2, "_Float16 must be 2 bytes");
138     static_assert(sizeof(f32) == 4, "_Float32 must be 4 bytes");
139     static_assert(sizeof(f64) == 8, "_Float64 must be 8 bytes");
140     static_assert(sizeof(f128) == 16, "_Float128 must be 16 bytes");
141 #elif defined(__clang__)
142     static_assert(sizeof(f16) == 4, "f16 fallback must be 4 bytes (float)");
143     static_assert(sizeof(f32) == 4, "f32 fallback must be 4 bytes (float)");
144     static_assert(sizeof(f64) == 8, "f64 fallback must be 8 bytes (double)");
145     static_assert(sizeof(f128) >= 8, "f128 fallback must be at least 8 bytes (long double)");
146 #else
147     static_assert(sizeof(float) == 4, "float must be 4 bytes");
148     static_assert(sizeof(double) == 8, "double must be 8 bytes");
149 #endif
150 
151     static_assert(sizeof(flong) >= 8, "long double must be at least 8 bytes");
152 
153     static_assert(sizeof(c8) == 1, "char8_t must be 1 byte");
154     static_assert(sizeof(c16) == 2, "char16_t must be 2 bytes");
155     static_assert(sizeof(c32) == 4, "char32_t must be 4 bytes");
156 
157     return true;
158   }
159 
160   static constexpr bool
161   check_posix()
162   {
163     static_assert(sizeof(__s16_type) == 2, "__s16_type must be 2 bytes");
164     static_assert(sizeof(__u16_type) == 2, "__u16_type must be 2 bytes");
165     static_assert(sizeof(__s32_type) == 4, "__s32_type must be 4 bytes");
166     static_assert(sizeof(__u32_type) == 4, "__u32_type must be 4 bytes");
167     static_assert(sizeof(__slongword_type) == sizeof(long), "__slongword_type must match long");
168     static_assert(sizeof(__ulongword_type) == sizeof(unsigned long), "__ulongword_type must match unsigned long");
169 
170     static_assert(sizeof(__sword_type) == sizeof(ssize_t), "__sword_type must match ssize_t");
171     static_assert(sizeof(__uword_type) == sizeof(usize), "__uword_type must match usize");
172     static_assert(sizeof(__squad_type) == 8 || sizeof(__squad_type) == sizeof(long), "__squad_type must be 8 bytes or long");
173     static_assert(sizeof(__uquad_type) == 8 || sizeof(__uquad_type) == sizeof(unsigned long),
174                   "__uquad_type must be 8 bytes or unsigned long");
175 
176     static_assert(sizeof(__slong32_type) == 4, "__slong32_type must be 4 bytes");
177     static_assert(sizeof(__ulong32_type) == 4, "__ulong32_type must be 4 bytes");
178     static_assert(sizeof(__s64_type) == 8, "__s64_type must be 8 bytes");
179     static_assert(sizeof(__u64_type) == 8, "__u64_type must be 8 bytes");
180 
181     static_assert(sizeof(__ptrdiff_type) == sizeof(ptrdiff_t), "__ptrdiff_type must match ptrdiff_t");
182     static_assert(sizeof(__syscall_slong_type) == sizeof(long), "__syscall_slong_type must match long");
183     static_assert(sizeof(__syscall_ulong_type) == sizeof(unsigned long), "__syscall_ulong_type must match unsigned long");
184 
185     static_assert(sizeof(__dev_t_type) == sizeof(__uquad_type), "__dev_t_type must match __uquad_type");
186     static_assert(sizeof(__uid_t_type) == 4, "__uid_t_type must be 4 bytes");
187     static_assert(sizeof(__gid_t_type) == 4, "__gid_t_type must be 4 bytes");
188     static_assert(sizeof(__ino_t_type) == sizeof(__uquad_type), "__ino_t_type must be 64-bit (LFS) == __uquad_type");
189     static_assert(sizeof(__ino64_t_type) == sizeof(__uquad_type), "__ino64_t_type must match __uquad_type");
190     static_assert(sizeof(__mode_t_type) == 4, "__mode_t_type must be 4 bytes");
191 
192 #ifdef __micron_arch_amd64
193     static_assert(sizeof(__nlink_t_type) == sizeof(__syscall_ulong_type), "amd64 kernel st_nlink is __kernel_ulong_t (8)");
194 #else
195     static_assert(sizeof(__nlink_t_type) == 4, "arm64/arm32/i386: asm-generic st_nlink is unsigned int (4)");
196 #endif
197     static_assert(sizeof(__fsword_t_type) == sizeof(__syscall_slong_type), "__fsword_t_type must match __syscall_slong_type");
198 
199     static_assert(sizeof(__off_t_type) == sizeof(__squad_type), "__off_t_type must be 64-bit (LFS) == __squad_type");
200     static_assert(sizeof(__off64_t_type) == sizeof(__squad_type), "__off64_t_type must match __squad_type");
201     static_assert(sizeof(__pid_t_type) == 4, "__pid_t_type must be 4 bytes");
202 
203     static_assert(sizeof(__rlim_t_type) == sizeof(__uquad_type), "__rlim_t_type must be 64-bit (LFS) == __uquad_type");
204     static_assert(sizeof(__rlim64_t_type) == sizeof(__uquad_type), "__rlim64_t_type must match __uquad_type");
205 
206     static_assert(sizeof(__blkcnt_t_type) == sizeof(__squad_type), "__blkcnt_t_type must be 64-bit (LFS) == __squad_type");
207     static_assert(sizeof(__blkcnt64_t_type) == sizeof(__squad_type), "__blkcnt64_t_type must match __squad_type");
208 
209     static_assert(sizeof(__fsblkcnt_t_type) == sizeof(__uquad_type), "__fsblkcnt_t_type must be 64-bit (LFS) == __uquad_type");
210     static_assert(sizeof(__fsblkcnt64_t_type) == sizeof(__uquad_type), "__fsblkcnt64_t_type must match __uquad_type");
211 
212     static_assert(sizeof(__fsfilcnt_t_type) == sizeof(__uquad_type), "__fsfilcnt_t_type must be 64-bit (LFS) == __uquad_type");
213     static_assert(sizeof(__fsfilcnt64_t_type) == sizeof(__uquad_type), "__fsfilcnt64_t_type must match __uquad_type");
214 
215     static_assert(sizeof(__id_t_type) == 4, "__id_t_type must be 4 bytes");
216 
217     static_assert(sizeof(__clock_t_type) == sizeof(__syscall_slong_type), "__clock_t_type must match __syscall_slong_type");
218     static_assert(sizeof(__clockid_t_type) == 4, "__clockid_t_type must be 4 bytes");
219     static_assert(sizeof(__time_t_type) == sizeof(__squad_type), "__time_t_type must be 64-bit (time64) == __squad_type");
220 
221     static_assert(sizeof(__timer_t_type) == sizeof(void *), "__timer_t_type must match void*");
222 
223     static_assert(sizeof(__useconds_t_type) == 4, "__useconds_t_type must be 4 bytes");
224     static_assert(sizeof(__suseconds_t_type) == sizeof(__squad_type), "__suseconds_t_type must be 64-bit (time64) == __squad_type");
225     static_assert(sizeof(__suseconds64_t_type) == sizeof(__squad_type), "__suseconds64_t_type must match __squad_type");
226 
227     static_assert(sizeof(__daddr_t_type) == 4, "__daddr_t_type must be 4 bytes");
228     static_assert(sizeof(__key_t_type) == 4, "__key_t_type must be 4 bytes");
229     static_assert(sizeof(__blksize_t_type) == sizeof(__syscall_slong_type), "__blksize_t_type must match __syscall_slong_type");
230     static_assert(sizeof(__ssize_t_type) == sizeof(__sword_type), "__ssize_t_type must match __sword_type");
231     static_assert(sizeof(__cpu_mask_type) == sizeof(__syscall_ulong_type), "__cpu_mask_type must match __syscall_ulong_type");
232 
233     return true;
234   }
235 };
