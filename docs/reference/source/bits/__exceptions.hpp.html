65 template<typename T>
66 inline void
67 __write_unsigned(T n)
68 {
69   char buf[32];
70   int i = 0;
71   if ( n == 0 ) {
72     buf[i++] = '0';
73   } else {
74     while ( n > 0 ) {
75       buf[i++] = static_cast<char>('0' + (n % 10));
76       n /= 10;
77     }
78     // reverse
79     for ( int j = 0; j < i / 2; ++j ) {
80       char tmp = buf[j];
81       buf[j] = buf[i - j - 1];
82       buf[i - j - 1] = tmp;
83     }
84   }
85   __write(buf, static_cast<usize>(i));
86 }

88 template<typename T, typename... Args>
89 __attribute__((always_inline, noreturn)) inline void
90 raise(Args &&...args)
91 {
92   T ex(args...);
93   __write("exception raised: ");
94   __write_n(ex.what());
95   __write("errno: ");
96   __write_unsigned(errno);
97   __write("\n");
98   abort(ex.which());
99 }

124 template<typename T, typename... Args>
125 __attribute__((always_inline, noreturn)) inline void
126 exc_silent(Args &&...args)
127 {
128   if constexpr ( micron::except::__use_exceptions )
