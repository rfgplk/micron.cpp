14 export function bytes(memory, offset = 0, length = memory.buffer.byteLength - offset) {
15   if (!Number.isInteger(offset) || !Number.isInteger(length) || offset < 0 || length < 0 ||
16       offset > memory.buffer.byteLength || length > memory.buffer.byteLength - offset)
17     throw new RangeError("Micron memory range is outside linear memory");
18   return new Uint8Array(memory.buffer, offset, length);
19 }

29 async function compile(source) {
30   if (source instanceof WebAssembly.Module) return source;
31   if (source instanceof Uint8Array || source instanceof ArrayBuffer) return WebAssembly.compile(source);
32   if (typeof Response !== "undefined" && source instanceof Response)
33     return WebAssembly.compile(await source.arrayBuffer());
34   if (node && (source instanceof URL ? source.protocol === "file:" : !/^https?:/.test(String(source)))) {
35     const fs = await import("node:fs/promises");
36     return WebAssembly.compile(await fs.readFile(source));
37   }
38   const response = await fetch(source);
39   if (!response.ok) throw new Error(`Micron fetch failed: ${response.status}`);
40   return WebAssembly.compile(await response.arrayBuffer());
41 }

45 export async function createLocal(module, memory, options = {}) {
46   const children = new Set();
47   const output = options.onWrite ?? ((text, stream) => {
48     if (node) (stream === 2 ? process.stderr : process.stdout).write(text);
49     else (stream === 2 ? console.error : console.log)(text);
50   });
51   const decoders = new Map();
52   function flushOutput() {
53     for (const [stream, decoder] of decoders) {
54       const text = decoder.decode();
55       if (text) output(text, stream);
56     }
57   }
58   const imports = {
59     env: {memory},
60     micron: {
61       write(pointer, length, stream) {
62         try {
63           const data = bytes(memory, pointer >>> 0, length >>> 0);
64           // TextDecoder's BufferSource excludes shared buffers in some browsers.
65           const input = typeof SharedArrayBuffer !== "undefined" && data.buffer instanceof SharedArrayBuffer
66             ? new Uint8Array(data) : data;
67           let decoder = decoders.get(stream);
68           if (!decoder) { decoder = new TextDecoder(); decoders.set(stream, decoder); }
69           const text = decoder.decode(input, {stream: true});
70           if (text) output(text, stream);
71           return length;
72         }
73         catch { return -14; }
74       },
75       random(pointer, length) {
76         try {
77           const out = bytes(memory, pointer >>> 0, length >>> 0);
78           const scratch = new Uint8Array(Math.min(out.length, 65536));
79           for (let i = 0; i < out.length; i += 65536) {
80             const chunk = scratch.subarray(0, Math.min(65536, out.length - i));
81             crypto.getRandomValues(chunk);
82             out.set(chunk, i);
83           }
84           return 0;
85         } catch { return -5; }
86       },
87       clock(kind) {
88         if (kind === 0) return BigInt(Date.now()) * 1000000n;
89         if (kind === 1) return BigInt(Math.floor((performance.timeOrigin + performance.now()) * 1e6));
90         if (kind === 2) return 1000000n;
91         if (kind === 3) return 1000000n;
92         return -95n;
93       },
94       panic(code) { throw new MicronExit(code); },
95       thread_exit() {
96         if (!options.threadExit) throw new Error("Micron worker exit is unavailable");
97         options.threadExit();
98         throw new Error("Micron worker exit handler returned");
99       },
100       yield() {
101         if (options.scheduler) Atomics.wait(options.scheduler, 0, 0, 0);
102       },
103       worker_spawn(slot, stack, tls, control) {
104         if (!options.scheduler || Atomics.load(options.scheduler, 0)) return -95;
105         for (let i = 1; i < options.scheduler.length; ++i) {
106           if (Atomics.compareExchange(options.scheduler, i, 0, 1) !== 0) continue;
107           options.dispatch({type: "spawn", pool: i - 1, slot, stack, tls, control});
108           return 0;
109         }
110         return -11;
111       }
112     }
113   };
114   const instance = await WebAssembly.instantiate(module, imports);
115   const e = instance.exports;
116   if (typeof e.__micron_wasm_abi_version !== "function")
117     throw new Error("Micron module is missing its host ABI version export");
118   const version = e.__micron_wasm_abi_version();
119   if (version !== 1) throw new Error(`Micron host ABI version ${version} is unsupported; expected 1`);
120   if (!e.__stack_pointer || !e.__micron_wasm_initialize)
121     throw new Error("Micron module is missing its startup exports");
122   function initialize() {
123     const tls = e.__micron_wasm_prepare();
124     if (e.__wasm_init_tls) {
125       if (!tls) throw new Error("Micron could not reserve initial TLS");
126       e.__wasm_init_tls(tls);
127     }
128     const initialized = e.__micron_wasm_initialize(options.worker ? 1 : 0);
129     if (initialized < 0) throw new Error(`Micron initialization failed: ${initialized}`);
130     return initialized;
131   }
132   if (!options.deferInitialize) initialize();
133   let stopped = false, shuttingDown = false;
134   function requireRunning() {
135     if (stopped) throw new Error("Micron runtime has stopped");
136     if (shuttingDown) throw new MicronStateError("Micron runtime is shutting down", -16);
137   }
138   return {
139     memory, instance, exports: e, initialize, flushOutput,
140     view: (offset, length) => bytes(memory, offset, length),
141     call(name, ...args) {
142       requireRunning();
143       if (typeof e[name] !== "function") throw new MicronStateError(`Micron export is missing: ${name}`, -38);
144       try { return e[name](...args); }
145       catch (error) { stopped = true; throw error; }
146     },
147     runMain(argc = 0, argv = 0) {
148       requireRunning();
149       try { return e.__micron_wasm_run_main(argc, argv); }
150       catch (error) { stopped = true; throw error; }
151     },
152     shutdown() {
153       if (stopped) return 0;
154       requireRunning();
155       shuttingDown = true;
156       let result;
157       try { result = e.__micron_wasm_shutdown(); }
158       catch (error) { stopped = true; throw error; }
159       finally { shuttingDown = false; }
160       if (result < 0) throw new MicronStateError(`Micron shutdown failed: ${result}`, result);
161       stopped = true;
162       flushOutput();
163       for (const child of children) child.terminate();
164       children.clear();
165       return result;
166     },
167     terminate() {
168       stopped = true;
169       for (const child of children) child.terminate();
170       children.clear();
171     }
172   };
173 }

175 export async function createMicron(source, options = {}) {
176   const threads = options.threads ?? false;
177   if (threads && !node && !globalThis.crossOriginIsolated)
178     throw new Error("Micron threads require cross-origin isolation (COOP: same-origin and COEP: require-corp)");
179   const module = await compile(source);
180   const memory = options.memory ?? new WebAssembly.Memory({
181     initial: options.initialPages ?? 256, maximum: options.maximumPages ?? 8192, shared: threads
182   });
183   if (threads !== (typeof SharedArrayBuffer !== "undefined" && memory.buffer instanceof SharedArrayBuffer))
184     throw new Error("Micron shared-memory profile does not match the supplied memory");
185   if (!threads) return createLocal(module, memory, options);
186   if (!WorkerClass) throw new Error("Micron threads require module workers");
187   const poolSize = options.poolSize ?? 4;
188   if (!Number.isInteger(poolSize) || poolSize < 1 || poolSize > 1023)
189     throw new RangeError("Micron poolSize must be in 1..1023");
190   const scheduler = new Int32Array(new SharedArrayBuffer((poolSize + 1) * 4));
191   const all = [];
192   try { for (let i = 0; i <= poolSize; ++i) all.push(newWorker()); }
193   catch (error) { await Promise.allSettled(all.map(w => w.terminate())); throw error; }
194   const [worker, ...pool] = all;
195   const pending = new Map();
196   const retiring = new Set();
197   let serial = 0, stopped = false, poisoned = false, shutdownPromise, termination;
198   const output = options.onWrite ?? ((text, stream) => {
199     if (node) (stream === 2 ? process.stderr : process.stdout).write(text);
200     else (stream === 2 ? console.error : console.log)(text);
201   });
202   const ready = [];
203   function rejectPending(error) {
204     for (const {reject} of pending.values()) reject(error);
205     pending.clear();
206   }
207   function stopWorkers() {
208     if (!termination) termination = Promise.allSettled([...new Set([...all, ...retiring])].map(w => {
209       try { return w.terminate(); } catch (error) { return Promise.reject(error); }
210     })).then(() => {});
211     return termination;
212   }
213   function poison(error) {
214     if (poisoned) return;
215     poisoned = true; stopped = true;
216     Atomics.store(scheduler, 0, 1);
217     for (const entry of ready) entry.reject(error);
218     rejectPending(error);
219     stopWorkers();
220   }
221   function bootWorker(w, index) {
222     const entry = {};
223     const readiness = new Promise((resolve, reject) => { entry.resolve = resolve; entry.reject = reject; });
224     // Posting may fail before the caller receives the readiness promise.
225     readiness.catch(() => {});
226     ready[index] = entry;
227     listen(w, message => {
228       if (stopped || all[index] !== w || retiring.has(w)) return;
229       try {
230         if (message.type === "ready") entry.resolve();
231         else if (message.type === "write") output(message.text, message.stream);
232         else if (message.type === "failed") poison(message.name === "MicronExit" ? new MicronExit(message.code) : new Error(message.error));
233         else if (message.type === "complete") {
234           Atomics.store(scheduler, message.pool + 1, 0);
235           const state = new Int32Array(memory.buffer, message.control >>> 0, 1);
236           Atomics.store(state, 0, message.result < 0 ? 4 : 3);
237           Atomics.notify(state, 0);
238         } else if (message.type === "worker-exit") {
239           if (index === 0 || message.pool !== index - 1) throw new Error("Invalid Micron worker exit");
240           replaceWorker(w, index, message).catch(poison);
241         } else if (message.type === "spawn") {
242           if (!Number.isInteger(message.pool) || !pool[message.pool]) throw new Error("Invalid Micron worker dispatch");
243           pool[message.pool].postMessage({...message, type: "run"});
244         } else if (message.type === "reply") {
245           const item = pending.get(message.id);
246           if (!item) return;
247           pending.delete(message.id);
248           if (message.error) item.reject(message.name === "MicronStateError"
249             ? new MicronStateError(message.error, message.code) : new Error(message.error));
250           else item.resolve(message.result);
251         }
252       } catch (error) { poison(error); }
253     });
254     failure(w, error => { if (all[index] === w && !retiring.has(w)) poison(error); });
255     if (node) w.on("exit", code => {
256       if (!termination && all[index] === w && !retiring.has(w))
257         poison(new Error(`Micron worker exited unexpectedly with status ${code}`));
258     });
259     else w.addEventListener("messageerror", () => {
260       if (all[index] === w && !retiring.has(w)) poison(new Error("Micron worker message could not be decoded"));
261     });
262     w.postMessage({type: "boot", module, memory, primary: index === 0, scheduler, pool: index - 1});
263     return readiness;
264   }
265   async function replaceWorker(previous, index, message) {
266     retiring.add(previous);
267     try { await previous.terminate(); }
268     finally { retiring.delete(previous); }
269     if (stopped) return;
270     const replacement = newWorker();
271     all[index] = replacement;
272     pool[index - 1] = replacement;
273     await bootWorker(replacement, index);
274     if (!stopped) replacement.postMessage({...message, type: "cleanup"});
275   }
276   function rpc(operation, args = []) {
277     if (stopped) return Promise.reject(new Error("Micron runtime has stopped"));
278     if (shutdownPromise && operation !== "shutdown")
279       return Promise.reject(new MicronStateError("Micron runtime is shutting down", -16));
280     const id = ++serial;
281     return new Promise((resolve, reject) => {
282       pending.set(id, {resolve, reject});
283       try { worker.postMessage({type: "call", id, operation, args}); }
284       catch (error) { pending.delete(id); reject(error); }
285     });
286   }
287   try { await Promise.all(all.map(bootWorker)); await rpc("initialize"); }
288   catch (error) { poison(error); await stopWorkers(); throw error; }
289   return {
290     memory, view: (offset, length) => bytes(memory, offset, length),
291     call: (name, ...args) => rpc("call", [name, ...args]),
292     runMain: (argc = 0, argv = 0) => rpc("runMain", [argc, argv]),
293     shutdown() {
294       if (shutdownPromise) return shutdownPromise;
295       if (stopped) return Promise.resolve(0);
296       const attempt = (async () => {
297         try {
298           const result = await rpc("shutdown");
299           stopped = true;
300           Atomics.store(scheduler, 0, 1);
301           rejectPending(new Error("Micron runtime has stopped"));
302           await stopWorkers();
303           return result;
304         } catch (error) {
305           if (!(error instanceof MicronStateError)) poison(error);
306           throw error;
307         } finally { if (shutdownPromise === attempt) shutdownPromise = undefined; }
308       })();
309       shutdownPromise = attempt;
310       return attempt;
311     },
312     terminate() {
313       poison(new Error("Micron runtime terminated"));
314       return stopWorkers();
315     }
316   };
317 }
