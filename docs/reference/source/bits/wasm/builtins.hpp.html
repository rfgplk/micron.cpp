7 __attribute__((weak, used)) __int128
8 __multi3(__int128 a, __int128 b) noexcept
9 {
10   struct words {
11     unsigned long long word[2];
12   };
13 
14   const auto x = __builtin_bit_cast(words, a), y = __builtin_bit_cast(words, b);
15   words r{};
16   const unsigned long long al = static_cast<unsigned>(x.word[0]), ah = x.word[0] >> 32;
17   const unsigned long long bl = static_cast<unsigned>(y.word[0]), bh = y.word[0] >> 32;
18   const unsigned long long p0 = al * bl, p1 = al * bh, p2 = ah * bl, p3 = ah * bh;
19   const unsigned long long mid = (p0 >> 32) + static_cast<unsigned>(p1) + static_cast<unsigned>(p2);
20   r.word[0] = (mid << 32) | static_cast<unsigned>(p0);
21   r.word[1] = p3 + (p1 >> 32) + (p2 >> 32) + (mid >> 32) + x.word[1] * y.word[0] + x.word[0] * y.word[1];
22   return __builtin_bit_cast(__int128, r);
23 }

34 __attribute__((weak, used)) void *
35 memmove(void *d, const void *s, __SIZE_TYPE__ n)
36 {
37   auto *out = static_cast<volatile unsigned char *>(d);
38   const auto *in = static_cast<const volatile unsigned char *>(s);
39   if ( reinterpret_cast<__UINTPTR_TYPE__>(d) < reinterpret_cast<__UINTPTR_TYPE__>(s) )
40     for ( __SIZE_TYPE__ i = 0; i < n; ++i ) out[i] = in[i];
41   else
42     for ( __SIZE_TYPE__ i = n; i; --i ) out[i - 1] = in[i - 1];
43   return d;
44 }

56 {
57   const auto *x = static_cast<const unsigned char *>(a), *y = static_cast<const unsigned char *>(b);
