5 import { Worker, isMainThread, parentPort, workerData } from 'node:worker_threads';
6 import { readFile } from 'node:fs/promises';
7 
8 if (isMainThread) {
9   const [file, ...args] = process.argv.slice(2);
10   const options = { file, threads: false, initialPages: 256, maximumPages: 8192, timeout: 30, minCases: 0 };
11   try {
12     if (!file || file.startsWith('-')) throw new Error('usage: node node.mjs module.wasm [--threads] [--timeout=seconds]');
13     for (const arg of args) {
14       if (arg === '--threads') { options.threads = true; continue; }
15       const match = /^--(initial-pages|maximum-pages|timeout|min-cases)=([0-9]+)$/.exec(arg);
16       if (!match) throw new Error(`unknown option: ${arg}`);
17       const value = Number(match[2]);
18       if (!Number.isSafeInteger(value) || value <= 0) throw new Error(`invalid option: ${arg}`);
19       options[{ 'initial-pages': 'initialPages', 'maximum-pages': 'maximumPages', timeout: 'timeout', 'min-cases': 'minCases' }[match[1]]] = value;
20     }
21     const worker = new Worker(new URL(import.meta.url), { workerData: options });
22     const timer = setTimeout(() => {
23       process.stderr.write(`Micron WASM timed out after ${options.timeout}s\n`);
24       process.exit(124);
25     }, options.timeout * 1000);
26     let completed = false;
27     const finish = (code) => {
28       if (completed) return;
29       completed = true;
30       clearTimeout(timer);
31       Promise.all([process.stdout, process.stderr].map(stream => new Promise(resolve => stream.write('', resolve))))
32         .then(() => process.exit(code));
33     };
34     worker.on('message', message => {
35       if (message.type === 'write') {
36         (message.stream === 2 ? process.stderr : process.stdout).write(message.text);
37         return;
38       }
39       if (message.error) process.stderr.write(`${message.error}\n`);
40       finish(message.code);
41     });
42     worker.on('error', error => { process.stderr.write(`${error.stack}\n`); finish(70); });
43     worker.on('exit', code => { if (!completed) finish(code || 70); });
44   } catch (error) {
45     process.stderr.write(`${error.message}\n`);
46     process.exit(64);
47   }
48 } else {
49   let runtime;
50   try {
51     const { createMicron } = await import('./host.mjs');
52     runtime = await createMicron(await readFile(workerData.file), {
53       ...workerData, onWrite: (text, stream) => parentPort.postMessage({type: 'write', text, stream})
54     });
55     const code = await runtime.runMain();
56     if (workerData.minCases && code === 1) {
57       const count = async name => {
58         const value = await runtime.call(name);
59         if (!Number.isInteger(value) || value < -0x80000000 || value > 0xffffffff)
60           throw new Error(`invalid unsigned test coverage export ${name}: ${String(value)}`);
61         return value >>> 0;
62       };
63       const cases = await count('__micron_wasm_test_cases');
64       const skips = await count('__micron_wasm_test_skips');
65       if (cases < workerData.minCases || skips !== 0)
66         throw new Error(`Micron test coverage failed: ${cases} cases, ${skips} skips; expected >=${workerData.minCases} cases and no skips`);
67     }
68     await runtime.shutdown();
69     if (!Number.isInteger(code) || code < 0 || code > 255) throw new Error(`invalid Micron exit code: ${code}`);
70     parentPort.postMessage({ code });
71   } catch (error) {
72     try { await runtime?.terminate(); } catch {}
73     parentPort.postMessage({ code: error.name === 'MicronExit' && Number.isInteger(error.code) ? error.code : 70,
74                              error: error.stack || String(error) });
75   }
76 }
