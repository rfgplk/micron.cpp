11 template<typename F> struct format {
12   using uint_type = micron::conditional_t<sizeof(F) == 4, u32, micron::conditional_t<sizeof(F) == 8, u64, wide>>;
13   static constexpr unsigned precision = sizeof(F) == 4 ? 24 : sizeof(F) == 8 ? 53 : 113;
14   static constexpr unsigned mantissa = precision - 1;
15   static constexpr unsigned exponent = sizeof(F) == 4 ? 8 : sizeof(F) == 8 ? 11 : 15;
16   static constexpr int bias = (1 << (exponent - 1)) - 1;
17   static constexpr unsigned emax = (1u << exponent) - 1;
18   static constexpr wide implicit = wide(1) << mantissa;
19   static constexpr wide fraction = implicit - 1;
20   static constexpr wide sign = wide(1) << (mantissa + exponent);
21   static constexpr wide infinity = wide(emax) << mantissa;
22   static constexpr wide quiet = implicit >> 1;
23 };

74 inline constexpr int
75 msb(wide x) noexcept
76 {
77   const u64 hi = u64(x >> 64);
78   return hi ? 127 - __builtin_clzll(hi) : u64(x) ? 63 - __builtin_clzll(u64(x)) : -1;
79 }

81 inline constexpr wide
82 jam(wide x, unsigned n) noexcept
83 {
84   return n == 0 ? x : n >= 128 ? wide(x != 0) : (x >> n) | wide((x << (128 - n)) != 0);
85 }

87 template<typename F>
88 inline constexpr F
89 pack(wide sign, int exponent, wide significand) noexcept
90 {
91   using T = format<F>;
92   if ( exponent >= int(T::emax) ) return from<F>(sign | T::infinity);
93   if ( exponent <= 0 ) {
94     significand = jam(significand, unsigned(1 - exponent));
95     exponent = 0;
96   }
97   const unsigned grs = unsigned(significand & 7);
98   wide result = (wide(exponent) << T::mantissa) | ((significand >> 3) & T::fraction);
99   if ( grs > 4 || (grs == 4 && ((significand >> 3) & 1)) ) ++result;
100   return from<F>(sign | result);
101 }

103 struct parts {

108 template<typename F>
109 inline constexpr parts
110 decompose(F x) noexcept
111 {
112   using T = format<F>;
113   const wide b = bits(x) & ~T::sign;
114   const int e = int(b >> T::mantissa);
115   return { (b & T::fraction) | (e ? T::implicit : 0), (e ? e : 1) - T::bias - int(T::mantissa) };
116 }

118 template<typename F>
119 inline constexpr F
120 scalbn(F x, int exponent) noexcept
121 {
122   using T = format<F>;
123   if ( x == F(0) || isnan(x) || isinf(x) ) return x;
124   auto p = decompose(x);
125   const int leading = msb(p.significand);
126   p.significand <<= T::mantissa - leading;
127   if ( exponent > 4 * T::bias ) exponent = 4 * T::bias;
128   if ( exponent < -4 * T::bias ) exponent = -4 * T::bias;
129   return pack<F>(bits(x) & T::sign, p.exponent + leading + T::bias + exponent, p.significand << 3);
130 }

132 template<typename F>
133 inline constexpr F
134 frexp(F x, int *exponent) noexcept
135 {
136   using T = format<F>;
137   if ( x == F(0) || isnan(x) || isinf(x) ) {
138     *exponent = 0;
139     return x;
140   }
141   const auto p = decompose(x);
142   const int leading = msb(p.significand);
143   *exponent = p.exponent + leading + 1;
144   return from<F>((bits(x) & T::sign) | (wide(T::bias - 1) << T::mantissa) | ((p.significand << (T::mantissa - leading)) & T::fraction));
145 }

147 template<typename F>
148 inline constexpr F
149 trunc(F x) noexcept
150 {
151   using T = format<F>;
152   const wide b = bits(x);
153   const int e = int((b & ~T::sign) >> T::mantissa) - T::bias;
154   if ( e < 0 ) return from<F>(b & T::sign);
155   if ( e >= int(T::mantissa) ) return x;
156   return from<F>(b & ~((wide(1) << (T::mantissa - e)) - 1));
157 }

175 template<typename F>
176 inline constexpr F
177 nearest(F x) noexcept
178 {
179   using T = format<F>;
180   if ( isnan(x) || isinf(x) || abs(x) >= from<F>(wide(T::bias + T::mantissa) << T::mantissa) ) return x;
181   const F a = abs(x), integral = trunc(a), fraction = a - integral;
182   const int exponent = int((bits(integral) & ~T::sign) >> T::mantissa) - T::bias;
183   const bool odd = exponent >= 0 && ((bits(integral) | T::implicit) >> (T::mantissa - exponent) & 1) != 0;
184   return copysign(integral + F(fraction > F(0.5) || (fraction == F(0.5) && odd)), x);
185 }

187 template<typename F>
188 inline constexpr F
189 round(F x) noexcept
190 {
191   if ( isnan(x) || isinf(x) ) return x;
192   F a = abs(x), integral = trunc(a);
193   return copysign(integral + F(a - integral >= F(0.5)), x);
194 }

196 template<typename F>
197 inline constexpr F
198 sqrt(F x) noexcept
199 {
200   using T = format<F>;
201   if ( isnan(x) ) return from<F>(bits(x) | T::quiet);
202   if ( x == F(0) ) return x;
203   if ( x < F(0) ) return nan<F>();
204   if ( isinf(x) ) return x;
205   auto p = decompose(x);
206   int leading = msb(p.significand), exponent = p.exponent + leading;
207   p.significand <<= T::mantissa - leading;
208   if ( exponent & 1 ) {
209     p.significand <<= 1;
210     --exponent;
211   }
212   constexpr unsigned shift = T::mantissa + 4;
213   wide root = 0, remainder = 0;
214   for ( int pair = int(T::precision + 1); pair >= 0; --pair ) {
215     unsigned digits = 0;
216     for ( int bit = 1; bit >= 0; --bit ) {
217       const int pos = 2 * pair + bit - int(shift);
218       digits = (digits << 1) | (pos >= 0 && pos < 128 ? unsigned((p.significand >> pos) & 1) : 0);
219     }
220     remainder = (remainder << 2) | digits;
221     root <<= 1;
222     const wide trial = (root << 1) | 1;
223     if ( remainder >= trial ) {
224       remainder -= trial;
225       ++root;
226     }
227   }
228   return pack<F>(0, exponent / 2 + T::bias, (root << 1) | wide(remainder != 0));
229 }

235 inline constexpr accumulator
236 term(const u64 *words, unsigned count, int offset) noexcept
237 {
238   accumulator out{};
239   bool sticky = false;
240   for ( unsigned i = 0; i < count * 64; ++i )
241     if ( (words[i / 64] >> (i % 64)) & 1 ) {
242       const int bit = int(i) + offset;
243       if ( bit < 0 )
244         sticky = true;
245       else if ( bit < 512 )
246         out.limb[bit / 64] |= u64(1) << (bit % 64);
247     }
248   if ( sticky ) out.limb[0] |= 1;
249   return out;
250 }

252 inline constexpr int
253 compare(const accumulator &a, const accumulator &b) noexcept
254 {
255   for ( int i = 7; i >= 0; --i )
256     if ( a.limb[i] != b.limb[i] ) return a.limb[i] < b.limb[i] ? -1 : 1;
257   return 0;
258 }

260 inline constexpr accumulator
261 add(accumulator a, const accumulator &b) noexcept
262 {
263   u64 carry = 0;
264   for ( unsigned i = 0; i < 8; ++i ) {
265     const wide v = wide(a.limb[i]) + b.limb[i] + carry;
266     a.limb[i] = u64(v);
267     carry = u64(v >> 64);
268   }
269   return a;
270 }
271 
272 inline constexpr accumulator
273 subtract(accumulator a, const accumulator &b) noexcept
274 {
275   u64 borrow = 0;
276   for ( unsigned i = 0; i < 8; ++i ) {
277     const u64 old = a.limb[i], sub = b.limb[i] + borrow;
278     borrow = sub < b.limb[i] || old < sub;
279     a.limb[i] = old - sub;
280   }
281   return a;
282 }

284 template<typename F>
285 inline constexpr F
286 fma(F a, F b, F c) noexcept
287 {
288   using T = format<F>;
289   const wide ab = bits(a), bb = bits(b), cb = bits(c), sign = (ab ^ bb) & T::sign;
290   if ( isnan(a) ) return from<F>(ab | T::quiet);
291   if ( isnan(b) ) return from<F>(bb | T::quiet);
292   if ( isnan(c) ) return from<F>(cb | T::quiet);
293   if ( isinf(a) || isinf(b) ) {
294     if ( a == F(0) || b == F(0) || (isinf(c) && ((cb ^ sign) & T::sign)) ) return nan<F>();
295     return from<F>(sign | T::infinity);
296   }
297   if ( isinf(c) ) return c;
298   if ( a == F(0) || b == F(0) ) return c == F(0) ? from<F>(sign & cb & T::sign) : c;
299   const auto ap = decompose(a), bp = decompose(b), cp = decompose(c);
300   const u64 al = u64(ap.significand), ah = u64(ap.significand >> 64), bl = u64(bp.significand), bh = u64(bp.significand >> 64);
301   const wide ll = wide(al) * bl, lh = wide(al) * bh, hl = wide(ah) * bl, hh = wide(ah) * bh;
302   const wide mid = (ll >> 64) + u64(lh) + u64(hl), hi = hh + (lh >> 64) + (hl >> 64) + (mid >> 64);
303   const u64 product[4]{ u64(ll), u64(mid), u64(hi), u64(hi >> 64) };
304   const u64 cwords[2]{ u64(cp.significand), u64(cp.significand >> 64) };
305   const int product_msb = hi ? 128 + msb(hi) : 64 + (u64(mid) ? 63 - __builtin_clzll(u64(mid)) : -64 + msb(wide(u64(ll))));
306   const int pe = ap.exponent + bp.exponent;
307   const int top_p = pe + product_msb, top_c = cp.significand ? cp.exponent + msb(cp.significand) : top_p;
308   const int top = top_p > top_c ? top_p : top_c;
309   auto p = term(product, 4, 500 + pe - top), q = term(cwords, 2, 500 + cp.exponent - top);
310   wide result_sign = sign;
311   if ( (cb & T::sign) == sign )
312     p = add(p, q);
313   else if ( compare(p, q) < 0 ) {
314     p = subtract(q, p);
315     result_sign = cb & T::sign;
316   } else
317     p = subtract(p, q);
318   int leading = -1;
319   for ( int i = 7; i >= 0; --i )
320     if ( p.limb[i] ) {
321       leading = i * 64 + 63 - __builtin_clzll(p.limb[i]);
322       break;
323     }
324   if ( leading < 0 ) return F(0);
325   const int shift = leading - int(T::mantissa) - 3;
326   wide significand = 0;
327   bool sticky = false;
328   for ( int i = 0; i <= leading; ++i )
329     if ( (p.limb[i / 64] >> (i % 64)) & 1 ) {
330       if ( i < shift )
331         sticky = true;
332       else
333         significand |= wide(1) << (i - shift);
334     }
335   if ( sticky ) significand |= 1;
336   return pack<F>(result_sign, top - 500 + leading + T::bias, significand);
337 }
