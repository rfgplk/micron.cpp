9 template<typename F>
10 inline F
11 integer_float(wide magnitude, bool negative) noexcept
12 {
13   using T = format<F>;
14   if ( !magnitude ) return F(0);
15   const int leading = msb(magnitude), shift = leading - int(T::mantissa) - 3;
16   const wide extended = shift > 0 ? jam(magnitude, unsigned(shift)) : magnitude << -shift;
17   return pack<F>(negative ? T::sign : 0, leading + T::bias, extended);
18 }

20 template<typename F, bool Signed>
21 inline wide
22 float_integer(F x) noexcept
23 {
24   using T = format<F>;
25   const bool negative = (bits(x) & T::sign) != 0;
26   const wide maximum = Signed ? (wide(1) << 127) - !negative : ~wide(0);
27   if constexpr ( !Signed )
28     if ( negative ) return 0;
29   if ( isnan(x) || isinf(x) ) return negative && Signed ? wide(0) - maximum : maximum;
30   const auto p = decompose(x);
31   const int leading = msb(p.significand) + p.exponent;
32   if ( leading < 0 ) return 0;
33   if ( leading >= (Signed ? 127 : 128) ) return negative && Signed ? wide(0) - maximum : maximum;
34   const wide value = p.exponent < 0 ? p.significand >> -p.exponent : p.significand << p.exponent;
35   return negative ? wide(0) - value : value;
36 }
