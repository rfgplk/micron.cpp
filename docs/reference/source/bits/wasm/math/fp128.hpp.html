38   inline constexpr int __tf_mantbits = 112;      // stored mantissa bits
39   inline constexpr int __tf_bias = 16383;
40   inline constexpr unsigned __tf_einf = 0x7FFFu;                                            // all-ones exponent field
41   inline constexpr __tf_rep __tf_implicit = static_cast<__tf_rep>(1) << __tf_mantbits;      // 2^112
42   inline constexpr __tf_rep __tf_mantmask = __tf_implicit - 1;                              // low 112 bits
43   inline constexpr __tf_rep __tf_signbit = static_cast<__tf_rep>(1) << 127;
44   inline constexpr __tf_rep __tf_absmask = ~__tf_signbit;

46   inline constexpr __tf_rep __tf_qnanbit = static_cast<__tf_rep>(1) << (__tf_mantbits - 1);
47   inline constexpr __tf_rep __tf_qnan = __tf_infrep | __tf_qnanbit;

53   [[gnu::always_inline]] inline int __tf_clz(__tf_rep x) noexcept      // precondition: x != 0
54   {
55     const unsigned long long hi = static_cast<unsigned long long>(x >> 64);
56     if ( hi ) return __builtin_clzll(hi);
57     return 64 + __builtin_clzll(static_cast<unsigned long long>(x));
58   }

60   // shift a (sub)normal significand so its leading 1 lands at bit __tf_mantbits
61   [[gnu::always_inline]] inline int __tf_normalize(__tf_rep & sig) noexcept
62   {
63     const int shift = __tf_clz(sig) - (127 - __tf_mantbits);
64     sig <<= shift;
65     return 1 - shift;
66   }

68   [[gnu::always_inline]] inline __tf_rep __tf_shift_jam(__tf_rep sig, unsigned shift) noexcept
69   {
70     if ( shift == 0 ) return sig;
71     if ( shift >= 128 ) return sig != 0;
72     return (sig >> shift) | ((sig << (128 - shift)) != 0);
73   }

75   [[gnu::always_inline]] inline __tf_t __tf_round_pack(__tf_rep sign, int exp, __tf_rep extended) noexcept
76   {
77     if ( exp >= static_cast<int>(__tf_einf) ) return __tf_from(sign | __tf_infrep);
78     if ( exp <= 0 ) {
79       extended = __tf_shift_jam(extended, unsigned(1 - exp));
80       exp = 0;
81     }
82     const unsigned grs = unsigned(extended & 7);
83     __tf_rep result = (static_cast<__tf_rep>(exp) << __tf_mantbits) | ((extended >> 3) & __tf_mantmask);
84     if ( grs > 4 || (grs == 4 && ((extended >> 3) & 1)) ) ++result;
85     return __tf_from(sign | result);
86   }

88   inline __tf_t __tf_addsub(__tf_rep a, __tf_rep b) noexcept
89   {
90     __tf_rep aAbs = a & __tf_absmask;
91     __tf_rep bAbs = b & __tf_absmask;
92 
93     if ( aAbs >= __tf_infrep || bAbs >= __tf_infrep || aAbs == 0 || bAbs == 0 ) {
94       if ( aAbs > __tf_infrep ) return __tf_from(a | __tf_qnanbit);      // a is NaN -> quiet
95       if ( bAbs > __tf_infrep ) return __tf_from(b | __tf_qnanbit);
96       if ( aAbs == __tf_infrep ) {
97         if ( bAbs == __tf_infrep && ((a ^ b) >> 127) ) return __tf_from(__tf_qnan);      // inf + -inf
98         return __tf_from(a);
99       }
100       if ( bAbs == __tf_infrep ) return __tf_from(b);
101       if ( aAbs == 0 ) return __tf_from(bAbs == 0 ? (a & b & __tf_signbit) : b);      // -0 + -0 = -0
102       return __tf_from(a);                                                            // b == 0
103     }
104 
105     if ( bAbs > aAbs ) {      // keep |a| >= |b|
106       __tf_rep t = a;
107       a = b;
108       b = t;
109       t = aAbs;
110       aAbs = bAbs;
111       bAbs = t;
112     }
113 
114     int aExp = static_cast<int>(aAbs >> __tf_mantbits);
115     int bExp = static_cast<int>(bAbs >> __tf_mantbits);
116     __tf_rep aSig = aAbs & __tf_mantmask;
117     __tf_rep bSig = bAbs & __tf_mantmask;
118     if ( aExp == 0 )
119       aExp = __tf_normalize(aSig);
120     else
121       aSig |= __tf_implicit;
122     if ( bExp == 0 )
123       bExp = __tf_normalize(bSig);
124     else
125       bSig |= __tf_implicit;
126 
127     const bool subtraction = ((a ^ b) >> 127) != 0;
128     const __tf_rep resultSign = a & __tf_signbit;
129 
130     aSig <<= 3;      // make room for guard/round/sticky in the low 3 bits
131     bSig <<= 3;
132     const int align = aExp - bExp;
133     if ( align ) {
134       if ( align < 128 ) {
135         const __tf_rep sticky = (bSig << (128 - align)) != 0 ? static_cast<__tf_rep>(1) : 0;
136         bSig = (bSig >> align) | sticky;
137       } else {
138         bSig = 1;      // wholly below the sticky bit
139       }
140     }
141 
142     if ( subtraction ) {
143       aSig -= bSig;
144       if ( aSig == 0 ) return __tf_from(0);      // exact cancellation -> +0
145       if ( aSig < (__tf_implicit << 3) ) {       // renormalize after cancellation
146         const int shift = __tf_clz(aSig) - (127 - (__tf_mantbits + 3));
147         aSig <<= shift;
148         aExp -= shift;
149       }
150     } else {
151       aSig += bSig;
152       if ( aSig & (__tf_implicit << 4) ) {      // carry out of the top
153         const __tf_rep sticky = aSig & 1;
154         aSig = (aSig >> 1) | sticky;
155         ++aExp;
156       }
157     }
158 
159     return __tf_round_pack(resultSign, aExp, aSig);
160   }

162   [[gnu::always_inline]] inline void __tf_wide_mul(__tf_rep a, __tf_rep b, __tf_rep & hi, __tf_rep & lo) noexcept
163   {
164     const unsigned long long al = static_cast<unsigned long long>(a), ah = static_cast<unsigned long long>(a >> 64);
165     const unsigned long long bl = static_cast<unsigned long long>(b), bh = static_cast<unsigned long long>(b >> 64);
166     const __tf_rep ll = static_cast<__tf_rep>(al) * bl;
167     const __tf_rep lh = static_cast<__tf_rep>(al) * bh;
168     const __tf_rep hl = static_cast<__tf_rep>(ah) * bl;
169     const __tf_rep hh = static_cast<__tf_rep>(ah) * bh;
170     const __tf_rep mid = (ll >> 64) + static_cast<unsigned long long>(lh) + static_cast<unsigned long long>(hl);
171     lo = static_cast<unsigned long long>(ll) | (mid << 64);
172     hi = hh + (lh >> 64) + (hl >> 64) + (mid >> 64);
173   }

175   // three-way ordered compare (callers handle NaN): -1/0/+1; +-0 compare equal
176   [[gnu::always_inline]] inline int __tf_cmp3(__tf_rep a, __tf_rep b) noexcept
177   {
178     const __tf_rep aAbs = a & __tf_absmask, bAbs = b & __tf_absmask;
179     if ( (aAbs | bAbs) == 0 ) return 0;
180     const bool aNeg = (a >> 127) != 0, bNeg = (b >> 127) != 0;
181     if ( aNeg != bNeg ) return aNeg ? -1 : 1;
182     if ( aAbs == bAbs ) return 0;
183     return ((aAbs < bAbs) != aNeg) ? -1 : 1;
184   }

191   // round a binary128 to a narrower binary format
192   template<typename U, int MBITS, int EBITS, int BIAS> [[gnu::always_inline]] inline U __tf_trunc_to(__tf_rep b) noexcept
193   {
194     const U sign = static_cast<U>(b >> 127) << (MBITS + EBITS);
195     const int e = static_cast<int>((b >> __tf_mantbits) & __tf_einf);
196     const __tf_rep m = b & __tf_mantmask;
197     const U emax = (static_cast<U>(1) << EBITS) - 1;
198     if ( e == static_cast<int>(__tf_einf) ) {
199       if ( m == 0 ) return sign | (emax << MBITS);      // inf
200       U pay = static_cast<U>(m >> (__tf_mantbits - MBITS));
201       pay |= static_cast<U>(1) << (MBITS - 1);
202       return sign | (emax << MBITS) | pay;
203     }
204     int ue;
205     __tf_rep sig;
206     if ( e == 0 ) {
207       if ( m == 0 ) return sign;      // +-0
208       ue = 1 - __tf_bias;             // f128 subnormal: far below any double/float subnormal
209       sig = m;
210     } else {
211       ue = e - __tf_bias;
212       sig = __tf_implicit | m;
213     }
214     const int te = ue + BIAS;
215     if ( te >= static_cast<int>(emax) ) return sign | (emax << MBITS);      // overflow -> inf
216     int rshift = __tf_mantbits - MBITS;
217     if ( te <= 0 ) rshift += 1 - te;
218     if ( rshift > 113 ) return sign;
219     U kept = static_cast<U>(sig >> rshift);
220     const __tf_rep rem = sig & ((static_cast<__tf_rep>(1) << rshift) - 1);
221     const __tf_rep half = static_cast<__tf_rep>(1) << (rshift - 1);
222     if ( rem > half || (rem == half && (kept & 1)) ) ++kept;
223     const U base = te > 0 ? static_cast<U>(te - 1) << MBITS : 0;
224     return sign | (base + kept);
225   }

243 extern "C" __mc_libgcc_sym __tf_t
244 __multf3(__tf_t fa, __tf_t fb) noexcept
245 {
246   const __tf_rep a = __tf_bits(fa), b = __tf_bits(fb);
247   const __tf_rep aAbs = a & __tf_absmask, bAbs = b & __tf_absmask;
248   const __tf_rep sign = (a ^ b) & __tf_signbit;
249 
250   if ( aAbs > __tf_infrep ) return __tf_from(a | __tf_qnanbit);      // NaN
251   if ( bAbs > __tf_infrep ) return __tf_from(b | __tf_qnanbit);
252   if ( aAbs == __tf_infrep || bAbs == __tf_infrep ) {
253     if ( aAbs == 0 || bAbs == 0 ) return __tf_from(__tf_qnan);      // inf * 0
254     return __tf_from(sign | __tf_infrep);
255   }
256   if ( aAbs == 0 || bAbs == 0 ) return __tf_from(sign);      // signed zero
257 
258   int aExp = static_cast<int>(aAbs >> __tf_mantbits);
259   int bExp = static_cast<int>(bAbs >> __tf_mantbits);
260   __tf_rep aSig = aAbs & __tf_mantmask;
261   __tf_rep bSig = bAbs & __tf_mantmask;
262   if ( aExp == 0 )
263     aExp = __tf_normalize(aSig);
264   else
265     aSig |= __tf_implicit;
266   if ( bExp == 0 )
267     bExp = __tf_normalize(bSig);
268   else
269     bSig |= __tf_implicit;
270 
271   __tf_rep hi, lo;
272   __tf_wide_mul(aSig, bSig, hi, lo);      // 226-bit product in (hi:lo)
273 
274   int exp = aExp + bExp - __tf_bias;
275   int shift;                                          // bits below the kept 113-bit significand
276   if ( hi & (static_cast<__tf_rep>(1) << 97) ) {      // leading bit at 225
277     shift = 113;
278     exp += 1;
279   } else {      // leading bit at 224
280     shift = 112;
281   }
282 
283   const __tf_rep sig = (lo >> shift) | (hi << (128 - shift));      // P >> shift, a 113-bit significand
284   const unsigned guard = static_cast<unsigned>((lo >> (shift - 1)) & 1);
285   const bool sticky = (lo & ((static_cast<__tf_rep>(1) << (shift - 1)) - 1)) != 0;
286 
287   return __tf_round_pack(sign, exp, (sig << 3) | (__tf_rep(guard) << 2) | __tf_rep(sticky));
288 }

290 extern "C" __mc_libgcc_sym __tf_t
291 __divtf3(__tf_t fa, __tf_t fb) noexcept
292 {
293   const __tf_rep a = __tf_bits(fa), b = __tf_bits(fb);
294   const __tf_rep aAbs = a & __tf_absmask, bAbs = b & __tf_absmask;
295   const __tf_rep sign = (a ^ b) & __tf_signbit;
296 
297   if ( aAbs > __tf_infrep ) return __tf_from(a | __tf_qnanbit);      // NaN propagates
298   if ( bAbs > __tf_infrep ) return __tf_from(b | __tf_qnanbit);
299   if ( aAbs == __tf_infrep ) {
300     if ( bAbs == __tf_infrep ) return __tf_from(__tf_qnan);      // inf / inf
301     return __tf_from(sign | __tf_infrep);
302   }
303   if ( bAbs == __tf_infrep ) return __tf_from(sign);      // finite / inf -> signed zero
304   if ( bAbs == 0 ) {
305     if ( aAbs == 0 ) return __tf_from(__tf_qnan);      // 0 / 0
306     return __tf_from(sign | __tf_infrep);              // x / 0 -> signed inf
307   }
308   if ( aAbs == 0 ) return __tf_from(sign);      // 0 / x -> signed zero
309 
310   int aExp = static_cast<int>(aAbs >> __tf_mantbits);
311   int bExp = static_cast<int>(bAbs >> __tf_mantbits);
312   __tf_rep aSig = aAbs & __tf_mantmask;
313   __tf_rep bSig = bAbs & __tf_mantmask;
314   if ( aExp == 0 )
315     aExp = __tf_normalize(aSig);
316   else
317     aSig |= __tf_implicit;
318   if ( bExp == 0 )
319     bExp = __tf_normalize(bSig);
320   else
321     bSig |= __tf_implicit;
322 
323   int exp = aExp - bExp + __tf_bias;
324   if ( aSig < bSig ) {      // keep the quotient in [1, 2)
325     aSig <<= 1;
326     --exp;
327   }
328 
329   // restoring long division: implicit leading 1, then 112 fraction bits + one guard bit
330   __tf_rep rem = aSig - bSig;
331   __tf_rep q = 1;
332   for ( int i = 0; i < __tf_mantbits + 1; ++i ) {
333     rem <<= 1;
334     q <<= 1;
335     if ( rem >= bSig ) {
336       rem -= bSig;
337       q |= 1;
338     }
339   }
340   const unsigned guard = static_cast<unsigned>(q & 1);
341   const bool sticky = rem != 0;
342   const __tf_rep sig = q >> 1;      // 113-bit significand
343 
344   return __tf_round_pack(sign, exp, (sig << 3) | (__tf_rep(guard) << 2) | __tf_rep(sticky));
345 }

356 extern "C" __mc_libgcc_sym __tf_t
357 __floatditf(long long i) noexcept      // signed 64-bit int -> binary128 (exact)
358 {
359   if ( i == 0 ) return __tf_from(0);
360   const bool neg = i < 0;
361   const unsigned long long mag = neg ? (0ull - static_cast<unsigned long long>(i)) : static_cast<unsigned long long>(i);
362   const int msb = 63 - __builtin_clzll(mag);
363   __tf_rep sig = static_cast<__tf_rep>(mag) << (__tf_mantbits - msb);      // leading 1 -> bit 112; exact (64 <= 113)
364   const __tf_rep exp = static_cast<__tf_rep>(msb + __tf_bias);
365   return __tf_from((neg ? __tf_signbit : 0) | (exp << __tf_mantbits) | (sig & __tf_mantmask));
366 }

369 __floatunditf(unsigned long long u) noexcept      // unsigned 64-bit int -> binary128 (exact)
370 {

374   const __tf_rep exp = static_cast<__tf_rep>(msb + __tf_bias);
375   return __tf_from((exp << __tf_mantbits) | (sig & __tf_mantmask));

394 extern "C" __mc_libgcc_sym long long
395 __fixtfdi(__tf_t fa) noexcept      // binary128 -> signed 64-bit int (truncate toward zero)
396 {
397   const __tf_rep a = __tf_bits(fa);
398   const bool neg = (a >> 127) != 0;
399   const unsigned exp = static_cast<unsigned>((a >> __tf_mantbits) & __tf_einf);
400   const __tf_rep mant = a & __tf_mantmask;
401   constexpr long long imin = -9223372036854775807LL - 1;
402   constexpr long long imax = 9223372036854775807LL;
403   if ( exp < static_cast<unsigned>(__tf_bias) ) return 0;      // |x| < 1 (incl. zero/subnormal)
404   if ( exp == __tf_einf ) return neg ? imin : imax;            // inf/NaN saturate
405   const int unbiased = static_cast<int>(exp) - __tf_bias;      // >= 0
406   if ( unbiased >= 63 ) {
407     if ( neg && unbiased == 63 && mant == 0 ) return imin;      // exactly -2^63
408     return neg ? imin : imax;                                   // out of range -> saturate
409   }
410   const __tf_rep sig = __tf_implicit | mant;      // 113-bit
411   const unsigned long long m = static_cast<unsigned long long>(sig >> (__tf_mantbits - unbiased));
412   return neg ? -static_cast<long long>(m) : static_cast<long long>(m);
413 }

415 extern "C" __mc_libgcc_sym unsigned long long
416 __fixunstfdi(__tf_t fa) noexcept      // binary128 -> unsigned 64-bit int (truncate toward zero)
417 {
418   const __tf_rep a = __tf_bits(fa);
419   const bool neg = (a >> 127) != 0;
420   const unsigned exp = static_cast<unsigned>((a >> __tf_mantbits) & __tf_einf);
421   if ( neg ) return 0;                                         // negatives (incl. -inf/-NaN) saturate low
422   if ( exp == __tf_einf ) return ~0ull;                        // +inf/+NaN saturate high
423   if ( exp < static_cast<unsigned>(__tf_bias) ) return 0;      // |x| < 1
424   const int unbiased = static_cast<int>(exp) - __tf_bias;
425   if ( unbiased >= 64 ) return ~0ull;
426   const __tf_rep sig = __tf_implicit | (a & __tf_mantmask);
427   return static_cast<unsigned long long>(sig >> (__tf_mantbits - unbiased));
428 }

449 extern "C" __mc_libgcc_sym __tf_t
450 __extenddftf2(double x) noexcept
451 {
452   const unsigned long long db = __builtin_bit_cast(unsigned long long, x);
453   const __tf_rep s = static_cast<__tf_rep>(db >> 63) << 127;
454   const int de = static_cast<int>((db >> 52) & 0x7ff);
455   const unsigned long long dm = db & 0xfffffffffffffull;
456   if ( de == 0x7ff )      // inf/nan: shift the payload up, quietness rides along in the top bit
457     return __tf_from(s | __tf_infrep | (static_cast<__tf_rep>(dm) << 60));
458   if ( de == 0 ) {
459     if ( dm == 0 ) return __tf_from(s);      // +-0
460     // subnormal double (value = dm * 2^-1074)
461     const int p = 63 - __builtin_clzll(dm);
462     const __tf_rep m = (static_cast<__tf_rep>(dm) << (__tf_mantbits - p)) & __tf_mantmask;
463     return __tf_from(s | (static_cast<__tf_rep>((p - 1074) + __tf_bias) << __tf_mantbits) | m);
464   }
465   return __tf_from(s | (static_cast<__tf_rep>(de - 1023 + __tf_bias) << __tf_mantbits) | (static_cast<__tf_rep>(dm) << 60));
466 }

468 extern "C" __mc_libgcc_sym __tf_t
469 __extendsftf2(float x) noexcept
470 {
471   const unsigned int fb = __builtin_bit_cast(unsigned int, x);
472   const __tf_rep s = static_cast<__tf_rep>(fb >> 31) << 127;
473   const int fe = static_cast<int>((fb >> 23) & 0xff);
474   const unsigned int fm = fb & 0x7fffffu;
475   if ( fe == 0xff ) return __tf_from(s | __tf_infrep | (static_cast<__tf_rep>(fm) << 89));
476   if ( fe == 0 ) {
477     if ( fm == 0 ) return __tf_from(s);
478     // subnormal float (value = fm * 2^-149)
479     const int p = 31 - __builtin_clz(fm);
480     const __tf_rep m = (static_cast<__tf_rep>(fm) << (__tf_mantbits - p)) & __tf_mantmask;
481     return __tf_from(s | (static_cast<__tf_rep>((p - 149) + __tf_bias) << __tf_mantbits) | m);
482   }
483   return __tf_from(s | (static_cast<__tf_rep>(fe - 127 + __tf_bias) << __tf_mantbits) | (static_cast<__tf_rep>(fm) << 89));
484 }
