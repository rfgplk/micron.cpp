19 inline constexpr real
20 exp(real x) noexcept
21 {
22   if ( isnan(x) ) return x;
23   if ( x > 11357.L ) return inf;
24   if ( x < -11434.L ) return 0;
25   const int n = int(nearest(x / ln2));
26   const real r = fma(-real(n), ln2_hi, x) - real(n) * ln2_lo;
27   real sum = 1, term = 1;
28   for ( int i = 1; i <= 40; ++i ) {
29     term *= r / real(i);
30     sum += term;
31   }
32   return scalbn(sum, n);
33 }

51 inline constexpr real
52 expm1(real x) noexcept
53 {
54   if ( abs(x) > 0.5L ) return exp(x) - 1;

64 inline constexpr real
65 log(real x) noexcept
66 {
67   if ( isnan(x) || x == inf ) return x;
68   if ( x == 0 ) return -inf;
69   if ( x < 0 ) return nan<real>();
70   int e;
71   real m = frexp(x, &e);
72   if ( m < 0.707106781186547524400844362104849039284835937688474L ) {
73     m *= 2;
74     --e;
75   }
76   const real z = (m - 1) / (m + 1), z2 = z * z;
77   real sum = z, power = z;
78   for ( int i = 3; i <= 101; i += 2 ) {
79     power *= z2;
80     sum += power / real(i);
81   }
82   return fma(real(e), ln2_hi, 2 * sum + real(e) * ln2_lo);
83 }

89 inline constexpr __ddreal
90 __ddsum(real a, real b) noexcept
91 {
92   const real s = a + b, bp = s - a;
93   return { s, (a - (s - bp)) + (b - bp) };

124 inline constexpr __ddreal
125 __ddlog(real x) noexcept
126 {
127   int e;
128   real m = frexp(x, &e);
129   if ( m < 0.707106781186547524400844362104849039284835937688474L ) {
130     m *= 2;
131     --e;
132   }
133   const auto z = __dddiv({ m - 1, 0 }, __ddsum(m, 1)), z2 = __ddmul(z, z);
134   auto sum = z, power = z;
135   for ( int i = 3; i <= 151; i += 2 ) {
136     power = __ddmul(power, z2);
137     sum = __ddadd(sum, __dddiv(power, real(i)));
138   }
139   return __ddadd(
140       __ddmul(
141           { real(e), 0 },
142           { ln2,
143             -7.008139474549585163412662008771625673777725287234181290190813091556509261166929314266314479764241869442967329248365E-36L }),
144       __ddmul({ 2, 0 }, sum));

147 inline constexpr real
148 __ddexp(__ddreal x) noexcept
149 {
150   if ( isnan(x.hi) ) return x.hi;
151   if ( x.hi > 11357.L ) return inf;
152   if ( x.hi < -11434.L ) return 0;
153   const int n = int(nearest(x.hi / ln2));
154   const real r = (fma(-real(n), ln2_hi, x.hi) - real(n) * ln2_lo) + x.lo;
155   return scalbn(exp(r), n);

173 inline constexpr real
174 log1p(real x) noexcept
175 {
176   if ( x == 0 ) return x;
177   if ( abs(x) < 0.5L ) {
178     const real z = x / (2 + x), z2 = z * z;
179     real sum = z, power = z;
180     for ( int i = 3; i <= 151; i += 2 ) {
181       power *= z2;
182       sum += power / real(i);
183     }
184     return 2 * sum;
185   }
186   const real u = 1 + x;
187   if ( u <= 0 || isnan(u) || isinf(u) ) return log(u);
188   return log(u) + (x - (u - 1)) / u;
189 }

191 inline constexpr real
192 cbrt(real x) noexcept
193 {
194   if ( x == 0 || isinf(x) || isnan(x) ) return x;
195   int e;
196   const real m = frexp(abs(x), &e);
197   int n = e / 3, r = e % 3;
198   if ( r < 0 ) {
199     r += 3;
200     --n;
201   }
202   const real a = scalbn(m, r);
203   real root = 1;
204   for ( int i = 0; i < 12; ++i ) root = (2 * root + a / (root * root)) / 3;
205   return copysign(scalbn(root, n), x);
206 }

208 inline constexpr real
209 hypot(real x, real y) noexcept
210 {
211   x = abs(x);
212   y = abs(y);
213   if ( isinf(x) || isinf(y) ) return inf;
214   if ( isnan(x) || isnan(y) ) return nan<real>();
215   if ( x < y ) {
216     const real t = x;
217     x = y;
218     y = t;
219   }
220   if ( y == 0 ) return x;
221   int e;
222   const real sx = frexp(x, &e), sy = scalbn(y, -e);
223   return scalbn(sqrt(fma(sx, sx, sy * sy)), e);
224 }

233 inline constexpr real
234 pow(real x, real y) noexcept
235 {
236   if ( y == 0 || x == 1 ) return 1;
237   if ( isnan(x) || isnan(y) ) return nan<real>();
238   const real ax = abs(x);
239   const bool integral = trunc(y) == y, odd = integral && odd_integer(y);
240   if ( isinf(y) ) return ax == 1 ? 1 : ((ax > 1) == (y > 0) ? inf : 0);
241   if ( x == 0 ) return copysign(y < 0 ? inf : 0.L, odd ? x : 1.L);
242   if ( isinf(x) ) return copysign(y < 0 ? 0.L : inf, odd ? x : 1.L);
243   if ( x < 0 && !integral ) return nan<real>();
244   real result;
245   // Repeated squaring amplifies rounded products in proportion to the power.
246   // Keep the cheap path for small powers; retain compensation for larger ones.
247   if ( integral && abs(y) <= 32 ) {
248     i64 n = i64(y);
249     real base = n < 0 ? 1 / ax : ax;
250     if ( n < 0 ) n = -n;
251     result = 1;
252     while ( n ) {
253       if ( n & 1 ) result *= base;
254       n >>= 1;
255       if ( n ) base *= base;
256     }
257   } else {
258     const auto logarithm = __ddlog(ax);
259     const real estimate = y * logarithm.hi;
260     result = estimate > 11357.L ? inf : estimate < -11434.L ? 0.L : __ddexp(__ddmul({ y, 0 }, logarithm));
261   }
262   return x < 0 && odd ? -result : result;
263 }

265 // Full-exponent Payne-Hanek reduction. The 528-limb constant is integer data;
266 // no binary64 conversion limits the input exponent or significand.
267 inline constexpr int
268 reduce(real x, real &r) noexcept
269 {
270   const auto p = decompose(abs(x));
271   u32 product[532]{};
272   for ( unsigned j = 0; j < 4; ++j ) {
273     const u32 word = u32(p.significand >> (32 * j));
274     u64 carry = 0;
275     for ( unsigned i = 0; i < 528; ++i ) {
276       const u64 v = u64(__two_over_pi[i]) * word + product[i + j] + carry;
277       product[i + j] = u32(v);
278       carry = v >> 32;
279     }
280     product[528 + j] = u32(carry);
281   }
282   const int point = int(__two_over_pi_bits) - p.exponent;
283   auto bit = [&](int at) constexpr -> unsigned { return at < 0 || at >= 532 * 32 ? 0u : (product[at / 32] >> (at % 32)) & 1u; };
284   unsigned quadrant = bit(point) | (bit(point + 1) << 1);
285   const bool up = bit(point - 1) != 0;
286   u64 fraction[4]{};
287   for ( int i = 0; i < 256; ++i )
288     if ( bit(point - 256 + i) ) fraction[i / 64] |= u64(1) << (i % 64);
289   if ( up ) {
290     ++quadrant;
291     // Values below the retained 256 bits are nonzero for finite binary inputs;
292     // floor(1-f) at this precision is the one's complement of floor(f).
293     for ( auto &v : fraction ) v = ~v;
294   }
295   int leading = -1;
296   for ( int i = 3; i >= 0; --i )
297     if ( fraction[i] ) {
298       leading = i * 64 + 63 - __builtin_clzll(fraction[i]);
299       break;
300     }
301   if ( leading < 0 ) {
302     r = 0;
303     return int(quadrant & 3);
304   }
305   const int shift = leading - 115;
306   wide extended = 0;
307   bool sticky = true;
308   for ( int i = 0; i <= leading; ++i )
309     if ( (fraction[i / 64] >> (i % 64)) & 1 ) {
310       if ( i >= shift ) extended |= wide(1) << (i - shift);
311     }
312   if ( sticky ) extended |= 1;
313   r = pack<real>(up ? format<real>::sign : 0, leading - 256 + 16383, extended) * half_pi;
314   if ( bits(x) & format<real>::sign ) {
315     r = -r;
316     quadrant = 0u - quadrant;
317   }
318   return int(quadrant & 3);
319 }

321 inline constexpr void
322 sincos(real x, real &s, real &c) noexcept
323 {
324   if ( isnan(x) || isinf(x) ) {
325     s = c = nan<real>();
326     return;
327   }
328   if ( x == 0 ) {
329     s = x;
330     c = 1;
331     return;
332   }
333   real r = x;
334   const int quadrant = abs(x) <= pi / 4 ? 0 : reduce(x, r);
335   const real r2 = -r * r;
336   real st = r, ct = 1;
337   s = st;
338   c = ct;
339   for ( int i = 1; i <= 24; ++i ) {
340     st *= r2 / real((2 * i) * (2 * i + 1));
341     ct *= r2 / real((2 * i - 1) * (2 * i));
342     s += st;
343     c += ct;
344   }
345   const real saved = s;
346   switch ( quadrant ) {
347   case 1:
348     s = c;
349     c = -saved;
350     break;
351   case 2:
352     s = -s;
353     c = -c;
354     break;
355   case 3:
356     s = -c;
357     c = saved;
358     break;
359   }
360 }

386 inline constexpr real
387 atan(real x) noexcept
388 {
389   if ( isnan(x) ) return x;
390   if ( isinf(x) ) return copysign(half_pi, x);
391   if ( x == 0 ) return x;
392   real a = abs(x);
393   const bool reciprocal = a > 1;
394   if ( reciprocal ) a = 1 / a;
395   const bool translate = a > 0.414213562373095048801688724209698078569671875376948L;
396   if ( translate ) a = (a - 1) / (a + 1);
397   const real a2 = -a * a;
398   real term = a, result = a;
399   for ( int i = 3; i <= 191; i += 2 ) {
400     term *= a2;
401     result += term / real(i);
402   }
403   if ( translate ) result += pi / 4;
404   if ( reciprocal ) result = half_pi - result;
405   return copysign(result, x);
406 }

408 inline constexpr real
409 atan2(real y, real x) noexcept
410 {
411   if ( isnan(x) || isnan(y) ) return nan<real>();
412   const bool negative_x = (bits(x) & format<real>::sign) != 0;
413   if ( y == 0 ) return negative_x ? copysign(pi, y) : y;
414   if ( x == 0 ) return copysign(half_pi, y);
415   if ( isinf(x) && isinf(y) ) return copysign(negative_x ? 3 * pi / 4 : pi / 4, y);
416   real r = atan(abs(y / x));
417   if ( negative_x ) r = pi - r;
418   return copysign(r, y);
419 }

494 inline constexpr real
495 fmod(real x, real y) noexcept
496 {
497   if ( isnan(x) || isnan(y) || isinf(x) || y == 0 ) return nan<real>();
498   if ( x == 0 || isinf(y) ) return x;
499   const real sign = x;
500   x = abs(x);
501   y = abs(y);
502   if ( x < y ) return sign;
503   auto a = decompose(x), b = decompose(y);
504   const int am = msb(a.significand), bm = msb(b.significand);
505   a.exponent += am;
506   b.exponent += bm;
507   a.significand <<= 112 - am;
508   b.significand <<= 112 - bm;
509   for ( int e = a.exponent; e >= b.exponent; --e ) {
510     if ( a.significand >= b.significand ) a.significand -= b.significand;
511     if ( e != b.exponent ) a.significand <<= 1;
512   }
513   if ( !a.significand ) return copysign(0.L, sign);
514   const int leading = msb(a.significand);
515   return pack<real>(bits(sign) & format<real>::sign, b.exponent - 112 + leading + 16383, a.significand << (115 - leading));
516 }

518 inline constexpr real
519 remainder(real x, real y) noexcept
520 {
521   if ( isnan(x) || isnan(y) || isinf(x) || y == 0 ) return nan<real>();
522   if ( isinf(y) ) return x;
523   real r = fmod(x, y), ay = abs(y), ar = abs(r);
524   // Above half the finite range the truncating quotient can only be zero or one.
525   const bool odd = ay > 0x1.ffffffffffffffffffffffffffffp16382L ? abs(x) >= ay : abs(fmod(x, 2 * ay)) >= ay;
526   if ( ar > ay - ar || (ar == ay - ar && odd) ) r -= copysign(ay, x);
527   return r;
528 }
