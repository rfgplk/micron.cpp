19 // A reservation is permanent. Heap regions, worker stacks and TLS all use this
20 // cursor; releasing an allocation never returns a linear-memory reservation.
21 inline constinit u64 __linear_cursor = 0;
22 inline constinit u32 __linear_growth_epoch = 0;
23 
24 struct __mutex {
25   u32 word{ 0 };
26 
27   void
28   lock() noexcept
29   {
30 #if defined(__micron_wasm_threads)
31     for ( ;; ) {
32       u32 expected = 0;
33       if ( __atomic_compare_exchange_n(&word, &expected, 1u, false, __ATOMIC_ACQUIRE, __ATOMIC_RELAXED) ) return;
34       while ( __atomic_load_n(&word, __ATOMIC_RELAXED) != 0 ) {
35       }
36     }
37 #endif
38   }
39 
40   void
41   unlock() noexcept
42   {
43 #if defined(__micron_wasm_threads)
44     __atomic_store_n(&word, 0u, __ATOMIC_RELEASE);
45 #endif
46   }
47 };

49 struct __lock_guard {
50   __mutex &mutex;
51 
52   explicit __lock_guard(__mutex &m) noexcept : mutex(m) { mutex.lock(); }
53 
54   ~__lock_guard() { mutex.unlock(); }
55 
56   __lock_guard(const __lock_guard &) = delete;
57   __lock_guard &operator=(const __lock_guard &) = delete;
58 };

62 [[nodiscard]] inline u64
63 linear_memory_bytes() noexcept
64 {
65   return static_cast<u64>(__builtin_wasm_memory_size(0)) * __micron_wasm_page_size;
66 }
67 
68 inline void
69 linear_memory_init() noexcept
70 {
71   __lock_guard lock{ __linear_mutex };
72   if ( __linear_cursor == 0 ) __linear_cursor = reinterpret_cast<uintptr_t>(&__heap_base);
73 }

75 [[nodiscard]] inline void *
76 linear_reserve(usize bytes, usize alignment = 16) noexcept
77 {
78   if ( bytes == 0 || alignment == 0 || (alignment & (alignment - 1)) != 0 ) return nullptr;
79   __lock_guard lock{ __linear_mutex };
80   const u64 cursor = __linear_cursor ? __linear_cursor : reinterpret_cast<uintptr_t>(&__heap_base);
81   const u64 begin = (cursor + alignment - 1) & ~(static_cast<u64>(alignment) - 1);
82   const u64 end = begin + bytes;
83   // Keep the one-past pointer representable in the wasm32 address space.
84   if ( begin == 0 || end > 0xffffffffull ) return nullptr;
85   const u64 have = linear_memory_bytes();
86   if ( end > have ) {
87     const usize pages = static_cast<usize>((end - have + __micron_wasm_page_size - 1) / __micron_wasm_page_size);
88     if ( __builtin_wasm_memory_grow(0, pages) == static_cast<usize>(-1) ) return nullptr;
89 #if defined(__micron_wasm_threads)
90     __atomic_add_fetch(&__linear_growth_epoch, 1u, __ATOMIC_RELEASE);
91 #else
92     ++__linear_growth_epoch;
93 #endif
94   }
95   __linear_cursor = end;
96   return reinterpret_cast<void *>(static_cast<uintptr_t>(begin));
97 }

99 [[nodiscard]] inline u32
100 memory_growth_epoch() noexcept
101 {
102 #if defined(__micron_wasm_threads)
103   return __atomic_load_n(&__linear_growth_epoch, __ATOMIC_ACQUIRE);
104 #else
105   return __linear_growth_epoch;
106 #endif
107 }
