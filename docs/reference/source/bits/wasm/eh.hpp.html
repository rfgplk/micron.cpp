25 alignas(16) inline byte emergency[16][2048]{};

29 inline void
30 link(__cxxabiv1::__cxa_exception *h) noexcept
31 {
32   // This VM unwinder owns the private words; no native stack walk uses them.
33   h->unwindHeader.private_1 = 0;
34   h->unwindHeader.private_2 = reinterpret_cast<usize>(allocations);
35   if ( allocations ) allocations->unwindHeader.private_1 = reinterpret_cast<usize>(h);
36   allocations = h;
37 }
38 
39 inline void
40 unlink(__cxxabiv1::__cxa_exception *h) noexcept
41 {
42   auto *previous = reinterpret_cast<__cxxabiv1::__cxa_exception *>(h->unwindHeader.private_1);
43   auto *next = reinterpret_cast<__cxxabiv1::__cxa_exception *>(h->unwindHeader.private_2);
44   if ( previous )
45     previous->unwindHeader.private_2 = reinterpret_cast<usize>(next);
46   else
47     allocations = next;
48   if ( next ) next->unwindHeader.private_1 = reinterpret_cast<usize>(previous);
49 }

59 inline bool
60 match(const std::type_info *caught, __cxxabiv1::__cxa_exception *h, void *&adjusted) noexcept
61 {
62   if ( !caught ) return true;
63   if ( std::__micron_ti_strcmp(h->exceptionType->name(), "Dn") == 0 && caught->name()[0] == 'M' ) {
64     const auto *member = static_cast<const __cxxabiv1::__pbase_type_info *>(caught);
65     auto *words = static_cast<usize *>(adjusted);
66     words[0] = member->__pointee->__is_function_p() ? 0 : ~usize(0);
67     words[1] = 0;
68     return true;
69   }
70   if ( h->exceptionType->__is_pointer_p() ) adjusted = *static_cast<void **>(adjusted);
71   return caught->__do_catch(h->exceptionType, &adjusted, 1);
72 }

78 __attribute__((weak, used)) void *
79 __cxa_allocate_exception(usize size) noexcept
80 {
81   using namespace micron::wasm::eh;
82   constexpr usize overhead = sizeof(__cxxabiv1::__cxa_exception);
83   if ( size > ~usize(0) - overhead ) terminate();
84   const usize total = overhead + (size < 8 ? 8 : size);
85   void *p = micron::wasmalloc::try_alloc(total);
86   if ( !p && total <= sizeof(emergency[0]) ) {
87     for ( unsigned i = 0; i < 16; ++i ) {
88       unsigned expected = 0;
89       if ( __atomic_compare_exchange_n(&emergency_used[i], &expected, 1u, false, __ATOMIC_ACQUIRE, __ATOMIC_RELAXED) ) {
90         p = emergency[i];
91         break;
92       }
93     }
94   }
95   if ( !p ) terminate();
96   auto *h = static_cast<__cxxabiv1::__cxa_exception *>(p);
97   *h = {};
98   link(h);
99   return h + 1;
100 }

102 __attribute__((weak, used)) void
103 __cxa_free_exception(void *obj) noexcept
104 {
105   using namespace micron::wasm::eh;
106   auto *p = static_cast<__cxxabiv1::__cxa_exception *>(obj) - 1;
107   unlink(p);
108   const usize a = reinterpret_cast<usize>(p), b = reinterpret_cast<usize>(emergency);
109   if ( a >= b && a < b + sizeof(emergency) )
110     __atomic_store_n(&emergency_used[(a - b) / sizeof(emergency[0])], 0u, __ATOMIC_RELEASE);
111   else
112     micron::wasmalloc::dealloc(p);
113 }

142 inline void
143 __micron_wasm_exception_cleanup(_Unwind_Reason_Code, _Unwind_Exception *ue)
144 {
145   auto *h = __cxxabiv1::__get_exception_header_from_ue(ue);
146   void *obj = h + 1;
147   const auto dtor = h->exceptionDestructor;
148   h->exceptionDestructor = nullptr;
149   // Wasm's C++ ABI returns this from complete/base destructors.
150   if ( dtor ) (void)reinterpret_cast<void *(*)(void *)>(dtor)(obj);
151   __cxa_free_exception(obj);
152 }

154 __attribute__((weak, used)) void
155 __micron_wasm_finish_exceptions() noexcept
156 {
157   auto *g = __cxa_get_globals();
158   g->caughtExceptions = nullptr;
159   g->uncaughtExceptions = 0;
160   // A checkpoint may abandon catches, propagation, or an unfinished throw
161   // construction. Only fully constructed, not-yet-destroyed objects have a dtor.
162   while ( auto *h = micron::wasm::eh::allocations ) __micron_wasm_exception_cleanup(_URC_NO_REASON, &h->unwindHeader);
163   __wasm_lpad_context = {};
164 }

166 __attribute__((weak, used, noreturn)) void
167 __cxa_throw(void *obj, void *type, void (*dtor)(void *))
168 {
169   auto *h = __cxxabiv1::__get_exception_header_from_object(obj);
170   h->exceptionType = static_cast<std::type_info *>(type);
171   h->exceptionDestructor = dtor;
172   h->adjustedPtr = obj;
173   h->unwindHeader.exception_class = __cxxabiv1::__gxx_exception_class;
174   h->unwindHeader.exception_cleanup = __micron_wasm_exception_cleanup;
175   ++__cxa_get_globals()->uncaughtExceptions;
176   __builtin_wasm_throw(0, &h->unwindHeader);
177   __builtin_unreachable();
178 }

198 __attribute__((weak, used)) void *
199 __cxa_begin_catch(void *ue) noexcept
200 {
201   auto *h = __cxxabiv1::__get_exception_header_from_ue(static_cast<_Unwind_Exception *>(ue));
202   auto *g = __cxa_get_globals();
203   h->handlerCount = h->handlerCount < 0 ? -h->handlerCount + 1 : h->handlerCount + 1;
204   if ( g->caughtExceptions != h ) {
205     h->nextException = static_cast<__cxxabiv1::__cxa_exception *>(g->caughtExceptions);
206     g->caughtExceptions = h;
207   }
208   if ( g->uncaughtExceptions ) --g->uncaughtExceptions;
209   return h->adjustedPtr;
210 }

212 __attribute__((weak, used)) void
213 __cxa_end_catch()
214 {
215   auto *g = __cxa_get_globals();
216   auto *h = static_cast<__cxxabiv1::__cxa_exception *>(g->caughtExceptions);
217   if ( !h ) return;
218   if ( h->handlerCount < 0 ) {
219     if ( ++h->handlerCount == 0 ) g->caughtExceptions = h->nextException;
220   } else if ( --h->handlerCount == 0 ) {
221     g->caughtExceptions = h->nextException;
222     _Unwind_DeleteException(&h->unwindHeader);
223   }
224 }

226 __attribute__((weak, used, noreturn)) void
227 __cxa_rethrow()
228 {
229   auto *g = __cxa_get_globals();
230   auto *h = static_cast<__cxxabiv1::__cxa_exception *>(g->caughtExceptions);
231   if ( !h ) micron::wasm::eh::terminate();
232   ++g->uncaughtExceptions;
233   h->handlerCount = -h->handlerCount;
234   __builtin_wasm_throw(0, &h->unwindHeader);
235   __builtin_unreachable();
236 }

257 __attribute__((weak, used)) int
258 _Unwind_CallPersonality(void *exception)
259 {
260   using namespace micron::eh;
261   auto &ctx = __wasm_lpad_context;
262 #if defined(MICRON_WASM_EH_TRACE)
263   micron::wasm::write_diag("personality\n");
264 #endif
265   ctx.selector = 0;
266   if ( !ctx.lsda ) return _URC_CONTINUE_UNWIND;
267   const byte *p = ctx.lsda;
268   const u8 landing_encoding = *p++;
269   if ( landing_encoding != DW_EH_PE_omit ) (void)read_encoded(landing_encoding, p, {});
270   const u8 type_encoding = *p++;
271   const byte *type_table = nullptr;
272   if ( type_encoding != DW_EH_PE_omit ) {
273     const usize off = read_uleb128(p);
274     type_table = p + off;
275   }
276   const u8 call_encoding = *p++;
277   if ( call_encoding != DW_EH_PE_uleb128 ) micron::wasm::eh::terminate();
278   const usize call_length = read_uleb128(p);
279   const byte *action_table = p + call_length;
280   usize action = 0;
281   bool found = false;
282   usize ordinal = 0;
283   while ( p < action_table ) {
284     (void)read_uleb128(p);      // landing-pad number; the VM already selected it
285     const usize a = read_uleb128(p);
286     if ( ordinal++ == ctx.index ) {
287       action = a;
288       found = true;
289       break;
290     }
291   }
292   if ( !found || !action ) return _URC_CONTINUE_UNWIND;
293   auto *ue = static_cast<_Unwind_Exception *>(exception);
294   if ( ue->exception_class != __cxxabiv1::__gxx_exception_class ) micron::wasm::eh::terminate();
295   auto *h = __cxxabiv1::__get_exception_header_from_ue(ue);
296   p = action_table + action - 1;
297   for ( ;; ) {
298     const i64 filter = read_sleb128(p);
299     const byte *next_base = p;
300     const i64 next = read_sleb128(p);
301     if ( filter > 0 ) {
302       const usize width = encoded_size(type_encoding);
303       if ( !type_table || !width ) micron::wasm::eh::terminate();
304       const byte *entry = type_table - static_cast<usize>(filter) * width;
305       const auto *caught = reinterpret_cast<const std::type_info *>(read_encoded(type_encoding, entry, {}));
306 #if defined(MICRON_WASM_EH_TRACE)
307       micron::wasm::write_diag(caught ? caught->name() : "...");
308       micron::wasm::write_diag(" <- ");
309       micron::wasm::write_diag(h->exceptionType->name());
310       micron::wasm::write_diag("\n");
311 #endif
312       void *adjusted = h + 1;
313       if ( micron::wasm::eh::match(caught, h, adjusted) ) {
314         h->adjustedPtr = adjusted;
315         ctx.selector = static_cast<unsigned>(filter);
316         return _URC_HANDLER_FOUND;
317       }
318     } else if ( filter < 0 ) {
319       // Dynamic exception specifications were removed before Micron's C++23 floor.
320       micron::wasm::eh::terminate();
321     }
322     if ( !next ) break;
323     p = next_base + next;
324   }
325   return _URC_CONTINUE_UNWIND;
326 }
