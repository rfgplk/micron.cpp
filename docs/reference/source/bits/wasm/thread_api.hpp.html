18 struct thread_packet {
19   unsigned long long value = 0;

22 template<typename F, typename... A> struct invocation_packet: thread_packet {
23   F function;
24   tuple<A...> arguments;
25 
26   template<typename Fn, typename... Args>
27   invocation_packet(Fn &&f, Args &&...a) : function(micron::forward<Fn>(f)), arguments(micron::forward<Args>(a)...)
28   {
29   }
30 
31   static int
32   run(void *p)
33   {
34     auto &self = *static_cast<invocation_packet *>(p);
35     if constexpr ( is_void_v<decltype(micron::apply(self.function, self.arguments))> ) {
36       micron::apply(self.function, self.arguments);
37     } else {
38       auto value = micron::apply(self.function, self.arguments);
39       if constexpr ( is_trivially_copyable_v<decltype(value)> && sizeof(value) <= sizeof(self.value) )
40         __builtin_memcpy(&self.value, &value, sizeof(value));
41     }
42     return 0;
43   }
44 
45   static void
46   destroy(thread_packet *p)
47   {
48     delete static_cast<invocation_packet *>(p);
49   }
50 };

53 template<usize Stack_Size = thread_stack_size> class thread
54 {
55   wasm::thread_handle __handle{};
56   wasm::thread_packet *__packet = nullptr;
57   void (*__destroy)(wasm::thread_packet *) = nullptr;
58   unsigned long long __result = 0;
59   int __status = 0;
60 
61   template<typename Fn, typename... Args>
62   void
63   launch(Fn &&fn, Args &&...args)
64   {
65     static_assert(Stack_Size <= MICRON_WASM_STACK_SIZE, "thread stack exceeds MICRON_WASM_STACK_SIZE");
66     using packet = wasm::invocation_packet<decay_t<Fn>, decay_t<Args>...>;
67     auto *p = new packet(micron::forward<Fn>(fn), micron::forward<Args>(args)...);
68     const int r = wasm::spawn(packet::run, p, __handle);
69     if ( r ) {
70       delete p;
71       exc<except::thread_error>("micron: browser worker pool is unavailable or full");
72     }
73     __packet = p;
74     __destroy = packet::destroy;
75   }
76 
77 public:
78   using thread_type = thread_tag;
79   thread() = default;
80 
81   template<typename Fn, typename... Args>
82     requires(is_invocable_v<decay_t<Fn>, decay_t<Args> &...>)
83   explicit thread(Fn &&fn, Args &&...args)
84   {
85     launch(micron::forward<Fn>(fn), micron::forward<Args>(args)...);
86   }
87 
88   thread(const thread &) = delete;
89   thread &operator=(const thread &) = delete;
90 
91   thread(thread &&o) noexcept
92       : __handle(o.__handle), __packet(o.__packet), __destroy(o.__destroy), __result(o.__result), __status(o.__status)
93   {
94     o.__handle = {};
95     o.__packet = nullptr;
96   }
97 
98   thread &
99   operator=(thread &&o) noexcept
100   {
101     if ( this == &o ) return *this;
102     join();
103     __handle = o.__handle;
104     __packet = o.__packet;
105     __destroy = o.__destroy;
106     __result = o.__result;
107     __status = o.__status;
108     o.__handle = {};
109     o.__packet = nullptr;
110     return *this;
111   }
112 
113   ~thread()
114   {
115     if ( join() < 0 && __handle.slot ) wasm::panic();
116   }
117 
118   bool
119   alive() const noexcept
120   {
121     if ( !__handle.slot ) return false;
122     const auto state = __atomic_load_n(&wasm::__threads[__handle.slot].state, __ATOMIC_ACQUIRE);
123     return state == 1 || state == 2;
124   }
125 
126   bool
127   active() const noexcept
128   {
129     return alive();
130   }
131 
132   int
133   can_join() const noexcept
134   {
135     return __handle.slot ? (alive() ? 16 : 1) : 0;
136   }
137 
138   void
139   wait_for() const noexcept
140   {
141     if ( !__handle.slot ) return;
142     auto &slot = wasm::__threads[__handle.slot];
143     unsigned state;
144     while ( (state = __atomic_load_n(&slot.state, __ATOMIC_ACQUIRE)) == 1 || state == 2 )
145       if ( wasm::wait32(&slot.state, state) == -95 ) wasm::panic();
146   }
147 
148   int
149   join() noexcept
150   {
151     if ( !__handle.slot ) return __status;
152     __status = wasm::join(__handle);
153     if ( __handle.slot ) return __status;
154     if ( __packet ) {
155       __result = __packet->value;
156       __destroy(__packet);
157       __packet = nullptr;
158     }
159     return __status;
160   }
161 
162   int
163   dismiss() noexcept
164   {
165     return join();
166   }
167 
168   int
169   try_join() noexcept
170   {
171     return alive() ? 16 : join();
172   }
173 
174   unsigned
175   thread_id() const noexcept
176   {
177     return __handle.slot ? __handle.slot + 1 : 0;
178   }
179 
180   wasm::thread_handle
181   native_handle() const noexcept
182   {
183     return __handle;
184   }
185 
186   int
187   cancel() noexcept
188   {
189     if ( !__handle.slot ) return -22;
190     if ( __handle.slot + 1 == wasm::thread_id() ) return -35;
191     const int r = wasm::request_cancel(__handle);
192     return r < 0 ? r : join();
193   }
194 
195   int
196   sleep() noexcept
197   {
198     return wasm::__set_park(__handle, true);
199   }
200 
201   int
202   sleep_second() noexcept
203   {
204     return sleep();
205   }
206 
207   int
208   awaken() noexcept
209   {
210     return wasm::__set_park(__handle, false);
211   }
212 
213   template<typename R>
214     requires(is_trivially_copyable_v<R> && sizeof(R) <= sizeof(void *))
215   R
216   result()
217   {
218     wait_for();
219     const auto value = __packet ? __packet->value : __result;
220     R r{};
221     __builtin_memcpy(&r, &value, sizeof(R));
222     return r;
223   }
224 
225   template<typename Fn, typename... Args>
226   thread &
227   operator[](Fn &&fn, Args &&...args)
228   {
229     join();
230     launch(micron::forward<Fn>(fn), micron::forward<Args>(args)...);
231     return *this;
232   }
233 
234   addr_t *
235   stack() noexcept
236   {
237     return __handle.slot ? reinterpret_cast<addr_t *>(wasm::__threads[__handle.slot].stack) : nullptr;
238   }
239 
240   constexpr usize
241   stack_size() const noexcept
242   {
243     return MICRON_WASM_STACK_SIZE;
244   }
245 
246   int terminate() = delete;
247   int kill() = delete;
248 };
