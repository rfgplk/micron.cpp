15 template<typename U, unsigned Op>
16 inline constexpr U
17 __wasm_shift(U a, U b, u64 count) noexcept
18 {
19   constexpr unsigned width = sizeof(U) * 8;
20   if constexpr ( Op >= 3 ) count &= width - 1;
21   if ( count == 0 ) return a;
22   if constexpr ( Op == 0 ) return count >= width ? U(0) : U(u64(a) << count);
23   if constexpr ( Op == 1 ) return count >= width ? U(0) : U(a >> count);
24   if constexpr ( Op == 2 ) {
25     const U fill = U(0) - U(a >> (width - 1));
26     return count >= width ? fill : U((a >> count) | (u64(fill) << (width - count)));
27   }
28   if constexpr ( Op == 3 ) return U((u64(a) << count) | (a >> (width - count)));
29   if constexpr ( Op == 4 ) return U((a >> count) | (u64(a) << (width - count)));
30   if constexpr ( Op == 5 ) return U((u64(a) << count) | (b >> (width - count)));
31   if constexpr ( Op == 6 ) return U((a >> count) | (u64(b) << (width - count)));
32 }

34 template<typename U, unsigned Op, bool Variable, __wasm_simd_class T>
35 inline T
36 __wasm_shift_apply(T a, T b, T counts, u64 count) noexcept
37 {
38   using B = typename T::bit_width;
39   auto av = __wasm_unpack<U>(B(a));
40   const auto bv = __wasm_unpack<U>(B(b)), cv = __wasm_unpack<U>(B(counts));
41   for ( usize i = 0; i < sizeof(B) / sizeof(U); ++i )
42     av.lane[i] = __wasm_shift<U, Op>(av.lane[i], bv.lane[i], Variable ? cv.lane[i] : count);
43   return __wasm_pack<B>(av);
44 }

46 template<typename U, __wasm_simd_class T, typename M>
47 inline T
48 __wasm_mask_merge(T src, M mask, T value) noexcept
49 {
50   using B = typename T::bit_width;
51   auto out = __wasm_unpack<U>(B(src));
52   const auto v = __wasm_unpack<U>(B(value));
53   for ( usize i = 0; i < sizeof(B) / sizeof(U); ++i )
54     if ( (u64(mask) >> i) & 1 ) out.lane[i] = v.lane[i];
55   return __wasm_pack<B>(out);
56 }

58 template<__wasm_simd_class T>
59 inline T
60 byte_shift_left(T &o, int count) noexcept
61 {
62   using B = typename T::bit_width;
63   auto v = __wasm_unpack<u8>(B(o));
64   __wasm_lanes<u8, sizeof(B)> out{};
65   if ( unsigned(count) < 16 )
66     for ( usize i = 0; i < sizeof(B); ++i )
67       if ( i % 16 >= unsigned(count) ) out.lane[i] = v.lane[i - count];
68   return __wasm_pack<B>(out);
69 }
70 
71 template<__wasm_simd_class T>
72 inline T
73 byte_shift_right(T &o, int count) noexcept
74 {
75   using B = typename T::bit_width;
76   auto v = __wasm_unpack<u8>(B(o));
77   __wasm_lanes<u8, sizeof(B)> out{};
78   if ( unsigned(count) < 16 )
79     for ( usize i = 0; i < sizeof(B); ++i )
80       if ( i % 16 + unsigned(count) < 16 ) out.lane[i] = v.lane[i + count];
81   return __wasm_pack<B>(out);
82 }
