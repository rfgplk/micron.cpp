120 template<typename F, is_simd_type B>
121 [[gnu::always_inline]] inline constexpr auto
122 __wasm_unpack(B value) noexcept
123 {
124   using lanes = __wasm_lanes<F, sizeof(B) / sizeof(F)>;
125   return __builtin_bit_cast(lanes, value);
126 }
127 
128 template<is_simd_type B, typename F, usize N>
129 [[gnu::always_inline]] inline constexpr B
130 __wasm_pack(__wasm_lanes<F, N> value) noexcept
131 {
132   static_assert(sizeof(value) == sizeof(B));
133   return __builtin_bit_cast(B, value);
134 }
135 
136 template<is_simd_type B, typename F>
137 [[gnu::always_inline]] inline constexpr B
138 __wasm_splat(F value) noexcept
139 {
140   __wasm_lanes<F, sizeof(B) / sizeof(F)> lanes{};
141   for ( auto &lane : lanes.lane ) lane = value;
142   return __wasm_pack<B>(lanes);
143 }
