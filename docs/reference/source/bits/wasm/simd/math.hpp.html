14 template<typename F, is_simd_type B, typename Fn>
15 inline B
16 __wasm_math_map(B value, Fn fn) noexcept
17 {
18   auto a = __wasm_unpack<F>(value);
19   for ( auto &x : a.lane ) x = fn(x);
20   return __wasm_pack<B>(a);
21 }
22 
23 template<typename F, is_simd_type B, typename Fn>
24 inline B
25 __wasm_math_zip(B a, B b, Fn fn) noexcept
26 {
27   auto av = __wasm_unpack<F>(a);
28   const auto bv = __wasm_unpack<F>(b);
29   for ( usize i = 0; i < sizeof(B) / sizeof(F); ++i ) av.lane[i] = fn(av.lane[i], bv.lane[i]);
30   return __wasm_pack<B>(av);
31 }

33 template<typename F, is_simd_type B, typename M>
34 inline B
35 __wasm_math_mask(B src, M mask, B value) noexcept
36 {
37   auto out = __wasm_unpack<F>(src);
38   const auto v = __wasm_unpack<F>(value);
39   for ( usize i = 0; i < sizeof(B) / sizeof(F); ++i )
40     if ( (u64(mask) >> i) & 1 ) out.lane[i] = v.lane[i];
41   return __wasm_pack<B>(out);
42 }

155 template<__wasm_real_packet B>
156 inline B
157 round(B a, int mode) noexcept
158 {
159   switch ( mode & 7 ) {
160   case 1:
161     return floor(a);
162   case 2:
163     return ceil(a);
164   case 3:
165     return trunc(a);
166   default:
167     return rint(a);
168   }
169 }
