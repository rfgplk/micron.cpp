40 inline int
41 wait32(unsigned *word, unsigned expected, long long timeout_ns = -1) noexcept
42 {
43   if ( !word || (reinterpret_cast<__UINTPTR_TYPE__>(word) & 3u) ) return -22;
44   if ( timeout_ns < -1 ) return -22;
45   if ( __atomic_load_n(word, __ATOMIC_ACQUIRE) != expected ) return -11;
46 #if defined(__micron_wasm_threads)
47   if ( !__can_wait ) return -95;
48   const int r = __builtin_wasm_memory_atomic_wait32(reinterpret_cast<int *>(word), static_cast<int>(expected), timeout_ns);
49   return r == 0 ? 0 : (r == 1 ? -11 : -110);
50 #else
51   (void)timeout_ns;
52   return -95;
53 #endif
54 }

56 inline int
57 wake32(unsigned *word, unsigned count = 0x7fffffffu) noexcept
58 {
59   if ( !word || (reinterpret_cast<__UINTPTR_TYPE__>(word) & 3u) ) return -22;
60 #if defined(__micron_wasm_threads)
61   return __builtin_wasm_memory_atomic_notify(reinterpret_cast<int *>(word), count);
62 #else
63   (void)count;
64   return 0;
65 #endif
66 }

68 struct mutex {
69   unsigned word = 0;
70 
71   void
72   lock() noexcept
73   {
74 #if defined(__micron_wasm_threads)
75     unsigned expected = 0;
76     if ( __atomic_compare_exchange_n(&word, &expected, 1u, false, __ATOMIC_ACQUIRE, __ATOMIC_RELAXED) ) return;
77     if ( !__can_wait ) {
78       write_diag("micron: blocking mutex requires a worker\n");
79       panic();
80     }
81     while ( __atomic_exchange_n(&word, 2u, __ATOMIC_ACQUIRE) ) (void)wait32(&word, 2u);
82 #else
83     if ( word ) {
84       write_diag("micron: recursive runtime mutex\n");
85       panic();
86     }
87     word = 1;
88 #endif
89   }
90 
91   void
92   unlock() noexcept
93   {
94 #if defined(__micron_wasm_threads)
95     if ( __atomic_exchange_n(&word, 0u, __ATOMIC_RELEASE) == 2u ) (void)wake32(&word, 1);
96 #else
97     word = 0;
98 #endif
99   }
100 };
