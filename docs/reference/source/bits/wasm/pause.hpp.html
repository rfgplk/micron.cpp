11 inline constexpr u64 __wasm_sleep_max_ns = 0x7fffffffffffffffull;
12 
13 template<u64 Scale>
14 [[nodiscard]] inline constexpr bool
15 __wasm_sleep_unsigned(umax_t duration, i64 &ns) noexcept
16 {
17   static_assert(Scale != 0);
18   if ( duration > __wasm_sleep_max_ns / Scale ) return false;
19   ns = static_cast<i64>(duration * Scale);
20   return true;
21 }
22 
23 [[nodiscard]] inline constexpr bool
24 __wasm_sleep_milliseconds(fduration_t milliseconds, i64 &ns) noexcept
25 {
26   if ( !__builtin_isfinite(milliseconds) ) return false;
27   if ( milliseconds <= 0 ) {
28     ns = 0;
29     return true;
30   }
31   if ( milliseconds > 0x1p63 / 1000000.0 ) return false;
32   const fduration_t scaled = milliseconds * 1000000.0;
33   if ( scaled >= 0x1p63 ) return false;
34   ns = static_cast<i64>(scaled);
35   return true;
36 }

38 inline void
39 __wasm_sleep_checked(i64 ns)
40 {
41   __micron_park_checkpoint();
42   const auto r = chrono::sleep_ns(ns);
43   __micron_park_checkpoint();
44   if ( r == -22 ) exc<except::thread_error>("micron: sleep duration is out of range");
45   if ( r < 0 ) exc<except::thread_error>("micron: synchronous sleep requires a shared-memory worker");
46 }

48 template<u64 Scale>
49 inline void
50 __wasm_sleep_units(umax_t duration)
51 {
52   i64 ns = 0;
53   if ( !__wasm_sleep_unsigned<Scale>(duration, ns) ) exc<except::thread_error>("micron: sleep duration is out of range");
54   __wasm_sleep_checked(ns);
55 }
