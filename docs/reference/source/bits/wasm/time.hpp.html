34 inline ssize_t
35 clock_gettime(clockid_t id, timespec_t &out) noexcept
36 {
37   unsigned which;
38   if ( id == clock_realtime || id == clock_realtime_coarse )
39     which = 0;
40   else if ( id == clock_monotonic || id == clock_monotonic_coarse )
41     which = 1;
42   else
43     return -95;
44   const i64 n = __micron_wasm_host_clock(which);
45   if ( n < 0 ) return static_cast<ssize_t>(n);
46   out = { n / 1000000000LL, n % 1000000000LL };
47   return 0;
48 }

56 inline ssize_t
57 clock_getres(clockid_t id, timespec_t &out) noexcept
58 {
59   timespec_t probe;
60   const auto r = clock_gettime(id, probe);
61   if ( r < 0 ) return r;
62   // The host reports its guaranteed resolution, not a fabricated nanosecond clock.
63   const i64 n = __micron_wasm_host_clock(id == clock_realtime || id == clock_realtime_coarse ? 2 : 3);
64   if ( n <= 0 ) return -95;
65   out = { n / 1000000000LL, n % 1000000000LL };
66   return 0;
67 }

77 inline ssize_t
78 clock_nanosleep(clockid_t id, int flags, const timespec_t &ts, timespec_t *rem = nullptr) noexcept
79 {
80   if ( ts.tv_sec < 0 || ts.tv_nsec < 0 || ts.tv_nsec >= 1000000000LL || (flags & ~timer_abstime) ) return -22;
81   if ( ts.tv_sec > (0x7fffffffffffffffLL - ts.tv_nsec) / 1000000000LL ) return -22;
82   i64 ns = ts.tv_sec * 1000000000LL + ts.tv_nsec;
83   timespec_t now;
84   const auto r = clock_gettime(id, now);
85   if ( r < 0 ) return r;
86   if ( flags == timer_abstime ) ns -= now.tv_sec * 1000000000LL + now.tv_nsec;
87   if ( rem ) *rem = {};
88   if ( ns <= 0 ) return 0;
89   const auto wait = wasm::wait32(&__wasm_sleep_word, 0, ns);
90   return wait == -110 ? 0 : wait;
91 }

105 inline time_t
106 time(time_t *out = nullptr) noexcept
107 {
108   timespec_t ts;
109   if ( clock_gettime(clock_realtime, ts) < 0 ) return -1;
110   if ( out ) *out = static_cast<time_t>(ts.tv_sec);
111   return static_cast<time_t>(ts.tv_sec);
112 }

114 inline ssize_t
115 gettimeofday(timeval_t *out, void * = nullptr) noexcept
116 {
117   if ( !out ) return -22;
118   timespec_t ts;
119   const auto r = clock_gettime(clock_realtime, ts);
120   if ( r < 0 ) return r;
121   *out = { ts.tv_sec, ts.tv_nsec / 1000 };
122   return 0;
123 }
