59 inline int
60 spawn(thread_entry entry, void *arg, thread_handle &handle) noexcept
61 {
62 #if defined(__micron_wasm_threads)
63   if ( !entry || !can_wait() ) return -95;
64   const unsigned runtime = __atomic_load_n(&__runtime_state, __ATOMIC_ACQUIRE);
65   if ( runtime != 1 && runtime != 2 ) return -16;
66   __threads_lock.lock();
67   unsigned slot = 1;
68   const unsigned locked_runtime = __atomic_load_n(&__runtime_state, __ATOMIC_ACQUIRE);
69   if ( locked_runtime != 1 && locked_runtime != 2 ) {
70     __threads_lock.unlock();
71     return -16;
72   }
73   while ( slot < MICRON_WASM_MAX_THREADS && __atomic_load_n(&__threads[slot].state, __ATOMIC_ACQUIRE) ) ++slot;
74   if ( slot == MICRON_WASM_MAX_THREADS ) {
75     __threads_lock.unlock();
76     return -11;
77   }
78   auto &t = __threads[slot];
79   if ( !t.stack ) {
80     t.stack = reinterpret_cast<unsigned>(linear_reserve(MICRON_WASM_STACK_SIZE, 16));
81     if ( !t.stack ) {
82       __threads_lock.unlock();
83       return -12;
84     }
85     t.stack_bytes = MICRON_WASM_STACK_SIZE;
86   }
87   if ( !t.tls ) {
88     t.tls = reinterpret_cast<unsigned>(linear_reserve(__builtin_wasm_tls_size(), __builtin_wasm_tls_align()));
89     if ( !t.tls ) {
90       __threads_lock.unlock();
91       return -12;
92     }
93   }
94   const unsigned generation = __atomic_add_fetch(&t.generation, 1u, __ATOMIC_RELAXED);
95   t.entry = entry;
96   t.arg = arg;
97   t.result = 0;
98   t.cancelled = 0;
99   t.park = 0;
100   __atomic_store_n(&t.state, 1u, __ATOMIC_RELEASE);
101   __atomic_fetch_add(&__live_threads, 1u, __ATOMIC_ACQ_REL);
102   handle = { slot, generation };
103   __threads_lock.unlock();
104   const int r = __micron_wasm_host_worker_spawn(slot, t.stack + t.stack_bytes, t.tls, reinterpret_cast<unsigned>(&t.state));
105   if ( r < 0 ) {
106     __threads_lock.lock();
107     __atomic_store_n(&t.state, 0u, __ATOMIC_RELEASE);
108     __atomic_fetch_sub(&__live_threads, 1u, __ATOMIC_ACQ_REL);
109     __threads_lock.unlock();
110     handle = {};
111     return r;
112   }
113   return 0;
114 #else
115   (void)entry;
116   (void)arg;
117   (void)handle;
118   return -95;
119 #endif
120 }

122 inline int
123 request_cancel(thread_handle handle) noexcept
124 {
125   if ( !handle.slot || handle.slot >= MICRON_WASM_MAX_THREADS ) return -22;
126   __threads_lock.lock();
127   auto &t = __threads[handle.slot];
128   if ( __atomic_load_n(&t.generation, __ATOMIC_RELAXED) != handle.generation || !__atomic_load_n(&t.state, __ATOMIC_ACQUIRE) ) {
129     __threads_lock.unlock();
130     return -22;
131   }
132   __atomic_store_n(&t.cancelled, 1u, __ATOMIC_RELEASE);
133   __atomic_store_n(&t.park, micron::__park_dying, __ATOMIC_RELEASE);
134   __threads_lock.unlock();
135   (void)wake32(&t.park);
136   return 0;
137 }

139 inline int
140 __set_park(thread_handle handle, bool parked) noexcept
141 {
142   if ( !handle.slot || handle.slot >= MICRON_WASM_MAX_THREADS ) return -22;
143   __threads_lock.lock();
144   auto &t = __threads[handle.slot];
145   const unsigned state = __atomic_load_n(&t.state, __ATOMIC_ACQUIRE);
146   if ( __atomic_load_n(&t.generation, __ATOMIC_RELAXED) != handle.generation || (state != 1 && state != 2) ) {
147     __threads_lock.unlock();
148     return -22;
149   }
150   if ( __atomic_load_n(&t.cancelled, __ATOMIC_ACQUIRE) ) {
151     __threads_lock.unlock();
152     return -125;
153   }
154   __atomic_store_n(&t.park, parked ? micron::__park_parked : micron::__park_run, __ATOMIC_RELEASE);
155   __threads_lock.unlock();
156   if ( !parked ) (void)wake32(&t.park);
157   return 0;
158 }

160 template<void (*BeforeState)(thread_handle) = nullptr>
161 inline int
162 __join(thread_handle &handle, int *result = nullptr) noexcept
163 {
164   if ( !handle.slot || handle.slot >= MICRON_WASM_MAX_THREADS ) return -22;
165   if ( handle.slot + 1 == __thread_id ) return -35;
166   auto &t = __threads[handle.slot];
167   unsigned state;
168   for ( ;; ) {
169     if ( __atomic_load_n(&t.generation, __ATOMIC_ACQUIRE) != handle.generation ) return -22;
170     if constexpr ( BeforeState != nullptr ) BeforeState(handle);
171     state = __atomic_load_n(&t.state, __ATOMIC_ACQUIRE);
172     if ( state != 1 && state != 2 ) break;
173     // A copied handle can be reaped and its slot reused between the two loads.
174     // Bound the state-only wait so it cannot wait indefinitely on that successor.
175     const int r = wait32(&t.state, state, __join_poll_ns);
176     if ( r < 0 && r != -11 && r != -110 ) return r;
177   }
178   __threads_lock.lock();
179   state = __atomic_load_n(&t.state, __ATOMIC_ACQUIRE);
180   if ( __atomic_load_n(&t.generation, __ATOMIC_RELAXED) != handle.generation || (state != 3 && state != 4) ) {
181     __threads_lock.unlock();
182     return -22;
183   }
184   if ( result ) *result = t.result;
185   const int status = state == 4 ? -5 : 0;
186   __atomic_store_n(&t.state, 0u, __ATOMIC_RELEASE);
187   __atomic_fetch_sub(&__live_threads, 1u, __ATOMIC_ACQ_REL);
188   __threads_lock.unlock();
189   handle = {};
190   return status;
191 }

199 inline void
200 finish_thread(bool cancelled = false) noexcept
201 {
202   if ( __thread_finished ) return;
203   __thread_finished = true;
204   micron::__micron_thread_park = nullptr;
205   micron::__micron_thread_die = nullptr;
206 #if defined(__micron_eh)
207   if ( cancelled ) __micron_wasm_finish_exceptions();
208 #else
209   (void)cancelled;
210 #endif
211   __run_thread_dtors();
212   if ( __thread_exit_hook ) __thread_exit_hook();
213   if ( __dynamic_tls_exit_hook ) __dynamic_tls_exit_hook();
214 }
