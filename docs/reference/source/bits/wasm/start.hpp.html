67 __attribute__((weak, used)) int
68 __cxa_guard_acquire(__micron_guard_t *guard)
69 {
70   auto *p = reinterpret_cast<unsigned *>(guard);
71   for ( unsigned i = 0; i < micron::wasm::__guard_count; ++i )
72     if ( micron::wasm::__active_guards[i] == p ) micron::wasm::panic();
73   for ( ;; ) {
74     unsigned v = __atomic_load_n(p, __ATOMIC_ACQUIRE);
75     if ( v & 1u ) return 0;
76     if ( !(v & 0x100u) && __atomic_compare_exchange_n(p, &v, 0x100u, false, __ATOMIC_ACQUIRE, __ATOMIC_RELAXED) ) {
77       if ( micron::wasm::__guard_count == 64 ) micron::wasm::panic();
78       micron::wasm::__active_guards[micron::wasm::__guard_count++] = p;
79       return 1;
80     }
81     if ( micron::wasm::wait32(p, 0x100u) == -95 ) micron::wasm::panic();
82   }
83 }

119 __attribute__((export_name("__micron_wasm_initialize"))) int
120 __micron_wasm_initialize(unsigned worker) noexcept
121 {
122   using namespace micron::wasm;
123 #if defined(__micron_wasm_threads)
124   if ( !worker ) return -95;
125 #endif
126   __can_wait = worker != 0;
127   __thread_id = 1;
128   __stack_begin = reinterpret_cast<unsigned>(&__stack_low);
129   __stack_bytes = reinterpret_cast<unsigned>(&__stack_high) - __stack_begin;
130   unsigned expected = 0;
131   if ( !__atomic_compare_exchange_n(&__runtime_state, &expected, 1u, false, __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE) )
132     return expected == 2 ? 0 : -16;
133   linear_memory_init();
134 #if defined(__micron_eh)
135   try {
136     __wasm_call_ctors();
137   } catch ( ... ) {
138     __atomic_store_n(&__runtime_state, 4u, __ATOMIC_RELEASE);
139     return -5;
140   }
141 #else
142   __wasm_call_ctors();
143 #endif
144   __atomic_store_n(&__runtime_state, 2u, __ATOMIC_RELEASE);
145   return 0;
146 }

148 __attribute__((export_name("__micron_wasm_run_main"))) int
149 __micron_wasm_run_main(int argc, char **argv)
150 {
151   if ( __atomic_load_n(&micron::wasm::__runtime_state, __ATOMIC_ACQUIRE) != 2 ) return -16;
152   if ( !__micron_user_main_args && !__micron_user_main ) return -38;
153   char *empty_arguments[] = { nullptr };
154   if ( argc < 0 || (argc && !argv) ) return -22;
155   if ( !argv ) argv = empty_arguments;
156 #if defined(__micron_eh)
157   try {
158     return __micron_user_main_args ? __micron_user_main_args(argc, argv) : __micron_user_main();
159   } catch ( ... ) {
160     micron::wasm::write_diag("micron: uncaught exception at main boundary\n");
161     return -5;
162   }
163 #else
164   return __micron_user_main_args ? __micron_user_main_args(argc, argv) : __micron_user_main();
165 #endif
166 }

168 __attribute__((export_name("__micron_wasm_shutdown"))) int
169 __micron_wasm_shutdown() noexcept
170 {
171   using namespace micron::wasm;
172   __threads_lock.lock();
173   if ( __atomic_load_n(&__live_threads, __ATOMIC_ACQUIRE) ) {
174     __threads_lock.unlock();
175     return -16;
176   }
177   unsigned expected = 2;
178   if ( !__atomic_compare_exchange_n(&__runtime_state, &expected, 3u, false, __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE) ) {
179     __threads_lock.unlock();
180     return expected == 4 ? 0 : -16;
181   }
182   __threads_lock.unlock();
183   micron::__run_exit_sequence();
184   __atomic_store_n(&__runtime_state, 4u, __ATOMIC_RELEASE);
185   return 0;
186 }

188 __attribute__((export_name("__micron_wasm_worker_entry"))) int
189 __micron_wasm_worker_entry(unsigned slot)
190 {
191   using namespace micron::wasm;
192   if ( !slot || slot >= MICRON_WASM_MAX_THREADS ) return -22;
193   auto &t = __threads[slot];
194   __can_wait = true;
195   __thread_id = slot + 1;
196   __stack_begin = t.stack;
197   __stack_bytes = t.stack_bytes;
198   micron::__micron_thread_park = &t.park;
199   micron::__micron_thread_die = __exit_worker;
200   unsigned expected = 1;
201   if ( !__atomic_compare_exchange_n(&t.state, &expected, 2u, false, __ATOMIC_ACQUIRE, __ATOMIC_RELAXED) ) return -22;
202   bool failed = false;
203 #if defined(__micron_eh)
204   try {
205     t.result = t.entry(t.arg);
206   } catch ( ... ) {
207     t.result = -5;
208     failed = true;
209   }
210 #else
211   t.result = t.entry(t.arg);
212 #endif
213   finish_thread();
214   return failed ? -5 : 0;
215 }

217 __attribute__((export_name("__micron_wasm_worker_cleanup"))) int
218 __micron_wasm_worker_cleanup(unsigned slot) noexcept
219 {
220   using namespace micron::wasm;
221   if ( !slot || slot >= MICRON_WASM_MAX_THREADS || __thread_id != slot + 1 ) return -22;
222   auto &t = __threads[slot];
223   if ( __atomic_load_n(&t.state, __ATOMIC_ACQUIRE) != 2 ) return -22;
224   t.result = -125;
225   while ( __guard_count ) {
226     auto *p = __active_guards[--__guard_count];
227     __atomic_store_n(p, 0u, __ATOMIC_RELEASE);
228     (void)wake32(p);
229   }
230   finish_thread(true);
231   return 0;
232 }
