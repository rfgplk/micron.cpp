3 import {createLocal, MicronStateError} from "./host.mjs";
4 const node = typeof process !== "undefined" && Boolean(process.versions?.node);
5 const port = node ? (await import("node:worker_threads")).parentPort : globalThis;
6 const send = message => port.postMessage(message);
7 let runtime, scheduler, memory, active, busy = false;
8 function exitWorker() {
9   if (!active || !busy) throw new Error("Invalid Micron worker exit state");
10   const park = new Int32Array(new SharedArrayBuffer(4));
11   runtime.flushOutput();
12   send({...active, type: "worker-exit"});
13   for (;;) Atomics.wait(park, 0, 0);
14 }
15 async function receive(message) {
16   if (message.type === "boot") {
17     try {
18       scheduler = message.scheduler; memory = message.memory;
19       runtime = await createLocal(message.module, memory, {
20         worker: true, deferInitialize: true, scheduler, dispatch: send, threadExit: exitWorker,
21         onWrite: (text, stream) => send({type: "write", text, stream})
22       });
23       send({type: "ready"});
24     } catch (error) { send({type: "failed", error: String(error?.stack ?? error), name: error?.name, code: error?.code}); }
25   } else if (message.type === "run" || message.type === "cleanup") {
26     if (busy || Atomics.load(scheduler, 0)) { send({type: "failed", error: "Invalid Micron worker state"}); return; }
27     busy = true; active = message;
28     try {
29       runtime.exports.__stack_pointer.value = message.stack;
30       let result;
31       if (message.type === "cleanup") {
32         runtime.exports.__tls_base.value = message.tls;
33         result = runtime.exports.__micron_wasm_worker_cleanup(message.slot);
34         if (result !== 0) throw new Error(`Micron worker cleanup failed: ${result}`);
35       } else {
36         runtime.exports.__wasm_init_tls(message.tls);
37         result = runtime.exports.__micron_wasm_worker_entry(message.slot);
38       }
39       runtime.flushOutput();
40       // Publish only after Wasm has returned: its stack and TLS may now be reused.
41       busy = false; active = undefined;
42       // The broker sees every preceding output message before publishing completion.
43       send({type: "complete", pool: message.pool, control: message.control, result});
44     } catch (error) {
45       Atomics.store(scheduler, 0, 1);
46       send({type: "failed", error: String(error?.stack ?? error), name: error?.name, code: error?.code});
47     }
48   } else if (message.type === "call") {
49     try {
50       const result = await runtime[message.operation](...message.args);
51       send({type: "reply", id: message.id, result});
52     } catch (error) {
53       send({type: error instanceof MicronStateError ? "reply" : "failed", id: message.id,
54         error: String(error?.stack ?? error), name: error?.name, code: error?.code});
55     }
56   }
57 }
58 if (node) port.on("message", receive); else port.addEventListener("message", e => receive(e.data));
