34 __micron_ti_strcmp(const char *a, const char *b) noexcept
35 {
36   while ( *a && (*a == *b) ) {
37     ++a;
38     ++b;
39   }

139 struct __upcast_result {
140   void *object = nullptr;
141   usize identity = 0;
142   bool found = false, ambiguous = false, accessible = false;
143 
144   void
145   add(void *p, usize id, bool access) noexcept
146   {
147     if ( !found ) {
148       object = p;
149       identity = id;
150       found = true;
151       accessible = access;
152     } else if ( object != p || (!p && identity != id) )
153       ambiguous = true;
154     else
155       accessible = accessible || access;
156   }

367 [[gnu::weak]] void
368 __vmi_class_type_info::__collect(const __class_type_info *target, void *object, __upcast_result &result, bool access, usize identity) const
369 {
370   if ( *this == *target ) {
371     result.add(object, identity, access);
372     return;
373   }
374   for ( unsigned i = 0; i < __base_count; ++i ) {
375     const auto &base = __base_info[i];
376     long offset = base.__offset_flags >> __base_class_type_info::__offset_shift;
377     const bool virt = base.__offset_flags & __base_class_type_info::__virtual_mask;
378     void *candidate = object;
379     if ( candidate ) {

381       candidate = static_cast<char *>(candidate) + offset;

438 [[gnu::weak]] bool
439 __pbase_type_info::__do_catch(const type_info *__thr_type, void **__thr_obj, unsigned __outer) const
440 {
441   if ( *this == *__thr_type ) return true;
442   if ( std::__micron_ti_strcmp(__thr_type->name(), "Dn") == 0 && __is_pointer_p() ) {
443     *__thr_obj = nullptr;
444     return true;
445   }
446   if ( __thr_type->__is_pointer_p() != __is_pointer_p() || !(__outer & 1) ) return false;
447   if ( !__is_pointer_p() && (name()[0] != 'M' || __thr_type->name()[0] != 'M') ) return false;
448   const auto *thrown = static_cast<const __pbase_type_info *>(__thr_type);
449   if ( (thrown->__flags & ~__flags) & (__const_mask | __volatile_mask | __restrict_mask) ) return false;
450   if ( (__flags & __noexcept_mask) && !(thrown->__flags & __noexcept_mask) ) return false;
451   if ( !(__flags & __const_mask) ) __outer &= ~1u;
452   return __pointer_catch(thrown, __thr_obj, __outer);
453 }

502 struct dynamic_target {
503   dynamic_source source;
504   const __cxxabiv1::__class_type_info *target;
505   void *found = nullptr;
506   bool ambiguous = false;
507 
508   static void
509   visit(const __cxxabiv1::__class_type_info *type, void *object, bool, void *p)
510   {
511     auto &s = *static_cast<dynamic_target *>(p);
512     if ( *type != *s.target ) return;
513     auto probe = s.source;
514     type->__walk(object, true, dynamic_source::visit, &probe);
515     if ( !probe.found ) return;
516     if ( s.found && s.found != object )
517       s.ambiguous = true;
518     else

520   }

524 extern "C" [[gnu::weak, gnu::used]] void *
525 __dynamic_cast(const void *source, const __cxxabiv1::__class_type_info *source_type, const __cxxabiv1::__class_type_info *target_type,
526                __PTRDIFF_TYPE__) noexcept
527 {
528   if ( !source ) return nullptr;
529   const auto *vtable = *static_cast<const usize *const *>(source);
530   const auto *complete_type = reinterpret_cast<const __cxxabiv1::__class_type_info *>(vtable[-1]);
531   void *complete = const_cast<char *>(static_cast<const char *>(source)) + static_cast<__PTRDIFF_TYPE__>(vtable[-2]);
532   micron::wasm::eh::dynamic_target search{ { source, source_type }, target_type };
533   complete_type->__walk(complete, true, micron::wasm::eh::dynamic_target::visit, &search);
534   if ( search.ambiguous ) return nullptr;
535   if ( search.found ) return search.found;
536   auto src = search.source;
537   complete_type->__walk(complete, true, micron::wasm::eh::dynamic_source::visit, &src);
538   if ( !src.found ) return nullptr;
539   __cxxabiv1::__upcast_result target;
540   complete_type->__collect(target_type, complete, target, true, 1);
541   return target.found && target.accessible && !target.ambiguous ? target.object : nullptr;
542 }
