37 // arity is decided by the types own aliases, never by probing a bare for_each
38 // requires expression does not instantiate the body
39 template<typename T>
40 concept __kv_visitable
41     = (micron::is_tagged_map<T> || micron::is_tree<T>) && requires { typename __U<T>::mapped_type; } && requires(const __U<T> t) {
42         // const auto & for the key, NOT const key_type &; certain maps store hash key not the key itself (hopscotch), this also dodges any
43         // issues with future maps
44         t.for_each([](const auto &, const typename __U<T>::mapped_type &) { });
45       };

235 template<typename T>
236 constexpr kind
237 classify(void)
238 {
239   using U = micron::remove_cvref_t<T>;
240 
241   // scalars, strings, raw pointers, and enums
242   if constexpr ( micron::has_cstr<U> || micron::is_arithmetic_v<U> || micron::is_pointer_v<U> || micron::is_enum_v<U> ) return kind::none;
243 
244   // types that are a number but expose an indexable interface over their storage
245   else if constexpr ( __numeric_tagged<U> )
246     return kind::none;
247 
248   else if constexpr ( __semantic<U> )
249     return kind::semantic;
250 
251   // node chains
252   else if constexpr ( __node_chain<U> )
253     return kind::node_chain;
254 
255   // keyed containers
256   else if constexpr ( __kv_visitable<U> )
257     return kind::kv_visit;
258   else if constexpr ( __set_visitable<U> )
259     return kind::set_visit;
260 
261   // sums before nullable/indexed
262   else if constexpr ( micron::is_option<U> )
263     return kind::option_like;
264   else if constexpr ( __sum_visitable<U> )
265     return kind::sum_visit;
266   else if constexpr ( micron::tuple_like<U> )
267     return kind::tuple_like;
268 
269   // shapes
270   else if constexpr ( __sparse_matrix<U> )
271     return kind::sparse_mat;
272   else if constexpr ( __sparse_vector<U> )
273     return kind::sparse_vec;
274   else if constexpr ( __dense_matrix<U> )
275     return kind::matrix;
276   else if constexpr ( __fixed_vector<U> )
277     return kind::vector_n;
278   else if constexpr ( __ndarray<U> )
279     return kind::ndarray;
280   else if constexpr ( __soa_like<U> )
281     return kind::soa_rows;
282   else if constexpr ( __bitfield_like<U> )
283     return kind::bits;
284 
285   // remaining maps
286   else if constexpr ( __kv_key_iter<U> )
287     return kind::kv_iter_key;
288   else if constexpr ( __kv_ab_iter<U> )
289     return kind::kv_iter_ab;
290 
291   else if constexpr ( micron::is_printable_container<U> )
292     return kind::sequence;
293 
294   // enumerable but not iterable
295   else if constexpr ( __seq_visitable<U> )
296     return kind::set_visit;
297   else if constexpr ( __indexable<U> )
298     return kind::indexed;
299 
300   // wrappers and last resorts
301   else if constexpr ( __smart_pointer<U> )
302     return kind::smart_ptr;
303   else if constexpr ( __ptr_range<U> )
304     return kind::ptr_range;
305   else if constexpr ( micron::nullable<U> )
306     return kind::nullable_opaque;
307   else if constexpr ( __consuming<U> )
308     return kind::consuming;
309   else if constexpr ( __opaque_tagged<U> )
310     return kind::opaque;
311   else
312     return kind::none;
313 }

323 template<typename Out, typename T, usize... Is>
324 constexpr void
325 emit_product(Out &o, const T &x, micron::index_sequence<Is...>)
326 {
327   bool first = true;
328   auto one = [&](const auto &v) {
329     if ( !first ) o.raw(", ", 2);
330     first = false;
331     o.elem(v);
332   };
333   (one(micron::get<Is>(x)), ...);

349 template<typename Out, typename T>
350 constexpr void
351 emit_nd(Out &o, const T &x, usize axis, usize off, usize span)
352 {
353   const usize n = x.shape(axis);
354   o.raw("{ ", 2);
355   if ( axis + 1u == static_cast<usize>(micron::remove_cvref_t<T>::rank) ) {
356     for ( usize i = 0; i < n; ++i ) {
357       if ( i ) o.raw(", ", 2);
358       o.elem(x.begin()[off + i]);
359     }
360   } else {
361     const usize sub = n ? span / n : 0;
362     for ( usize i = 0; i < n; ++i ) {
363       if ( i ) o.raw(", ", 2);
364       micron::__print::__impl::emit_nd(o, x, axis + 1u, off + i * sub, sub);
365     }
366   }
367   o.raw(" }", 2);
368 }

372 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
373 // main print renderer
374 //
375 // { } collection, [ ] fixed arity product, ( ) mathematical vector, name(x) tagged value
376 
377 template<typename Out, typename T>
378 constexpr void
379 render(Out &o, const T &x)
380 {
381   using U = micron::remove_cvref_t<T>;
382   constexpr micron::__print::kind K = micron::__print::kind_of_v<U>;
383 
384   if constexpr ( K == kind::semantic ) {
385     x.__micron_print(o);
386 
387   } else if constexpr ( K == kind::node_chain ) {
388     o.raw("{ ", 2);
389     bool first = true;
390     for ( auto *p = x.ibegin(); p != nullptr; p = p->next ) {
391       if ( !first ) o.raw(", ", 2);
392       first = false;
393       o.elem(p->data);
394     }
395     o.raw(" }", 2);
396 
397   } else if constexpr ( K == kind::kv_visit ) {
398     o.raw("{ ", 2);
399     bool first = true;
400     x.for_each([&](const auto &k, const auto &v) {
401       if ( !first ) o.raw(", ", 2);
402       first = false;
403       o.elem(k);      // recursed, so stringy keys compile
404       o.raw(": ", 2);
405       o.elem(v);
406     });
407     o.raw(" }", 2);
408 
409   } else if constexpr ( K == kind::set_visit ) {
410     o.raw("{ ", 2);
411     bool first = true;
412     x.for_each([&](const auto &v) {
413       if ( !first ) o.raw(", ", 2);
414       first = false;
415       o.elem(v);
416     });
417     o.raw(" }", 2);
418 
419   } else if constexpr ( K == kind::option_like ) {
420     using F1 = typename U::first_type;
421     using F2 = typename U::second_type;
422     if ( !x.has_value() ) {
423       o.raw("none", 4);
424     } else if ( x.is_first() ) {
425       o.raw("first(", 6);
426       if constexpr ( !micron::is_void_v<F1> ) o.elem(x.template cast<F1>());
427       o.raw(")", 1);
428     } else {
429       o.raw("second(", 7);
430       o.elem(x.template cast<F2>());
431       o.raw(")", 1);
432     }
433 
434   } else if constexpr ( K == kind::sum_visit ) {
435     // the index is the only stable identity for an any<Ts...>
436     if ( !x.has_value() ) {
437       o.raw("none", 4);
438     } else {
439       o.raw("any#", 4);
440       o.num(static_cast<u64>(x.index()));
441       o.raw("(", 1);
442       x.visit([&](const auto &v) { o.elem(v); });
443       o.raw(")", 1);
444     }
445 
446   } else if constexpr ( K == kind::tuple_like ) {
447     o.raw("[", 1);
448     micron::__print::__impl::emit_product(o, x, micron::make_index_sequence<micron::tuple_size_v<U>>{});
449     o.raw("]", 1);
450 
451   } else if constexpr ( K == kind::sparse_mat ) {
452     constexpr bool by_row = requires(const U t, usize i) { t.row_begin(i); };
453     const usize major = x.outer.size() ? x.outer.size() - 1u : 0u;
454     const usize lim = x.values.size() < x.inner.size() ? x.values.size() : x.inner.size();
455     o.raw("{ ", 2);
456     bool first = true;
457     for ( usize m = 0; m < major; ++m ) {
458       usize e = static_cast<usize>(x.outer.data()[m + 1]);
459       if ( e > lim ) e = lim;      // a malformed/partially-built outer must not walk off inner
460       for ( usize p = static_cast<usize>(x.outer.data()[m]); p < e; ++p ) {
461         if ( !first ) o.raw(", ", 2);
462         first = false;
463         const usize minor = static_cast<usize>(x.inner.data()[p]);
464         o.raw("(", 1);
465         o.num(static_cast<u64>(by_row ? m : minor));
466         o.raw(", ", 2);
467         o.num(static_cast<u64>(by_row ? minor : m));
468         o.raw("): ", 3);
469         o.elem(x.values.data()[p]);
470       }
471     }
472     o.raw(" }", 2);
473 
474   } else if constexpr ( K == kind::sparse_vec ) {
475     o.raw("{ ", 2);
476     for ( usize p = 0; p < x.values.size(); ++p ) {
477       if ( p ) o.raw(", ", 2);
478       o.num(static_cast<u64>(x.idx.data()[p]));
479       o.raw(": ", 2);
480       o.elem(x.values.data()[p]);
481     }
482     o.raw(" }", 2);
483 
484   } else if constexpr ( K == kind::matrix ) {
485     const usize R = static_cast<usize>(x.rows);
486     const usize C = static_cast<usize>(x.cols);
487     o.raw("{ ", 2);
488     for ( usize r = 0; r < R; ++r ) {
489       if ( r ) o.raw(", ", 2);
490       o.raw("{ ", 2);
491       for ( usize c = 0; c < C; ++c ) {
492         if ( c ) o.raw(", ", 2);
493         o.elem(x.at(r, c));
494       }
495       o.raw(" }", 2);
496     }
497     o.raw(" }", 2);
498 
499   } else if constexpr ( K == kind::vector_n ) {
500     o.raw("(", 1);
501     for ( usize i = 0; i < static_cast<usize>(U::length); ++i ) {
502       if ( i ) o.raw(", ", 2);
503       o.elem(x[i]);
504     }
505     o.raw(")", 1);
506 
507   } else if constexpr ( K == kind::ndarray ) {
508     if ( !x.size() ) [[unlikely]] {
509       o.raw("{ }", 3);
510     } else {
511       micron::__print::__impl::emit_nd(o, x, 0, 0, static_cast<usize>(x.size()));
512     }
513 
514   } else if constexpr ( K == kind::soa_rows ) {
515     o.raw("{ ", 2);
516     for ( usize r = 0; r < x.size(); ++r ) {
517       if ( r ) o.raw(", ", 2);
518       o.raw("[", 1);
519       micron::__print::__impl::emit_soa_row(o, x, r, micron::make_index_sequence<U::column_count>{});
520       o.raw("]", 1);
521     }
522     o.raw(" }", 2);
523 
524   } else if constexpr ( K == kind::bits ) {
525     const usize nbits = static_cast<usize>(x.size()) * 8u;
526     o.raw("0b", 2);
527     char b[64];
528     usize k = 0;
529     for ( usize i = 0; i < nbits; ++i ) {
530       b[k++] = x[i] ? '1' : '0';
531       if ( k == sizeof(b) ) {
532         o.raw(b, k);
533         k = 0;
534       }
535     }
536     if ( k ) o.raw(b, k);
537 
538   } else if constexpr ( K == kind::kv_iter_key ) {
539     // residuals for untagged third party key/value maps
540     o.raw("{ ", 2);
541     bool first = true;
542     for ( auto itr = x.begin(); itr != x.end(); ++itr ) {
543       if constexpr ( requires {
544                        { !itr->key } -> micron::convertible_to<bool>;
545                      } ) {
546         if ( !itr->key ) continue;
547       }
548       if ( !first ) o.raw(", ", 2);
549       first = false;
550       o.elem(itr->key);
551       o.raw(": ", 2);
552       o.elem(itr->value);
553     }
554     o.raw(" }", 2);
555 
556   } else if constexpr ( K == kind::kv_iter_ab ) {
557     o.raw("{ ", 2);
558     bool first = true;
559     for ( auto itr = x.begin(); itr != x.end(); ++itr ) {
560       if ( !first ) o.raw(", ", 2);
561       first = false;
562       auto entry = *itr;
563       o.elem(entry.a);
564       o.raw(": ", 2);
565       o.elem(entry.b);
566     }
567     o.raw(" }", 2);
568 
569   } else if constexpr ( K == kind::sequence ) {
570     o.raw("{ ", 2);
571     bool first = true;
572     for ( auto itr = x.cbegin(); itr != x.cend(); ++itr ) {
573       if ( !first ) o.raw(", ", 2);
574       first = false;
575       // deducing from *itr mishandles cv-qualified and proxy iterators
576       if constexpr ( requires { typename U::value_type; } )
577         o.elem(static_cast<const typename U::value_type &>(*itr));
578       else
579         o.elem(*itr);
580     }
581     o.raw(" }", 2);
582 
583   } else if constexpr ( K == kind::indexed ) {
584     o.raw("{ ", 2);
585     for ( usize i = 0; i < static_cast<usize>(x.size()); ++i ) {
586       if ( i ) o.raw(", ", 2);
587       o.elem(x[i]);
588     }
589     o.raw(" }", 2);
590 
591   } else if constexpr ( K == kind::smart_ptr ) {
592     if ( x.get() == nullptr ) [[unlikely]]
593       o.raw("null", 4);
594     else
595       o.elem(*x.get());
596 
597   } else if constexpr ( K == kind::ptr_range ) {
598     o.raw("<view ", 6);
599     o.elem(static_cast<const void *>(x.begin()));
600     if ( x.end() == nullptr ) [[unlikely]] {
601       o.raw(" +?>", 4);
602     } else {
603       o.raw(" +", 2);
604       o.num(static_cast<u64>(reinterpret_cast<const byte *>(x.end()) - reinterpret_cast<const byte *>(x.begin())));
605       o.raw(">", 1);
606     }
607 
608   } else if constexpr ( K == kind::nullable_opaque ) {
609     if ( x.has_value() )
610       o.raw("<opaque>", 8);
611     else
612       o.raw("none", 4);
613 
614   } else if constexpr ( K == kind::opaque ) {
615     // '~' marks a container that cannot be enumerated coherently
616     o.raw("{ ~", 3);
617     if constexpr ( requires(const U t) {
618                      { t.size() } -> micron::convertible_to<usize>;
619                    } )
620       o.num(static_cast<u64>(x.size()));
621     o.raw(" }", 2);
622 
623   } else if constexpr ( K == kind::consuming ) {
624     static_assert(sizeof(U) == 0, "micron::io: this range is single-pass and consuming (micron::generator); printing it would "
625                                   "drain it. collect it first, e.g. io::echo(micron::vector<T>(g.begin(), g.end()))");
626   }
627 }
628 
629 };      // namespace __print
630 };      // namespace micron
