12 // cpuid

15 namespace micron
16 {
17 
18 struct cpuid_regs {
19   u32 eax;
20   u32 ebx;
21   u32 ecx;
22   u32 edx;
23 };
24 
25 #if defined(__micron_arch_x86_any)
26 
27 [[gnu::always_inline]] inline cpuid_regs
28 __cpuid_read(u32 leaf, u32 sub = 0) noexcept
29 {
30   cpuid_regs r{ 0, 0, 0, 0 };
31 #if defined(__micron_arch_x86) && (defined(__PIC__) || defined(__pic__))
32   // NOTE: on 32-bit PIC ebx is the GOT pointer and cannot be an output operand
33   __asm__ __volatile__("xchgl %%ebx, %1\n\t"
34                        "cpuid\n\t"
35                        "xchgl %%ebx, %1"
36                        : "=a"(r.eax), "=&r"(r.ebx), "=c"(r.ecx), "=d"(r.edx)
37                        : "0"(leaf), "2"(sub));
38 #else
39   __asm__ __volatile__("cpuid" : "=a"(r.eax), "=b"(r.ebx), "=c"(r.ecx), "=d"(r.edx) : "0"(leaf), "2"(sub));
40 #endif
41   return r;
42 }
43 
44 inline u32
45 __cpuid_max_leaf(void) noexcept
46 {
47   return __cpuid_read(0u).eax;
48 }
49 
50 inline u32
51 __cpuid_max_ext_leaf(void) noexcept
52 {
53   return __cpuid_read(0x8000'0000u).eax;
54 }
55 
56 inline bool
57 __cpuid_has_leaf(u32 leaf) noexcept
58 {
59   if ( leaf >= 0x8000'0000u ) return __cpuid_max_ext_leaf() >= leaf;
60   return __cpuid_max_leaf() >= leaf;
61 }
62 
63 #endif      // __micron_arch_x86_any
