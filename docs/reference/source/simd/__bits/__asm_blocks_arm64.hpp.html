31     __inline_g void __block_copy_16(u8 *__restrict d, const u8 *__restrict s) noexcept
32 {
33   uint8x16_t t;

35   __asm__("str %q1, %0" : "=Q"(*reinterpret_cast<uint8x16_t *>(d)) : "w"(t));

205 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
206 // 16-byte memcmp (CMEQ.16B + UMINV.B)
207 __inline_g bool
208 __block_all_eq_16(const u8 *a, const u8 *b) noexcept
209 {
210   uint8x16_t va, vb, ceq;
211   __asm__("ldr %q0, %1" : "=w"(va) : "Q"(*reinterpret_cast<const uint8x16_t *>(a)));
212   __asm__("ldr %q0, %1" : "=w"(vb) : "Q"(*reinterpret_cast<const uint8x16_t *>(b)));
213   __asm__("cmeq %0.16b, %1.16b, %2.16b" : "=w"(ceq) : "w"(va), "w"(vb));
214   uint8x16_t minv;
215   __asm__("uminv %b0, %1.16b" : "=w"(minv) : "w"(ceq));
216   return reinterpret_cast<const u8 *>(&minv)[0] == 0xFFu;
217 }

219 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
220 // 16-byte byte-search mask (CMEQ + SHRN narrowed to 64-bit nibble mask)
221 // Returns a 64-bit value with 4 bits per source byte.
222 // Consumer uses __builtin_ctzll(m) >> 2 / (63 - __builtin_clzll(m)) >> 2 for byte index.
223 __inline_g u64
224 __block_eq_mask_16(const u8 *p, u8 c) noexcept
225 {
226   uint8x16_t v, bc, ceq;
227   __asm__("dup %0.16b, %w1" : "=w"(bc) : "r"(static_cast<u32>(c)));
228   __asm__("ldr %q0, %1" : "=w"(v) : "Q"(*reinterpret_cast<const uint8x16_t *>(p)));
229   __asm__("cmeq %0.16b, %1.16b, %2.16b" : "=w"(ceq) : "w"(v), "w"(bc));
230   uint8x16_t narrowed;
231   __asm__("shrn %0.8b, %1.8h, #4" : "=w"(narrowed) : "w"(ceq));
232   u64 m;
233   __asm__("umov %0, %1.d[0]" : "=r"(m) : "w"(narrowed));
234   return m;
235 }
