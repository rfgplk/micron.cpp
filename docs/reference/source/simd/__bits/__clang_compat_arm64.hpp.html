21 template<typename V>
22 [[gnu::always_inline]] static inline V
23 __clang_arm_absdiff(V a, V b) noexcept
24 {
25   V out{};
26   constexpr unsigned n = sizeof(V) / sizeof(a[0]);
27   for ( unsigned i = 0; i < n; ++i ) out[i] = a[i] < b[i] ? b[i] - a[i] : a[i] - b[i];
28   return out;
29 }

71 template<typename Out, bool High, typename V>
72 [[gnu::always_inline]] static inline Out
73 __clang_arm_widen_absdiff(V a, V b) noexcept
74 {
75   Out out{};
76   constexpr unsigned n = sizeof(Out) / sizeof(out[0]);
77   const unsigned base = High ? n : 0;
78   for ( unsigned i = 0; i < n; ++i ) {
79     long long av = a[base + i];
80     long long bv = b[base + i];
81     out[i] = av < bv ? bv - av : av - bv;
82   }
83   return out;
84 }

86 template<typename Acc, typename V>
87 [[gnu::always_inline]] static inline Acc
88 __clang_arm_widen_acc_absdiff(Acc acc, V a, V b) noexcept
89 {
90   constexpr unsigned n = sizeof(Acc) / sizeof(acc[0]);
91   for ( unsigned i = 0; i < n; ++i ) {
92     long long av = a[i];
93     long long bv = b[i];
94     acc[i] += av < bv ? bv - av : av - bv;
95   }
96   return acc;
97 }

99 template<typename Out, typename V>
100 [[gnu::always_inline]] static inline Out
101 __clang_arm_narrow_sat(V v, long long low, unsigned long long high) noexcept
102 {
103   Out out{};
104   constexpr unsigned n = sizeof(Out) / sizeof(out[0]);
105   for ( unsigned i = 0; i < n; ++i ) {
106     long long value = v[i];
107     if ( value < low ) value = low;
108     if ( value > static_cast<long long>(high) ) value = high;
109     out[i] = value;
110   }
111   return out;
112 }

162 [[gnu::always_inline]] static inline V
163 __clang_arm_tbl(V table, V indexes) noexcept
164 {
165   V out{};
166   constexpr unsigned n = sizeof(V) / sizeof(table[0]);
167   for ( unsigned i = 0; i < n; ++i ) {
168     const unsigned index = static_cast<unsigned char>(indexes[i]);
169     out[i] = index < n ? table[index] : 0;
