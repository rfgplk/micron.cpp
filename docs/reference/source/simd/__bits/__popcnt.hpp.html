24 #define __inline_g [[gnu::always_inline, gnu::artificial]] static inline

26     // POPCNT only exists after SSE4.2); SWAR workarounds
27     __inline_g int _mm_popcnt_u32(unsigned int v) noexcept
28 {
29 #if defined(__micron_x86_popcnt)
30   return __builtin_popcount(v);
31 #else
32   v = v - ((v >> 1) & 0x55555555u);
33   v = (v & 0x33333333u) + ((v >> 2) & 0x33333333u);
34   v = (v + (v >> 4)) & 0x0f0f0f0fu;
35   return (int)((v * 0x01010101u) >> 24);
36 #endif
37 }
38 
39 __inline_g int
40 _mm_popcnt_u64(unsigned long long v) noexcept
41 {
42 #if defined(__micron_x86_popcnt)
43   return __builtin_popcountll(v);
44 #elif defined(__micron_arch_width_64)
45   v = v - ((v >> 1) & 0x5555555555555555ull);
46   v = (v & 0x3333333333333333ull) + ((v >> 2) & 0x3333333333333333ull);
47   v = (v + (v >> 4)) & 0x0f0f0f0f0f0f0f0full;
48   return (int)((v * 0x0101010101010101ull) >> 56);
49 #else
50   return _mm_popcnt_u32((unsigned)v) + _mm_popcnt_u32((unsigned)(v >> 32));
51 #endif
52 }
53 
54 __inline_g int
55 _popcnt32(int v) noexcept
56 {
57   return _mm_popcnt_u32((unsigned)v);
58 }
59 
60 __inline_g int
61 _popcnt64(long long v) noexcept
62 {
63   return _mm_popcnt_u64((unsigned long long)v);
64 }
