123 __inline_g unsigned int
124 _bextr_u32(unsigned int v, unsigned int start, unsigned int len) noexcept
125 {
126 #if defined(__micron_x86_bmi1)
127   return __builtin_ia32_bextr_u32(v, ((start & 0xff) | ((len & 0xff) << 8)));
128 #else
129   unsigned s = start & 0xff, l = len & 0xff;
130   if ( s >= 32 ) return 0u;
131   unsigned r = v >> s;
132   return (l >= 32) ? r : (r & ((1u << l) - 1u));
133 #endif
134 }

195 __inline_g unsigned int
196 _pext_u32(unsigned int v, unsigned int mask) noexcept
197 {
198 #if defined(__micron_x86_bmi2)
199   return __builtin_ia32_pext_si(v, mask);
200 #else
201   unsigned r = 0;
202   int k = 0;
203   while ( mask ) {
204     unsigned lo = mask & (~mask + 1u);      // lowest set bit
205     if ( v & lo ) r |= (1u << k);
206     ++k;
207     mask &= mask - 1u;
208   }
209   return r;
210 #endif
211 }

267 __inline_g unsigned int
268 _mulx_u32(unsigned int a, unsigned int b, unsigned int *hi) noexcept
269 {
270   unsigned long long p = (unsigned long long)a * b;
271   *hi = (unsigned int)(p >> 32);
272   return (unsigned int)p;
273 }
274 
275 __inline_g unsigned long long
276 _mulx_u64(unsigned long long a, unsigned long long b, unsigned long long *hi) noexcept
277 {
278 #if defined(__micron_arch_width_64)
279   unsigned __int128 p = (unsigned __int128)a * b;
280   *hi = (unsigned long long)(p >> 64);
281   return (unsigned long long)p;
282 #else
283   // 32x32 -> 64 schoolbook (no __int128 on i386)
284   unsigned long long al = (unsigned int)a, ah = a >> 32;
285   unsigned long long bl = (unsigned int)b, bh = b >> 32;
286   unsigned long long ll = al * bl, lh = al * bh, hl = ah * bl, hh = ah * bh;
287   unsigned long long cross = (ll >> 32) + (lh & 0xFFFFFFFFull) + (hl & 0xFFFFFFFFull);
288   *hi = hh + (lh >> 32) + (hl >> 32) + (cross >> 32);
289   return (ll & 0xFFFFFFFFull) | (cross << 32);
290 #endif
291 }
