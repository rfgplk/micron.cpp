98 // scheduling for batched bulk loops
99 // makes every subsequent store dependent on all four loaded values
100 [[gnu::always_inline]] static inline void
101 __pin4(__v16 &a, __v16 &b, __v16 &c, __v16 &e) noexcept
102 {
103 #if defined(__micron_arch_x86_any)
104   __asm__("" : "+x"(a), "+x"(b), "+x"(c), "+x"(e));
