179 [[nodiscard]] consteval int
180 __mm_shuffle_imm(int fp3, int fp2, int fp1, int fp0) noexcept
181 {
182   return (fp3 << 6) | (fp2 << 4) | (fp1 << 2) | fp0;
183 }
184 
185 [[nodiscard]] consteval int
186 __mm_shuffle2_imm(int fp1, int fp0) noexcept
187 {
188   return (fp1 << 1) | fp0;
189 }
190 
191 [[nodiscard]] consteval int
192 __mm_mk_insertps_ndx_imm(int s, int d, int m) noexcept
193 {
194   return (s << 6) | (d << 4) | m;
195 }
