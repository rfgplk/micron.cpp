30     // 16-byte copy (VLD1.8 / VST1.8)
31     __inline_g void __block_copy_16(u8 *__restrict d, const u8 *__restrict s) noexcept
32 {
33   uint8x16_t t;
34   __asm__("vld1.8 {%q0}, [%1]" : "=w"(t) : "r"(s), "m"(*reinterpret_cast<const u8(*)[16]>(s)));
35   __asm__("vst1.8 {%q2}, [%1]" : "=m"(*reinterpret_cast<u8(*)[16]>(d)) : "r"(d), "w"(t));
36 }

62 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
63 // 16-byte broadcast + set (VDUP.8 / VST1.8)
64 __inline_g uint8x16_t
65 __broadcast_byte_16(u8 b) noexcept
66 {
67   uint8x16_t v;
68   __asm__("vdup.8 %q0, %1" : "=w"(v) : "r"(static_cast<u32>(b)));
69   return v;
70 }

72 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
73 // 16-byte word broadcast: u64 -> 2x 64-bit lanes via vmov d, r, r + vmov q.d[1], q.d[0]
74 __inline_g uint8x16_t
75 __broadcast_word_16(u64 w) noexcept
76 {
77   uint8x16_t v;
78   __asm__("vmov %e0, %Q1, %R1\n\t"
79           "vmov %f0, %e0"
80           : "=w"(v)
81           : "r"(w));
82   return v;
83 }

103 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
104 // 16-byte memcmp (VCEQ.I8)
105 __inline_g uint8x16_t
106 __block_cmpeq_16(const u8 *a, const u8 *b) noexcept
107 {
108   uint8x16_t va, vb, ceq;
109   __asm__("vld1.8 {%q0}, [%1]" : "=w"(va) : "r"(a), "m"(*reinterpret_cast<const u8(*)[16]>(a)));
110   __asm__("vld1.8 {%q0}, [%1]" : "=w"(vb) : "r"(b), "m"(*reinterpret_cast<const u8(*)[16]>(b)));
111   __asm__("vceq.i8 %q0, %q1, %q2" : "=w"(ceq) : "w"(va), "w"(vb));
112   return ceq;
113 }

115 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
116 // 16-byte byte-search mask (VCEQ + VSHRN narrowed to 64-bit nibble mask)
117 __inline_g u64
118 __block_eq_mask_16(const u8 *p, u8 c) noexcept
119 {
120   uint8x16_t v, bc, ceq;
121   __asm__("vdup.8 %q0, %1" : "=w"(bc) : "r"(static_cast<u32>(c)));
122   __asm__("vld1.8 {%q0}, [%1]" : "=w"(v) : "r"(p), "m"(*reinterpret_cast<const u8(*)[16]>(p)));
123   __asm__("vceq.i8 %q0, %q1, %q2" : "=w"(ceq) : "w"(v), "w"(bc));
124   uint8x8_t narrowed;
125   __asm__("vshrn.i16 %P0, %q1, #4" : "=w"(narrowed) : "w"(ceq));
126   u64 m;
127   __asm__("vmov %Q0, %R0, %P1" : "=r"(m) : "w"(narrowed));
128   return m;
129 }
