101 template<typename V>
102 [[gnu::always_inline]] static inline unsigned long long
103 __clang_compare_mask(V a, V b, int predicate) noexcept
104 {
105   unsigned long long mask = 0;
106   constexpr unsigned n = sizeof(V) / sizeof(a[0]);
107   for ( unsigned i = 0; i < n; ++i ) {
108     const bool unord = __builtin_isunordered(a[i], b[i]);
109     bool match = false;
110     switch ( predicate & 31 ) {
111     case 0:
112       match = !unord and a[i] == b[i];
113       break;
114     case 1:
115       match = !unord and a[i] < b[i];
116       break;
117     case 2:
118       match = !unord and a[i] <= b[i];
119       break;
120     case 3:
121       match = unord;
122       break;
123     case 4:
124       match = unord or a[i] != b[i];
125       break;
126     case 5:
127       match = unord or !(a[i] < b[i]);
128       break;
129     case 6:
130       match = unord or !(a[i] <= b[i]);
131       break;
132     case 7:
133       match = !unord;
134       break;
135     case 8:
136       match = unord or a[i] == b[i];
137       break;
138     case 9:
139       match = unord or !(a[i] >= b[i]);
140       break;
141     case 10:
142       match = unord or !(a[i] > b[i]);
143       break;
144     case 11:
145       match = false;
146       break;
147     case 12:
148       match = !unord and a[i] != b[i];
149       break;
150     case 13:
151       match = !unord and a[i] >= b[i];
152       break;
153     case 14:
154       match = !unord and a[i] > b[i];
155       break;
156     case 15:
157       match = true;
158       break;
159     default:
160       match = __clang_compare_mask(a, b, predicate & 15) & (1ull << i);
161       break;
162     }
163     if ( match ) mask |= 1ull << i;
164   }
165   return mask;
166 }

171     [[gnu::always_inline]] static inline V
172     __clang_unpack(V a, V b) noexcept
173 {
174   V out{};
175   constexpr unsigned lanes = sizeof(V) / sizeof(a[0]);
176   constexpr unsigned block_lanes = 16 / sizeof(a[0]);
177   constexpr unsigned half = block_lanes / 2;
178   for ( unsigned block = 0; block < lanes; block += block_lanes ) {
179     const unsigned base = block + (High ? half : 0);
180     for ( unsigned i = 0; i < half; ++i ) {
181       out[block + i * 2] = a[base + i];
182       out[block + i * 2 + 1] = b[base + i];
183     }
184   }
185   return out;
186 }

283 [[gnu::always_inline]] static inline V
284 __clang_gather(V src, Base base, I indexes, M mask, int scale) noexcept
285 {
286   constexpr unsigned n = sizeof(V) / sizeof(src[0]);
287   const char *bytes = (const char *)base;
288   for ( unsigned i = 0; i < n; ++i ) {
289     if ( mask[i] != 0 ) {
290       __typeof__(src[0]) value;
291       __builtin_memcpy(&value, bytes + indexes[i] * scale, sizeof(value));
292       src[i] = value;
293     }
294   }
295   return src;
296 }

428 template<typename Out, typename In>
429 [[gnu::always_inline]] static inline Out
430 __clang_pack(In a, In b, long long low, unsigned long long high) noexcept
431 {
432   Out out{};
433   const long long high_value = static_cast<long long>(high);
434   constexpr unsigned blocks = sizeof(In) / 16;
435   constexpr unsigned in_lanes = 16 / sizeof(a[0]);
436   constexpr unsigned out_lanes = 16 / sizeof(out[0]);
437   for ( unsigned block = 0; block < blocks; ++block ) {
438     for ( unsigned i = 0; i < in_lanes; ++i ) {
439       long long av = a[block * in_lanes + i];
440       long long bv = b[block * in_lanes + i];
441       if ( av < low ) av = low;
442       if ( bv < low ) bv = low;
443       if ( av > high_value ) av = high_value;
444       if ( bv > high_value ) bv = high_value;
445       out[block * out_lanes + i] = static_cast<__typeof__(out[0])>(av);
446       out[block * out_lanes + in_lanes + i] = static_cast<__typeof__(out[0])>(bv);
447     }
448   }
449   return out;
450 }
