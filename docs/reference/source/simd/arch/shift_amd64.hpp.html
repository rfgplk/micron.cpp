31     template<int imm8, is_simd_class T>
32     inline T byte_shift_left(T &o)
33 {
34   if constexpr ( micron::is_same_v<typename T::bit_width, i128> ) {
35     return _mm_bslli_si128(o, imm8);
36   } else if constexpr ( micron::is_same_v<typename T::bit_width, i256> ) {
37 
38     return _mm256_alignr_epi8(o, _mm256_setzero_si256(), 16 - imm8);
39   } else if constexpr ( micron::is_same_v<typename T::bit_width, i512> ) {
40     static_assert(!sizeof(T), "512-bit byte_shift_left has no micron intrinsic");
41   } else {
42     static_assert(!sizeof(T), "unsupported simd lane width");
43   }
44 }
