498 {
499   int32_t idx[4], msrc[4], mk[4], out[4];
500   vst1q_s32(idx, vindex);
501   vst1q_s32(msrc, src);
502   vst1q_s32(mk, mask);

505       __builtin_memcpy(&out[i], __neon_scale_ptr<scale>(base, idx[i]), 4);

507       out[i] = msrc[i];
508   }
509   return vld1q_s32(out);

671 inline d128
672 maskgather_pd_i64(d128 src, double *base, i128 vindex, d128 mask)
673 {
674   i64 idx[2];
675   double msrc[2], out[2];
676   i64 mk[2];
677   vst1q_s64(idx, vreinterpretq_s64_s32(vindex));
678   vst1q_f64(msrc, src);
679   vst1q_s64(mk, vreinterpretq_s64_f64(mask));
680   for ( int i = 0; i < 2; ++i ) {
681     if ( mk[i] < 0 )
682       __builtin_memcpy(&out[i], __neon_scale_ptr<scale>(base, idx[i]), 8);
683     else
684       out[i] = msrc[i];
685   }
686   return vld1q_f64(out);
687 }
