251   inline B
252   maskload_32(int *ptr, B mask)
253   {
254     static_assert(micron::is_same_v<B, i128>, "maskload_32 mask/result must be i128");
255     const i128 loaded = vld1q_s32(reinterpret_cast<const int32_t *>(ptr));
256     const i128 smask = vshrq_n_s32(mask, 31);
257     return vandq_s32(loaded, smask);
258   }

407   template<is_simd_type B, int scale>
408   inline i128
409   maskgather_32_i32(i128 src, int *base, i128 vindex, i128 mask)
410   {
411     int32_t idx[4], msrc[4], mk[4], out[4];
412     vst1q_s32(idx, vindex);
413     vst1q_s32(msrc, src);
414     vst1q_s32(mk, mask);
415     for ( int i = 0; i < 4; ++i ) {
416       if ( mk[i] < 0 )
417         __builtin_memcpy(&out[i], __neon_scale_ptr<scale>(base, idx[i]), 4);
418       else
419         out[i] = msrc[i];
420     }
421     return vld1q_s32(out);
422   }
