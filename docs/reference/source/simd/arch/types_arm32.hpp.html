95   inline int
96   neon_movemask_f32(float32x4_t v) noexcept
97   {
98     const uint32x4_t m = vshrq_n_u32(vreinterpretq_u32_f32(v), 31);
99     return static_cast<int>(vgetq_lane_u32(m, 0) | (vgetq_lane_u32(m, 1) << 1) | (vgetq_lane_u32(m, 2) << 2) | (vgetq_lane_u32(m, 3) << 3));

136   inline float32x4_t
137   neon_v7_divq_f32(float32x4_t a, float32x4_t b) noexcept
138   {
139     float32x4_t recip = vrecpeq_f32(b);
140     recip = vmulq_f32(vrecpsq_f32(b, recip), recip);
141     recip = vmulq_f32(vrecpsq_f32(b, recip), recip);
142     return vmulq_f32(a, recip);

153   inline uint64x2_t
154   neon_v7_cgtq_s64(int64x2_t a, int64x2_t b) noexcept
155   {
156 
157     uint32x4_t hi_gt = vcgtq_s32(vreinterpretq_s32_s64(a), vreinterpretq_s32_s64(b));
158     uint32x4_t hi_eq = vceqq_s32(vreinterpretq_s32_s64(a), vreinterpretq_s32_s64(b));
159     uint32x4_t lo_gt = vcgtq_u32(vreinterpretq_u32_s64(a), vreinterpretq_u32_s64(b));
160 
161     uint32x4_t hi_gt_r = vrev64q_u32(hi_gt);
162     uint32x4_t hi_eq_r = vrev64q_u32(hi_eq);
163 
164     uint32x4_t r = vorrq_u32(hi_gt_r, vandq_u32(hi_eq_r, lo_gt));
165 
166     uint32x4x2_t z = vzipq_u32(r, r);
167     return vreinterpretq_u64_u32(vcombine_u32(vget_low_u32(z.val[0]), vget_low_u32(z.val[1])));
168   }

199   __attribute__((always_inline)) static inline uint16x8_t
200   __expand_mask_u16(uint8_t k) noexcept
201   {
202     static const uint16_t bp[8] = { 1, 2, 4, 8, 16, 32, 64, 128 };
203     uint16x8_t m = vdupq_n_u16(k), p = vld1q_u16(bp);
204     return vceqq_u16(vandq_u16(m, p), p);
205   }
206 
207   __attribute__((always_inline)) static inline uint8x16_t
208   __expand_mask_u8(uint16_t k) noexcept
209   {
210     static const uint8_t bp[16] = { 1, 2, 4, 8, 16, 32, 64, 128, 1, 2, 4, 8, 16, 32, 64, 128 };
211 
212     uint8x16_t sel = vld1q_u8(bp);
213     uint8x8_t lo = vdup_n_u8(static_cast<uint8_t>(k & 0xFFu));
214     uint8x8_t hi = vdup_n_u8(static_cast<uint8_t>((k >> 8) & 0xFFu));
215     uint8x16_t m = vcombine_u8(lo, hi);
216     return vceqq_u8(vandq_u8(m, sel), sel);
217   }

219   __attribute__((always_inline)) static inline uint32x4_t
220   __expand_mask_u32(uint8_t k) noexcept
221   {
222     static const uint32_t bp[4] = { 1, 2, 4, 8 };
223     uint32x4_t m = vdupq_n_u32(k), p = vld1q_u32(bp);
224     return vceqq_u32(vandq_u32(m, p), p);
225   }

227   __attribute__((always_inline)) static inline uint64x2_t
228   __expand_mask_u64(uint8_t k) noexcept
229   {
230     uint64_t lanes[2];
231     lanes[0] = (k & 1) ? ~static_cast<uint64_t>(0) : 0;
232     lanes[1] = (k & 2) ? ~static_cast<uint64_t>(0) : 0;
233     return vld1q_u64(lanes);
234   }
