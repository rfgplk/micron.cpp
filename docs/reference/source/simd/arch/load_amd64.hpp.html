17     // load/aligned loads
18     template<is_simd_type B>
19     inline B load(B *ptr)
20 {
21   if constexpr ( micron::is_same_v<B, i128> ) {
22     return _mm_load_si128(static_cast<i128 *>(__builtin_assume_aligned(ptr, 16)));
23   } else if constexpr ( micron::is_same_v<B, f128> ) {
24     return _mm_load_ps(reinterpret_cast<float *>(__builtin_assume_aligned(ptr, 16)));
25   } else if constexpr ( micron::is_same_v<B, d128> ) {
26     return _mm_load_pd(reinterpret_cast<double *>(__builtin_assume_aligned(ptr, 16)));
27   } else if constexpr ( micron::is_same_v<B, i256> ) {
28     return _mm256_load_si256(static_cast<i256 *>(__builtin_assume_aligned(ptr, 32)));
29   } else if constexpr ( micron::is_same_v<B, f256> ) {
30     return _mm256_load_ps(reinterpret_cast<float *>(__builtin_assume_aligned(ptr, 32)));
31   } else if constexpr ( micron::is_same_v<B, d256> ) {
32     return _mm256_load_pd(reinterpret_cast<double *>(__builtin_assume_aligned(ptr, 32)));
33   } else
34     static_assert(!sizeof(B), "load<B>: unsupported simd width (b64/512-bit have no aligned-load path)");
35 }
