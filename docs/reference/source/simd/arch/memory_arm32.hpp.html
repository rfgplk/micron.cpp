32 template<typename T>
33 __attribute__((nonnull)) T *
34 memcpy128(T *__restrict dest, const T *__restrict src, const u64 count) noexcept
35 {
36   static_assert(micron::is_trivially_copyable_v<T>, "memcpy128 requires trivially copyable type");
37 
38   auto *__restrict d = reinterpret_cast<u8 *>(dest);
39   const auto *__restrict s = reinterpret_cast<const u8 *>(src);
40   const u64 bytes = count * sizeof(T);
41   const u64 n = bytes / 16;
42 
43   for ( u64 i = 0; i < n; i++ ) __bits::__block_copy_16(d + i * 16, s + i * 16);
44 
45   const u64 rem = bytes % 16;
46   if ( rem ) {
47     u8 tmp[16] = {};
48     for ( u64 i = 0; i < rem; i++ ) tmp[i] = s[n * 16 + i];
49     for ( u64 i = 0; i < rem; i++ ) d[n * 16 + i] = tmp[i];
50   }
51 
52   return dest;
53 }

114 template<typename T>
115 __attribute__((nonnull)) T *
116 memmove128(T *dest, const T *src, const u64 count) noexcept
117 {
118   static_assert(micron::is_trivially_copyable_v<T>, "memmove128 requires trivially copyable type");
119 
120   auto *d = reinterpret_cast<u8 *>(dest);
121   const auto *s = reinterpret_cast<const u8 *>(src);
122   const u64 bytes = count * sizeof(T);
123 
124   if ( d == s || bytes == 0 ) return dest;
125 
126   if ( d < s || d >= s + bytes ) {
127     const u64 n = bytes / 16;
128     for ( u64 i = 0; i < n; i++ ) __bits::__block_copy_16(d + i * 16, s + i * 16);
129     const u64 rem = bytes % 16;
130     if ( rem ) {
131       for ( u64 i = 0; i < rem; i++ ) d[n * 16 + i] = s[n * 16 + i];
132     }
133   } else {
134     // overlapping backward path: use __block_move_16 (no __restrict)
135     const u64 n = bytes / 16;
136     const u64 rem = bytes % 16;
137     if ( rem ) {
138       for ( u64 i = rem; i > 0; i-- ) d[n * 16 + i - 1] = s[n * 16 + i - 1];
139     }
140     for ( u64 i = n; i > 0; i-- ) __bits::__block_move_16(d + (i - 1) * 16, s + (i - 1) * 16);
141   }
142 
143   return dest;
144 }

238 template<typename T>
239 __attribute__((nonnull)) i64
240 memcmp128(const T *__restrict src, const T *__restrict dest, const u64 count) noexcept
241 {
242   const auto *s = reinterpret_cast<const u8 *>(src);
243   const auto *d = reinterpret_cast<const u8 *>(dest);
244   const u64 bytes = count * sizeof(T);
245   const u64 n = bytes / 16;
246 
247   for ( u64 i = 0; i < n; i++ ) {
248     uint8x16_t ceq = __bits::__block_cmpeq_16(s + i * 16, d + i * 16);
249     if ( !__neon_v7_all_eq(ceq) ) {
250       const u64 base = i * 16;
251       const u64 limit = (base + 16 < bytes) ? base + 16 : bytes;
252       for ( u64 j = base; j < limit; j++ )
253         if ( s[j] != d[j] ) return static_cast<i64>(static_cast<unsigned>(s[j])) - static_cast<i64>(static_cast<unsigned>(d[j]));
254     }
255   }
256 
257   const u64 rem = bytes % 16;
258   if ( rem ) {
259     const u64 base = n * 16;
260     for ( u64 i = 0; i < rem; i++ )
261       if ( s[base + i] != d[base + i] )
262         return static_cast<i64>(static_cast<unsigned>(s[base + i])) - static_cast<i64>(static_cast<unsigned>(d[base + i]));
263   }
264 
265   return 0;
266 }

320 __attribute__((nonnull)) T *
321 memchr128(const T *src, u8 c, const u64 count) noexcept
322 {
323   const auto *p = reinterpret_cast<const u8 *>(src);
324   const u64 bytes = count * sizeof(T);
325   const u64 n = bytes / 16;
326 
327   for ( u64 i = 0; i < n; i++ ) {
328     const u64 m = __bits::__block_eq_mask_16(p + i * 16, c);
329     if ( m ) return const_cast<T *>(reinterpret_cast<const T *>(p + i * 16 + (__builtin_ctzll(m) >> 2)));
330   }
331 
332   const u64 base = n * 16;
333   const u64 rem = bytes % 16;
334   for ( u64 i = 0; i < rem; i++ )
335     if ( p[base + i] == c ) return const_cast<T *>(reinterpret_cast<const T *>(p + base + i));
336 
337   return nullptr;

343 template<typename T>
344 __attribute__((nonnull)) T *
345 memrchr128(const T *src, u8 c, const u64 count) noexcept
346 {
347   const auto *p = reinterpret_cast<const u8 *>(src);
348   const u64 bytes = count * sizeof(T);
349   const u64 n = bytes / 16;
350   const u64 rem = bytes % 16;
351   const u64 base = n * 16;
352 
353   for ( u64 i = rem; i > 0; i-- )
354     if ( p[base + i - 1] == c ) return const_cast<T *>(reinterpret_cast<const T *>(p + base + i - 1));
355 
356   for ( u64 i = n; i > 0; i-- ) {
357     const u64 m = __bits::__block_eq_mask_16(p + (i - 1) * 16, c);
358     if ( m ) {
359       const u32 hi = (63u - __builtin_clzll(m)) >> 2;
360       return const_cast<T *>(reinterpret_cast<const T *>(p + (i - 1) * 16 + hi));
361     }
362   }
363 
364   return nullptr;
365 }

395 // memmem - substring search

397 template<typename T>
398 T *
399 memmem128(const T *hay, const u64 hlen, const T *nee, const u64 nlen) noexcept
400 {
401   const auto *h = reinterpret_cast<const u8 *>(hay);
402   const auto *ne = reinterpret_cast<const u8 *>(nee);
403   const u64 hbytes = hlen * sizeof(T);
404   const u64 nbytes = nlen * sizeof(T);
405 
406   if ( nbytes == 0 ) return const_cast<T *>(hay);
407   if ( nbytes > hbytes ) return nullptr;
408 
409   const u64 limit = hbytes - nbytes + 1;
410   const u8 first = ne[0];
411 
412   u64 i = 0;
413   while ( i + 16 <= limit ) {
414     const u64 m = __bits::__block_eq_mask_16(h + i, first);
415     if ( m ) {
416       u64 mm = m;
417       while ( mm ) {
418         const u32 off = __builtin_ctzll(mm) >> 2;
419         const u64 cand = i + off;
420         if ( cand >= limit ) return nullptr;
421         if ( nbytes == 1 ) return const_cast<T *>(reinterpret_cast<const T *>(h + cand));
422         bool match = true;
423         for ( u64 j = 1; j < nbytes; j++ )
424           if ( h[cand + j] != ne[j] ) {
425             match = false;
426             break;
427           }
428         if ( match ) return const_cast<T *>(reinterpret_cast<const T *>(h + cand));
429         mm &= ~(0xFull << (__builtin_ctzll(mm) & ~3u));
430       }
431     }
432     i += 16;
433   }
434 
435   for ( ; i < limit; i++ ) {
436     if ( h[i] != first ) continue;
437     bool match = true;
438     for ( u64 j = 1; j < nbytes; j++ )
439       if ( h[i + j] != ne[j] ) {
440         match = false;
441         break;
442       }
443     if ( match ) return const_cast<T *>(reinterpret_cast<const T *>(h + i));
444   }
445 
446   return nullptr;
447 }

449 __attribute__((nonnull)) inline u8 *
450 __memset_bulk(u8 *__restrict d, const u8 v, const u64 n) noexcept
451 {
452   const uint8x16_t vv = __bits::__broadcast_byte_16(v);
453   __bits::__block_set_16(d, vv);
454   u8 *p = reinterpret_cast<u8 *>((reinterpret_cast<uintptr_t>(d) + 16) & ~static_cast<uintptr_t>(15));
455   u8 *const e = d + n;
456   for ( ; p + 64 <= e; p += 64 ) {
457     __bits::__block_set_16(p, vv);
458     __bits::__block_set_16(p + 16, vv);
459     __bits::__block_set_16(p + 32, vv);
460     __bits::__block_set_16(p + 48, vv);
461   }
462   __bits::__block_set_16(e - 64, vv);
463   __bits::__block_set_16(e - 48, vv);
464   __bits::__block_set_16(e - 32, vv);
465   __bits::__block_set_16(e - 16, vv);
466   return d;
467 }

469 __attribute__((nonnull)) inline u8 *
470 __wordset_bulk(u8 *__restrict d, const u64 w, const u64 n) noexcept
471 {
472   // WARNING: n must be > 64 and n % period[8] == 0, not guarding behind branch
473   // dst is aligned for the bulk loop by rotating the pattern to the aligned
474   // phase
475   const uint8x16_t vv = __bits::__broadcast_word_16(w);
476   if ( n <= 256 ) {
477     u64 i = 0;
478     for ( ; i + 16 <= n; i += 16 ) __bits::__block_set_16(d + i, vv);
479     if ( i < n ) __bits::__block_set_16(d + n - 16, vv);
480     return d;
481   }
482   __bits::__block_set_16(d, vv);
483   u8 *p = reinterpret_cast<u8 *>((reinterpret_cast<uintptr_t>(d) + 16) & ~static_cast<uintptr_t>(15));
484   const u32 rot = (static_cast<u32>(p - d) & 7u) * 8u;
485   const u64 w2 = rot ? ((w >> rot) | (w << (64u - rot))) : w;
486   const uint8x16_t vv2 = __bits::__broadcast_word_16(w2);
487   u8 *const e = d + n;
488   for ( ; p + 64 <= e; p += 64 ) {
489     __bits::__block_set_16_a(p, vv2);
490     __bits::__block_set_16_a(p + 16, vv2);
491     __bits::__block_set_16_a(p + 32, vv2);
492     __bits::__block_set_16_a(p + 48, vv2);
493   }
494   for ( ; p + 16 <= e; p += 16 ) __bits::__block_set_16_a(p, vv2);
495   if ( p < e ) __bits::__block_set_16(e - 16, vv);
496   return d;
497 }

499 __attribute__((nonnull)) inline u8 *
500 __memcpy_bulk(u8 *__restrict d, const u8 *__restrict s, const u64 n) noexcept
501 {
502   // NOTE: batch all 4 loads before any store
503   // in order cores (ones we care about) an interleaved vld1/vst1 pairing stalls on NEON load to store latency every 16 B block
504   const __ml::__v16 h = __ml::__ld16(s);
505   const __ml::__v16 t0 = __ml::__ld16(s + n - 64);
506   const __ml::__v16 t1 = __ml::__ld16(s + n - 48);
507   const __ml::__v16 t2 = __ml::__ld16(s + n - 32);
508   const __ml::__v16 t3 = __ml::__ld16(s + n - 16);
509   __ml::__st16(d, h);
510   const u64 off = 16 - (reinterpret_cast<uintptr_t>(d) & 15);
511   u8 *pd = d + off;
512   const u8 *ps = s + off;
513   u64 rem = n - off;
514   for ( ; rem >= 64; rem -= 64, pd += 64, ps += 64 ) {
515     __bits::__pld(ps + 256);
516     __ml::__v16 a = __ml::__ld16(ps);
517     __ml::__v16 b = __ml::__ld16(ps + 16);
518     __ml::__v16 c = __ml::__ld16(ps + 32);
519     __ml::__v16 e = __ml::__ld16(ps + 48);
520     __ml::__pin4(a, b, c, e);
521     __ml::__st16(pd, a);
522     __ml::__st16(pd + 16, b);
523     __ml::__st16(pd + 32, c);
524     __ml::__st16(pd + 48, e);
525   }
526   __ml::__st16(d + n - 64, t0);
527   __ml::__st16(d + n - 48, t1);
528   __ml::__st16(d + n - 32, t2);
529   __ml::__st16(d + n - 16, t3);
530   return d;
531 }

533 __attribute__((nonnull)) inline u8 *
534 __memmove_bulk_fwd(u8 *d, const u8 *s, const u64 n) noexcept
535 {
536   const __ml::__v16 h = __ml::__ld16(s);
537   const __ml::__v16 t0 = __ml::__ld16(s + n - 64);
538   const __ml::__v16 t1 = __ml::__ld16(s + n - 48);
539   const __ml::__v16 t2 = __ml::__ld16(s + n - 32);
540   const __ml::__v16 t3 = __ml::__ld16(s + n - 16);
541   const u64 off = 16 - (reinterpret_cast<uintptr_t>(d) & 15);
542   u8 *pd = d + off;
543   const u8 *ps = s + off;
544   u64 rem = n - off;
545   for ( ; rem >= 64; rem -= 64, pd += 64, ps += 64 ) {
546     __bits::__pld(ps + 256);
547     __ml::__v16 a = __ml::__ld16(ps);
548     __ml::__v16 b = __ml::__ld16(ps + 16);
549     __ml::__v16 c = __ml::__ld16(ps + 32);
550     __ml::__v16 e = __ml::__ld16(ps + 48);
551     __ml::__pin4(a, b, c, e);
552     __ml::__st16(pd, a);
553     __ml::__st16(pd + 16, b);
554     __ml::__st16(pd + 32, c);
555     __ml::__st16(pd + 48, e);
556   }
557   __ml::__st16(d + n - 64, t0);
558   __ml::__st16(d + n - 48, t1);
559   __ml::__st16(d + n - 32, t2);
560   __ml::__st16(d + n - 16, t3);
561   __ml::__st16(d, h);
562   return d;
563 }

565 __attribute__((nonnull)) inline u8 *
566 __memmove_bulk_bwd(u8 *d, const u8 *s, const u64 n) noexcept
567 {
568   const __ml::__v16 t = __ml::__ld16(s + n - 16);
569   const __ml::__v16 h0 = __ml::__ld16(s);
570   const __ml::__v16 h1 = __ml::__ld16(s + 16);
571   const __ml::__v16 h2 = __ml::__ld16(s + 32);
572   const __ml::__v16 h3 = __ml::__ld16(s + 48);
573   u8 *pe = reinterpret_cast<u8 *>(reinterpret_cast<uintptr_t>(d + n) & ~static_cast<uintptr_t>(15));
574   const u8 *pse = s + (pe - d);
575   while ( static_cast<u64>(pe - d) > 64 ) {
576     pe -= 64;
577     pse -= 64;
578     __ml::__v16 a = __ml::__ld16(pse + 48);
579     __ml::__v16 b = __ml::__ld16(pse + 32);
580     __ml::__v16 c = __ml::__ld16(pse + 16);
581     __ml::__v16 e = __ml::__ld16(pse);
582     __ml::__pin4(a, b, c, e);
583     __ml::__st16(pe + 48, a);
584     __ml::__st16(pe + 32, b);
585     __ml::__st16(pe + 16, c);
586     __ml::__st16(pe, e);
587   }
588   __ml::__st16(d, h0);
589   __ml::__st16(d + 16, h1);
590   __ml::__st16(d + 32, h2);
591   __ml::__st16(d + 48, h3);
592   __ml::__st16(d + n - 16, t);
593   return d;
594 }
