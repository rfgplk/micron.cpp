52 __attribute__((always_inline)) static inline int32x4_t
53 __neon_bslli(int32x4_t v, int n) noexcept
54 {
55   if ( n <= 0 ) return v;
56   if ( n >= 16 ) return vdupq_n_s32(0);
57   uint8x16_t a = vreinterpretq_u8_s32(v), z = vdupq_n_u8(0);
58   switch ( n ) {
59   case 1:
60     return vreinterpretq_s32_u8(vextq_u8(z, a, 15));
61   case 2:
62     return vreinterpretq_s32_u8(vextq_u8(z, a, 14));
63   case 3:
64     return vreinterpretq_s32_u8(vextq_u8(z, a, 13));
65   case 4:
66     return vreinterpretq_s32_u8(vextq_u8(z, a, 12));
67   case 5:
68     return vreinterpretq_s32_u8(vextq_u8(z, a, 11));
69   case 6:
70     return vreinterpretq_s32_u8(vextq_u8(z, a, 10));
71   case 7:
72     return vreinterpretq_s32_u8(vextq_u8(z, a, 9));
73   case 8:
74     return vreinterpretq_s32_u8(vextq_u8(z, a, 8));
75   case 9:
76     return vreinterpretq_s32_u8(vextq_u8(z, a, 7));
77   case 10:
78     return vreinterpretq_s32_u8(vextq_u8(z, a, 6));
79   case 11:
80     return vreinterpretq_s32_u8(vextq_u8(z, a, 5));
81   case 12:
82     return vreinterpretq_s32_u8(vextq_u8(z, a, 4));
83   case 13:
84     return vreinterpretq_s32_u8(vextq_u8(z, a, 3));
85   case 14:
86     return vreinterpretq_s32_u8(vextq_u8(z, a, 2));
87   case 15:
88     return vreinterpretq_s32_u8(vextq_u8(z, a, 1));
89   default:
90     return vdupq_n_s32(0);
91   }
92 }
