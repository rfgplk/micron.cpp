39   inline unsigned long long
40   __read_xcr0(void) noexcept
41   {
42     unsigned eax, edx;
43     __asm__ volatile(".byte 0x0f, 0x01, 0xd0" : "=a"(eax), "=d"(edx) : "c"(0u));
44     return ((unsigned long long)edx << 32) | eax;
45   }

47   inline __simd_flags
48   __get_runtime_features(void)
49   {
50     __simd_flags flags;
51     processor_t cpu;
52     info(&cpu);
53     flags.fma = cpu.features.fma;
54     flags.fma3 = cpu.features.fma3;
55     flags.fma4 = cpu.features.fma4;
56     flags.sse = cpu.features.sse;
57     flags.sse2 = cpu.features.sse2;
58     flags.sse3 = cpu.features.sse3;
59     flags.ssse3 = cpu.features.ssse3;
60     flags.sse4_1 = cpu.features.sse4_1;
61     flags.sse4_2 = cpu.features.sse4_2;
62     flags.avx = cpu.features.avx;
63     flags.avx2 = cpu.features.avx2;
64     flags.osxsave = cpu.features.osxsave;
65     return flags;
66   }

68   // true iff the OS has enabled XSAVE (OSXSAVE) and XCR0 advertises SSE+AVX (YMM) state save (bits 1,2)
69   inline bool
70   __os_avx_enabled(const __simd_flags &flags)
71   {
72     return flags.osxsave and ((__read_xcr0() & 0x6ull) == 0x6ull);
73   }
