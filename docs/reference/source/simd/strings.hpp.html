13 inline usize
14 sse_strlen(const char *str)
15 {
16 #if defined(__micron_x86_sse2)
17   const char *p = str;
18   i128 zero = _mm_setzero_si128();
19 
20   while ( reinterpret_cast<uintptr_t>(p) & 0xF ) {
21     if ( *p == '\0' ) return static_cast<usize>(p - str);
22     ++p;
23   }
24 
25   for ( ;; ) {
26     i128 chunk = _mm_load_si128(reinterpret_cast<const i128 *>(p));
27     i128 cmp = _mm_cmpeq_epi8(chunk, zero);
28     int mask = _mm_movemask_epi8(cmp);
29     if ( mask != 0 ) {
30       return static_cast<usize>((p - str) + __builtin_ctz(static_cast<unsigned int>(mask)));
31     }
32     p += 16;
33   }
34 #else
35   const char *p = str;
36   while ( *p ) ++p;
37   return static_cast<usize>(p - str);
38 #endif
39 }

351 find_first_elem(const T *p, usize len, T ch) noexcept
352 {
353   usize i = 0;
354 #if defined(__micron_scan_lanes)
355   if constexpr ( __lane_ok<T> ) {
356     constexpr usize EPV = __lane_elems<T>;
357     const auto needle = __lane_bcast(ch);

542 template<typename T>
543 [[gnu::always_inline]] static inline usize
544 find_substr_elem(const T *h, usize hlen, const T *n, usize nlen) noexcept
545 {
546   if ( nlen == 0 ) return 0;
547   if ( nlen > hlen ) return hlen;
548   const T first = n[0];
549   usize i = 0;
550   while ( i + nlen <= hlen ) {
551     usize f = find_first_elem(h + i, hlen - i, first);
552     if ( f == hlen - i ) return hlen;
553     i += f;
554     if ( i + nlen > hlen ) return hlen;
555     usize j = 1;
556     for ( ; j < nlen; ++j )
557       if ( h[i + j] != n[j] ) break;
558     if ( j == nlen ) return i;
559     ++i;
560   }
561   return hlen;
562 }

578 template<typename T>
579 [[gnu::always_inline]] static inline void
580 to_lower_elem(T *p, usize len) noexcept
581 {
582   usize i = 0;
583 #if defined(__micron_x86_avx2)
584   if constexpr ( sizeof(T) == 1 || sizeof(T) == 2 || sizeof(T) == 4 ) {
585     constexpr usize EPV = 32 / sizeof(T);
586     for ( ; i + EPV <= len; i += EPV ) {
587       __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(p + i));
588       __m256i ge, le, delta, mask;
589       if constexpr ( sizeof(T) == 1 ) {
590         ge = _mm256_cmpgt_epi8(v, _mm256_set1_epi8('A' - 1));
591         le = _mm256_cmpgt_epi8(_mm256_set1_epi8('Z' + 1), v);
592         mask = _mm256_and_si256(ge, le);
593         delta = _mm256_and_si256(mask, _mm256_set1_epi8(0x20));
594       } else if constexpr ( sizeof(T) == 2 ) {
595         ge = _mm256_cmpgt_epi16(v, _mm256_set1_epi16('A' - 1));
596         le = _mm256_cmpgt_epi16(_mm256_set1_epi16('Z' + 1), v);
597         mask = _mm256_and_si256(ge, le);
598         delta = _mm256_and_si256(mask, _mm256_set1_epi16(0x20));
599       } else {
600         ge = _mm256_cmpgt_epi32(v, _mm256_set1_epi32('A' - 1));
601         le = _mm256_cmpgt_epi32(_mm256_set1_epi32('Z' + 1), v);
602         mask = _mm256_and_si256(ge, le);
603         delta = _mm256_and_si256(mask, _mm256_set1_epi32(0x20));
604       }
605       _mm256_storeu_si256(reinterpret_cast<__m256i *>(p + i), _mm256_add_epi8(v, delta));
606     }
607   }
608 #endif
609   for ( ; i < len; ++i ) {
610     auto c = static_cast<unsigned long>(p[i]);
611     if ( c >= 'A' && c <= 'Z' ) p[i] = static_cast<T>(c + 0x20);
612   }
613 }

628         le = _mm256_cmpgt_epi8(_mm256_set1_epi8('z' + 1), v);
629         mask = _mm256_and_si256(ge, le);
630         delta = _mm256_and_si256(mask, _mm256_set1_epi8(0x20));
631       } else if constexpr ( sizeof(T) == 2 ) {
632         ge = _mm256_cmpgt_epi16(v, _mm256_set1_epi16('a' - 1));
633         le = _mm256_cmpgt_epi16(_mm256_set1_epi16('z' + 1), v);
634         mask = _mm256_and_si256(ge, le);
635         delta = _mm256_and_si256(mask, _mm256_set1_epi16(0x20));
636       } else {
637         ge = _mm256_cmpgt_epi32(v, _mm256_set1_epi32('a' - 1));
638         le = _mm256_cmpgt_epi32(_mm256_set1_epi32('z' + 1), v);
639         mask = _mm256_and_si256(ge, le);
640         delta = _mm256_and_si256(mask, _mm256_set1_epi32(0x20));
641       }
642       _mm256_storeu_si256(reinterpret_cast<__m256i *>(p + i), _mm256_sub_epi8(v, delta));
643     }
644   }
645 #endif
646   for ( ; i < len; ++i ) {
647     auto c = static_cast<unsigned long>(p[i]);
648     if ( c >= 'a' && c <= 'z' ) p[i] = static_cast<T>(c - 0x20);
649   }
650 }

652 template<typename T>
653 [[gnu::always_inline]] static inline void
654 reverse_elem(T *p, usize len) noexcept
655 {
656   for ( usize a = 0, b = (len ? len - 1 : 0); a < b; ++a, --b ) {
657     T t = p[a];
658     p[a] = p[b];
659     p[b] = t;
660   }
661 }

663 enum class __byte_op { __xor, __and, __or };
664 
665 template<__byte_op Op>
666 [[gnu::always_inline]] static inline byte
667 __apply_byte_op(byte x, byte y) noexcept
668 {
669   if constexpr ( Op == __byte_op::__xor )
670     return static_cast<byte>(x ^ y);
671   else if constexpr ( Op == __byte_op::__and )
672     return static_cast<byte>(x & y);
673   else
674     return static_cast<byte>(x | y);
675 }
676 
677 template<__byte_op Op>
678 [[gnu::always_inline]] static inline void
679 __bytes_cycle(byte *dst, const byte *a, usize an, const byte *b, usize bn) noexcept
680 {
681   if ( bn == 0 ) {
682     for ( usize i = 0; i < an; ++i ) dst[i] = a[i];
683     return;
684   }
685   usize i = 0;
686 #if defined(__micron_x86_avx2)
687   if ( bn >= an ) {
688     for ( ; i + 32 <= an; i += 32 ) {
689       __m256i va = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(a + i));
690       __m256i vb = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(b + i));
691       __m256i vr;
692       if constexpr ( Op == __byte_op::__xor )
693         vr = _mm256_xor_si256(va, vb);
694       else if constexpr ( Op == __byte_op::__and )
695         vr = _mm256_and_si256(va, vb);
696       else
697         vr = _mm256_or_si256(va, vb);
698       _mm256_storeu_si256(reinterpret_cast<__m256i *>(dst + i), vr);
699     }
700   }
701 #endif
702   for ( ; i < an; ++i ) dst[i] = __apply_byte_op<Op>(a[i], b[i % bn]);
703 }

705 [[gnu::always_inline]] static inline void
706 xor_bytes_cycle(byte *dst, const byte *a, usize an, const byte *b, usize bn) noexcept
707 {
708   __bytes_cycle<__byte_op::__xor>(dst, a, an, b, bn);
709 }

712 and_bytes_cycle(byte *dst, const byte *a, usize an, const byte *b, usize bn) noexcept

718 or_bytes_cycle(byte *dst, const byte *a, usize an, const byte *b, usize bn) noexcept
