1732 __inline_sse void
1733 decimal_digits_16(char *out, u64 value) noexcept
1734 {
1735   constexpr u64 hundred_million = 100000000ull;
1736   constexpr u32 div10k_sig = static_cast<u32>((1ull << 40) / 10000 + 1);
1737   constexpr u32 neg10k = static_cast<u32>((1ull << 32) - 10000);
1738   constexpr u32 div100_sig = (1u << 19) / 100 + 1;
1739   constexpr u16 div10_sig = (1u << 16) / 10 + 1;
1740 
1741   const u32 hi = static_cast<u32>(value / hundred_million);
1742   const u32 lo = static_cast<u32>(value - static_cast<u64>(hi) * hundred_million);
1743   const __m128i x = set_i64(hi, lo);
1744   const __m128i div10k = splat_i64(div10k_sig);
1745   const __m128i n10k = splat_i64(neg10k);
1746   __m128i y = add_i64(x, mul_2x32_to_2x64_u(n10k, shr_i64(mul_2x32_to_2x64_u(x, div10k), 40)));
1747   y = shuffle_i32<27>(y);
1748 
1749   const __m128i div100 = splat_i32(static_cast<i32>(div100_sig));
1750   const __m128i hundred = splat_i32(100);
1751   const __m128i div10 = splat_i16(static_cast<i16>(div10_sig));
1752   const __m128i moddiv10 = splat_i16(10 * (1 << 8) - 1);
1753   const __m128i y_div_100 = shr_i16(mul_hi_u16(y, div100), 3);
1754   const __m128i y_mod_100 = sub_i16(y, mul_lo_i16(y_div_100, hundred));
1755   const __m128i z = or_i128(shl_i32(y_mod_100, 16), y_div_100);
1756   const __m128i bcd = sub_i16(shl_i16(z, 8), mul_lo_i16(moddiv10, mul_hi_u16(z, div10)));
1757   storeu_i128(reinterpret_cast<__m128i_u *>(out), or_i128(bcd, splat_i8('0')));
1758 }
