23 template<typename K>
24 concept reducible_key = (micron::is_integral_v<K> || micron::is_floating_point_v<K>)
25                         && (sizeof(K) == 1 || sizeof(K) == 2 || sizeof(K) == 4 || sizeof(K) == 8);

27 template<bool Less, typename K>
28 [[gnu::always_inline]] inline usize
29 __arg_extreme_scalar(const K *p, usize n) noexcept
30 {
31   usize bi = 0;
32   K best = p[0];
33   for ( usize i = 1; i < n; ++i ) {
34     if constexpr ( Less ) {
35       if ( p[i] < best ) {
36         best = p[i];
37         bi = i;
38       }
39     } else {
40       if ( p[i] > best ) {
41         best = p[i];
42         bi = i;
43       }
44     }
45   }
46   return bi;
47 }

49 template<typename K>
50 consteval bool
51 __arg_extreme_has_simd() noexcept
52 {
53 #if defined(__micron_x86_avx2)
54   return (micron::is_integral_v<K> && sizeof(K) == 4) || micron::is_same_v<K, float> || micron::is_same_v<K, double>;
55 #elif defined(__micron_arch_arm64) && defined(__micron_arm_neon)
56   return (micron::is_integral_v<K> && sizeof(K) <= 4) || micron::is_same_v<K, float> || micron::is_same_v<K, double>;
57 #elif defined(__micron_arch_arm32) && defined(__micron_arm_neon)
58   return (micron::is_integral_v<K> && sizeof(K) == 4) || micron::is_same_v<K, float>;
59 #else
60   (void)0;
61   return false;
62 #endif
63 }

65 template<bool Less, typename K>
66 [[gnu::always_inline]] inline K
67 __chunk_extreme(const K *p) noexcept
68 {
69 #if defined(__micron_x86_avx2)
70   if constexpr ( micron::is_integral_v<K> && sizeof(K) == 4 ) {
71     __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(p));
72     __m128i lo = _mm256_castsi256_si128(v);
73     __m128i hi = _mm256_extracti128_si256(v, 1);
74     __m128i m;
75     if constexpr ( micron::is_signed_v<K> ) {
76       m = Less ? _mm_min_epi32(lo, hi) : _mm_max_epi32(lo, hi);
77       m = Less ? _mm_min_epi32(m, _mm_shuffle_epi32(m, 0x4E)) : _mm_max_epi32(m, _mm_shuffle_epi32(m, 0x4E));
78       m = Less ? _mm_min_epi32(m, _mm_shuffle_epi32(m, 0xB1)) : _mm_max_epi32(m, _mm_shuffle_epi32(m, 0xB1));
79     } else {
80       m = Less ? _mm_min_epu32(lo, hi) : _mm_max_epu32(lo, hi);
81       m = Less ? _mm_min_epu32(m, _mm_shuffle_epi32(m, 0x4E)) : _mm_max_epu32(m, _mm_shuffle_epi32(m, 0x4E));
82       m = Less ? _mm_min_epu32(m, _mm_shuffle_epi32(m, 0xB1)) : _mm_max_epu32(m, _mm_shuffle_epi32(m, 0xB1));
83     }
84     return static_cast<K>(_mm_cvtsi128_si32(m));
85   } else if constexpr ( micron::is_same_v<K, float> ) {
86     __m256 v = _mm256_loadu_ps(p);
87     __m128 lo = _mm256_castps256_ps128(v);
88     __m128 hi = _mm256_extractf128_ps(v, 1);
89     __m128 m = Less ? _mm_min_ps(lo, hi) : _mm_max_ps(lo, hi);
90     m = Less ? _mm_min_ps(m, _mm_movehl_ps(m, m)) : _mm_max_ps(m, _mm_movehl_ps(m, m));
91     m = Less ? _mm_min_ps(m, _mm_shuffle_ps(m, m, 0x55)) : _mm_max_ps(m, _mm_shuffle_ps(m, m, 0x55));
92     return _mm_cvtss_f32(m);
93   } else {      // double
94     __m256d v = _mm256_loadu_pd(p);
95     __m128d lo = _mm256_castpd256_pd128(v);
96     __m128d hi = _mm256_extractf128_pd(v, 1);
97     __m128d m = Less ? _mm_min_pd(lo, hi) : _mm_max_pd(lo, hi);
98     m = Less ? _mm_min_pd(m, _mm_unpackhi_pd(m, m)) : _mm_max_pd(m, _mm_unpackhi_pd(m, m));
99     return _mm_cvtsd_f64(m);
100   }
101 #elif defined(__micron_arch_arm64) && defined(__micron_arm_neon)
102   if constexpr ( micron::is_integral_v<K> && sizeof(K) == 1 ) {
103     if constexpr ( micron::is_signed_v<K> ) {
104       int8x16_t v = vld1q_s8(reinterpret_cast<const signed char *>(p));
105       return static_cast<K>(Less ? vminvq_s8(v) : vmaxvq_s8(v));
106     } else {
107       uint8x16_t v = vld1q_u8(reinterpret_cast<const unsigned char *>(p));
108       return static_cast<K>(Less ? vminvq_u8(v) : vmaxvq_u8(v));
109     }
110   } else if constexpr ( micron::is_integral_v<K> && sizeof(K) == 2 ) {
111     if constexpr ( micron::is_signed_v<K> ) {
112       int16x8_t v = vld1q_s16(reinterpret_cast<const signed short *>(p));
113       return static_cast<K>(Less ? vminvq_s16(v) : vmaxvq_s16(v));
114     } else {
115       uint16x8_t v = vld1q_u16(reinterpret_cast<const unsigned short *>(p));
116       return static_cast<K>(Less ? vminvq_u16(v) : vmaxvq_u16(v));
117     }
118   } else if constexpr ( micron::is_integral_v<K> && sizeof(K) == 4 ) {
119     if constexpr ( micron::is_signed_v<K> ) {
120       int32x4_t v = vld1q_s32(reinterpret_cast<const signed int *>(p));
121       return static_cast<K>(Less ? vminvq_s32(v) : vmaxvq_s32(v));
122     } else {
123       uint32x4_t v = vld1q_u32(reinterpret_cast<const unsigned int *>(p));
124       return static_cast<K>(Less ? vminvq_u32(v) : vmaxvq_u32(v));
125     }
126   } else if constexpr ( micron::is_same_v<K, float> ) {
127     float32x4_t v = vld1q_f32(p);
128     return Less ? vminvq_f32(v) : vmaxvq_f32(v);
129   } else {      // double : 2-lane fold (no vminvq_f64)
130     float64x2_t v = vld1q_f64(p);
131     double a = vgetq_lane_f64(v, 0), b = vgetq_lane_f64(v, 1);
132     return Less ? (a < b ? a : b) : (a > b ? a : b);
133   }
134 #elif defined(__micron_arch_arm32) && defined(__micron_arm_neon)
135   if constexpr ( micron::is_integral_v<K> && sizeof(K) == 4 ) {
136     if constexpr ( micron::is_signed_v<K> ) {
137       int32x4_t v = vld1q_s32(reinterpret_cast<const signed int *>(p));
138       int32x2_t r = Less ? vpmin_s32(vget_low_s32(v), vget_high_s32(v)) : vpmax_s32(vget_low_s32(v), vget_high_s32(v));
139       r = Less ? vpmin_s32(r, r) : vpmax_s32(r, r);
140       return static_cast<K>(vget_lane_s32(r, 0));
141     } else {
142       uint32x4_t v = vld1q_u32(reinterpret_cast<const unsigned int *>(p));
143       uint32x2_t r = Less ? vpmin_u32(vget_low_u32(v), vget_high_u32(v)) : vpmax_u32(vget_low_u32(v), vget_high_u32(v));
144       r = Less ? vpmin_u32(r, r) : vpmax_u32(r, r);
145       return static_cast<K>(vget_lane_u32(r, 0));
146     }
147   } else {      // float
148     float32x4_t v = vld1q_f32(p);
149     float32x2_t r = Less ? vpmin_f32(vget_low_f32(v), vget_high_f32(v)) : vpmax_f32(vget_low_f32(v), vget_high_f32(v));
150     r = Less ? vpmin_f32(r, r) : vpmax_f32(r, r);
151     return vget_lane_f32(r, 0);
152   }
153 #else
154   return p[0];
155 #endif
156 }

158 template<typename K>
159 consteval usize
160 __chunk_lanes() noexcept
161 {
162 #if defined(__micron_x86_avx2)
163   return 32 / sizeof(K);
164 #elif (defined(__micron_arch_arm64) || defined(__micron_arch_arm32)) && defined(__micron_arm_neon)
165   return 16 / sizeof(K);
166 #else
167   return 1;
168 #endif
169 }

171 template<bool Less, typename K>
172 inline usize
173 arg_extreme(const K *p, usize n) noexcept
174 {
175   if ( n <= 1 ) return 0;
176 
177   if constexpr ( reducible_key<K> && __arg_extreme_has_simd<K>() ) {
178     {
179       constexpr usize L = __chunk_lanes<K>();
180       usize bi = 0;
181       K best = p[0];
182       usize i = 0;
183       for ( ; i + L <= n; i += L ) {
184         K cv = __chunk_extreme<Less, K>(p + i);
185         const bool better = Less ? (cv < best) : (cv > best);
186         if ( better ) {
187           for ( usize j = i; j < i + L; ++j ) {
188             if constexpr ( Less ) {
189               if ( p[j] < best ) {
190                 best = p[j];
191                 bi = j;
192               }
193             } else {
194               if ( p[j] > best ) {
195                 best = p[j];
196                 bi = j;
197               }
198             }
199           }
200         }
201       }
202       for ( ; i < n; ++i ) {
203         if constexpr ( Less ) {
204           if ( p[i] < best ) {
205             best = p[i];
206             bi = i;
207           }
208         } else {
209           if ( p[i] > best ) {
210             best = p[i];
211             bi = i;
212           }
213         }
214       }
215       return bi;
216     }
217   }
218   return __arg_extreme_scalar<Less, K>(p, n);
219 }

239 inline usize
240 first_quake_violation(const u32 *nc, usize H, u32 an, u32 ad) noexcept
241 {
242   if ( H < 2 ) return H;
243   const usize lim = H - 1;      // i ranges over [0, lim)
244   usize i = 0;
245 
246 #if defined(__micron_x86_avx2)
247   {
248     const __m256i van = _mm256_set1_epi32(static_cast<int>(an));
249     const __m256i vad = _mm256_set1_epi32(static_cast<int>(ad));
250     for ( ; i + 8 <= lim; i += 8 ) {
251       __m256i a = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(nc + i));
252       __m256i b = _mm256_loadu_si256(reinterpret_cast<const __m256i *>(nc + i + 1));
253       __m256i lhs = _mm256_mullo_epi32(b, vad);
254       __m256i rhs = _mm256_mullo_epi32(a, van);
255       __m256i gt = _mm256_cmpgt_epi32(lhs, rhs);
256       unsigned m = static_cast<unsigned>(_mm256_movemask_epi8(gt));
257       if ( m ) return i + (static_cast<usize>(__builtin_ctz(m)) / 4);
258     }
259   }
260 #elif (defined(__micron_arch_arm64) || defined(__micron_arch_arm32)) && defined(__micron_arm_neon)
261   {
262     const uint32x4_t van = vdupq_n_u32(an);
263     const uint32x4_t vad = vdupq_n_u32(ad);
264     for ( ; i + 4 <= lim; i += 4 ) {
265       uint32x4_t a = vld1q_u32(nc + i);
266       uint32x4_t b = vld1q_u32(nc + i + 1);
267       uint32x4_t gt = vcgtq_u32(vmulq_u32(b, vad), vmulq_u32(a, van));
268       u32 buf[4];
269       vst1q_u32(buf, gt);
270       for ( int k = 0; k < 4; ++k )
271         if ( buf[k] ) return i + static_cast<usize>(k);
272     }
273   }
274 #endif
275 
276   for ( ; i < lim; ++i )
277     if ( static_cast<u64>(ad) * nc[i + 1] > static_cast<u64>(an) * nc[i] ) return i;
278   return H;
279 }
