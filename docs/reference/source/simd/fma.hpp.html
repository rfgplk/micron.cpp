23 __micron_diagnostic_push __micron_diagnostic_ignored("-Wignored-attributes") template<is_simd_class T>
24 inline typename T::bit_width __unwrap(const T &o) noexcept
25 {
26   static_assert(__is_standard_layout(T) && sizeof(T) == sizeof(typename T::bit_width),
27                 "fma: wrapper must be standard-layout with a sole register member");
28   return *reinterpret_cast<const typename T::bit_width *>(&o);
29 }
30 
31 template<is_simd_class T>
32 inline T
33 __wrap(typename T::bit_width r) noexcept
34 {
35   T o;
36   *reinterpret_cast<typename T::bit_width *>(&o) = r;
37   return o;
38 }
