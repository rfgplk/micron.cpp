72 template<class T, class Maker> struct __fork_awaitable {
73   void *__dst;      // eventual<T>* | T* | nullptr, interpreted per __mode
74   u8 __mode;
75   Maker __make;      // () -> task<T>
76 
77   bool
78   await_ready() const noexcept
79   {
80     return false;
81   }
82 
83   template<class P>
84   std::coroutine_handle<>
85   await_suspend(std::coroutine_handle<P> __parent) noexcept
86   {
87     __frame_base *__par = &__parent.promise();
88     const char *__msg = nullptr;
89     typename task<T>::__handle __ch{};
90 #if !defined(__micron_freestanding) || defined(__micron_eh)
91     try {
92       task<T> __child = __make();
93       __ch = __child.release();
94     } catch ( ... ) {
95       __msg = "micron::fork: child construction threw";
96     }
97 #else
98     {
99       task<T> __child = __make();
100       __ch = __child.release();
101     }
102 #endif
103     if ( !__ch && __msg == nullptr ) [[unlikely]]
104       __msg = "micron::fork: child frame allocation failed (arena exhausted)";
105     if ( __msg != nullptr ) [[unlikely]] {
106       // no fork happened
107       if ( __mode == __rm_eventual && __dst != nullptr ) static_cast<eventual<T> *>(__dst)->__set_error(__msg);
108       if ( __par->__first_err == nullptr ) __par->__first_err = __msg;
109       return __parent;
110     }
111     auto &__cb = __ch.promise();
112     __cb.__parent = __par;
113     __cb.__cancel = __par->__cancel;      // inherit the cooperative cancel flag (cancellation.hpp)
114     __cb.__ret = __dst;                   // deliver the result straight into the bound destination
115     __cb.__ret_mode = __mode;
116     __cb.__sibling = __par->__child_head;      // link into the parent's reclaim list (owns the frame)
117     __par->__child_head = &__cb;
118 
119     worker *__w = current_worker();
120     __par->__pushed_kind = __frame_base::__kind_fork;                          // tag before the release-publish
121     bool __published = (__w != nullptr) && __w->deque.push_bottom(__par);      // publish the parent continuation
122     if ( __published ) {
123       __notify_work();      // wake a parked worker if any (cheap when all busy)
124     } else {
125       __cb.__cont = __parent;
126     }
127     return __ch;      // run the child eagerly (symmetric transfer)
128   }
129 
130   void
131   await_resume() noexcept
132   {
133   }
134 };

136 template<class T, class Maker> struct __call_awaitable {
137   void *__dst;      // eventual<T>* | T* | nullptr, interpreted per __mode
138   u8 __mode;
139   Maker __make;      // () -> task<T>
140   typename task<T>::__handle __h{};
141 
142   ~__call_awaitable()
143   {
144     if ( __h ) __h.destroy();
145   }
146 
147   bool
148   await_ready() const noexcept
149   {
150     return false;
151   }

153   template<class P>
154   std::coroutine_handle<>
155   await_suspend(std::coroutine_handle<P> __parent) noexcept
156   {
157     const char *__msg = nullptr;
158 #if !defined(__micron_freestanding) || defined(__micron_eh)
159     try {
160       task<T> __child = __make();
161       __h = __child.release();
162     } catch ( ... ) {
163       __msg = "micron::call: child construction threw";
164     }
165 #else
166     {
167       task<T> __child = __make();
168       __h = __child.release();
169     }
170 #endif
171     if ( !__h && __msg == nullptr ) [[unlikely]]
172       __msg = "micron::call: child frame allocation failed (arena exhausted)";
173     if ( __msg != nullptr ) [[unlikely]] {
174       if ( __mode == __rm_eventual && __dst != nullptr ) static_cast<eventual<T> *>(__dst)->__set_error(__msg);
175       return __parent;
176     }
177     auto &__cb = __h.promise();
178     __cb.__cont = __parent;                           // serial continuation (no __parent, no deque publish)
179     __cb.__cancel = __parent.promise().__cancel;      // inherit the cooperative cancel flag
180     __cb.__ret = __dst;                               // construct-once delivery into the bound destination
181     __cb.__ret_mode = __mode;
182     return __h;      // symmetric transfer into the child (eager, serial)
183   }

200 template<class Fn> struct __fork_binder {
201   void *__dst;
202   u8 __mode;
203   Fn __fn;
204 
205   template<class... Args>
206   [[nodiscard]] auto
207   operator()(Args &&...__args)
208   {
209     static_assert(__impl::__fork_args_ok<Fn>(),
210                   "micron::fork: a forked callable must take its arguments by value; a reference parameter binds "
211                   "to the awaitable-owned argument copy and dangles when the continuation is stolen");
212     using T = coro::__task_value_t<decltype(__fn(micron::forward<Args>(__args)...))>;
213     auto __make
214         = [__f = micron::move(__fn), ... __a = micron::forward<Args>(__args)]() mutable -> task<T> { return __f(micron::move(__a)...); };
215     return __fork_awaitable<T, decltype(__make)>{ __dst, __mode, micron::move(__make) };
216   }
217 };

241 // fork[&a, fn](args...) / fork(&a, fn)(..)
242 // fork(discard, fn)(args...)
243 struct __fork_fn {
244   template<class T, class Fn, class... Args>
245   [[nodiscard]] auto
246   operator()(eventual<T> &__out, Fn &&__fn, Args &&...__args) const
247   {
248     static_assert(__impl::__fork_args_ok<Fn>(), "micron::fork: a forked callable must take its arguments BY VALUE");
249     auto __make = [__fn = micron::forward<Fn>(__fn), ... __a = micron::forward<Args>(__args)]() mutable -> task<T> {
250       return __fn(micron::move(__a)...);
251     };
252     return __fork_awaitable<T, decltype(__make)>{ static_cast<void *>(&__out), __rm_eventual, micron::move(__make) };
253   }
254 
255   template<class I, class Fn>
256     requires(!micron::is_function_v<I>)
257   [[nodiscard]] auto
258   operator()(I *__ret, Fn &&__fn) const
259   {
260     constexpr u8 __m = __impl::__is_eventual<I>::value ? __rm_eventual : __rm_lvalue;
261     return __impl::__fork_binder<micron::decay_t<Fn>>{ static_cast<void *>(__ret), __m, micron::forward<Fn>(__fn) };
262   }

336 struct __join_awaitable {
337   __frame_base *__self = nullptr;
338 
339   bool
340   await_ready() const noexcept
341   {
342     return false;
343   }
344 
345   template<class P>
346   std::coroutine_handle<>
347   await_suspend(std::coroutine_handle<P> __h) noexcept
348   {
349     __self = &__h.promise();
350     return __join_suspend(__self);
351   }
352 
353   void
354   await_resume()
355   {
356     if ( __self != nullptr ) {
357       __join_resume(__self);
358       if ( __self->__first_err != nullptr ) [[unlikely]] {
359         const char *__e = __self->__first_err;
360         __self->__first_err = nullptr;
361         __raise_eventual(__e);
362       }
363     }
364   }
