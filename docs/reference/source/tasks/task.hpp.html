69 // by value spec
70 template<class T> struct __task_promise: coro::__frame_base {
71   std::coroutine_handle<> __cont{};      // awaiting continuation (call/awaited) or inline-fork-fallback parent
72   coro::manual_lifetime<T> __val;        // holds the result ONLY in __rm_unbound (plain co_await / root)
73   void *__ret = nullptr;                 // bound destination: eventual<T>* (__rm_eventual) or T* (__rm_lvalue)
74   const char *__err = nullptr;
75   u8 __state = 0;      // 0 empty / 1 value / 2 error
76   u8 __ret_mode = coro::__rm_unbound;
77 
78   __task_promise() noexcept { }
79 
80   ~__task_promise()
81   {
82     if ( __state == 1 && __ret_mode == coro::__rm_unbound ) __val.destroy();
83   }
84 
85   task<T> get_return_object() noexcept;
86 
87   // next handle at final_suspend
88   std::coroutine_handle<>
89   __on_final() noexcept
90   {
91     if ( this->__parent != nullptr ) {
92       if ( __state == 2 ) {      // a failed child
93         if ( __ret_mode == coro::__rm_eventual && __ret != nullptr )
94           static_cast<coro::eventual<T> *>(__ret)->__set_error(__err);                          // explicit-read path raises with this
95         if ( this->__parent->__first_err == nullptr ) this->__parent->__first_err = __err;      // first-child-error-wins (surfaced at join)
96       }
97       if ( __cont ) return __cont;      // inline-fork fallback
98       return coro::__child_complete(this);
99     }
100     return __cont ? __cont : std::noop_coroutine();
101   }
102 
103   std::suspend_always
104   initial_suspend() noexcept
105   {
106     return {};
107   }
108 
109   __task_impl::__final_awaiter<__task_promise>
110   final_suspend() noexcept
111   {
112     return {};
113   }

115   template<class U = T>
116   void
117   return_value(U &&__v) noexcept(micron::is_nothrow_constructible_v<T, U &&>)
118   {
119     if ( __ret_mode == coro::__rm_eventual ) {
120       static_cast<coro::eventual<T> *>(__ret)->__set(micron::forward<U>(__v));      // construct into the eventual
121     } else if ( __ret_mode == coro::__rm_lvalue ) {
122       if constexpr ( requires(T &__d, U &&__s) { __d = micron::forward<U>(__s); } )
123         *static_cast<T *>(__ret) = micron::forward<U>(__v);      // assign into the caller lvalue
124       else
125         __builtin_trap();      // raw-lvalue fork binding requires an assignable T; use eventual<T>
126     } else if ( __ret_mode == coro::__rm_discard ) {
127       (void)micron::forward<U>(__v);      // evaluate, drop
128     } else {
129       __val.construct(micron::forward<U>(__v));      // unbound: into the slot
130     }
131     __state = 1;
132   }
133 
134   void
135   unhandled_exception() noexcept
136   {
137     __state = 2;
138 #if !defined(__micron_freestanding) || defined(__micron_eh)
139     __err = "micron::task: unhandled exception";
140 #else
141     __builtin_trap();      // unreachable
142 #endif
143   }

245 template<class T>
246 class [[nodiscard("a dropped micron::task never runs and immediately destroys its frame; "
247                   "co_await it, sync_wait it, or hand it to the scheduler")]] task
248 {
249 public:
250   using promise_type = __task_promise<T>;
251   using __handle = std::coroutine_handle<promise_type>;
252 
253   task() noexcept = default;
254 
255   explicit task(__handle __h) noexcept : __coro(__h) { }
256 
257   task(const task &) = delete;
258   task &operator=(const task &) = delete;
259 
260   task(task &&__o) noexcept : __coro(__o.__coro) { __o.__coro = {}; }
261 
262   task &
263   operator=(task &&__o) noexcept
264   {
265     if ( this != &__o ) {
266       if ( __coro ) __coro.destroy();
267       __coro = __o.__coro;
268       __o.__coro = {};
269     }
270     return *this;
271   }
272 
273   ~task()
274   {
275     if ( __coro ) __coro.destroy();
276   }
277 
278   bool
279   valid() const noexcept
280   {
281     return static_cast<bool>(__coro);
282   }
283 
284   bool
285   done() const noexcept
286   {
287     return __coro && __coro.done();
288   }
289 
290   __handle
291   handle() const noexcept
292   {
293     return __coro;
294   }
295 
296   __handle
297   release() noexcept
298   {
299     __handle __h = __coro;
300     __coro = {};
301     return __h;
302   }
303 
304   decltype(auto)
305   result()
306   {
307     if ( !__coro ) [[unlikely]]
308       __task_impl::__raise_broken("micron::task: result() on an empty/moved-from/allocation-failed task");
309     if ( !__coro.done() ) [[unlikely]]
310       __task_impl::__raise_broken("micron::task: result() called before the task completed");
311     return __coro.promise().__take();
312   }
313 
314   auto
315   operator co_await() && noexcept
316   {
317     struct awaiter {
318       __handle __h;
319 
320       bool
321       await_ready() const noexcept
322       {
323         return !__h || __h.done();
324       }
325 
326       std::coroutine_handle<>
327       await_suspend(std::coroutine_handle<> __awaiting) noexcept
328       {
329         __h.promise().__cont = __awaiting;
330         return __h;
331       }
332 
333       decltype(auto)
334       await_resume()
335       {
336         if ( !__h ) [[unlikely]]
337           __task_impl::__raise_broken("micron::task: co_await on a null/allocation-failed task");
338         return __h.promise().__take();
339       }
340     };
341 
342     return awaiter{ __coro };
343   }
344 
345 private:
346   __handle __coro{};
347 };
