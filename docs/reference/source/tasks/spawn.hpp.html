30 template<class I, class T>
31 [[nodiscard]] auto
32 fork_task(I *__ret, micron::task<T> &&__t)
33 {
34   constexpr u8 __m = __impl::__is_eventual<I>::value ? __rm_eventual : __rm_lvalue;
35   auto __make = [__tk = micron::move(__t)]() mutable -> micron::task<T> { return micron::move(__tk); };
36   return __fork_awaitable<T, decltype(__make)>{ static_cast<void *>(__ret), __m, micron::move(__make) };
37 }
38 
39 template<class Fn> using __spawn_result_t = __task_value_t<micron::invoke_result_t<Fn &, usize>>;
40 
41 template<class Fn>
42 micron::task<micron::vector<__spawn_result_t<Fn>>>
43 spawn_many(usize __n, Fn __fn)
44 {
45   using T = __spawn_result_t<Fn>;
46   if ( __n == 0 ) co_return micron::vector<T>{};
47   micron::vector<T> __out(__n);      // n default-constructed slots; addresses stable (no realloc in the loop)
48   for ( usize __i = 0; __i < __n; ++__i ) co_await fork(__builtin_addressof(__out[__i]), __fn)(__i);
49   co_await join;
50   co_return __out;
51 }
52 
53 template<usize... Is, class... Tk>
54 micron::task<micron::tuple<__task_value_t<Tk>...>>
55 __spawn_tuple_impl(micron::index_sequence<Is...>, Tk... __tasks)
56 {
57   micron::tuple<__task_value_t<Tk>...> __out{};
58   ((co_await fork_task(__builtin_addressof(micron::get<Is>(__out)), micron::move(__tasks))), ...);
59   co_await join;
60   co_return __out;
61 }
62 
63 template<class... Tk>
64 [[nodiscard]] auto
65 spawn_tuple(Tk... __tasks)
66 {
67   return __spawn_tuple_impl(micron::make_index_sequence<sizeof...(Tk)>{}, micron::move(__tasks)...);
68 }

70 template<class... Tk>
71 [[nodiscard]] auto
72 when_all(Tk... __tasks)
73 {
74   return __spawn_tuple_impl(micron::make_index_sequence<sizeof...(Tk)>{}, micron::move(__tasks)...);
