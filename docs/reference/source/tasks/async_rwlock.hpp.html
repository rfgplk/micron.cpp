26 class async_rwlock
27 {
28   struct __rw_node {
29     micron::atomic_token<usize> __st{ 0 };      // 0 idle | 1 granted | else parked frame pointer
30     __rw_node *__next = nullptr;
31   };
32 
33   micron::atomic_token<u32> __lk{ 0 };      // internal spinlock for the fields below
34   i64 __readers = 0;                        // active readers
35   bool __writer = false;                    // a writer holds
36   u32 __wwaiting = 0;                       // writers parked
37   __rw_node *__rhead = nullptr;             // parked readers
38   __rw_node *__whead = nullptr;             // parked writers
39 
40   void
41   __acq() noexcept
42   {
43     u32 __e = 0;
44     while ( !__lk.compare_exchange_weak(__e, 1u, memory_order_acq_rel, memory_order_acquire) ) {
45       __e = 0;
46       __cpu_pause();
47     }
48   }
49 
50   void
51   __rel() noexcept
52   {
53     __lk.store(0u, memory_order_release);
54   }
55 
56   static void
57   __grant(__rw_node *__w) noexcept
58   {
59     const usize __old = __w->__st.swap(1u, memory_order_acq_rel);
60     if ( __old != 0u ) __resume_waiter(reinterpret_cast<__frame_base *>(__old));
61   }
62 
63 public:
64   async_rwlock() noexcept = default;
65   async_rwlock(const async_rwlock &) = delete;
66   async_rwlock &operator=(const async_rwlock &) = delete;
67 
68   struct [[nodiscard]] __rlock_awaiter {
69     async_rwlock *__rw;
70     __rw_node __node{};
71 
72     bool
73     await_ready() noexcept
74     {
75       __rw->__acq();
76       if ( !__rw->__writer && __rw->__wwaiting == 0u ) {
77         ++__rw->__readers;
78         __rw->__rel();
79         return true;
80       }
81       __node.__next = __rw->__rhead;
82       __rw->__rhead = &__node;
83       __rw->__rel();
84       return false;
85     }
86 
87     template<class P>
88     bool
89     await_suspend(std::coroutine_handle<P> __h) noexcept
90     {
91       usize __exp = 0u;
92       const usize __f = reinterpret_cast<usize>(static_cast<__frame_base *>(&__h.promise()));
93       return __node.__st.compare_exchange_strong(__exp, __f, memory_order_acq_rel, memory_order_acquire);
94     }
95 
96     void
97     await_resume() const noexcept
98     {
99     }
100   };
101 
102   __rlock_awaiter
103   lock_shared() noexcept
104   {
105     return __rlock_awaiter{ this };
106   }
107 
108   void
109   unlock_shared() noexcept
110   {
111     __acq();
112     --__readers;
113     if ( __readers == 0 && __wwaiting > 0u ) {
114       __rw_node *__w = __whead;
115       __whead = __w->__next;
116       --__wwaiting;
117       __writer = true;
118       __rel();
119       __grant(__w);
120       return;
121     }
122     __rel();
123   }

125   struct [[nodiscard]] __wlock_awaiter {
126     async_rwlock *__rw;
127     __rw_node __node{};
128 
129     bool
130     await_ready() noexcept
131     {
132       __rw->__acq();
133       if ( !__rw->__writer && __rw->__readers == 0 ) {
134         __rw->__writer = true;
135         __rw->__rel();
136         return true;
137       }
138       ++__rw->__wwaiting;
139       __node.__next = __rw->__whead;
140       __rw->__whead = &__node;
141       __rw->__rel();
142       return false;
143     }
144 
145     template<class P>
146     bool
147     await_suspend(std::coroutine_handle<P> __h) noexcept
148     {
149       usize __exp = 0u;
150       const usize __f = reinterpret_cast<usize>(static_cast<__frame_base *>(&__h.promise()));
151       return __node.__st.compare_exchange_strong(__exp, __f, memory_order_acq_rel, memory_order_acquire);
152     }
153 
154     void
155     await_resume() const noexcept
156     {
157     }
158   };
159 
160   __wlock_awaiter
161   lock() noexcept
162   {
163     return __wlock_awaiter{ this };
164   }
165 
166   void
167   unlock() noexcept
168   {
169     __acq();
170     __writer = false;
171     if ( __wwaiting > 0u ) {
172       __rw_node *__w = __whead;
173       __whead = __w->__next;
174       --__wwaiting;
175       __writer = true;
176       __rel();
177       __grant(__w);
178       return;
179     }
180 
181     __rw_node *__rs = __rhead;
182     __rhead = nullptr;
183     i64 __cnt = 0;
184     for ( __rw_node *__p = __rs; __p != nullptr; __p = __p->__next ) ++__cnt;
185     __readers = __cnt;
186     __rel();
187     while ( __rs != nullptr ) {
188       __rw_node *__nx = __rs->__next;
189       __grant(__rs);
190       __rs = __nx;
191     }
192   }
