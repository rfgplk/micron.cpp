24 class async_condvar
25 {
26   struct __cv_waiter {
27     micron::atomic_token<usize> __st{ 0 };
28     __cv_waiter *__next = nullptr;
29   };
30 
31   __cv_waiter *__head = nullptr;
32   micron::atomic_token<u32> __lk{ 0 };
33 
34   void
35   __lock() noexcept
36   {
37     u32 __e = 0;
38     while ( !__lk.compare_exchange_weak(__e, 1u, memory_order_acq_rel, memory_order_acquire) ) {
39       __e = 0;
40       __cpu_pause();
41     }
42   }
43 
44   void
45   __unlock() noexcept
46   {
47     __lk.store(0u, memory_order_release);
48   }
49 
50   void
51   __push(__cv_waiter *__w) noexcept
52   {
53     __lock();
54     __w->__next = __head;
55     __head = __w;
56     __unlock();
57   }
58 
59   __cv_waiter *
60   __pop() noexcept
61   {
62     __lock();
63     __cv_waiter *__w = __head;
64     if ( __w != nullptr ) __head = __w->__next;
65     __unlock();
66     return __w;
67   }
68 
69   __cv_waiter *
70   __swap_all() noexcept
71   {
72     __lock();
73     __cv_waiter *__w = __head;
74     __head = nullptr;
75     __unlock();
76     return __w;
77   }
78 
79   static void
80   __signal(__cv_waiter *__w) noexcept
81   {
82     const usize __old = __w->__st.swap(1u, memory_order_acq_rel);
83     if ( __old != 0u ) __resume_waiter(reinterpret_cast<__frame_base *>(__old));
84   }
85 
86   struct [[nodiscard]] __park_awaiter {
87     __cv_waiter *__w;
88 
89     bool
90     await_ready() const noexcept
91     {
92       return __w->__st.get(memory_order_acquire) == 1u;
93     }
94 
95     template<class P>
96     bool
97     await_suspend(std::coroutine_handle<P> __h) noexcept
98     {
99       usize __exp = 0u;
100       const usize __f = reinterpret_cast<usize>(static_cast<__frame_base *>(&__h.promise()));
101 
102       return __w->__st.compare_exchange_strong(__exp, __f, memory_order_acq_rel, memory_order_acquire);
103     }
104 
105     void
106     await_resume() const noexcept
107     {
108     }
109   };
110 
111 public:
112   async_condvar() noexcept = default;
113   async_condvar(const async_condvar &) = delete;
114   async_condvar &operator=(const async_condvar &) = delete;
115 
116   micron::task<void>
117   wait(async_mutex &__m)
118   {
119     __cv_waiter __w;
120     __push(&__w);
121     __m.unlock();
122     co_await __park_awaiter{ &__w };
123     co_await __m.lock();
124   }
125 
126   void
127   notify_one() noexcept
128   {
129     __cv_waiter *__w = __pop();
130     if ( __w != nullptr ) __signal(__w);
131   }
132 
133   void
134   notify_all() noexcept
135   {
136     __cv_waiter *__w = __swap_all();
137     while ( __w != nullptr ) {
138       __cv_waiter *__nx = __w->__next;
139       __signal(__w);
140       __w = __nx;
141     }
142   }
143 };
