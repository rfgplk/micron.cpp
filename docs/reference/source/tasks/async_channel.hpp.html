18 // a bounded, coroutine-awaitable MPMC channel
19 
20 namespace micron
21 {
22 
23 enum class chan_status : u8 { ok = 0, empty = 1, closed = 2 };
24 
25 template<class T> struct chan_pull {
26   bool ok;
27   T value;
28 };
29 
30 template<class T, usize Cap = 256> class async_channel
31 {
32   micron::crossbeam<T, Cap> __ring;
33   micron::coro::async_semaphore __space{ static_cast<i64>(Cap) };      // free slots (<= ring capacity)
34   micron::coro::async_semaphore __items{ 0 };                          // available items
35   micron::atomic_token<u32> __closed{ 0 };
36 
37 public:
38   async_channel() = default;
39   async_channel(const async_channel &) = delete;
40   async_channel &operator=(const async_channel &) = delete;
41 
42   micron::task<bool>
43   push(T __v)
44   {
45     co_await __space.acquire();
46     if ( __closed.get(memory_order_acquire) ) co_return false;
47     while ( !__ring.push(micron::move(__v)) ) __cpu_pause();      // holds a space permit -> succeeds
48     __items.release();
49     co_return true;
50   }
51 
52   micron::task<chan_pull<T>>
53   pull()
54   {
55     const bool __got = co_await __items.acquire();
56     T __v{};
57     if ( __got ) {
58       while ( !__ring.pop(__v) ) __cpu_pause();      // permit held -> an item is guaranteed (may be mid-publish)
59       __space.release();
60       co_return chan_pull<T>{ true, micron::move(__v) };
61     }
62     // aborted/closed: no permit
63     if ( __ring.pop(__v) ) {
64       __space.release();
65       co_return chan_pull<T>{ true, micron::move(__v) };
66     }
67     co_return chan_pull<T>{ false, T{} };      // empty -> closed & drained
68   }
69 
70   micron::task<usize>
71   push_bulk(const T *__src, usize __n)
72   {
73     usize __done = 0;
74     while ( __done < __n ) {
75       if ( __closed.get(memory_order_acquire) ) break;
76       const i64 __got = __space.try_acquire_many(static_cast<i64>(__n - __done));
77       if ( __got > 0 ) {
78         for ( i64 __k = 0; __k < __got; ++__k ) {
79           while ( !__ring.push(__src[__done]) ) __cpu_pause();
80           ++__done;
81         }
82         __items.release(__got);
83       } else {
84         co_await __space.acquire();      // backpressure: block for one permit
85         if ( __closed.get(memory_order_acquire) ) break;
86         while ( !__ring.push(__src[__done]) ) __cpu_pause();
87         ++__done;
88         __items.release(1);
89       }
90     }
91     co_return __done;
92   }
93 
94   micron::task<usize>
95   pull_bulk(T *__dst, usize __n)
96   {
97     usize __done = 0;
98     while ( __done < __n ) {
99       const i64 __got = __items.try_acquire_many(static_cast<i64>(__n - __done));
100       if ( __got > 0 ) {
101         for ( i64 __k = 0; __k < __got; ++__k ) {
102           T __v{};
103           while ( !__ring.pop(__v) ) __cpu_pause();
104           __dst[__done] = micron::move(__v);
105           ++__done;
106         }
107         __space.release(__got);
108       } else {
109         const bool __got_one = co_await __items.acquire();      // block for one item (or wake on close)
110         T __v{};
111         if ( __got_one ) {
112           while ( !__ring.pop(__v) ) __cpu_pause();      // permit held -> an item is guaranteed
113           __dst[__done] = micron::move(__v);
114           ++__done;
115           __space.release(1);
116         } else if ( __ring.pop(__v) ) {      // closed: drain any still-buffered item
117           __dst[__done] = micron::move(__v);
118           ++__done;
119           __space.release(1);
120         } else {
121           break;      // closed & drained
122         }
123       }
124     }
125     co_return __done;
126   }
127 
128   // non-blocking enqueue
129   bool
130   post(T __v) noexcept
131   {
132     if ( __closed.get(memory_order_acquire) || !__space.try_acquire() ) return false;
133     while ( !__ring.push(micron::move(__v)) ) __cpu_pause();
134     __items.release();
135     return true;
136   }
137 
138   // non-blocking dequeue
139   chan_status
140   try_pull(T &__out) noexcept
141   {
142     if ( __items.try_acquire() ) {
143       while ( !__ring.pop(__out) ) __cpu_pause();      // permit held -> an item is guaranteed
144       __space.release();
145       return chan_status::ok;
146     }
147     if ( __closed.get(memory_order_acquire) ) {
148       if ( __ring.pop(__out) ) {
149         __space.release();
150         return chan_status::ok;
151       }
152       return chan_status::closed;
153     }
154     return chan_status::empty;
155   }
156 
157   void
158   close() noexcept
159   {
160     __closed.store(1, memory_order_release);
161     __items.abort();
162     __space.abort();
163   }
164 
165   bool
166   is_closed() const noexcept
167   {
168     return __closed.get(memory_order_acquire) != 0;
169   }
170 };
