18 // a reusable (cyclic) barrier
19 
20 namespace micron
21 {
22 namespace coro
23 {
24 
25 class async_barrier
26 {
27   const u32 __threshold;
28   micron::atomic_token<u32> __count;      // arrivals still needed this generation
29   micron::atomic_token<u32> __gen;        // generation counter (prevents arrive/fire ABA)
30   waiter_list __waiters;
31 
32 public:
33   explicit async_barrier(u32 __n) noexcept : __threshold(__n), __count(__n), __gen(0) { }
34 
35   async_barrier(const async_barrier &) = delete;
36   async_barrier &operator=(const async_barrier &) = delete;
37 
38   struct [[nodiscard]] __arrive_awaiter {
39     async_barrier *__b;
40     u32 __mygen = 0;
41     waiter_node __node{};
42 
43     bool
44     await_ready() noexcept
45     {
46       __mygen = __b->__gen.get(memory_order_acquire);
47       const u32 __prev = __b->__count.fetch_sub(1, memory_order_acq_rel);
48       if ( __prev == 1u ) {      // last arriver: complete the generation
49         __b->__count.store(__b->__threshold, memory_order_relaxed);
50         // the bump and the drain must be one step wrt push_unless
51         __wake_all(__b->__waiters.swap_all_under([__bp = __b]() {
52           __bp->__gen.fetch_add(1, memory_order_acq_rel);
53           return true;
54         }));
55         return true;      // last arriver continues without suspending
56       }
57       return false;      // not last: suspend in await_suspend
58     }
59 
60     template<class P>
61     bool
62     await_suspend(std::coroutine_handle<P> __h) noexcept
63     {
64       __node.__frame = &__h.promise();
65       // park unless the generation already advanced (barrier fired between our fetch_sub and here)
66       return __b->__waiters.push_unless(&__node, [__bp = __b, __g = __mygen]() { return __bp->__gen.get(memory_order_acquire) != __g; });
67     }
68 
69     void
70     await_resume() const noexcept
71     {
72     }
73   };
74 
75   __arrive_awaiter
76   arrive_and_wait() noexcept
77   {
78     return __arrive_awaiter{ this };
79   }
80 };
