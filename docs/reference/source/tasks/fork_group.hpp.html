20 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
21 // dynamic task groups (Go/WaitGroup inspired)
22 //
23 // a fork_group owns its own completion counter, so you can spawn() incrementally and co_await join() from
24 // any frame; each spawned task runs as an independent root on the pool
25 
26 namespace micron

28 namespace coro

31 struct __group_bridge {
32   struct promise_type: __frame_base {
33     __group_bridge
34     get_return_object() noexcept
35     {
36       auto __h = std::coroutine_handle<promise_type>::from_promise(*this);
37       this->__self = __h;
38       return __group_bridge{ __h };
39     }
40 
41     std::suspend_always
42     initial_suspend() noexcept
43     {
44       return {};
45     }
46 
47     std::suspend_never
48     final_suspend() noexcept
49     {
50       return {};
51     }
52 
53     void
54     return_void() noexcept
55     {
56     }
57 
58     void
59     unhandled_exception() noexcept
60     {
61       __builtin_trap();
62     }
63 
64     static void *
65     operator new(usize __n) noexcept
66     {
67       return ::operator new(__n);
68     }
69 
70     static void
71     operator delete(void *__p, usize) noexcept
72     {
73       ::operator delete(__p);
74     }
75 
76     static __group_bridge
77     get_return_object_on_allocation_failure() noexcept
78     {
79       return {};
80     }
81   };
82 
83   std::coroutine_handle<promise_type> __h{};
84   __group_bridge() noexcept = default;
85 
86   explicit __group_bridge(std::coroutine_handle<promise_type> __hh) noexcept : __h(__hh) { }
87 };

89 class fork_group
90 {
91   micron::atomic_token<i64> __outstanding{ 0 };
92   waiter_list __joiners;
93   micron::atomic_token<u32> __cancelled{ 0 };
94 
95 public:
96   fork_group() noexcept = default;
97   fork_group(const fork_group &) = delete;
98   fork_group &operator=(const fork_group &) = delete;
99 
100   void
101   __complete() noexcept
102   {
103     if ( __outstanding.sub_fetch(1, memory_order_acq_rel) == 0 )
104       __wake_all(__joiners.swap_all_under([this]() { return __outstanding.get(memory_order_acquire) <= 0; }));
105   }
106 
107   [[nodiscard]] i64
108   outstanding() const noexcept
109   {
110     return __outstanding.get(memory_order_acquire);
111   }
112 
113   // cancellation
114   void
115   request_cancel() noexcept
116   {
117     __cancelled.store(1u, memory_order_release);
118   }
119 
120   [[nodiscard]] bool
121   cancelled() const noexcept
122   {
123     return __cancelled.get(memory_order_relaxed) != 0u;
124   }
125 
126   [[nodiscard]] cancellation_token
127   token() const noexcept
128   {
129     return cancellation_token(&__cancelled);
130   }
131 
132   template<class Fn, class... Args>
133   void
134   spawn(Fn __fn, Args... __args)
135   {
136     __outstanding.fetch_add(1, memory_order_acq_rel);
137     start_coroutine_runtime();
138     const micron::atomic_token<u32> *__cflag = &__cancelled;
139     fork_group *__self = this;
140     __group_bridge __b = [](fork_group *__g, const micron::atomic_token<u32> *__cf, Fn __f, Args... __a) -> __group_bridge {
141       auto __t = __f(micron::move(__a)...);                           // the spawned task<R>
142       if ( __t.valid() ) __t.handle().promise().__cancel = __cf;      // children inherit the group cancel flag
143       co_await micron::move(__t);
144       __g->__complete();
145     }(__self, __cflag, micron::move(__fn), micron::move(__args)...);
146     if ( __b.__h )
147       __global_engine->submit(&__b.__h.promise());
148     else
149       __complete();      // bridge allocation failed: undo the outstanding increment so join() still completes
150   }
151 
152   struct [[nodiscard]] __join_awaiter {
153     fork_group *__g;
154     waiter_node __node{};
155 
156     bool
157     await_ready() const noexcept
158     {
159       return __g->__outstanding.get(memory_order_acquire) <= 0;
160     }
161 
162     template<class P>
163     bool
164     await_suspend(std::coroutine_handle<P> __h) noexcept
165     {
166       __node.__frame = &__h.promise();
167       return __g->__joiners.push_unless(&__node, [__gp = __g]() { return __gp->__outstanding.get(memory_order_acquire) <= 0; });
168     }
169 
170     void
171     await_resume() const noexcept
172     {
173     }
174   };
175 
176   __join_awaiter
177   join() noexcept
178   {
179     return __join_awaiter{ this };
180   }
181 };
