18 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
19 // stackful coroutine porcelain
20 //
21 // spin / launch / jump / yield
22 //
23 // companion to the solo:: thread porcelain
24 //
25 // issues:
26 //   .. destroying a suspended routine abandons it: the stack is released with no unwinding
27 //   .. yield(v) with a type other than the routine's Y reads back type-confused (TODO: will be enforced later)
28 //   .. a suspended routine may be jumped or destroyed from another thread, but never concurrently.
29 
30 // example:
31 //   auto r = coro::spin<int>(fn, args...);   // create, paused (lazy; ~create_fiber cost)
32 //   auto r = coro::launch<int>(fn, args...); // create + run to the first yield (eager)
33 //   int *v = coro::jump(r);                  // enter r; returns at its next yield (value) or finish (nullptr)
34 //   coro::yield(42);                         // inside fn: hand 42 to the jumper, park here
35 //   coro::yield();                           // inside fn: park without a value
36 //   r.result();                              // finish() + move out fn's return value
37 namespace micron

39 namespace coro

42 struct __spin_ctl {
43   ar::__fiber_ctx jumper;                                  // jump() parks the jumper here; f->link -> &jumper
44   void *xfer = nullptr;                                    // yielded-value pointer; nulled past completion
45   void *res = nullptr;                                     // -> the block's __spin_result<R> base (exact upcast)
46   void (*drop)(__spin_ctl *, u32) noexcept = nullptr;      // type-erased cleanup, see __drop_*
47   const char *err = nullptr;                               // escaped-exception message (hosted/EH)
48 };
49 
50 inline constexpr u32 __drop_block = 0;       // destroy fn + args (never consumed: fresh routine dropped)
51 inline constexpr u32 __drop_result = 1;      // destroy an unconsumed result (done routine dropped)
52 
53 template<class R> struct __spin_result {
54   alignas(R) byte __slot[sizeof(R)];
55   bool __has = false;
56 };
57 
58 template<> struct __spin_result<void> {
59 };

61 template<class R, class F, class... A> struct __spin_block: __spin_ctl, __spin_result<R> {
62   F __fn;
63   micron::tuple<A...> __args;      // by-value homes for the launch args; consumed as rvalues at invocation
64 
65   template<class Fw, class... Aw>
66   explicit __spin_block(Fw &&__f, Aw &&...__a) : __fn(micron::forward<Fw>(__f)), __args(micron::forward<Aw>(__a)...)
67   {
68   }
69 };

74 template<class B, usize... Is>
75 [[gnu::always_inline]] inline decltype(auto)
76 __spin_apply(B *__b, micron::index_sequence<Is...>)
77 {
78   return __b->__fn(micron::move(micron::get<Is>(__b->__args))...);
79 }

81 template<class R, class F, class... A>
82 inline void
83 __spin_drop(__spin_ctl *__c, u32 __what) noexcept
84 {
85   auto *__b = static_cast<__spin_block<R, F, A...> *>(__c);
86   if ( __what == __drop_block ) {
87     __b->__fn.~F();
88     __b->__args.~tuple<A...>();
89     return;
90   }
91   if constexpr ( !micron::is_void_v<R> ) {
92     if ( __b->__has ) {
93       micron::ptr_cast<R *>(__b->__slot)->~R();
94       __b->__has = false;
95     }
96   }
97 }

99 // runs on the routine's fiber
100 template<class R, class F, class... A>
101 inline void
102 __spin_entry(fiber::fiber *__f) noexcept
103 {
104   auto *__b = static_cast<__spin_block<R, F, A...> *>(static_cast<__spin_ctl *>(__f->arg));
105 #if !defined(__micron_freestanding) || defined(__micron_eh)
106   try {
107 #endif
108     if constexpr ( micron::is_void_v<R> )
109       __spin_apply(__b, micron::index_sequence_for<A...>{});
110     else {
111       ::new (static_cast<void *>(__b->__slot)) R(__spin_apply(__b, micron::index_sequence_for<A...>{}));
112       __b->__has = true;
113     }
114 #if !defined(__micron_freestanding) || defined(__micron_eh)
115   } catch ( ... ) {
116     __b->err = "micron::coro::routine: unhandled exception";
117   }
118 #endif
119   __b->xfer = nullptr;      // no yield value past completion
120   __b->__fn.~F();           // consume the launch state on-fiber, before the final park
121   __b->__args.~tuple<A...>();
122 }

124 template<class Y = void, class R = void>
125 class [[nodiscard("a dropped micron::coro::routine releases its fiber; jump it, finish it, or keep the handle")]] routine
126 {
127   fiber::fiber *__f = nullptr;
128   __spin_ctl *__c = nullptr;      // == __f->arg, cached
129 
130   [[gnu::always_inline]] u32
131   __state() const noexcept
132   {
133     return __f->state.get(micron::memory_order_acquire);
134   }
135 
136   void
137   __destroy() noexcept
138   {
139     if ( __f == nullptr ) return;
140     const u32 __st = __state();
141     if ( __st == static_cast<u32>(fiber::fiber_state::running) ) __builtin_trap();      // destroying a live activation
142     if ( __st == static_cast<u32>(fiber::fiber_state::fresh) )
143       __c->drop(__c, __drop_block);      // fn/args were never consumed; run their dtors exactly once
144     else if ( __st == static_cast<u32>(fiber::fiber_state::done) )
145       __c->drop(__c, __drop_result);      // unconsumed result, if any
146     fiber::destroy_fiber(__f);
147     __f = nullptr;
148     __c = nullptr;
149   }
150 
151 public:
152   using yield_type = Y;
153   using result_type = R;
154 
155   // internal: handles are minted by spin()/launch()
156   routine(fiber::fiber *__ff, __spin_ctl *__cc) noexcept : __f(__ff), __c(__cc) { }
157 
158   routine() noexcept = default;
159   routine(const routine &) = delete;
160   routine &operator=(const routine &) = delete;
161 
162   routine(routine &&__o) noexcept : __f(__o.__f), __c(__o.__c)
163   {
164     __o.__f = nullptr;
165     __o.__c = nullptr;
166   }
167 
168   routine &
169   operator=(routine &&__o) noexcept
170   {
171     if ( this != &__o ) {
172       __destroy();
173       __f = __o.__f;
174       __c = __o.__c;
175       __o.__f = nullptr;
176       __o.__c = nullptr;
177     }
178     return *this;
179   }
180 
181   ~routine() { __destroy(); }
182 
183   bool
184   done() const noexcept
185   {
186     return __f == nullptr || __state() == static_cast<u32>(fiber::fiber_state::done);
187   }
188 
189   bool
190   alive() const noexcept
191   {
192     return __f != nullptr && __state() != static_cast<u32>(fiber::fiber_state::done);
193   }
194 
195   bool
196   started() const noexcept
197   {
198     return __f != nullptr && __state() != static_cast<u32>(fiber::fiber_state::fresh);
199   }
200 
201   explicit
202   operator bool() const noexcept
203   {
204     return alive();
205   }
206 
207   auto
208   jump() noexcept(false) -> micron::conditional_t<micron::is_void_v<Y>, bool, Y *>
209   {
210     constexpr auto __none = [] {
211       if constexpr ( micron::is_void_v<Y> )
212         return false;
213       else
214         return static_cast<Y *>(nullptr);
215     }();
216     if ( __f == nullptr ) return __none;
217     const u32 __st = __state();      // acquire: pairs with the suspend/done release for cross-thread jumps
218     if ( __st == static_cast<u32>(fiber::fiber_state::done) ) return __none;
219     if ( __st == static_cast<u32>(fiber::fiber_state::running) ) __builtin_trap();      // self/concurrent jump
220     __f->state.store(static_cast<u32>(fiber::fiber_state::running), micron::memory_order_relaxed);
221     __spin_ctl *__prev = __cur_routine;
222     __cur_routine = __c;
223     fiber::resume(__f, &__c->jumper);
224     __cur_routine = __prev;
225     if ( __c->err != nullptr ) [[unlikely]] {
226       const char *__e = __c->err;
227       __c->err = nullptr;
228       micron::exc<except::library_error>(__e);
229     }
230     if ( __f->state.get(micron::memory_order_relaxed) == static_cast<u32>(fiber::fiber_state::done) ) return __none;
231     if constexpr ( micron::is_void_v<Y> )
232       return true;
233     else
234       return static_cast<Y *>(__c->xfer);
235   }
236 
237   Y *
238   value() const noexcept
239     requires(!micron::is_void_v<Y>)
240   {
241     return __f != nullptr ? static_cast<Y *>(__c->xfer) : nullptr;
242   }
243 
244   void
245   finish()
246   {
247     while ( !done() ) (void)jump();
248   }
249 
250   R
251   result()
252     requires(!micron::is_void_v<R>)
253   {
254     finish();
255     if ( __f == nullptr ) [[unlikely]]
256       micron::exc<except::library_error>("micron::coro::routine: result() on an empty handle");
257     auto *__res = static_cast<__spin_result<R> *>(__c->res);
258     if ( !__res->__has ) [[unlikely]]
259       micron::exc<except::library_error>("micron::coro::routine: no result (fn did not complete)");
260     __res->__has = false;
261     R __r = micron::move(*micron::ptr_cast<R *>(__res->__slot));
262     micron::ptr_cast<R *>(__res->__slot)->~R();
263     return __r;
264   }
265 
266   void
267   dismiss() noexcept
268   {
269     __destroy();
270   }
271 
272   fiber::fiber *
273   native_handle() const noexcept
274   {
275     return __f;
276   }
277 
278   usize
279   stack_size() const noexcept
280   {
281     return __f != nullptr ? __f->region.stack_bytes : 0u;
282   }

288 // create a stackful coroutine running fn(args...) on its own fiber
289 template<class Y = void, usize Stack = static_cast<usize>(micro_stack_size), class F, class... Args>
290   requires(micron::is_invocable_v<micron::decay_t<F>, micron::decay_t<Args>...>)
291 [[nodiscard]] auto
292 spin(F &&__fn, Args &&...__args)
293 {
294   using DF = micron::decay_t<F>;
295   using R = micron::invoke_result_t<DF, micron::decay_t<Args>...>;
296   using B = __spin_block<R, DF, micron::decay_t<Args>...>;
297 
298   fiber::fiber *__f = fiber::create_fiber(&__spin_entry<R, DF, micron::decay_t<Args>...>, nullptr, Stack);
299   if ( __f == nullptr ) [[unlikely]]
300     micron::exc<except::memory_error>("micron::coro::spin: fiber segment exhausted");
301   const usize __p = (reinterpret_cast<usize>(__f->frame_sp) + alignof(B) - 1) & ~(alignof(B) - 1);
302   if ( __p + sizeof(B) > reinterpret_cast<usize>(__f->region.frame_limit) ) [[unlikely]] {
303     fiber::destroy_fiber(__f);
304     micron::exc<except::memory_error>("micron::coro::spin: launch state exceeds the frame arena");
305   }
306   B *__b;
307 #if !defined(__micron_freestanding) || defined(__micron_eh)
308   try {
309     __b = ::new (reinterpret_cast<void *>(__p)) B(micron::forward<F>(__fn), micron::forward<Args>(__args)...);
310   } catch ( ... ) {
311     fiber::destroy_fiber(__f);      // a throwing fn/arg copy must not leak the fresh segment
312     throw;
313   }
314 #else
315   __b = ::new (reinterpret_cast<void *>(__p)) B(micron::forward<F>(__fn), micron::forward<Args>(__args)...);
316 #endif
317   __b->drop = &__spin_drop<R, DF, micron::decay_t<Args>...>;
318   __b->res = static_cast<__spin_result<R> *>(__b);
319   __f->frame_sp = reinterpret_cast<byte *>((__p + sizeof(B) + 15) & ~static_cast<usize>(15));
320   __f->arg = static_cast<__spin_ctl *>(__b);
321   return routine<Y, R>{ __f, static_cast<__spin_ctl *>(__b) };
322 }

324 // launch: eager spin
325 template<class Y = void, usize Stack = static_cast<usize>(micro_stack_size), class F, class... Args>
326   requires(micron::is_invocable_v<micron::decay_t<F>, micron::decay_t<Args>...>)
327 [[nodiscard]] auto
328 launch(F &&__fn, Args &&...__args)
329 {
330   auto __r = spin<Y, Stack>(micron::forward<F>(__fn), micron::forward<Args>(__args)...);
331   (void)__r.jump();
332   return __r;
333 }

343 [[gnu::noinline]] inline void
344 __yield_park(void *__xfer) noexcept
345 {
346   __spin_ctl *__c = __cur_routine;
347   if ( __c == nullptr ) [[unlikely]]
348     __builtin_trap();      // yield outside a spun routine
349   __c->xfer = __xfer;
350   fiber::suspend_to_scheduler();      // f->link == &ctl->jumper: straight back to the jumper
351 }
