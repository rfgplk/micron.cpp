152 struct __io_mop {
153   __frame_base *__f = nullptr;
154   micron::atomic_token<u32> __st{ __io_mop_idle };
155   micron::atomic_token<u32> __live{ 0 };      // armed and the kernel still promises more
156   micron::atomic_token<u32> __qh{ 0 };
157   micron::atomic_token<u32> __qt{ 0 };
158   u8 __ring = 0xff;      // worker ring the op is armed on
159   __io_mev __q[__io_mop_qcap];
160 
161   [[nodiscard]] bool
162   __pop(__io_mev &__out) noexcept      // single consumer
163   {
164     const u32 __h = __qh.get(micron::memory_order_relaxed);
165     if ( __qt.get(micron::memory_order_acquire) == __h ) return false;
166     __out = __q[__h & (__io_mop_qcap - 1u)];
167     __qh.store(__h + 1u, micron::memory_order_release);
168     return true;
169   }
170 
171   void
172   __push(const __io_mev &__e) noexcept      // dispatcher only; serialized by the ring's __cq_lk
173   {
174     const u32 __t = __qt.get(micron::memory_order_relaxed);
175     if ( __t - __qh.get(micron::memory_order_acquire) >= __io_mop_qcap ) [[unlikely]]
176       __builtin_trap();      // see the capacity note above
177     __q[__t & (__io_mop_qcap - 1u)] = __e;
178     // WARNING: must be seq_cst, not release; will cause cross read deadlocks otherwise
179     __qt.store(__t + 1u, micron::memory_order_seq_cst);
180   }
181 };

237 [[gnu::always_inline]] inline void
238 __io_lock(micron::atomic_token<u32> &__l) noexcept
239 {
240   u32 __e = 0;
241   while ( !__l.compare_exchange_weak(__e, 1u, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
242     __e = 0;
243     micron::cpu_pause<1>();
244   }

260 [[gnu::always_inline]] inline bool
261 __io_cq_acquire(__wring &__wr) noexcept
262 {
263   if ( __wr.__live.get(micron::memory_order_acquire) == 0 ) return false;
264   if ( !__io_trylock(__wr.__cq_lk) ) return false;
265   if ( __wr.__live.get(micron::memory_order_acquire) == 0 ) [[unlikely]] {      // died between the test and the lock
266     __io_unlock(__wr.__cq_lk);
267     return false;
268   }
269   return true;
270 }

316 inline i32
317 __io_pb_init(__wring &__wr) noexcept
318 {
319   const u32 __st = __wr.__pb.__state.get(micron::memory_order_acquire);
320   if ( __st == 1u ) return 0;
321   if ( __st == 2u ) return -95;      // EOPNOTSUPP: refused earlier
322   const usize __rb = static_cast<usize>(__io_pb_entries) * sizeof(micron::uring::buf);
323   auto *__br = micron::ptr_cast<micron::uring::buf_ring *>(
324       micron::mmap(nullptr, __rb, micron::prot_read | micron::prot_write, micron::map_private | micron::map_anonymous, -1, 0));
325   if ( micron::mmap_failed(reinterpret_cast<addr_t *>(__br)) ) {
326     __wr.__pb.__state.store(2u, micron::memory_order_release);
327     return -95;
328   }
329   byte *__ar
330       = reinterpret_cast<byte *>(micron::mmap(nullptr, static_cast<usize>(__io_pb_entries) * __io_pb_sz,
331                                               micron::prot_read | micron::prot_write, micron::map_private | micron::map_anonymous, -1, 0));
332   if ( micron::mmap_failed(micron::ptr_cast<addr_t *>(__ar)) ) {
333     micron::munmap(reinterpret_cast<addr_t *>(__br), __rb);
334     __wr.__pb.__state.store(2u, micron::memory_order_release);
335     return -95;
336   }
337   micron::uring::buf_reg __reg{};
338   __reg.ring_addr = reinterpret_cast<u64>(__br);
339   __reg.ring_entries = __io_pb_entries;
340   __reg.bgid = __io_pb_bgid;
341   if ( __wr.__r.register_pbuf_ring(__reg) < 0 ) {
342     micron::munmap(reinterpret_cast<addr_t *>(__br), __rb);
343     micron::munmap(micron::ptr_cast<addr_t *>(__ar), static_cast<usize>(__io_pb_entries) * __io_pb_sz);
344     __wr.__pb.__state.store(2u, micron::memory_order_release);
345     return -95;
346   }
347   for ( u32 __i = 0; __i < __io_pb_entries; ++__i ) {
348     micron::uring::buf *__b = &__br->bufs()[__i];
349     __b->addr = reinterpret_cast<u64>(__ar + static_cast<usize>(__i) * __io_pb_sz);
350     __b->len = __io_pb_sz;
351     __b->bid = static_cast<u16>(__i);
352   }
353   __wr.__pb.__br = __br;
354   __wr.__pb.__arena = __ar;
355   __wr.__pb.__tail_shadow = __io_pb_entries;
356   micron::atom::store(&__br->tail, static_cast<u16>(__io_pb_entries), __ATOMIC_RELEASE);
357   __wr.__pb.__state.store(1u, micron::memory_order_release);      // publishes __br/__arena
358   return 0;
359 }

373 inline void
374 __io_pb_recycle(u8 __ring, u16 __bid) noexcept
375 {
376   if ( __ring >= __io_ring_cap || __bid >= __io_pb_entries ) [[unlikely]]
377     return;
378   __wring &__wr = __io_rings[__ring];
379   if ( __wr.__pb.__state.get(micron::memory_order_acquire) != 1u ) return;
380   __io_lock(__wr.__pb.__lk);
381   if ( __wr.__pb.__state.get(micron::memory_order_acquire) != 1u ) [[unlikely]] {      // died between the test and the lock
382     __io_unlock(__wr.__pb.__lk);
383     return;
384   }
385   micron::uring::buf *__b = &__wr.__pb.__br->bufs()[__wr.__pb.__tail_shadow & (__io_pb_entries - 1u)];
386   __b->addr = reinterpret_cast<u64>(__wr.__pb.__arena + static_cast<usize>(__bid) * __io_pb_sz);
387   __b->len = __io_pb_sz;
388   __b->bid = __bid;
389   ++__wr.__pb.__tail_shadow;
390   micron::atom::store(&__wr.__pb.__br->tail, static_cast<u16>(__wr.__pb.__tail_shadow), __ATOMIC_RELEASE);
391   __io_unlock(__wr.__pb.__lk);
392 }

394 inline void
395 __io_pb_shutdown(__wring &__wr) noexcept      // owner thread, before the ring fd closes
396 {
397   if ( __wr.__pb.__state.get(micron::memory_order_acquire) != 1u ) return;
398   __wr.__pb.__state.store(0u, micron::memory_order_release);
399   __io_lock(__wr.__pb.__lk);
400   __io_unlock(__wr.__pb.__lk);
401   (void)__wr.__r.unregister_pbuf_ring(__io_pb_bgid);
402   micron::munmap(reinterpret_cast<addr_t *>(__wr.__pb.__br), static_cast<usize>(__io_pb_entries) * sizeof(micron::uring::buf));
403   micron::munmap(micron::ptr_cast<addr_t *>(__wr.__pb.__arena), static_cast<usize>(__io_pb_entries) * __io_pb_sz);
404   __wr.__pb.__br = nullptr;
405   __wr.__pb.__arena = nullptr;
406   __wr.__pb.__tail_shadow = 0;
407 }

421 __io_worker_ring_init(u32 __id, u32 __nworkers) noexcept
422 {
423   __wring &__wr = __io_rings[__id];
424   if ( __wr.__r.init_best(__io_sq_entries, __io_ring_flags(__nworkers)) != 0 ) return;
425   __wr.__defer = (__wr.__r.setup_flags & micron::uring::setup_defer_taskrun) != 0 ? 1 : 0;
426   (void)__wr.__r.register_ring_fd();
427   __wr.__pending.store(0, micron::memory_order_relaxed);
428   __wr.__park_fired.store(0, micron::memory_order_relaxed);

433 inline void
434 __io_worker_ring_shutdown(u32 __id) noexcept
435 {
436   __wring &__wr = __io_rings[__id];
437   if ( __wr.__live.get(micron::memory_order_acquire) == 0 ) return;
438   __io_flush_staged(__wr);
439   __io_pb_shutdown(__wr);      // unregister before the ring fd closes

441   __wr.__live.store(0, micron::memory_order_release);
442   __io_lock(__wr.__cq_lk);
443   __wr.__r.shutdown();
444   __wr.__defer = 0;
445   __wr.__bufs_reg.store(0, micron::memory_order_relaxed);
446   __wr.__files_reg.store(0, micron::memory_order_relaxed);
447   for ( u32 __i = 0; __i < __io_file_slots / 64; ++__i ) __wr.__file_free[__i] = 0;
448   __io_unlock(__wr.__cq_lk);
449 }

474 inline bool
475 __io_submit_own(__wring &__wr, const micron::uring::sqe &__q, u64 __ud) noexcept
476 {
477   micron::uring::sqe *__s = __wr.__r.get_sqe();
478   if ( __s == nullptr ) [[unlikely]] {
479 #if defined(MICRON_CORO_STATS)
480     __wr.__stat.sqe_full_flushes.fetch_add(1, micron::memory_order_relaxed);
481 #endif
482     __wr.__staged = 0;
483     (void)__wr.__r.enter(0);
484     __s = __wr.__r.get_sqe();
485     if ( __s == nullptr ) return false;
486   }
487   *__s = __q;
488   __s->user_data = __ud;
489   __wr.__r.advance_sq();
490 #if defined(MICRON_CORO_STAGED_SUBMIT)
491   if ( ++__wr.__staged < __io_stage_max ) {
492 #if defined(MICRON_CORO_STATS)
493     __wr.__stat.submits.fetch_add(1, micron::memory_order_relaxed);
494 #endif
495     return true;
496   }
497   __wr.__staged = 0;
498 #endif
499   (void)__wr.__r.enter(0);
500 #if defined(MICRON_CORO_STATS)
501   __wr.__stat.submits.fetch_add(1, micron::memory_order_relaxed);
502   __wr.__stat.enters.fetch_add(1, micron::memory_order_relaxed);
503 #endif
504   return true;
505 }

507 inline bool
508 __io_submit_fb(const micron::uring::sqe &__q, u64 __ud) noexcept
509 {
510   // spinlocked
511   __io_lock(__io_fb.__sq_lk);
512   if ( __io_fb.__live.get(micron::memory_order_acquire) == 0 ) [[unlikely]] {      // torn down under us
513     __io_unlock(__io_fb.__sq_lk);
514     return false;
515   }
516   micron::uring::sqe *__s = __io_fb.__r.get_sqe();
517   if ( __s == nullptr ) [[unlikely]] {
518 #if defined(MICRON_CORO_STATS)
519     __io_fb.__stat.sqe_full_flushes.fetch_add(1, micron::memory_order_relaxed);
520 #endif
521     (void)__io_fb.__r.enter(0);
522     __s = __io_fb.__r.get_sqe();
523     if ( __s == nullptr ) {
524       __io_unlock(__io_fb.__sq_lk);
525       return false;
526     }
527   }
528   *__s = __q;
529   __s->user_data = __ud;

713 struct __io_fixed_pool {
714   byte *__base = nullptr;
715   micron::atomic_token<u32> __map{ 0 };      // bit i set = slot busy
716   micron::atomic_token<u32> __lk{ 0 };
717   micron::atomic_token<u32> __ready{ 0 };

720 inline __io_fixed_pool __io_fixed;

722 inline bool
723 __io_fixed_ensure() noexcept
724 {
725   if ( __io_fixed.__ready.get(micron::memory_order_acquire) != 0 ) return true;
726   __io_lock(__io_fixed.__lk);
727   if ( __io_fixed.__ready.get(micron::memory_order_acquire) != 0 ) {
728     __io_unlock(__io_fixed.__lk);
729     return true;
730   }
731   void *__p = micron::mmap(nullptr, __io_fixed_slots * __io_fixed_sz, prot_read | prot_write, map_private | map_anonymous, -1, 0);
732   if ( reinterpret_cast<usize>(__p) >= static_cast<usize>(-4095) ) {
733     __io_unlock(__io_fixed.__lk);
734     return false;
735   }
736   __io_fixed.__base = reinterpret_cast<byte *>(__p);

742 inline bool
743 __io_fixed_reg(__wring &__wr) noexcept
744 {
745   const u32 __r0 = __wr.__bufs_reg.get(micron::memory_order_acquire);
746   if ( __r0 == 1 ) return true;
747   if ( __r0 == 2 ) return false;
748   __io_lock(__wr.__sq_lk);
749   const u32 __r1 = __wr.__bufs_reg.get(micron::memory_order_acquire);
750   if ( __r1 != 0 ) [[unlikely]] {      // another thread finished while we waited
751     __io_unlock(__wr.__sq_lk);
752     return __r1 == 1;
753   }
754   if ( !__io_fixed_ensure() ) {
755     __wr.__bufs_reg.store(2, micron::memory_order_release);
756     __io_unlock(__wr.__sq_lk);
757     return false;
758   }
759   micron::uring::iovec __iov[__io_fixed_slots];
760   for ( u32 __i = 0; __i < __io_fixed_slots; ++__i ) {
761     __iov[__i].iov_base = __io_fixed.__base + static_cast<usize>(__i) * __io_fixed_sz;
762     __iov[__i].iov_len = __io_fixed_sz;
763   }

765   const u32 __v = __wr.__r.register_buffers(__iov, __io_fixed_slots) != 0 ? 2u : 1u;
766   __wr.__bufs_reg.store(__v, micron::memory_order_release);

796 [[nodiscard]] inline i32
797 __io_slot_acquire(__wring &__wr) noexcept
798 {
799   if ( !__io_files_reg(__wr) ) return -1;
800   __io_lock(__wr.__file_lk);
801   for ( u32 __w = 0; __w < __io_file_slots / 64; ++__w ) {
802     if ( __wr.__file_free[__w] == ~0ull ) continue;
803     const u32 __b = static_cast<u32>(__builtin_ctzll(~__wr.__file_free[__w]));
804     __wr.__file_free[__w] |= (1ull << __b);
805     __io_unlock(__wr.__file_lk);
806     return static_cast<i32>(__w * 64u + __b);
807   }
808   __io_unlock(__wr.__file_lk);
809   return -1;      // exhausted; the caller stops staging and reports the rest unstaged
810 }

818   __io_unlock(__wr.__file_lk);
