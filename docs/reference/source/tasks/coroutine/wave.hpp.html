52 [[nodiscard]] inline __wring *
53 wave_ring_of(i32 __rid) noexcept
54 {
55   if ( __rid < 0 || static_cast<u32>(__rid) >= __io_ring_cap ) return nullptr;
56   __wring &__wr = __io_rings[__rid];
57   return __wr.__live.get(micron::memory_order_acquire) != 0 ? &__wr : nullptr;
58 }

74 // evict a direct descriptor whose close never ran (severed chain, cancelled batch)
75 [[nodiscard]] inline bool
76 wave_slot_close(i32 __rid, i32 __slot) noexcept
77 {
78   // WARNING: the slot may only go back in the free bitmap once this succeeds
79   __wring *__r = wave_ring_of(__rid);
80   if ( __r == nullptr ) return false;
81   if ( __slot < 0 || static_cast<u32>(__slot) >= __io_file_slots ) return false;
82   const i32 __none = -1;
83   return __r->__r.register_files_update(static_cast<u32>(__slot), &__none, 1) >= 0;
84 }

86 struct [[nodiscard]] __wave_awaitable {
87   micron::uring::sqe *__q = nullptr;
88   __io_wop *__nodes = nullptr;
89   __io_wave *__wv = nullptr;
90   u32 __k = 0;
91   i32 __err = 0;

93   bool
94   await_ready() noexcept
95   {
96     if ( __k == 0 ) {
97       __err = 0;
98       return true;
99     }
100     if ( __io.any_live.get(micron::memory_order_acquire) == 0 ) {
101       __err = -38;      // no ring on this system
102       return true;
103     }
104     return false;
105   }

107   template<class P>
108   bool
109   await_suspend(std::coroutine_handle<P> __h) noexcept
110   {
111     __frame_base *__f = &__h.promise();
112     __wv->__f = __f;
113     __wv->__fin.store(0, micron::memory_order_relaxed);
114     __wv->__st.store(__io_st_submitted, micron::memory_order_relaxed);
115     __wv->__left.store(__k, micron::memory_order_release);
116 
117     __wring *__r = __io_own_ring();
118     if ( __r == nullptr ) [[unlikely]] {
119       __err = -38;
120       return false;
121     }
122     for ( u32 __i = 0; __i < __k; ++__i ) {
123       if ( reinterpret_cast<u64>(&__nodes[__i]) >> __io_ud_tag_shift != 0 ) [[unlikely]]
124         __builtin_trap();
125       __nodes[__i].__w = __wv;
126       __nodes[__i].__res = 0;
127       __q[__i].user_data = __io_ud_make(__io_ud_wop, reinterpret_cast<u64>(&__nodes[__i]));
128     }
129 
130     __r->__pending.fetch_add(1, micron::memory_order_acq_rel);
131     if ( !__io_submit_own_n(*__r, __q, __k) ) [[unlikely]] {
132       __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
133       __wv->__left.store(0, micron::memory_order_release);
134       __err = -105;
135       return false;
136     }
137 
138     if ( __io_cq_acquire(*__r) ) {
139       micron::uring::cqe __c{};
140       u32 __budget = __k + 8u;
141       while ( __budget-- != 0 && __wv->__left.get(micron::memory_order_acquire) != 0 && __r->__r.peek_cqe(&__c) )
142         __global_engine->__dispatch_cqe(*__r, __c);
143       __io_unlock(__r->__cq_lk);
144     }
145 
146     u32 __exp = __io_st_submitted;
147     if ( __wv->__st.compare_exchange_strong(__exp, __io_st_suspended, micron::memory_order_acq_rel, micron::memory_order_acquire) )
148       return true;
149     __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
150     return false;
151   }

153   i32
154   await_resume() const noexcept
155   {
156     return __err;
157   }

167 [[nodiscard]] inline bool
168 __wave_settle(__io_wave &__wv) noexcept
169 {
170   if ( __wv.__st.get(micron::memory_order_acquire) != __io_st_resumed ) return true;
171   for ( u64 __i = 0; __wv.__fin.get(micron::memory_order_acquire) == 0; ++__i ) {
172     if ( __i >= (1ull << 24) ) {
173       __io_wave_stranded.fetch_add(1, micron::memory_order_acq_rel);
174       return false;
175     }
176     if ( (__i & 1023u) == 1023u )
177       micron::yield();
178     else
179       micron::cpu_pause<1>();
180   }
181   return true;
182 }

184 inline void
185 __wave_drain_cancel(__io_wave &__wv, __io_wop *__nodes, u32 __k, i32 __rid) noexcept
186 {
187   if ( __wv.__left.get(micron::memory_order_acquire) == 0 ) return;
188   __wring *__r = wave_ring_of(__rid);
189   if ( __r == nullptr || __global_engine == nullptr ) {
190     __wv.__left.store(0, micron::memory_order_release);      // never reached a real ring
191     return;
192   }
193   for ( u32 __i = 0; __i < __k; ++__i ) __io_sync_cancel_ud(__r->__r.fd, __io_ud_make(__io_ud_wop, reinterpret_cast<u64>(&__nodes[__i])));
194   while ( __wv.__left.get(micron::memory_order_acquire) != 0 ) {
195     if ( __io_cq_acquire(*__r) ) {
196       micron::uring::cqe __c{};
197       while ( __r->__r.peek_cqe(&__c) ) __global_engine->__dispatch_cqe(*__r, __c);
198       __io_unlock(__r->__cq_lk);
199       continue;
200     }
201     if ( __r->__live.get(micron::memory_order_acquire) == 0 ) break;      // ring torn down
202     micron::cpu_pause<1>();
203   }
