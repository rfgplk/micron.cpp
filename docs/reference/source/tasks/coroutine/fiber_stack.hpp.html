18 // cactus fiber-stack backing memory
19 //
20 // raw carve/release of contiguous, double-guard-paged segments for stackful fibers
21 //
22 //   [ low guard | coroutine-frame arena --grows up--> | mid guard | <--grows down-- C exec stack | FCB ]
23 //     low                                                                                          high
24 namespace micron

26 namespace fiber

29 // geometric size classes (usable bytes, before the guard page)
30 inline constexpr usize __seg_size_table[] = {
31   static_cast<usize>(micro_stack_size),       // 256 KiB
32   static_cast<usize>(thread_stack_size),      // 512 KiB
33   static_cast<usize>(small_stack_size),       // 1 MiB
34   static_cast<usize>(4) * 1024 * 1024,        // 4 MiB
35 };
36 inline constexpr u32 __seg_class_count = sizeof(__seg_size_table) / sizeof(__seg_size_table[0]);
37 
38 // share of the class size given to the coroutine-frame arena; the remainder is the C exec stack
39 inline constexpr usize __seg_frame_ratio_num = 3;
40 inline constexpr usize __seg_frame_ratio_den = 4;
41 
42 // a carved fiber segment (two regions + two guard pages)
43 struct __region {
44   byte *run = nullptr;              // mapping base == low guard page (PROT_NONE)
45   byte *frame_base = nullptr;       // coroutine-frame arena low (grows up)
46   byte *frame_limit = nullptr;      // coroutine-frame arena high (== mid guard)
47   byte *mid_guard = nullptr;        // PROT_NONE page between the arena and the C stack
48   byte *stack_base = nullptr;       // C exec stack low (above mid guard)
49   byte *stack_top = nullptr;        // C exec stack high; FCB + initial SP live here
50   usize frame_bytes = 0;            // coroutine-frame arena size
51   usize stack_bytes = 0;            // C exec stack size
52   usize total = 0;                  // whole mapping size
53   bool from_va = false;             // carved from __va_carve (vs addrmap fallback)
54   u32 cls = 0;                      // size-class index
55 };
56 
57 [[gnu::always_inline]] inline u32
58 __seg_class_for(usize want) noexcept
59 {
60   for ( u32 i = 0; i < __seg_class_count; ++i )
61     if ( __seg_size_table[i] >= want ) return i;
62   return __seg_class_count - 1;      // clamp to the largest class
63 }
64 
65 // [low guard | arena (all the slack) | mid guard | stack]
66 [[gnu::always_inline]] inline void
67 __region_geometry(__region &out, byte *run, usize total, usize stack_bytes, u32 cls, bool from_va) noexcept
68 {
69   constexpr usize page = __micron_page_size_default;
70   const usize frame_bytes = total - stack_bytes - 2 * page;
71   out.run = run;
72   out.frame_base = run + page;
73   out.frame_limit = out.frame_base + frame_bytes;
74   out.mid_guard = out.frame_limit;
75   out.stack_base = out.mid_guard + page;
76   out.stack_top = out.stack_base + stack_bytes;
77   out.frame_bytes = frame_bytes;
78   out.stack_bytes = stack_bytes;
79   out.total = total;
80   out.from_va = from_va;
81   out.cls = cls;
82 }
83 
84 [[gnu::cold]] inline bool
85 __carve_region(__region &out, u32 cls) noexcept
86 {
87   if ( cls >= __seg_class_count ) cls = __seg_class_count - 1;
88   constexpr usize page = __micron_page_size_default;
89   const usize usable = __seg_size_table[cls];
90   const usize frame_want = (usable * __seg_frame_ratio_num / __seg_frame_ratio_den + page - 1) & ~(page - 1);
91   usize stack_bytes = ((usable - frame_want) + page - 1) & ~(page - 1);
92   if ( stack_bytes < page ) stack_bytes = page;
93 
94   {
95     const usize want = page + frame_want + page + stack_bytes;
96     const usize total = (want + abc::__sheet_align_mask) & ~abc::__sheet_align_mask;      // granule-rounded
97     byte *run = reinterpret_cast<byte *>(abc::__va_carve_reserved(total));
98     if ( run != nullptr ) {
99       __region_geometry(out, run, total, stack_bytes, cls, true);
100       if ( abc::__va_commit(micron::ptr_cast<addr_t *>(out.frame_base), out.frame_bytes) != nullptr
101            && abc::__va_commit(micron::ptr_cast<addr_t *>(out.stack_base), out.stack_bytes) != nullptr ) [[likely]]
102         return true;
103       abc::__va_release(micron::ptr_cast<addr_t *>(run), total);      // commit failed: hand the run back
104     }
105   }
106 
107   const usize total = page + frame_want + page + stack_bytes;
108   byte *run = reinterpret_cast<byte *>(micron::addrmap(total));
109   if ( micron::mmap_failed(run) || run == nullptr ) {
110     out = __region{};
111     return false;
112   }
113   __region_geometry(out, run, total, stack_bytes, cls, false);
114   (void)micron::mprotect(micron::ptr_cast<addr_t *>(out.run), page, micron::prot_none);
115   (void)micron::mprotect(micron::ptr_cast<addr_t *>(out.mid_guard), page, micron::prot_none);
116   return true;
117 }
118 
119 // release a carved run back to the VA reservation (or unmap the fallback mapping)
120 inline void
121 __release_region(const __region &reg) noexcept
122 {
123   if ( reg.run == nullptr ) return;
124   if ( reg.from_va )
125     abc::__va_release(micron::ptr_cast<addr_t *>(reg.run), reg.total);
126   else
127     micron::try_unmap(reg.run, reg.total);
128 }
129 
130 [[gnu::always_inline]] inline void
131 __decommit_region(const __region &reg) noexcept
132 {
133   if ( reg.frame_base != nullptr && reg.stack_top > reg.frame_base )
134     (void)micron::madvise(micron::ptr_cast<addr_t *>(reg.frame_base), static_cast<usize>(reg.stack_top - reg.frame_base),
135                           /*MADV_DONTNEED*/ 4);
136 }
