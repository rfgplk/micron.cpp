49 inline void
50 __cl_hot_entry(micron::fiber::fiber *self) noexcept
51 {
52   for ( ;; ) {
53     void *a = self->arg;
54     if ( a == nullptr ) return;
55     static_cast<__frame_base *>(a)->__self.resume();
56     self->arg = nullptr;
57     micron::ar::__micron_swap_context(&self->ctx, self->link);
58   }

90   [[gnu::always_inline]] void
91   __run(worker *w, __frame_base *cont) noexcept
92   {
93     // steal accounting happens at the take sites
94     micron::fiber::fiber *f = w->hot;
95     if ( f == nullptr ) [[unlikely]] {
96       f = micron::fiber::create_fiber(&__cl_hot_entry, nullptr, static_cast<usize>(small_stack_size));
97       while ( f == nullptr ) [[unlikely]] {
98         // stack/VA reservation is exhausted
99         if ( stopping.get(micron::memory_order_acquire) ) return;
100         micron::yield();
101         f = micron::fiber::create_fiber(&__cl_hot_entry, nullptr, static_cast<usize>(small_stack_size));
102       }
103       w->hot = f;
104     }
105     f->arg = cont;
106     micron::fiber::resume(f);
107 #if !defined(__micron_freestanding) || defined(__micron_eh)
108     // hard trap out if an exception escaped the coroutine
109     if ( f->escaped ) [[unlikely]]
110       __builtin_trap();
111 #endif
112     if ( f->refs.get(micron::memory_order_acquire) != 1 ) [[unlikely]]
113       __retire_hot(w);
114     else
115       f->frame_sp = f->region.frame_base;
116   }

120   __frame_base *
121   __steal(worker *w, u32 &seed) noexcept
122   {
123     if ( n <= 1 ) return nullptr;
124     for ( u32 sweep = 0; sweep < 2u; ++sweep ) {
125       seed ^= seed << 13;      // one xorshift per sweep; random start avoids thief convoys
126       seed ^= seed >> 17;
127       seed ^= seed << 5;
128       // Lemire reduction instead of seed % n (integer division), then a linear probe (faster)
129       u32 v = static_cast<u32>((static_cast<u64>(seed) * n) >> 32);
130       for ( u32 i = 0; i < n; ++i ) {
131         if ( v != w->id ) {
132 #if defined(MICRON_CORO_STEAL_MIN_DEPTH)
133           // experiment: first sweep skips depth-1 victims
134           if ( sweep == 0 && workers[v].deque.size() <= 1 ) {
135             if ( ++v == n ) v = 0;
136             continue;
137           }
138 #endif
139           for ( u32 r = 0; r < __cl_steal_retries; ++r ) {
140             const micron::steal_result<__frame_base *> s = workers[v].deque.try_steal();
141             if ( s.__st == micron::steal_status::got ) {
142               if ( s.__more ) __notify_work();      // wake propagation
143               return s.__v;
144             }
145             if ( s.__st == micron::steal_status::empty ) break;
146             __cpu_pause();      // lost
147           }
148         }
149         if ( ++v == n ) v = 0;
150       }
151     }
152     return nullptr;
153   }

155   __frame_base *
156   __search(worker *w, u32 &seed) noexcept
157   {
158 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
159     if ( __cl_searchers.get(micron::memory_order_relaxed) >= __cl_max_searchers ) return nullptr;
160 #else
161     const u32 __awake = n - static_cast<u32>(__builtin_popcount(__cl_sleeper_mask.get(micron::memory_order_relaxed)));
162     const u32 __s = __cl_searchers.get(micron::memory_order_relaxed);
163     if ( __s != 0 && 2u * __s >= __awake ) return nullptr;
164 #endif
165     __cl_searchers.fetch_add(1, micron::memory_order_acq_rel);
166     __frame_base *cont = nullptr;
167     u32 backoff = 16;
168     for ( u32 round = 0; round < 6; ++round ) {      // 16+32+..+512 pauses ~= 1-2 us total
169       for ( u32 p = 0; p < backoff; ++p ) __cpu_pause();
170       if ( stopping.get(micron::memory_order_acquire) ) break;
171       cont = __find(w, seed);
172       if ( cont != nullptr ) break;
173       if ( backoff < 512 ) backoff <<= 1;
174     }
175     __cl_searchers.sub_fetch(1, micron::memory_order_acq_rel);
176     return cont;
177   }

179   __frame_base *
180   __find(worker *w, u32 &seed) noexcept
181   {
182     __frame_base *cont = nullptr;
183     if ( (++w->tick & 63u) == 0u ) {      // periodic inbox-first so roots aren't starved by deep local work
184       if ( inbox.pop(cont) && cont != nullptr ) return cont;
185     }
186     cont = w->deque.pop_bottom();
187     if ( cont != nullptr ) {
188       __count_take(cont);      // a locally-popped fork continuation was detached from its child (it suspended)
189       return cont;
190     }
191     if ( w->ovf != nullptr ) [[unlikely]] {      // deque-overflow spill
192       cont = w->ovf;
193       w->ovf = cont->__ovf_next;
194       cont->__ovf_next = nullptr;
195       return cont;
196     }
197     if ( inbox.pop(cont) && cont != nullptr ) return cont;
198 #if defined(MICRON_CORO_URING)
199     cont = __io_ovf_pop();      // completions the inbox had no room for
200     if ( cont != nullptr ) return cont;
201     if ( __io.any_live.get(micron::memory_order_acquire) != 0 ) {
202       if ( __drain_io(w, seed) ) {
203         if ( inbox.pop(cont) && cont != nullptr ) return cont;
204         cont = __io_ovf_pop();
205         if ( cont != nullptr ) return cont;
206       }
207     }
208 #endif
209     cont = __steal(w, seed);
210     if ( cont != nullptr ) __count_take(cont);
211     return cont;
212   }

216   void
217   worker_main(u32 id) noexcept
218   {
219     worker *w = &workers[id];
220     __cur_worker = w;
221     u32 seed = id * 2654435761u + 1u;
222 #if defined(MICRON_CORO_URING)
223     __io_worker_ring_init(id, n);
224 #endif
225     for ( u32 __i = 0; __i < __cl_prewarm_segments; ++__i ) {
226       micron::fiber::fiber *__pf = micron::fiber::create_fiber(&__cl_hot_entry, nullptr, static_cast<usize>(small_stack_size));
227       if ( __pf == nullptr ) break;
228       micron::fiber::destroy_fiber(__pf);
229     }
230     // active covers the interval where a continuation left the inbox/deque but is still running here
231     for ( ;; ) {
232       if ( stopping.get(micron::memory_order_acquire) ) break;
233       w->active.store(1, micron::memory_order_release);
234       __frame_base *cont = __find(w, seed);
235       if ( cont == nullptr ) cont = __search(w, seed);      // bounded pre-park spin (capped searchers)
236       if ( cont != nullptr ) {
237         __run(w, cont);
238         continue;
239       }
240       w->active.store(0, micron::memory_order_release);
241 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
242       __cl_sleepers.fetch_add(1, micron::memory_order_seq_cst);      // announce parked (Dekker: see submit)
243       const u32 sig = __cl_signal.get(micron::memory_order_seq_cst);
244 #else
245       const u32 __bit = 1u << w->id;
246       // NOTE: a waker can only bump the epoch after claiming our bit, so any wake between the announce and the futex_wait leaves epoch !=
247       // __ep -> EAGAIN
248       const u32 __ep = __cl_park[w->id].epoch.get(micron::memory_order_relaxed);
249       __cl_sleeper_mask.fetch_or(__bit, micron::memory_order_seq_cst);      // announce parked (Dekker: see __cl_wake_one<true>)
250 #endif
251       w->active.store(1, micron::memory_order_release);
252       cont = __find(w, seed);
253       if ( cont != nullptr ) {
254 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
255         __cl_sleepers.sub_fetch(1, micron::memory_order_acq_rel);
256 #else
257         __cl_sleeper_mask.fetch_and(~__bit, micron::memory_order_acq_rel);
258 #endif
259         if ( !stopping.get(micron::memory_order_acquire) ) __run(w, cont);
260         continue;
261       }
262       w->active.store(0, micron::memory_order_release);
263       if ( !stopping.get(micron::memory_order_acquire) ) {
264 #if defined(MICRON_CORO_URING)
265         // (>=6.7)
266         if ( __io.futex_ok.get(micron::memory_order_acquire) != 0 ) {
267           __wring &__own = __io_rings[w->id];
268           if ( __own.__live.get(micron::memory_order_acquire) != 0 && __own.__pending.get(micron::memory_order_relaxed) != 0 ) {
269             __ring_park(w, __ep);
270             __cl_sleeper_mask.fetch_and(~__bit, micron::memory_order_acq_rel);
271             continue;
272           }
273         }
274         if ( __io.any_live.get(micron::memory_order_acquire) != 0 && __io_pending_total() != 0 ) {
275           i32 __exp = -1;
276           if ( __io.watcher.compare_exchange_strong(__exp, static_cast<i32>(w->id), micron::memory_order_acq_rel,
277                                                     micron::memory_order_acquire) ) {
278             timespec_t __wts{ 0, 1000000 };
279             micron::__futex(__cl_park[w->id].epoch.ptr(), futex_wait | futex_private_flag, __ep, &__wts, nullptr, 0);
280             __io.watcher.store(-1, micron::memory_order_release);
281             __drain_all();
282             __cl_sleeper_mask.fetch_and(~__bit, micron::memory_order_acq_rel);
283             continue;
284           }
285         }
286 #endif
287         timespec_t __ts{ 0, 100000000 };
288 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
289         micron::__futex(__cl_signal.ptr(), futex_wait | futex_private_flag, sig, &__ts, nullptr, 0);
290 #else
291         micron::__futex(__cl_park[w->id].epoch.ptr(), futex_wait | futex_private_flag, __ep, &__ts, nullptr, 0);
292 #endif
293       }
294 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
295       __cl_sleepers.sub_fetch(1, micron::memory_order_acq_rel);
296 #else
297       __cl_sleeper_mask.fetch_and(~__bit, micron::memory_order_acq_rel);
298 #endif
299     }
300     w->active.store(0, micron::memory_order_release);
301     __retire_hot(w);      // before drain: the retired segment recycles into the freelist being drained
302     micron::fiber::drain_freelist();
303 #if defined(MICRON_CORO_URING)
304     __io_worker_ring_shutdown(id);      // own thread: last touch of the single-issuer ring
305 #endif
306   }

308   void
309   submit(__frame_base *fb) noexcept
310   {
311     while ( !inbox.push(fb) ) micron::yield();      // inbox full
312 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
313     __cl_signal.fetch_add(1, micron::memory_order_seq_cst);
314     if ( __cl_sleepers.get(micron::memory_order_seq_cst) != 0 ) micron::wake_futex(__cl_signal.ptr(), 1);
315 #else
316     __cl_wake_one<true>();      // Strong: the seq_cst RMW mask read orders after the inbox push
317 #endif
318   }

323   __submit_io(__frame_base *fb) noexcept
324   {

335   __frame_base *
336   __io_ovf_pop() noexcept
337   {
338     if ( __io_ovf_n.get(micron::memory_order_acquire) == 0 ) return nullptr;
339     __io_lock(__io_ovf_lk);
340     __frame_base *fb = __io_ovf;
341     if ( fb != nullptr ) {
342       __io_ovf = fb->__io_next;
343       fb->__io_next = nullptr;
344       __io_ovf_n.sub_fetch(1, micron::memory_order_acq_rel);
345     }
346     __io_unlock(__io_ovf_lk);
347     return fb;
348   }

356   // cqe popping is serialized per ring
357   void
358   __dispatch_cqe(__wring &__wr, const micron::uring::cqe &__c) noexcept
359   {
360     const u8 __tag = __io_ud_tag(__c.user_data);
361     if ( __tag == __io_ud_park ) {
362       if ( __c.res == -22 || __c.res == -95 ) __io.futex_ok.store(0, micron::memory_order_release);
363       __wr.__park_fired.store(1, micron::memory_order_release);
364       return;
365     }
366     if ( __tag == __io_ud_mop ) {
367       __io_mop *__m = reinterpret_cast<__io_mop *>(__io_ud_payload(__c.user_data));
368       __io_mev __e;
369       __e.__res = __c.res;
370       __e.__ring = __m->__ring;
371       __e.__fl = 0;
372       if ( (__c.flags & micron::uring::cqe_f_buffer) != 0 ) {
373         __e.__bid = micron::uring::cqe_buffer_id(__c.flags);
374         __e.__fl |= __io_mev_buf;
375       }
376       if ( (__c.flags & micron::uring::cqe_f_more) != 0 )
377         __e.__fl |= __io_mev_more;
378       else {
379         __m->__live.store(0, micron::memory_order_release);
380         __wr.__pending.sub_fetch(1, micron::memory_order_acq_rel);
381       }
382       __m->__push(__e);
383       // three state park handoff
384       u32 __cur = __m->__st.get(micron::memory_order_seq_cst);
385       for ( ;; ) {
386         if ( __cur == __io_mop_parked ) {
387           if ( !__m->__st.compare_exchange_weak(__cur, __io_mop_idle, micron::memory_order_seq_cst, micron::memory_order_seq_cst) )
388             continue;
389           __frame_base *__fb = __m->__f;      // lst read of *__m
390           __submit_io(__fb);
391           break;
392         }
393         if ( __cur == __io_mop_parking ) {
394           if ( !__m->__st.compare_exchange_weak(__cur, __io_mop_pending, micron::memory_order_seq_cst, micron::memory_order_seq_cst) )
395             continue;
396           break;
397         }
398         break;
399       }
400       return;
401     }
402     if ( __tag == __io_ud_wop ) {
403       __io_wop *__wp = reinterpret_cast<__io_wop *>(__io_ud_payload(__c.user_data));
404       __wp->__res = __c.res;
405       __io_wave *__wv = __wp->__w;
406       if ( __wv->__left.sub_fetch(1, micron::memory_order_acq_rel) != 0 ) return;
407       u32 __exp = __io_st_submitted;
408       if ( __wv->__st.compare_exchange_strong(__exp, __io_st_done_early, micron::memory_order_acq_rel, micron::memory_order_acquire) )
409         return;
410       __wr.__pending.sub_fetch(1, micron::memory_order_acq_rel);
411       u32 __sus = __io_st_suspended;
412       if ( !__wv->__st.compare_exchange_strong(__sus, __io_st_resumed, micron::memory_order_acq_rel, micron::memory_order_acquire) )
413         return;      // the owner is tearing the wave down and frees it itself
414       __submit_io(__wv->__f);
415       return;
416     }
417     if ( __tag != __io_ud_op ) return;      // cancel / ltimer / reclaim
418     __io_op *__op = reinterpret_cast<__io_op *>(__c.user_data);
419     __op->__res = __c.res;
420     u32 __exp = __io_st_submitted;
421     if ( __op->__st.compare_exchange_strong(__exp, __io_st_done_early, micron::memory_order_acq_rel, micron::memory_order_acquire) ) return;
422     __wr.__pending.sub_fetch(1, micron::memory_order_acq_rel);
423     __submit_io(__op->__f);
424   }

427   __drain_ring(__wring &__wr) noexcept
428   {
429     if ( !__io_cq_acquire(__wr) ) return false;

440   // own ring -> fallback ring -> one seed-rotated foreign ring
441   bool
442   __drain_io(worker *__w, u32 &__seed) noexcept
443   {
444     bool __any = false;
445     __wring &__own = __io_rings[__w->id];
446     if ( __own.__live.get(micron::memory_order_acquire) != 0 ) {
447       __io_flush_staged(__own);
448       if ( __own.__r.cq_overflowed() || (__own.__defer != 0 && __own.__r.taskrun_pending()) )
449         (void)__own.__r.enter2(0, 0, micron::uring::enter_getevents, nullptr, 0);
450       if ( __own.__pending.get(micron::memory_order_relaxed) != 0 ) __any |= __drain_ring(__own);
451     }
452     if ( __io_fb.__pending.get(micron::memory_order_relaxed) != 0 ) {
453       if ( __io_fb.__r.cq_overflowed() ) (void)__io_fb.__r.enter2(0, 0, micron::uring::enter_getevents, nullptr, 0);
454       __any |= __drain_ring(__io_fb);
455     }
456     if ( n > 1 ) {
457       __seed ^= __seed << 13;
458       __seed ^= __seed >> 17;
459       __seed ^= __seed << 5;
460       u32 __v = static_cast<u32>((static_cast<u64>(__seed) * n) >> 32);
461       if ( __v == __w->id && ++__v == n ) __v = 0;
462       if ( __v != __w->id && __io_rings[__v].__pending.get(micron::memory_order_relaxed) != 0 ) __any |= __drain_ring(__io_rings[__v]);
463     }
464     return __any;
465   }

476   void
477   __ring_park(worker *__w, u32 __ep) noexcept
478   {
479     __wring &__wr = __io_rings[__w->id];
480     __wr.__park_fired.store(0, micron::memory_order_relaxed);
481     micron::uring::sqe __q;
482     micron::uring::prep_futex_wait(&__q, __cl_park[__w->id].epoch.ptr(), __ep);
483     if ( !__io_submit_own(__wr, __q, __io_ud_make(__io_ud_park, __w->id)) ) {
484       timespec_t __ts{ 0, 100000000 };
485       micron::__futex(__cl_park[__w->id].epoch.ptr(), futex_wait | futex_private_flag, __ep, &__ts, nullptr, 0);
486       return;
487     }
488 #if defined(MICRON_CORO_STATS)
489     __wr.__stat.parks.fetch_add(1, micron::memory_order_relaxed);
490 #endif
491     const micron::uring::ktimespec __kt{ 0, 100000000 };
492     (void)__wr.__r.submit_and_wait_timeout(1, &__kt);
493 #if defined(MICRON_CORO_STATS)
494     __wr.__stat.wakes.fetch_add(1, micron::memory_order_relaxed);
495 #endif
496     __drain_ring(__wr);
497     if ( __wr.__park_fired.get(micron::memory_order_acquire) == 0 ) {
498       micron::uring::sqe __cq;
499       micron::uring::prep_cancel(&__cq, __io_ud_make(__io_ud_park, __w->id));
500       while ( !__io_submit_own(__wr, __cq, __io_ud_make(__io_ud_cancel, 0)) ) {
501         if ( __wr.__park_fired.get(micron::memory_order_acquire) != 0 ) break;
502         (void)__wr.__r.submit_and_wait_timeout(1, &__kt);
503         __drain_ring(__wr);
504       }
505       while ( __wr.__park_fired.get(micron::memory_order_acquire) == 0 ) {
506         (void)__wr.__r.submit_and_wait_timeout(1, &__kt);
507         __drain_ring(__wr);
508       }
509     }

569 [[gnu::always_inline]] inline void
570 __ts_add_ns(timespec_t &__t, u64 __ns) noexcept
571 {
572   __t.tv_sec += static_cast<decltype(__t.tv_sec)>(__ns / 1000000000ull);
573   __t.tv_nsec += static_cast<decltype(__t.tv_nsec)>(__ns % 1000000000ull);

580 inline void
581 __timer_main() noexcept
582 {
583   for ( ;; ) {
584     engine *__e = __global_engine;
585     if ( __e == nullptr || __e->stopping.get(micron::memory_order_acquire) ) break;
586     const u32 __ep = __timer_epoch.get(micron::memory_order_acquire);
587     timespec_t __now{};
588     micron::clock_gettime(micron::clock_monotonic, __now);
589     __timer_node *__fired = nullptr;
590     bool __have_next = false;
591     timespec_t __next{};
592     __timer_lock();
593     __timer_node **__pp = &__timer_head;
594     while ( *__pp != nullptr ) {
595       if ( __ts_le((*__pp)->__deadline, __now) ) {
596         __timer_node *__n = *__pp;
597         *__pp = __n->__next;
598         __n->__next = __fired;
599         __fired = __n;
600       } else {
601         if ( !__have_next || __ts_le((*__pp)->__deadline, __next) ) {
602           __next = (*__pp)->__deadline;
603           __have_next = true;
604         }
605         __pp = &(*__pp)->__next;
606       }
607     }
608     __timer_unlock();
609     while ( __fired != nullptr ) {
610       __timer_node *__nx = __fired->__next;      // read before submit (the resumed frame may free the node)
611       __frame_base *__f = __fired->__frame;
612       __e->submit(__f);
613       __e->pending_timers.sub_fetch(1, micron::memory_order_acq_rel);
614       __fired = __nx;
615     }
616     if ( __have_next ) {
617       timespec_t __rel{ __next.tv_sec - __now.tv_sec, __next.tv_nsec - __now.tv_nsec };
618       if ( __rel.tv_nsec < 0 ) {
619         __rel.tv_sec -= 1;
620         __rel.tv_nsec += 1000000000l;
621       }
622       if ( __rel.tv_sec < 0 ) continue;      // already due; rescan
623       micron::__futex(__timer_epoch.ptr(), futex_wait | futex_private_flag, __ep, &__rel, nullptr, 0);
624     } else {
625       micron::__futex(__timer_epoch.ptr(), futex_wait | futex_private_flag, __ep, nullptr, nullptr, 0);
626     }
627   }
628 }

657 inline void
658 start_coroutine_runtime(u32 nworkers = 0) noexcept
659 {
660   if ( __engine_state.get(micron::memory_order_acquire) == 2u ) return;
661 
662   u32 __exp = 0u;
663   if ( !__engine_state.compare_exchange_strong(__exp, 1u, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
664     while ( __engine_state.get(micron::memory_order_acquire) != 2u ) micron::yield();      // wait for ready
665     return;
666   }
667 
668   __arm_runtime_reaper();
669 
670   if ( nworkers == 0 ) nworkers = micron::cpu_count();
671   if ( nworkers > __cl_max_workers ) nworkers = __cl_max_workers;
672   if ( nworkers < 1 ) nworkers = 1;
673 
674   engine *e = new engine();
675   e->n = nworkers;
676   e->workers = new worker[nworkers];
677   for ( u32 i = 0; i < nworkers; ++i ) e->workers[i].id = i;
678 #if defined(MICRON_CORO_URING)
679   __io_fb_init();      // worker rings init on their own threads (single_issuer)
680   __io.futex_ok.store(micron::kernel::has(micron::kernel::feature::uring_futex) ? 1u : 0u, micron::memory_order_relaxed);
681   __io.watcher.store(-1, micron::memory_order_relaxed);
682   __io_cancel_hook = &__io_cancel_walker;      // cancellation_source::cancel() reaches in-flight ops
683 #endif
684   __global_engine = e;
685   for ( u32 i = 0; i < nworkers; ++i )
686     e->threads[i] = micron::solo::spawn<micron::auto_thread<>>([](engine *eng, u32 id) { eng->worker_main(id); }, e, i);
687   // the timer thread spawns lazily on the first sleep/timer
688   __engine_state.store(2u, micron::memory_order_release);
689 }

691 inline void
692 stop_coroutine_runtime() noexcept
693 {
694   engine *e = __global_engine;
695   if ( e == nullptr ) return;
696 
697   {
698     u32 __quiet = 0;
699     timespec_t __tm_dl{ 0, 0 };      // re-armed whenever a timer fires
700     u32 __tm_seen = 0;
701     bool __tm_armed = false;
702     bool __tm_cut = false;
703 #if defined(MICRON_CORO_URING)
704     timespec_t __io_dl{ 0, 0 };      // grace deadline; re-armed whenever io makes progress
705     u64 __seen = 0;                  // in-flight count the deadline was armed against
706     bool __armed = false;
707 #endif
708     while ( __quiet < 4 ) {
709       // WARNING: a still-armed timer may _NEVER_ stall indefinitely
710       const u32 __tm = e->pending_timers.get(micron::memory_order_acquire);
711       if ( __tm == 0 ) {
712         __tm_armed = false;
713       } else if ( !__tm_cut ) {
714         timespec_t __now{};
715         micron::clock_gettime(micron::clock_monotonic, __now);
716         if ( !__tm_armed || __tm != __tm_seen ) {
717           __tm_seen = __tm;
718           __tm_armed = true;
719           __tm_dl = __now;
720           __ts_add_ns(__tm_dl, __cl_timer_drain_grace_ns);
721         } else if ( !__ts_le(__now, __tm_dl) ) {
722           __tm_cut = true;
723         }
724         if ( !__tm_cut ) {
725           timespec_t __nap{ 0, 200000 };
726           (void)micron::nanosleep(__nap);
727           __quiet = 0;
728           continue;
729         }
730       }
731       bool __q = e->inbox.empty() && (__tm == 0 || __tm_cut);
732 #if defined(MICRON_CORO_URING)
733       if ( __q && !e->__io_ovf_empty() ) __q = false;
734 #endif
735       for ( u32 __i = 0; __q && __i < e->n; ++__i )
736         if ( e->workers[__i].active.get(micron::memory_order_acquire) != 0 || !e->workers[__i].deque.empty()
737              || e->workers[__i].ovf != nullptr )
738           __q = false;
739 #if defined(MICRON_CORO_URING)
740       if ( __q ) {
741         const u64 __pend = __io_pending_total();
742         if ( __pend != 0 ) {
743           timespec_t __now{};
744           micron::clock_gettime(micron::clock_monotonic, __now);
745           if ( !__armed || __pend != __seen ) {
746             __seen = __pend;
747             __armed = true;
748             __io_dl = __now;
749             __ts_add_ns(__io_dl, __cl_io_drain_grace_ns);
750           } else if ( !__ts_le(__now, __io_dl) ) {      // grace burned with zero progress
751             micron::uring::sync_cancel_reg __sc{};
752             __sc.fd = -1;
753             __sc.flags = micron::uring::async_cancel_any;
754             for ( u32 __i = 0; __i < e->n; ++__i ) {
755               __wring &__wr = __io_rings[__i];
756               if ( __wr.__live.get(micron::memory_order_acquire) != 0 )
757                 (void)micron::uring::__io_uring_register(__wr.__r.fd, micron::uring::reg_register_sync_cancel, &__sc, 1);
758             }
759             if ( __io_fb.__live.get(micron::memory_order_acquire) != 0 )
760               (void)micron::uring::__io_uring_register(__io_fb.__r.fd, micron::uring::reg_register_sync_cancel, &__sc, 1);
761             __armed = false;      // re-arm: a cancel that does not land must not spin the register syscall
762           }
763           __q = false;
764           // the grace window is seconds long: poll it, do not sched_yield a core
765           // flat out while the workers are the ones that need the cpu
766           timespec_t __nap{ 0, 200000 };
767           (void)micron::nanosleep(__nap);
768           __quiet = 0;
769           continue;
770         }
771         __armed = false;
772       }
773 #endif
774       if ( __q )
775         ++__quiet;
776       else
777         __quiet = 0;
778       micron::yield();
779     }
780   }
781 
782   e->stopping.store(1, micron::memory_order_release);
783   if ( __timer_state.get(micron::memory_order_acquire) != 0u ) {
784     __timer_epoch.fetch_add(1, micron::memory_order_release);      // the timer may be in an indefinite wait
785     micron::wake_futex(__timer_epoch.ptr(), 1);
786     __timer_thread.reset();
787     __timer_state.store(0u, micron::memory_order_release);
788   }
789   __timer_lock();
790   __timer_head = nullptr;
791   __timer_unlock();
792   e->pending_timers.store(0, micron::memory_order_release);
793 #if defined(MICRON_CORO_GLOBAL_SIGNAL)
794   __cl_signal.fetch_add(1, micron::memory_order_release);
795   micron::wake_futex(__cl_signal.ptr(), static_cast<int>(e->n));
796 #else
797   for ( u32 i = 0; i < e->n; ++i ) {
798     __cl_park[i].epoch.fetch_add(1, micron::memory_order_release);
799     micron::wake_futex(__cl_park[i].epoch.ptr(), 1);
800   }
801 #endif
802   for ( u32 i = 0; i < e->n; ++i ) e->threads[i].reset();
803   micron::fiber::drain_seg_pool();      // segments parked by crossworker finalize
804 #if defined(MICRON_CORO_URING)
805   __io_cancel_hook = nullptr;
806   __io_fb_shutdown();
807   __io_fixed_shutdown();
808   __io.any_live.store(0, micron::memory_order_release);
809   __io.watcher.store(-1, micron::memory_order_relaxed);
810 #endif
811   __global_engine = nullptr;
812   delete e;
813   __engine_state.store(0u, micron::memory_order_release);
814 }
