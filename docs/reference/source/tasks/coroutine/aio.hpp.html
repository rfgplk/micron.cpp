47   template<class P>
48   bool
49   await_suspend(std::coroutine_handle<P> __h) noexcept
50   {
51     __frame_base *__f = &__h.promise();
52     __op.__f = __f;
53     if ( reinterpret_cast<u64>(&__op) >> __io_ud_tag_shift != 0 ) [[unlikely]]
54       __builtin_trap();      // non-canonical op address would alias a user_data tag
55     __wring *__r = __io_own_ring();
56     const bool __own = (__r != nullptr);
57     if ( !__own ) {
58       if ( __io_fb.__live.get(micron::memory_order_acquire) == 0 ) {
59         __op.__res = -38;
60         return false;
61       }
62       __r = &__io_fb;
63     }
64     if ( __fixed != 0 && !__io_fixed_reg(*__r) ) {
65       __op.__res = -95;      // EOPNOTSUPP: slab registration refused (memlock/old kernel)
66       return false;
67     }
68     const micron::atomic_token<u32> *__cxf = __f->__cancel;
69     if ( __cxf != nullptr ) {
70       if ( __cxf->get(micron::memory_order_acquire) != 0 ) {
71         __op.__res = -125;      // ECANCELED before submission
72         return false;
73       }
74       __cxl.__flag = __cxf;
75       __cxl.__ud = reinterpret_cast<u64>(&__op);
76       __cxl.__ring_fd = __r->__r.fd;
77       __io_cxl_push(&__cxl);
78     }
79     __r->__pending.fetch_add(1, micron::memory_order_acq_rel);
80     bool __ok = __own ? __io_submit_own(*__r, __sqe, reinterpret_cast<u64>(&__op)) : __io_submit_fb(__sqe, reinterpret_cast<u64>(&__op));
81     if ( !__ok ) [[unlikely]] {
82       // kernel refused our enter (CQ overflow backpressure)
83       if ( __io_cq_acquire(*__r) ) {
84         micron::uring::cqe __c{};
85         while ( __r->__r.peek_cqe(&__c) ) __global_engine->__dispatch_cqe(*__r, __c);
86         __io_unlock(__r->__cq_lk);
87       }
88       __ok = __own ? __io_submit_own(*__r, __sqe, reinterpret_cast<u64>(&__op)) : __io_submit_fb(__sqe, reinterpret_cast<u64>(&__op));
89     }
90     if ( !__ok ) {
91       __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
92       // ENOBUFS
93       __op.__res = -105;
94       return false;
95     }
96     if ( __cxf != nullptr && __cxf->get(micron::memory_order_acquire) != 0 )
97       __io_sync_cancel_ud(__r->__r.fd, reinterpret_cast<u64>(&__op));      // cancel raced the submit
98     if ( __io_cq_acquire(*__r) ) {
99       micron::uring::cqe __c{};
100       u32 __budget = 8;
101       bool __mine = false;
102       while ( __budget-- != 0 && __r->__r.peek_cqe(&__c) ) {
103         if ( __io_ud_tag(__c.user_data) == __io_ud_op && reinterpret_cast<__io_op *>(__c.user_data) == &__op ) {
104           __op.__res = __c.res;
105           __mine = true;
106           break;
107         }
108         __global_engine->__dispatch_cqe(*__r, __c);      // foreign cqe: normal completer path
109       }
110       __io_unlock(__r->__cq_lk);
111       if ( __mine ) {
112         __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
113 #if defined(MICRON_CORO_STATS)
114         __r->__stat.inline_completions.fetch_add(1, micron::memory_order_relaxed);
115 #endif
116         return false;
117       }
118     }
119     // commit to suspension; on failure a completer already delivered __res
120     u32 __exp = __io_st_submitted;
121     if ( __op.__st.compare_exchange_strong(__exp, __io_st_suspended, micron::memory_order_acq_rel, micron::memory_order_acquire) )
122       return true;      // LAST access to this awaitable/frame: the frame may already be running elsewhere
123     __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
124     return false;
125   }

127   i32
128   await_resume() noexcept
129   {
130     if ( __cxl.__flag != nullptr ) {
131       __io_cxl_unlink(&__cxl);
132       __cxl.__flag = nullptr;
133     }
134     return __op.__res;
135   }

177   {

180     if ( reinterpret_cast<u64>(&__op) >> __io_ud_tag_shift != 0 ) [[unlikely]]

182     __wring *__r = __io_own_ring();
183     const bool __own = (__r != nullptr);
184     if ( !__own ) {
185       if ( __io_fb.__live.get(micron::memory_order_acquire) == 0 ) {
186         __op.__res = -38;
187         return false;
188       }
189       __r = &__io_fb;
190     }
191     if ( __fixed != 0 && !__io_fixed_reg(*__r) ) {      // same lazy per-ring registration the untimed path does
192       __op.__res = -95;                                 // EOPNOTSUPP: slab registration refused (memlock/old kernel)
193       return false;
194     }
195     const micron::atomic_token<u32> *__cxf = __f->__cancel;
196     if ( __cxf != nullptr ) {
197       if ( __cxf->get(micron::memory_order_acquire) != 0 ) {
198         __op.__res = -125;
199         return false;
200       }
201       __cxl.__flag = __cxf;
202       __cxl.__ud = reinterpret_cast<u64>(&__op);
203       __cxl.__ring_fd = __r->__r.fd;
204       __io_cxl_push(&__cxl);
205     }
206     micron::uring::sqe __lt;
207     micron::uring::prep_link_timeout(&__lt, &__kts);      // kernel copies the timespec at prep
208     __r->__pending.fetch_add(1, micron::memory_order_acq_rel);
209     bool __ok = __own ? __io_submit_own2(*__r, __sqe, reinterpret_cast<u64>(&__op), __lt, __io_ud_make(__io_ud_ltimer, 0))
210                       : __io_submit_fb2(__sqe, reinterpret_cast<u64>(&__op), __lt, __io_ud_make(__io_ud_ltimer, 0));
211     if ( !__ok ) [[unlikely]] {
212       // reap then retry
213       if ( __io_cq_acquire(*__r) ) {
214         micron::uring::cqe __c{};
215         while ( __r->__r.peek_cqe(&__c) ) __global_engine->__dispatch_cqe(*__r, __c);
216         __io_unlock(__r->__cq_lk);
217       }
218       __ok = __own ? __io_submit_own2(*__r, __sqe, reinterpret_cast<u64>(&__op), __lt, __io_ud_make(__io_ud_ltimer, 0))
219                    : __io_submit_fb2(__sqe, reinterpret_cast<u64>(&__op), __lt, __io_ud_make(__io_ud_ltimer, 0));
220     }
221     if ( !__ok ) {
222       __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
223       __op.__res = -105;      // ENOBUFS
224       return false;
225     }
226     if ( __cxf != nullptr && __cxf->get(micron::memory_order_acquire) != 0 ) __io_sync_cancel_ud(__r->__r.fd, reinterpret_cast<u64>(&__op));
227     if ( __io_cq_acquire(*__r) ) {
228       micron::uring::cqe __c{};
229       u32 __budget = 8;
230       bool __mine = false;
231       while ( __budget-- != 0 && __r->__r.peek_cqe(&__c) ) {
232         if ( __io_ud_tag(__c.user_data) == __io_ud_op && reinterpret_cast<__io_op *>(__c.user_data) == &__op ) {
233           __op.__res = __c.res;
234           __mine = true;
235           break;
236         }
237         __global_engine->__dispatch_cqe(*__r, __c);
238       }
239       __io_unlock(__r->__cq_lk);
240       if ( __mine ) {
241         __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
242 #if defined(MICRON_CORO_STATS)
243         __r->__stat.inline_completions.fetch_add(1, micron::memory_order_relaxed);
244 #endif
245         return false;
246       }
247     }
248     u32 __exp = __io_st_submitted;
249     if ( __op.__st.compare_exchange_strong(__exp, __io_st_suspended, micron::memory_order_acq_rel, micron::memory_order_acquire) )
250       return true;      // LAST access
251     __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
252     return false;
253   }

543 [[nodiscard]] inline i32
544 mrecv_arm(mrecv_t &__mr, i32 __fd, u32 __msg_flags = 0) noexcept
545 {
546   worker *__w = current_worker();
547   __wring *__r = __io_own_ring();
548   if ( __w == nullptr || __r == nullptr ) return -95;      // EOPNOTSUPP
549   if ( __io_pb_init(*__r) != 0 ) return -95;
550   __io_mop &__m = __mr.__op;
551   if ( __m.__live.get(micron::memory_order_acquire) != 0 ) return -114;      // EALREADY armed
552   if ( reinterpret_cast<u64>(&__m) >> __io_ud_tag_shift != 0 ) [[unlikely]]
553     __builtin_trap();      // non-canonical address would alias the user_data tag
554   __m.__f = nullptr;
555   __m.__st.store(__io_mop_idle, micron::memory_order_relaxed);
556   __m.__ring = static_cast<u8>(__w->id);
557   __m.__live.store(1u, micron::memory_order_release);
558   micron::uring::sqe __q;
559   micron::uring::prep_recv_multishot(&__q, __fd, __io_pb_bgid, __msg_flags);
560   const u64 __ud = __io_ud_make(__io_ud_mop, reinterpret_cast<u64>(&__m));

569     __ok = __io_submit_own(*__r, __q, __ud);
570   }
571   if ( !__ok ) {
572     __r->__pending.sub_fetch(1, micron::memory_order_acq_rel);
573     __m.__live.store(0u, micron::memory_order_release);
574     return -105;      // ENOBUFS: ring full even after flush+reap
575   }
576   return 0;

585   await_ready() noexcept
586   {
587     if ( __m->__pop(__e) ) {

613     if ( __m->__st.compare_exchange_strong(__exp, __io_mop_parked, micron::memory_order_seq_cst, micron::memory_order_seq_cst) )
614       return true;      // last access to this frame; may already be running elsewhere
615     __m->__st.store(__io_mop_idle, micron::memory_order_release);
616     return false;
617   }

623     if ( __m->__pop(__e) ) return __e;

661     const u8 __rid = __m.__ring;

664       const u64 __ud = __io_ud_make(__io_ud_mop, reinterpret_cast<u64>(&__m));
665       __io_sync_cancel_ud(__wr.__r.fd, __ud);
666       for ( u32 __spin = 0; __m.__live.get(micron::memory_order_acquire) != 0; ++__spin ) {
667         if ( __io_cq_acquire(__wr) ) {
668           micron::uring::cqe __c{};

672         }
