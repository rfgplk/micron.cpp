24 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
25 // stackful cactus fiber
26 //
27 // fiber is a schedulable execution context that runs on its own contiguous stack segment
28 // can be suspended at any point (including inside arbitrary blocking code)
29 //
30 // the fiber control block (FCB) is embedded at the top of its own stack region (highest address)
31 
32 namespace micron
33 {
34 namespace fiber
35 {
36 
37 enum class fiber_state : u32 { fresh = 0, running, suspended, done };
38 
39 struct alignas(64) fiber {
40   ar::__fiber_ctx ctx;                   // this fiber's saved register context (resume point)
41   ar::__fiber_ctx *link = nullptr;       // where to swap on suspend/finish (host worker's scheduler ctx)
42   __region region;                       // backing stack run (FCB is embedded inside it)
43   byte *frame_sp = nullptr;              // coroutine-frame bump cursor (M3)
44   fiber *parent = nullptr;               // cactus spine link
45   void (*entry)(fiber *) = nullptr;      // user entry, invoked by __micron_fiber_enter
46   void *arg = nullptr;                   // user arg
47   void *coro = nullptr;                  // std::coroutine_handle<>::address() (M3); void* to avoid <coroutine> here
48   micron::atomic_token<u32> state{ static_cast<u32>(fiber_state::fresh) };
49   micron::atomic_token<u32> refs{ 1 };
50   i32 owner_worker = -1;           // micthread::self() of the allocating worker (cross-worker free routing)
51   bool escaped = false;            // an exception escaped entry(); the engine (cl_sched __run) fails fast on it
52   fiber *free_next = nullptr;      // per-worker recycle free-list link
53 };

62 [[gnu::always_inline]] inline i32
63 __self_tid() noexcept
64 {
65   i32 t = __cached_tid;
66   if ( t < 0 ) [[unlikely]]
67     t = __cached_tid = static_cast<i32>(micthread::self());
68   return t;

83 inline void
84 __micron_fiber_enter(void *p) noexcept
85 {
86   fiber *f = static_cast<fiber *>(p);
87   f->state.store(static_cast<u32>(fiber_state::running), micron::memory_order_relaxed);
88 #if !defined(__micron_freestanding) || defined(__micron_eh)
89   try {
90     f->entry(f);
91   } catch ( ... ) {
92     f->escaped = true;      // an exception escaped the noexcept coroutine chain; cl_sched __run fails fast on it
93   }
94 #else
95   f->entry(f);
96 #endif
97   f->state.store(static_cast<u32>(fiber_state::done), micron::memory_order_release);
98   ar::__micron_swap_context(&f->ctx, f->link);      // final park back to the scheduler; never resumed
99   __builtin_unreachable();
100 }

102 // create a fiber that will run entry(fcb) on a fresh (or recycled) guard-paged stack segment.
103 [[nodiscard]] inline fiber *
104 create_fiber(void (*entry)(fiber *), void *arg, usize stack_size = static_cast<usize>(micro_stack_size), fiber *parent = nullptr) noexcept
105 {
106   const u32 cls = __seg_class_for(stack_size);
107   fiber *f = nullptr;
108 
109   if ( __seg_freelist[cls] != nullptr ) {      // recycle a same-class FCB+region
110     f = __seg_freelist[cls];
111     __seg_freelist[cls] = f->free_next;
112   } else if ( (f = __seg_pool[cls].pop()) != nullptr ) {
113     // recycled
114   } else {
115     __region reg;
116     if ( !__carve_region(reg, cls) ) return nullptr;
117     // embed the FCB at the top of the C-stack region, cache-line aligned
118     const usize fcb_addr = (reinterpret_cast<usize>(reg.stack_top) - sizeof(fiber)) & ~static_cast<usize>(63);
119     f = reinterpret_cast<fiber *>(fcb_addr);
120     new (static_cast<void *>(f)) fiber();
121     f->region = reg;
122   }
123 
124   // execution stack top sits just below the embedded FCB (16-aligned)
125   byte *stack_top = reinterpret_cast<byte *>(reinterpret_cast<usize>(f) & ~static_cast<usize>(15));
126 
127   f->link = &__worker_sched_ctx;
128   f->parent = parent;
129   f->entry = entry;
130   f->arg = arg;
131   f->coro = nullptr;
132   f->frame_sp = f->region.frame_base;                  // coroutine-frame arena bump cursor (grows up)
133   f->refs.store(1, micron::memory_order_relaxed);      // execution holds the initial ref
134   f->owner_worker = __self_tid();
135   f->escaped = false;
136   f->free_next = nullptr;
137   // relaxed: the fiber is unpublished until the creating thread resumes or hands it off
138   f->state.store(static_cast<u32>(fiber_state::fresh), micron::memory_order_relaxed);
139 
140   ar::make_context(&f->ctx, stack_top, &__micron_fiber_enter, f);
141   return f;
142 }

144 // run f from the current (host) context; returns when f suspends or finishes.
145 [[gnu::always_inline]] inline void
146 resume(fiber *f, ar::__fiber_ctx *host = &__worker_sched_ctx) noexcept
147 {
148   f->link = host;
149   fiber *prev = __current_fiber;
150   __current_fiber = f;
151   ar::__micron_swap_context(host, &f->ctx);      // -> trampoline (fresh) or the saved resume point
152   __current_fiber = prev;                        // control returned to the host
153 }

155 [[gnu::always_inline]] inline void
156 suspend_to_scheduler() noexcept
157 {
158   fiber *f = __current_fiber;
159   f->state.store(static_cast<u32>(fiber_state::suspended), micron::memory_order_release);
160   ar::__micron_swap_context(&f->ctx, f->link);
161 }

170 inline void
171 __finalize_fiber(fiber *f) noexcept
172 {
173   if ( f->owner_worker == __self_tid() ) {
174     f->free_next = __seg_freelist[f->region.cls];
175     __seg_freelist[f->region.cls] = f;      // keep the FCB constructed for reuse
176     return;
177   }
178   if ( __seg_pool[f->region.cls].push(f) ) return;
179   __region reg = f->region;
180   f->~fiber();
181   __release_region(reg);
182 }

184 [[gnu::always_inline]] inline void
185 __drop_ref(fiber *f) noexcept
186 {
187   // acq_rel so the release pairs with the acquire that observes refs==0 -> all prior frame writes are visible
188   if ( f->refs.sub_fetch(1, micron::memory_order_acq_rel) == 0 ) __finalize_fiber(f);
189 }

191 // a coroutine frame must satisfy the ABI's strictest alignment, not just 16
192 // gcc freely places an over aligned local at a frame offset computed as if the frame base were
193 // __BIGGEST_ALIGNMENT__-aligned
194 // max(__BIGGEST_ALIGNMENT__, 16), selected by the preprocessor: on a target where the two coincide
195 // a ?: here reads as a condition with identical branches
196 #if __BIGGEST_ALIGNMENT__ < 16
197 inline constexpr usize __frame_align = usize{ 16 };
198 #else
199 inline constexpr usize __frame_align = usize{ __BIGGEST_ALIGNMENT__ };
200 #endif
201 inline constexpr usize __frame_hdr = __frame_align;
202 
203 [[gnu::always_inline]] inline usize
204 __frame_round(usize n) noexcept
205 {
206   return (n + (__frame_align - 1)) & ~(__frame_align - 1);
207 }
208 
209 // over-allocate and hand back an aligned interior pointer
210 [[gnu::always_inline]] inline void *
211 __frame_alloc_heap(usize need) noexcept
212 {
213   byte *raw = static_cast<byte *>(::operator new(need + __frame_align));
214   byte *h = reinterpret_cast<byte *>((reinterpret_cast<usize>(raw + __frame_hdr) + (__frame_align - 1)) & ~(__frame_align - 1));
215   *micron::ptr_cast<fiber **>(h - __frame_hdr) = nullptr;
216   *micron::ptr_cast<byte **>(h - __frame_hdr + sizeof(void *)) = raw;
217   return h;
218 }
219 
220 [[gnu::always_inline]] inline void *
221 __frame_alloc(usize n) noexcept
222 {
223   fiber *f = __current_fiber;
224   const usize need = __frame_hdr + __frame_round(n);
225   if ( f == nullptr ) return __frame_alloc_heap(need);
226   byte *h = reinterpret_cast<byte *>((reinterpret_cast<usize>(f->frame_sp) + (__frame_align - 1)) & ~(__frame_align - 1));
227   byte *next = h + need;
228   if ( next > f->region.frame_limit ) [[unlikely]]
229     return __frame_alloc_heap(need);
230   f->frame_sp = next;
231   *micron::ptr_cast<fiber **>(h) = f;
232   f->refs.fetch_add(1, micron::memory_order_relaxed);
233   return h + __frame_hdr;
234 }

236 [[gnu::always_inline]] inline void
237 __frame_free(void *p, usize n) noexcept
238 {
239   if ( p == nullptr ) return;
240   byte *h = reinterpret_cast<byte *>(p) - __frame_hdr;
241   fiber *owner = *micron::ptr_cast<fiber **>(h);
242   if ( owner == nullptr ) {
243     ::operator delete(static_cast<void *>(*micron::ptr_cast<byte **>(h + sizeof(void *))));
244     return;
245   }
246   if ( owner == __current_fiber && h + __frame_hdr + __frame_round(n) == owner->frame_sp ) owner->frame_sp = h;
247   __drop_ref(owner);
248 }

250 inline void
251 destroy_fiber(fiber *f) noexcept
252 {
253   if ( f == nullptr ) return;
254   __drop_ref(f);
255 }

257 inline void
258 drain_freelist() noexcept
259 {
260   for ( u32 c = 0; c < __seg_class_count; ++c ) {
261     fiber *f = __seg_freelist[c];
262     __seg_freelist[c] = nullptr;
263     while ( f != nullptr ) {
264       fiber *nx = f->free_next;
265       __region reg = f->region;
266       f->~fiber();
267       __release_region(reg);
268       f = nx;
269     }
270   }
271 }
272 
273 inline void
274 drain_seg_pool() noexcept
275 {
276   for ( u32 c = 0; c < __seg_class_count; ++c )
277     __seg_pool[c].drain([](fiber *f) {
278       __region reg = f->region;
279       f->~fiber();
280       __release_region(reg);
281     });
282 }
