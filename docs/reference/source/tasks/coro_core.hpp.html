70 [[gnu::always_inline]] inline void
71 __notify_work() noexcept
72 {
73   if ( __cl_sleepers.get(micron::memory_order_relaxed) != 0 ) {

86 // claim-and-wake one parked worker
87 template<bool Strong>
88 [[gnu::always_inline]] inline void
89 __cl_wake_one() noexcept
90 {
91   u32 __m;
92   if constexpr ( Strong )
93     __m = __cl_sleeper_mask.fetch_or(0, micron::memory_order_seq_cst);
94   else
95     __m = __cl_sleeper_mask.get(micron::memory_order_relaxed);
96   while ( __m != 0 ) {
97     const u32 __i = static_cast<u32>(__builtin_ctz(__m));
98     const u32 __bit = 1u << __i;
99     const u32 __old = __cl_sleeper_mask.fetch_and(~__bit, micron::memory_order_acq_rel);
100     if ( __old & __bit ) {      // claimed an actual sleeper
101       __cl_park[__i].epoch.fetch_add(1, micron::memory_order_release);
102       micron::wake_futex(__cl_park[__i].epoch.ptr(), 1);
103       return;
104     }
105     __m = __old & ~__bit;      // raced the sleepers own clear
106   }
107 }

122 [[gnu::always_inline]] inline std::coroutine_handle<>
123 __child_complete(__frame_base *child) noexcept
124 {
125   __frame_base *parent = child->__parent;
126   worker *w = __cur_worker;
127   __frame_base *cont = (w != nullptr) ? w->deque.pop_bottom() : nullptr;
128   // balanced pop
129   if ( cont == parent && cont != nullptr && cont->__pushed_kind == __frame_base::__kind_fork ) return parent->__self;
130   // detached
131   const u32 r = parent->__joins.add_fetch(1, micron::memory_order_acq_rel);
132   const bool ready = (r & __frame_base::__joiner_bit)
133                      && (r & __frame_base::__credit_mask) == ((r >> __frame_base::__expected_shift) & __frame_base::__expected_mask);
134   if ( cont != nullptr ) {
135     if ( cont == parent ) return parent->__self;
136     __count_take(cont);
137     if ( ready ) {
138       cont->__pushed_kind = __frame_base::__kind_plain;
139       if ( w->deque.push_bottom(cont) ) [[likely]] {
140         __notify_work();
141       } else {
142         // deque full
143         // WARNING: __NEVER__ spin here; this thread is the only valid popper, so yielding cannot make room
144         cont->__ovf_next = w->ovf;
145         w->ovf = cont;
146       }
147       return parent->__self;
148     }
149     return cont->__self;
150   }
151   return ready ? parent->__self : std::noop_coroutine();
152 }

154 [[gnu::always_inline]] inline std::coroutine_handle<>
155 __join_suspend(__frame_base *self) noexcept
156 {
157   const u32 s = self->__steals.get(micron::memory_order_acquire);
158   if ( s == 0 ) return self->__self;
159   if ( s > __frame_base::__expected_mask ) [[unlikely]]
160     __builtin_trap();      // > 32767 outstanding steals in one join epoch exceeds the packed word
161   // TODO: error out properly
162   const u32 r = self->__joins.add_fetch(__frame_base::__joiner_bit | (s << __frame_base::__expected_shift), micron::memory_order_acq_rel);
163   if ( (r & __frame_base::__credit_mask) == s ) {
164     self->__joins.store(0, micron::memory_order_relaxed);
165     self->__steals.store(0, micron::memory_order_relaxed);
166     return self->__self;
167   }
168   return std::noop_coroutine();
169 }

171 [[gnu::always_inline]] inline void
172 __join_resume(__frame_base *self) noexcept
173 {
174   self->__joins.store(0, micron::memory_order_relaxed);
175   self->__steals.store(0, micron::memory_order_relaxed);
176   __frame_base *__c = self->__child_head;
177   self->__child_head = nullptr;
178   while ( __c != nullptr ) {
179     __frame_base *__next = __c->__sibling;
180     if ( __c->__self ) __c->__self.destroy();
181     __c = __next;
182   }
183 }
