23 // pmap
24 //
25 // persistent (immutable) hash array mapped trie (HAMT) with 32-way branching
26 // every mutation returns a fresh pmap that path-copies the modified spine and
27 // shares the rest with the source
28 //
29 // reclamation is driven by atomic intrusive refcounts on each shared node;
30 // the count saturates at __immortal_refs and the node becomes permanently live
31 //
32 // COLLISION CHAINS: keys whose 64-bit hashes are equal (or that stay equal past
33 // __max_depth re-mixes) share a leaf's overflow chain, which has no length bound
34 template<typename K, typename V>
35   requires micron::is_copy_constructible_v<K> and micron::is_copy_constructible_v<V> and micron::is_move_constructible_v<K>
36            and micron::is_move_constructible_v<V>
37 class pmap

39   static constexpr usize __log2B = 5;
40   static constexpr usize __B = 1u << __log2B;      // 32-way branching
41   static constexpr u32 __mask32 = static_cast<u32>(__B - 1u);
42   static constexpr usize __natural_depth = 12;                            // 12 * 5 = 60 natural hash bits
43   static constexpr usize __max_depth = 24;                                // one re-mix grants 60 more bits

172   __copy_chain(const __chain *src)
173   {
174     __chain *head = nullptr;
175     __chain **tail = &head;

192   // once a refcount reaches __immortal_refs the node is treated as permanently live
193   static inline __attribute__((always_inline)) void
194   __retain_refs(u32 *p) noexcept
195   {
196     u32 cur = atom::load(p, atomic_relaxed);
197     for ( ;; ) {
198       if ( cur == __immortal_refs ) [[unlikely]]
199         return;
200       if ( atom::compare_exchange(p, &cur, cur + 1u, true, atomic_relaxed, atomic_relaxed) ) return;
201     }
202   }
203 
204   static inline __attribute__((always_inline)) bool
205   __dec_refs(u32 *p) noexcept
206   {
207     u32 cur = atom::load(p, atomic_relaxed);
208     for ( ;; ) {
209       if ( cur == __immortal_refs ) [[unlikely]]
210         return false;
211       if ( atom::compare_exchange(p, &cur, cur - 1u, true, atomic_release, atomic_relaxed) ) {
212         if ( cur != 1u ) [[likely]]
213           return false;
214         atom::thread_fence(atomic_acquire);
215         return true;
216       }
217     }
218   }

230   static void
231   __release(uintptr_t p) noexcept
232   {
233     if ( !p ) return;
234     if ( __is_leaf(p) ) {
235       __leaf *l = __as_leaf(p);
236       if ( !__dec_refs(&l->refs) ) return;
237       __destroy_chain(l->overflow);
238       delete l;
239       return;
240     }
241     __internal *in = __as_internal(p);
242     if ( !__dec_refs(&in->refs) ) return;
243     usize n = static_cast<usize>(micron::popcount(in->bitmap));
244     uintptr_t *cs = __children_of(in);
245     for ( usize i = 0; i < n; ++i ) __release(cs[i]);
246     in->~__internal();
247     ::operator delete(in);
248   }

282   __idx_at(hash64_t h, usize depth) noexcept
283   {
284     if ( depth < __natural_depth ) [[likely]]
285       return static_cast<u32>(h >> (depth * __log2B)) & __mask32;
286     return static_cast<u32>(__mix_hash(h) >> ((depth - __natural_depth) * __log2B)) & __mask32;

289   static __internal *
290   __internal_replace(const __internal *src, u32 logical_idx, uintptr_t new_child)
291   {
292     u32 bm = src->bitmap;
293     usize n = static_cast<usize>(micron::popcount(bm));
294     __internal *out = __alloc_internal(bm);
295     const uintptr_t *src_cs = __children_of(src);
296     uintptr_t *dst_cs = __children_of(out);
297     usize phys = __phys_idx(bm, logical_idx);
298     for ( usize i = 0; i < n; ++i ) {
299       if ( i == phys ) {
300         dst_cs[i] = new_child;
301       } else {
302         dst_cs[i] = src_cs[i];
303         __retain(dst_cs[i]);
304       }
305     }
306     return out;
307   }

309   static __internal *
310   __internal_insert(const __internal *src, u32 logical_idx, uintptr_t new_child)
311   {
312     u32 nbm = src->bitmap | (1u << logical_idx);
313     __internal *out = __alloc_internal(nbm);
314     usize n_old = static_cast<usize>(micron::popcount(src->bitmap));
315     const uintptr_t *src_cs = __children_of(src);
316     uintptr_t *dst_cs = __children_of(out);
317     usize phys = __phys_idx(nbm, logical_idx);
318     for ( usize i = 0; i < phys; ++i ) {
319       dst_cs[i] = src_cs[i];
320       __retain(dst_cs[i]);
321     }
322     dst_cs[phys] = new_child;
323     for ( usize i = phys; i < n_old; ++i ) {
324       dst_cs[i + 1] = src_cs[i];
325       __retain(dst_cs[i + 1]);
326     }
327     return out;
328   }

330   static __internal *
331   __internal_remove(const __internal *src, u32 logical_idx)
332   {
333     u32 nbm = src->bitmap & ~(1u << logical_idx);
334     if ( nbm == 0 ) return nullptr;
335     __internal *out = __alloc_internal(nbm);
336     usize n_old = static_cast<usize>(micron::popcount(src->bitmap));
337     const uintptr_t *src_cs = __children_of(src);
338     uintptr_t *dst_cs = __children_of(out);
339     usize phys = __phys_idx(src->bitmap, logical_idx);
340     for ( usize i = 0; i < phys; ++i ) {
341       dst_cs[i] = src_cs[i];
342       __retain(dst_cs[i]);
343     }
344     for ( usize i = phys + 1; i < n_old; ++i ) {
345       dst_cs[i - 1] = src_cs[i];
346       __retain(dst_cs[i - 1]);
347     }
348     return out;
349   }

424   {

446   static __internal *
447   __split_leaf_with_extra(const __leaf *l, hash64_t new_h, const K &new_k, V new_v, usize depth)
448   {
449     u32 bm = 0;
450     bm |= 1u << __idx_at(l->h, depth);
451     bm |= 1u << __idx_at(new_h, depth);
452     for ( const __chain *c = l->overflow; c; c = c->next ) bm |= 1u << __idx_at(c->h, depth);
453 
454     __internal *in = __alloc_internal(bm);
455     uintptr_t *cs = __children_of(in);
456 
457     auto place = [&]<typename Vf>(hash64_t h, const K &k, Vf &&v) {
458       u32 idx = __idx_at(h, depth);
459       __child_absorb(&cs[__phys_idx(bm, idx)], h, k, micron::forward<Vf>(v));
460     };
461 
462 #if !defined(__micron_freestanding) || defined(__micron_eh)
463     try {
464 #endif
465       place(l->h, l->key, l->value);
466       for ( const __chain *c = l->overflow; c; c = c->next ) place(c->h, c->key, c->value);
467       place(new_h, new_k, micron::move(new_v));
468 #if !defined(__micron_freestanding) || defined(__micron_eh)
469     } catch ( ... ) {
470       __release(__tag_internal_ptr(in));
471       throw;
472     }
473 #endif
474 
475     return in;
476   }

478   static uintptr_t
479   __insert_into(uintptr_t n, hash64_t h, const K &k, V v, usize depth, bool &existed)
480   {
481     if ( n == 0 ) {
482       existed = false;
483       return __tag_leaf_ptr(new __leaf(h, k, micron::move(v)));
484     }
485     if ( __is_leaf(n) ) {
486       const __leaf *l = __as_leaf(n);
487 
488       if ( l->h == h && l->key == k ) {
489         existed = true;
490         __leaf *nl = new __leaf(h, k, micron::move(v));
491         __node_guard g{ __tag_leaf_ptr(nl) };
492         nl->overflow_len = l->overflow_len;
493         nl->overflow = __copy_chain(l->overflow);      // if it throws, g frees nl (overflow still null)
494         g.dismiss();
495         return __tag_leaf_ptr(nl);
496       }
497 
498       if ( __chain_has(l->overflow, h, k) ) {
499         existed = true;
500         __leaf *nl = new __leaf(l->h, l->key, l->value);
501         __node_guard g{ __tag_leaf_ptr(nl) };
502         nl->overflow_len = l->overflow_len;
503         nl->overflow = __chain_copy_with_replace(l->overflow, h, k, micron::move(v));
504         g.dismiss();
505         return __tag_leaf_ptr(nl);
506       }
507 
508       existed = false;
509       u32 total_after = 1u + l->overflow_len + 1u;
510       if ( total_after > __split_threshold && depth < __max_depth && __leaf_would_diverge(l, h, depth) ) {
511         return __tag_internal_ptr(__split_leaf_with_extra(l, h, k, micron::move(v), depth));
512       }
513 
514       __leaf *nl = new __leaf(l->h, l->key, l->value);
515       __node_guard g{ __tag_leaf_ptr(nl) };
516       nl->overflow_len = l->overflow_len + 1u;
517       __chain *rest = __copy_chain(l->overflow);
518       nl->overflow = rest;      // nl now owns rest; if the prepend below throws, g frees nl + rest
519       nl->overflow = new __chain(h, k, micron::move(v), rest);
520       g.dismiss();
521       return __tag_leaf_ptr(nl);
522     }
523 
524     const __internal *in = __as_internal(n);
525     u32 idx = __idx_at(h, depth);
526     if ( !__has(in->bitmap, idx) ) {
527       existed = false;
528       __leaf *nl = new __leaf(h, k, micron::move(v));
529       __node_guard g{ __tag_leaf_ptr(nl) };      // freed if __internal_insert's alloc throws (OOM)
530       uintptr_t res = __tag_internal_ptr(__internal_insert(in, idx, __tag_leaf_ptr(nl)));
531       g.dismiss();
532       return res;
533     }
534     usize phys = __phys_idx(in->bitmap, idx);
535     uintptr_t old_child = __children_of(in)[phys];
536     uintptr_t new_child = __insert_into(old_child, h, k, micron::move(v), depth + 1u, existed);
537     __node_guard g{ new_child };      // freed if __internal_replace's alloc throws (OOM)
538     uintptr_t res = __tag_internal_ptr(__internal_replace(in, idx, new_child));
539     g.dismiss();
540     return res;
541   }

543   static const V *
544   __find_in(uintptr_t n, hash64_t h, const K &k, usize depth)
545   {
546     while ( n != 0 ) {
547       if ( __is_leaf(n) ) {
548         const __leaf *l = __as_leaf(n);
549         if ( l->h == h && l->key == k ) return micron::addressof(l->value);
550         return __chain_find(l->overflow, h, k);
551       }
552       const __internal *in = __as_internal(n);
553       u32 idx = __idx_at(h, depth);
554       if ( !__has(in->bitmap, idx) ) return nullptr;
555       n = __children_of(in)[__phys_idx(in->bitmap, idx)];
556       ++depth;
557     }
558     return nullptr;
559   }

561   static uintptr_t
562   __erase_from(uintptr_t n, hash64_t h, const K &k, usize depth, bool &erased)
563   {
564     if ( n == 0 ) {
565       erased = false;
566       return 0;
567     }
568     if ( __is_leaf(n) ) {
569       const __leaf *l = __as_leaf(n);
570 
571       if ( l->h == h && l->key == k ) {
572         erased = true;
573         if ( l->overflow == nullptr ) return 0;
574 
575         __chain *first = l->overflow;
576         __leaf *nl = new __leaf(first->h, first->key, first->value);
577         __node_guard g{ __tag_leaf_ptr(nl) };
578         nl->overflow_len = l->overflow_len - 1u;
579         nl->overflow = __copy_chain(first->next);
580         g.dismiss();
581         return __tag_leaf_ptr(nl);
582       }
583 
584       if ( __chain_has(l->overflow, h, k) ) {
585         erased = true;
586         __leaf *nl = new __leaf(l->h, l->key, l->value);
587         __node_guard g{ __tag_leaf_ptr(nl) };
588         nl->overflow_len = l->overflow_len - 1u;
589         nl->overflow = __chain_copy_with_erase(l->overflow, h, k);
590         g.dismiss();
591         return __tag_leaf_ptr(nl);
592       }
593 
594       erased = false;
595       __retain(n);
596       return n;
597     }
598 
599     const __internal *in = __as_internal(n);
600     u32 idx = __idx_at(h, depth);
601     if ( !__has(in->bitmap, idx) ) {
602       erased = false;
603       __retain(n);
604       return n;
605     }
606     usize phys = __phys_idx(in->bitmap, idx);
607     uintptr_t old_child = __children_of(in)[phys];
608     uintptr_t new_child = __erase_from(old_child, h, k, depth + 1u, erased);
609     if ( !erased ) {
610 
611       __release(new_child);
612       __retain(n);
613       return n;
614     }
615     if ( new_child == 0 ) {
616       __internal *nn = __internal_remove(in, idx);
617       if ( nn == nullptr ) return 0;
618       return __tag_internal_ptr(nn);
619     }
620     __node_guard g{ new_child };      // freed if __internal_replace's alloc throws (OOM)
621     uintptr_t res = __tag_internal_ptr(__internal_replace(in, idx, new_child));
622     g.dismiss();
623     return res;
624   }

626   template<typename Fn>
627   static void
628   __walk(uintptr_t n, Fn &&fn)
629   {
630     if ( n == 0 ) return;
631     if ( __is_leaf(n) ) {
632       const __leaf *l = __as_leaf(n);
633       fn(l->key, l->value);
634       for ( const __chain *c = l->overflow; c; c = c->next ) fn(c->key, c->value);
635       return;
636     }
637     const __internal *in = __as_internal(n);
638     usize cnt = static_cast<usize>(micron::popcount(in->bitmap));
639     const uintptr_t *cs = __children_of(in);
640     for ( usize i = 0; i < cnt; ++i ) __walk(cs[i], micron::forward<Fn>(fn));
641   }
