81 __attribute__((always_inline)) static inline u32
82 hash_scan_eq(const hash64_t *p, usize n, hash64_t h) noexcept
83 {
84   u32 mask = 0;
85 #if defined(__micron_x86_avx2)
86   const __m256i target = simd::avx::splat_i64(static_cast<long long>(h));
87   usize i = 0;
88   while ( i + 4 <= n ) {
89     __m256i v = simd::avx::loadu_i256(reinterpret_cast<const __m256i_u *>(p + i));
90     __m256i eq = simd::avx2::eq_i64(v, target);
91     u32 lane = static_cast<u32>(simd::avx::movemask_f64(simd::avx::cast_i256_to_f64(eq)));
92     mask |= lane << i;
93     i += 4;
94   }
95   while ( i < n ) {
96     if ( p[i] == h ) mask |= 1u << i;
97     ++i;
98   }
99 #elif defined(__micron_x86_sse2)
100   const __m128i target = simd::sse::splat_i64(static_cast<long long>(h));
101   usize i = 0;
102   while ( i + 2 <= n ) {
103     __m128i v = simd::sse::loadu_i128(reinterpret_cast<const __m128i_u *>(p + i));
104 #if defined(__micron_x86_sse4_1)
105     __m128i eq = simd::sse::eq_i64(v, target);
106     u32 lane = static_cast<u32>(simd::sse::movemask_f64(simd::sse::cast_i128_to_f64(eq)));
107 #else
108     __m128i eq = simd::sse::eq_i32(v, target);
109     u32 bits = static_cast<u32>(simd::sse::movemask_f32(simd::sse::cast_i128_to_f32(eq)));
110     bits &= bits >> 1u;
111     u32 lane = (bits & 1u) | ((bits >> 1u) & 2u);
112 #endif
113     mask |= lane << i;
114     i += 2;
115   }
116   while ( i < n ) {
117     if ( p[i] == h ) mask |= 1u << i;
118     ++i;
119   }
120 #elif defined(__micron_arm_neon)
121 
122   for ( usize i = 0; i < n; ++i )
123     if ( p[i] == h ) mask |= 1u << i;
124 #else
125   for ( usize i = 0; i < n; ++i )
126     if ( p[i] == h ) mask |= 1u << i;
127 #endif
128   return mask;
129 }

139 __attribute__((always_inline)) static inline usize
140 hash_bsearch_gt(const hash64_t *p, usize n, hash64_t h) noexcept
141 {
142   usize lo = 0;
143   usize hi = n;
144   while ( lo < hi ) {
145     const usize mid = lo + ((hi - lo) >> 1);
146     if ( p[mid] <= h )
147       lo = mid + 1;
148     else
149       hi = mid;
150   }
151   return lo;
152 }

598                   if constexpr ( micron::is_nothrow_move_constructible_v<V> )
599                     new (target_op.value_ptr()) V(micron::move(*source_op.value_ptr()));
600                   else if constexpr ( micron::is_copy_constructible_v<V> )
601                     new (target_op.value_ptr()) V(*source_op.value_ptr());
602                   else
603                     new (target_op.value_ptr()) V(micron::move(*source_op.value_ptr()));
604 #if !defined(__micron_freestanding) || defined(__micron_eh)
605                 } catch ( ... ) {
606                   target_op.key_ptr()->~K();
607                   throw;
608                 }
609 #endif
610               }
611               ++target->buf_used;
612 #if !defined(__micron_freestanding) || defined(__micron_eh)
613             } catch ( ... ) {
614               target->~internal_node();
615               throw;
616             }
617 #endif
618           }
619         }
620       }
621 #if !defined(__micron_freestanding) || defined(__micron_eh)
622     } catch ( ... ) {
623       for ( usize i = 0; i < completed; ++i ) {
624         if ( next.memory[i].raw[0] == __k_fsentinel ) continue;
625         if ( next.memory[i].raw[1] != 0 )
626           micron::ptr_cast<leaf_node *>(next.memory[i].raw)->~leaf_node();
627         else
628           micron::ptr_cast<internal_node *>(next.memory[i].raw)->~internal_node();
629       }
630       throw;
631     }
632 #endif
633 
634     for ( usize i = 0; i < live_slots; ++i ) {
635       if ( __slab.memory[i].raw[0] == __k_fsentinel ) continue;
636       if ( __slab.memory[i].raw[1] != 0 )
637         micron::ptr_cast<leaf_node *>(__slab.memory[i].raw)->~leaf_node();
638       else
639         micron::ptr_cast<internal_node *>(__slab.memory[i].raw)->~internal_node();
640     }
641 
642     chunk<byte> old_chunk = __slab.data();
643     __slab.memory = next.memory;
644     __slab.capacity = next.capacity;
645     __slab.length = live_slots;
646     next.memory = nullptr;
647     next.capacity = 0;
648     next.length = 0;
649     if ( old_chunk.ptr ) __slab_alloc::destroy(old_chunk);
650   }

779   usize
780   leaf_find_slot(const leaf_node &L, hash64_t h, const K &k) const
781   {
782     u32 m = __btree_impl::hash_scan_eq(L.hashes, L.nkeys, h);
783     while ( m ) {
784       u32 bit = static_cast<u32>(__builtin_ctz(m));
785       if ( L.keys()[bit] == k ) return bit;
786       m &= m - 1;
787     }
788     return __leaf_fanout;
789   }

799   int
800   buffer_probe(const internal_node &N, hash64_t h, const K &k, V **out_value_ptr) const
801   {
802     for ( u8 i = N.buf_used; i > 0; --i ) {
803       const pending_op &op = N.buffer[i - 1u];
804       if ( op.hash != h || !(*op.key_ptr() == k) ) continue;
805       if ( op.tag == OP_ERASE ) return 2;
806       *out_value_ptr = const_cast<V *>(op.value_ptr());
807       return 1;
808     }
809     return 0;

812   V *
813   find_impl(hash64_t h, const K &k)
814   {
815     if ( !buckets_ || n_buckets_ == 0 ) return nullptr;
816     bucket_head &b = buckets_[h & mask_];
817     if ( b.root == __k_empty ) return nullptr;
818     node_idx cur = b.root;
819     while ( !slot_is_leaf(cur) ) {
820       internal_node &N = inode(cur);
821       V *bp = nullptr;
822       int r = buffer_probe(N, h, k, &bp);
823       if ( r == 1 ) return bp;
824       if ( r == 2 ) return nullptr;
825       usize ci = __btree_impl::hash_route_gt<__int_fanout>(N.pivots, N.nkeys, h);
826       cur = N.children[ci];
827       if ( cur == __k_empty ) return nullptr;
828       __builtin_prefetch(slot_raw(cur), 0, 1);
829     }
830     while ( cur != __k_empty ) {
831       leaf_node &L = lnode(cur);
832       const usize p = leaf_find_slot(L, h, k);
833       if ( p != __leaf_fanout ) return micron::addr(L.values()[p]);
834       cur = L.overflow_next;
835     }
836     return nullptr;
837   }

845   V *
846   leaf_insert_no_split(node_idx li, hash64_t h, const K &k, V &&v, bool *replaced)
847   {
848     leaf_node &L = lnode(li);
849     usize p = leaf_find_slot(L, h, k);
850     if ( p != __leaf_fanout ) {
851       L.values()[p] = micron::move(v);
852       *replaced = true;
853       return micron::addr(L.values()[p]);
854     }
855     const usize ip = leaf_insert_pos(L, h);
856     const usize n_before = static_cast<usize>(L.nkeys);
857 
858     if ( ip < n_before ) {
859       const usize count = n_before - ip;
860       __btree_impl::trivial_memmove(micron::addr(L.hashes[ip + 1]), micron::addr(L.hashes[ip]), count * sizeof(hash64_t));
861       __btree_impl::typed_shift_right(L.keys(), ip, n_before);
862       __btree_impl::typed_shift_right(L.values(), ip, n_before);
863     }
864     L.hashes[ip] = h;
865     new (L.keys() + ip) K(k);
866     new (L.values() + ip) V(micron::move(v));
867     ++L.nkeys;
868     *replaced = false;
869     return micron::addr(L.values()[ip]);
870   }

879   micron::pair<node_idx, hash64_t>
880   split_leaf_mixed(node_idx li)
881   {
882     usize mid;
883     {
884       const leaf_node &Lpre = lnode(li);
885       const usize n = static_cast<usize>(Lpre.nkeys);
886       mid = __leaf_fanout / 2;
887       while ( mid < n && Lpre.hashes[mid - 1] == Lpre.hashes[mid] ) ++mid;
888       if ( mid >= n ) {
889         mid = __leaf_fanout / 2;
890         while ( mid > 0 && Lpre.hashes[mid - 1] == Lpre.hashes[mid] ) --mid;
891       }
892     }
893     node_idx ri = alloc_leaf();
894     leaf_node &L = lnode(li);      // re-fetch after potential slab grow
895     leaf_node &R = lnode(ri);
896     R.nkeys = static_cast<u8>(L.nkeys - mid);
897     __btree_impl::trivial_memmove(micron::addr(R.hashes[0]), micron::addr(L.hashes[mid]), R.nkeys * sizeof(hash64_t));
898     __btree_impl::typed_relocate_n(R.keys(), L.keys() + mid, R.nkeys);
899     __btree_impl::typed_relocate_n(R.values(), L.values() + mid, R.nkeys);
900     L.nkeys = static_cast<u8>(mid);
901     leaf_list_insert_after(li, ri);
902     return { ri, R.hashes[0] };
903   }

905   micron::pair<node_idx, hash64_t>
906   asymmetric_split_leaf(node_idx ni, hash64_t h, const K &k, V &&v, V **out_value, bool *out_inserted)
907   {
908     const hash64_t leaf_hash = lnode(ni).hashes[0];
909     node_idx ri = alloc_leaf();
910 
911     if ( h > leaf_hash ) {
912       leaf_node &R = lnode(ri);
913       R.hashes[0] = h;
914       new (R.keys()) K(k);
915       new (R.values()) V(micron::move(v));
916       R.nkeys = 1;
917       const node_idx tail = leaf_chain_tail(ni);
918       leaf_list_insert_after(tail, ri);
919       *out_value = micron::addr(R.values()[0]);
920       *out_inserted = true;
921       return { ri, h };
922     }
923     leaf_node &L = lnode(ni);
924     leaf_node &R = lnode(ri);
925     const u8 n = L.nkeys;
926     __btree_impl::trivial_memmove(R.hashes, L.hashes, static_cast<usize>(n) * sizeof(hash64_t));
927     __btree_impl::typed_relocate_n(R.keys(), L.keys(), static_cast<usize>(n));
928     __btree_impl::typed_relocate_n(R.values(), L.values(), static_cast<usize>(n));
929     R.nkeys = n;
930     R.overflow_next = L.overflow_next;
931     L.overflow_next = __k_empty;
932     L.nkeys = 0;
933     L.hashes[0] = h;
934     new (L.keys()) K(k);
935     new (L.values()) V(micron::move(v));
936     L.nkeys = 1;
937     leaf_list_insert_after(ni, ri);
938     *out_value = micron::addr(L.values()[0]);
939     *out_inserted = true;
940     return { ri, leaf_hash };
941   }

943   V *
944   chain_insert_same_hash(node_idx primary, hash64_t h, const K &k, V &&v, bool *replaced)
945   {
946     node_idx cur = primary;
947     node_idx tail = primary;
948     node_idx with_space = (lnode(primary).nkeys < __leaf_fanout) ? primary : __k_empty;
949     while ( cur != __k_empty ) {
950       leaf_node &C = lnode(cur);
951       const usize p = leaf_find_slot(C, h, k);
952       if ( p != __leaf_fanout ) {
953         C.values()[p] = micron::move(v);
954         *replaced = true;
955         return micron::addr(C.values()[p]);
956       }
957       if ( with_space == __k_empty && C.nkeys < __leaf_fanout ) with_space = cur;
958       tail = cur;
959       cur = C.overflow_next;
960     }
961     if ( with_space != __k_empty ) {
962       return leaf_insert_no_split(with_space, h, k, micron::move(v), replaced);

964     node_idx new_li = alloc_leaf();
965     leaf_node &Lnew = lnode(new_li);
966     Lnew.hashes[0] = h;
967     new (Lnew.keys()) K(k);
968     new (Lnew.values()) V(micron::move(v));
969     Lnew.nkeys = 1;
970     lnode(tail).overflow_next = new_li;
971     leaf_list_insert_after(tail, new_li);
972     *replaced = false;
973     return micron::addr(Lnew.values()[0]);
974   }

1037   void
1038   internal_insert_pivot_or_split(node_idx ii, usize ip, hash64_t new_pivot, node_idx right_child, hash64_t *out_split_pivot,
1039                                  node_idx *out_split_right)
1040   {
1041     *out_split_right = __k_empty;
1042     {
1043       internal_node &N = inode(ii);
1044       if ( N.nkeys < __int_fanout - 1 ) {
1045         internal_insert_pivot(N, ip, new_pivot, right_child);
1046         return;
1047       }
1048     }
1049     hash64_t exp_pivots[__int_fanout];
1050     node_idx exp_children[__int_fanout + 1];
1051     usize nb;
1052     {
1053       const internal_node &N = inode(ii);
1054       nb = static_cast<usize>(N.nkeys);
1055       for ( usize j = 0; j < nb; ++j ) exp_pivots[j] = N.pivots[j];
1056       for ( usize j = 0; j <= nb; ++j ) exp_children[j] = N.children[j];
1057     }
1058     for ( usize j = nb; j > ip; --j ) exp_pivots[j] = exp_pivots[j - 1];
1059     for ( usize j = nb + 1; j > ip + 1; --j ) exp_children[j] = exp_children[j - 1];
1060     exp_pivots[ip] = new_pivot;
1061     exp_children[ip + 1] = right_child;
1062     const usize total = nb + 1;
1063 
1064     node_idx ri = alloc_internal();
1065     const usize lmid = total / 2;
1066     const hash64_t promoted = exp_pivots[lmid];
1067     {
1068       internal_node &L = inode(ii);
1069       L.nkeys = static_cast<u8>(lmid);
1070       for ( usize j = 0; j < lmid; ++j ) L.pivots[j] = exp_pivots[j];
1071       for ( usize j = 0; j <= lmid; ++j ) L.children[j] = exp_children[j];
1072     }
1073     {
1074       internal_node &R = inode(ri);
1075       R.nkeys = static_cast<u8>(total - lmid - 1);
1076       for ( usize j = 0; j < R.nkeys; ++j ) R.pivots[j] = exp_pivots[lmid + 1 + j];
1077       for ( usize j = 0; j <= R.nkeys; ++j ) R.children[j] = exp_children[lmid + 1 + j];
1078     }
1079     *out_split_pivot = promoted;
1080     *out_split_right = ri;
1081   }

1083   V *
1084   descend_apply(node_idx ni, hash64_t h, const K &k, V *v_for_insert, op_tag tag, bool *out_inserted, hash64_t *out_split_pivot,
1085                 node_idx *out_split_right)
1086   {
1087     *out_split_right = __k_empty;
1088     if ( slot_is_leaf(ni) ) {
1089       if ( tag == OP_ERASE ) {
1090         node_idx cur = ni;
1091         while ( cur != __k_empty ) {
1092           leaf_node &C = lnode(cur);
1093           const usize p = leaf_find_slot(C, h, k);
1094           if ( p != __leaf_fanout ) {
1095             const usize n_before = static_cast<usize>(C.nkeys);
1096             if ( p + 1 < n_before ) {
1097               const usize tail = n_before - p - 1;
1098               __btree_impl::trivial_memmove(micron::addr(C.hashes[p]), micron::addr(C.hashes[p + 1]), tail * sizeof(hash64_t));
1099             }
1100             __btree_impl::typed_erase_at(C.keys(), p, n_before);
1101             __btree_impl::typed_erase_at(C.values(), p, n_before);
1102             --C.nkeys;
1103             *out_inserted = true;
1104             return nullptr;
1105           }
1106           cur = C.overflow_next;
1107         }
1108         *out_inserted = false;
1109         return nullptr;
1110       }
1111 
1112       {
1113         node_idx cur = ni;
1114         while ( cur != __k_empty ) {
1115           leaf_node &C = lnode(cur);
1116           const usize p = leaf_find_slot(C, h, k);
1117           if ( p != __leaf_fanout ) {
1118             C.values()[p] = micron::move(*v_for_insert);
1119             *out_inserted = false;
1120             return micron::addr(C.values()[p]);
1121           }
1122           cur = C.overflow_next;
1123         }
1124       }
1125 
1126       bool replaced = false;
1127       leaf_node &L = lnode(ni);
1128       if ( L.nkeys < __leaf_fanout ) {
1129         V *vp = leaf_insert_no_split(ni, h, k, micron::move(*v_for_insert), &replaced);
1130         *out_inserted = !replaced;
1131         return vp;
1132       }
1133       if ( leaf_is_single_class(L) ) {
1134         const hash64_t leaf_hash = L.hashes[0];
1135         if ( h == leaf_hash ) {
1136           V *vp = chain_insert_same_hash(ni, h, k, micron::move(*v_for_insert), &replaced);
1137           *out_inserted = !replaced;
1138           return vp;
1139         }
1140         V *vp = nullptr;
1141         auto sp = asymmetric_split_leaf(ni, h, k, micron::move(*v_for_insert), &vp, out_inserted);
1142         *out_split_right = sp.a;
1143         *out_split_pivot = sp.b;
1144         return vp;
1145       }
1146 
1147       auto sp = split_leaf_mixed(ni);
1148       const node_idx right = sp.a;
1149       const hash64_t med = sp.b;
1150       const node_idx target = (h < med) ? ni : right;
1151       V *vp = leaf_insert_no_split(target, h, k, micron::move(*v_for_insert), &replaced);
1152       *out_inserted = !replaced;
