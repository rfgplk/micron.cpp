26 // rb_map
27 //
28 // hash map with two-tier collision handling:
29 //  a) light bins use a singly-linked chain through a dense SoA storage
30 //  b) heavy bins are *treeified* into a per-bin red-black tree once the chain
31 //     length exceeds __treeify_threshold
32 //
33 // the SoA layout means _hashes[] is contiguous and has no key/value padding so it stays cache-resident for repeated lookups
34 
35 template<typename K, typename V, class Alloc = micron::allocator_serial<>>
36   requires micron::is_copy_constructible_v<V> and micron::is_move_constructible_v<V>

48     hash64_t hash;
49     K key;

51 
52     __tree_entry() : hash(0), key{}, value{} { }

54     __tree_entry(hash64_t h, const K &k, const V &v) : hash(h), key(k), value(v) { }

56     __tree_entry(hash64_t h, K &&k, V &&v) : hash(h), key(micron::move(k)), value(micron::move(v)) { }

64     operator<(const __tree_entry &o) const

66       if ( hash != o.hash ) return hash < o.hash;

128   static usize
129   __round_pow2(usize n) noexcept
130   {
131     if ( n <= __min_bins ) return __min_bins;
132     usize p = 1;
133     while ( p < n ) {
134       usize next = p << 1u;
135       if ( next <= p ) return p;
136       p = next;
137     }
138     return p;
139   }

141   static constexpr usize
142   __load_limit(usize bins) noexcept

144     return (bins / __load_denom) * __load_num + ((bins % __load_denom) * __load_num) / __load_denom;

147   void
148   __alloc_bins(usize count)
149   {
150     if ( count > static_cast<usize>(numeric_limits<i32>::max()) ) exc<except::library_error>("rb_map: bin index capacity overflow");

167   void
168   __grow_soa()
169   {
170     if ( __cap_soa > static_cast<usize>(numeric_limits<i32>::max()) / 2u )
171       exc<except::library_error>("rb_map: SoA index capacity overflow");
172     usize new_cap = __cap_soa == 0 ? 16 : __cap_soa * 2;
173     hash64_t *nh = nullptr;
174     i32 *nn = nullptr;
175     i32 *nb = nullptr;
176     K *nk = nullptr;
177     V *nv = nullptr;
178 #if !defined(__micron_freestanding) || defined(__micron_eh)
179     try {
180 #endif
181       nh = __alloc_array<hash64_t>(new_cap);
182       nn = __alloc_array<i32>(new_cap);
183       nb = __alloc_array<i32>(new_cap);
184       nk = __alloc_array<K>(new_cap);
185       nv = __alloc_array<V>(new_cap);
186 #if !defined(__micron_freestanding) || defined(__micron_eh)
187     } catch ( ... ) {
188       __free_array(nv);
189       __free_array(nk);
190       __free_array(nb);
191       __free_array(nn);
192       __free_array(nh);
193       throw;
194     }
195 #endif
196     usize moved = 0;
197     if ( __n_soa > 0 ) {
198 #if !defined(__micron_freestanding) || defined(__micron_eh)
199       try {
200         for ( ; moved < __n_soa; ++moved ) {
201           if constexpr ( micron::is_nothrow_move_constructible_v<K> && micron::is_nothrow_move_constructible_v<V> )
202             new (nk + moved) K(micron::move(__keys[moved]));
203           else if constexpr ( micron::is_copy_constructible_v<K> )
204             new (nk + moved) K(__keys[moved]);
205           else
206             new (nk + moved) K(micron::move(__keys[moved]));
207           try {
208             if constexpr ( micron::is_nothrow_move_constructible_v<K> && micron::is_nothrow_move_constructible_v<V> )
209               new (nv + moved) V(micron::move(__values[moved]));
210             else if constexpr ( micron::is_copy_constructible_v<K> )
211               new (nv + moved) V(__values[moved]);
212             else
213               new (nv + moved) V(micron::move(__values[moved]));
214           } catch ( ... ) {
215             nk[moved].~K();
216             throw;
217           }
218         }
219       } catch ( ... ) {
220         for ( usize i = 0; i < moved; ++i ) {
221           nv[i].~V();
222           nk[i].~K();
223         }
224         __free_array(nv);
225         __free_array(nk);
226         __free_array(nb);
227         __free_array(nn);
228         __free_array(nh);
229         throw;
230       }
231 #else
232       for ( ; moved < __n_soa; ++moved ) {
233         new (nk + moved) K(micron::move(__keys[moved]));
234         new (nv + moved) V(micron::move(__values[moved]));
235       }
236 #endif
237       micron::memcpy(reinterpret_cast<byte *>(nh), reinterpret_cast<byte *>(__hashes), __n_soa * sizeof(hash64_t));
238       micron::memcpy(reinterpret_cast<byte *>(nn), reinterpret_cast<byte *>(__next), __n_soa * sizeof(i32));
239       micron::memcpy(reinterpret_cast<byte *>(nb), reinterpret_cast<byte *>(__home_bin), __n_soa * sizeof(i32));
240       for ( usize i = 0; i < __n_soa; ++i ) {
241         __keys[i].~K();
242         __values[i].~V();
243       }
244     }
245     __free_array(__hashes);
246     __free_array(__keys);
247     __free_array(__values);
248     __free_array(__next);
249     __free_array(__home_bin);
250     __hashes = nh;
251     __keys = nk;
252     __values = nv;
253     __next = nn;
254     __home_bin = nb;
255     __cap_soa = new_cap;
256   }

258   void
259   __soa_remove(i32 idx)
260   {
261     i32 last = static_cast<i32>(__n_soa) - 1;
262     if ( idx != last ) {
263 
264       i32 last_home = __home_bin[last];
265       __bin_t &lb = __bins[last_home];
266       if ( !lb.tree ) {
267 
268         if ( lb.list_head == last ) {
269           lb.list_head = idx;
270         } else {
271           i32 prev = lb.list_head;
272           while ( prev != -1 && __next[prev] != last ) prev = __next[prev];
273           if ( prev != -1 ) __next[prev] = idx;
274         }
275       }
276       __hashes[idx] = __hashes[last];
277       __next[idx] = __next[last];
278       __home_bin[idx] = __home_bin[last];
279       __keys[idx] = micron::move(__keys[last]);
280       __values[idx] = micron::move(__values[last]);
281     }
282     __keys[__n_soa - 1].~K();
283     __values[__n_soa - 1].~V();
284     --__n_soa;
285   }

287   void
288   __treeify(usize bin_idx)
289   {
290     __bin_t &b = __bins[bin_idx];
291     if ( b.tree ) return;
292     auto *t = __make_tree();
293 
294     i32 cur = b.list_head;
295 #if !defined(__micron_freestanding) || defined(__micron_eh)
296     try {
297 #endif
298       while ( cur != -1 ) {
299         i32 nxt = __next[cur];
300         if constexpr ( micron::is_copy_constructible_v<K> )
301           t->insert(__tree_entry(__hashes[cur], __keys[cur], __values[cur]));
302         else
303           t->insert(__tree_entry(__hashes[cur], micron::move(__keys[cur]), micron::move(__values[cur])));
304         cur = nxt;
305       }
306 #if !defined(__micron_freestanding) || defined(__micron_eh)
307     } catch ( ... ) {
308       __free_tree(t);
309       throw;
310     }
311 #endif
312 
313     b.list_head = -1;
314     b.tree = t;
315     b.chain_len = 0;
316 
317     i32 i = 0;
318     while ( i < static_cast<i32>(__n_soa) ) {
319       if ( __home_bin[i] == static_cast<i32>(bin_idx) ) {
320         __soa_remove(i);
321       } else {
322         ++i;
323       }
324     }
325   }

327   void
328   __untreeify(usize bin_idx)
329   {
330     __bin_t &b = __bins[bin_idx];
331     if ( !b.tree ) return;
332     __tree_t *t = b.tree;
333 
334     while ( __n_soa + t->size() > __cap_soa ) __grow_soa();
335 
336     b.tree = nullptr;
337     b.list_head = -1;
338     b.chain_len = 0;
339     while ( !t->empty() ) {
340       __tree_entry e = t->extract_min();
341       i32 ix = static_cast<i32>(__n_soa);
342       new (__keys + ix) K(micron::move(e.key));
343       new (__values + ix) V(micron::move(e.value));
344       __hashes[ix] = e.hash;
345       __next[ix] = b.list_head;
346       __home_bin[ix] = static_cast<i32>(bin_idx);
347       b.list_head = ix;
348       ++b.chain_len;
349       ++__n_soa;
350     }
351     __free_tree(t);
352   }

363   void
364   __rehash(usize new_n_bins)
365   {
366     if ( new_n_bins < __min_bins ) new_n_bins = __min_bins;
367     new_n_bins = __round_pow2(new_n_bins);
368 
369     if constexpr ( micron::is_copy_constructible_v<K> ) {
370       rb_map next(__raw_bins_tag{}, new_n_bins);
371       for ( usize i = 0; i < __n_soa; ++i ) next.__insert_absent(__hashes[i], __keys[i], __values[i]);
372       for ( usize i = 0; i < __n_bins; ++i ) {
373         const __tree_t *tree = __bins[i].tree;
374         if ( !tree ) continue;
375         tree->for_each([&next](const __tree_entry &entry) { next.__insert_absent(entry.hash, entry.key, entry.value); });
376       }
377       swap(next);
378       return;
379     }
380 
381     usize total = __total;
382     auto *snap = total ? __alloc_array<__tree_entry>(total) : nullptr;
383     usize w = 0;
384 
385 #if !defined(__micron_freestanding) || defined(__micron_eh)
386     try {
387       for ( usize i = 0; i < __n_soa; ++i ) {
388         new (snap + w) __tree_entry(__hashes[i], micron::move(__keys[i]), micron::move(__values[i]));
389         ++w;
390       }
391     } catch ( ... ) {
392       for ( usize j = 0; j < w; ++j ) snap[j].~__tree_entry();
393       __free_array(snap);
394       throw;
395     }
396 #else
397     for ( usize i = 0; i < __n_soa; ++i ) {
398       new (snap + w) __tree_entry(__hashes[i], micron::move(__keys[i]), micron::move(__values[i]));
399       ++w;
400     }
401 #endif
402     for ( usize i = 0; i < __n_soa; ++i ) {
403       __keys[i].~K();
404       __values[i].~V();
405     }
406     __n_soa = 0;
407 
408     if ( __bins ) {
409 #if !defined(__micron_freestanding) || defined(__micron_eh)
410       try {
411         for ( usize i = 0; i < __n_bins; ++i ) {
412           __tree_t *t = __bins[i].tree;
413           if ( !t ) continue;
414           while ( !t->empty() ) {
415             new (snap + w) __tree_entry(t->extract_min());
416             ++w;
417           }
418         }
419       } catch ( ... ) {
420         for ( usize j = 0; j < w; ++j ) snap[j].~__tree_entry();
421         __free_array(snap);
422         for ( usize i = 0; i < __n_bins; ++i )
423           if ( __bins[i].tree ) __free_tree(__bins[i].tree);
424         __free_array(__bins);
425         __bins = nullptr;
426         __n_bins = 0;
427         __bin_mask = 0;
428         __total = 0;
429         throw;
430       }
431 #else
432       for ( usize i = 0; i < __n_bins; ++i ) {
433         __tree_t *t = __bins[i].tree;
434         if ( !t ) continue;
435         while ( !t->empty() ) {
436           new (snap + w) __tree_entry(t->extract_min());
437           ++w;
438         }
439       }
440 #endif
441       for ( usize i = 0; i < __n_bins; ++i )
442         if ( __bins[i].tree ) __free_tree(__bins[i].tree);
443       __free_array(__bins);
444       __bins = nullptr;
445     }
446 
447     __bins = __alloc_array<__bin_t>(new_n_bins);
448     for ( usize i = 0; i < new_n_bins; ++i ) {
449       __bins[i].list_head = -1;
450       __bins[i].tree = nullptr;
451       __bins[i].chain_len = 0;
452     }
453     __n_bins = new_n_bins;
454     __bin_mask = new_n_bins - 1u;
455     __total = 0;
456 
457     usize consumed = 0;
458 #if !defined(__micron_freestanding) || defined(__micron_eh)
459     try {
460 #endif
461       for ( ; consumed < w; ++consumed ) {
462         __tree_entry &e = snap[consumed];
463         usize bi = __bin_of(e.hash);
464         __bin_t &b = __bins[bi];
465         if ( b.tree ) {
466           b.tree->insert(micron::move(e));
467           ++__total;
468         } else {
469           if ( __n_soa == __cap_soa ) __grow_soa();
470           i32 ix = static_cast<i32>(__n_soa);
471           new (__keys + ix) K(micron::move(e.key));
472           new (__values + ix) V(micron::move(e.value));
473           __hashes[ix] = e.hash;
474           __next[ix] = b.list_head;
475           __home_bin[ix] = static_cast<i32>(bi);
476           b.list_head = ix;
477           ++b.chain_len;
478           ++__n_soa;
479           ++__total;
480           if ( b.chain_len >= __treeify_threshold && __n_bins >= __min_treeify_cap ) {
481             __treeify(bi);
482           }
483         }
484         e.~__tree_entry();
485       }
486 #if !defined(__micron_freestanding) || defined(__micron_eh)
487     } catch ( ... ) {
488       for ( usize j = consumed; j < w; ++j ) snap[j].~__tree_entry();
489       __free_array(snap);
490       throw;
491     }
492 #endif
493     __free_array(snap);
494   }

528   V *
529   __find_hash(hash64_t h, const K &key)
530   {
531     if ( !__bins ) return nullptr;
532     __bin_t &bin = __bins[__bin_of(h)];
533     if ( bin.tree ) {
534       __tree_entry *entry
535           = bin.tree->find_by([h, &key](const __tree_entry &data) { return h != data.hash ? h < data.hash : key < data.key; },
536                               [h, &key](const __tree_entry &data) { return data.hash != h ? data.hash < h : data.key < key; });
537       return entry ? micron::addressof(entry->value) : nullptr;
538     }
539     for ( i32 i = bin.list_head; i != -1; i = __next[i] )
540       if ( __hashes[i] == h && __keys[i] == key ) return micron::addressof(__values[i]);
541     return nullptr;
542   }

550   template<class KK, class VV>
551   micron::pair<bool, V *>
552   __insert_absent(hash64_t h, KK &&key, VV &&value)
553   {
554     if ( __total == numeric_limits<usize>::max() ) exc<except::library_error>("rb_map: size overflow");
555     if ( __total + 1u > __load_limit(__n_bins) ) {
556       if ( __n_bins > static_cast<usize>(numeric_limits<i32>::max()) / 2u )
557         exc<except::library_error>("rb_map: bin index capacity overflow");
558       __rehash(__n_bins * 2u);
559     }
560     usize bin_index = __bin_of(h);
561     __bin_t &bin = __bins[bin_index];
562 
563     if ( !bin.tree && bin.chain_len + 1u >= __treeify_threshold && __n_bins >= __min_treeify_cap ) __treeify(bin_index);
564     if ( bin.tree ) {
565       __tree_entry &inserted = bin.tree->insert(__tree_entry(h, K(micron::forward<KK>(key)), V(micron::forward<VV>(value))));
566       ++__total;
567       return { true, micron::addressof(inserted.value) };
568     }
569 
570     if ( __n_soa == __cap_soa ) __grow_soa();
571     i32 index = static_cast<i32>(__n_soa);
572     new (__keys + index) K(micron::forward<KK>(key));
573 #if !defined(__micron_freestanding) || defined(__micron_eh)
574     try {
575       new (__values + index) V(micron::forward<VV>(value));
576     } catch ( ... ) {
577       __keys[index].~K();
578       throw;
579     }
580 #else
581     new (__values + index) V(micron::forward<VV>(value));
582 #endif
583     __hashes[index] = h;
584     __next[index] = bin.list_head;
585     __home_bin[index] = static_cast<i32>(bin_index);
586     bin.list_head = index;
587     ++bin.chain_len;
588     ++__n_soa;
589     ++__total;
590     return { true, micron::addressof(__values[index]) };
591   }

792   bool
793   erase(const K &k)
794   {
795     hash64_t h = hash<hash64_t>(k);
796     usize bi = __bin_of(h);
797     __bin_t &b = __bins[bi];
798     if ( b.tree ) {
799       bool ok = b.tree->erase_by([h, &k](const __tree_entry &d) { return h != d.hash ? h < d.hash : k < d.key; },
800                                  [h, &k](const __tree_entry &d) { return d.hash != h ? d.hash < h : d.key < k; });
801       if ( ok ) {
802         --__total;
803         if ( b.tree->size() < __untreeify_threshold ) __untreeify(bi);
804       }
805       return ok;
806     }
807     i32 prev = -1;
808     for ( i32 i = b.list_head; i != -1; i = __next[i] ) {
809       if ( __hashes[i] == h && __keys[i] == k ) {
810         if ( prev == -1 )
811           b.list_head = __next[i];
812         else
813           __next[prev] = __next[i];
814         --b.chain_len;
815         --__total;
816         __soa_remove(i);
817         return true;
818       }
819       prev = i;
820     }
821     return false;
822   }

847   // two-phase cursor
848   class const_iterator
849   {
850     const rb_map *__m = nullptr;
851     usize __soa_i = 0;
852     usize __bin_i = 0;
853     typename __tree_t::const_iterator __ti{};
854 
855     void
856     __enter_bin(usize b)
857     {
858       __bin_i = b;
859       while ( __bin_i < __m->__n_bins ) {
860         const __tree_t *t = __m->__bins[__bin_i].tree;
861         if ( t ) {
862           __ti = t->begin();
863           if ( __ti != t->end() ) return;
864         }
865         ++__bin_i;
866       }
867       __ti = typename __tree_t::const_iterator{};
868     }
869 
870   public:
871     using value_type = micron::pair<const K &, const V &>;
872     using reference = micron::pair<const K &, const V &>;
873     using difference_type = ssize_t;
874 
875     constexpr const_iterator() = default;
876 
877     const_iterator(const rb_map *__mm, bool __end) : __m(__mm), __soa_i(__end ? __mm->__n_soa : 0), __bin_i(__end ? __mm->__n_bins : 0)
878     {
879       if ( __end ) return;
880       if ( __soa_i >= __m->__n_soa ) __enter_bin(0);
881     }
882 
883     reference
884     operator*() const
885     {
886       if ( __soa_i < __m->__n_soa ) return reference{ __m->__keys[__soa_i], __m->__values[__soa_i] };
887       return reference{ __ti->key, __ti->value };
888     }
889 
890     const_iterator &
891     operator++()
892     {
893       if ( __soa_i < __m->__n_soa ) {
894         ++__soa_i;
895         if ( __soa_i >= __m->__n_soa ) __enter_bin(0);
896         return *this;
897       }
898       ++__ti;
899       const __tree_t *t = __m->__bins[__bin_i].tree;
900       if ( __ti == t->end() ) __enter_bin(__bin_i + 1);
901       return *this;
902     }
903 
904     const_iterator
905     operator++(int)
906     {
907       const_iterator __t = *this;
908       ++*this;
909       return __t;
910     }
911 
912     bool
913     operator==(const const_iterator &__o) const noexcept
914     {
915       return __soa_i == __o.__soa_i && __bin_i == __o.__bin_i && __ti == __o.__ti;
916     }
917 
918     bool
919     operator!=(const const_iterator &__o) const noexcept
920     {
921       return !(*this == __o);
922     }
923   };
