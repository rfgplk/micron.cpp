76   static constexpr u8
77   __h2(hash64_t h)
78   {
79     // 7-bit tag for the control byte
80     u8 v = static_cast<u8>((h >> 40) & 0x7Fu);
81     u8 c = static_cast<u8>(v | __sentinel);
82     // NEVER alias the reserved control bytes __deleted (0xFE) / __empty (0xFF)
83     return c >= __deleted ? static_cast<u8>(c - 2u) : c;
84   }

141   __mask
142   __match(u8 hash_val, usize ind) const
143   {
144     // NOTE: must be unaligned load, no guarantee ind will always be aligned
145 #if defined(__micron_arch_x86_any)
146     simd::i128 match = simd::sse::splat_i8(static_cast<char>(hash_val));
147     simd::i128 meta = simd::sse::loadu_i128(reinterpret_cast<const __m128i_u *>(&__control_bytes[ind]));
148     int mask = simd::sse::movemask_i8(simd::sse::eq_i8(match, meta));
149     return __mask(mask);
150 #elif defined(__micron_arm_neon)
151     uint8x16_t match = simd::neon::splat_u8(hash_val);
152     uint8x16_t meta = simd::neon::load_u8(&__control_bytes[ind]);
153     return __mask(static_cast<i32>(simd::neon::movemask_u8(simd::neon::eq(meta, match))));
154 #else
155     i32 mask = 0;
156     for ( usize i = 0; i < 16; ++i )
157       if ( __control_bytes[ind + i] == hash_val ) mask |= (1 << i);
158     return __mask(mask);
159 #endif
160   }

182   __mask
183   __match_empty_or_deleted(usize ind) const
184   {
185 #if defined(__micron_arch_x86_any)
186     simd::i128 meta = simd::sse::loadu_i128(reinterpret_cast<const __m128i_u *>(&__control_bytes[ind]));
187     simd::i128 empty = simd::sse::splat_i8(static_cast<char>(__empty));
188     simd::i128 deleted = simd::sse::splat_i8(static_cast<char>(__deleted));
189     int mask = simd::sse::movemask_i8(simd::sse::or_i128(simd::sse::eq_i8(meta, empty), simd::sse::eq_i8(meta, deleted)));
190     return __mask(mask);
191 #elif defined(__micron_arm_neon)
192     uint8x16_t meta = simd::neon::load_u8(&__control_bytes[ind]);
193     uint8x16_t empty = simd::neon::splat_u8(__empty);
194     uint8x16_t deleted = simd::neon::splat_u8(__deleted);
195     uint8x16_t combined = simd::neon::or_(simd::neon::eq(meta, empty), simd::neon::eq(meta, deleted));
196     return __mask(static_cast<i32>(simd::neon::movemask_u8(combined)));
197 #else
198     i32 mask = 0;
199     for ( usize i = 0; i < 16; ++i ) {
200       u8 c = __control_bytes[ind + i];
201       if ( c == __empty || c == __deleted ) mask |= (1 << i);
202     }
203     return __mask(mask);
204 #endif
205   }

217   usize
218   __find_insert_slot(usize start) const noexcept
219   {
220     for ( usize i = 0; i < NH; i += 16 ) {
221       usize group_start = (start + i) % N;
222       usize remaining = NH - i;
223 
224       if ( group_start + 16 <= N ) {
225         __mask m = __clip(__match_empty_or_deleted(group_start), remaining);
226         if ( m.any() ) return group_start + static_cast<usize>(m.lowest());
227       } else {
228         usize span = remaining < 16 ? remaining : 16;
229         for ( usize j = 0; j < span; ++j ) {
230           usize probe = (group_start + j) % N;
231           if ( __control_bytes[probe] == __empty || __control_bytes[probe] == __deleted ) return probe;
232         }
233       }
234     }
235     return static_cast<usize>(-1);
236   }

238   const V *
239   __find_hash_impl(hash64_t kh, const K &key) const
240   {
241     u8 h2 = __h2(kh);
242     usize start = __b_index_key(kh);
243 
244     for ( usize i = 0; i < NH; i += 16 ) {
245       usize group_start = (start + i) % N;
246       usize remaining = NH - i;
247 
248       if ( group_start + 16 <= N ) {
249         __mask m = __clip(__match(h2, group_start), remaining);
250         while ( m.any() ) {
251           usize probe = group_start + static_cast<usize>(m.lowest());
252           if ( __entry(probe).key == key ) return micron::addressof(__entry(probe).value);
253           m.clear_lowest();
254         }
255         if ( __clip(__match_empty(group_start), remaining).any() ) return nullptr;
256       } else {
257         usize span = remaining < 16 ? remaining : 16;
258         for ( usize j = 0; j < span; ++j ) {
259           usize probe = (group_start + j) % N;
260           if ( __control_bytes[probe] == h2 && __entry(probe).key == key ) return micron::addressof(__entry(probe).value);
261           if ( __control_bytes[probe] == __empty ) return nullptr;
262         }
263       }
264     }
265     return nullptr;
266   }

531   bool
532   erase(const K &key)
533   {
534     hash64_t kh = hash<hash64_t>(key);
535     u8 h2 = __h2(kh);
536     usize start = __b_index_key(kh);
537 
538     for ( usize i = 0; i < NH; i += 16 ) {
539       usize group_start = (start + i) % N;
540       usize remaining = NH - i;
541 
542       if ( group_start + 16 <= N ) {
543         __mask m = __clip(__match(h2, group_start), remaining);
544         while ( m.any() ) {
545           usize probe = group_start + static_cast<usize>(m.lowest());
546           if ( __entry(probe).key == key ) {
547             __destroy_entry(probe);
548             __control_bytes[probe] = __deleted;
549             --__size;
550             return true;
551           }
552           m.clear_lowest();
553         }
554         if ( __clip(__match_empty(group_start), remaining).any() ) return false;
555       } else {
556         usize span = remaining < 16 ? remaining : 16;
557         for ( usize j = 0; j < span; ++j ) {
558           usize probe = (group_start + j) % N;
559           if ( __control_bytes[probe] == h2 && __entry(probe).key == key ) {
560             __destroy_entry(probe);
561             __control_bytes[probe] = __deleted;
562             --__size;
563             return true;
564           }
565           if ( __control_bytes[probe] == __empty ) return false;
566         }
567       }
568     }
569 
570     return false;
571   }
