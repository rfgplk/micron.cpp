44   struct __node {
45     uintptr_t __packed_left;      // left pointer/red bit
46     __node *right;
47     mutable u32 refs;
48     K key;
49     V value;

118     if ( !n ) [[unlikely]] {      // micron::__alloc returns nullptr on OOM (it does NOT throw)

196   static __attribute__((noinline)) void
197   __release_subtree(__node *n) noexcept
198   {
199     constexpr usize __stack_cap = 64;
200     __node *stack[__stack_cap];
201     usize depth = 0;
202 
203     while ( n ) [[likely]] {
204       if ( !__release_ref(n) ) [[likely]]
205         break;
206       __node *l = n->left();
207       __node *r = n->right;
208       __dealloc_node(n);
209       if ( l ) {
210         if ( depth < __stack_cap ) [[likely]]
211           stack[depth++] = l;
212         else
213           __release_subtree(l);      // overflow
214       }
215       n = r;
216     }
217     while ( depth > 0 ) __release_subtree(stack[--depth]);
218   }

245   static inline __node *
246   __acquire_mutable(const __node *h)
247   {
248     // refs==1 means this version is the sole owner -> unreachable by any other
249     // thread, so an in-place mutation is safe
250     if ( atom::load(&h->refs, atomic_acquire) == 1u ) [[likely]] {
251       __node *n = const_cast<__node *>(h);
252       atom::fetch_add(&n->refs, 1u, atomic_relaxed);
253       return n;
254     }
255     return __copy_node(h);

273   static inline __node *
274   __rotate_left(__node *h)
275   {
276     __node *x = h->right;
277 
278     if ( atom::load(&x->refs, atomic_acquire) == 1u ) [[likely]] {
279       // both exclusive relink
280       bool h_color = h->red();
281       h->right = x->left();
282       h->set_red(true);
283       x->set_left_and_red(h, h_color);
284       return x;
285     }
286 
287     // x shared
288     K xk = x->key;
289     V xv = x->value;
290     bool h_color = h->red();
291 
292     __node *xl = __retain(x->left());
293     __node *xr = __retain(x->right);
294     __release(x);
295 
296     h->right = xl;
297     h->set_red(true);
298 
299     return __make_node(micron::move(xk), micron::move(xv), h, xr, h_color);
300   }

316     K xk = x->key;
317     V xv = x->value;

321     __node *xr = __retain(x->right);

394   static __node *
395   __insert_impl(const __node *h, const K &k, Vf &&v, bool &inserted)
396   {
397     if ( !h ) [[unlikely]] {
398       inserted = true;
399       return __make_node(k, micron::forward<Vf>(v), nullptr, nullptr, true);
400     }
401 
402     // cache first comparison
403     const bool less = k < h->key;
404 
405     // build the recursive child into a local BEFORE retaining the sibling
406     __node *n;
407     if ( less ) [[likely]] {
408       __node *nl = __insert_impl(h->left(), k, micron::forward<Vf>(v), inserted);
409       __node *nr = __retain(h->right);
410       n = __clone_with(h, nl, nr, h->red());
411     } else if ( h->key < k ) {
412       __node *nr = __insert_impl(h->right, k, micron::forward<Vf>(v), inserted);
413       __node *nl = __retain(h->left());
414       n = __clone_with(h, nl, nr, h->red());
415     } else {
416       // update value
417       __node *nl = __retain(h->left());
418       __node *nr = __retain(h->right);
419       n = __make_node(h->key, micron::forward<Vf>(v), nl, nr, h->red());
420       inserted = false;
421     }
422 
423     return __fixup(n);
424   }

466   __erase_min_impl(const __node *h)
467   {

485   static __node *
486   __erase_impl(const __node *h, const K &k, bool &erased)
487   {
488     if ( !h ) [[unlikely]] {
489       erased = false;
490       return nullptr;
491     }
492 
493     __node *n = __acquire_mutable(h);
494     const bool less = k < n->key;
495 
496     if ( less ) {
497       if ( !n->left() ) [[unlikely]] {
498         erased = false;
499         return n;
500       }
501 
502       if ( !__is_red(n->left()) && n->left() && !__is_red(n->left()->left()) ) [[unlikely]]
503         n = __move_red_left(n);
504 
505       __node *old_left = n->left();
506       n->set_left(__erase_impl(old_left, k, erased));
507       __release(old_left);
508     } else {
509       if ( __is_red(n->left()) ) [[unlikely]]
510         n = __rotate_right(n);
511 
512       const bool equal = !less && !(n->key < k);
513 
514       if ( equal && !n->right ) [[unlikely]] {
515         erased = true;
516         __release(n);
517         return nullptr;
518       }
519 
520       if ( !n->right ) [[unlikely]] {
521         erased = false;
522         return __fixup(n);
523       }
524 
525       if ( !__is_red(n->right) && n->right && !__is_red(n->right->left()) ) [[unlikely]]
526         n = __move_red_right(n);
527 
528       if ( equal ) [[unlikely]] {
529         const __node *m = __find_min(n->right);
530 
531         __node *replacement = __make_node(m->key, m->value, __retain(n->left()), __retain(n->right), n->red());
532         __release(n);
533         n = replacement;
534 
535         __node *old_right = n->right;
536         n->right = __erase_min_impl(old_right);
537         __release(old_right);
538         erased = true;
539       } else {
540         __node *old_right = n->right;
541         n->right = __erase_impl(old_right, k, erased);
542         __release(old_right);
543       }
544     }
545 
546     return __fixup(n);
547   }

782   class const_iterator
783   {
784     static constexpr usize __max_depth = 64;
785     const __node *__stack[__max_depth];
786     usize __depth;
787     const __node *__pin;
788 
789     inline void
790     __push_left(const __node *n)
791     {
792       while ( n ) [[likely]] {
793         __stack[__depth++] = n;
794         n = n->left();
795       }
796     }
797 
798   public:
799     const_iterator() : __depth(0), __pin(nullptr) { }
800 
801     explicit const_iterator(const __node *root) : __depth(0), __pin(root)
802     {
803       __retain(const_cast<__node *>(root));
804       __push_left(root);
805     }
806 
807     const_iterator(const const_iterator &o) : __depth(o.__depth), __pin(o.__pin)
808     {
809       for ( usize i = 0; i < __depth; ++i ) __stack[i] = o.__stack[i];
810       __retain(const_cast<__node *>(__pin));
811     }

814     {
815       for ( usize i = 0; i < __depth; ++i ) __stack[i] = o.__stack[i];
816       o.__pin = nullptr;
817       o.__depth = 0;
818     }

821     operator=(const const_iterator &o)
822     {
823       if ( this != &o ) {
824         __release(const_cast<__node *>(__pin));
825         __depth = o.__depth;
826         __pin = o.__pin;
827         for ( usize i = 0; i < __depth; ++i ) __stack[i] = o.__stack[i];
828         __retain(const_cast<__node *>(__pin));

837         __release(const_cast<__node *>(__pin));
838         __depth = o.__depth;
839         __pin = o.__pin;
840         for ( usize i = 0; i < __depth; ++i ) __stack[i] = o.__stack[i];
841         o.__pin = nullptr;

844       return *this;

851     {
852       if ( __depth != o.__depth ) return false;
853       return __depth == 0 || __stack[__depth - 1] == o.__stack[o.__depth - 1];
854     }

940   {
