18 template<typename T> struct double_list_node {
19   T data;

41   // tear the whole chain down, destroying each node exactly once
42   inline void
43   __destroy()
44   {
45     double_list_node_t *ptr = root;
46     while ( ptr != nullptr ) {
47       double_list_node_t *nx = ptr->next;
48       delete ptr;      // ~double_list_node destroys data once
49       ptr = nx;
50     }
51     root = nullptr;
52   }

54   // deep-copy src's chain into a freshly empty (root == nullptr) list
55   inline void
56   __clone_from(const double_list_node_t *src)
57   {
58     double_list_node_t **tail = &root;
59     double_list_node_t *prev = nullptr;
60     for ( const double_list_node_t *p = src; p != nullptr; p = p->next ) {
61       __impl_heap(p->data, tail);      // *tail = new node(p->data, nullptr, nullptr)
62       (*tail)->prev = prev;
63       prev = *tail;
64       tail = &(*tail)->next;
65     }
66   }

179   double_list(const size_t cnt) : root(nullptr)
180   {
181     if ( cnt == 0 ) return;
182     __impl_heap(T(), &root);
183     double_list_node_t *ptr = root;
184     for ( size_t i = 1; i < cnt; i++ ) {
185       __impl_heap(T(), &ptr->next);
186       ptr->next->prev = ptr;
187       ptr = ptr->next;
188     }
189   }

350   {

371   // remove the node itr points to, fixing up both neighbours
372   void
373   erase(const_pointer itr)
374   {
375     if ( itr == nullptr ) exc<except::runtime_error>("micron::erase invalid iterator.");
376     auto *p = const_cast<pointer>(itr);
377     if ( p->prev != nullptr )
378       p->prev->next = p->next;
379     else
380       root = p->next;      // p was the head
381     if ( p->next != nullptr ) p->next->prev = p->prev;
382     delete p;      // ~double_list_node destroys data once
383   }

469   // splice o's entire chain in right after pos
470   void
471   splice(const_pointer pos, double_list &o)
472   {
473     if ( &o == this or o.root == nullptr ) return;
474     if ( pos == nullptr ) exc<except::runtime_error>("micron::splice invalid position.");
475     auto *p = const_cast<pointer>(pos);
476     auto *ohead = o.root;
477     auto *otail = const_cast<pointer>(o.iend());
478     otail->next = p->next;
479     if ( p->next != nullptr ) p->next->prev = otail;
480     p->next = ohead;
481     ohead->prev = p;
482     o.root = nullptr;
483   }

485   bool
486   operator==(const double_list &o) const
487   {
488     const double_list_node_t *a = root;
489     const double_list_node_t *b = o.root;
490     while ( a != nullptr and b != nullptr ) {
491       if ( !(a->data == b->data) ) return false;
492       a = a->next;
493       b = b->next;
494     }
495     return a == nullptr and b == nullptr;
496   }
