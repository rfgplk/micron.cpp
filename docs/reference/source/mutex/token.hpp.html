6 #pragma once
7 
8 #include "mutex.hpp"
9 
10 namespace micron
11 {
12 // tokens are used similar to stop_tokens from the STL (in place of mutex)
13 // used to signal whether an action should be taken, if it should, execute the callback
14 template<typename R, typename... Args> using token_function = R (*)(Args...);
15 
16 template<typename R = void, typename T = micron::mutex *, typename A = void (micron::mutex::*)()> class token
17 {
18   static_assert(micron::is_same_v<T, micron::mutex *>, "token: T must be micron::mutex* (callback dispatch is hardwired)");
19   static_assert(micron::is_same_v<A, void (micron::mutex::*)()>,
20                 "token: A must be micron::mutex's reset PMF (callback dispatch is hardwired)");
21 
22   token_function<R, T, A> fptr;
23   micron::mutex mtx;      // retained only so the callback receives a (non-null) mutex* + its unlock PMF (legacy ABI)
24   atomic_token<bool> fired;
25 
26 public:
27   ~token() { }
28 
29   token() : fptr(nullptr), mtx(), fired(false) { }
30 
31   token(token_function<R, T, A> f) : fptr(f), mtx(), fired(false) { }
32 
33   token(const token &) = delete;
34   token(token &&) = delete;
35   token &operator=(const token &) = delete;
36   token &operator=(token &&) = delete;
37 
38   bool
39   operator()(void)
40   {
41     if ( fired.compare_and_swap(false, true) ) return true;      // first caller wins (single atomic op; no TOCTOU, no held lock)
42     if ( fptr ) fptr(&mtx, mtx.retrieve());                      // later callers dispatch the callback
43     return false;
44   }
45 };
46 };      // namespace micron
