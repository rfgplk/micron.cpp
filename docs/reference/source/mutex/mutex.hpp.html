13 // NOTE: a seq_cst test-and-set spin lock; hard correct,
14 // on x86 the unlock store compiles to a full barrier (xchg / mov+mfence)
15 // use only for _STRICT_ global ordering across unrelated atomics
16 class mutex
17 {
18   atomic_token<bool> tk;
19 
20   // one release definition per lock
21   void
22   reset()
23   {
24     unlock();
25   }
26 
27 public:
28   ~mutex() = default;
29   mutex() = default;
30 
31   auto
32   operator()()
33   {
34     default_backoff bo;
35     while ( !tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED) ) {
36       do {
37         bo.relax();
38       } while ( tk.get(memory_order::relaxed) == ATOMIC_LOCKED );      // TTAS: relaxed-read spin, CAS only when observed OPEN
39     };
40     return &mutex::reset;
41   }
42 
43   bool
44   operator!()
45   {
46     return (tk.get() != ATOMIC_LOCKED);
47   }
48 
49   auto
50   lock()
51   {
52     default_backoff bo;
53     while ( !tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED) ) {
54       do {
55         bo.relax();
56       } while ( tk.get(memory_order::relaxed) == ATOMIC_LOCKED );      // TTAS: relaxed-read spin, CAS only when observed OPEN
57     };
58     return &mutex::reset;
59   }
60 
61   bool
62   try_lock() noexcept
63   {
64     return tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED);
65   }
66 
67   void
68   unlock() noexcept
69   {
70     tk.store(ATOMIC_OPEN);
71   }
72 
73   auto
74   retrieve()
75   {
76     return &mutex::reset;
77   }
78 
79   bool
80   is_locked() const noexcept
81   {
82     return tk.get() == ATOMIC_LOCKED;
83   }
84 
85   template<typename... T> friend void unlock(T &...);
86 
87   mutex(const mutex &) = delete;
88   mutex(mutex &&) = delete;
89   mutex &operator=(const mutex &) = delete;
90 };

297 struct mcs_node {
298   atomic_token<mcs_node *> next;      // successor in the queue
299   atomic_token<bool> waiting;
300 
301   // constexpr so a thread_local array of these is CONSTANT-initialized -- mcs_lock's per-thread
302   // slot table would otherwise need a TLS guard variable and __cxa_thread_atexit, which is exactly
303   // the machinery that drops registrations past MICRON_TDTOR_CAP in freestanding
304   constexpr mcs_node() noexcept : next(nullptr), waiting(false) { }
305 };

307 class queuing_mutex
308 {
309 public:
310   using node_type = mcs_node;
311 
312 private:
313   atomic_token<mcs_node *> tail;
314 
315 #if defined(MICRON_LOCK_STATS)
316   // arrivals, counted at the ONLY point that fixes queue position: after the tail swap. a test
317   // cannot pin arrival order from outside -- a waiter blocks the instant it enqueues, so anything
318   // it publishes beforehand races the swap. absent unless MICRON_LOCK_STATS (see mutex/backoff.hpp).
319   atomic_token<u32> __enqueued{ 0 };
320 #endif
321 
322   void
323   do_unlock(mcs_node &node) noexcept
324   {
325     mcs_node *expected_next = node.next.get(memory_order::acquire);
326 
327     if ( !expected_next ) {
328       mcs_node *observed = &node;
329       if ( tail.compare_exchange_strong(observed, nullptr, memory_order::acq_rel, memory_order::acquire) ) {
330         return;
331       }
332       if ( observed == nullptr ) return;
333 
334       default_backoff bo;
335       [[maybe_unused]] u64 __rounds = 0;
336       while ( !(expected_next = node.next.get(memory_order::acquire)) ) {
337         bo.relax();
338         if ( ++__rounds == (1ull << 24) ) __micron_lock_misuse("queuing_mutex::unlock: node is not the queue head");
339       }
340     }
341 
342     expected_next->waiting.store(false, memory_order::release);
343   }
344 
345 public:
346   queuing_mutex() noexcept : tail(nullptr) { }
347 
348   queuing_mutex(const queuing_mutex &) = delete;
349   queuing_mutex(queuing_mutex &&) = delete;
350   queuing_mutex &operator=(const queuing_mutex &) = delete;
351 
352   auto
353   operator()(mcs_node &node) noexcept
354   {
355     node.next.store(nullptr, memory_order::relaxed);
356     node.waiting.store(true, memory_order::relaxed);
357 
358     mcs_node *prev = tail.swap(&node);
359 #if defined(MICRON_LOCK_STATS)
360     __enqueued.fetch_add(1, memory_order::acq_rel);
361 #endif
362 
363     if ( prev ) {
364       prev->next.store(&node, memory_order::release);
365       default_backoff bo;
366       while ( node.waiting.get(memory_order::acquire) ) bo.relax();
367     }
368     return &queuing_mutex::do_unlock;
369   }
370 
371   void
372   lock(mcs_node &node) noexcept
373   {
374     operator()(node);
375   }
376 
377   bool
378   try_lock(mcs_node &node) noexcept
379   {
380     if ( tail.get(memory_order::relaxed) != nullptr ) return false;
381 
382     node.next.store(nullptr, memory_order::relaxed);
383     node.waiting.store(false, memory_order::relaxed);
384 
385     mcs_node *expected = nullptr;
386     return tail.compare_and_swap(expected, &node);
387   }
388 
389   void
390   unlock(mcs_node &node) noexcept
391   {
392     do_unlock(node);
393   }
394 
395   bool
396   is_locked() const noexcept
397   {
398     return tail.get(memory_order::relaxed) != nullptr;
399   }
400 
401   // threads that have taken a queue position, in arrival order. 0 unless MICRON_LOCK_STATS.
402   [[nodiscard]] u32
403   enqueued() const noexcept
404   {
405 #if defined(MICRON_LOCK_STATS)
406     return __enqueued.get(memory_order::acquire);
407 #else
408     return 0;
409 #endif
410   }
411 };
