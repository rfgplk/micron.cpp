31 template<typename T = void> class rcu_domain
32 {
33   atomic_token<usize> global_epoch;
34   atomic_token<usize> sync_epoch;
35 
36   struct retire_entry {
37     void (*deleter)(void *);
38     void *ptr;
39     usize retire_epoch;
40     retire_entry *next;
41 
42     retire_entry(void (*d)(void *), void *p, usize e) : deleter(d), ptr(p), retire_epoch(e), next(nullptr) { }
43   };
44 
45   atomic<retire_entry *> retire_head;
46   atomic<retire_entry *> retire_tail;
47   atomic_token<usize> retire_count;
48 
49   static constexpr usize MAX_READERS = 256;
50   atomic<rcu_reader_state *> readers[MAX_READERS];
51   atomic_token<usize> reader_count;
52 
53   atomic_token<bool> queue_lock;
54 
55   void
56   lock_queue()
57   {
58     while ( !queue_lock.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED) ) {
59       __cpu_pause();
60     }
61   }
62 
63   void
64   unlock_queue()
65   {
66     queue_lock.store(ATOMIC_OPEN);
67   }
68 
69   void
70   register_reader()
71   {
72     auto &state = rcu_tls_state<T>::state;
73 
74     usize idx = reader_count.fetch_add(1, memory_order_acq_rel);
75     if ( idx >= MAX_READERS ) {
76       reader_count.fetch_sub(1, memory_order_acq_rel);
77       return;
78     }
79 
80     readers[idx].operator=(&state);
81   }
82 
83   void
84   wait_for_readers(usize target_epoch)
85   {
86     usize num_readers = reader_count.get(memory_order_acquire);
87 
88     for ( ;; ) {
89       bool all_clear = true;
90 
91       for ( usize i = 0; i < num_readers; ++i ) {
92         rcu_reader_state *reader = *readers[i].get();
93         if ( !reader ) continue;
94 
95         bool is_active = reader->active.get(memory_order_acquire);
96         if ( !is_active ) continue;
97 
98         usize reader_epoch = reader->epoch.get(memory_order_acquire);
99         if ( reader_epoch < target_epoch ) {
100           all_clear = false;
101           break;
102         }
103       }
104 
105       if ( all_clear ) break;
106 
107       __cpu_pause();
108     }
109   }
110 
111   void
112   process_retirements(usize safe_epoch)
113   {
114     lock_queue();
115 
116     retire_entry *prev = nullptr;
117     retire_entry *curr = *retire_head.get();
118 
119     while ( curr ) {
120       if ( curr->retire_epoch < safe_epoch ) {
121         retire_entry *to_delete = curr;
122 
123         if ( prev ) {
124           prev->next = curr->next;
125         } else {
126           retire_head.operator=(curr->next);
127         }
128 
129         retire_entry *next = curr->next;
130 
131         unlock_queue();
132 
133         to_delete->deleter(to_delete->ptr);
134         delete to_delete;
135         retire_count.fetch_sub(1, memory_order_acq_rel);
136 
137         lock_queue();
138 
139         curr = next;
140         if ( !prev && curr ) {
141           retire_head.operator=(curr);
142         }
143       } else {
144         prev = curr;
145         curr = curr->next;
146       }
147     }
148 
149     if ( !*retire_head.get() ) {
150       retire_tail.operator=(static_cast<retire_entry *>(nullptr));
151     }
152 
153     unlock_queue();
154   }
155 
156 public:
157   rcu_domain() : global_epoch(0), sync_epoch(0), retire_count(0), queue_lock(ATOMIC_OPEN), reader_count(0)
158   {
159     retire_head.operator=(static_cast<retire_entry *>(nullptr));
160     retire_tail.operator=(static_cast<retire_entry *>(nullptr));
161 
162     for ( usize i = 0; i < MAX_READERS; ++i ) {
163       readers[i].operator=(static_cast<rcu_reader_state *>(nullptr));
164     }
165   }

167   ~rcu_domain() { rcu_barrier(*this); }

169   void
170   lock()
171   {
172     auto &state = rcu_tls_state<T>::state;
173 
174     if ( !state.active.get(memory_order_relaxed) ) {
175       register_reader();
176     }
177 
178     state.active.store(true, memory_order_release);
179     state.epoch.store(global_epoch.get(memory_order_acquire), memory_order_release);
180   }
181 
182   void
183   unlock()
184   {
185     auto &state = rcu_tls_state<T>::state;
186     state.active.store(false, memory_order_release);
187   }

214   template<typename U>
215   void
216   retire_impl(U *ptr, void (*deleter)(void *))
217   {
218     usize current_epoch = global_epoch.get(memory_order_acquire);
219     auto *entry = new retire_entry(deleter, static_cast<void *>(ptr), current_epoch);
220 
221     lock_queue();
222 
223     retire_entry *tail = *retire_tail.get();
224     if ( tail ) {
225       tail->next = entry;
226     } else {
227       retire_head.operator=(entry);
228     }
229     retire_tail.operator=(entry);
230     retire_count.fetch_add(1, memory_order_acq_rel);
231 
232     unlock_queue();
233 
234     if ( retire_count.get(memory_order_acquire) > 16 ) {
235       advance_epoch();
236       usize safe_epoch = current_epoch;
237       wait_for_readers(current_epoch + 1);
238       process_retirements(safe_epoch);
239     }
240   }
241 
242   void
243   synchronize_impl()
244   {
245     usize current = global_epoch.fetch_add(1, memory_order_acq_rel);
246     wait_for_readers(current + 1);
247     sync_epoch.store(current + 1, memory_order_release);
248   }

251   barrier_impl()
252   {
253     usize current = global_epoch.fetch_add(1, memory_order_acq_rel);
254 
255     wait_for_readers(current + 1);
256 
257     process_retirements(current + 1);
258 
259     sync_epoch.store(current + 1, memory_order_release);
260   }

430   bool
431   compare_exchange(T *expected, T *desired)
432   {
433     auto *p = ptr.get();
434     T *current = *p;
435 
436     if ( current == expected ) {
437       ptr.release();
438       ptr.operator=(desired);
439 
440       if ( expected ) {

454 template<typename F, typename Domain = void>
455 void
456 rcu_call(F &&func, rcu_domain<Domain> &domain)
457 {
458   struct wrapper {
459     F f;
460 
461     wrapper(F &&fn) : f(micron::forward<F>(fn)) { }
462   };
463 
464   auto *w = new wrapper(micron::forward<F>(func));
465 
466   auto deleter = [](void *p) {
467     auto *wp = static_cast<wrapper *>(p);
468     wp->f();
469     delete wp;
470   };
471 
472   domain.retire_impl(w, deleter);
473 }

482 template<typename T, typename Domain = void> class rcu_batch
483 {
484   rcu_domain<Domain> &domain;
485   T **objects;
486   usize count;
487   usize capacity;
488 
489   void
490   grow()
491   {
492     usize new_cap = capacity * 2;
493     T **new_objs = new T *[new_cap];
494 
495     for ( usize i = 0; i < count; ++i ) {
496       new_objs[i] = objects[i];
497     }
498 
499     delete[] objects;
500     objects = new_objs;
501     capacity = new_cap;
502   }
503 
504 public:
505   explicit rcu_batch(rcu_domain<Domain> &d, usize initial_capacity = 16) : domain(d), count(0), capacity(initial_capacity)
506   {
507     objects = new T *[capacity];
508   }
509 
510   ~rcu_batch()
511   {
512     flush();
513     delete[] objects;
514   }
515 
516   void
517   add(T *ptr)
518   {
519     if ( count >= capacity ) {
520       grow();
521     }
522     objects[count++] = ptr;
523   }
524 
525   void
526   flush()
527   {
528     for ( usize i = 0; i < count; ++i ) {
529       rcu_retire(objects[i], domain);
530     }
531     count = 0;
532   }
533 
534   usize
535   size() const
536   {
537     return count;
538   }
539 
540   rcu_batch(const rcu_batch &) = delete;
541   rcu_batch &operator=(const rcu_batch &) = delete;
542 };
