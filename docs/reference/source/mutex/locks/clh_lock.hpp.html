17 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
18 // CLH queue lock
19 //
20 // FIFO like MCS, and cheaper to release;
21 // no spin waiting
22 
23 struct alignas(cache_line_size()) clh_node {
24   atomic_token<bool> locked;
25 
26   constexpr clh_node() noexcept : locked(false) { }
27 };
28 
29 template<usize MaxThreads = 32, spin_policy P = spin_yield> class basic_clh_lock
30 {
31   static_assert(MaxThreads > 0, "basic_clh_lock needs room for at least one thread");
32 
33   static constexpr u32 __node_count = static_cast<u32>(MaxThreads) + 1u;      // + 1 for the initial dummy
34 
35   alignas(cache_line_size()) atomic_token<clh_node *> __tail;
36   atomic_token<clh_node *> __held;          // the node the holder published, or null when unheld
37   atomic_token<clh_node *> __consumed;      // the node it inherited; recycled by the release
38 
39   alignas(cache_line_size()) atomic_token<u8> __pooled[__node_count];
40   atomic_token<u32> __drawn;      // nodes in existence; only grows when the pool comes up empty
41   atomic_token<u32> __hint;       // most recently returned index: the scan's first guess
42 
43   clh_node __arena[__node_count];
44   [[no_unique_address]] __lock_stats st;
45 
46   clh_node *
47   __take() noexcept
48   {
49     for ( u32 pass = 0; pass < 2u; ++pass ) {
50       const u32 n = __drawn.get(memory_order::relaxed);
51       u32 i = __hint.get(memory_order::relaxed);
52       if ( i >= n ) i = 0u;
53       for ( u32 k = 0; k < n; ++k ) {
54         if ( __pooled[i].get(memory_order::relaxed) != 0u
55              and __pooled[i].compare_and_swap(1u, 0u, memory_order::acq_rel, memory_order::relaxed) )
56           return &__arena[i];
57         if ( ++i >= n ) i = 0u;
58       }
59       u32 d = n;
60       while ( d < __node_count )
61         if ( __drawn.compare_exchange_weak(d, d + 1u, memory_order::acq_rel, memory_order::relaxed) ) return &__arena[d];
62     }
63     return nullptr;
64   }
65 
66   void
67   __give(clh_node *n) noexcept
68   {
69     const u32 i = static_cast<u32>(n - __arena);
70     __pooled[i].store(1u, memory_order::release);
71     __hint.store(i, memory_order::relaxed);
72   }
73 
74   void
75   reset() noexcept
76   {
77     unlock();
78   }
79 
80 public:
81   using node_type = clh_node;
82 
83   ~basic_clh_lock() = default;
84 
85   basic_clh_lock() noexcept : __tail(nullptr), __held(nullptr), __consumed(nullptr), __drawn(1), __hint(0)
86   {
87     for ( u32 i = 0; i < __node_count; ++i ) __pooled[i].store(0u, memory_order::relaxed);
88     __arena[0].locked.store(false, memory_order::relaxed);
89     __tail.store(&__arena[0], memory_order::release);
90   }
91 
92   basic_clh_lock(const basic_clh_lock &) = delete;
93   basic_clh_lock(basic_clh_lock &&) = delete;
94   basic_clh_lock &operator=(const basic_clh_lock &) = delete;
95 
96   auto
97   operator()()
98   {
99     clh_node *me = __take();
100     if ( me == nullptr ) micron::exc<except::thread_error>("clh_lock arena exhausted; raise its MaxThreads");
101 
102     me->locked.store(true, memory_order::relaxed);
103     clh_node *pred = __tail.swap(me, memory_order::acq_rel);
104 
105     if ( pred->locked.get(memory_order::acquire) ) {
106       __counted_backoff<P> bo(st);
107       while ( pred->locked.get(memory_order::acquire) ) bo.relax();
108     }
109 
110     __consumed.store(pred, memory_order::relaxed);
111     __held.store(me, memory_order::release);      // release: pairs with the acquire in unlock()
112     st.note_acquire();
113     return &basic_clh_lock::reset;
114   }
115 
116   auto
117   lock()
118   {
119     return operator()();
120   }
121 
122   // CLH has no way to back out of the tail swap;
123   // this succeeds only when the queue is provably empty
124   bool
125   try_lock() noexcept
126   {
127     clh_node *t = __tail.get(memory_order::acquire);
128     if ( t == nullptr or t->locked.get(memory_order::acquire) ) return false;
129 
130     clh_node *me = __take();
131     if ( me == nullptr ) return false;      // the arena is full; try_lock may not throw
132 
133     me->locked.store(true, memory_order::relaxed);
134     if ( !__tail.compare_and_swap(t, me, memory_order::acq_rel, memory_order::acquire) ) {
135       me->locked.store(false, memory_order::relaxed);
136       __give(me);
137       return false;
138     }
139 
140     if ( t->locked.get(memory_order::acquire) ) {
141       if ( __tail.compare_and_swap(me, t, memory_order::acq_rel, memory_order::relaxed) ) {
142         me->locked.store(false, memory_order::relaxed);
143         __give(me);
144         return false;
145       }
146       __counted_backoff<P> bo(st);
147       while ( t->locked.get(memory_order::acquire) ) bo.relax();
148       __give(t);
149       me->locked.store(false, memory_order::release);
150       return false;
151     }
152 
153     __consumed.store(t, memory_order::relaxed);
154     __held.store(me, memory_order::release);
155     st.note_acquire();
156     return true;
157   }
158 
159   // pooling the inherited node before the handoff is safe
160   void
161   unlock() noexcept
162   {
163     clh_node *me = __held.get(memory_order::acquire);
164     if ( me == nullptr ) return;
165 
166     clh_node *pred = __consumed.get(memory_order::relaxed);
167     __held.store(nullptr, memory_order::release);
168     __give(pred);
169     me->locked.store(false, memory_order::release);
170   }
171 
172   auto
173   retrieve() noexcept
174   {
175     return &basic_clh_lock::reset;
176   }
177 
178   bool
179   operator!() const noexcept
180   {
181     return !is_locked();
182   }
183 
184   bool
185   is_locked() const noexcept
186   {
187     const clh_node *t = __tail.get(memory_order::relaxed);
188     return t != nullptr and t->locked.get(memory_order::relaxed);
189   }
190 
191   [[nodiscard]] u32
192   participants() const noexcept
193   {
194     return __drawn.get(memory_order::relaxed) - 1u;
195   }
196 
197   [[nodiscard]] const __lock_stats &
198   stats() const noexcept
199   {
200     return st;
201   }
202 
203   template<typename... T> friend void unlock(T &...);
204 };
