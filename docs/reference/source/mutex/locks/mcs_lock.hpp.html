21 // %%%%%%%%%%%%%%%%%%
22 // queuing_mutex
23 //
24 // WARNING: RELEASE IS BY THE LOCK, NOT BY THE THREAD;
25 // unlock() with nothing held is noop; a second seqd unlock() of one acquisition is a noop; not recursive; THE ACQUIRING THREAD MUST NEVER
26 // EXIT AND THE LOCK MUST NOT BE DTORD WHILE AN ACQ IS PENDING
27 enum class __mcs_slot_state : u8 { idle = 0, busy = 1, releasing = 2 };
28 
29 struct __mcs_slot {
30   mcs_node node;
31   atomic_token<__mcs_slot_state> state;
32 
33   constexpr __mcs_slot() noexcept : node(), state(__mcs_slot_state::idle) { }
34 };
35 
36 class mcs_lock
37 {
38   using __pool = __lock_slot_table<__mcs_slot>;
39 
40   const u64 __id;
41   queuing_mutex __q;
42   atomic_token<__mcs_slot *> __holder;
43   [[no_unique_address]] __lock_stats st;
44 
45   static bool
46   __idle(const __mcs_slot &s) noexcept
47   {
48     return s.state.get(memory_order::acquire) == __mcs_slot_state::idle;
49   }
50 
51   [[gnu::noinline]] void
52   __await_node(__mcs_slot *s)
53   {
54     default_backoff bo;
55     for ( ;; ) {
56       // avoid shadowing
57       const __mcs_slot_state __st = s->state.get(memory_order::acquire);
58       if ( __st == __mcs_slot_state::idle ) return;
59       if ( __st == __mcs_slot_state::busy ) {
60         if ( __holder.get(memory_order::acquire) != s )
61           micron::exc<except::thread_error>("mcs_lock re-entered while this thread was still acquiring it");
62         if ( bo.rounds() >= MICRON_MCS_RECURSION_ROUNDS )
63           micron::exc<except::thread_error>("mcs_lock is not recursive: this thread already holds it");
64       }
65       bo.relax();
66     }
67   }
68 
69   void
70   reset() noexcept
71   {
72     unlock();
73   }
74 
75 public:
76   using node_type = mcs_node;
77 
78   ~mcs_lock() { __pool::release(this); }
79 
80   mcs_lock() noexcept : __id(__next_lock_slot_id()), __q(), __holder(nullptr) { }
81 
82   mcs_lock(const mcs_lock &) = delete;
83   mcs_lock(mcs_lock &&) = delete;
84   mcs_lock &operator=(const mcs_lock &) = delete;
85 
86   auto
87   operator()()
88   {
89     bool fresh = false;
90     __mcs_slot *s = __pool::claim_evicting(this, __id, fresh, __idle);
91     if ( fresh ) s->state.store(__mcs_slot_state::idle, memory_order::relaxed);
92 
93     if ( s->state.get(memory_order::acquire) != __mcs_slot_state::idle ) __await_node(s);
94 
95     s->state.store(__mcs_slot_state::busy, memory_order::relaxed);
96     __q.lock(s->node);
97     __holder.store(s, memory_order::release);
98     st.note_acquire();
99     return &mcs_lock::reset;
100   }
101 
102   auto
103   lock()
104   {
105     return operator()();
106   }
107 
108   bool
109   try_lock() noexcept
110   {
111     bool fresh = false;
112 
113     __mcs_slot *s = __pool::try_claim(this, __id, fresh, __idle);
114     if ( s == nullptr ) return false;
115     if ( fresh ) s->state.store(__mcs_slot_state::idle, memory_order::relaxed);
116 
117     if ( s->state.get(memory_order::acquire) != __mcs_slot_state::idle ) return false;
118 
119     s->state.store(__mcs_slot_state::busy, memory_order::relaxed);
120     if ( __q.try_lock(s->node) ) {
121       __holder.store(s, memory_order::release);
122       st.note_acquire();
123       return true;
124     }
125     s->state.store(__mcs_slot_state::idle, memory_order::relaxed);
126     return false;
127   }
128 
129   void
130   unlock() noexcept
131   {
132     __mcs_slot *s = __holder.get(memory_order::acquire);
133     if ( s == nullptr ) return;
134     s->state.store(__mcs_slot_state::releasing, memory_order::release);
135     __holder.store(nullptr, memory_order::release);
136     __q.unlock(s->node);
137     s->state.store(__mcs_slot_state::idle, memory_order::release);
138   }
139 
140   auto
141   retrieve() noexcept
142   {
143     return &mcs_lock::reset;
144   }
145 
146   bool
147   operator!() const noexcept
148   {
149     return !__q.is_locked();
150   }
151 
152   bool
153   is_locked() const noexcept
154   {
155     return __q.is_locked();
156   }
157 
158   [[nodiscard]] bool
159   holds() const noexcept
160   {
161     const __mcs_slot *s = __pool::find(this, __id);
162     return s != nullptr and __holder.get(memory_order::acquire) == s;
163   }
