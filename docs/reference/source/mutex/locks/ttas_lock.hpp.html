16 // ttas_lock
17 //
18 // test-and-test-and-set with exponential backoff
19 template<spin_policy P = spin_yield> class basic_ttas_lock
20 {
21   atomic_token<bool> tk;
22   [[no_unique_address]] __lock_stats st;
23 
24   void
25   reset()
26   {
27     unlock();
28   }
29 
30 public:
31   ~basic_ttas_lock() = default;
32 
33   basic_ttas_lock() : tk(ATOMIC_OPEN) { }
34 
35   explicit basic_ttas_lock(bool state) : tk(state) { }
36 
37   basic_ttas_lock(const basic_ttas_lock &) = delete;
38   basic_ttas_lock(basic_ttas_lock &&) = delete;
39   basic_ttas_lock &operator=(const basic_ttas_lock &) = delete;
40 
41   auto
42   operator()()
43   {
44     if ( tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED, memory_order::acquire, memory_order::relaxed) ) {
45       st.note_acquire();
46       return &basic_ttas_lock::reset;      // uncontended: one CAS, no backoff state touched
47     }
48 
49     __counted_backoff<P> bo(st);
50     for ( ;; ) {
51       while ( tk.get(memory_order::relaxed) == ATOMIC_LOCKED ) bo.relax();
52       if ( tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED, memory_order::acquire, memory_order::relaxed) ) break;
53       bo.relax();      // lost the race to another spinner; back off rather than re-CAS immediately
54     }
55     st.note_acquire();
56     return &basic_ttas_lock::reset;
57   }
58 
59   auto
60   lock()
61   {
62     return operator()();
63   }
64 
65   bool
66   try_lock() noexcept
67   {
68     if ( tk.get(memory_order::relaxed) == ATOMIC_LOCKED ) return false;
69     return tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED, memory_order::acquire, memory_order::relaxed);
70   }
71 
72   void
73   unlock() noexcept
74   {
75     tk.store(ATOMIC_OPEN, memory_order::release);
76   }
77 
78   auto
79   retrieve()
80   {
81     return &basic_ttas_lock::reset;
82   }
83 
84   bool
85   operator!() const
86   {
87     return tk.get(memory_order::relaxed) != ATOMIC_LOCKED;
88   }
89 
90   bool
91   is_locked() const noexcept
92   {
93     return tk.get(memory_order::relaxed) == ATOMIC_LOCKED;
94   }
95 
96   [[nodiscard]] const __lock_stats &
97   stats() const noexcept
98   {
99     return st;
100   }
101 
102   template<typename... T> friend void unlock(T &...);
103 };
104 
105 using ttas_lock = basic_ttas_lock<spin_yield>;
