14 namespace micron
15 {
16 // naming it like this to avoid ridiculous conflicts
17 template<auto F> class do_once
18 {
19   // WARN: one class + one static gate is instantiated per F -> binary bloat
20   static constexpr u8 ST_UNINIT = 0;
21   static constexpr u8 ST_RUNNING = 1;
22   static constexpr u8 ST_DONE = 2;
23 
24   static auto &
25   _gate(void)
26   {
27     static atomic_token<u8> g(ST_UNINIT);
28     return g;
29   }
30 
31   // NOTE: for the "exactly once" guarantee F must be effectively noexcept
32   template<typename... Args>
33   static void
34   _run(Args &&...args)
35   {
36     auto &g = _gate();
37     for ( ;; ) {
38       const u8 s = g.get(memory_order::acquire);
39       if ( s == ST_DONE ) return;      // completed: acquire pairs with the release store below -> F's writes visible
40       if ( s == ST_UNINIT && g.compare_and_swap(ST_UNINIT, ST_RUNNING) ) {
41         struct __unwind {
42           atomic_token<u8> *gp;
43           bool armed;
44 
45           ~__unwind()
46           {
47             if ( armed ) gp->store(ST_UNINIT, memory_order::release);
48           }
49         } guard{ &g, true };
50 
51         F(micron::forward<Args>(args)...);
52         guard.armed = false;
53         g.store(ST_DONE, memory_order::release);      // publish completion AFTER F
54         return;
55       }
56       micron::yield();      // ST_RUNNING (or lost CAS): block until the winner publishes DONE
57     }
58   }
59 
60 public:
61   ~do_once() { }
62 
63   do_once() { _run(); }
64 
65   template<typename... Args> do_once(Args &&...args) { _run(micron::forward<Args>(args)...); }
66 
67   // delete everything else explicitly
68   do_once(const do_once &) = delete;
69   do_once(do_once &&) = delete;
70   do_once &operator=(do_once &&) = delete;
71   do_once &operator=(const do_once &) = delete;
72 };
73 };      // namespace micron
