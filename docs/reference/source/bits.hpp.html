188 constexpr T
189 reverse_bits(T t)
190 {
191   T c{};
192   unsigned char *p = reinterpret_cast<unsigned char *>(&t);
193   unsigned char *q = reinterpret_cast<unsigned char *>(&c);
194   for ( size_t i = 0; i < sizeof(T); ++i ) q[sizeof(T) - 1 - i] = BitReverseTable256[p[i]];
195   return c;

206 template<typename T>
207 constexpr T
208 reverse_bits_per_byte(T x) noexcept
209 {
210   T result = T(0);
211   for ( size_t b = 0; b < sizeof(T); ++b ) {
212     u8 byte = static_cast<u8>(x >> (b * 8));
213     byte = static_cast<u8>((byte * 0x0202020202ULL & 0x010884422010ULL) % 1023);
214     result |= T(byte) << (b * 8);
215   }
216   return result;

364 template<typename T>
365   requires(micron::is_arithmetic_v<T>)
366 constexpr int
367 bitcount(T x) noexcept
368 {
369   if constexpr ( micron::is_same_v<T, int> || micron::is_same_v<T, char> || micron::is_same_v<T, signed char>
370                  || micron::is_same_v<T, unsigned char> || micron::is_same_v<T, unsigned int> || micron::is_same_v<T, short>
371                  || micron::is_same_v<T, unsigned short> )
372     return __builtin_popcount(static_cast<unsigned int>(x) & (~0u >> ((sizeof(unsigned int) - sizeof(T)) * 8)));
373   else if constexpr ( micron::is_same_v<T, long> || micron::is_same_v<T, unsigned long> )
374     return __builtin_popcountl(static_cast<unsigned long>(x));
375   else if constexpr ( micron::is_same_v<T, long long> || micron::is_same_v<T, unsigned long long> )
376     return __builtin_popcountll(static_cast<unsigned long long>(x));
377   else if constexpr ( micron::is_integral_v<T> && sizeof(T) > sizeof(unsigned long long) ) {
378     const micron::make_unsigned_t<T> u = static_cast<micron::make_unsigned_t<T>>(x);
379     return __builtin_popcountll(static_cast<unsigned long long>(u))
380            + __builtin_popcountll(static_cast<unsigned long long>(u >> __word_bits));
381   } else if constexpr ( sizeof(T) < sizeof(unsigned long long) )
382     return __builtin_popcountll(static_cast<unsigned long long>(x) & (~0ULL >> ((sizeof(unsigned long long) - sizeof(T)) * 8)));
383   else
384     return __builtin_popcountll(static_cast<unsigned long long>(x));
385 }

443 template<typename T>
444   requires(micron::is_arithmetic_v<T>)
445 constexpr int
446 countl_zero(T x) noexcept
447 {
448   if ( x == 0 ) return static_cast<int>(sizeof(T) * 8);
449   if constexpr ( micron::is_same_v<T, int> || micron::is_same_v<T, unsigned int> )
450     return __builtin_clz(static_cast<unsigned int>(x));
451   else if constexpr ( micron::is_same_v<T, long> || micron::is_same_v<T, unsigned long> )
452     return __builtin_clzl(static_cast<unsigned long>(x));
453   else if constexpr ( micron::is_same_v<T, long long> || micron::is_same_v<T, unsigned long long> )
454     return __builtin_clzll(static_cast<unsigned long long>(x));
455   else if constexpr ( micron::is_integral_v<T> && sizeof(T) > sizeof(unsigned long long) ) {
456     const micron::make_unsigned_t<T> u = static_cast<micron::make_unsigned_t<T>>(x);
457     const unsigned long long hi = static_cast<unsigned long long>(u >> __word_bits);
458     return hi ? __builtin_clzll(hi) : __word_bits + __builtin_clzll(static_cast<unsigned long long>(u));
459   } else if constexpr ( sizeof(T) < sizeof(unsigned long long) ) {
460     // narrow types: mask off the widening's sign extension, then adjust for padding
461     constexpr int pad = static_cast<int>((sizeof(unsigned long long) - sizeof(T)) * 8);
462     return __builtin_clzll(static_cast<unsigned long long>(x) & (~0ULL >> pad)) - pad;
463   } else
464     return __builtin_clzll(static_cast<unsigned long long>(x));
465 }

706 // Swap bytes i and j within a multi-byte word
707 template<typename T>
708   requires(micron::is_arithmetic_v<T>)
709 constexpr T
710 swap_bytes(T x, int i, int j) noexcept
711 {
712   u8 bi = static_cast<u8>(x >> (i * 8));
713   u8 bj = static_cast<u8>(x >> (j * 8));
714   T result = x;
715   result &= ~((T(0xFF) << (i * 8)) | (T(0xFF) << (j * 8)));
716   result |= (T(bj) << (i * 8)) | (T(bi) << (j * 8));
717   return result;
718 }

819 constexpr u32
820 spread_bits_16(u16 x) noexcept
821 {
822   u32 v = x;
823   v = (v | (v << 8)) & 0x00FF00FFu;
824   v = (v | (v << 4)) & 0x0F0F0F0Fu;
825   v = (v | (v << 2)) & 0x33333333u;
826   v = (v | (v << 1)) & 0x55555555u;
827   return v;
828 }

830 constexpr u16
831 compact_bits_16(u32 x) noexcept
832 {
833   x &= 0x55555555u;
834   x = (x | (x >> 1)) & 0x33333333u;
835   x = (x | (x >> 2)) & 0x0F0F0F0Fu;
836   x = (x | (x >> 4)) & 0x00FF00FFu;
837   x = (x | (x >> 8)) & 0x0000FFFFu;
838   return static_cast<u16>(x);
839 }
