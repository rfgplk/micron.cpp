56 inline max_t
57 __read_small(const char *path, char *buf, usize cap) noexcept
58 {
59   const i32 fd = static_cast<i32>(posix::open(path, posix::o_rdonly | posix::o_cloexec, 0));
60   if ( fd < 0 ) [[unlikely]]
61     return fd;
62   max_t n = posix::read(fd, buf, cap - 1);
63   (void)posix::close(fd);
64   if ( n < 0 ) [[unlikely]]
65     return n;
66   buf[n] = '\0';
67   return n;
68 }
69 
70 inline i32
71 __write_one(const char *path, const char *data, usize len) noexcept
72 {
73   const i32 fd = static_cast<i32>(posix::open(path, posix::o_wronly | posix::o_cloexec, 0));
74   if ( fd < 0 ) [[unlikely]]
75     return fd;
76   const max_t w = posix::write(fd, data, len);
77   (void)posix::close(fd);
78   if ( w < 0 ) [[unlikely]]
79     return static_cast<i32>(w);
80   return w == static_cast<max_t>(len) ? 0 : -error::invalid_arg;
81 }

83 inline i32
84 __parse_int(const char *s) noexcept
85 {
86   i32 v = 0;
87   bool any = false;
88   while ( *s == ' ' || *s == '\t' ) ++s;
89   while ( *s >= '0' && *s <= '9' ) {
90     v = v * 10 + (*s - '0');
91     ++s;
92     any = true;
93   }
94   return any ? v : -1;
95 }

97 inline bool
98 __join(char *out, usize cap, const char *dir, const char *leaf) noexcept
99 {
100   usize i = 0;
101   for ( const char *p = dir; *p; ++p ) {
102     if ( i + 2 >= cap ) return false;
103     out[i++] = *p;
104   }
105   if ( i + 2 >= cap ) return false;
106   out[i++] = '/';
107   for ( const char *p = leaf; *p; ++p ) {
108     if ( i + 1 >= cap ) return false;
109     out[i++] = *p;
110   }
111   out[i] = '\0';
112   return true;
113 }

115 inline usize
116 __len(const char *s) noexcept
117 {
118   usize n = 0;
119   while ( s[n] ) ++n;
120   return n;
121 }

125 __fs_join(char *out, usize cap, const char *leaf) noexcept

127   return __join(out, cap, fs_root, leaf);

130 inline max_t
131 __read_fs(const char *leaf, char *buf, usize cap) noexcept
132 {
133   char path[384];
134   if ( !__fs_join(path, sizeof(path), leaf) ) [[unlikely]]
135     return -error::name_too_long;
136   return __read_small(path, buf, cap);

141 // policy state
142 [[nodiscard]] inline status_t
143 status(void) noexcept
144 {
145   status_t st{};
146   char buf[32];
147 
148   const max_t e = __impl::__read_fs("enforce", buf, sizeof(buf));
149   if ( e < 0 ) return st;      // selinuxfs is not mounted -> SELinux is not in play
150 
151   st.present = true;
152   st.state = (__impl::__parse_int(buf) == 1) ? mode::enforcing : mode::permissive;
153 
154   if ( __impl::__read_fs("policyvers", buf, sizeof(buf)) >= 0 ) st.policyvers = __impl::__parse_int(buf);
155   if ( __impl::__read_fs("mls", buf, sizeof(buf)) >= 0 ) st.mls = __impl::__parse_int(buf) == 1;
156   if ( __impl::__read_fs("deny_unknown", buf, sizeof(buf)) >= 0 ) st.deny_unknown = __impl::__parse_int(buf) == 1;
157   return st;
158 }

202 // security context
203 class context
204 {
205 public:
206   static constexpr usize max_len = 512;
207 
208   struct part {
209     const char *data = nullptr;
210     usize size = 0;
211 
212     [[nodiscard]] constexpr bool
213     empty(void) const noexcept
214     {
215       return size == 0;
216     }
217   };
218 
219 private:
220   char __raw[max_len]{};
221   usize __len = 0;
222   u16 __sep[3]{};
223   u8 __seps = 0;
224   bool __trunc = false;
225 
226 public:
227   constexpr context(void) noexcept = default;
228 
229   // an over-long label does not become a SHORTER label: it becomes no label, flagged, because a
230   // silently clipped MLS range compares equal to things it is not
231   [[nodiscard]] static context
232   parse(const char *s, usize n) noexcept
233   {
234     context c{};
235     if ( s == nullptr ) return c;
236     while ( n > 0 && s[n - 1] == '\0' ) --n;
237     while ( n > 0 && s[n - 1] == '\n' ) --n;
238     if ( n >= max_len ) {
239       c.__trunc = true;
240       return c;
241     }
242     if ( n == 0 ) return c;
243 
244     for ( usize i = 0; i < n; ++i ) {
245       c.__raw[i] = s[i];
246       if ( s[i] == ':' && c.__seps < 3 ) c.__sep[c.__seps++] = static_cast<u16>(i);
247     }
248     c.__raw[n] = '\0';
249     c.__len = n;
250     return c;
251   }
252 
253   [[nodiscard]] static context
254   parse(const char *s) noexcept
255   {
256     return parse(s, s ? __impl::__len(s) : 0);
257   }
258 
259   [[nodiscard]] constexpr bool
260   valid(void) const noexcept
261   {
262     return __len > 0 && __seps >= 2;
263   }
264 
265   // the input did not fit in max_len. Never valid(), and never equal to anything
266   [[nodiscard]] constexpr bool
267   truncated(void) const noexcept
268   {
269     return __trunc;
270   }
271 
272   [[nodiscard]] constexpr explicit
273   operator bool(void) const noexcept
274   {
275     return valid();
276   }
277 
278   [[nodiscard]] constexpr const char *
279   str(void) const noexcept
280   {
281     return __raw;
282   }
283 
284   [[nodiscard]] constexpr usize
285   size(void) const noexcept
286   {
287     return __len;
288   }
289 
290   [[nodiscard]] constexpr bool
291   has_range(void) const noexcept
292   {
293     return __seps == 3;
294   }
295 
296   [[nodiscard]] constexpr part
297   user(void) const noexcept
298   {
299     return __seps >= 1 ? part{ __raw, __sep[0] } : part{};
300   }
301 
302   [[nodiscard]] constexpr part
303   role(void) const noexcept
304   {
305     return __seps >= 2 ? part{ __raw + __sep[0] + 1, static_cast<usize>(__sep[1] - __sep[0] - 1) } : part{};
306   }
307 
308   [[nodiscard]] constexpr part
309   type(void) const noexcept
310   {
311     if ( __seps < 2 ) return part{};
312     const usize start = __sep[1] + 1;
313     const usize end = (__seps == 3) ? __sep[2] : __len;
314     return part{ __raw + start, end - start };
315   }
316 
317   [[nodiscard]] constexpr part
318   range(void) const noexcept
319   {
320     if ( __seps < 3 ) return part{};
321     return part{ __raw + __sep[2] + 1, __len - __sep[2] - 1 };
322   }
323 
324   // WARNING: two contexts that did not parse are NOT equal. This is an access predicate before it is
325   // a value comparison, and "both unreadable" must never read as "the same label"
326   [[nodiscard]] constexpr bool
327   operator==(const context &o) const noexcept
328   {
329     if ( !valid() || !o.valid() ) return false;
330     if ( __len != o.__len ) return false;
331     for ( usize i = 0; i < __len; ++i )
332       if ( __raw[i] != o.__raw[i] ) return false;
333     return true;
334   }
335 };

349 [[nodiscard]] constexpr const char *

351 {
352   switch ( a ) {

371 // a context that did not fit is an error, not a value -- otherwise is_first() reports success on a

373 inline result<context>
374 __wrap(const context &c) noexcept
375 {
376   if ( c.truncated() ) [[unlikely]]
377     return result<context>{ error_t(-error::name_too_long) };
378   return result<context>{ c };
379 }

382 [[nodiscard]] inline result<context>
383 get_attr(attr a) noexcept
384 {
385   char path[64];
386   if ( !__impl::__join(path, sizeof(path), "/proc/self/attr", name_of(a)) ) [[unlikely]]
387     return result<context>{ error_t(-error::name_too_long) };
388 
389   // one byte MORE than a context can hold, so an over-long attr reaches parse() as over-long
390   // instead of arriving pre-clipped and looking well-formed
391   char buf[context::max_len + 1];
392   const max_t n = __impl::__read_small(path, buf, sizeof(buf));
393   if ( n < 0 ) [[unlikely]]
394     return result<context>{ error_t(static_cast<i32>(n)) };
395   return __impl::__wrap(context::parse(buf, static_cast<usize>(n)));
396 }

398 [[nodiscard]] inline result<context>

404 [[nodiscard]] inline i32
405 set_attr(attr a, const context &c) noexcept
406 {
407   if ( !c.valid() ) [[unlikely]]
408     return -error::invalid_arg;
409   char path[64];
410   if ( !__impl::__join(path, sizeof(path), "/proc/self/attr", name_of(a)) ) [[unlikely]]
411     return -error::name_too_long;
412   return __impl::__write_one(path, c.str(), c.size());
413 }

420     return -error::name_too_long;

422 }
