30 template<typename T>
31 concept is_syscall_group = requires {
32   typename T::policy_tag;
33   { T::calls } -> micron::convertible_to<const i32 *>;
34   { T::count } -> micron::convertible_to<usize>;
35 } && micron::is_same_v<typename T::policy_tag, syscall_group_tag>;
