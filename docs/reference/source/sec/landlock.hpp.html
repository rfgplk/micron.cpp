65 #define __MICRON_SEC_BITOPS(T)                                                                                                             \
66   [[nodiscard]] constexpr T operator|(T a, T b) noexcept { return static_cast<T>(static_cast<u64>(a) | static_cast<u64>(b)); }             \
67   [[nodiscard]] constexpr T operator&(T a, T b) noexcept { return static_cast<T>(static_cast<u64>(a) & static_cast<u64>(b)); }             \
68   [[nodiscard]] constexpr T operator~(T a) noexcept { return static_cast<T>(~static_cast<u64>(a)); }                                       \
69   constexpr T &operator|=(T &a, T b) noexcept { return a = a | b; }                                                                        \
70   constexpr T &operator&=(T &a, T b) noexcept { return a = a & b; }                                                                        \
71   [[nodiscard]] constexpr bool any(T a) noexcept { return static_cast<u64>(a) != 0; }                                                      \
72   [[nodiscard]] constexpr u64 bits(T a) noexcept { return static_cast<u64>(a); }

95 {
96   const i32 v = posix::landlock_abi_version();

103 [[nodiscard]] inline i32
104 abi_level(void) noexcept
105 {
106   const i32 v = atom::load(&__impl::__abi_slot, micron::atomic_relaxed);
107   if ( v != 0 ) [[likely]]
108     return v;
109   return __impl::__probe_abi();
110 }

139 class ruleset
140 {
141   i32 __fd = -error::bad_file_number;
142   i32 __abi = 0;
143   access_fs __handled = access_fs::none;
144   access_net __handled_net = access_net::none;
145   scope __scoped = scope::none;
146 
147   void
148   __alive(void) const
149   {
150     if ( __fd < 0 ) [[unlikely]]
151       exc<except::system_error>("micron::sec::landlock::ruleset, no open ruleset fd");
152   }
153 
154   [[nodiscard]] i32
155   __check(void) const noexcept
156   {
157     if ( __fd < 0 ) [[unlikely]]
158       return __fd;
159     return 0;
160   }
161 
162   constexpr ruleset(i32 fd, i32 ab, access_fs h, access_net hn, scope sc) noexcept
163       : __fd(fd), __abi(ab), __handled(h), __handled_net(hn), __scoped(sc)
164   {
165   }
166 
167   void
168   __create(access_fs handled, access_net handled_net, scope scoped)
169   {
170     __abi = abi_level();
171     if ( __abi <= 0 ) [[unlikely]]
172       exc<except::system_error>("micron::sec::landlock::ruleset, landlock unavailable on this kernel");
173 
174     __handled = handled & static_cast<access_fs>(posix::landlock_fs_mask_for(__abi));
175     __handled_net = handled_net & static_cast<access_net>(posix::landlock_net_mask_for(__abi));
176     __scoped = scoped & static_cast<scope>(posix::landlock_scope_mask_for(__abi));
177 
178     if ( !any(__handled) && !any(__handled_net) && !any(__scoped) ) [[unlikely]]
179       exc<except::invalid_argument>("micron::sec::landlock::ruleset, nothing left to handle after ABI narrowing");
180 
181     posix::landlock_ruleset_attr_t attr{ bits(__handled), bits(__handled_net), bits(__scoped) };
182     __fd = posix::landlock_create_ruleset(&attr, posix::landlock_ruleset_size_for(__abi), 0);
183     if ( __fd < 0 ) [[unlikely]]
184       exc<except::system_error>("micron::sec::landlock::ruleset, landlock_create_ruleset failed");
185   }
186 
187 public:
188   ~ruleset() { close(); }
189 
190   explicit ruleset(access_fs handled, access_net handled_net = access_net::none, scope scoped = scope::none)
191   {
192     __create(handled, handled_net, scoped);
193   }

286   [[nodiscard]] i32
287   allow_fd(i32 dir_fd, access_fs allowed) noexcept
288   {
289     if ( i32 e = __check(); e ) [[unlikely]]
290       return e;
291     if ( dir_fd < 0 ) [[unlikely]]
292       return -error::bad_file_number;
293 
294     const access_fs granted = allowed & __handled;
295     if ( !any(granted) ) [[unlikely]]
296       return -error::invalid_arg;
297     return posix::landlock_add_path_beneath(__fd, dir_fd, bits(granted));
298   }

312 private:
313   [[nodiscard]] i32
314   __open_rule_dir(const char *path, bool follow) noexcept
315   {
316     constexpr i32 base = posix::o_path | posix::o_cloexec | posix::o_directory;
317     if ( follow ) return static_cast<i32>(posix::open(path, base, 0));
318 
319     if ( micron::kernel::since(micron::kernel::feature::openat2) ) {
320       posix::open_how how{};
321       how.flags = static_cast<u64>(base);
322       how.mode = 0;
323       how.resolve = posix::resolve_no_symlinks | posix::resolve_no_magiclinks;
324       const i32 fd = posix::openat2(posix::at_fdcwd, path, how);
325       if ( fd >= 0 ) return fd;
326       if ( fd != -error::bad_syscall ) return fd;
327     }
328     return static_cast<i32>(posix::open(path, base | posix::o_nofollow, 0));
329   }

331   [[nodiscard]] i32
332   __add_path(const char *path, access_fs allowed, bool follow) noexcept
333   {
334     if ( i32 e = __check(); e ) [[unlikely]]
335       return e;
336     if ( path == nullptr ) [[unlikely]]
337       return -error::invalid_arg;
338 
339     const i32 pf = __open_rule_dir(path, follow);
340     if ( pf < 0 ) [[unlikely]]
341       return pf;
342     const i32 r = allow_fd(pf, allowed);
343     (void)posix::close(pf);
344     return r;
345   }

349   allow_port(u64 port, access_net allowed) noexcept
350   {
351     if ( i32 e = __check(); e ) [[unlikely]]
352       return e;
353     const access_net granted = allowed & __handled_net;

361   {
362     if ( i32 e = __check(); e ) [[unlikely]]
363       return e;
364     if ( i32 nnp = micron::prctl(PR_SET_NO_NEW_PRIVS, 1UL); nnp < 0 ) [[unlikely]]
365       return nnp;
366     return posix::landlock_restrict_self(__fd, flags);
367   }

370 enum class abi_policy : u8 {
371   narrow = 0,
372   strict,
373 };

375 [[nodiscard]] inline ruleset
376 try_ruleset(access_fs handled, access_net handled_net = access_net::none, scope scoped = scope::none,
377             abi_policy pol = abi_policy::narrow) noexcept
378 {
379   const i32 a = abi_level();
380   if ( a <= 0 ) [[unlikely]]
381     return ruleset::__failed(a < 0 ? a : -error::bad_syscall);      // abi_level() never answers 0 or -1
382 
383   const access_fs h = handled & static_cast<access_fs>(posix::landlock_fs_mask_for(a));
384   const access_net hn = handled_net & static_cast<access_net>(posix::landlock_net_mask_for(a));
385   const scope sc = scoped & static_cast<scope>(posix::landlock_scope_mask_for(a));
386 
387   if ( pol == abi_policy::strict ) {
388     if ( any(handled & ~h) || any(handled_net & ~hn) || any(scoped & ~sc) ) [[unlikely]]
389       return ruleset::__failed(-error::op_not_supported);
390   }
391 
392   if ( !any(h) && !any(hn) && !any(sc) ) [[unlikely]]
393     return ruleset::__failed(-error::invalid_arg);
394 
395   posix::landlock_ruleset_attr_t attr{ bits(h), bits(hn), bits(sc) };
396   const i32 fd = posix::landlock_create_ruleset(&attr, posix::landlock_ruleset_size_for(a), 0);
397   if ( fd < 0 ) [[unlikely]]
398     return ruleset::__failed(fd);
399   return ruleset::__adopt(fd, a, h, hn, sc);
400 }
