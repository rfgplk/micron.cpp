372 template<typename Fn>
373   requires(micron::invocable<Fn, landlock::ruleset &> && __resultable<micron::invoke_result_t<Fn, landlock::ruleset &>>)
374 auto
375 with_ruleset(landlock::access_fs handled, Fn &&fn) -> result<__unit_if_void_t<micron::invoke_result_t<Fn, landlock::ruleset &>>>
376 {
377   using R = result<__unit_if_void_t<micron::invoke_result_t<Fn, landlock::ruleset &>>>;
378   landlock::ruleset rs = landlock::try_ruleset(handled);
379   if ( !rs.valid() ) [[unlikely]]
380     return R{ error_t(rs.error()) };
381   if constexpr ( micron::is_void_v<micron::invoke_result_t<Fn, landlock::ruleset &>> ) {
382     micron::forward<Fn>(fn)(rs);
383     return R{ unit_t{} };
384   } else {
385     return R{ micron::forward<Fn>(fn)(rs) };
386   }
387 }

399 // WARNING: only seccomp
400 template<usize N>
401 [[nodiscard]] inline result<sandbox::exit_status>
402 confined(const seccomp::filter_builder<N> &fb, i32 (*body)(void)) noexcept
403 {
404   using R = result<sandbox::exit_status>;
405   if ( !fb.valid() ) [[unlikely]]
406     return R{ error_t(-error::invalid_arg) };
407   sandbox box;
408   box.seccomp(fb);
409   if ( !box.configured() ) [[unlikely]]
410     return R{ error_t(box.config_fault().err) };
411   return box.run_to_completion(body);
412 }
