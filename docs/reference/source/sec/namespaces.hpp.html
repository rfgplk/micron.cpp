139 inline void
140 __ns_path(char *out, usize cap, i32 pid, const char *name) noexcept
141 {
142   usize i = 0;
143   auto put = [&](const char *s) {
144     while ( *s && i + 1 < cap ) out[i++] = *s++;
145   };
146   put("/proc/");
147   if ( pid < 0 ) {
148     put("self");
149   } else {
150     char digits[12];
151     usize n = 0;
152     u32 v = static_cast<u32>(pid);
153     if ( v == 0 ) digits[n++] = '0';
154     while ( v ) {
155       digits[n++] = static_cast<char>('0' + (v % 10));
156       v /= 10;
157     }
158     while ( n && i + 1 < cap ) out[i++] = digits[--n];
159   }
160   put("/ns/");
161   put(name);
162   out[i < cap ? i : cap - 1] = '\0';
163 }

308 [[nodiscard]] inline ns_handle
309 open_of(i32 pid, ns_kind k) noexcept
310 {
311   const char *nm = name_of(k);
312   if ( nm == nullptr ) [[unlikely]]
313     return ns_handle::__adopt(-error::invalid_arg, k);
314 
315   char path[64];
316   __impl::__ns_path(path, sizeof(path), pid, nm);
317   const i32 fd = static_cast<i32>(posix::open(path, posix::o_rdonly | posix::o_cloexec, 0));
318   return ns_handle::__adopt(fd, k);
319 }

338 inline i32
339 __write_proc(const char *path, const char *data, usize len) noexcept
340 {
341   const i32 fd = static_cast<i32>(posix::open(path, posix::o_wronly | posix::o_cloexec, 0));
342   if ( fd < 0 ) [[unlikely]]
343     return fd;
344   const max_t w = posix::write(fd, data, len);
345   (void)posix::close(fd);
346   if ( w < 0 ) [[unlikely]]
347     return static_cast<i32>(w);
348   return w == static_cast<max_t>(len) ? 0 : -error::invalid_arg;
349 }

351 // "<inner> <outer> <count>\n" with no formatter
352 inline usize
353 __map_line(char *out, usize cap, u32 inner, u32 outer, u32 count) noexcept
354 {
355   auto put_u32 = [&](usize i, u32 v) -> usize {
356     char d[12];
357     usize n = 0;
358     if ( v == 0 ) d[n++] = '0';
359     while ( v ) {
360       d[n++] = static_cast<char>('0' + (v % 10));
361       v /= 10;
362     }
363     while ( n && i + 1 < cap ) out[i++] = d[--n];
364     return i;
365   };
366   usize i = put_u32(0, inner);
367   if ( i + 1 < cap ) out[i++] = ' ';
368   i = put_u32(i, outer);
369   if ( i + 1 < cap ) out[i++] = ' ';
370   i = put_u32(i, count);
371   if ( i + 1 < cap ) out[i++] = '\n';
372   out[i] = '\0';
373   return i;
374 }

399 [[nodiscard]] inline i32
400 map_to_root(u32 outer_uid, u32 outer_gid) noexcept
401 {
402   if ( const i32 r = deny_setgroups(); r < 0 ) [[unlikely]]
403     return r;

409 [[nodiscard]] inline i32
410 unshare_user_as_root(ns_kind extra = ns_kind::none) noexcept
411 {
412   const u32 uid = static_cast<u32>(posix::getuid());
413   const u32 gid = static_cast<u32>(posix::getgid());
414   if ( const i32 r = unshare(ns_kind::user); r < 0 ) [[unlikely]]
415     return r;
416   if ( const i32 r = map_to_root(uid, gid); r < 0 ) [[unlikely]]
417     return r;
418   return any(extra) ? unshare(extra) : 0;
419 }

439 class enter_scope
440 {
441   ns_handle __saved;
442   i32 __entered = -error::bad_file_number;
443   bool __restored = false;
444 
445   i32
446   __restore(void) noexcept
447   {
448     if ( __restored || __entered < 0 ) return 0;
449     if ( !__saved.valid() ) return __saved.error();
450     const i32 r = __saved.enter();
451     if ( r >= 0 ) __restored = true;
452     return r;
453   }
454 
455 public:
456   enter_scope(const enter_scope &) = delete;
457   enter_scope &operator=(const enter_scope &) = delete;
458 
459   explicit enter_scope(const ns_handle &target) noexcept : __saved(open_self(target.kind()))
460   {
461     __entered = target.enter();
462     if ( __entered < 0 ) __saved.close();
463   }
464 
465   ~enter_scope()
466   {
467     const i32 r = __restore();
468     if ( r < 0 ) __impl::__restore_slot = r;
469   }
470 
471   [[nodiscard]] i32
472   restore(void) noexcept
473   {
474     return __restore();
475   }
476 
477   [[nodiscard]] bool
478   ok(void) const noexcept
479   {
480     return __entered >= 0;
481   }

483   [[nodiscard]] i32
484   error(void) const noexcept
485   {
