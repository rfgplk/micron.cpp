38 [[nodiscard]] constexpr const char *
39 __rel_put_old(const char *nr, const char *po) noexcept
40 {
41   if ( nr == nullptr || po == nullptr ) return nullptr;
42   usize n = 0;
43   while ( nr[n] != '\0' ) ++n;
44   while ( n > 1 && nr[n - 1] == '/' ) --n;
45   if ( n == 0 ) return nullptr;
46   if ( n == 1 && nr[0] == '/' ) return po;
47   for ( usize i = 0; i < n; ++i )
48     if ( po[i] == '\0' || po[i] != nr[i] ) return nullptr;
49   if ( po[n] != '/' ) return nullptr;
50   return po + n;
51 }

53 // WARNING: MS_REC is ignored by a MS_REMOUNT|MS_BIND remount
54 [[nodiscard]] inline i32
55 __remount_attrs(const char *target, bool recursive, unsigned long keep_flags, u64 attr_set, bool rdonly) noexcept
56 {
57   if ( recursive ) {
58     posix::mount_attr_t at{};
59     at.attr_set = attr_set | (rdonly ? posix::mount_attr_rdonly : 0ull);
60     if ( at.attr_set == 0 ) return 0;
61     return posix::mount_setattr(posix::at_fdcwd, target, static_cast<u32>(posix::at_recursive), at);
62   }
63   unsigned long fl = (keep_flags & ~posix::ms_rec) | posix::ms_bind | posix::ms_remount;
64   if ( rdonly ) fl |= posix::ms_rdonly;
65   if ( attr_set & posix::mount_attr_nosuid ) fl |= posix::ms_nosuid;
66   if ( attr_set & posix::mount_attr_nodev ) fl |= posix::ms_nodev;
67   if ( attr_set & posix::mount_attr_noexec ) fl |= posix::ms_noexec;
68   return static_cast<i32>(posix::mount(nullptr, target, nullptr, fl, nullptr));
69 }

71 [[nodiscard]] inline i32
72 __remount_ro(const char *target, bool recursive, unsigned long keep_flags = 0) noexcept
73 {
74   return __remount_attrs(target, recursive, keep_flags, 0, true);
75 }

221 template<usize Max, typename... Rules> struct seccomp_policy_n {
222   using policy_tag = seccomp_policy_tag;
223   static constexpr usize capacity = Max;
224 
225   static constexpr seccomp::filter_builder<Max>
226   build(u32 default_action) noexcept
227   {
228     seccomp::filter_builder<Max> fb;
229     fb.require_native_arch();
230     (Rules::template emit<Max>(fb), ...);
231     fb.__seal(default_action);
232     return fb;
233   }
234 
235   // WARNING: TSYNC, see CVE-2017-5206
236   static int
237   apply(void) noexcept
238   {
239     auto fb = build(seccomp::act_errno(static_cast<u16>(error::permissions)));
240     return seccomp::load_tsync(fb, true);
241   }
242 
243   static int
244   apply_this_thread_only(void) noexcept
245   {
246     auto fb = build(seccomp::act_errno(static_cast<u16>(error::permissions)));
247     return seccomp::load(fb, true);
248   }

288 template<micron::cap... Cs> struct capability_policy {
289   using policy_tag = capability_policy_tag;
290 
291   static int
292   apply_pre_uid(void) noexcept
293   {
294     (void)posix::cap_set_keepcaps(1);
295     u64 keep = 0;
296     ((keep |= micron::cap_bit(Cs)), ...);
297     const u64 drop = ~keep & micron::cap_all_mask;
298     int err = 0;
299     for ( u32 i = 0; i < static_cast<u32>(micron::cap::__count); ++i ) {
300       if ( !(drop & (u64(1) << i)) ) continue;
301       const int r = micron::drop_bounding(static_cast<micron::cap>(i));
302       if ( r < 0 && r != -error::invalid_arg && err == 0 ) err = r;
303     }
304     return err;
305   }
306 
307   static int
308   apply_post_uid(void) noexcept
309   {
310     micron::ucap_set_t cs = micron::ucap_set_t::none();
311     (cs.grant(Cs), ...);
312     const int r = micron::set_caps(cs);
313     if ( r < 0 ) return r;
314     ((void)micron::raise_ambient(Cs), ...);
315     return 0;
316   }
317 
318   static int
319   apply(void) noexcept
320   {
321     const int r = apply_pre_uid();
322     if ( r < 0 ) return r;
323     return apply_post_uid();
324   }
325 };

431 // pivot_root, then detach the old root
432 template<micron::fixed_string New, micron::fixed_string Put> struct pivot_to {
433   static int
434   apply(void) noexcept
435   {
436     const char *rel = __impl::__rel_put_old(New.c_str(), Put.c_str());
437     if ( rel == nullptr ) return -error::invalid_arg;
438     int r = static_cast<int>(posix::pivot_root(New.c_str(), Put.c_str()));
439     if ( r < 0 ) return r;
440     r = static_cast<int>(posix::chdir("/"));
441     if ( r < 0 ) return r;
442     return static_cast<int>(posix::umount2(rel, posix::mnt_detach));
443   }

464 template<micron::fixed_string Src, micron::fixed_string Dst, bool ReadOnly = false, bool Recursive = true, bool Harden = true> struct bind {
465   static int
466   apply(void) noexcept
467   {
468     constexpr unsigned long fl = Recursive ? (posix::ms_bind | posix::ms_rec) : posix::ms_bind;
469     const int r = static_cast<int>(posix::mount(Src.c_str(), Dst.c_str(), nullptr, fl, nullptr));
470     if ( r < 0 ) return r;
471     constexpr u64 attrs = Harden ? __impl::__bind_default_attrs : 0ull;
472     if constexpr ( ReadOnly || Harden ) return __impl::__remount_attrs(Dst.c_str(), Recursive, fl, attrs, ReadOnly);
473     return r;
474   }

576 template<typename... Rs> struct landlock_policy {
577   using policy_tag = landlock_policy_tag;
578 
579   static constexpr landlock::access_fs handled = (landlock::access_fs::none | ... | Rs::access);
580 
581   static int
582   apply(void) noexcept
583   {
584     if ( !landlock::available() ) return -error::bad_syscall;
585 
586     landlock::ruleset rs = landlock::try_ruleset(handled);
587     if ( !rs.valid() ) return rs.error();
588 
589     int err = 0;
590     (
591         [&rs, &err]<typename R>(R *) {
592           const i32 r = R::add(rs);
593           if ( r < 0 && err == 0 ) err = r;
594         }(static_cast<Rs *>(nullptr)),
595         ...);
596     if ( err < 0 ) return err;
597     return rs.restrict_self();
598   }
599 };
