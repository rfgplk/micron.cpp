76 //  any<Ts...>
77 //
78 //  a value holds __exactly one__ alternative, tracked by a usize discriminant (npos/empty/valueless)
79 //
80 //  storage is always stack-allocated; capacity = element-wise max of sizeof / alignof over all Ts

82 template<typename... Ts>
83   requires(sizeof...(Ts) > 0)
84 class any
85 {
86   static constexpr usize npos = micron::numeric_limits<usize>::max();
87   static constexpr usize __storage_size = __impl::max_of_values<sizeof(Ts)...>::value;
88   static constexpr usize __storage_align = __impl::max_of_values<alignof(Ts)...>::value;
89 
90   alignas(__storage_align) unsigned char storage[__storage_size];
91   usize __idx = npos;
92 
93   void (*__destroy)(void *) noexcept = nullptr;
94   void (*__copy)(void *, const void *) = nullptr;
95   void (*__move)(void *, void *) noexcept = nullptr;
96 
97   template<typename T>
98   static void
99   destroy_fn(void *p) noexcept
100   {
101     static_cast<T *>(p)->~T();
102   }
103 
104   template<typename T>
105   static void
106   copy_fn(void *dst, const void *src)
107   {
108     if constexpr ( micron::is_copy_constructible_v<T> ) {
109       new (dst) T(*static_cast<const T *>(src));
110     }
111     // else: any's copy ctor / copy-assign are deleted whenever some
112     // alternative isn't copy-constructible, so this branch can never
113     // run at runtime; keeping the function instantiable lets emplace<U>
114     // assign &copy_fn<U> unconditionally for any U in the pack.
115   }
116 
117   template<typename T>
118   static void
119   move_fn(void *dst, void *src) noexcept
120   {
121     new (dst) T(micron::move(*static_cast<T *>(src)));
122     destroy_fn<T>(src);
123   }

125   void
126   reset_impl(void) noexcept
127   {
128     if ( __idx != npos ) {
129       __destroy(storage);
130       __idx = npos;

132       __copy = nullptr;
133       __move = nullptr;

195   template<typename T, typename... Args>

198   emplace(Args &&...args)
199   {

376 // option<T, F>  binary sum type/tunion of exactly two alternatives
377 //  equivalent to any<T,F> but uses a 3-valued enum discriminant
378 //
379 //  T and F must be distinct types!
380 
381 template<typename T, typename F>
382   requires(micron::distinct<T, F> && !micron::is_void_v<F>)
383 class option
384 {
385   static constexpr usize __t_size = __impl::__safe_sizeof<T>;
386   static constexpr usize __t_align = __impl::__safe_alignof<T>;
387   static constexpr usize storage_size = __t_size > sizeof(F) ? __t_size : sizeof(F);
388   static constexpr usize storage_aln = __t_align > alignof(F) ? __t_align : alignof(F);
389 
390   alignas(storage_aln) unsigned char storage[storage_size];
391 
392   enum class which : unsigned char { none = 0, first = 1, second = 2 } __active = which::none;
393 
394   void (*__destroy)(void *) noexcept = nullptr;
395   void (*__copy)(void *, const void *) = nullptr;
396   void (*__move)(void *, void *) noexcept = nullptr;
397 
398   template<typename U>
399   static void
400   destroy_fn(void *p) noexcept
401   {
402     static_cast<U *>(p)->~U();
403   }
404 
405   template<typename U>
406   static void
407   copy_fn(void *dst, const void *src)
408   {
409     if constexpr ( micron::is_copy_constructible_v<U> ) {
410       new (dst) U(*static_cast<const U *>(src));
411     }
412   }
413 
414   template<typename U>
415   static void
416   move_fn(void *dst, void *src) noexcept
417   {
418     new (dst) U(micron::move(*static_cast<U *>(src)));
419     destroy_fn<U>(src);
420   }

422   void
423   reset_impl(void) noexcept
424   {
425     if ( __active != which::none ) {
426       if ( __destroy ) __destroy(storage);
427       __active = which::none;
428       __destroy = nullptr;
429       __copy = nullptr;
430       __move = nullptr;
431     }
432   }

434   template<typename U>

513   template<typename U, typename... Args>
514     requires(micron::same_as<micron::decay_t<U>, T> or micron::same_as<micron::decay_t<U>, F>)
515   decltype(auto)
516   emplace(Args &&...args)
517   {
518     using D = micron::decay_t<U>;
519     reset_impl();
520     if constexpr ( micron::is_void_v<D> ) {
521       static_assert(sizeof...(Args) == 0, "option::emplace<void>() takes no constructor arguments");
522       __active = which::first;
523       // __destroy/__copy/__move stay nullptr; reset_impl + copy/move null-check
524     } else {
525       new (storage) D(micron::forward<Args>(args)...);
526       __active = which_of<D>();
527       __destroy = &destroy_fn<D>;
528       __copy = &copy_fn<D>;
529       __move = &move_fn<D>;
530       return *micron::ptr_cast<D *>(storage);
531     }
532   }
