19 // a span is a stack-allocated, fixed-capacity view of contiguous memory
20 // always owns its memory (on the stack), never heap allocates
21 // always moves, never copies (where avoidable)

519   span &
520   insert(size_type ind, const T &val)
521   {
522     if ( length + 1 > N ) [[unlikely]]
523       exc<except::runtime_error>("micron::span insert() exceeds stack capacity N");
524     if ( ind > length ) [[unlikely]]
525       exc<except::runtime_error>("micron::span insert() index out of range");
526     if ( ind < length ) {
527       new (micron::addr(stack[length])) T(micron::move(stack[length - 1]));
528       for ( size_type i = length - 1; i > ind; --i ) stack[i] = micron::move(stack[i - 1]);
529       stack[ind] = val;
530     } else {
531       new (micron::addr(stack[length])) T(val);
532     }
533     ++length;
534     return *this;
535   }

1107   reverse()
1108   {
1109     if ( length < 2 ) return *this;
1110     pointer lo = stack;
1111     pointer hi = stack + length - 1;
1112     while ( lo < hi ) {
1113       T tmp = micron::move(*lo);

1122   span &
1123   rotate_left(size_type n)
1124   {
1125     if ( length == 0 ) return *this;
1126     n %= length;
1127     if ( n == 0 ) return *this;
1128     auto rev = [](pointer a, pointer b) {
1129       while ( a < b ) {
1130         T tmp = micron::move(*a);
1131         *a = micron::move(*b);
1132         *b = micron::move(tmp);
1133         ++a;
1134         --b;
1135       }
1136     };
1137     rev(stack, stack + n - 1);
1138     rev(stack + n, stack + length - 1);
1139     rev(stack, stack + length - 1);
1140     return *this;
1141   }

1186   template<typename U>
1187   align_result<T, U>
1188   align_to() const
1189   {
1190     const byte *raw = reinterpret_cast<const byte *>(stack);
1191     const size_type total = length * sizeof(T);
1192     const size_type align = alignof(U);
1193 
1194     size_type prefix_bytes = 0;
1195     auto raw_addr = reinterpret_cast<uintptr_t>(raw);
1196     if ( raw_addr % align != 0 ) prefix_bytes = align - (raw_addr % align);
1197     if ( prefix_bytes > total ) prefix_bytes = total;
1198 
1199     size_type remaining = total - prefix_bytes;
1200     size_type middle_count = remaining / sizeof(U);
1201     size_type middle_bytes = middle_count * sizeof(U);
1202     size_type suffix_bytes = remaining - middle_bytes;
1203     size_type prefix_elems = prefix_bytes / sizeof(T);
1204     size_type suffix_elems = suffix_bytes / sizeof(T);
1205 
1206     return { raw_slice<T>(stack, prefix_elems), raw_slice<U>(reinterpret_cast<U *>(const_cast<byte *>(raw) + prefix_bytes), middle_count),
1207              raw_slice<T>(stack + length - suffix_elems, suffix_elems) };
1208   }
