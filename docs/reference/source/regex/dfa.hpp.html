21 inline constexpr i32 kDfaMaxStates = 255;      // state ids fit in a u8
22 inline constexpr u8 kShengAccept = 0x10;       // bit 4 -- $-less accept
23 inline constexpr u8 kShengEol = 0x20;          // bit 5 -- accept only at end ($)

32 struct dfa {
33   i32 nstates = 0;
34   i32 start = 0;
35   u8 *table = nullptr;           // nstates*256: next state id
36   u8 *accept = nullptr;          // nstates: 0/1  ($-less match here)
37   u8 *eol_accept = nullptr;      // nstates: 0/1  (match here iff at end)
38   bool has_sheng = false;
39   u8 masks[256][16];      // valid iff has_sheng
40   u8 start_byte = 0;
41   // in a state that self-loops on all but a few "escape" bytes
42   bool has_accel = false;
43   u8 *accel_type = nullptr;                // nstates: 0 none | 1 single byte (memchr) | 2 small set (truffle)
44   u8 *accel_byte = nullptr;                // nstates: the lone escape byte (type 1)
45   truffle_masks *accel_set = nullptr;      // nstates: escape-set masks (type 2)
46 };

48 inline void
49 dfa_eclose(prog_view pv, u32 pc, char *seen, u64 *key, bool &accept, bool &eol, bool at_start, bool in_eol) noexcept
50 {
51   if ( pc >= pv.ncode ) return;
52   const inst &I = pv.code[pc];
53   if ( I.code == op::Match ) {      // terminal -- never deduped, so both flags can set
54     if ( in_eol )
55       eol = true;
56     else
57       accept = true;
58     return;
59   }
60   if ( seen[pc] ) return;
61   seen[pc] = 1;
62   switch ( I.code ) {
63   case op::Char:
64   case op::Class:
65   case op::Any:
66     if ( !in_eol ) key[pc >> 6] |= (u64(1) << (pc & 63));      // cannot consume after $
67     break;
68   case op::Jmp:
69     dfa_eclose(pv, I.x, seen, key, accept, eol, at_start, in_eol);
70     break;
71   case op::Split:
72     dfa_eclose(pv, I.x, seen, key, accept, eol, at_start, in_eol);
73     dfa_eclose(pv, I.y, seen, key, accept, eol, at_start, in_eol);
74     break;
75   case op::Save:
76     dfa_eclose(pv, pc + 1, seen, key, accept, eol, at_start, in_eol);
77     break;
78   case op::Bol:
79     if ( at_start ) dfa_eclose(pv, pc + 1, seen, key, accept, eol, at_start, in_eol);      // ^: pass only at offset 0
80     break;
81   case op::Eol:
82     dfa_eclose(pv, pc + 1, seen, key, accept, eol, at_start, true);      // $: everything past is end-only
83     break;
84   case op::Match:
85     break;
86   }
87 }

89 inline bool
90 dfa_consumes(const inst &I, u8 c, const charreach *cls, usize ncls) noexcept
91 {
92   switch ( I.code ) {
93   case op::Char:
94     return (u8)I.x == c;
95   case op::Class:
96     return I.x < ncls && cls[I.x].test(c);
97   case op::Any:
98     return true;
99   default:
100     return false;
101   }
102 }

104 inline dfa *
105 build_dfa(prog_view pv, char *seen) noexcept
106 {
107   if ( pv.ncode == 0 || pv.ncode > 256 ) return nullptr;
108 
109   bool has_bol = false;
110   for ( usize i = 0; i < pv.ncode; ++i )
111     if ( pv.code[i].code == op::Bol ) {
112       has_bol = true;
113       break;
114     }
115 
116   // start-anchored iff there is NO way to begin matching without a ^
117   bool start_anchored = false;
118   if ( has_bol ) {
119     u64 k[4] = {};
120     bool a = false, e = false;
121     for ( usize i = 0; i < pv.ncode; ++i ) seen[i] = 0;
122     dfa_eclose(pv, 0, seen, k, a, e, false, false);
123     start_anchored = (k[0] == 0 && k[1] == 0 && k[2] == 0 && k[3] == 0 && !a && !e);
124     if ( !start_anchored ) return nullptr;      // mixed ^ -> Pike
125   }
126   const bool restart = !start_anchored;      // unanchored DFAs re-seed the start every step
127 
128   const i32 __cap = kDfaMaxStates;
129   u64 *keys = micron::alloc<u64>((usize)(__cap + 1) * 4 * sizeof(u64));
130   u8 *acc = micron::alloc<u8>((usize)(__cap + 1));
131   u8 *eolacc = micron::alloc<u8>((usize)(__cap + 1));
132   u8 *tbl = micron::alloc<u8>((usize)(__cap + 1) * 256);
133   i32 nstates = 0;
134   bool overflow = false;
135 
136   auto same = [&](i32 i, const u64 *k) -> bool {
137     const u64 *ki = keys + (usize)i * 4;
138     return ki[0] == k[0] && ki[1] == k[1] && ki[2] == k[2] && ki[3] == k[3];
139   };
140   auto intern = [&](const u64 *k, bool a, bool e) -> i32 {
141     for ( i32 i = 0; i < nstates; ++i )
142       if ( acc[i] == (a ? 1 : 0) && eolacc[i] == (e ? 1 : 0) && same(i, k) ) return i;
143     if ( nstates >= __cap ) {
144       overflow = true;
145       return -1;
146     }
147     u64 *ki = keys + (usize)nstates * 4;
148     ki[0] = k[0];
149     ki[1] = k[1];
150     ki[2] = k[2];
151     ki[3] = k[3];
152     acc[nstates] = a ? 1 : 0;
153     eolacc[nstates] = e ? 1 : 0;
154     return nstates++;
155   };
156 
157   u64 k0[4] = {};
158   bool a0 = false, e0 = false;
159   for ( usize i = 0; i < pv.ncode; ++i ) seen[i] = 0;
160   dfa_eclose(pv, 0, seen, k0, a0, e0, /*at_start=*/true, false);
161   i32 start = intern(k0, a0, e0);
162 
163   for ( i32 sid = 0; sid < nstates && !overflow; ++sid ) {
164     const u64 *ks = keys + (usize)sid * 4;
165     for ( i32 c = 0; c < 256; ++c ) {
166       u64 nk[4] = {};
167       bool na = false, ne = false;
168       for ( usize i = 0; i < pv.ncode; ++i ) seen[i] = 0;
169       if ( restart ) dfa_eclose(pv, 0, seen, nk, na, ne, false, false);      // unanchored restart
170       for ( usize pc = 0; pc < pv.ncode; ++pc ) {
171         if ( !((ks[pc >> 6] >> (pc & 63)) & 1) ) continue;
172         if ( dfa_consumes(pv.code[pc], (u8)c, pv.cls, pv.ncls) ) dfa_eclose(pv, static_cast<u32>(pc + 1), seen, nk, na, ne, false, false);
173       }
174       i32 nid = intern(nk, na, ne);
175       if ( nid < 0 ) break;
176       tbl[(usize)sid * 256 + (usize)c] = (u8)nid;
177     }
178   }
179 
180   if ( overflow || start < 0 ) {
181     micron::free(keys);
182     micron::free(acc);
183     micron::free(eolacc);
184     micron::free(tbl);
185     return nullptr;
186   }
187 
188   dfa *d = micron::alloc<dfa>(sizeof(dfa));
189   d->nstates = nstates;
190   d->start = start;
191   d->table = micron::alloc<u8>((usize)nstates * 256);
192   d->accept = micron::alloc<u8>((usize)nstates);
193   d->eol_accept = micron::alloc<u8>((usize)nstates);
194   for ( i32 s = 0; s < nstates; ++s ) {
195     d->accept[s] = acc[s];
196     d->eol_accept[s] = eolacc[s];
197     for ( i32 c = 0; c < 256; ++c ) d->table[(usize)s * 256 + (usize)c] = tbl[(usize)s * 256 + (usize)c];
198   }
199 
200   d->has_accel = false;
201   d->accel_type = micron::alloc<u8>((usize)nstates);
202   d->accel_byte = micron::alloc<u8>((usize)nstates);
203   d->accel_set = micron::alloc<truffle_masks>((usize)nstates * sizeof(truffle_masks));
204   for ( i32 s = 0; s < nstates; ++s ) {
205     d->accel_type[s] = 0;
206     d->accel_byte[s] = 0;
207     if ( d->accept[s] ) continue;      // accepting states return before any self-loop scan
208     const u8 *row = d->table + (usize)s * 256;
209     i32 nesc = 0, first = -1;
210     charreach esc;
211     for ( i32 c = 0; c < 256; ++c )
212       if ( row[c] != (u8)s ) {
213         ++nesc;
214         if ( first < 0 ) first = c;
215         esc.set((u8)c);
216       }
217     if ( nesc == 1 ) {      // one escape byte -> memchr
218       d->accel_type[s] = 1;
219       d->accel_byte[s] = (u8)first;
220       d->has_accel = true;
221     } else if ( nesc >= 2 && nesc <= 32 ) {      // few escape bytes -> truffle set-scan
222       d->accel_type[s] = 2;
223       d->accel_set[s] = truffle_build(esc);
224       d->has_accel = true;
225     }
226   }
227 
228   d->has_sheng = kShengAvailable && (nstates <= 16);
229   if constexpr ( kShengAvailable ) {
230     if ( d->has_sheng ) {
231       auto sbyte = [&](i32 s) -> u8 { return (u8)(s | (d->accept[s] ? kShengAccept : 0) | (d->eol_accept[s] ? kShengEol : 0)); };
232       for ( i32 c = 0; c < 256; ++c )
233         for ( i32 s = 0; s < 16; ++s ) {
234           i32 ns = (s < nstates) ? d->table[(usize)s * 256 + (usize)c] : 0;
235           d->masks[c][s] = sbyte(ns);
236         }
237       d->start_byte = sbyte(start);
238     }
239   }
240 
241   micron::free(keys);
242   micron::free(acc);
243   micron::free(eolacc);
244   micron::free(tbl);
245   return d;
246 }

248 inline void
249 dfa_free(dfa *d) noexcept
250 {
251   if ( !d ) return;
252   if ( d->table ) micron::free(d->table);
253   if ( d->accept ) micron::free(d->accept);
254   if ( d->eol_accept ) micron::free(d->eol_accept);
255   if ( d->accel_type ) micron::free(d->accel_type);
256   if ( d->accel_byte ) micron::free(d->accel_byte);
257   if ( d->accel_set ) micron::free(d->accel_set);
258   micron::free(d);
259 }

261 // one PSHUFB per byte (<=16 states), unrolled 4 bytes per iteration; four loads are independent, shuffles chained on cur
262 inline bool
263 dfa_sheng_has_match(const dfa *d, const char *in, usize n) noexcept
264 {
265   if ( d->start_byte & kShengAccept ) return true;
266   if ( n == 0 ) return (d->start_byte & kShengEol) != 0;
267   const u8(*masks)[16] = d->masks;
268   micron::simd::v8 cur((signed char)d->start_byte);
269   usize i = 0;
270 
271   for ( ; i + 4 <= n; i += 4 ) {
272     micron::simd::v8 m1;
273     m1.uload(const_cast<u8 *>(masks[(u8)in[i + 0]]));
274     cur = m1.shuffle(cur);
275     u8 a1 = cur.byte0();
276     micron::simd::v8 m2;
277     m2.uload(const_cast<u8 *>(masks[(u8)in[i + 1]]));
278     cur = m2.shuffle(cur);
279     u8 a2 = cur.byte0();
280     micron::simd::v8 m3;
281     m3.uload(const_cast<u8 *>(masks[(u8)in[i + 2]]));
282     cur = m3.shuffle(cur);
283     u8 a3 = cur.byte0();
284     micron::simd::v8 m4;
285     m4.uload(const_cast<u8 *>(masks[(u8)in[i + 3]]));
286     cur = m4.shuffle(cur);
287     u8 a4 = cur.byte0();
288     if ( (a1 | a2 | a3 | a4) & kShengAccept ) return true;      // any of the 4 accepted
289   }
290 
291   for ( ; i < n; ++i ) {
292     micron::simd::v8 m;
293     m.uload(const_cast<u8 *>(masks[(u8)in[i]]));
294     cur = m.shuffle(cur);
295     if ( cur.byte0() & kShengAccept ) return true;
296   }
297   return (cur.byte0() & kShengEol) != 0;
298 }

300 inline bool
301 dfa_table_has_match(const dfa *d, const char *in, usize n) noexcept
302 {
303   i32 s = d->start;
304   if ( d->accept[s] ) return true;
305   if ( n == 0 ) return d->eol_accept[s] != 0;
306   for ( usize i = 0; i < n; ++i ) {
307     s = d->table[(usize)s * 256 + (u8)in[i]];
308     if ( d->accept[s] ) return true;
309   }
310   return d->eol_accept[s] != 0;
311 }

313 inline bool
314 dfa_accel_has_match(const dfa *d, const char *in, usize n) noexcept
315 {
316   i32 s = d->start;
317   if ( d->accept[s] ) return true;
318   usize i = 0;
319   while ( i < n ) {
320     u8 at = d->accel_type[s];
321     if ( at == 1 ) {
322       const char *fp = micron::memchr(in + i, d->accel_byte[s], n - i);
323       if ( !fp ) break;      // escape byte absent -> state self-loops to the end
324       i = (usize)(fp - in);
325     } else if ( at == 2 ) {
326       usize idx = truffle_find_first(in + i, n - i, d->accel_set[s]);
327       if ( idx >= n - i ) break;
328       i += idx;
329     }
330     s = d->table[(usize)s * 256 + (u8)in[i]];
331     if ( d->accept[s] ) return true;
332     ++i;
333   }
334   return d->eol_accept[s] != 0;
335 }

337 inline bool
338 dfa_has_match(const dfa *d, const char *in, usize n) noexcept
339 {
340   if ( d->has_accel ) return dfa_accel_has_match(d, in, n);
341   if constexpr ( kShengAvailable ) {
342     if ( d->has_sheng ) return dfa_sheng_has_match(d, in, n);
343   }
344   return dfa_table_has_match(d, in, n);
345 }
