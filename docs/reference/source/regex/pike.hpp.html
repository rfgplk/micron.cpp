11 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
12 // the Pike virtual machine
13 // Thompson-NFA simulation that runs the compiled program against the input
14 
15 namespace micron
16 {
17 namespace rgx
18 {
19 
20 struct vm {
21   prog_view pv;
22   usize nslots = 0;      // == 2 * (ngroups + 1)
23   usize cap = 0;         // max threads per list == pv.ncode
24   const char *in = nullptr;
25   usize n = 0;
26   bool anchored = false;
27   usize start_at = 0;
28 
29   u32 *a_pc = nullptr;
30   i64 *a_sav = nullptr;
31   usize a_n = 0;
32   u32 *b_pc = nullptr;
33   i64 *b_sav = nullptr;
34   usize b_n = 0;
35 
36   u32 *seen = nullptr;
37   u32 gen = 0;
38   i64 *work = nullptr;
39   i64 *mcaps = nullptr;
40   bool matched = false;
41   i64 mstart = 0;
42   i64 mend = 0;
43 
44   constexpr void
45   addthread(u32 *lpc, i64 *lsav, usize &ln, u32 pc, usize sp, i64 *sav) noexcept
46   {
47     if ( seen[pc] == gen ) return;
48     seen[pc] = gen;
49     const inst &I = pv.code[pc];
50     switch ( I.code ) {
51     case op::Jmp:
52       addthread(lpc, lsav, ln, I.x, sp, sav);
53       break;
54     case op::Split:
55       addthread(lpc, lsav, ln, I.x, sp, sav);
56       addthread(lpc, lsav, ln, I.y, sp, sav);
57       break;
58     case op::Save:
59       if ( I.x < nslots ) {
60         i64 old = sav[I.x];
61         sav[I.x] = (i64)sp;
62         addthread(lpc, lsav, ln, pc + 1, sp, sav);
63         sav[I.x] = old;
64       } else {
65         addthread(lpc, lsav, ln, pc + 1, sp, sav);
66       }
67       break;
68     case op::Bol:
69       if ( sp == 0 ) addthread(lpc, lsav, ln, pc + 1, sp, sav);
70       break;
71     case op::Eol:
72       if ( sp == n ) addthread(lpc, lsav, ln, pc + 1, sp, sav);
73       break;
74     default:
75       if ( ln < cap ) {
76         lpc[ln] = pc;
77         for ( usize s = 0; s < nslots; ++s ) lsav[ln * nslots + s] = sav[s];
78         ++ln;
79       }
80       break;
81     }
82   }
83 
84   constexpr bool
85   run() noexcept
86   {
87     for ( usize s = 0; s < nslots; ++s ) {
88       work[s] = -1;
89       mcaps[s] = -1;
90     }
91     matched = false;
92     a_n = 0;
93     b_n = 0;
94     ++gen;
95     addthread(a_pc, a_sav, a_n, 0, start_at, work);
96 
97     for ( usize sp = start_at;; ++sp ) {
98       if ( a_n == 0 && (matched || anchored) ) break;
99       int c = (sp < n) ? (int)(unsigned char)in[sp] : -1;
100       ++gen;
101       b_n = 0;
102       for ( usize i = 0; i < a_n; ++i ) {
103         u32 pc = a_pc[i];
104         i64 *sav = &a_sav[i * nslots];
105         const inst &I = pv.code[pc];
106         switch ( I.code ) {
107         case op::Char:
108           if ( c == (int)I.x ) addthread(b_pc, b_sav, b_n, pc + 1, sp + 1, sav);
109           break;
110         case op::Class:
111           if ( c >= 0 && pv.cls[I.x].test((unsigned char)c) ) addthread(b_pc, b_sav, b_n, pc + 1, sp + 1, sav);
112           break;
113         case op::Any:
114           if ( c >= 0 ) addthread(b_pc, b_sav, b_n, pc + 1, sp + 1, sav);
115           break;
116         case op::Match: {
117           i64 st = sav[0], en = (i64)sp;
118           if ( !matched || st < mstart || (st == mstart && en > mend) ) {
119             for ( usize s = 0; s < nslots; ++s ) mcaps[s] = sav[s];
120             mstart = st;
121             mend = en;
122             matched = true;
123           }
124           break;
125         }
126         default:
127           break;
128         }
129       }
130       if ( !matched && !anchored && (sp + 1) <= n ) addthread(b_pc, b_sav, b_n, 0, sp + 1, work);
131 
132       u32 *tp = a_pc;
133       a_pc = b_pc;
134       b_pc = tp;
135       i64 *ts = a_sav;
136       a_sav = b_sav;
137       b_sav = ts;
138       a_n = b_n;
139 
140       if ( c < 0 ) break;
141     }
142     return matched;
143   }
144 };
