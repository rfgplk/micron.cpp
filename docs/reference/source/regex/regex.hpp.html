34 inline constexpr usize kMaxGroups = 32;      // capture groups 1..32 (+ whole-match 0)
35 
36 // {ptr,len} normalisation: accept a raw C string or any has_cstr object.
37 struct subject_t {
38   const char *p = nullptr;
39   usize n = 0;
40 };
41 
42 constexpr subject_t
43 as_subject(const char *s) noexcept
44 {
45   usize n = 0;
46   if ( s )
47     while ( s[n] ) ++n;
48   return { s, n };
49 }
50 
51 template<has_cstr S>
52 constexpr subject_t
53 as_subject(const S &s) noexcept
54 {
55   return { s.c_str(), s.size() };
56 }

58 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
59 // rmatch
60 struct rmatch {
61   bool matched = false;
62   u32 ng = 0;      // number of groups, including group 0
63   const char *base = nullptr;
64   i64 caps[2 * (kMaxGroups + 1)] = {};
65 
66   constexpr bool
67   has_match() const noexcept
68   {
69     return matched;
70   }
71 
72   constexpr explicit
73   operator bool() const noexcept
74   {
75     return matched;
76   }
77 
78   constexpr usize
79   groups() const noexcept
80   {
81     return ng;
82   }
83 
84   constexpr i64
85   group_start(usize i) const noexcept
86   {
87     return (matched && i < ng) ? caps[2 * i] : -1;
88   }
89 
90   constexpr i64
91   group_end(usize i) const noexcept
92   {
93     return (matched && i < ng) ? caps[2 * i + 1] : -1;
94   }
95 
96   // span of capture group i, empty/invalid if unset
97   constexpr raw_slice<const char>
98   group(usize i) const noexcept
99   {
100     if ( !matched || i >= ng ) return raw_slice<const char>{};
101     i64 a = caps[2 * i], b = caps[2 * i + 1];
102     if ( a < 0 || b < 0 || b < a ) return raw_slice<const char>{};
103     return raw_slice<const char>{ base + a, (usize)(b - a) };
104   }
105 };
106 
107 constexpr void
108 finish_match(rmatch &r, vm &m, const char *base, u32 ngroups) noexcept
109 {
110   bool ok = m.run();
111   r.matched = ok;
112   r.base = base;
113   r.ng = ngroups + 1;
114   if ( ok )
115     for ( usize s = 0; s < m.nslots && s < 2 * (kMaxGroups + 1); ++s ) r.caps[s] = m.mcaps[s];
116 }

118 // %%%%%%%%%%%%%%%%%%%%%%%%%%
119 // cmatch<"pattern">(input)
120 template<fixed_string P, class S>
121 constexpr rmatch
122 cmatch(const S &input) noexcept
123 {
124   constexpr usize n = P.size();
125   constexpr usize maxn = n * 4 + 8;
126   constexpr usize maxc = n + 4;
127   constexpr usize maxi = n * 8 + 16;
128   constexpr usize maxg = (n < kMaxGroups ? n : kMaxGroups);
129   constexpr usize maxslots = 2 * (maxg + 1);
130 
131   rmatch r;
132 
133   node nodes[maxn]{};
134   charreach cls[maxc]{};
135   inst code[maxi]{};
136   usize ncode = 0, ncls = 0;
137   u32 ngroups = 0;
138   if ( !compile_regex(P.data(), n, nodes, maxn, cls, maxc, code, maxi, ncode, ncls, ngroups) ) return r;
139   if ( ngroups > maxg ) return r;
140 
141   usize nslots = 2 * (ngroups + 1);
142 
143   u32 a_pc[maxi]{};
144   u32 b_pc[maxi]{};
145   i64 a_sav[maxi * maxslots]{};
146   i64 b_sav[maxi * maxslots]{};
147   u32 seen[maxi]{};
148   i64 work[maxslots]{};
149   i64 mcaps[maxslots]{};
150 
151   subject_t sub = as_subject(input);
152 
153   vm m;
154   m.pv = prog_view{ code, ncode, cls, ncls, ngroups };
155   m.nslots = nslots;
156   m.cap = ncode;
157   m.in = sub.p;
158   m.n = sub.n;
159   m.a_pc = a_pc;
160   m.a_sav = a_sav;
161   m.b_pc = b_pc;
162   m.b_sav = b_sav;
163   m.seen = seen;
164   m.gen = 0;
165   m.work = work;
166   m.mcaps = mcaps;
167 
168   finish_match(r, m, sub.p, ngroups);
169   return r;
170 }

172 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
173 // regex: runtime pattern; heap allocated, move only
174 class regex
175 {
176   inst *__code = nullptr;
177   charreach *__cls = nullptr;
178   usize __ncode = 0;
179   usize __ncls = 0;
180   u32 __ngroups = 0;
181   bool __ok = false;
182   charreach __first{};            // bytes that can begin a match (prefilter)
183   bool __nullable = false;        // pattern can match the empty string
184   dfa *__dfa = nullptr;           // SIMD/table has_match accelerator (or null)
185   bool __prefer_dfa = false;      // pattern has .* / wide-class loop -> DFA beats the prefilter
186 
187   void
188   build(const char *pat, usize len) noexcept
189   {
190     usize maxn = len * 4 + 16, maxc = len + 8, maxi = len * 8 + 32;
191     node *nodes = micron::alloc<node>(maxn * sizeof(node));
192     __cls = micron::alloc<charreach>(maxc * sizeof(charreach));
193     __code = micron::alloc<inst>(maxi * sizeof(inst));
194     usize ncls = 0;
195     u32 ng = 0;
196     __ok = compile_regex(pat, len, nodes, maxn, __cls, maxc, __code, maxi, __ncode, ncls, ng);
197     __ncls = ncls;
198     __ngroups = ng;
199     micron::free(nodes);
200     if ( ng > kMaxGroups ) __ok = false;
201     if ( __ok ) {
202       prog_view pv{ __code, __ncode, __cls, __ncls, __ngroups };
203       char *seen = micron::alloc<char>(__ncode ? __ncode : 1);
204       first_info fi = compute_first(pv, seen);
205       __first = fi.first;
206       __nullable = fi.nullable;
207       __dfa = build_dfa(pv, seen);      // null if unsuitable -> Pike VM fallback
208       micron::free(seen);
209       if ( __dfa ) {
210         usize fc = __first.count();
211         bool pf_usable = !__nullable && fc >= 1 && fc < 256;
212         if ( !pf_usable ) {
213           __prefer_dfa = true;      // no prefilter exists -> Pike walk
214         } else if ( __dfa->accel_type[__dfa->start] != 0 ) {
215           __prefer_dfa = true;
216         } else {
217           // NOTE: Pike is slow when a match can span a long run (`.*`, `[^x]*`)
218           for ( usize k = 0; k < __ncode; ++k ) {
219             if ( __code[k].code == op::Any || __code[k].code == op::Bol ) {      // .* / ^-anchored -> DFA
220               __prefer_dfa = true;
221               break;
222             }
223             if ( __code[k].code == op::Class && __code[k].x < __ncls && __cls[__code[k].x].count() > 200 ) {
224               __prefer_dfa = true;
225               break;
226             }
227           }
228         }
229       }
230     }
231   }

233   rmatch
234   do_search(const char *p, usize n, bool anchored) const noexcept
235   {
236     rmatch r;
237     if ( !__ok ) return r;
238     usize nslots = 2 * (__ngroups + 1);
239 
240     u32 *a_pc = micron::alloc<u32>(__ncode * sizeof(u32));
241     u32 *b_pc = micron::alloc<u32>(__ncode * sizeof(u32));
242     i64 *a_sav = micron::alloc<i64>(__ncode * nslots * sizeof(i64));
243     i64 *b_sav = micron::alloc<i64>(__ncode * nslots * sizeof(i64));
244     u32 *seen = micron::alloc<u32>(__ncode * sizeof(u32));
245     i64 *work = micron::alloc<i64>(nslots * sizeof(i64));
246     i64 *mcaps = micron::alloc<i64>(nslots * sizeof(i64));
247     for ( usize i = 0; i < __ncode; ++i ) seen[i] = 0;
248 
249     vm m;
250     m.pv = prog_view{ __code, __ncode, __cls, __ncls, __ngroups };
251     m.nslots = nslots;
252     m.cap = __ncode;
253     m.in = p;
254     m.n = n;
255     m.anchored = anchored;
256     m.a_pc = a_pc;
257     m.a_sav = a_sav;
258     m.b_pc = b_pc;
259     m.b_sav = b_sav;
260     m.seen = seen;
261     m.gen = 0;
262     m.work = work;
263     m.mcaps = mcaps;
264 
265     usize fc = __first.count();
266     bool prefiltered = false;
267     if ( !anchored && !__nullable && fc >= 1 && fc < 256 ) {
268       const int kMaxFailedAttempts = 32;
269       unsigned char single = (fc == 1) ? (unsigned char)__first.single() : 0;
270       truffle_masks tm;
271       if ( fc > 1 ) tm = truffle_build(__first);
272 
273       m.anchored = true;
274       bool hit = false;
275       int attempts = 0;
276       for ( usize pos = 0; pos < n; ) {
277         usize cstart;
278         if ( fc == 1 ) {
279           const char *fp = micron::memchr(p + pos, single, n - pos);
280           if ( !fp ) break;
281           cstart = (usize)(fp - p);
282         } else {
283           usize idx = truffle_find_first(p + pos, n - pos, tm);
284           if ( idx >= n - pos ) break;
285           cstart = pos + idx;
286         }
287         m.start_at = cstart;
288         if ( m.run() ) {
289           hit = true;
290           break;
291         }
292         pos = cstart + 1;
293         if ( ++attempts > kMaxFailedAttempts ) break;      // dense candidates -> fall back
294       }
295       if ( hit ) {
296         r.matched = true;
297         r.base = p;
298         r.ng = __ngroups + 1;
299         for ( usize s = 0; s < nslots && s < 2 * (kMaxGroups + 1); ++s ) r.caps[s] = m.mcaps[s];
300         prefiltered = true;
301       } else if ( attempts <= kMaxFailedAttempts ) {
302         // candidates exhausted with no match -> genuinely no match
303         r.matched = false;
304         r.base = p;
305         r.ng = __ngroups + 1;
306         prefiltered = true;
307       }
308     }
309     if ( !prefiltered ) {
310       m.anchored = anchored;
311       m.start_at = 0;
312       finish_match(r, m, p, __ngroups);
313     }
314 
315     micron::free(a_pc);
316     micron::free(b_pc);
317     micron::free(a_sav);
318     micron::free(b_sav);
319     micron::free(seen);
320     micron::free(work);
321     micron::free(mcaps);
322     return r;
323   }

410   int
411   has_match_path() const noexcept
412   {
413     if ( __dfa && __prefer_dfa ) return __dfa->has_accel ? 4 : (__dfa->has_sheng ? 3 : 2);
414     if ( !__nullable ) {
415       usize fc = __first.count();
416       if ( fc >= 1 && fc < 256 ) return 1;

439   has_match(const S &input) const noexcept
440   {

443     if ( __dfa && __prefer_dfa ) return dfa_has_match(__dfa, s.p, s.n);      // .*/wide -> DFA
