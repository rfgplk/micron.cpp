97 struct __atexit_entry {
98   __atexit_fn_t func;
99   void *arg;
100   void *dso;

103 #ifndef MICRON_ATEXIT_CAP
104 #define MICRON_ATEXIT_CAP 4096
105 #endif
106 inline constexpr usize __atexit_cap = MICRON_ATEXIT_CAP;
107 static_assert(__atexit_cap >= 8, "micron: MICRON_ATEXIT_CAP must leave room for the runtime's own handlers.");
108 
109 inline __atexit_entry __atexit_table[__atexit_cap] = {};
110 
111 // WARNING: this is a monotonic allocation cursor, must never be lowered
112 inline u32 __atexit_count = 0;
113 
114 inline u64 __atexit_pub[(__atexit_cap + 63) / 64] = {};
115 inline u32 __atexit_dropped = 0;      // reserved but never published entries the drain gave up on; should stay 0
116 inline bool __exit_in_progress = false;
117 inline bool __fini_fired = false;      // .fini_array is one shot
118 
119 // how many no progress passes the drain will make waiting on a publication before giving up
120 inline constexpr u32 __atexit_publish_rounds = 1u << 16;
121 
122 [[gnu::always_inline]] inline bool
123 __published(u32 idx) noexcept
124 {
125   return (__atomic_load_n(&__atexit_pub[idx / 64], __ATOMIC_ACQUIRE) & (1ull << (idx % 64))) != 0;
126 }
127 
128 inline void
129 __atexit_thunk(void *p) noexcept
130 {
131   using __noarg_fn = void (*)();
132   (reinterpret_cast<__noarg_fn>(p))();
133 }
134 
135 inline int
136 __push(__atexit_fn_t func, void *arg, void *dso = nullptr) noexcept
137 {
138   u32 idx = micron::atom::load(&__atexit_count, micron::atomic_acquire);
139   for ( ;; ) {
140     if ( idx >= __atexit_cap ) return -1;
141     if ( micron::atom::compare_exchange(&__atexit_count, &idx, idx + 1u, false, micron::atomic_acq_rel, __ATOMIC_ACQUIRE) ) break;
142   }
143   __atexit_table[idx].arg = arg;
144   __atexit_table[idx].dso = dso;
145   __atomic_store_n(&__atexit_table[idx].func, func, __ATOMIC_RELAXED);
146   __atomic_fetch_or(&__atexit_pub[idx / 64], 1ull << (idx % 64), __ATOMIC_ACQ_REL);      // releases both fields
147   return 0;
148 }
149 
150 // WARNING: arg may only be read after this returns nonnull
151 [[gnu::always_inline]] inline __atexit_fn_t
152 __claim(u32 idx) noexcept
153 {
154   if ( !__published(idx) ) return nullptr;
155   return __atomic_exchange_n(&__atexit_table[idx].func, static_cast<__atexit_fn_t>(nullptr), __ATOMIC_ACQ_REL);
156 }
157 
158 // WARNING: claim destructors (of potential external libs/so/dlls/drivers) while the DSO is still mapped
159 inline void
160 __finalize_dso(void *dso) noexcept
161 {
162   for ( ;; ) {
163     const u32 count = micron::atom::load(&__atexit_count, micron::atomic_acquire);
164     bool called = false;
165     for ( u32 i = count; i-- > 0; ) {
166       if ( !__published(i) || (dso && __atexit_table[i].dso != dso) ) continue;
167       if ( auto function = __claim(i); function ) {
168         function(__atexit_table[i].arg);
169         called = true;
170         break;      // NOTE: destructors registered by this callback precede the remaining older ones
171       }
172     }
173     if ( !called ) return;
174   }
175 }

191 inline void
192 __drain_atexit_table() noexcept
193 {
194   using namespace __exit_internal;
195   u32 lo = 0;
196   u32 hi = micron::atom::load(&__atexit_count, micron::atomic_acquire);
197   u32 stall = 0;
198   while ( lo != hi ) {
199     u32 pending = hi;      // lowest index seen reserved but unpublished this pass
200     bool progress = false;
201     bool restart = false;
202     for ( u32 i = hi; i-- != lo; ) {
203       if ( !__published(i) ) {
204         pending = i;
205         continue;
206       }
207       __atexit_fn_t f = __claim(i);
208       if ( f == nullptr ) continue;      // another drainer got there first
209       progress = true;
210       f(__atexit_table[i].arg);      // arg is safe to read
211       const u32 next = micron::atom::load(&__atexit_count, micron::atomic_acquire);
212       if ( next != hi ) {
213         hi = next;
214         restart = true;
215         break;      // new registrations precede the remaining older callbacks
216       }
217     }
218     if ( restart ) {
219       stall = 0;
220       continue;
221     }
222     const u32 next = micron::atom::load(&__atexit_count, micron::atomic_acquire);
223     if ( pending == hi ) {      // everything below hi is accounted for
224       if ( next == hi ) break;
225       lo = hi;
226       hi = next;
227       stall = 0;
228       continue;
229     }
230     if ( !progress && next == hi ) {      // only unpublished entries left and nobody moved
231       if ( ++stall > __atexit_publish_rounds ) {
232         for ( u32 i = pending; i < hi; ++i )
233           if ( !__published(i) ) __atomic_fetch_add(&__atexit_dropped, 1u, __ATOMIC_RELAXED);
234         break;
235       }
236       __cpu_pause();
237     } else {
238       stall = 0;
239     }
240     lo = pending;
241     hi = next;
242   }
243 }

245 inline void
246 __run_fini_array() noexcept
247 {
248 #if !defined(MICRON_ATTACH_MODULE)
249   using namespace __exit_internal;
250   bool __fini_expected = false;
251   if ( !micron::atom::compare_exchange(&__fini_fired, &__fini_expected, true, false, micron::atomic_seq_cst, __ATOMIC_RELAXED) ) return;
252   if ( __fini_array_start && __fini_array_end ) {
253     for ( void (**p)(void) = __fini_array_end; p > __fini_array_start; ) {
254       --p;
255       if ( *p ) (*p)();
256     }
257   }
258 #endif
259 }

261 inline void
262 __settle_exit_callbacks() noexcept
263 {
264   for ( ;; ) {
265     const u32 before = micron::atom::load(&__exit_internal::__atexit_count, micron::atomic_acquire);
266     __drain_atexit_table();
267 #if !defined(MICRON_ATTACH_MODULE)
268     micron::__run_thread_dtors();
269 #endif
270     if ( micron::atom::load(&__exit_internal::__atexit_count, micron::atomic_acquire) == before ) return;
271   }
272 }

274 // WARNING: a guest module must not do this; its __run_thread_dtors() forwards to the hosts list, so a guest calling exit() would destroy
275 // the host thread's thread_locals
276 //
277 // WARNING: these run arbitrary user dtors
278 __attribute__((always_inline)) inline void
279 __run_exit_sequence(void) noexcept
280 {
281 #if !defined(MICRON_ATTACH_MODULE)
282   micron::__run_thread_dtors();
283 #endif
284   __settle_exit_callbacks();
285   __run_fini_array();
286   __settle_exit_callbacks();
287 #if !defined(MICRON_ATTACH_MODULE)
288   if ( micron::__dynamic_tls_exit_hook ) micron::__dynamic_tls_exit_hook();
289 #endif
290 }

292 // soft exit routine, fires destructors
293 __attribute__((noreturn)) inline void
294 exit(int s = exit_ok)
295 {
296   using namespace __exit_internal;
297 
298   bool expected = false;
299   if ( !micron::atom::compare_exchange(&__exit_in_progress, &expected, true, false, micron::atomic_seq_cst, __ATOMIC_RELAXED) ) {
300     sys_exit(s);
301   }
302 
303   __run_exit_sequence();
304 
305   sys_exit(s);
306   __builtin_unreachable();
307 }

309 __attribute__((noreturn)) inline void
310 group_exit(int s = exit_ok)
311 {
312   using namespace __exit_internal;
313 
314   bool expected = false;
315   if ( !micron::atom::compare_exchange(&__exit_in_progress, &expected, true, false, micron::atomic_seq_cst, __ATOMIC_RELAXED) ) {
316     sys_group_exit(s);
317   }
318 
319   __run_exit_sequence();
320 
321   proc_exit(s);
322   __builtin_unreachable();
323 }
