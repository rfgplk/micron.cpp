115   static void
116   __on_global(void *data, wl_registry *registry, u32 name, const char *interface, u32 version) noexcept
117   {
118     auto *self = static_cast<wayland_display_t *>(data);
119     if ( micron::strcmp(interface, "wl_compositor") == 0 ) {
120       const u32 use_ver = version < 6 ? version : 6u;
121       void *p = self->__lib->wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(registry), xdg_shell_ops::wl_registry_bind,
122                                                     &__wl_compositor_interface, use_ver, 0, name, "wl_compositor", use_ver, nullptr);
123       self->__compositor = reinterpret_cast<wl_compositor *>(p);
124       self->__compositor_name = name;
125       self->__compositor_version = use_ver;
126     } else if ( micron::strcmp(interface, "xdg_wm_base") == 0 ) {
127       const u32 use_ver = version < 1 ? version : 1u;
128       void *p = self->__lib->wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(registry), xdg_shell_ops::wl_registry_bind,
129                                                     &__xdg_wm_base_interface, use_ver, 0, name, "xdg_wm_base", use_ver, nullptr);
130       self->__shell = reinterpret_cast<xdg_wm_base *>(p);
131       self->__shell_name = name;
132       self->__shell_version = use_ver;
133 
134       static xdg_wm_base_listener_t listener{
135         /*ping*/ [](void *d, xdg_wm_base *base, u32 serial) {
136           auto *me = static_cast<wayland_display_t *>(d);
137           me->__lib->wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(base), xdg_shell_ops::xdg_wm_base_pong, nullptr,
138                                             me->__shell_version, 0, serial);
139         },
140       };
141       self->__lib->wl_proxy_add_listener(reinterpret_cast<wl_proxy *>(self->__shell), reinterpret_cast<void (**)(void)>(&listener), self);
142     }
143   }

209   void
210   open(wayland_lib_t &w)
211   {
212     __lib = &w;
213     __display = w.wl_display_connect(nullptr);
214     if ( !__display )
215       throw except::network_error("wayland: wl_display_connect returned null — $WAYLAND_DISPLAY unset or socket unreachable");
216     __fd = static_cast<i32>(w.wl_display_get_fd(__display));
217 
218     __registry = w.wl_display_get_registry(__display);
219     static wl_registry_listener_t reg_listener{ &__on_global, &__on_global_remove };
220     w.wl_proxy_add_listener(reinterpret_cast<wl_proxy *>(__registry), reinterpret_cast<void (**)(void)>(&reg_listener), this);
221 
222     w.wl_display_roundtrip(__display);
223     w.wl_display_roundtrip(__display);
224 
225     if ( !__compositor ) throw except::library_error("wayland: compositor missing — wl_compositor not advertised by server");
226     if ( !__shell ) throw except::library_error("wayland: xdg_wm_base missing — server does not support xdg-shell");
227   }

229   void
230   flush() noexcept
231   {
232     if ( __lib && __display ) __lib->wl_display_flush(__display);
233   }
234 
235   void
236   roundtrip() noexcept
237   {
238     if ( __lib && __display ) __lib->wl_display_roundtrip(__display);
239   }
240 
241   void
242   dispatch_pending() noexcept
243   {
244     if ( !__lib || !__display ) return;
245     while ( __lib->wl_display_prepare_read(__display) != 0 )
246       if ( __lib->wl_display_dispatch_pending(__display) < 0 ) return;
247     __lib->wl_display_flush(__display);
248     micron::posix::pollfd p{ __fd, micron::posix::poll_in, 0 };
249     if ( micron::posix::poll(p, 1, 0) > 0 && (p.revents & posix::poll_in) )
250       __lib->wl_display_read_events(__display);
251     else
252       __lib->wl_display_cancel_read(__display);
253     __lib->wl_display_dispatch_pending(__display);
254   }
255 
256   void
257   close() noexcept
258   {
259     if ( __lib && __shell ) __lib->wl_proxy_destroy(reinterpret_cast<wl_proxy *>(__shell));
260     if ( __lib && __compositor ) __lib->wl_proxy_destroy(reinterpret_cast<wl_proxy *>(__compositor));
261     if ( __lib && __registry ) __lib->wl_proxy_destroy(reinterpret_cast<wl_proxy *>(__registry));
262     if ( __lib && __display ) __lib->wl_display_disconnect(__display);
263     __display = nullptr;
264     __registry = nullptr;
265     __compositor = nullptr;
266     __shell = nullptr;
267     __fd = -1;
268   }

293   static void
294   __on_xdg_surface_configure(void *data, xdg_surface *s, u32 serial) noexcept
295   {
296     auto *self = static_cast<wayland_window_t *>(data);
297     self->__last_serial = serial;

304     if ( self->__cb_expose ) self->__cb_expose(self->__owner);
305   }

308   __on_toplevel_configure(void *data, xdg_toplevel *, i32 w, i32 h, void *) noexcept
309   {
310     auto *self = static_cast<wayland_window_t *>(data);
311 
312     if ( w > 0 && h > 0 && (w != self->__width || h != self->__height) ) {
313       self->__width = w;
314       self->__height = h;
315       if ( self->__cb_resize ) self->__cb_resize(self->__owner, self->width(), self->height());

468   void
469   create(wayland_display_t &dpy, i32 w, i32 h, const char *title)
470   {
471     __disp = &dpy;
472     __width = w;
473     __height = h;
474     auto &lib = *dpy.lib();
475 
476     void *surf = lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(dpy.compositor()), xdg_shell_ops::wl_compositor_create_surface,
477                                             &__wl_surface_interface, dpy.compositor_version(), 0, nullptr);
478     __surface = reinterpret_cast<wl_surface *>(surf);
479     if ( !__surface ) throw except::library_error("wayland: wl_compositor.create_surface returned null");
480 
481     struct surface_listener {
482       void (*enter)(void *, wl_surface *, wl_output *);
483       void (*leave)(void *, wl_surface *, wl_output *);
484       void (*scale)(void *, wl_surface *, i32);
485       void (*transform)(void *, wl_surface *, u32);
486     };
487 
488     static const surface_listener surface_events{ +[](void *, wl_surface *, wl_output *) { }, +[](void *, wl_surface *, wl_output *) { },
489                                                   +[](void *data, wl_surface *, i32 scale) {
490                                                     if ( scale < 1 || scale > 8 ) return;
491                                                     auto &self = *static_cast<wayland_window_t *>(data);
492                                                     self.__disp->lib()->wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(self.__surface),
493                                                                                                8, nullptr,
494                                                                                                self.__disp->compositor_version(), 0, scale);
495                                                     self.set_render_scale(static_cast<f32>(scale));
496                                                   },
497                                                   +[](void *, wl_surface *, u32) { } };
498     lib.wl_proxy_add_listener(reinterpret_cast<wl_proxy *>(__surface),
499                               reinterpret_cast<void (**)(void)>(const_cast<surface_listener *>(&surface_events)), this);
500 
501     void *xsurf
502         = lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(dpy.shell()), xdg_shell_ops::xdg_wm_base_get_xdg_surface,
503                                      &__xdg_surface_interface, dpy.shell_version(), 0, nullptr, reinterpret_cast<wl_proxy *>(__surface));
504     __xsurface = reinterpret_cast<xdg_surface *>(xsurf);
505     if ( !__xsurface ) throw except::library_error("wayland: xdg_wm_base.get_xdg_surface returned null");
506 
507     static xdg_surface_listener_t xdg_surf_listener{ &__on_xdg_surface_configure };
508     lib.wl_proxy_add_listener(reinterpret_cast<wl_proxy *>(__xsurface), reinterpret_cast<void (**)(void)>(&xdg_surf_listener), this);
509 
510     void *top = lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__xsurface), xdg_shell_ops::xdg_surface_get_toplevel,
511                                            &__xdg_toplevel_interface, dpy.shell_version(), 0, nullptr);
512     __toplevel = reinterpret_cast<xdg_toplevel *>(top);
513     if ( !__toplevel ) throw except::library_error("wayland: xdg_surface.get_toplevel returned null");
514 
515     static xdg_toplevel_listener_t top_listener{
516       &__on_toplevel_configure,
517       &__on_toplevel_close,
518       &__on_toplevel_configure_bounds,
519       &__on_toplevel_wm_capabilities,
520     };
521     lib.wl_proxy_add_listener(reinterpret_cast<wl_proxy *>(__toplevel), reinterpret_cast<void (**)(void)>(&top_listener), this);
522 
523     if ( title ) {
524       lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__toplevel), xdg_shell_ops::xdg_toplevel_set_title, nullptr,
525                                  dpy.shell_version(), 0, title);
526     }
527 
528     lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__surface), xdg_shell_ops::wl_surface_commit, nullptr, dpy.compositor_version(),
529                                0);
530     lib.wl_display_roundtrip(dpy.display());
531   }

554   void
555   destroy() noexcept
556   {
557     if ( !__disp ) return;
558     auto &lib = *__disp->lib();
559     if ( __toplevel ) {
560       lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__toplevel), xdg_shell_ops::xdg_toplevel_destroy, nullptr,
561                                  __disp->shell_version(), 1 /*WL_MARSHAL_FLAG_DESTROY*/);
562       __toplevel = nullptr;
563     }
564     if ( __xsurface ) {
565       lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__xsurface), xdg_shell_ops::xdg_surface_destroy, nullptr,
566                                  __disp->shell_version(), 1);
567       __xsurface = nullptr;
568     }
569     if ( __surface ) {
570       lib.wl_proxy_marshal_flags(reinterpret_cast<wl_proxy *>(__surface), xdg_shell_ops::wl_surface_destroy, nullptr,
571                                  __disp->compositor_version(), 1);
572       __surface = nullptr;
573     }
