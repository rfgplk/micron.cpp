32 inline constexpr u32 __scope_count = 5;

34 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
35 // block header
36 //   user - 8    offset_to_raw   what abc::dealloc has to be given
37 //   user - 16   size            reqd size; abc::query_size only knows the granted one
38 //   user - 24   scope
39 //   user - 32   magic           flags a free of memory this allocator never handed out
40 
41 inline constexpr usize __off_raw = 8;
42 inline constexpr usize __off_size = 16;
43 inline constexpr usize __off_scope = 24;
44 inline constexpr usize __off_magic = 32;
45 inline constexpr usize __header_size = 32;
46 
47 inline constexpr usize __min_align = 16;
48 
49 inline constexpr usize __magic = static_cast<usize>(0x564b484cul);      // 'VKHL'
50 
51 // __alloc rounds with ~(align_eff - 1), and align_eff is floored by __min_align
52 static_assert(__min_align != 0 and (__min_align & (__min_align - 1)) == 0, "__min_align feeds an alignment mask; power of two only");
53 static_assert(__header_size >= __off_magic, "every header field must fit below the user pointer");
54 static_assert(__off_magic >= __off_scope + sizeof(usize) and __off_scope >= __off_size + sizeof(usize)
55                   and __off_size >= __off_raw + sizeof(usize) and __off_raw >= sizeof(usize),
56               "header fields overlap");
57 static_assert(__magic != 0, "the magic must survive truncation to a 32-bit usize");

79 inline void
80 __stat_on_alloc([[maybe_unused]] usize size, [[maybe_unused]] VkSystemAllocationScope scope) noexcept
81 {
82 #if defined(MICRON_VK_HOST_STATS)
83   __stat.allocs.fetch_add(1, micron::memory_order_relaxed);
84   __stat.live_blocks.fetch_add(1, micron::memory_order_relaxed);
85   const u64 now = __stat.live_bytes.add_fetch(static_cast<u64>(size), micron::memory_order_relaxed);
86 
87   u64 peak = __stat.peak_bytes.get(micron::memory_order_relaxed);
88   while ( now > peak and !__stat.peak_bytes.compare_exchange_weak(peak, now, micron::memory_order_relaxed, micron::memory_order_relaxed) );
89 
90   const u32 i = static_cast<u32>(static_cast<i32>(scope));
91   if ( i < __scope_count ) [[likely]] {
92     __stat.scope_blocks[i].fetch_add(1, micron::memory_order_relaxed);
93     __stat.scope_bytes[i].fetch_add(static_cast<u64>(size), micron::memory_order_relaxed);
94   }
95 #endif
96 }

98 // reads the header, so it must run before the block goes back to abc
99 inline void
100 __stat_on_free([[maybe_unused]] byte *user) noexcept
101 {
102 #if defined(MICRON_VK_HOST_STATS)
103   if ( *reinterpret_cast<usize *>(user - __off_magic) != __magic ) [[unlikely]] {
104     __stat.foreign_frees.fetch_add(1, micron::memory_order_relaxed);
105     return;
106   }
107   const u64 size = static_cast<u64>(*reinterpret_cast<usize *>(user - __off_size));
108   const u32 i = static_cast<u32>(*reinterpret_cast<usize *>(user - __off_scope));
109   *reinterpret_cast<usize *>(user - __off_magic) = 0;
110 
111   __stat.frees.fetch_add(1, micron::memory_order_relaxed);
112   __stat.live_blocks.fetch_sub(1, micron::memory_order_relaxed);
113   __stat.live_bytes.fetch_sub(size, micron::memory_order_relaxed);
114   if ( i < __scope_count ) [[likely]] {
115     __stat.scope_blocks[i].fetch_sub(1, micron::memory_order_relaxed);
116     __stat.scope_bytes[i].fetch_sub(size, micron::memory_order_relaxed);
117   }
118 #endif
119 }

121 inline void *
122 __alloc(usize size, usize align, VkSystemAllocationScope scope) noexcept
123 {
124   if ( size == 0 ) return nullptr;
125   // the mask below is only a rounding op for a power of two, and Vulkan promises one
126   if ( align == 0 or (align & (align - 1)) != 0 ) [[unlikely]]
127     return nullptr;
128 
129   const usize align_eff = align > __min_align ? align : __min_align;
130   usize total;
131   if ( abc::check_add_overflow(size, align_eff + __header_size, total) ) [[unlikely]]
132     return nullptr;
133 
134   byte *raw = abc::alloc(total);
135   if ( !raw ) return nullptr;
136   const uintptr_t raw_addr = reinterpret_cast<uintptr_t>(raw);
137   const uintptr_t user_addr = (raw_addr + __header_size + align_eff - 1) & ~(static_cast<uintptr_t>(align_eff) - 1);
138   byte *user = reinterpret_cast<byte *>(user_addr);
139 
140   *micron::ptr_cast<usize *>(user - __off_raw) = user_addr - raw_addr;
141   *micron::ptr_cast<usize *>(user - __off_size) = size;
142   *micron::ptr_cast<usize *>(user - __off_scope) = static_cast<usize>(static_cast<u32>(static_cast<i32>(scope)));
143   *micron::ptr_cast<usize *>(user - __off_magic) = __magic;
144 
145   __stat_on_alloc(size, scope);
146   return user;
147 }

149 inline void
150 __free(void *user_ptr) noexcept
151 {
152   if ( !user_ptr ) return;
153   byte *user = reinterpret_cast<byte *>(user_ptr);
154   __stat_on_free(user);
155   const usize offset_to_raw = *micron::ptr_cast<usize *>(user - __off_raw);
156   abc::dealloc(user - offset_to_raw);
157 }

159 inline usize
160 __size_of(void *user_ptr) noexcept
161 {
162   return *micron::ptr_cast<usize *>(reinterpret_cast<byte *>(user_ptr) - __off_size);
163 }

172 inline void *
173 __realloc(void *original, usize size, usize align, VkSystemAllocationScope scope) noexcept
174 {
175   if ( !original ) return __alloc(size, align, scope);
176   if ( size == 0 ) {
177     __free(original);
178     return nullptr;
179   }
180   const usize old_size = __size_of(original);
181   void *fresh = __alloc(size, align, scope);
182   if ( !fresh ) return nullptr;
183   const usize copy = old_size < size ? old_size : size;
184   micron::memcpy(reinterpret_cast<byte *>(fresh), reinterpret_cast<byte *>(original), copy);
185   __free(original);
186 #if defined(MICRON_VK_HOST_STATS)
187   __stat.reallocs.fetch_add(1, micron::memory_order_relaxed);
188 #endif
189   return fresh;
190 }

255 [[nodiscard]] inline host_alloc_stats_t
256 host_alloc_stats() noexcept
257 {
258   host_alloc_stats_t o{};
259 #if defined(MICRON_VK_HOST_STATS)
260   o.enabled = true;
261   o.live_blocks = __vk_host::__stat.live_blocks.get(micron::memory_order_relaxed);
262   o.live_bytes = __vk_host::__stat.live_bytes.get(micron::memory_order_relaxed);
263   o.peak_bytes = __vk_host::__stat.peak_bytes.get(micron::memory_order_relaxed);
264   o.allocs = __vk_host::__stat.allocs.get(micron::memory_order_relaxed);
265   o.frees = __vk_host::__stat.frees.get(micron::memory_order_relaxed);
266   o.reallocs = __vk_host::__stat.reallocs.get(micron::memory_order_relaxed);
267   o.foreign_frees = __vk_host::__stat.foreign_frees.get(micron::memory_order_relaxed);
268   for ( u32 i = 0; i < __vk_host::__scope_count; ++i ) {
269     o.scope_blocks[i] = __vk_host::__stat.scope_blocks[i].get(micron::memory_order_relaxed);
270     o.scope_bytes[i] = __vk_host::__stat.scope_bytes[i].get(micron::memory_order_relaxed);
271   }
272   o.internal_allocs = __vk_host::__stat.internal_allocs.get(micron::memory_order_relaxed);
273   o.internal_alloc_bytes = __vk_host::__stat.internal_alloc_bytes.get(micron::memory_order_relaxed);
274   o.internal_frees = __vk_host::__stat.internal_frees.get(micron::memory_order_relaxed);
275   o.internal_free_bytes = __vk_host::__stat.internal_free_bytes.get(micron::memory_order_relaxed);
276 #endif
277   return o;
