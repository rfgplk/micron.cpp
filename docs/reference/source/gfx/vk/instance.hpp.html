39   static u32
40   __append(const char **buf, u32 i, u32 cap, const char *name) noexcept
41   {
42     if ( !name || i >= cap ) return i;
43 
44     for ( u32 k = 0; k < i; ++k ) {
45       if ( buf[k] == name || micron::strcmp(buf[k], name) == 0 ) return i;
46     }
47     buf[i] = name;
48     return i + 1;
49   }
50 
51   static u32
52   __build_layers(const instance_hints &h, const char **buf, u32 cap) noexcept
53   {
54     u32 n = 0;
55     if ( h.debug && has_instance_layer(VK_LAYER_KHRONOS_VALIDATION_NAME) ) n = __append(buf, n, cap, VK_LAYER_KHRONOS_VALIDATION_NAME);
56     for ( usize i = 0; i < h.extra_layers_count && n < cap; ++i ) n = __append(buf, n, cap, h.extra_layers[i]);
57     return n;
58   }
59 
60   static u32
61   __build_extensions(const instance_hints &h, const char **buf, u32 cap) noexcept
62   {
63     u32 n = 0;
64     if ( h.surface ) {
65       n = __append(buf, n, cap, VK_KHR_SURFACE_EXTENSION_NAME);
66 
67       if ( has_instance_extension(VK_KHR_XLIB_SURFACE_EXTENSION_NAME) ) n = __append(buf, n, cap, VK_KHR_XLIB_SURFACE_EXTENSION_NAME);
68       if ( has_instance_extension(VK_KHR_WAYLAND_SURFACE_EXTENSION_NAME) ) n = __append(buf, n, cap, VK_KHR_WAYLAND_SURFACE_EXTENSION_NAME);
69     }
70     if ( h.debug ) n = __append(buf, n, cap, VK_EXT_DEBUG_UTILS_EXTENSION_NAME);
71     for ( usize i = 0; i < h.extra_extensions_count && n < cap; ++i ) n = __append(buf, n, cap, h.extra_extensions[i]);
72     return n;
73   }

76   ~instance() { reset(); }

80   explicit instance(const instance_hints &h)
81   {
82 
83     (void)vk_lib();
84     __load_global();
85     if ( !vkCreateInstance ) throw except::library_error("vk::instance: vkCreateInstance unresolved — wrong libvulkan?");
86 
87     const char *layers[max_enabled_layers];
88     const char *exts[max_enabled_extensions];
89     const u32 num_layers = __build_layers(h, layers, max_enabled_layers);
90     const u32 num_exts = __build_extensions(h, exts, max_enabled_extensions);
91 
92     VkApplicationInfo app{};
93     app.sType = structure_type_of_v<VkApplicationInfo>;
94     app.pApplicationName = h.app_name;
95     app.applicationVersion = h.app_version;
96     app.pEngineName = h.engine_name;
97     app.engineVersion = h.engine_version;
98     app.apiVersion = h.api_version;
99 
100     VkDebugUtilsMessengerCreateInfoEXT debug_ci = default_messenger_info();
101 
102     VkInstanceCreateInfo ci{};
103     ci.sType = structure_type_of_v<VkInstanceCreateInfo>;
104     ci.pApplicationInfo = &app;
105     ci.enabledLayerCount = num_layers;
106     ci.ppEnabledLayerNames = num_layers ? layers : nullptr;
107     ci.enabledExtensionCount = num_exts;
108     ci.ppEnabledExtensionNames = num_exts ? exts : nullptr;
109     if ( h.debug ) ci.pNext = &debug_ci;
110 
111     check_vk(vkCreateInstance(&ci, host_allocation_callbacks(), &__h), "vkCreateInstance");
112 
113     __load_instance_1_3(__h);
114 
115     __debug_enabled = h.debug;
116     if ( h.debug && vkCreateDebugUtilsMessengerEXT ) {
117 
118       __dbg = messenger(__h);
119     }
120   }

186   struct physical_device_list {
187     physical_device devices[max_physical_devices]{};
188     u32 count = 0;
189 
190     physical_device *
191     begin() noexcept
192     {
193       return devices;
194     }
195 
196     physical_device *
197     end() noexcept
198     {
199       return devices + count;
200     }
201 
202     const physical_device *
203     begin() const noexcept
204     {
205       return devices;
206     }
207 
208     const physical_device *
209     end() const noexcept
210     {
211       return devices + count;
212     }
213 
214     physical_device &
215     operator[](u32 i) noexcept
216     {
217       return devices[i];
218     }

220     const physical_device &
221     operator[](u32 i) const noexcept
222     {
223       return devices[i];
224     }

227   physical_device_list
228   enumerate_physical_devices() const
229   {
230     if ( !__h || !vkEnumeratePhysicalDevices )
231       throw except::library_error("vk::instance::enumerate_physical_devices: vkEnumeratePhysicalDevices not loaded");
232     u32 n = 0;
233     check_vk(vkEnumeratePhysicalDevices(__h, &n, nullptr), "vkEnumeratePhysicalDevices");
234     if ( n > max_physical_devices ) n = max_physical_devices;
235     VkPhysicalDevice raw[max_physical_devices]{};
236     check_vk(vkEnumeratePhysicalDevices(__h, &n, raw), "vkEnumeratePhysicalDevices");
237     physical_device_list out;
238     out.count = n;
239     for ( u32 i = 0; i < n; ++i ) out.devices[i] = physical_device(raw[i]);
240     return out;
241   }
242 
243   physical_device
244   pick_physical_device(bool prefer_discrete = true) const
245   {
246     auto list = enumerate_physical_devices();
247     if ( list.count == 0 ) return {};
248     if ( prefer_discrete ) {
249       for ( u32 i = 0; i < list.count; ++i ) {
250         if ( list.devices[i].is_discrete() ) return list.devices[i];
251       }
252     }
253     return list.devices[0];
254   }
