23 class query_pool
24 {
25   VkDevice __dev = nullptr;
26   VkQueryPool __h = nullptr;
27   VkQueryType __type = static_cast<VkQueryType>(0);
28   u32 __count = 0;
29 
30 public:
31   ~query_pool() { reset(); }
32 
33   query_pool() = default;
34 
35   query_pool(device &dev, VkQueryType type, u32 count, VkQueryPipelineStatisticFlags pipeline_statistics = {})
36       : __dev(dev.handle()), __type(type), __count(count)
37   {
38     if ( !__dev || !vkCreateQueryPool ) throw except::library_error("vk::query_pool: vkCreateQueryPool unresolved");
39     VkQueryPoolCreateInfo ci{};
40     ci.sType = structure_type_of_v<VkQueryPoolCreateInfo>;
41     ci.queryType = type;
42     ci.queryCount = count;
43     ci.pipelineStatistics = pipeline_statistics;
44     check_vk(vkCreateQueryPool(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateQueryPool");
45   }
46 
47   query_pool(const query_pool &) = delete;
48 
49   query_pool(query_pool &&o) noexcept : __dev(o.__dev), __h(o.__h), __type(o.__type), __count(o.__count)
50   {
51     o.__dev = nullptr;
52     o.__h = nullptr;
53     o.__count = 0;
54   }
55 
56   query_pool &operator=(const query_pool &) = delete;
57 
58   query_pool &
59   operator=(query_pool &&o) noexcept
60   {
61     if ( this != &o ) {
62       reset();
63       __dev = o.__dev;
64       __h = o.__h;
65       __type = o.__type;
66       __count = o.__count;
67       o.__dev = nullptr;
68       o.__h = nullptr;
69       o.__count = 0;
70     }
71     return *this;
72   }
73 
74   VkQueryPool
75   handle() const noexcept
76   {
77     return __h;
78   }
79 
80   VkQueryType
81   type() const noexcept
82   {
83     return __type;
84   }
85 
86   u32
87   count() const noexcept
88   {
89     return __count;
90   }
91 
92   bool
93   valid() const noexcept
94   {
95     return __h != nullptr;
96   }
97 
98   explicit
99   operator bool() const noexcept
100   {
101     return valid();
102   }
103 
104   void
105   reset_range(u32 first, u32 count)
106   {
107     if ( !__h || !__dev || !vkResetQueryPool )
108       throw except::library_error("vk::query_pool::reset_range: vkResetQueryPool unresolved (needs 1.2 host_query_reset)");
109     vkResetQueryPool(__dev, __h, first, count);
110   }
111 
112   VkResult
113   get_results(u32 first, u32 count, void *out_buf, usize buf_size_bytes, VkDeviceSize stride, VkQueryResultFlags flags = {}) noexcept
114   {
115     if ( !__h || !__dev || !vkGetQueryPoolResults ) return VK_ERROR_INITIALIZATION_FAILED;
116     return vkGetQueryPoolResults(__dev, __h, first, count, buf_size_bytes, out_buf, stride, flags);
117   }
118 
119   void
120   reset() noexcept
121   {
122     if ( __h && __dev && vkDestroyQueryPool ) vkDestroyQueryPool(__dev, __h, host_allocation_callbacks());
123     __h = nullptr;
124     __dev = nullptr;
125     __count = 0;
126   }
127 };
