88 inline void
89 check_vk(VkResult r, const char *where)
90 {
91   if ( r == VK_SUCCESS ) return;
92   (void)where;
93   throw except::library_error(result_name(r));
94 }
95 
96 inline VkResult
97 check_vk_ok(VkResult r, const char *where)
98 {
99   switch ( r ) {
100   case VK_SUCCESS:
101   case VK_NOT_READY:
102   case VK_TIMEOUT:
103   case VK_EVENT_SET:
104   case VK_EVENT_RESET:
105   case VK_INCOMPLETE:
106   case VK_SUBOPTIMAL_KHR:
107     return r;
108   default:
109     (void)where;
110     throw except::library_error(result_name(r));
111   }
112 }
