21 inline constexpr u32 max_queue_families = 16;
22 
23 class physical_device
24 {
25   VkPhysicalDevice __h = nullptr;
26   VkPhysicalDeviceProperties __props{};
27   VkPhysicalDeviceFeatures __features{};
28   VkPhysicalDeviceMemoryProperties __mem_props{};
29   VkQueueFamilyProperties __qf_props[max_queue_families]{};
30   u32 __qf_count = 0;
31 
32   void
33   __refresh() noexcept
34   {
35     if ( !__h ) return;
36     if ( vkGetPhysicalDeviceProperties ) vkGetPhysicalDeviceProperties(__h, &__props);
37     if ( vkGetPhysicalDeviceFeatures ) vkGetPhysicalDeviceFeatures(__h, &__features);
38     if ( vkGetPhysicalDeviceMemoryProperties ) vkGetPhysicalDeviceMemoryProperties(__h, &__mem_props);
39     if ( vkGetPhysicalDeviceQueueFamilyProperties ) {
40       u32 n = max_queue_families;
41       vkGetPhysicalDeviceQueueFamilyProperties(__h, &n, __qf_props);
42       __qf_count = n > max_queue_families ? max_queue_families : n;
43     }
44   }
45 
46 public:
47   physical_device() = default;
48 
49   explicit physical_device(VkPhysicalDevice h) noexcept : __h(h) { __refresh(); }
50 
51   VkPhysicalDevice
52   handle() const noexcept
53   {
54     return __h;
55   }
56 
57   bool
58   valid() const noexcept
59   {
60     return __h != nullptr;
61   }
62 
63   explicit
64   operator bool() const noexcept
65   {
66     return valid();
67   }
68 
69   const VkPhysicalDeviceProperties &
70   properties() const noexcept
71   {
72     return __props;
73   }
74 
75   const VkPhysicalDeviceFeatures &
76   features() const noexcept
77   {
78     return __features;
79   }
80 
81   const VkPhysicalDeviceMemoryProperties &
82   memory_properties() const noexcept
83   {
84     return __mem_props;
85   }
86 
87   const VkQueueFamilyProperties *
88   queue_family_properties() const noexcept
89   {
90     return __qf_props;
91   }
92 
93   u32
94   queue_family_count() const noexcept
95   {
96     return __qf_count;
97   }
98 
99   u32
100   select_queue_family(VkQueueFlags required, VkSurfaceKHR surf = VK_NULL_HANDLE) const noexcept
101   {
102     for ( u32 i = 0; i < __qf_count; ++i ) {
103       if ( (__qf_props[i].queueFlags & required) != required ) continue;
104       if ( surf == VK_NULL_HANDLE ) return i;
105       if ( !vkGetPhysicalDeviceSurfaceSupportKHR ) continue;
106       VkBool32 ok = 0;
107       if ( vkGetPhysicalDeviceSurfaceSupportKHR(__h, i, surf, &ok) != VK_SUCCESS ) continue;
108       if ( ok ) return i;
109     }
110     return ~0u;
111   }
112 
113   bool
114   supports_extension(const char *name) const noexcept
115   {
116     if ( !__h || !name || !vkEnumerateDeviceExtensionProperties ) return false;
117     u32 n = 0;
118     if ( vkEnumerateDeviceExtensionProperties(__h, nullptr, &n, nullptr) != VK_SUCCESS ) return false;
119     if ( n == 0 ) return false;
120     constexpr u32 cap = 512;
121     if ( n > cap ) n = cap;
122     VkExtensionProperties props[cap]{};
123     if ( vkEnumerateDeviceExtensionProperties(__h, nullptr, &n, props) != VK_SUCCESS ) return false;
124     for ( u32 i = 0; i < n; ++i ) {
125       if ( micron::strcmp(props[i].extensionName, name) == 0 ) return true;
126     }
127     return false;
128   }
129 
130   bool
131   is_discrete() const noexcept
132   {
133     return __props.deviceType == static_cast<VkPhysicalDeviceType>(VK_PHYSICAL_DEVICE_TYPE_DISCRETE_GPU);
134   }
135 };
