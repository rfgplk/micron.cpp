31 // own libvulkan.so.1; cached
32 // NOTE: libvulkan must be linked against this at comptime, otherwise errors out
33 struct vk_lib_t {
34   host_dso host;
35   PFN_vkGetInstanceProcAddr vkGetInstanceProcAddr = nullptr;
36 
37   vk_lib_t() : host("libvulkan.so.1")
38   {
39     vkGetInstanceProcAddr = host.sym_as<PFN_vkGetInstanceProcAddr>("vkGetInstanceProcAddr");
40     if ( !vkGetInstanceProcAddr ) {
41       throw except::library_error("vulkan: vkGetInstanceProcAddr missing in libvulkan.so.1");
42     }
43   }
44 };

55 // 0 global
56 inline void
57 __load_global() noexcept
58 {
59   auto get = vk_lib().vkGetInstanceProcAddr;
60   for ( usize i = 0; i < vk_table_size; ++i ) {
61     if ( vk_table[i].dispatch_kind != 0 ) continue;
62     *__vk_fnptr_slots[i] = reinterpret_cast<void *>(get(VK_NULL_HANDLE, vk_table[i].name));
63   }
64 }

66 // 1 instance
67 inline void
68 __load_instance_1_3(VkInstance instance) noexcept
69 {
70   __current_instance_for_loader = instance;
71   auto get = vk_lib().vkGetInstanceProcAddr;
72   for ( usize i = 0; i < vk_table_size; ++i ) {
73     if ( vk_table[i].dispatch_kind != 1 ) continue;
74     *__vk_fnptr_slots[i] = reinterpret_cast<void *>(get(instance, vk_table[i].name));
75   }
76   if ( !vkGetDeviceProcAddr ) {
77     vkGetDeviceProcAddr = reinterpret_cast<PFN_vkGetDeviceProcAddr>(get(instance, "vkGetDeviceProcAddr"));
78   }
79 }

82 inline void
83 __load_device_1_3(VkDevice device) noexcept
84 {
85   if ( !vkGetDeviceProcAddr ) return;      // should be populated by __load_instance_1_3

88     *__vk_fnptr_slots[i] = reinterpret_cast<void *>(vkGetDeviceProcAddr(device, vk_table[i].name));
89   }

92 // detect at rt whether the loader supports a named instance extension
93 inline bool
94 has_instance_extension(const char *name) noexcept
95 {
96   if ( !name || !vkEnumerateInstanceExtensionProperties ) return false;
97   u32 n = 0;
98   if ( vkEnumerateInstanceExtensionProperties(nullptr, &n, nullptr) != VK_SUCCESS ) return false;
99   if ( n == 0 ) return false;
100   constexpr u32 cap = 256;
101   if ( n > cap ) n = cap;
102   VkExtensionProperties props[cap]{};
103   const VkResult r = vkEnumerateInstanceExtensionProperties(nullptr, &n, props);
104   if ( r != VK_SUCCESS && r != VK_INCOMPLETE ) return false;
105   for ( u32 i = 0; i < n; ++i ) {
106     if ( micron::strcmp(props[i].extensionName, name) == 0 ) return true;
107   }
108   return false;
109 }

129 inline bool
130 has_device_extension(VkPhysicalDevice pd, const char *name) noexcept
131 {
132   if ( !pd || !name || !vkEnumerateDeviceExtensionProperties ) return false;
133   u32 n = 0;
134   if ( vkEnumerateDeviceExtensionProperties(pd, nullptr, &n, nullptr) != VK_SUCCESS ) return false;
135   if ( n == 0 ) return false;
136   constexpr u32 cap = 512;
137   if ( n > cap ) n = cap;
138   VkExtensionProperties props[cap]{};
139   if ( vkEnumerateDeviceExtensionProperties(pd, nullptr, &n, props) != VK_SUCCESS ) return false;
140   for ( u32 i = 0; i < n; ++i ) {
141     if ( micron::strcmp(props[i].extensionName, name) == 0 ) return true;
142   }
143   return false;
144 }

146 template<u16... Indices> class subset
147 {
148   void *__fns[sizeof...(Indices)] = {};
149 
150 public:
151   static constexpr usize count = sizeof...(Indices);
152   static constexpr u16 indices[count] = { Indices... };
153 
154   void
155   load_global() noexcept
156   {
157     auto get = vk_lib().vkGetInstanceProcAddr;
158     usize k = 0;
159     ((__fns[k++] = reinterpret_cast<void *>(get(VK_NULL_HANDLE, vk_table[Indices].name))), ...);
160   }
161 
162   void
163   load_instance(VkInstance instance) noexcept
164   {
165     auto get = vk_lib().vkGetInstanceProcAddr;
166     usize k = 0;
167     ((__fns[k++] = reinterpret_cast<void *>(get(instance, vk_table[Indices].name))), ...);
168   }
169 
170   void
171   load_device(VkDevice device) noexcept
172   {
173     if ( !vkGetDeviceProcAddr ) return;
174     usize k = 0;
175     ((__fns[k++] = reinterpret_cast<void *>(vkGetDeviceProcAddr(device, vk_table[Indices].name))), ...);
176   }
177 
178   template<u16 Idx>
179   static consteval usize
180   position_of() noexcept
181   {
182     static_assert(((Idx == Indices) || ...), "subset does not contain that index");
183     usize pos = 0;
184     bool found = false;
185     auto check = [&](u16 idx) {
186       if ( !found ) {
187         if ( idx == Idx )
188           found = true;
189         else
190           ++pos;
191       }
192     };
193     ((check(Indices)), ...);
194     return pos;
195   }
196 
197   template<u16 Idx>
198   void *
199   raw() const noexcept
200   {
201     return __fns[position_of<Idx>()];
202   }
203 
204   template<u16 Idx, class Fn>
205   Fn
206   as() const noexcept
207   {
208     return reinterpret_cast<Fn>(raw<Idx>());
209   }
210 };
