46   static u32
47   __add_unique(u32 *out, u32 n, u32 family) noexcept
48   {
49     if ( family == ~0u ) return n;
50     for ( u32 i = 0; i < n; ++i ) {
51       if ( out[i] == family ) return n;
52     }
53     out[n] = family;
54     return n + 1;
55   }

58   ~device() { reset(); }

62   device(const physical_device &pd, const device_hints &h, const surface *surf = nullptr) : __pd(pd)
63   {
64     if ( !pd.valid() ) throw except::logic_error("vk::device: invalid physical_device");
65     if ( !vkCreateDevice ) throw except::library_error("vk::device: vkCreateDevice unresolved — was vk::instance constructed first?");
66 
67     VkQueueFlags gfx_req{}, cmp_req{}, xfer_req{};
68     if ( h.graphics ) gfx_req |= VK_QUEUE_GRAPHICS_BIT;
69     if ( h.compute ) cmp_req |= VK_QUEUE_COMPUTE_BIT;
70     if ( h.transfer ) xfer_req |= VK_QUEUE_TRANSFER_BIT;
71 
72     if ( h.graphics ) __graphics_family = pd.select_queue_family(gfx_req);
73     if ( h.compute ) __compute_family = pd.select_queue_family(cmp_req);
74     if ( h.transfer ) __transfer_family = pd.select_queue_family(xfer_req);
75     if ( h.present && surf && surf->valid() ) __present_family = pd.select_queue_family(VkQueueFlags{}, surf->handle());
76 
77     u32 unique_families[max_unique_queue_families];
78     u32 unique_n = 0;
79     unique_n = __add_unique(unique_families, unique_n, __graphics_family);
80     unique_n = __add_unique(unique_families, unique_n, __compute_family);
81     unique_n = __add_unique(unique_families, unique_n, __transfer_family);
82     unique_n = __add_unique(unique_families, unique_n, __present_family);
83     if ( unique_n == 0 ) throw except::logic_error("vk::device: no requested capability could be satisfied");
84 
85     VkDeviceQueueCreateInfo queue_cis[max_unique_queue_families]{};
86     const float priority = 1.0f;
87     for ( u32 i = 0; i < unique_n; ++i ) {
88       queue_cis[i].sType = structure_type_of_v<VkDeviceQueueCreateInfo>;
89       queue_cis[i].queueFamilyIndex = unique_families[i];
90       queue_cis[i].queueCount = 1;
91       queue_cis[i].pQueuePriorities = &priority;
92     }
93 
94     const char *exts[max_device_extensions];
95     u32 num_exts = 0;
96     if ( h.present && surf && surf->valid() ) {
97       exts[num_exts++] = VK_KHR_SWAPCHAIN_EXTENSION_NAME;
98     }
99     for ( usize i = 0; i < h.extra_extensions_count && num_exts < max_device_extensions; ++i ) {
100       exts[num_exts++] = h.extra_extensions[i];
101     }
102 
103     VkPhysicalDeviceVulkan13Features vk13 = h.required_vk13_features;
104     vk13.sType = structure_type_of_v<VkPhysicalDeviceVulkan13Features>;
105     vk13.pNext = nullptr;
106 
107     VkDeviceCreateInfo ci{};
108     ci.sType = structure_type_of_v<VkDeviceCreateInfo>;
109     ci.queueCreateInfoCount = unique_n;
110     ci.pQueueCreateInfos = queue_cis;
111     ci.enabledExtensionCount = num_exts;
112     ci.ppEnabledExtensionNames = num_exts ? exts : nullptr;
113     ci.pEnabledFeatures = &h.required_features;
114     ci.pNext = &vk13;
115 
116     check_vk(vkCreateDevice(pd.handle(), &ci, host_allocation_callbacks(), &__h), "vkCreateDevice");
117 
118     __load_device_1_3(__h);
119 
120     auto pull = [&](u32 family, queue &out) {
121       if ( family == ~0u ) return;
122       VkQueue q = nullptr;
123       if ( vkGetDeviceQueue ) vkGetDeviceQueue(__h, family, 0, &q);
124       out = queue(q, family, 0);
125     };
126     pull(__graphics_family, __graphics);
127     pull(__compute_family, __compute);
128     pull(__transfer_family, __transfer);
129     pull(__present_family, __present);
130   }

162   }
