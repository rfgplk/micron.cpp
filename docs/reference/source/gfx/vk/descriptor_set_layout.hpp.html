23 class descriptor_set_layout
24 {

26   VkDescriptorSetLayout __h = nullptr;

28 public:

30 
31   descriptor_set_layout() = default;
32 
33   descriptor_set_layout(device &dev, const VkDescriptorSetLayoutCreateInfo &ci) : __dev(dev.handle())
34   {
35     if ( !__dev || !vkCreateDescriptorSetLayout )
36       throw except::library_error("vk::descriptor_set_layout: vkCreateDescriptorSetLayout unresolved");
37     check_vk(vkCreateDescriptorSetLayout(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateDescriptorSetLayout");
38   }
39 
40   descriptor_set_layout(device &dev, const VkDescriptorSetLayoutBinding *bindings, u32 binding_count,
41                         VkDescriptorSetLayoutCreateFlags flags = {})
42       : __dev(dev.handle())
43   {
44     if ( !__dev || !vkCreateDescriptorSetLayout )
45       throw except::library_error("vk::descriptor_set_layout: vkCreateDescriptorSetLayout unresolved");
46     VkDescriptorSetLayoutCreateInfo ci{};
47     ci.sType = structure_type_of_v<VkDescriptorSetLayoutCreateInfo>;
48     ci.flags = flags;
49     ci.bindingCount = binding_count;
50     ci.pBindings = bindings;
51     check_vk(vkCreateDescriptorSetLayout(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateDescriptorSetLayout");
52   }
53 
54   descriptor_set_layout(const descriptor_set_layout &) = delete;
55 
56   descriptor_set_layout(descriptor_set_layout &&o) noexcept : __dev(o.__dev), __h(o.__h)
57   {
58     o.__dev = nullptr;
59     o.__h = nullptr;
60   }
61 
62   descriptor_set_layout &operator=(const descriptor_set_layout &) = delete;
63 
64   descriptor_set_layout &
65   operator=(descriptor_set_layout &&o) noexcept
66   {
67     if ( this != &o ) {
68       reset();
69       __dev = o.__dev;
70       __h = o.__h;
71       o.__dev = nullptr;
72       o.__h = nullptr;
73     }
74     return *this;
75   }

77   VkDescriptorSetLayout
78   handle() const noexcept
79   {
80     return __h;
81   }
82 
83   bool
84   valid() const noexcept
85   {
86     return __h != nullptr;
87   }
88 
89   explicit
90   operator bool() const noexcept
91   {
92     return valid();
93   }
94 
95   void
96   reset() noexcept
97   {
98     if ( __h && __dev && vkDestroyDescriptorSetLayout ) vkDestroyDescriptorSetLayout(__dev, __h, host_allocation_callbacks());
99     __h = nullptr;
100     __dev = nullptr;
101   }
