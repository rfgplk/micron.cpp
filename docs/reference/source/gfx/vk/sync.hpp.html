28 class fence
29 {
30   VkDevice __dev = nullptr;
31   VkFence __h = nullptr;
32 
33 public:
34   ~fence() { reset(); }
35 
36   fence() = default;
37 
38   explicit fence(device &dev, bool start_signaled = false) : __dev(dev.handle())
39   {
40     if ( !__dev || !vkCreateFence ) throw except::library_error("vk::fence: vkCreateFence unresolved");
41     VkFenceCreateInfo ci{};
42     ci.sType = structure_type_of_v<VkFenceCreateInfo>;
43     ci.flags = start_signaled ? VkFenceCreateFlags{ VK_FENCE_CREATE_SIGNALED_BIT } : VkFenceCreateFlags{};
44     check_vk(vkCreateFence(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateFence");
45   }
46 
47   fence(const fence &) = delete;
48 
49   fence(fence &&o) noexcept : __dev(o.__dev), __h(o.__h)
50   {
51     o.__dev = nullptr;
52     o.__h = nullptr;
53   }
54 
55   fence &operator=(const fence &) = delete;
56 
57   fence &
58   operator=(fence &&o) noexcept
59   {
60     if ( this != &o ) {
61       reset();
62       __dev = o.__dev;
63       __h = o.__h;
64       o.__dev = nullptr;
65       o.__h = nullptr;
66     }
67     return *this;
68   }
69 
70   VkFence
71   handle() const noexcept
72   {
73     return __h;
74   }
75 
76   bool
77   valid() const noexcept
78   {
79     return __h != nullptr;
80   }
81 
82   explicit
83   operator bool() const noexcept
84   {
85     return valid();
86   }
87 
88   bool
89   wait(u64 timeout_ns = ~0ull)
90   {
91     if ( !__h || !__dev || !vkWaitForFences ) return false;
92     VkResult r = vkWaitForFences(__dev, 1, &__h, VK_TRUE, timeout_ns);
93     if ( r == VK_SUCCESS ) return true;
94     if ( r == VK_TIMEOUT ) return false;
95     check_vk(r, "vkWaitForFences");
96     return false;
97   }
98 
99   void
100   reset_state()
101   {
102     if ( !__h || !__dev || !vkResetFences ) return;
103     check_vk(vkResetFences(__dev, 1, &__h), "vkResetFences");
104   }
105 
106   bool
107   is_signaled() const noexcept
108   {
109     if ( !__h || !__dev || !vkGetFenceStatus ) return false;
110     return vkGetFenceStatus(__dev, __h) == VK_SUCCESS;
111   }
112 
113   void
114   reset() noexcept
115   {
116     if ( __h && __dev && vkDestroyFence ) vkDestroyFence(__dev, __h, host_allocation_callbacks());
117     __h = nullptr;
118     __dev = nullptr;
119   }

127 class semaphore
128 {
129   VkDevice __dev = nullptr;
130   VkSemaphore __h = nullptr;
131   bool __is_timeline = false;
132 
133 public:
134   ~semaphore() { reset(); }
135 
136   semaphore() = default;
137 
138   explicit semaphore(device &dev) : __dev(dev.handle())
139   {
140     if ( !__dev || !vkCreateSemaphore ) throw except::library_error("vk::semaphore: vkCreateSemaphore unresolved");
141     VkSemaphoreCreateInfo ci{};
142     ci.sType = structure_type_of_v<VkSemaphoreCreateInfo>;
143     check_vk(vkCreateSemaphore(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateSemaphore");
144   }
145 
146   semaphore(device &dev, timeline_t, u64 initial_value = 0) : __dev(dev.handle()), __is_timeline(true)
147   {
148     if ( !__dev || !vkCreateSemaphore ) throw except::library_error("vk::semaphore: vkCreateSemaphore unresolved");
149     VkSemaphoreTypeCreateInfo type_ci{};
150     type_ci.sType = structure_type_of_v<VkSemaphoreTypeCreateInfo>;
151     type_ci.semaphoreType = static_cast<VkSemaphoreType>(VK_SEMAPHORE_TYPE_TIMELINE);
152     type_ci.initialValue = initial_value;
153 
154     VkSemaphoreCreateInfo ci{};
155     ci.sType = structure_type_of_v<VkSemaphoreCreateInfo>;
156     ci.pNext = &type_ci;
157     check_vk(vkCreateSemaphore(__dev, &ci, host_allocation_callbacks(), &__h), "vkCreateSemaphore (timeline)");
158   }
159 
160   semaphore(const semaphore &) = delete;
161 
162   semaphore(semaphore &&o) noexcept : __dev(o.__dev), __h(o.__h), __is_timeline(o.__is_timeline)
163   {
164     o.__dev = nullptr;
165     o.__h = nullptr;
166     o.__is_timeline = false;
167   }
168 
169   semaphore &operator=(const semaphore &) = delete;

173   {
174     if ( this != &o ) {
175       reset();
176       __dev = o.__dev;
177       __h = o.__h;
178       __is_timeline = o.__is_timeline;
179       o.__dev = nullptr;
180       o.__h = nullptr;
181       o.__is_timeline = false;

184   }

197 
198   bool
199   is_timeline() const noexcept

202   }
203 
204   explicit

207     return valid();

209 
210   u64
211   current_value() const noexcept
212   {
213     if ( !__is_timeline || !__h || !__dev || !vkGetSemaphoreCounterValue ) return 0;
214     u64 v = 0;
215     (void)vkGetSemaphoreCounterValue(__dev, __h, &v);
216     return v;
217   }
218 
219   bool
220   wait_value(u64 value, u64 timeout_ns = ~0ull)
221   {
222     if ( !__is_timeline || !__h || !__dev || !vkWaitSemaphores ) return false;
223     VkSemaphoreWaitInfo info{};
224     info.sType = structure_type_of_v<VkSemaphoreWaitInfo>;
225     info.semaphoreCount = 1;
226     info.pSemaphores = &__h;
227     info.pValues = &value;
228     VkResult r = vkWaitSemaphores(__dev, &info, timeout_ns);
229     if ( r == VK_SUCCESS ) return true;
230     if ( r == VK_TIMEOUT ) return false;
231     check_vk(r, "vkWaitSemaphores");
232     return false;
233   }
234 
235   void
236   signal_value(u64 value)
237   {
238     if ( !__is_timeline || !__h || !__dev || !vkSignalSemaphore )
239       throw except::library_error("vk::semaphore::signal_value: not a timeline semaphore");
240     VkSemaphoreSignalInfo info{};
241     info.sType = structure_type_of_v<VkSemaphoreSignalInfo>;
242     info.semaphore = __h;
243     info.value = value;
244     check_vk(vkSignalSemaphore(__dev, &info), "vkSignalSemaphore");
245   }
246 
247   void
248   reset() noexcept
249   {
250     if ( __h && __dev && vkDestroySemaphore ) vkDestroySemaphore(__dev, __h, host_allocation_callbacks());
251     __h = nullptr;
252     __dev = nullptr;
253     __is_timeline = false;
254   }
