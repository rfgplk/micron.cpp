54   VkExtent2D
55   __compose_create_info(VkSwapchainCreateInfoKHR &ci, const swapchain_request &r, VkSwapchainKHR old) const
56   {
57     if ( !__dev || !__surf || !vkGetPhysicalDeviceSurfaceCapabilitiesKHR )
58       throw except::library_error("vk::swapchain: surface caps probe missing");
59 
60     VkSurfaceCapabilitiesKHR caps{};
61     check_vk(vkGetPhysicalDeviceSurfaceCapabilitiesKHR(__dev->physical().handle(), __surf->handle(), &caps),
62              "vkGetPhysicalDeviceSurfaceCapabilitiesKHR");
63 
64     VkExtent2D ext = r.extent;
65     if ( ext.width == 0 || ext.height == 0 ) {
66 
67       if ( caps.currentExtent.width != ~0u )
68         ext = caps.currentExtent;
69       else
70         ext = caps.minImageExtent;
71     }
72 
73     if ( ext.width < caps.minImageExtent.width ) ext.width = caps.minImageExtent.width;
74     if ( ext.height < caps.minImageExtent.height ) ext.height = caps.minImageExtent.height;
75     if ( caps.maxImageExtent.width && ext.width > caps.maxImageExtent.width ) ext.width = caps.maxImageExtent.width;
76     if ( caps.maxImageExtent.height && ext.height > caps.maxImageExtent.height ) ext.height = caps.maxImageExtent.height;
77 
78     u32 min_count = r.min_image_count ? r.min_image_count : (caps.minImageCount + 1);
79     if ( caps.maxImageCount > 0 && min_count > caps.maxImageCount ) min_count = caps.maxImageCount;
80     if ( min_count < caps.minImageCount ) min_count = caps.minImageCount;
81 
82     VkCompositeAlphaFlagBitsKHR alpha = VK_COMPOSITE_ALPHA_OPAQUE_BIT_KHR;
83     if ( !static_cast<bool>(caps.supportedCompositeAlpha & VK_COMPOSITE_ALPHA_OPAQUE_BIT_KHR) ) {
84 
85       const u32 mask = static_cast<u32>(caps.supportedCompositeAlpha);
86       for ( u32 b = 1; b < 16; b <<= 1 ) {
87         if ( mask & b ) {
88           alpha = static_cast<VkCompositeAlphaFlagBitsKHR>(b);
89           break;
90         }
91       }
92     }
93 
94     ci = {};
95     ci.sType = structure_type_of_v<VkSwapchainCreateInfoKHR>;
96     ci.surface = __surf->handle();
97     ci.minImageCount = min_count;
98     ci.imageFormat = r.format.format;
99     ci.imageColorSpace = r.format.colorSpace;
100     ci.imageExtent = ext;
101     ci.imageArrayLayers = 1;
102     ci.imageUsage = r.image_usage;
103     ci.imageSharingMode = static_cast<VkSharingMode>(VK_SHARING_MODE_EXCLUSIVE);
104     ci.preTransform = caps.currentTransform;
105     ci.compositeAlpha = alpha;
106     ci.presentMode = r.present_mode;
107     ci.clipped = VK_TRUE;
108     ci.oldSwapchain = old;
109     return ext;
110   }

112   void
113   __retrieve_images()
114   {
115     if ( !__h || !vkGetSwapchainImagesKHR )
116       throw except::library_error("vk::swapchain::__retrieve_images: vkGetSwapchainImagesKHR unresolved");
117     u32 n = 0;
118     check_vk(vkGetSwapchainImagesKHR(__dev->handle(), __h, &n, nullptr), "vkGetSwapchainImagesKHR");
119     if ( n > max_swapchain_images ) n = max_swapchain_images;
120     VkImage raw[max_swapchain_images]{};
121     check_vk(vkGetSwapchainImagesKHR(__dev->handle(), __h, &n, raw), "vkGetSwapchainImagesKHR");
122 
123     VkExtent3D ext3{ __extent.width, __extent.height, 1 };
124     for ( u32 i = 0; i < n; ++i ) {
125       __images[i] = image::wrap_external(raw[i], __format, ext3);
126       __views[i] = image_view(*__dev, __images[i], VK_IMAGE_ASPECT_COLOR_BIT, static_cast<VkImageViewType>(VK_IMAGE_VIEW_TYPE_2D));
127     }
128     __image_count = n;
129   }

131   void
132   __build(const swapchain_request &r, VkSwapchainKHR old)
133   {
134     if ( !__dev || !__dev->valid() ) throw except::logic_error("vk::swapchain: invalid device");
135     if ( !__surf || !__surf->valid() ) throw except::logic_error("vk::swapchain: invalid surface");
136     if ( !vkCreateSwapchainKHR )
137       throw except::library_error("vk::swapchain: vkCreateSwapchainKHR unresolved — enable VK_KHR_swapchain on the device");
138 
139     VkSwapchainCreateInfoKHR ci{};
140     VkExtent2D final_ext = __compose_create_info(ci, r, old);
141 
142     check_vk(vkCreateSwapchainKHR(__dev->handle(), &ci, host_allocation_callbacks(), &__h), "vkCreateSwapchainKHR");
143 
144     __format = r.format.format;
145     __color_space = r.format.colorSpace;
146     __extent = final_ext;
147     __present_mode = r.present_mode;
148     __image_usage = r.image_usage;
149     __min_image_count = ci.minImageCount;
150     __retrieve_images();
151   }

153   void
154   __destroy_views() noexcept
155   {
156     for ( u32 i = 0; i < __image_count; ++i ) __views[i].reset();
157     for ( u32 i = 0; i < __image_count; ++i ) __images[i].reset();
158     __image_count = 0;
159   }

282   VkResult
283   acquire_next_image(semaphore *signal_sem, fence *signal_fence, u64 timeout_ns, u32 *out_index) noexcept
284   {
285     if ( !__h || !__dev || !vkAcquireNextImageKHR ) return VK_ERROR_INITIALIZATION_FAILED;
286     VkSemaphore s = signal_sem ? signal_sem->handle() : VK_NULL_HANDLE;
287     VkFence f = signal_fence ? signal_fence->handle() : VK_NULL_HANDLE;
288     return vkAcquireNextImageKHR(__dev->handle(), __h, timeout_ns, s, f, out_index);
289   }

291   VkResult
292   present(queue &q, u32 image_index, semaphore *wait = nullptr) noexcept
293   {
294     if ( !__h || !q.valid() || !vkQueuePresentKHR ) return VK_ERROR_INITIALIZATION_FAILED;
295 
296     VkSemaphore w = wait ? wait->handle() : VK_NULL_HANDLE;
297     VkSwapchainKHR sc = __h;
298     VkResult per_swapchain = VK_SUCCESS;
299 
300     VkPresentInfoKHR info{};
301     info.sType = structure_type_of_v<VkPresentInfoKHR>;
302     info.waitSemaphoreCount = w ? 1u : 0u;
303     info.pWaitSemaphores = w ? &w : nullptr;
304     info.swapchainCount = 1;
305     info.pSwapchains = &sc;
306     info.pImageIndices = &image_index;
307     info.pResults = &per_swapchain;
308 
309     const VkResult r = vkQueuePresentKHR(q.handle(), &info);
310     if ( r == VK_SUCCESS && per_swapchain != VK_SUCCESS ) return per_swapchain;
311     return r;
312   }

314   void
315   recreate(VkExtent2D new_extent, const swapchain_request *overrides = nullptr)
316   {
317     swapchain_request r;
318     if ( overrides ) r = *overrides;
319     r.extent = new_extent;
320     if ( !overrides ) {
321       r.format.format = __format;
322       r.format.colorSpace = __color_space;
323       r.present_mode = __present_mode;
324       r.image_usage = __image_usage;
325       r.min_image_count = __min_image_count;
326     }
327 
328     VkSwapchainKHR old = __h;
329     __destroy_views();
330     __h = nullptr;
331     __build(r, old);
332     if ( old && vkDestroySwapchainKHR ) vkDestroySwapchainKHR(__dev->handle(), old, host_allocation_callbacks());
333   }

335   void
336   reset() noexcept
337   {

340     __h = nullptr;
