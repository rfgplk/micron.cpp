24 inline VkBool32 VKAPI_PTR
25 default_debug_messenger_callback(VkDebugUtilsMessageSeverityFlagBitsEXT /*severity*/, VkDebugUtilsMessageTypeFlagsEXT /*types*/,
26                                  const VkDebugUtilsMessengerCallbackDataEXT *data, void * /*user*/) noexcept
27 {
28   if ( !data || !data->pMessage ) return VK_FALSE;
29   const char *m = data->pMessage;
30   usize n = micron::strlen(m);
31   micron::syscall(SYS_write, 2, m, n);
32   const char nl = '\n';
33   micron::syscall(SYS_write, 2, &nl, 1);
34   return VK_FALSE;
35 }

37 inline VkDebugUtilsMessengerCreateInfoEXT
38 default_messenger_info() noexcept
39 {
40   VkDebugUtilsMessengerCreateInfoEXT ci{};
41   ci.sType = structure_type_of_v<VkDebugUtilsMessengerCreateInfoEXT>;
42   ci.messageSeverity = static_cast<VkDebugUtilsMessageSeverityFlagsEXT>(VK_DEBUG_UTILS_MESSAGE_SEVERITY_WARNING_BIT_EXT
43                                                                         | VK_DEBUG_UTILS_MESSAGE_SEVERITY_ERROR_BIT_EXT);
44   ci.messageType = static_cast<VkDebugUtilsMessageTypeFlagsEXT>(VK_DEBUG_UTILS_MESSAGE_TYPE_GENERAL_BIT_EXT
45                                                                 | VK_DEBUG_UTILS_MESSAGE_TYPE_VALIDATION_BIT_EXT
46                                                                 | VK_DEBUG_UTILS_MESSAGE_TYPE_PERFORMANCE_BIT_EXT);
47   ci.pfnUserCallback = &default_debug_messenger_callback;
48   ci.pUserData = nullptr;
49   return ci;
50 }

52 class messenger
53 {
54 private:
55   VkInstance __inst = nullptr;
56   VkDebugUtilsMessengerEXT __h = nullptr;
57 
58 public:
59   ~messenger() { reset(); }
60 
61   messenger() = default;
62 
63   explicit messenger(VkInstance inst) : __inst(inst)
64   {
65     if ( !inst || !vkCreateDebugUtilsMessengerEXT )
66       throw except::library_error("vk::messenger: VK_EXT_debug_utils not loaded — was hints.debug enabled?");
67     auto ci = default_messenger_info();
68     VkResult r = vkCreateDebugUtilsMessengerEXT(inst, &ci, host_allocation_callbacks(), &__h);
69     check_vk(r, "vkCreateDebugUtilsMessengerEXT");
70   }
71 
72   messenger(VkInstance inst, const VkDebugUtilsMessengerCreateInfoEXT &ci) : __inst(inst)
73   {
74     if ( !inst || !vkCreateDebugUtilsMessengerEXT )
75       throw except::library_error("vk::messenger: VK_EXT_debug_utils not loaded — was hints.debug enabled?");
76     VkResult r = vkCreateDebugUtilsMessengerEXT(inst, &ci, host_allocation_callbacks(), &__h);
77     check_vk(r, "vkCreateDebugUtilsMessengerEXT");
78   }
79 
80   messenger(const messenger &) = delete;
81 
82   messenger(messenger &&o) noexcept : __inst(o.__inst), __h(o.__h)
83   {
84     o.__inst = nullptr;
85     o.__h = nullptr;
86   }
87 
88   messenger &operator=(const messenger &) = delete;
89 
90   messenger &
91   operator=(messenger &&o) noexcept
92   {
93     if ( this != &o ) {
94       reset();
95       __inst = o.__inst;
96       __h = o.__h;
97       o.__inst = nullptr;
98       o.__h = nullptr;
99     }
100     return *this;
101   }
102 
103   VkDebugUtilsMessengerEXT
104   handle() const noexcept
105   {
106     return __h;
107   }
108 
109   bool
110   valid() const noexcept
111   {
112     return __h != nullptr;
113   }
114 
115   explicit
116   operator bool() const noexcept
117   {
118     return valid();
119   }
120 
121   void
122   reset() noexcept
123   {
124     if ( __h && __inst && vkDestroyDebugUtilsMessengerEXT ) vkDestroyDebugUtilsMessengerEXT(__inst, __h, host_allocation_callbacks());
125     __h = nullptr;
126     __inst = nullptr;
127   }

142 inline void
143 begin_label(VkCommandBuffer cb, const char *name, const float color[4] = nullptr) noexcept
144 {
145   if ( !cb || !name || !vkCmdBeginDebugUtilsLabelEXT ) return;
146   VkDebugUtilsLabelEXT label{};
147   label.sType = structure_type_of_v<VkDebugUtilsLabelEXT>;
148   label.pLabelName = name;
149   if ( color ) {
150     label.color[0] = color[0];
151     label.color[1] = color[1];
152     label.color[2] = color[2];
153     label.color[3] = color[3];
154   }
155   vkCmdBeginDebugUtilsLabelEXT(cb, &label);
156 }
