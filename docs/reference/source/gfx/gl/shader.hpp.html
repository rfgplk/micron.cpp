21 enum class shader_kind : GLenum {
22   vertex = GL_VERTEX_SHADER,
23   fragment = GL_FRAGMENT_SHADER,
24   geometry = GL_GEOMETRY_SHADER,
25   tess_control = GL_TESS_CONTROL_SHADER,
26   tess_evaluation = GL_TESS_EVALUATION_SHADER,
27   compute = GL_COMPUTE_SHADER,
28 };
29 
30 class shader
31 {
32 private:
33   GLuint __h = 0;
34 
35 public:
36   ~shader() { reset(); }
37 
38   shader() = default;
39 
40   explicit shader(shader_kind k)
41   {
42     if ( !glCreateShader ) throw except::library_error("shader: gl not loaded");
43     __h = glCreateShader(static_cast<GLenum>(k));
44     if ( !__h ) throw except::library_error("shader: glCreateShader returned 0");
45   }
46 
47   shader(shader_kind k, const char *src) : shader(k) { compile(src); }
48 
49   template<micron::has_cstr S> shader(shader_kind k, const S &src) : shader(k) { compile(src.c_str()); }
50 
51   shader(const shader &) = delete;
52 
53   shader(shader &&o) noexcept : __h(o.__h) { o.__h = 0; }
54 
55   shader &operator=(const shader &) = delete;
56 
57   shader &
58   operator=(shader &&o) noexcept
59   {
60     if ( this != &o ) {
61       reset();
62       __h = o.__h;
63       o.__h = 0;
64     }
65     return *this;
66   }
67 
68   void
69   reset() noexcept
70   {
71     if ( __h && glDeleteShader ) glDeleteShader(__h);
72     __h = 0;
73   }
74 
75   GLuint
76   handle() const noexcept
77   {
78     return __h;
79   }
80 
81   bool
82   valid() const noexcept
83   {
84     return __h != 0;
85   }
86 
87   explicit
88   operator bool() const noexcept
89   {
90     return valid();
91   }
92 
93   void
94   compile(const char *src)
95   {
96     if ( !__h ) throw except::logic_error("shader: empty handle");
97     const GLchar *arr[1] = { src };
98     glShaderSource(__h, 1, arr, nullptr);
99     glCompileShader(__h);
100     GLint ok = GL_FALSE;
101     glGetShaderiv(__h, GL_COMPILE_STATUS, &ok);
102     if ( ok == GL_FALSE ) {
103       static char log[1024] = { 0 };
104       GLsizei len = 0;
105       glGetShaderInfoLog(__h, sizeof(log) - 1, &len, log);
106       throw except::library_error(log[0] ? log : "shader: compile failed");
107     }
108   }
109 
110   void
111   debug_label(const char *name) noexcept
112   {
113     if ( __h && glObjectLabel && name ) glObjectLabel(0x8B31 /* GL_SHADER */, __h, -1, name);
114   }
115 };
