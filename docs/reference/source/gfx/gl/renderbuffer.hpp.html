20 class renderbuffer
21 {
22   GLuint __h = 0;
23 
24 public:
25   ~renderbuffer() { reset(); }
26 
27   renderbuffer()
28   {
29     if ( !glGenRenderbuffers ) throw except::library_error("renderbuffer: gl not loaded");
30     glGenRenderbuffers(1, &__h);
31     if ( !__h ) throw except::library_error("renderbuffer: glGenRenderbuffers returned 0");
32   }
33 
34   renderbuffer(const renderbuffer &) = delete;
35 
36   renderbuffer(renderbuffer &&o) noexcept : __h(o.__h) { o.__h = 0; }
37 
38   renderbuffer &operator=(const renderbuffer &) = delete;
39 
40   renderbuffer &
41   operator=(renderbuffer &&o) noexcept
42   {
43     if ( this != &o ) {
44       reset();
45       __h = o.__h;
46       o.__h = 0;
47     }
48     return *this;
49   }
50 
51   void
52   reset() noexcept
53   {
54     if ( __h && glDeleteRenderbuffers ) glDeleteRenderbuffers(1, &__h);
55     __h = 0;
56   }
57 
58   GLuint
59   handle() const noexcept
60   {
61     return __h;
62   }
63 
64   bool
65   valid() const noexcept
66   {
67     return __h != 0;
68   }
69 
70   explicit
71   operator bool() const noexcept
72   {
73     return valid();
74   }
75 
76   void
77   bind() const noexcept
78   {
79     if ( __h && glBindRenderbuffer ) glBindRenderbuffer(GL_RENDERBUFFER, __h);
80   }
81 
82   void
83   storage(GLenum internalformat, GLsizei width, GLsizei height) noexcept
84   {
85     if ( !__h || !glRenderbufferStorage ) return;
86     bind();
87     glRenderbufferStorage(GL_RENDERBUFFER, internalformat, width, height);
88   }
89 
90   void
91   debug_label(const char *name) noexcept
92   {
93     if ( __h && glObjectLabel && name ) glObjectLabel(0x8D41 /* GL_RENDERBUFFER */, __h, -1, name);
94   }
95 };
