96   static bool
97   __ext_present(const char *list, const char *name) noexcept
98   {
99     if ( !list || !name ) return false;
100     const usize nlen = micron::strlen(name);
101     if ( nlen == 0 ) return false;
102     const char *p = list;
103     while ( (p = micron::strstr(p, name)) != nullptr ) {
104       const bool left_ok = (p == list) || (p[-1] == ' ');
105       const bool right_ok = (p[nlen] == ' ' || p[nlen] == '\0');
106       if ( left_ok && right_ok ) return true;
107       p += nlen;
108     }
109     return false;
110   }

112   bool
113   has_display_extension(EGLDisplay dpy, const char *name) noexcept

115     if ( !eglQueryString || dpy == EGL_NO_DISPLAY || !name ) return false;

210   void
211   open(egl_lib_t &e, EGLenum platform, void *native)
212   {
213     __lib = &e;
214     if ( e.eglGetPlatformDisplayEXT ) {
215       const EGLint attribs[] = { EGL_NONE };
216       __display = e.eglGetPlatformDisplayEXT(platform, native, attribs);
217     } else {
218       __display = e.eglGetDisplay(reinterpret_cast<EGLNativeDisplayType>(native));
219     }
220     if ( __display == EGL_NO_DISPLAY ) throw except::library_error("egl: eglGetDisplay returned EGL_NO_DISPLAY");
221     if ( !e.eglInitialize(__display, &__major, &__minor) ) throw except::library_error("egl: eglInitialize failed");
222     if ( !e.eglBindAPI(EGL_OPENGL_API) ) throw except::library_error("egl: eglBindAPI(EGL_OPENGL_API) failed");
223   }

225   void
226   choose_config(const egl_config_request_t &r)
227   {
228     if ( !__lib || __display == EGL_NO_DISPLAY ) throw except::logic_error("egl_context: open() not called");
229     const EGLint attribs[] = {
230       EGL_SURFACE_TYPE, r.surface_type, EGL_RENDERABLE_TYPE, r.renderable,          EGL_RED_SIZE, r.red_size,     EGL_GREEN_SIZE,
231       r.green_size,     EGL_BLUE_SIZE,  r.blue_size,         EGL_ALPHA_SIZE,        r.alpha_size, EGL_DEPTH_SIZE, r.depth_size,
232       EGL_STENCIL_SIZE, r.stencil_size, EGL_SAMPLE_BUFFERS,  r.samples > 0 ? 1 : 0, EGL_SAMPLES,  r.samples,      EGL_NONE,
233     };
234     EGLint n = 0;
235     if ( !__lib->eglChooseConfig(__display, attribs, &__config, 1, &n) || n == 0 ) {
236       throw except::library_error("egl: eglChooseConfig found no matching config");
237     }
238   }

240   void
241   create_context(const egl_context_request_t &r)
242   {
243     if ( !__lib || !__config ) throw except::logic_error("egl_context: choose_config not called");
244     EGLint flags = 0;
245     if ( r.debug ) flags |= EGL_CONTEXT_OPENGL_DEBUG_BIT_KHR;
246     if ( r.forward_compat ) flags |= EGL_CONTEXT_OPENGL_FORWARD_COMPATIBLE_BIT_KHR;
247     const EGLint profile = r.core_profile ? EGL_CONTEXT_OPENGL_CORE_PROFILE_BIT : EGL_CONTEXT_OPENGL_COMPATIBILITY_PROFILE_BIT;
248     const EGLint attribs[] = {
249       EGL_CONTEXT_MAJOR_VERSION,
250       r.major,
251       EGL_CONTEXT_MINOR_VERSION,
252       r.minor,
253       EGL_CONTEXT_OPENGL_PROFILE_MASK,
254       profile,
255       EGL_CONTEXT_FLAGS_KHR,
256       flags,
257       EGL_NONE,
258     };
259     __ctx = __lib->eglCreateContext(__display, __config, r.share, attribs);
260     if ( __ctx == EGL_NO_CONTEXT ) throw except::library_error("egl: eglCreateContext failed");
261   }

263   void
264   create_window_surface(void *native_window)
265   {
266     if ( !__lib || !__config ) throw except::logic_error("egl_context: choose_config not called");
267     if ( __lib->eglCreatePlatformWindowSurfaceEXT ) {
268       const EGLint attribs[] = { EGL_NONE };
269       __surface = __lib->eglCreatePlatformWindowSurfaceEXT(__display, __config, native_window, attribs);
270     } else {
271       __surface = __lib->eglCreateWindowSurface(__display, __config, reinterpret_cast<EGLNativeWindowType>(native_window), nullptr);
272     }
273     if ( __surface == EGL_NO_SURFACE ) throw except::library_error("egl: eglCreateWindowSurface failed");
274   }

276   EGLSurface
277   create_pbuffer_surface(i32 width, i32 height) noexcept
278   {
279     if ( !__lib || !__lib->eglCreatePbufferSurface || !__config ) return EGL_NO_SURFACE;
280     const EGLint attribs[] = {
281       EGL_WIDTH, static_cast<EGLint>(width), EGL_HEIGHT, static_cast<EGLint>(height), EGL_NONE,

283     return __lib->eglCreatePbufferSurface(__display, __config, attribs);
284   }

323   void
324   destroy() noexcept
325   {
326     if ( !__lib ) return;
327     if ( __surface != EGL_NO_SURFACE ) __lib->eglDestroySurface(__display, __surface);
328     if ( __ctx != EGL_NO_CONTEXT ) __lib->eglDestroyContext(__display, __ctx);
329     if ( __display != EGL_NO_DISPLAY ) __lib->eglTerminate(__display);
330     __display = EGL_NO_DISPLAY;
331     __ctx = EGL_NO_CONTEXT;
332     __surface = EGL_NO_SURFACE;
333   }
