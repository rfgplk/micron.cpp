75   bool
76   has_extension(x11_display_t &dpy, const char *name) noexcept
77   {
78     if ( !glXQueryExtensionsString || !name ) return false;
79     const char *exts = glXQueryExtensionsString(dpy.display(), dpy.screen());
80     if ( !exts ) return false;
81     const usize nlen = micron::strlen(name);
82     if ( nlen == 0 ) return false;
83     const char *p = exts;
84     while ( (p = micron::strstr(p, name)) != nullptr ) {
85       const bool left_ok = (p == exts) || (p[-1] == ' ');
86       const bool right_ok = (p[nlen] == ' ' || p[nlen] == '\0');
87       if ( left_ok && right_ok ) return true;
88       p += nlen;
89     }
90     return false;
91   }

195   void
196   pick_visual(glx_lib_t &g, x11_display_t &dpy, const glx_fbconfig_request_t &r)
197   {
198     __lib = &g;
199     __disp = &dpy;
200     const int attribs[] = {
201       GLX_X_RENDERABLE,
202       1,
203       GLX_DRAWABLE_TYPE,
204       GLX_WINDOW_BIT,
205       GLX_RENDER_TYPE,
206       GLX_RGBA_BIT,
207       GLX_X_VISUAL_TYPE,
208       GLX_TRUE_COLOR,
209       GLX_RED_SIZE,
210       r.red_size,
211       GLX_GREEN_SIZE,
212       r.green_size,
213       GLX_BLUE_SIZE,
214       r.blue_size,
215       GLX_ALPHA_SIZE,
216       r.alpha_size,
217       GLX_DEPTH_SIZE,
218       r.depth_size,
219       GLX_STENCIL_SIZE,
220       r.stencil_size,
221       GLX_DOUBLEBUFFER,
222       r.double_buffer ? 1 : 0,
223       GLX_SAMPLE_BUFFERS,
224       r.samples > 0 ? 1 : 0,
225       GLX_SAMPLES,
226       r.samples,
227       0,
228     };
229     int n = 0;
230     GLXFBConfig *cfgs = g.glXChooseFBConfig(dpy.display(), dpy.screen(), attribs, &n);
231     if ( !cfgs || n == 0 ) {
232       if ( cfgs ) dpy.lib()->XFree(cfgs);
233       throw except::library_error("glx: no matching FBConfig");
234     }
235     __fbconfig = cfgs[0];
236     dpy.lib()->XFree(cfgs);
237 
238     __visual = g.glXGetVisualFromFBConfig(dpy.display(), __fbconfig);
239     if ( !__visual ) throw except::library_error("glx: glXGetVisualFromFBConfig returned null");
240   }

284   void
285   create(const glx_context_request_t &r)
286   {
287     if ( !__lib || !__disp || !__fbconfig ) throw except::logic_error("glx_context: pick_visual not called");
288 
289     if ( __lib->glXCreateContextAttribsARB ) {
290       int flags = 0;
291       if ( r.debug ) flags |= GLX_CONTEXT_DEBUG_BIT_ARB;
292       if ( r.forward_compat ) flags |= GLX_CONTEXT_FORWARD_COMPATIBLE_BIT_ARB;
293       const int profile = r.core_profile ? GLX_CONTEXT_CORE_PROFILE_BIT_ARB : GLX_CONTEXT_COMPATIBILITY_PROFILE_BIT_ARB;
294       const int attribs[] = {
295         GLX_CONTEXT_MAJOR_VERSION_ARB,
296         r.major,
297         GLX_CONTEXT_MINOR_VERSION_ARB,
298         r.minor,
299         GLX_CONTEXT_PROFILE_MASK_ARB,
300         profile,
301         GLX_CONTEXT_FLAGS_ARB,
302         flags,
303         0,
304       };
305       __ctx = __lib->glXCreateContextAttribsARB(__disp->display(), __fbconfig, r.share, 1, attribs);
306     }
307     if ( !__ctx && __lib->glXCreateNewContext ) {
308       __ctx = __lib->glXCreateNewContext(__disp->display(), __fbconfig, /*GLX_RGBA_TYPE*/ 0x8014, r.share, 1);
309     }
310     if ( !__ctx ) throw except::library_error("glx: context creation failed");
311   }
