694 {
695   for ( u16 i = 0; i < gl_table_size; ++i ) {
696     const char *a = gl_table[i].name;
697     const char *b = name;
698     while ( *a && *a == *b ) {
699       ++a;
700       ++b;
701     }
702     if ( *a == 0 && *b == 0 ) return i;
703   }
704   return u16(-1);
705 }
