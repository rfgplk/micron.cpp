20 enum class buffer_target : GLenum {
21   array = GL_ARRAY_BUFFER,
22   element_array = GL_ELEMENT_ARRAY_BUFFER,
23   uniform = GL_UNIFORM_BUFFER,
24   shader_storage = GL_SHADER_STORAGE_BUFFER,
25   copy_read = GL_COPY_READ_BUFFER,
26   copy_write = GL_COPY_WRITE_BUFFER,
27   pixel_pack = GL_PIXEL_PACK_BUFFER,
28   pixel_unpack = GL_PIXEL_UNPACK_BUFFER,
29 };

31 enum class buffer_usage : GLenum {
32   static_draw = GL_STATIC_DRAW,
33   static_read = GL_STATIC_READ,
34   static_copy = GL_STATIC_COPY,
35   stream_draw = GL_STREAM_DRAW,
36   stream_read = GL_STREAM_READ,
37   stream_copy = GL_STREAM_COPY,
38   dynamic_draw = GL_DYNAMIC_DRAW,
39   dynamic_read = GL_DYNAMIC_READ,
40   dynamic_copy = GL_DYNAMIC_COPY,
41 };

43 class buffer
44 {
45 private:
46   GLuint __h = 0;
47   buffer_target __target = buffer_target::array;
48 
49 public:
50   ~buffer() { reset(); }
51 
52   buffer() = default;
53 
54   explicit buffer(buffer_target t) : __target(t)
55   {
56     if ( !glGenBuffers ) throw except::library_error("buffer: gl not loaded");
57     glGenBuffers(1, &__h);
58     if ( !__h ) throw except::library_error("buffer: glGenBuffers returned 0");
59   }
60 
61   buffer(const buffer &) = delete;
62 
63   buffer(buffer &&o) noexcept : __h(o.__h), __target(o.__target) { o.__h = 0; }
64 
65   buffer &operator=(const buffer &) = delete;
66 
67   buffer &
68   operator=(buffer &&o) noexcept
69   {
70     if ( this != &o ) {
71       reset();
72       __h = o.__h;
73       __target = o.__target;
74       o.__h = 0;
75     }
76     return *this;
77   }
78 
79   void
80   reset() noexcept
81   {
82     if ( __h && glDeleteBuffers ) glDeleteBuffers(1, &__h);
83     __h = 0;
84   }
85 
86   GLuint
87   handle() const noexcept
88   {
89     return __h;
90   }
91 
92   buffer_target
93   target() const noexcept
94   {
95     return __target;
96   }
97 
98   bool
99   valid() const noexcept
100   {
101     return __h != 0;
102   }
103 
104   explicit
105   operator bool() const noexcept
106   {
107     return valid();
108   }
109 
110   void
111   bind() const noexcept
112   {
113     if ( __h && glBindBuffer ) glBindBuffer(static_cast<GLenum>(__target), __h);
114   }
115 
116   void
117   unbind() const noexcept
118   {
119     if ( glBindBuffer ) glBindBuffer(static_cast<GLenum>(__target), 0);
120   }
121 
122   void
123   data(const void *src, GLsizeiptr size, buffer_usage usage) noexcept
124   {
125     if ( !__h || !glBufferData ) return;
126     bind();
127     glBufferData(static_cast<GLenum>(__target), size, src, static_cast<GLenum>(usage));
128   }
129 
130   void
131   sub_data(GLintptr offset, GLsizeiptr size, const void *src) noexcept
132   {
133     if ( !__h || !glBufferSubData ) return;
134     bind();
135     glBufferSubData(static_cast<GLenum>(__target), offset, size, src);
136   }
137 
138   void *
139   map(GLintptr offset, GLsizeiptr length, GLbitfield access) noexcept
140   {
141     if ( !__h || !glMapBufferRange ) return nullptr;
142     bind();
143     return glMapBufferRange(static_cast<GLenum>(__target), offset, length, access);
144   }
145 
146   bool
147   unmap() noexcept
148   {
149     if ( !__h || !glUnmapBuffer ) return false;
150     bind();
151     return glUnmapBuffer(static_cast<GLenum>(__target)) != GL_FALSE;
152   }
153 
154   void
155   debug_label(const char *name) noexcept
156   {
157     if ( __h && glObjectLabel && name ) glObjectLabel(0x82E0 /* GL_BUFFER */, __h, -1, name);
158   }
159 };
