20 enum class query_target : GLenum {
21   samples_passed = 0x8914,                       // GL_SAMPLES_PASSED
22   any_samples_passed = 0x8C2F,                   // GL_ANY_SAMPLES_PASSED
23   any_samples_passed_conservative = 0x8D6A,      // GL_ANY_SAMPLES_PASSED_CONSERVATIVE
24   primitives_generated = 0x8C87,                 // GL_PRIMITIVES_GENERATED
25   time_elapsed = 0x88BF,                         // GL_TIME_ELAPSED
26   timestamp = 0x8E28,                            // GL_TIMESTAMP
27 };

29 class query
30 {
31   GLuint __h = 0;
32   query_target __target = query_target::samples_passed;
33 
34 public:
35   ~query() { reset(); }
36 
37   query() = default;
38 
39   explicit query(query_target t) : __target(t)
40   {
41     if ( !glGenQueries ) throw except::library_error("query: gl not loaded");
42     glGenQueries(1, &__h);
43     if ( !__h ) throw except::library_error("query: glGenQueries returned 0");
44   }
45 
46   query(const query &) = delete;
47 
48   query(query &&o) noexcept : __h(o.__h), __target(o.__target) { o.__h = 0; }
49 
50   query &operator=(const query &) = delete;
51 
52   query &
53   operator=(query &&o) noexcept
54   {
55     if ( this != &o ) {
56       reset();
57       __h = o.__h;
58       __target = o.__target;
59       o.__h = 0;
60     }
61     return *this;
62   }
63 
64   void
65   reset() noexcept
66   {
67     if ( __h && glDeleteQueries ) glDeleteQueries(1, &__h);
68     __h = 0;
69   }
70 
71   GLuint
72   handle() const noexcept
73   {
74     return __h;
75   }
76 
77   query_target
78   target() const noexcept
79   {
80     return __target;
81   }
82 
83   bool
84   valid() const noexcept
85   {
86     return __h != 0;
87   }
88 
89   explicit
90   operator bool() const noexcept
91   {
92     return valid();
93   }
94 
95   void
96   begin() noexcept
97   {
98     if ( __h && glBeginQuery ) glBeginQuery(static_cast<GLenum>(__target), __h);
99   }
100 
101   void
102   end() noexcept
103   {
104     if ( glEndQuery ) glEndQuery(static_cast<GLenum>(__target));
105   }
106 
107   bool
108   available() const noexcept
109   {
110     GLuint r = 0;
111     if ( __h && glGetQueryObjectuiv ) glGetQueryObjectuiv(__h, 0x8867 /* GL_QUERY_RESULT_AVAILABLE */, &r);
112     return r != 0;
113   }
114 
115   GLuint
116   result_u32() const noexcept
117   {
118     GLuint r = 0;
119     if ( __h && glGetQueryObjectuiv ) glGetQueryObjectuiv(__h, 0x8866 /* GL_QUERY_RESULT */, &r);
120     return r;
121   }
122 
123   GLint
124   result_i32() const noexcept
125   {
126     GLint r = 0;
127     if ( __h && glGetQueryObjectiv ) glGetQueryObjectiv(__h, 0x8866 /* GL_QUERY_RESULT */, &r);
128     return r;
129   }
130 
131   GLuint64
132   result_u64() const noexcept
133   {
134     GLuint64 r = 0;
135     if ( __h && glGetQueryObjectui64v ) glGetQueryObjectui64v(__h, 0x8866 /* GL_QUERY_RESULT */, &r);
136     return r;
137   }
138 
139   GLint64
140   result_i64() const noexcept
141   {
142     GLint64 r = 0;
143     if ( __h && glGetQueryObjecti64v ) glGetQueryObjecti64v(__h, 0x8866 /* GL_QUERY_RESULT */, &r);
144     return r;

147   void
148   debug_label(const char *name) noexcept
149   {
150     if ( __h && glObjectLabel && name ) glObjectLabel(0x8253 /* GL_QUERY */, __h, -1, name);
151   }

154 inline void
155 query_timestamp(query &q) noexcept
156 {
157   if ( q.handle() && glQueryCounter ) glQueryCounter(q.handle(), 0x8E28 /* GL_TIMESTAMP */);
158 }
