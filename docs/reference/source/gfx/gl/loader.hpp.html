12 // Two-track GL function loading:
13 //
14 //   1. Runtime full-table — populates every inline-variable PFN pointer in
15 //      opengl.hpp by walking gl_table[] in lockstep with __gl_fnptr_slots[].
16 //      Called once after the context is current; subsequent gl::glClear(...)
17 //      calls hit the resolved pointer directly with zero indirection cost.
18 //
19 //   2. Constexpr subset — `gl::subset<I0, I1, ...>` packs only the indices
20 //      the user names at compile time. Storage is exactly N pointers; no
21 //      ~100-pointer table linked in when you only need three calls.
22 //      Use the MICRON_GL_SYM(name) macro to translate a name to an index
23 //      at compile time:
24 //
25 //          gl::subset<MICRON_GL_SYM(glClear), MICRON_GL_SYM(glDrawArrays)> t;
26 //          t.load(get_proc);
27 //          auto clear = t.as<MICRON_GL_SYM(glClear), gl::PFNGLCLEARPROC>();
28 //          clear(gl::GL_COLOR_BUFFER_BIT);
29

39 inline get_proc_t __get_proc = nullptr;

41 inline void
42 __load_core_4_6(get_proc_t gp) noexcept
43 {
44   __get_proc = gp;
45   for ( usize i = 0; i < gl_table_size; ++i ) {
46     *__gl_fnptr_slots[i] = gp(gl_table[i].name);
47   }
48 }

50 inline void *
51 get_proc_address(const char *name) noexcept
52 {
53   return __get_proc ? __get_proc(name) : nullptr;
54 }

56 // Returns true if `ext` is present, case-sensitive and word-bounded, in a
57 // space-separated extension list. GL ES-style. Used by the legacy
58 // GL_EXTENSIONS path here and by the EGL / GLX extension probes.
59 inline bool
60 __extension_in(const char *ext, const char *list) noexcept
61 {
62   if ( !ext || !list ) return false;
63   const usize elen = micron::strlen(ext);
64   if ( elen == 0 ) return false;
65   const char *p = list;
66   while ( (p = micron::strstr(p, ext)) != nullptr ) {
67     const bool left_ok = (p == list) || (p[-1] == ' ');
68     const bool right_ok = (p[elen] == ' ' || p[elen] == '\0');
69     if ( left_ok && right_ok ) return true;
70     p += elen;
71   }
72   return false;
73 }

75 // Walks the GL 3.0+ extension enumeration via glGetStringi(GL_EXTENSIONS, i)
76 // when available. The caller must have a current context.
77 inline bool
78 has_extension(const char *name) noexcept
79 {
80   if ( !glGetIntegerv || !glGetStringi || !glGetString ) return false;
81   GLint n = 0;
82   glGetIntegerv(GL_NUM_EXTENSIONS, &n);
83   if ( n > 0 ) {
84     for ( GLint i = 0; i < n; ++i ) {
85       const GLubyte *s = glGetStringi(GL_EXTENSIONS, static_cast<GLuint>(i));
86       if ( !s ) continue;
87       if ( micron::strcmp(reinterpret_cast<const char *>(s), name) == 0 ) return true;
88     }
89     return false;
90   }
91   const GLubyte *legacy = glGetString(GL_EXTENSIONS);
92   return __extension_in(name, reinterpret_cast<const char *>(legacy));
93 }

95 // --- Comptime subset ----------------------------------------------------
96 // Storage is sizeof...(Indices) pointers. Names are looked up at compile
97 // time via gl_table[Indices].name; the runtime cost of `load()` is one
98 // get_proc call per index.
99 
100 template<u16... Indices> class subset
101 {
102 private:
103   void *__fns[sizeof...(Indices)] = {};
104 
105 public:
106   static constexpr usize count = sizeof...(Indices);
107   static constexpr u16 indices[count] = { Indices... };
108 
109   void
110   load(get_proc_t gp) noexcept
111   {
112     usize k = 0;
113     ((__fns[k++] = gp(gl_table[Indices].name)), ...);
114   }
115 
116   // Compute the local array position of a globally-numbered index. consteval
117   // so the cost lives entirely at compile time.
118   template<u16 Idx>
119   static consteval usize
120   position_of() noexcept
121   {
122     static_assert(((Idx == Indices) || ...), "subset does not contain that index");
123     usize pos = 0;
124     bool found = false;
125     auto check = [&](u16 idx) {
126       if ( !found ) {
127         if ( idx == Idx )
128           found = true;
129         else
130           ++pos;
131       }
132     };
133     ((check(Indices)), ...);
134     return pos;
135   }
136 
137   template<u16 Idx>
138   void *
139   raw() const noexcept
140   {
141     return __fns[position_of<Idx>()];
142   }
143 
144   template<u16 Idx, class Fn>
145   Fn
146   as() const noexcept
147   {
148     return reinterpret_cast<Fn>(raw<Idx>());
149   }
150 };
