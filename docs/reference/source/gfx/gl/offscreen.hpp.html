26 class offscreen_surface
27 {
28   display *__dpy = nullptr;
29   i32 __width = 0;
30   i32 __height = 0;
31   context_hints __hints{};
32   glx_context_t __glx_ctx{};
33   GLXPbuffer __glx_pbuf = 0;
34   egl_context_t __egl_ctx{};
35   EGLSurface __egl_pbuf = EGL_NO_SURFACE;
36 
37   void
38   __build_x11()
39   {
40     auto *xd = __dpy->as_x11();
41     if ( !xd ) throw except::logic_error("gl::offscreen_surface: display is not x11");
42     glx_fbconfig_request_t fbreq;
43     fbreq.depth_size = __hints.depth_bits;
44     fbreq.stencil_size = __hints.stencil_bits;
45     fbreq.samples = __hints.ms_samples;
46     fbreq.double_buffer = false;
47     __glx_ctx.pick_pbuffer_fbconfig(glx_lib(), *xd, fbreq);
48     __glx_pbuf = __glx_ctx.create_pbuffer(__width, __height);
49     if ( !__glx_pbuf ) throw except::library_error("offscreen_surface: glXCreatePbuffer returned 0");
50   }
51 
52   void
53   __build_wayland()
54   {
55     auto *wd = __dpy->as_wayland();
56     if ( !wd ) throw except::logic_error("gl::offscreen_surface: display is not wayland");
57     egl_config_request_t cr_cfg;
58     cr_cfg.depth_size = __hints.depth_bits;
59     cr_cfg.stencil_size = __hints.stencil_bits;
60     cr_cfg.samples = __hints.ms_samples;
61     cr_cfg.surface_type = EGL_PBUFFER_BIT;
62     __egl_ctx.open(egl_lib(), EGL_PLATFORM_WAYLAND_EXT, wd->display());
63     __egl_ctx.choose_config(cr_cfg);
64     __egl_pbuf = __egl_ctx.create_pbuffer_surface(__width, __height);
65     if ( __egl_pbuf == EGL_NO_SURFACE ) throw except::library_error("offscreen_surface: eglCreatePbufferSurface failed");
66     __egl_ctx.attach_surface(__egl_pbuf);
67   }
68 
69 public:
70   ~offscreen_surface()
71   {
72     if ( __glx_pbuf ) __glx_ctx.destroy_pbuffer(__glx_pbuf);
73   }
74 
75   offscreen_surface() = default;
76 
77   offscreen_surface(display &dpy, i32 width, i32 height, const context_hints &hints = {})
78       : __dpy(&dpy), __width(width), __height(height), __hints(hints)
79   {
80     switch ( dpy.backend() ) {
81     case backend_tag_t::x11:
82       __build_x11();
83       break;
84     case backend_tag_t::wayland:
85       __build_wayland();
86       break;
87     case backend_tag_t::none:
88     default:
89       throw except::logic_error("gl::offscreen_surface: display has no backend");
90     }
91   }
92 
93   offscreen_surface(const offscreen_surface &) = delete;
94   offscreen_surface(offscreen_surface &&) = delete;
95 
96   offscreen_surface &operator=(const offscreen_surface &) = delete;
97   offscreen_surface &operator=(offscreen_surface &&) = delete;
