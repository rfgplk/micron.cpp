22 enum class fb_target : GLenum {
23   both = GL_FRAMEBUFFER,
24   draw = GL_DRAW_FRAMEBUFFER,
25   read = GL_READ_FRAMEBUFFER,
26 };
27 
28 class framebuffer
29 {
30 private:
31   GLuint __h = 0;
32   fb_target __target = fb_target::both;
33 
34 public:
35   ~framebuffer() { reset(); }
36 
37   explicit framebuffer(fb_target t = fb_target::both) : __target(t)
38   {
39     if ( !glGenFramebuffers ) throw except::library_error("framebuffer: gl not loaded");
40     glGenFramebuffers(1, &__h);
41     if ( !__h ) throw except::library_error("framebuffer: glGenFramebuffers returned 0");
42   }
43 
44   framebuffer(const framebuffer &) = delete;
45 
46   framebuffer(framebuffer &&o) noexcept : __h(o.__h), __target(o.__target) { o.__h = 0; }
47 
48   framebuffer &operator=(const framebuffer &) = delete;
49 
50   framebuffer &
51   operator=(framebuffer &&o) noexcept
52   {
53     if ( this != &o ) {
54       reset();
55       __h = o.__h;
56       __target = o.__target;
57       o.__h = 0;
58     }
59     return *this;
60   }
61 
62   void
63   reset() noexcept
64   {
65     if ( __h && glDeleteFramebuffers ) glDeleteFramebuffers(1, &__h);
66     __h = 0;
67   }
68 
69   GLuint
70   handle() const noexcept
71   {
72     return __h;
73   }
74 
75   bool
76   valid() const noexcept
77   {
78     return __h != 0;
79   }
80 
81   explicit
82   operator bool() const noexcept
83   {
84     return valid();
85   }
86 
87   void
88   bind() const noexcept
89   {
90     if ( __h && glBindFramebuffer ) glBindFramebuffer(static_cast<GLenum>(__target), __h);
91   }
92 
93   static void
94   bind_default(fb_target t = fb_target::both) noexcept
95   {
96     if ( glBindFramebuffer ) glBindFramebuffer(static_cast<GLenum>(t), 0);
97   }
98 
99   void
100   attach_color(GLuint slot, const texture &tex, GLint level = 0) noexcept
101   {
102     if ( !__h || !glFramebufferTexture2D ) return;
103     bind();
104     glFramebufferTexture2D(static_cast<GLenum>(__target), GL_COLOR_ATTACHMENT0 + slot, static_cast<GLenum>(tex.target()), tex.handle(),
105                            level);
106   }
107 
108   void
109   attach_depth(const texture &tex, GLint level = 0) noexcept
110   {
111     if ( !__h || !glFramebufferTexture2D ) return;
112     bind();
113     glFramebufferTexture2D(static_cast<GLenum>(__target), GL_DEPTH_ATTACHMENT, static_cast<GLenum>(tex.target()), tex.handle(), level);
114   }
115 
116   void
117   attach_depth_rb(const renderbuffer &rb) noexcept
118   {
119     if ( !__h || !glFramebufferRenderbuffer ) return;
120     bind();
121     glFramebufferRenderbuffer(static_cast<GLenum>(__target), GL_DEPTH_ATTACHMENT, GL_RENDERBUFFER, rb.handle());
122   }
123 
124   void
125   attach_depth_stencil_rb(const renderbuffer &rb) noexcept
126   {
127     if ( !__h || !glFramebufferRenderbuffer ) return;
128     bind();
129     glFramebufferRenderbuffer(static_cast<GLenum>(__target), GL_DEPTH_STENCIL_ATTACHMENT, GL_RENDERBUFFER, rb.handle());
130   }
131 
132   GLenum
133   check() const
134   {
135     if ( !__h || !glCheckFramebufferStatus ) throw except::logic_error("framebuffer: empty handle");
136     bind();
137     return glCheckFramebufferStatus(static_cast<GLenum>(__target));
138   }
139 
140   void
141   require_complete() const
142   {
143     GLenum s = check();
144     if ( s != GL_FRAMEBUFFER_COMPLETE ) throw except::library_error("framebuffer: not complete");
145   }
146 
147   void
148   debug_label(const char *name) noexcept
149   {
150     if ( __h && glObjectLabel && name ) glObjectLabel(0x8D40 /* GL_FRAMEBUFFER */, __h, -1, name);
151   }
152 };
