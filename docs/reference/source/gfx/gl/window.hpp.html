25 class owned_display_base
26 {
27   using __display_t = platform::display;
28 
29 protected:
30   micron::aligned_storage_t<sizeof(__display_t), alignof(__display_t)> __dpy_storage;
31   bool __owns_dpy = false;
32 
33   struct own_tag {
34   };
35 
36   owned_display_base() noexcept = default;
37 
38   explicit owned_display_base(own_tag)
39   {
40     new (micron::addr(__dpy_storage)) __display_t();
41     __owns_dpy = true;
42   }
43 
44   owned_display_base(const owned_display_base &) = delete;
45   owned_display_base &operator=(const owned_display_base &) = delete;
46 
47   ~owned_display_base()
48   {
49     if ( __owns_dpy ) __owned_ref().~__display_t();
50   }
51 
52   __display_t &
53   __owned_ref() noexcept
54   {
55     return *reinterpret_cast<__display_t *>(&__dpy_storage);

64   static platform::x11_visual_choice
65   __x11_pick(platform::x11_display_t &xd, platform::x11_lib_t &, void *user) noexcept
66   {
67     auto *h = static_cast<context_hints *>(user);
68     glx_context_t tmp;
69     glx_fbconfig_request_t fbreq;
70     fbreq.depth_size = h->depth_bits;
71     fbreq.stencil_size = h->stencil_bits;
72     fbreq.samples = h->ms_samples;
73     fbreq.double_buffer = h->double_buffer;
74     tmp.pick_visual(glx_lib(), xd, fbreq);
75     platform::x11_visual_choice c{};
76     c.visual = tmp.take_visual();
77     c.extra = reinterpret_cast<void *>(tmp.fbconfig());
78     return c;
79   }
80

99   static platform::window_hooks
100   __build_hooks(const context_hints &h) noexcept

111   }
