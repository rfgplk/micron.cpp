35   void
36   __build_x11()
37   {
38     glx_context_request_t cr;
39     cr.major = __hints.major;
40     cr.minor = __hints.minor;
41     cr.core_profile = __hints.core_profile;
42     cr.forward_compat = __hints.forward_compat;
43     cr.debug = __hints.debug;
44     if ( __hints.share_with ) cr.share = __hints.share_with->__glx.ctx();
45 
46     auto *xd = __win->dpy().as_x11();
47     if ( !xd ) throw except::logic_error("gl::context: x11 window/display missing");
48     __glx.bind(glx_lib(), *xd, __win->glx_fbconfig());
49     __glx.create(cr);
50 
51     __has_swap_tear = glx_lib().has_extension(*xd, "GLX_EXT_swap_control_tear");
52   }
53 
54   void
55   __build_wayland()
56   {
57     auto *wd = __win->dpy().as_wayland();
58     auto *wwin = __win->as_wayland();
59     if ( !wd || !wwin ) throw except::logic_error("gl::context: wayland window/display missing");
60 
61     egl_config_request_t cr_cfg;
62     cr_cfg.depth_size = __hints.depth_bits;
63     cr_cfg.stencil_size = __hints.stencil_bits;
64     cr_cfg.samples = __hints.ms_samples;
65     egl_context_request_t cr_ctx;
66     cr_ctx.major = __hints.major;
67     cr_ctx.minor = __hints.minor;
68     cr_ctx.core_profile = __hints.core_profile;
69     cr_ctx.forward_compat = __hints.forward_compat;
70     cr_ctx.debug = __hints.debug;
71     if ( __hints.share_with ) cr_ctx.share = __hints.share_with->__egl.ctx();
72 
73     __egl.open(egl_lib(), EGL_PLATFORM_WAYLAND_EXT, wd->display());
74     __egl.choose_config(cr_cfg);
75     __egl.create_context(cr_ctx);
76 
77     __egl.create_window_surface(__win->wl_native());
78 
79     __has_swap_tear = egl_lib().has_display_extension(__egl.display(), "EGL_EXT_swap_control_tear");
80   }

90   static void
91   __on_window_resize(void *data, i32 w, i32 h) noexcept
92   {
93     auto *self = static_cast<context *>(data);
94     if ( self->__loaded ) {
95 
96       (void)self->make_current();
97     }
98     if ( glViewport ) glViewport(0, 0, w, h);
99   }

102   ~context()
103   {
104

106   }

108   context(window &w, const context_hints &h = {}) : __win(&w), __hints(h)
109   {
110     switch ( w.dpy().backend() ) {
111     case backend_tag_t::x11:
112       __build_x11();
113       break;
114     case backend_tag_t::wayland:
115       __build_wayland();
116       break;
117     case backend_tag_t::none:
118     default:
119       throw except::logic_error("gl::context: window's display has no backend");
120     }
121 
122     __win->set_internal_resize_cb(&__on_window_resize, this);
123   }

125   context(window &w, bool make_current_now, const context_hints &h = {}) : context(w, h)
126   {
127     if ( make_current_now && !make_current() ) throw except::logic_error("gl::context: make_current() failed");
128   }

154   bool
155   make_current()
156   {
157     bool ok = false;
158     switch ( __win->dpy().backend() ) {
159     case backend_tag_t::x11:
160       ok = __glx.make_current(static_cast<GLXDrawable>(__win->as_x11()->window()));
161       break;
162     case backend_tag_t::wayland:
163       ok = __egl.make_current();
164       break;
165     default:
166       ok = false;
167       break;
168     }
169     if ( ok && !__loaded ) {
170 
171       __current_for_thunk = this;
172       __load_core_4_6(__get_proc_thunk);
173       __loaded = true;
174       if ( __hints.debug ) install_default_debug_callback();
175     }
176     return ok;
177   }
178 
179   void
180   release() noexcept
181   {
182     if ( __win->dpy().backend() == backend_tag_t::x11 )
183       __glx.release();
184     else if ( __win->dpy().backend() == backend_tag_t::wayland )
185       __egl.release();
186   }
187 
188   void
189   swap() noexcept
190   {
191     switch ( __win->dpy().backend() ) {
192     case backend_tag_t::x11:
193       if ( auto *xw = __win->as_x11() ) __glx.swap(static_cast<GLXDrawable>(xw->window()));
194       break;
195     case backend_tag_t::wayland:
196       __egl.swap();
197 
198       if ( auto *wd = __win->dpy().as_wayland() ) wd->flush();
199       break;
200     default:
201       break;
202     }
203   }
204 
205   void
206   set_swap_interval(int interval) noexcept
207   {
208     if ( interval < 0 && !__has_swap_tear ) interval = -interval;
209     if ( __win->dpy().backend() == backend_tag_t::x11 ) {
210       if ( auto *xw = __win->as_x11() ) __glx.set_swap_interval(static_cast<GLXDrawable>(xw->window()), interval);
211     } else if ( __win->dpy().backend() == backend_tag_t::wayland ) {
212       __egl.set_swap_interval(interval);
213     }
214   }

216   void *
217   proc_address(const char *name) noexcept
218   {
219     switch ( __win->dpy().backend() ) {
220     case backend_tag_t::x11:
221       return __glx.proc_address(name);
222     case backend_tag_t::wayland:
223       return __egl.proc_address(name);
224     default:
225       return nullptr;
226     }
