22 struct atomic_flag {
23   atomic_token<bool> tk;
24 
25   constexpr atomic_flag() noexcept : tk(false) { }
26 
27   atomic_flag(const atomic_flag &) = delete;
28   atomic_flag &operator=(const atomic_flag &) = delete;
29 
30   bool
31   test(memory_order order = memory_order::seq_cst) const noexcept
32   {
33     return tk.get(order);
34   }
35 
36   bool
37   test_and_set(memory_order order = memory_order::seq_cst) noexcept
38   {
39     return tk.swap(true, order);
40   }
41 
42   void
43   ttas(memory_order success, memory_order failure) noexcept
44   {
45     __pause_backoff bo;
46     for ( ;; ) {
47       while ( tk.get(memory_order::acquire) ) bo.relax();
48       bool exp = ATOMIC_OPEN;
49       if ( tk.compare_exchange_weak(exp, ATOMIC_LOCKED, success, failure) ) break;
50       bo.relax();
51     }
52   }
53 
54   void
55   clear(memory_order order = memory_order::seq_cst) noexcept
56   {
57     tk.store(false, order);
58   }
59 
60   void
61   wait(bool old, [[maybe_unused]] memory_order order = memory_order::seq_cst) const noexcept
62   {
63     while ( tk.get(memory_order::relaxed) == old ) micron::yield();      // relaxed poll + yield, not a bare busy-spin
64   }
65 };
66 };      // namespace micron
