60 template<typename T>
61 __attribute__((always_inline)) inline bool
62 __clang_i386_atomic_compare_exchange(T *ptr, T *expected, T desired) noexcept
63 {
64   auto *word = reinterpret_cast<volatile __clang_i386_atomic_word *>(ptr);
65   const __clang_i386_atomic_word wanted = __clang_i386_atomic_bits(*expected);
66   const __clang_i386_atomic_word observed = __sync_val_compare_and_swap(word, wanted, __clang_i386_atomic_bits(desired));
67   if ( observed == wanted ) return true;
68   *expected = __clang_i386_atomic_value<T>(observed);
69   return false;
70 }

74 template<bool ReturnOld, __clang_i386_atomic_op Op, typename T>
75 __attribute__((always_inline)) inline T
76 __clang_i386_atomic_rmw(T *ptr, T value) noexcept
77 {
78   auto *word = reinterpret_cast<volatile __clang_i386_atomic_word *>(ptr);
79   const __clang_i386_atomic_word operand = __clang_i386_atomic_bits(value);
80   __clang_i386_atomic_word old = __sync_val_compare_and_swap(word, 0, 0);
81   for ( ;; ) {
82     __clang_i386_atomic_word next;
83     if constexpr ( Op == __clang_i386_atomic_op::add )
84       next = old + operand;
85     else if constexpr ( Op == __clang_i386_atomic_op::sub )
86       next = old - operand;
87     else if constexpr ( Op == __clang_i386_atomic_op::bit_and )
88       next = old & operand;
89     else if constexpr ( Op == __clang_i386_atomic_op::bit_xor )
90       next = old ^ operand;
91     else if constexpr ( Op == __clang_i386_atomic_op::bit_or )
92       next = old | operand;
93     else
94       next = ~(old & operand);
95     const __clang_i386_atomic_word observed = __sync_val_compare_and_swap(word, old, next);
96     if ( observed == old ) return __clang_i386_atomic_value<T>(ReturnOld ? old : next);
97     old = observed;
98   }
99 }

102 template<typename T>
103 constexpr __attribute__((always_inline)) inline T
104 load(const T *ptr, i32 memorder)
105 {
106 #if defined(__clang__) && defined(__micron_arch_x86)
107   if constexpr ( sizeof(T) == sizeof(__clang_i386_atomic_word) ) return __clang_i386_atomic_load(ptr);
108 #endif
109 #if defined(__clang__)
110   if constexpr ( __is_enum(T) ) {
111     T value;
112     __atomic_load(ptr, &value, memorder);
113     return value;
114   } else
115 #endif
116     return __atomic_load_n(ptr, memorder);
117 }
