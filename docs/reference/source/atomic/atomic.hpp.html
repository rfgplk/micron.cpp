45 constexpr memory_order
46 __cas_failure_clamp(memory_order failure) noexcept
47 {
48   return failure == memory_order::release ? memory_order::relaxed : failure == memory_order::acq_rel ? memory_order::acquire : failure;
49 }

51 // derive the failure order from the success order
52 constexpr memory_order
53 __cas_failure_from_success(memory_order success) noexcept
54 {
55   return success == memory_order::acq_rel ? memory_order::acquire : success == memory_order::release ? memory_order::relaxed : success;
56 }

307   void
308   lock_check()
309   {
310     // TTAS; backoff escalates to sched_yield rather than pausing forever
311     if ( tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED) ) return;
312     default_backoff bo;
313     for ( ;; ) {
314       while ( tk.get(memory_order::relaxed) == ATOMIC_LOCKED ) bo.relax();
315       if ( tk.compare_and_swap(ATOMIC_OPEN, ATOMIC_LOCKED) ) return;
316       bo.relax();
317     }
318   }

870     while ( !tk.compare_exchange_weak(old, old - 1, memory_order::acq_rel, memory_order::relaxed) );
871     return old;
872   }
873 };
