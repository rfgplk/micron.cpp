210   vector &
211   operator=(const vector &o)
212   {
213     if constexpr ( Sf == true ) {
214       if ( this == micron::addressof(const_cast<vector &>(o)) ) return *this;
215     }
216 
217     if ( o.length > __mem::capacity ) reserve(o.length);
218     if constexpr ( micron::is_trivially_copyable_v<T> ) {
219       __impl_container::copy_assign(__mem::memory, o.memory, o.length);
220     } else {
221       const size_type overlap = __mem::length < o.length ? __mem::length : o.length;
222       __impl_container::copy_assign(__mem::memory, o.memory, overlap);
223       if ( o.length > overlap )
224         __impl_container::copy(micron::addr(__mem::memory[overlap]), micron::addr(o.memory[overlap]), o.length - overlap);
225       if constexpr ( !micron::is_trivially_destructible_v<T> ) {
226         for ( size_type i = o.length; i < __mem::length; ++i ) __mem::memory[i].~T();
227       }
228     }
229     __mem::length = o.length;
230     return *this;
231   }

480   void
481   resize(size_type n)
482   {
483     if ( n == __mem::length ) return;
484     if ( n < __mem::length ) {
485       if constexpr ( !micron::is_trivially_destructible_v<T> ) {
486         for ( size_type i = n; i < __mem::length; ++i ) __mem::memory[i].~T();
487       }
488       __mem::length = n;
489       return;
490     }
491     if ( n > __mem::capacity ) reserve(n);
492     if constexpr ( micron::is_trivially_default_constructible<T>::value and micron::is_trivially_copyable<T>::value ) {
493       micron::byteset(__mem::memory + __mem::length, 0x0, (n - __mem::length) * sizeof(T));
494     } else {
495       T *f_ptr = __mem::memory;
496       for ( size_type i = __mem::length; i < n; i++ ) new (micron::addr(f_ptr[i])) T{};
497     }
498     __mem::length = n;
499   }

609   inline iterator
610   insert(size_type n, const T &val, size_type cnt)
611   {
612     if ( !__mem::length ) {
613       for ( size_type i = 0; i < cnt; ++i ) push_back(val);
614       return begin();
615     }
616     if ( !__mem::has_space(cnt) ) __grow_reserve(allocation_add_or_throw(__mem::length, cnt));
617     if constexpr ( Sf == true ) {
618       if ( n >= __mem::length ) [[unlikely]]
619         __chk::template __fail<except::library_error>("micron::vector insert(): out of allocated memory range.");
620     }
621     __impl_container::open_gap(__mem::memory, __mem::length, n, cnt);
622     __impl_container::fill_gap_copy(__mem::memory, __mem::length, n, cnt, val);      // rollback-safe gap fill
623     __mem::length += cnt;
624     return micron::addr(__mem::memory[n]);
625   }

693         __chk::template __fail<except::library_error>("micron::vector insert(): iterator out of range.");

696     if ( !__mem::has_space(cnt) ) __grow_reserve(allocation_add_or_throw(__mem::length, cnt));
697     __impl_container::open_gap(__mem::memory, __mem::length, p, cnt);
698     __impl_container::fill_gap_copy(__mem::memory, __mem::length, p, cnt, val);      // rollback-safe gap fill
699     __mem::length += cnt;
700     return micron::addr(__mem::memory[p]);

705   {
706     if ( !__mem::length ) {
707       push_back(val);
708       return begin();
709     }
710     if constexpr ( Sf == true ) {
711       if ( it > end() or it < begin() ) [[unlikely]]
712         __chk::template __fail<except::library_error>("micron::vector insert(): iterator out of range.");

715     if ( !__mem::has_space(1) ) __grow_reserve(allocation_add_or_throw(__mem::length, 1));
716     __impl_container::open_gap(__mem::memory, __mem::length, p, 1);
717     __impl_container::fill_gap_copy(__mem::memory, __mem::length, p, 1, val);      // rollback-safe gap fill
718     __mem::length++;

722   template<typename InputIt>
723   inline iterator
724   insert(iterator it, InputIt first, InputIt last)
725   {
726     const size_type cnt = static_cast<size_type>(last - first);
727     if ( cnt == 0 ) return it;
728     if ( !__mem::length ) {
729       for ( InputIt p = first; p != last; ++p ) push_back(*p);
730       return begin();
731     }
732     if constexpr ( Sf == true ) {
733       if ( it > end() or it < begin() ) [[unlikely]]
734         __chk::template __fail<except::library_error>("micron::vector insert(): iterator out of range.");
735     }
736     const size_type p = static_cast<size_type>(it - __mem::memory);
737     if ( !__mem::has_space(cnt) ) __grow_reserve(allocation_add_or_throw(__mem::length, cnt));
738     __impl_container::open_gap(__mem::memory, __mem::length, p, cnt);
739     InputIt q = first;
740     __impl_container::fill_gap(__mem::memory, __mem::length, p, cnt, [&](size_type idx) {
741       new (micron::addr(__mem::memory[idx])) T(*q);
742       ++q;
743     });
744     __mem::length += cnt;
745     return micron::addr(__mem::memory[p]);
746   }

757   template<typename U = T>
758   iterator
759   insert_sort(U &&val)
760   {
761     if ( __mem::length == 0 ) {
762       push_back(micron::move(val));
763       return begin();
764     }
765     if ( !__mem::has_space(1) ) __grow_reserve(allocation_add_or_throw(__mem::length, 1));
766 
767     size_type pos = 0;
768     for ( ; pos < __mem::length; ++pos )
769       if ( __mem::memory[pos] > val ) break;
770 
771     __impl_container::open_gap(__mem::memory, __mem::length, pos, 1);
772     __impl_container::fill_gap(__mem::memory, __mem::length, pos, 1,
773                                [&](size_type i) { new (micron::addr(__mem::memory[i])) T(micron::move(val)); });
774     ++__mem::length;
775     return micron::addr(__mem::memory[pos]);
776   }

840   {
841     if ( !__mem::length ) return;

843     for ( size_type r = 0; r < __mem::length; ++r ) {

851     }

853   }
