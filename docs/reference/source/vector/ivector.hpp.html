214   ivector(const std::initializer_list<T> &lst) : __mem(lst.size())
215   {
216     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_constructible_v<T> ) {

223         }

225         for ( usize j = 0; j < i; ++j ) __mem::memory[j].~T();
226         throw;
227       }
228 #else
229       for ( const T &value : lst ) {
230         new (micron::addr(__mem::memory[i])) T(value);
231         ++i;
232       }
233 #endif
234       __mem::length = lst.size();
235     } else {
236       usize i = 0;
237       for ( auto &&value : lst ) __mem::memory[i++] = value;
