101   struct __hold2 {
102     micron::fast_mutex *a;
103     micron::fast_mutex *b;      // nullptr when both args alias the same mutex
104 
105     [[gnu::always_inline]] __hold2(micron::fast_mutex &m1, micron::fast_mutex &m2) noexcept
106     {
107       if ( micron::addressof(m1) == micron::addressof(m2) ) {
108         a = micron::addressof(m1);
109         b = nullptr;
110         a->lock();
111       } else if ( reinterpret_cast<usize>(micron::addressof(m1)) < reinterpret_cast<usize>(micron::addressof(m2)) ) {
112         a = micron::addressof(m1);
113         b = micron::addressof(m2);
114         a->lock();
115         b->lock();
116       } else {
117         a = micron::addressof(m2);
118         b = micron::addressof(m1);
119         a->lock();
120         b->lock();
121       }
122     }
123 
124     [[gnu::always_inline]] ~__hold2() noexcept
125     {
126       if ( b ) b->unlock();
127       a->unlock();
128     }
129 
130     __hold2(const __hold2 &) = delete;
131     __hold2 &operator=(const __hold2 &) = delete;
132   };

240   convector(const std::initializer_list<T> &lst) : __mem(lst.size())
241   {
242     if constexpr ( micron::is_class_v<T> or !micron::is_trivially_constructible_v<T> ) {
243       size_type i = 0;
244 #if !defined(__micron_freestanding) || defined(__micron_eh)
245       try {
246         for ( const T &value : lst ) {
247           new (micron::addr(__mem::memory[i])) T(value);
248           ++i;
249         }
250       } catch ( ... ) {
251         for ( size_type j = 0; j < i; ++j ) __mem::memory[j].~T();
252         throw;
253       }
254 #else
255       for ( const T &value : lst ) {
256         new (micron::addr(__mem::memory[i])) T(value);
257         ++i;
258       }
259 #endif
260       __mem::length = lst.size();
261     } else {
262       size_type i = 0;
263       for ( auto &&value : lst ) __mem::memory[i++] = value;
264       __mem::length = lst.size();
265     }

317   convector &
318   operator=(const convector &o)
319   {
320     if ( this == micron::addressof(const_cast<convector &>(o)) ) return *this;
321     __hold2 __lock(__mtx, o.__mtx);
322     const T *const src = o.memory;
323     const size_type n = o.length;
324     if ( __mem::memory ) {
325       __unlocked_clear();
326       __mem::free();
327     }
328     __mem::realloc(n);
329     __impl_container::copy(__mem::memory, src, n);
330     __mem::length = n;
331     return *this;
332   }

705   template<typename F, bool Sf2 = Sf>
706     requires(sizeof(T) == sizeof(F))
707   inline convector &
708   append(const convector<F, Alloc, Sf2> &o)
709   {
710     __hold2 __lock(__mtx, o.__mtx);
711     const size_type on = o.length;
712     if ( on == 0 ) return *this;
713     if ( !__mem::has_space(on) ) __unlocked_grow(allocation_add_or_throw(__mem::length, on));
714     __impl_container::copy(__mem::memory + __mem::length, o.memory, on);
715     __mem::length += on;
716     return *this;
717   }

719   template<typename F, bool Sf2 = Sf>
720     requires(sizeof(T) == sizeof(F))
721   inline convector &
722   weld(convector<F, Alloc, Sf2> &&o)
723   {
724     if ( static_cast<const void *>(this) == static_cast<const void *>(micron::addressof(o)) ) return *this;
725     __hold2 __lock(__mtx, o.__mtx);
726     const size_type on = o.length;
727     if ( on == 0 ) return *this;
728     if ( !__mem::has_space(on) ) __unlocked_grow(allocation_add_or_throw(__mem::length, on));
729     __impl_container::move(__mem::memory + __mem::length, o.memory, on);
730     __mem::length += on;
731     o.length = 0;
732     return *this;
733   }

753   void
754   resize(size_type n)
755   {
756     __hold __lock(__mtx);
757 
758     if ( n == __mem::length ) return;
759     if ( n < __mem::length ) {
760       if constexpr ( !micron::is_trivially_destructible_v<T> ) {
761         for ( size_type i = n; i < __mem::length; ++i ) __mem::memory[i].~T();
762       }
763       __mem::length = n;
764       return;
765     }
766     if ( n >= __mem::capacity ) __unlocked_reserve(n);
767     if constexpr ( micron::is_trivially_default_constructible<T>::value and micron::is_trivially_copyable<T>::value ) {
768       micron::byteset(__mem::memory + __mem::length, 0x0, (n - __mem::length) * sizeof(T));
769     } else {
770       T *f_ptr = __mem::memory;
771       for ( size_type i = __mem::length; i < n; i++ ) new (micron::addr(f_ptr[i])) T{};
772     }
773     __mem::length = n;
774   }

1060   {
1061     __hold __lock(__mtx);
1062     if ( !__mem::length ) return;
1063     usize w = 0;
1064     for ( usize r = 0; r < __mem::length; ++r ) {
1065       if ( __mem::memory[r] == val ) continue;
1066       if ( w != r ) __mem::memory[w] = micron::move(__mem::memory[r]);
1067       ++w;
1068     }
1069     if ( w == __mem::length ) return;
1070     if constexpr ( !micron::is_trivially_destructible_v<T> ) {
1071       for ( usize i = w; i < __mem::length; ++i ) __mem::memory[i].~T();
1072     }
1073     __mem::length = w;
1074   }
