21 // reserved_vector
22 // table contiguous prefix of a bounded virtual address reservation
23 // growth commits pages and starts object lifetimes without moving existing objects
24 template<typename T> class reserved_vector
25 {
26   T *__data = nullptr;
27   usize __limit = 0, __mapped = 0, __committed = 0, __size = 0;
28 
29 public:
30   reserved_vector() noexcept = default;
31 
32   reserved_vector(const reserved_vector &other)
33   {
34     if ( !other.__data ) return;
35     if ( !initialize(other.__limit) || !try_copy(other) ) {
36       release();
37       exc<except::memory_error>("reserved_vector: copy allocation failed");
38     }
39   }
40 
41   reserved_vector &
42   operator=(const reserved_vector &other)
43   {
44     if ( !try_copy(other) ) exc<except::memory_error>("reserved_vector: copy allocation failed");
45     return *this;
46   }
47 
48   reserved_vector(reserved_vector &&other) noexcept
49       : __data(other.__data), __limit(other.__limit), __mapped(other.__mapped), __committed(other.__committed), __size(other.__size)
50   {
51     other.__data = nullptr;
52     other.__limit = other.__mapped = other.__committed = other.__size = 0;
53   }
54 
55   reserved_vector &
56   operator=(reserved_vector &&other) noexcept
57   {
58     if ( this == &other ) return *this;
59     release();
60     __data = other.__data;
61     __limit = other.__limit;
62     __mapped = other.__mapped;
63     __committed = other.__committed;
64     __size = other.__size;
65     other.__data = nullptr;
66     other.__limit = other.__mapped = other.__committed = other.__size = 0;
67     return *this;
68   }
69 
70   ~reserved_vector() { release(); }
71 
72   [[nodiscard]] bool
73   initialize(usize maximum) noexcept
74   {
75     if ( __data ) return maximum == __limit;
76     if ( !maximum || maximum > (~usize{ 0 } - (page_size - 1)) / sizeof(T) || alignof(T) > page_size ) return false;
77     const usize bytes = (maximum * sizeof(T) + page_size - 1) & ~(page_size - 1);
78     auto *memory = micron::mmap(nullptr, bytes, prot_none, map_private | map_anonymous, -1, 0);
79     if ( micron::mmap_failed(memory) ) return false;
80     __data = reinterpret_cast<T *>(memory);
81     __limit = maximum;
82     __mapped = bytes;
83     return true;
84   }
85 
86   [[nodiscard]] bool
87   try_reserve(usize count) noexcept
88   {
89     if ( count > __limit ) return false;
90     const usize bytes = (count * sizeof(T) + page_size - 1) & ~(page_size - 1);
91     if ( bytes <= __committed ) return true;
92     usize target = __committed ? __committed : page_size;
93     while ( target < bytes ) target = target > __mapped / 2 ? __mapped : target * 2;
94     auto *at = reinterpret_cast<addr_t *>(reinterpret_cast<byte *>(__data) + __committed);
95     auto *memory = micron::mmap(at, target - __committed, prot_read | prot_write, map_private | map_anonymous | map_fixed, -1, 0);
96     if ( micron::mmap_failed(memory) || memory != at ) return false;
97     __committed = target;
98     return true;
99   }
100 
101   [[nodiscard]] bool
102   try_resize(usize count) noexcept
103   {
104     if ( !try_reserve(count) ) return false;
105     usize built = __size;
106 #if !defined(__micron_freestanding) || defined(__micron_eh)
107     try {
108 #endif
109       for ( ; built < count; ++built ) new (__data + built) T{};
110 #if !defined(__micron_freestanding) || defined(__micron_eh)
111     } catch ( ... ) {
112       if constexpr ( !is_trivially_destructible_v<T> )
113         while ( built > __size ) __data[--built].~T();
114       return false;
115     }
116 #endif
117     if constexpr ( !is_trivially_destructible_v<T> )
118       for ( usize i = count; i < __size; ++i ) __data[i].~T();
119     __size = count;
120     return true;
121   }
122 
123   // A failed element assignment leaves a valid, partially assigned prefix.
124   [[nodiscard]] bool
125   try_copy(const reserved_vector &other) noexcept
126   {
127     if ( this == &other ) return true;
128     if ( !other.__data ) {
129       clear();
130       return true;
131     }
132     if ( !__data && !initialize(other.__limit) ) return false;
133     if ( !try_resize(other.__size) ) return false;
134 #if !defined(__micron_freestanding) || defined(__micron_eh)
135     try {
136 #endif
137       for ( usize i = 0; i < __size; ++i ) __data[i] = other.__data[i];
138 #if !defined(__micron_freestanding) || defined(__micron_eh)
139     } catch ( ... ) {
140       return false;
141     }
142 #endif
143     return true;
144   }
145 
146   void
147   clear() noexcept
148   {
149     if constexpr ( !is_trivially_destructible_v<T> )
150       for ( usize i = 0; i < __size; ++i ) __data[i].~T();
151     __size = 0;
152   }
153 
154   void
155   release() noexcept
156   {
157     clear();
158     if ( __data ) micron::munmap(reinterpret_cast<addr_t *>(__data), __mapped);
159     __data = nullptr;
160     __limit = __mapped = __committed = __size = 0;
161   }
162 
163   [[nodiscard]] usize
164   size() const noexcept
165   {
166     return __size;
167   }
168 
169   [[nodiscard]] usize
170   capacity() const noexcept
171   {
172     return micron::min(__committed / sizeof(T), __limit);
173   }
174 
175   [[nodiscard]] usize
176   max_size() const noexcept
177   {
178     return __limit;
179   }
180 
181   [[nodiscard]] T *
182   data() noexcept
183   {
184     return __data;
185   }
186 
187   [[nodiscard]] const T *
188   data() const noexcept
189   {
190     return __data;
191   }
192 
193   [[nodiscard]] T &
194   operator[](usize i) noexcept
195   {
196     return __data[i];
197   }
198 
199   [[nodiscard]] const T &
200   operator[](usize i) const noexcept
201   {
202     return __data[i];
203   }
204 };
