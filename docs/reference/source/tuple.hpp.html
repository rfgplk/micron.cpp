20 template<usize N, usize A = alignof(max_align_t)> class static_any
21 {
22   alignas(A) unsigned char storage_[N];
23   const void *type_id_ = nullptr;
24 
25   template<typename T>
26   static const void *
27   id() noexcept
28   {
29     static const int x = 0;
30     return &x;
31   }
32 
33   template<typename T>
34   static void
35   destroy_fn(void *p) noexcept
36   {
37     static_cast<T *>(p)->~T();
38   }
39 
40   template<typename T>
41   static void
42   copy_fn(void *dst, const void *src)
43   {
44     new (dst) T(*static_cast<const T *>(src));
45   }
46 
47   template<typename T>
48   static void
49   move_fn(void *dst, void *src)
50   {
51     new (dst) T(micron::move(*static_cast<T *>(src)));
52     destroy_fn<T>(src);
53   }
54 
55   void (*destroy_)(void *) noexcept = nullptr;
56   void (*copy_)(void *, const void *) = nullptr;
57   void (*move_)(void *, void *) = nullptr;
58 
59 public:
60   constexpr static_any() noexcept = default;
61 
62   constexpr ~static_any() { reset(); }
63 
64   constexpr static_any(const static_any &other)
65   {
66     if ( other.type_id_ ) {
67       other.copy_(storage_, other.storage_);
68       type_id_ = other.type_id_;
69       destroy_ = other.destroy_;
70       copy_ = other.copy_;
71       move_ = other.move_;
72     }
73   }
74 
75   constexpr static_any(static_any &&other) noexcept
76   {
77     if ( other.type_id_ ) {
78       other.move_(storage_, other.storage_);
79       type_id_ = other.type_id_;
80       destroy_ = other.destroy_;
81       copy_ = other.copy_;
82       move_ = other.move_;
83 
84       other.type_id_ = nullptr;
85       other.destroy_ = nullptr;
86       other.copy_ = nullptr;
87       other.move_ = nullptr;
88     }
89   }
90 
91   constexpr static_any &
92   operator=(const static_any &other)
93   {
94     if ( this != &other ) {
95       reset();
96       if ( other.type_id_ ) {
97         other.copy_(storage_, other.storage_);
98         type_id_ = other.type_id_;
99         destroy_ = other.destroy_;
100         copy_ = other.copy_;
101         move_ = other.move_;
102       }
103     }
104     return *this;
105   }
106 
107   constexpr static_any &
108   operator=(static_any &&other) noexcept
109   {
110     if ( this != &other ) {
111       reset();
112       if ( other.type_id_ ) {
113         other.move_(storage_, other.storage_);
114         type_id_ = other.type_id_;
115         destroy_ = other.destroy_;
116         copy_ = other.copy_;
117         move_ = other.move_;
118 
119         other.type_id_ = nullptr;
120         other.destroy_ = nullptr;
121         other.copy_ = nullptr;
122         other.move_ = nullptr;
123       }
124     }
125     return *this;
126   }
127 
128   template<typename T> constexpr static_any(T &&value) { emplace<micron::decay_t<T>>(micron::forward<T>(value)); }
129 
130   template<typename T, typename... Args>
131   constexpr T &
132   emplace(Args &&...args)
133   {
134     using U = micron::decay_t<T>;
135     static_assert(sizeof(U) <= N, "type too large for static_any");
136     static_assert(alignof(U) <= A, "alignment too strict for static_any");
137 
138     reset();
139 
140     new (storage_) U(micron::forward<Args>(args)...);
141 
142     type_id_ = id<U>();
143     destroy_ = &destroy_fn<U>;
144     copy_ = &copy_fn<U>;
145     move_ = &move_fn<U>;
146 
147     return *reinterpret_cast<U *>(storage_);
148   }
149 
150   constexpr void
151   reset() noexcept
152   {
153     if ( type_id_ ) {
154       destroy_(storage_);
155       type_id_ = nullptr;
156       destroy_ = nullptr;
157       copy_ = nullptr;
158       move_ = nullptr;
159     }
160   }
161 
162   constexpr bool
163   has_value() const noexcept
164   {
165     return type_id_ != nullptr;
166   }
167 
168   template<typename T>
169   constexpr T &
170   cast()
171   {
172     // static_assert(type_id_ == id<micron::decay_t<T>>());
173     return *reinterpret_cast<micron::decay_t<T> *>(storage_);
174   }
175 
176   template<typename T>
177   constexpr const T &
178   cast() const
179   {
180     // static_assert(type_id_ == id<micron::decay_t<T>>());
181     return *reinterpret_cast<const micron::decay_t<T> *>(storage_);
182   }
183 
184   template<typename T>
185   constexpr bool
186   is() const noexcept
187   {
188     return type_id_ == id<micron::decay_t<T>>();
189   }

198   // default-constructs then assigns, so reference members can never use these; without the
199   // exclusion braced { k, v } proxy pairs (map iterators) select this ctor and hard-error
200   template<typename C>
201     requires(((micron::is_same_v<T, C>) or (micron::is_same_v<F, C>)
202               or ((micron::is_convertible_v<T, C>) or micron::is_convertible_v<F, C>))
203              and !micron::is_reference_v<T> and !micron::is_reference_v<F>)
204   constexpr pair(std::initializer_list<C> &&lst) : a(T()), b(F())
205   {
206     auto it = lst.begin();
207 
208     if ( it != lst.end() ) {
209       a = static_cast<T>(*it);
210       ++it;
211     }
212 
213     if ( it != lst.end() ) {
214       b = static_cast<F>(*it);
215     }
216   }

221              and !micron::is_reference_v<T> and !micron::is_reference_v<F>)

778 template<typename... Ts, typename... Us, size_t... Is>
779 constexpr bool
780 tuple_lt_impl(const tuple<Ts...> &lhs, const tuple<Us...> &rhs, index_sequence<Is...>)
781 {
782   bool result = false;
783   bool done = false;
784 
785   (void)((done
786           || ((get<Is>(lhs) < get<Is>(rhs))   ? (result = true, done = true)
787               : (get<Is>(rhs) < get<Is>(lhs)) ? (done = true)
788                                               : false))
789          || ...);
790 
791   return result;
792 }
