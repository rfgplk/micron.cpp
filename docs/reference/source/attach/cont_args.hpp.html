17 inline constexpr u32 micron_cont_args_abi = 3u;
18 
19 inline constexpr u32 micron_cont_args_flag_graph = 1u << 0;        // ops names a real table
20 inline constexpr u32 micron_cont_args_flag_replace = 1u << 1;      // this call does not return
21 inline constexpr u32 micron_cont_args_flag_all = micron_cont_args_flag_graph | micron_cont_args_flag_replace;
22 
23 // _continue return codes
24 inline constexpr i64 cont_ok = 0;
25 inline constexpr i64 cont_efault = -1;            // null args
26 inline constexpr i64 cont_ebadargs = -2;          // abi/flags/span
27 inline constexpr i64 cont_enotattached = -3;      // the direct form was entered before _attach

29 struct micron_graph_ops {
30   i32 (*find)(void *ctx, const char *name, u32 *out_id);
31   i32 (*find_hash)(void *ctx, const byte *hash, u32 *out_id);
32   i32 (*addr_of)(void *ctx, u32 id, u64 *out_addr, u64 *out_len);
33   i32 (*replace)(void *ctx, u32 id, u64 new_addr, u64 new_len);
34   i32 (*resolve_extern)(void *ctx, u32 row);
35   i32 (*select)(void *ctx, const char *name, u32 id);
36   i32 (*resolve)(void *ctx, const char *name, u64 *out_addr);
37   void *ctx;
38 };

40 static_assert(sizeof(micron_graph_ops) == 8 * sizeof(void *), "micron_graph_ops must mirror mx_graph_ops");
41 
42 // 64 bytes, 8-aligned, no interior padding
43 struct micron_cont_args {
44   u32 abi;            // 0    == micron_cont_args_abi
45   u32 flags;          // 4    micron_cont_args_flag_*
46   u64 self_base;      // 8    where this continuation was placed
47   u64 self_span;      // 16
48   u64 host_base;      // 24   the image that called mx_continue()
49   u64 host_span;      // 32
50   u64 ops;            // 40   a const mx::mx_graph_ops *; valid iff micron_cont_args_flag_graph
51   u64 user;           // 48   the host's own argument, opaque to micron
52   u64 user_len;       // 56
53 };      // 64
54 
55 static_assert(sizeof(micron_cont_args) == 64, "micron_cont_args must be 64 bytes");
56 static_assert(alignof(micron_cont_args) == 8, "micron_cont_args must be 8-aligned");
57 static_assert(__builtin_offsetof(micron_cont_args, abi) == 0);
58 static_assert(__builtin_offsetof(micron_cont_args, flags) == 4);
59 static_assert(__builtin_offsetof(micron_cont_args, self_base) == 8);
60 static_assert(__builtin_offsetof(micron_cont_args, self_span) == 16);
61 static_assert(__builtin_offsetof(micron_cont_args, host_base) == 24);
62 static_assert(__builtin_offsetof(micron_cont_args, host_span) == 32);
63 static_assert(__builtin_offsetof(micron_cont_args, ops) == 40);
64 static_assert(__builtin_offsetof(micron_cont_args, user) == 48);
65 static_assert(__builtin_offsetof(micron_cont_args, user_len) == 56);
66 
67 }      // extern "C"

69 namespace micron

72 inline const micron_cont_args *__micron_cont_args = nullptr;
73 
74 inline __attribute__((always_inline)) bool
75 __cont_args_valid(const micron_cont_args *a) noexcept
76 {
77   if ( a == nullptr ) return false;
78   if ( a->abi != micron_cont_args_abi ) return false;
79   if ( (a->flags & ~micron_cont_args_flag_all) != 0 ) return false;
80   if ( (a->flags & micron_cont_args_flag_graph) != 0 && a->ops == 0 ) return false;
81   if ( a->self_span == 0 || a->host_span == 0 ) return false;
82   return true;
83 }
