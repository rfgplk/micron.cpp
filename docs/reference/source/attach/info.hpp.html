31 struct micron_attach_info {
32   u32 magic;        // == micron_attach_magc
33   u16 version;      // == micron_attach_version
34   u16 abi;          // == MICRON_ABI
35   u32 size;         // == sizeof(micron_attach_info) as the HOST compiled it
36   u32 flags;        // attach_f_*
37 
38   // [surplus (guests tdata) | host image | TCB]
39   // canary seeded from the live TCB; does not install the TP
40   int (*make_child_frame)(micron::__tls_frame *out) noexcept;
41   void (*free_frame)(const micron::__tls_frame *f) noexcept;
42 
43   // called instead of SYS_exit_group on any guest fatal path
44   void (*fatal)(int code) noexcept;
45 
46   int (*thread_atexit)(void (*dtor)(void *), void *obj) noexcept;
47   void (*run_thread_dtors)(void) noexcept;
48 
49   u64 surplus_size;          // host compile-time surplus S (guest self-check)
50   u64 host_image_memsz;      // host PT_TLS memsz (diagnostics)
51   u64 host_stack_lo;         // valid iff attach_f_have_stack
52   u64 host_stack_hi;
53   char **host_environ;      // may be null

60 inline const micron_attach_info *__micron_attach_info = nullptr;
61 
62 inline __attribute__((always_inline)) bool
63 __attach_info_valid(const micron_attach_info *i) noexcept
64 {
65   if ( i == nullptr ) return false;
66   if ( i->magic != micron_attach_magic ) return false;
67   if ( i->version != micron_attach_version ) return false;
68   if ( i->abi != static_cast<u32>(MICRON_ABI) ) return false;
69   // must cover every field the guest reads
70   if ( i->size < __builtin_offsetof(micron_attach_info, host_image_memsz) ) return false;
71   if ( i->make_child_frame == nullptr || i->free_frame == nullptr || i->fatal == nullptr ) return false;
72   if ( i->thread_atexit == nullptr || i->run_thread_dtors == nullptr ) return false;
73   if ( i->surplus_size == 0 ) return false;
74   return true;
75 }
