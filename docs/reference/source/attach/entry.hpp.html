47 // primary _attach() method
48 // NOTE: for external readers this may not make a lot of sense nor be technically useable, it's currently used mainly for internal .bmg
49 // modules; to make use of this you need a custom linking procedure or have your custom ELF jump into this without _start while running
50 // within the address space of a live _start()ed image
51 extern "C" __attribute__((used, retain, visibility("default"))) int
52 _attach(const micron_attach_info *info) noexcept
53 {
54   // NOTE: the bmg linker currently doesn't provide for frame unwinding, so we can't use exceptions at all; this is here for that reason
55   // only once we extend the linker remove this
56   static_assert(!micron::except::__use_exceptions, "modules must build freestanding without __micron_eh");
57 
58   if ( micron::__micron_attach_info != nullptr ) return attach_ealready;
59   if ( micron::__micron_module_detached ) return attach_ealready;      // spent: statics are destroyed, guards latched
60   if ( !micron::__attach_info_valid(info) ) return attach_ebadinfo;
61 
62   micron::__micron_attach_fatal = info->fatal;
63   micron::__micron_attach_thread_atexit = info->thread_atexit;
64   micron::__micron_attach_run_thread_dtors = info->run_thread_dtors;
65   micron::__micron_attach_info = info;
66 
67   if ( __micron_mem_init ) __micron_mem_init();
68   micron::__boot_threadpool();
69   if ( info->flags & attach_f_io_buffers ) micron::io::__boot_io_buffers();
70 
71   return attach_ok;
72 }
73 
74 extern "C" __attribute__((used, retain, visibility("default"))) void
75 _detach(void) noexcept
76 {
77   if ( micron::__micron_attach_info == nullptr || micron::__micron_module_detached ) return;
78   micron::__micron_module_detached = true;
79 
80   // WARNING: order matters
81   if ( micron::__global_parallelpool != nullptr ) {
82     micron::__global_parallelpool->stop_all();
83     micron::__global_parallelpool->join_all();
84   }
85   if ( micron::__global_threadpool != nullptr ) micron::__global_threadpool->join_all(500);
86 
87   micron::__run_thread_dtors();             // the attaching thread's thread_local dtors (LIFO)
88   micron::__drain_atexit_table();           // module __cxa_atexit registrations
89   micron::io::__shutdown_io_buffers();      // flush module stdio (no-op if never booted); never closes fd 0/1/2
90 
91   // WARNING: the hooks stay installed on purpose
92 }
