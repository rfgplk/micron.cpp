10 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
11 // mx entry descriptor
12 //
13 // what __micron_mxc hands a custom entry instead of (argc, argv, envp)
14 
15 extern "C" {
16 
17 inline constexpr u32 micron_mx_entry_abi = 1u;
18 
19 inline constexpr u32 micron_mx_entry_f_have_stack = 1u << 0;      // stack_lo/hi are real
20 inline constexpr u32 micron_mx_entry_f_have_tls = 1u << 1;        // tls_base is real
21 inline constexpr u32 micron_mx_entry_f_have_user = 1u << 2;       // user/user_len are real
22 inline constexpr u32 micron_mx_entry_f_all = micron_mx_entry_f_have_stack | micron_mx_entry_f_have_tls | micron_mx_entry_f_have_user;
23 
24 // 128 bytes, 8-aligned, no interior padding
25 struct alignas(8) micron_mx_entry_args {
26   u32 abi;              // 0    == micron_mx_entry_abi
27   u32 size;             // 4    == sizeof(micron_mx_entry_args) as the CRT compiled it
28   u32 flags;            // 8    micron_mx_entry_f_*
29   u32 argc;             // 12
30   u64 argv;             // 16   char **
31   u64 envp;             // 24   char **
32   u64 auxv;             // 32   const micron::auxv_t *
33   u64 self_base;        // 40   where the image was mapped (at_base); 0 when the loader gave none
34   u64 page_size;        // 48   at_pagesz
35   u64 stack_lo;         // 56   iff micron_mx_entry_f_have_stack
36   u64 stack_hi;         // 64
37   u64 tls_base;         // 72   the seated thread pointer; iff micron_mx_entry_f_have_tls
38   u64 user;             // 80   loader-opaque; iff micron_mx_entry_f_have_user
39   u64 user_len;         // 88
40   u64 reserved[4];      // 96   must be zero
41 };      // 128
42 
43 static_assert(sizeof(micron_mx_entry_args) == 128, "micron_mx_entry_args must be 128 bytes");
44 static_assert(alignof(micron_mx_entry_args) == 8, "micron_mx_entry_args must be 8-aligned");
45 static_assert(__builtin_offsetof(micron_mx_entry_args, abi) == 0);
46 static_assert(__builtin_offsetof(micron_mx_entry_args, size) == 4);
47 static_assert(__builtin_offsetof(micron_mx_entry_args, flags) == 8);
48 static_assert(__builtin_offsetof(micron_mx_entry_args, argc) == 12);
49 static_assert(__builtin_offsetof(micron_mx_entry_args, argv) == 16);
50 static_assert(__builtin_offsetof(micron_mx_entry_args, envp) == 24);
51 static_assert(__builtin_offsetof(micron_mx_entry_args, auxv) == 32);
52 static_assert(__builtin_offsetof(micron_mx_entry_args, self_base) == 40);
53 static_assert(__builtin_offsetof(micron_mx_entry_args, page_size) == 48);
54 static_assert(__builtin_offsetof(micron_mx_entry_args, stack_lo) == 56);
55 static_assert(__builtin_offsetof(micron_mx_entry_args, stack_hi) == 64);
56 static_assert(__builtin_offsetof(micron_mx_entry_args, tls_base) == 72);
57 static_assert(__builtin_offsetof(micron_mx_entry_args, user) == 80);
58 static_assert(__builtin_offsetof(micron_mx_entry_args, user_len) == 88);
59 static_assert(__builtin_offsetof(micron_mx_entry_args, reserved) == 96);
60 
61 }      // extern "C"

63 namespace micron
64 {
65 
66 inline __attribute__((always_inline)) bool
67 __mx_entry_args_valid(const micron_mx_entry_args *a) noexcept
68 {
69   if ( a == nullptr ) return false;
70   if ( a->abi != micron_mx_entry_abi ) return false;
71   if ( a->size < sizeof(micron_mx_entry_args) ) return false;
72   if ( (a->flags & ~micron_mx_entry_f_all) != 0 ) return false;
73   if ( a->argv == 0 || a->envp == 0 || a->auxv == 0 ) return false;
74   return true;
75 }
76 
77 inline __attribute__((always_inline)) bool
78 __mx_entry_has(const micron_mx_entry_args *a, u32 end_off) noexcept
79 {
80   return a != nullptr && a->size >= end_off;
81 }
