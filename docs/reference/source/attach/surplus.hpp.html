47 inline bool
48 __attach_host_tls_record(i64 surplus_tpoff, u64 image_block) noexcept
49 {
50   if constexpr ( __micron_tls_surplus == 0 ) {
51     return false;
52   } else {
53     if ( __micron_tls_inited ) return true;
54     __micron_surplus_tpoff = surplus_tpoff;
55     __micron_host_image_block = image_block;
56     __micron_tls_inited = true;
57     return true;
58   }
59 }

61 inline __attribute__((always_inline)) bool
62 __attach_host_tls_ready() noexcept
63 {
64   // whether the landlord can seat guests yet
65   return __micron_tls_inited && __micron_tls_surplus != 0;
66 }

95 inline constexpr u32 __micron_attach_max_frames = 4096;
96 inline byte *__attach_frames[__micron_attach_max_frames] = {};
97 inline u32 __attach_frame_hi = 0;
98 inline unsigned char __attach_frame_lock = 0;
99 
100 inline __attribute__((always_inline)) void
101 __attach_frames_lock() noexcept
102 {
103   while ( __atomic_exchange_n(&__attach_frame_lock, static_cast<unsigned char>(1), __ATOMIC_ACQUIRE) ) __cpu_pause();
104 }
105 
106 inline __attribute__((always_inline)) void
107 __attach_frames_unlock() noexcept
108 {
109   __atomic_store_n(&__attach_frame_lock, static_cast<unsigned char>(0), __ATOMIC_RELEASE);
110 }
111 
112 // NOTE: registration must happen before the frame is seeded, never after
113 inline void
114 __attach_frame_register(byte *base) noexcept
115 {
116   if ( base == nullptr ) return;
117   __attach_frames_lock();
118   for ( u32 i = 0; i < __micron_attach_max_frames; ++i ) {
119     if ( __attach_frames[i] == nullptr ) {
120       __attach_frames[i] = base;
121       if ( i >= __attach_frame_hi ) __attach_frame_hi = i + 1;
122       break;
123     }
124   }
125   // NOTE: a full registry is not fatal
126   __attach_frames_unlock();
127 }
128 
129 inline void
130 __attach_frame_unregister(byte *base) noexcept
131 {
132   if ( base == nullptr ) return;
133   __attach_frames_lock();
134   for ( u32 i = 0; i < __attach_frame_hi; ++i ) {
135     if ( __attach_frames[i] == base ) {
136       __attach_frames[i] = nullptr;
137       break;
138     }
139   }
140   __attach_frames_unlock();
141 }
