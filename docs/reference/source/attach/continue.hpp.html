24 extern "C" __attribute__((used, retain, visibility("default"))) i64
25 __micron_continuec(void *args) noexcept
26 {
27   static_assert(!micron::except::__use_exceptions, "continuations must build freestanding without __micron_eh");
28 
29   if ( args == nullptr ) return cont_efault;
30 
31   const micron_cont_args *a = static_cast<const micron_cont_args *>(args);
32   if ( !micron::__cont_args_valid(a) ) return cont_ebadargs;
33 
34   if ( (a->flags & micron_cont_args_flag_replace) == 0 && micron::__micron_attach_info == nullptr ) return cont_enotattached;
35 
36   micron::__micron_cont_args = a;
37 
38   return __micron_cont_user_entry(a);
39 }
