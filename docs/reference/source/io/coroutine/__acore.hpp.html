57 // cached fd classification, probed once per fd
58 inline constexpr i32 __fdc_nonblock = 1;      // O_NONBLOCK is set
59 inline constexpr i32 __fdc_seekable = 2;      // regular file or block device -- p{read,write} is legal, poll is not

61 [[gnu::always_inline]] inline i32
62 __fd_class(i32 fd, i32 &__c) noexcept
63 {
64   if ( __c < 0 ) {
65     i32 __v = 0;
66     const max_t __fl = posix::fcntl(fd_t{ fd }, posix::f_getfl);
67     if ( __fl >= 0 && (__fl & posix::o_nonblock) != 0 ) __v |= __fdc_nonblock;
68     stat_t __st{};
69     if ( posix::fstat(fd_t{ fd }, __st) == 0 && (posix::__impl::stat_is_reg(__st) || posix::__impl::stat_is_blk(__st)) )
70       __v |= __fdc_seekable;
71     __c = __v;
72   }
73   return __c;
74 }

76 [[gnu::always_inline]] inline bool
77 __fd_nonblocking(i32 fd, i32 &__c) noexcept
78 {
79   return (__fd_class(fd, __c) & (__fdc_nonblock | __fdc_seekable)) == __fdc_nonblock;
80 }

88 [[gnu::always_inline]] inline u64
89 __off_for(i32 fd, u64 off, i32 &__c) noexcept
90 {
91   return (off == static_cast<u64>(-1) || !__fd_seekable(fd, __c)) ? static_cast<u64>(-1) : off;

97 // suspend until fd is ready
98 [[nodiscard]] inline micron::task<i32>
99 __await_ready(i32 fd, u32 __mask)
100 {
101   const i32 __ev = co_await micron::coro::io::poll(fd, __mask);
102   if ( __ev < 0 ) co_return __ev;
103   if ( (static_cast<u32>(__ev) & micron::uring::poll_nval) != 0 ) co_return -error::bad_fd;
104   co_return 0;
105 }

107 // WARNING: *_some on a nonblocking pollable fd must __not__ go through the ring; io_uring arms
108 // its own internal poll and parks instead of returning -EAGAIN, which is right for read() and fatal
109 // for read_some()
110 [[nodiscard]] inline micron::task<max_t>
111 __read_once(i32 fd, void *p, usize n, u64 off, i32 &__nb)
112 {
113   const u64 __at = __off_for(fd, off, __nb);
114   if ( __fd_nonblocking(fd, __nb) )
115     co_return __at == static_cast<u64>(-1) ? posix::read(fd_t{ fd }, static_cast<byte *>(p), n)
116                                            : posix::pread(fd_t{ fd }, static_cast<byte *>(p), n, static_cast<off64_t>(__at));
117   const i32 r = co_await micron::coro::io::read(fd, p, static_cast<u32>(n), __at);
118   co_return static_cast<max_t>(r);
119 }

121 [[nodiscard]] inline micron::task<max_t>
122 __write_once(i32 fd, const void *p, usize n, u64 off, i32 &__nb)
123 {
124   const u64 __at = __off_for(fd, off, __nb);
125   if ( __fd_nonblocking(fd, __nb) )
126     co_return __at == static_cast<u64>(-1) ? posix::write(fd_t{ fd }, static_cast<const byte *>(p), n)
127                                            : posix::pwrite(fd_t{ fd }, static_cast<const byte *>(p), n, static_cast<off64_t>(__at));
128   const i32 w = co_await micron::coro::io::write(fd, p, static_cast<u32>(n), __at);
129   co_return static_cast<max_t>(w);

132 [[nodiscard]] inline micron::task<i32>
133 __retry_after(i32 __res, i32 fd, u32 __mask, i32 &__nb, u32 &__spun)
134 {
135   if ( __res == -error::no_buffer_space ) {        // ring full even after the submit path reaped
136     co_await micron::coro::reschedule_fair();      // pumps the ring, then yields behind the worker's other work
137     co_return 0;
138   }
139   if ( __res != -error::try_again ) co_return __res;
140   // WARNING: no progress bound must be set before the blocking/nonblocking split
141   if ( ++__spun > __ready_spins ) co_return __res;      // readiness keeps loopin; surrender
142   if ( !__fd_nonblocking(fd, __nb) ) {
143     co_await micron::coro::reschedule_fair();
144     co_return 0;
145   }
146   const i32 __r = co_await __await_ready(fd, __mask);
147   co_return __r < 0 ? __r : 0;
148 }

150 [[nodiscard]] inline micron::task<i32>
151 __retry_after2(i32 __res, i32 __in, i32 __out, u32 &__spun, u32 &__turn)
152 {
153   // WARNING: both ends must be polled regardless of O_NONBLOCK, unlike the single fd retry
154   if ( __res == -error::no_buffer_space ) {
155     co_await micron::coro::reschedule_fair();
156     co_return 0;
157   }
158   if ( __res != -error::try_again ) co_return __res;
159   const bool __probe_in = (__turn++ & 1u) == 0u;
160   if ( !__probe_in && ++__spun > __ready_spins ) co_return __res;      // both ends probed, nothing moved
161   const i32 __r = co_await __await_ready(__probe_in ? __in : __out, __probe_in ? micron::uring::poll_in : micron::uring::poll_out);
162   co_return __r < 0 ? __r : 0;
163 }

165 [[nodiscard]] inline micron::task<max_t>
166 __read_full(i32 fd, void *p, usize n, u64 off)
167 {
168   byte *dst = static_cast<byte *>(p);
169   usize got = 0;
170   i32 nb = __nb_unknown;
171   u32 spun = 0;
172   // NOTE: no seekability probe
173   while ( got < n ) {
174     usize want = n - got;
175     if ( want > __chunk_cap ) want = __chunk_cap;
176     const u64 at = off == static_cast<u64>(-1) ? off : off + got;
177     i32 r = co_await micron::coro::io::read(fd, dst + got, static_cast<u32>(want), at);
178     if ( r < 0 ) [[unlikely]] {
179       if ( r == -4 /*EINTR*/ ) continue;
180       const i32 a = co_await __retry_after(r, fd, micron::uring::poll_in, nb, spun);
181       if ( a == 0 ) continue;
182       co_return got ? static_cast<max_t>(got) : static_cast<max_t>(a);
183     }
184     if ( r == 0 ) break;
185     got += static_cast<usize>(r);
186     spun = 0;
187   }
188   co_return static_cast<max_t>(got);
189 }

191 [[nodiscard]] inline micron::task<max_t>
192 __write_full(i32 fd, const void *p, usize n, u64 off)
193 {
194   const byte *src = static_cast<const byte *>(p);
195   usize done = 0;
196   i32 nb = __nb_unknown;
197   u32 spun = 0;
198   while ( done < n ) {
199     usize want = n - done;
200     if ( want > __chunk_cap ) want = __chunk_cap;
201     const u64 at = off == static_cast<u64>(-1) ? off : off + done;
202     i32 w = co_await micron::coro::io::write(fd, src + done, static_cast<u32>(want), at);
203     if ( w < 0 ) [[unlikely]] {
204       if ( w == -4 /*EINTR*/ ) continue;
205       const i32 a = co_await __retry_after(w, fd, micron::uring::poll_out, nb, spun);
206       if ( a == 0 ) continue;
207       co_return done ? static_cast<max_t>(done) : static_cast<max_t>(a);
208     }
209     if ( w == 0 ) break;
210     done += static_cast<usize>(w);
211     spun = 0;
212   }
213   co_return static_cast<max_t>(done);
214 }

216 [[nodiscard]] inline micron::task<max_t>
217 __read_remainder(i32 fd, u64 off, micron::buffer &out)
218 {
219   stat_t st{};
220   usize total = 0;
221   if ( posix::fstat(fd_t{ fd }, st) == 0 && st.st_size > 0 ) total = static_cast<usize>(st.st_size);
222   if ( total > off ) {
223     const usize want = total - static_cast<usize>(off);
224     if ( out.size() < want ) out = micron::buffer(want);
225     max_t got = co_await __read_full(fd, out.data(), want, off);
226     co_return got;
227   }
228   usize cap = __probe_sz;
229   micron::buffer grow(cap);
230   usize got = 0;
231   i32 nb = __nb_unknown;
232   u32 spun = 0;
233   for ( ;; ) {
234     if ( got == cap ) {
235       cap *= 2;
236       micron::buffer bigger(cap);
237       micron::bytecpy(bigger.data(), grow.data(), got);
238       grow = micron::move(bigger);
239     }
240     usize want = cap - got;
241     if ( want > __chunk_cap ) want = __chunk_cap;
242     i32 r = co_await micron::coro::io::read(fd, reinterpret_cast<byte *>(grow.data()) + got, static_cast<u32>(want), off + got);
243     if ( r < 0 ) [[unlikely]] {
244       if ( r == -4 ) continue;
245       const i32 a = co_await __retry_after(r, fd, micron::uring::poll_in, nb, spun);
246       if ( a == 0 ) continue;
247       co_return static_cast<max_t>(a);
248     }
249     if ( r == 0 ) break;
250     got += static_cast<usize>(r);
251     spun = 0;
252   }
253   out = micron::move(grow);
254   co_return static_cast<max_t>(got);
255 }

260 template<typename T>
261   requires __readable_value<T>
262 [[nodiscard]] micron::task<max_t>
263 __write_value(i32 fd, u64 off, const T &v)
264 {
265   if constexpr ( __bulk_value<T> ) {
266     max_t w = co_await __write_full(fd, v.data(), v.size() * sizeof(typename T::value_type), off);
267     co_return w;
268   } else if constexpr ( __framed_value<T> ) {
269     max_t need = serialize::framed_size(v);
270     if ( need < 0 ) [[unlikely]]
271       co_return need;
272     micron::buffer out(static_cast<usize>(need));
273     max_t n = serialize::frame_into(reinterpret_cast<byte *>(out.data()), out.size(), v);
274     if ( n < 0 ) [[unlikely]]
275       co_return n;
276     max_t w = co_await __write_full(fd, out.data(), static_cast<usize>(n), off);
277     co_return w;
278   } else {
279     max_t w = co_await __write_full(fd, micron::addressof(v), sizeof(T), off);
280     co_return w;
281   }
282 }

284 template<typename T>
285 [[nodiscard]] inline max_t
286 __marshal_size(const T &v) noexcept
287 {
288   if constexpr ( __bulk_value<T> )
289     return static_cast<max_t>(v.size() * sizeof(typename T::value_type));
290   else if constexpr ( __framed_value<T> )
291     return serialize::framed_size(v);
292   else
293     return static_cast<max_t>(sizeof(T));

306 template<typename T>
307   requires __readable_value<T>
308 [[nodiscard]] micron::task<max_t>
309 __read_value_into(i32 fd, u64 off, T &out)
310 {
311   if constexpr ( __bulk_value<T> ) {
312     micron::buffer raw(0);
313     max_t got = co_await __read_remainder(fd, off, raw);
314     if ( got < 0 ) [[unlikely]]
315       co_return got;
316     const usize elems = static_cast<usize>(got) / sizeof(typename T::value_type);
317     if constexpr ( requires(T t, usize n) { t.reserve(n); } ) out.reserve(elems + 1);
318     if constexpr ( requires(const T &t) { t.max_size(); } ) {
319       if ( elems > out.max_size() ) [[unlikely]]
320         co_return -error::file_too_big;
321     }
322     __set_length(out, elems);
323     if ( elems != 0 ) micron::bytecpy(reinterpret_cast<byte *>(out.data()), raw.data(), elems * sizeof(typename T::value_type));
324     co_return got;
325   } else if constexpr ( __framed_value<T> ) {
326     micron::buffer raw(0);
327     max_t got = co_await __read_remainder(fd, off, raw);
328     if ( got < 0 ) [[unlikely]]
329       co_return got;
330     if ( got == 0 ) co_return 0;
331     max_t used = serialize::unframe_from(reinterpret_cast<const byte *>(raw.data()), static_cast<usize>(got), out);
332     if ( used < 0 ) [[unlikely]]
333       co_return used;
334     co_return got;
335   } else {
336     max_t got = co_await __read_full(fd, micron::addressof(out), sizeof(T), off);
337     if ( got < 0 ) [[unlikely]]
338       co_return got;
339     if ( static_cast<usize>(got) != sizeof(T) ) [[unlikely]]
340       co_return -error::io_error;      // a short object read is corruption
341     co_return got;
342   }
343 }
