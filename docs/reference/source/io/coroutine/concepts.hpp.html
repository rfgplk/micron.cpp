23 // direct awaitables (aio plumbing)
24 template<class A>
25 concept __plain_awaitable = requires(A a) {
26   { a.await_ready() } -> micron::convertible_to<bool>;
27   a.await_resume();
28 };
29 
30 template<class A>
31 concept __member_co_await = requires(A a) { static_cast<A &&>(a).operator co_await(); };
32 
33 template<class A>
34 concept __awaitable = __plain_awaitable<A> || __member_co_await<A>;
35 
36 namespace __impl
37 {
38 
39 template<class A> struct __await_result {
40 };
41 
42 template<__plain_awaitable A> struct __await_result<A> {
43   using type = decltype(micron::declval<A &>().await_resume());
44 };
45 
46 template<class A>
47   requires(__member_co_await<A> && !__plain_awaitable<A>)
48 struct __await_result<A> {
49   using type = decltype(micron::declval<A>().operator co_await().await_resume());
50 };
51 
52 };      // namespace __impl
53 
54 template<__awaitable A> using __await_result_t = typename __impl::__await_result<A>::type;
55 
56 template<class A>
57 concept __io_awaitable = __awaitable<A> && micron::convertible_to<__await_result_t<A>, max_t>;
58 
59 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
60 // the contracts
61 
62 template<class T>
63 concept async_readable = requires(T t, void *p, usize n) {
64   { t.read_some(p, n) } -> __io_awaitable;
65 };
66 
67 template<class T>
68 concept async_writable = requires(T t, const void *p, usize n) {
69   { t.write_some(p, n) } -> __io_awaitable;
70 };
71 
72 template<class T>
73 concept async_stream = async_readable<T> && async_writable<T>;
74 
75 template<class T>
76 concept async_seekable = async_readable<T> && requires(T t, u64 off, void *p, usize n) {
77   { t.read_at(off, p, n) } -> __io_awaitable;
78 };
79 
80 static_assert(async_readable<file>);
81 static_assert(async_writable<file>);
82 static_assert(async_seekable<file>);
83 static_assert(async_stream<fd_io>);
