15 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
16 // coro io waves
17 // (whole small files; batch per enter)
18 //
19 // normal perop route does an open + read + close as three separate io_uring_enters; for small
20 // files the overhead is expensive;
21 // a wave stages all three as one linked chain per file, W files at a time
22 //
23 // NOTE: this is for many small files per worker; a file that overflows the slab is reported partial
24 // and must be reread conventionally
25

33 #ifndef MICRON_CORO_WAVE_ITEMS
34 #define MICRON_CORO_WAVE_ITEMS 16u
35 #endif
36 #ifndef MICRON_CORO_WAVE_CAP
37 #define MICRON_CORO_WAVE_CAP (32u << 10)
38 #endif
39 
40 inline constexpr u32 wave_items = MICRON_CORO_WAVE_ITEMS;
41 inline constexpr usize wave_cap = MICRON_CORO_WAVE_CAP;
42 inline constexpr u32 wave_sqes = 3;              // open, read, close
43 inline constexpr usize wave_name_cap = 256;      // NAME_MAX + 1
44 
45 struct wave_result {
46   const char *name = nullptr;
47   const byte *data = nullptr;
48   usize len = 0;
49   i32 err = 0;
50   i32 close_err = 0;
51   bool unstaged = false;
52   bool partial = false;
53 };
54 
55 class wave
56 {
57   micron::sstr<wave_name_cap> __name[wave_items]{};
58   wave_result __res[wave_items]{};
59   i32 __slot[wave_items]{};
60   micron::uring::sqe __q[wave_items * wave_sqes]{};
61   micron::coro::__io_wop __nodes[wave_items * wave_sqes]{};
62   micron::coro::__io_wave __wv{};
63   byte *__slab = nullptr;
64   i32 __dirfd = -1;
65   i32 __rid = -1;          // ring the slots and sqes went to
66   u32 __n = 0;             // items pushed
67   u32 __staged = 0;        // items that reached the ring
68   u32 __seen = 0;          // items completed since construction
69   u32 __missed = 0;        // which overflowed the slab
70   u32 __stranded = 0;      // slots whose direct descriptor could not be evicted; should stay 0
71   bool __inert = false;
72 
73   static constexpr usize __slab_sz = static_cast<usize>(wave_items) * wave_cap;
74 
75   // WARNING: the slab must be prefaulted before the ring reads into it;
76   // io_uring cannot fault a page from the submit path
77   bool
78   __arm_slab() noexcept
79   {
80     if ( __slab != nullptr ) return true;
81     void *__p = micron::mmap(nullptr, __slab_sz, prot_read | prot_write, map_private | map_anonymous, -1, 0);
82     if ( micron::mmap_failed(reinterpret_cast<addr_t *>(__p)) ) return false;
83     __slab = reinterpret_cast<byte *>(__p);
84     usize __ps = micron::getpagesizelive();
85     if ( __ps == 0 || __ps > __slab_sz ) __ps = micron::page_size;      // probe refused; fall back to the arch default
86     if ( __ps == 0 ) __ps = 4096;
87     for ( usize __o = 0; __o < __slab_sz; __o += __ps ) __slab[__o] = 0;
88     return true;
89   }

91 public:
92   wave() noexcept
93   {
94     for ( u32 __i = 0; __i < wave_items; ++__i ) __slot[__i] = -1;
95   }

102   // WARNING: the kernel holds pointers into __nodes, so a live wave may not simply die
103   ~wave() noexcept
104   {
105     (void)micron::coro::io::__wave_abandon(__wv);
106     micron::coro::io::__wave_drain_cancel(__wv, __nodes, __staged * wave_sqes, __rid);
107     if ( !micron::coro::io::__wave_settle(__wv) ) return;      // resumed frame never reported; leak, do not corrupt
108     __reclaim_slots();
109     if ( __slab != nullptr ) micron::munmap(micron::ptr_cast<addr_t *>(__slab), __slab_sz);
110   }

166   [[nodiscard]] bool
167   push(const char *name) noexcept
168   {
169     if ( __n >= wave_items ) return false;
170     if ( name == nullptr || name[0] == 0 ) return false;
171     usize __l = 0;
172     while ( name[__l] != 0 )
173       if ( ++__l >= wave_name_cap ) return false;      // sstr<N> needs strlen < N, room for the nul
174     __name[__n] = micron::sstr<wave_name_cap>(name);
175     ++__n;
176     return true;
177   }
178 
179   [[nodiscard]] micron::task<i32>
180   run()
181   {
182     if ( __n == 0 ) co_return 0;
183     if ( !__arm_slab() ) co_return -error::out_of_memory;
184 
185     for ( u32 __i = 0; __i < __n; ++__i ) {
186       __res[__i] = wave_result{};
187       __res[__i].name = __name[__i].c_str();
188       __slot[__i] = -1;
189     }
190 
191     u32 __k = 0;
192     __staged = 0;
193     __rid = micron::coro::io::wave_ring_id();
194     const u32 __room = micron::coro::io::wave_room();
195     for ( u32 __i = 0; __i < __n; ++__i ) {
196       if ( __k + wave_sqes > __room ) break;
197       const i32 __s = micron::coro::io::wave_slot_acquire();
198       if ( __s < 0 ) break;
199       __slot[__i] = __s;
200 
201       micron::uring::sqe *__s0 = &__q[__k];
202       micron::uring::sqe *__s1 = &__q[__k + 1];
203       micron::uring::sqe *__s2 = &__q[__k + 2];
204 
205       micron::uring::prep_openat_direct(__s0, __dirfd, __name[__i].c_str(), posix::o_rdonly | posix::o_nofollow, 0, static_cast<u32>(__s));
206       __s0->flags |= micron::uring::sqe_io_link;
207 
208       micron::uring::prep_read(__s1, __s, __slab + static_cast<usize>(__i) * wave_cap, static_cast<u32>(wave_cap), 0);
209       micron::uring::sqe_set_fixed_file(__s1);
210       __s1->flags |= micron::uring::sqe_io_hardlink;
211 
212       micron::uring::prep_close_direct(__s2, static_cast<u32>(__s));
213 
214       for ( u32 __j = 0; __j < wave_sqes; ++__j ) {
215         __nodes[__k + __j].__idx = static_cast<u16>(__i);
216         __nodes[__k + __j].__step = static_cast<u8>(__j);
217         __nodes[__k + __j].__res = 0;      // a batch that never reaches the ring must not read as the last one's
218       }
219       __k += wave_sqes;
220       __staged = __i + 1;
221     }
222 
223     i32 __rc = 0;
224     if ( __k != 0 ) {
225       micron::coro::io::__wave_awaitable __aw{ __q, __nodes, &__wv, __k, 0 };
226       __rc = co_await __aw;
227     }
228 
229     __reclaim_slots();
230 
231     if ( __rc == 0 ) {
232       for ( u32 __i = 0; __i < __k; __i += wave_sqes ) {
233         const u32 __it = __nodes[__i].__idx;
234         const i32 __oerr = __nodes[__i].__res;
235         const i32 __rres = __nodes[__i + 1].__res;
236         __res[__it].close_err = __oerr < 0 ? 0 : __nodes[__i + 2].__res;
237         if ( __oerr < 0 ) {
238           __res[__it].err = __oerr;
239           continue;
240         }
241         if ( __rres < 0 ) {
242           __res[__it].err = __rres;
243           continue;
244         }
245         __res[__it].data = __slab + static_cast<usize>(__it) * wave_cap;
246         __res[__it].len = static_cast<usize>(__rres);
247         __res[__it].partial = static_cast<usize>(__rres) >= wave_cap;
248       }
249     } else {
250       for ( u32 __i = 0; __i < __staged; ++__i ) __res[__i].err = __rc;
251     }
252     for ( u32 __i = __staged; __i < __n; ++__i ) __res[__i].unstaged = true;
253 
254     __note();
255     __wv.__fin.store(1, micron::memory_order_release);
256     co_return __rc;
257   }
258 
259   void
260   clear() noexcept
261   {
262     __n = 0;
263     __staged = 0;
264   }
265 
266 private:
267   [[nodiscard]] bool
268   __slot_still_open(u32 __i) const noexcept
269   {
270     if ( __i >= __staged ) return false;      // never reached the ring, so nothing was opened
271     const u32 __b = __i * wave_sqes;
272     return __nodes[__b].__res >= 0 && __nodes[__b + 2].__res != 0;
273   }
274 
275   // WARNING: a slot goes back in the free bitmap only once it is known empty; hardlink keeps the close attached to a failed read, but a
276   // severed chain, a cancelled batch or a ring that went away can still leave one installed
277   void
278   __reclaim_slots() noexcept
279   {
280     for ( u32 __i = 0; __i < wave_items; ++__i ) {
281       if ( __slot[__i] < 0 ) continue;
282       if ( __slot_still_open(__i) && !micron::coro::io::wave_slot_close(__rid, __slot[__i]) ) {
283         __slot[__i] = -1;
284         ++__stranded;
285         continue;
286       }
287       micron::coro::io::wave_slot_release(__rid, __slot[__i]);
288       __slot[__i] = -1;
289     }
290   }
291 
292   void
293   __note() noexcept
294   {
295     __seen += __staged;
296     for ( u32 __i = 0; __i < __staged; ++__i )
297       if ( __res[__i].partial ) ++__missed;
298     if ( __seen >= wave_items && __missed * 2u > __seen ) __inert = true;
299   }
