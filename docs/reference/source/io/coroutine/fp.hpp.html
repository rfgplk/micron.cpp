22 template<typename Fn>
23   requires micron::invocable<Fn, file &>
24 [[nodiscard]] auto
25 with_file(io::path_t p, modes m, Fn fn)
26     -> micron::task<micron::option<__unit_if_void_t<micron::conditional_t<__impl::__is_task_v<micron::invoke_result_t<Fn, file &>>,
27                                                                           __impl::__task_inner_t<micron::invoke_result_t<Fn, file &>>,
28                                                                           micron::invoke_result_t<Fn, file &>>>,
29                                    io::error_t>>
30 {
31   using R0 = micron::invoke_result_t<Fn, file &>;
32   using RV = micron::conditional_t<__impl::__is_task_v<R0>, __impl::__task_inner_t<R0>, R0>;
33   using Ret = micron::option<__unit_if_void_t<RV>, io::error_t>;
34   file f = co_await coro::open_file(micron::move(p), m);
35   if ( !f.valid() ) [[unlikely]]
36     co_return Ret{ io::error_t(f.raw_fd()) };
37   if constexpr ( __impl::__is_task_v<R0> ) {
38     R0 t = fn(f);
39     if constexpr ( micron::is_void_v<RV> ) {
40       co_await micron::move(t);
41       co_return Ret{ unit_t{} };
42     } else {
43       RV v = co_await micron::move(t);
44       co_return Ret{ micron::move(v) };
45     }
46   } else {
47     if constexpr ( micron::is_void_v<R0> ) {
48       fn(f);
49       co_return Ret{ unit_t{} };
50     } else {
51       co_return Ret{ fn(f) };
52     }
53   }
54 }

77 template<typename C = micron::vector<micron::string>>
78   requires requires(C c, micron::string s) { c.push_back(micron::move(s)); }
79 [[nodiscard]] micron::task<micron::option<C, io::error_t>>
80 read_lines(io::path_t p)
81 {
82   using Ret = micron::option<C, io::error_t>;
83   file f = co_await coro::open_file(micron::move(p));
84   if ( !f.valid() ) [[unlikely]]
85     co_return Ret{ io::error_t(f.raw_fd()) };
86   C out{};
87   __aline_cursor cur = f.lines();
88   micron::string ln;
89   for ( ;; ) {
90     bool more = co_await cur.next(ln);
91     if ( !more ) break;
92     out.push_back(ln);
93   }
94   if ( i32 e = cur.error() ) [[unlikely]]
95     co_return Ret{ io::error_t(e) };
96   co_return Ret{ micron::move(out) };
97 }

99 template<typename C>
100   requires requires(C c, micron::string s) { c.push_back(micron::move(s)); }
101 [[nodiscard]] micron::task<max_t>
102 read_lines(io::path_t p, C &target)
103 {
104   file f = co_await coro::open_file(micron::move(p));
105   if ( !f.valid() ) [[unlikely]]
106     co_return f.raw_fd();
107   __aline_cursor cur = f.lines();
108   micron::string ln;
109   max_t count = 0;
110   for ( ;; ) {
111     bool more = co_await cur.next(ln);
112     if ( !more ) break;
113     target.push_back(ln);
114     ++count;
115   }
116   if ( i32 e = cur.error() ) [[unlikely]]
117     co_return e;
118   co_return count;
119 }

121 template<typename Fn>
122   requires(micron::is_invocable_v<Fn, micron::string &&> && is_string<micron::invoke_result_t<Fn, micron::string &&>>)
123 [[nodiscard]] micron::task<max_t>
124 interact(Fn fn)
125 {
126   micron::string in{};
127   micron::buffer chunk(__lines::chunk);
128   i32 nb = __impl::__nb_unknown;
129   u32 spun = 0;
130   for ( ;; ) {
131     i32 r = co_await micron::coro::io::read(0, chunk.data(), static_cast<u32>(chunk.size()));
132     if ( r < 0 ) [[unlikely]] {
133       if ( r == -4 /*EINTR*/ ) continue;
134       const i32 a = co_await __impl::__retry_after(r, 0, micron::uring::poll_in, nb, spun);
135       if ( a == 0 ) continue;
136       co_return static_cast<max_t>(a);
137     }
138     if ( r == 0 ) break;
139     in.append(reinterpret_cast<const char *>(chunk.data()), static_cast<usize>(r));
140     spun = 0;
141   }
142   auto out = fn(micron::move(in));
143   max_t w = co_await __impl::__write_full(1, out.c_str(), out.size() * sizeof(typename decltype(out)::value_type), static_cast<u64>(-1));
144   co_return w;
145 }
