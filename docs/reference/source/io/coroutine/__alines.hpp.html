25 class __aline_cursor
26 {
27   i32 __fd;
28   u64 __off;
29   micron::buffer __chunk;
30   usize __chunk_sz;
31   usize __pos = 0;
32   usize __valid = 0;
33   i32 __err = 0;
34   i32 __nb = __impl::__nb_unknown;
35   u32 __spun = 0;
36   bool __eof = false;
37   bool __stream;      // off == -1: unseekable fd (pipe/tty/stdin) -> kernel position
38 
39 public:
40   explicit __aline_cursor(i32 fd, u64 off = 0, usize chunk_sz = __lines::chunk)
41       : __fd(fd), __off(off), __chunk(chunk_sz ? chunk_sz : __lines::chunk), __chunk_sz(chunk_sz ? chunk_sz : __lines::chunk),
42         __stream(off == static_cast<u64>(-1))
43   {
44     if ( __stream ) __off = 0;
45   }
46 
47   __aline_cursor(const __aline_cursor &) = delete;
48   __aline_cursor &operator=(const __aline_cursor &) = delete;
49   __aline_cursor(__aline_cursor &&) = delete;
50   __aline_cursor &operator=(__aline_cursor &&) = delete;
51 
52   [[nodiscard]] micron::task<bool>
53   next(micron::string &out)
54   {
55     if ( __eof || __err != 0 ) co_return false;
56     out.set_size(0);
57     const char *base = reinterpret_cast<const char *>(__chunk.data());
58     for ( ;; ) {
59       for ( usize i = __pos; i < __valid; ++i ) {
60         if ( base[i] != '\n' ) continue;
61         if ( i > __pos ) out.append(base + __pos, i - __pos);
62         __pos = i + 1;
63         if ( out.size() && out[out.size() - 1] == '\r' ) out.set_size(out.size() - 1);
64         co_return true;
65       }
66       if ( __valid > __pos ) out.append(base + __pos, __valid - __pos);
67       __pos = __valid;
68       const u64 at = __stream ? static_cast<u64>(-1) : __off;
69       i32 r = co_await micron::coro::io::read(__fd, __chunk.data(), static_cast<u32>(__chunk_sz), at);
70       if ( r < 0 ) [[unlikely]] {
71         if ( r == -4 /*EINTR*/ ) continue;
72         const i32 a = co_await __impl::__retry_after(r, __fd, micron::uring::poll_in, __nb, __spun);
73         if ( a == 0 ) continue;
74         __err = a;
75         out.set_size(0);
76         co_return false;
77       }
78       if ( r == 0 ) {
79         __eof = true;
80         co_return !out.empty();      // final unterminated line, emitted once
81       }
82       __off += static_cast<u64>(r);
83       __pos = 0;
84       __valid = static_cast<usize>(r);
85       __spun = 0;
86     }
87   }
88 
89   [[nodiscard]] i32
90   error() const noexcept
91   {
92     return __err;
93   }
94 
95   [[nodiscard]] bool
96   at_eof() const noexcept
97   {
98     return __eof;
99   }
100 
101   [[nodiscard]] u64
102   offset() const noexcept
103   {
104     return __off - (__valid - __pos);
105   }
106 };
