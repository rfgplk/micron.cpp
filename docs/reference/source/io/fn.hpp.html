50 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%
51 // callable detection
52 namespace __fn_impl
53 {
54 struct __call_probe {
55   void operator()();      // participates in name lookup only
56 };
57 
58 template<typename T> struct __call_mixin: T, __call_probe {
59 };
60 
61 // resolves iff T contributes NO operator() of any kind
62 template<typename T>
63 concept __no_call_op = requires { &__call_mixin<T>::operator(); };
64 };      // namespace __fn_impl
65 
66 // anything that is semantically a callable
67 // (monomorphic and generic lambdas, functors with overloaded/templated operator(), function (pointer)s, member-fn pointers)
68 // NOTE: a final functor whose only operator() is templated/overloaded and that is not nullary-invocable is undetectable and still goes in
69 // the TC tier
70 template<typename T>
71 concept fn_like
72     = (micron::is_class_v<micron::remove_cvref_t<T>> && !micron::is_final_v<micron::remove_cvref_t<T>>
73        && !__fn_impl::__no_call_op<micron::remove_cvref_t<T>>)
74       || requires { &micron::remove_cvref_t<T>::operator(); }      // final monomorphic functors
75       || micron::is_function_v<micron::remove_reference_t<T>>
76       || (micron::is_pointer_v<micron::remove_cvref_t<T>> && micron::is_function_v<micron::remove_pointer_t<micron::remove_cvref_t<T>>>)
77       || micron::is_member_function_pointer_v<micron::remove_cvref_t<T>>
78       || micron::is_invocable_v<micron::remove_cvref_t<T>>;      // final nullary functors
79 
80 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
81 // signature deduction
82 // (monomorphic callables and plain functions only)
83 
84 template<typename Fn>
85 concept fn_deducible
86     = requires { &micron::remove_cvref_t<Fn>::operator(); } || micron::is_function_v<micron::remove_reference_t<Fn>>
87       || (micron::is_pointer_v<micron::remove_cvref_t<Fn>> && micron::is_function_v<micron::remove_pointer_t<micron::remove_cvref_t<Fn>>>);
88 
89 template<fn_deducible Fn> inline constexpr usize fn_arity_v = ::function_traits<micron::remove_cvref_t<Fn>>::arity;
90 
91 template<fn_deducible Fn> using fn_ret_t = typename ::function_traits<micron::remove_cvref_t<Fn>>::return_type;
92 
93 template<fn_deducible Fn> using fn_arg0_exact_t = typename ::function_traits<micron::remove_cvref_t<Fn>>::template arg_type<0>;
94 
95 template<fn_deducible Fn> using fn_arg0_t = micron::remove_cvref_t<fn_arg0_exact_t<Fn>>;

98 inline micron::option<max_t, io::error_t>
99 to_option(max_t n)
100 {
101   if ( n < 0 ) [[unlikely]]
102     return micron::option<max_t, io::error_t>{ io::error_t(static_cast<i32>(n)) };
