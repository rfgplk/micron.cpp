15 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
16 // line iterations for fp
17 // line policy:
18 //   a) terminator '\n'; one trailing '\r' before it is stripped ("a\r\r\n" -> "a\r")
19 //   b) the final unterminated line is emitted; a line with no terminator keeps its '\r'
20 //   c) a trailing '\n' does not produce an empty final line; empty file -> no lines
21 //   d) lines longer than the window grow the carry buffer (memory ~ longest line)
22

32 // chunked line cursor
33 class __line_cursor
34 {
35   i32 __fd;
36   micron::buffer __chunk;
37   usize __chunk_sz;
38   usize __pos = 0;        // consume offset into the window
39   usize __valid = 0;      // valid bytes in the window
40   micron::string __line;
41   i32 __err = 0;
42   bool __eof = false;
43 
44 public:
45   explicit __line_cursor(i32 fd, usize chunk_sz = __lines::chunk)
46       : __fd(fd), __chunk(chunk_sz ? chunk_sz : __lines::chunk), __chunk_sz(chunk_sz ? chunk_sz : __lines::chunk)
47   {
48   }
49 
50   ~__line_cursor()
51   {
52     if ( __fd >= 0 && __valid > __pos ) posix::lseek(__fd, -static_cast<posix::off64_t>(__valid - __pos), posix::seek_cur);
53   }
54 
55   // the cursor owns a side effect on a borrowed fd, so it is neither copyable nor movable
56   __line_cursor(const __line_cursor &) = delete;
57   __line_cursor &operator=(const __line_cursor &) = delete;
58   __line_cursor(__line_cursor &&) = delete;
59   __line_cursor &operator=(__line_cursor &&) = delete;
60 
61   bool
62   next()
63   {
64     if ( __eof || __err != 0 ) return false;
65     __line.set_size(0);
66     const char *base = reinterpret_cast<const char *>(__chunk.data());
67     for ( ;; ) {
68       for ( usize i = __pos; i < __valid; ++i ) {
69         if ( base[i] != '\n' ) continue;
70         if ( i > __pos ) __line.append(base + __pos, i - __pos);
71         __pos = i + 1;
72         if ( __line.size() && __line[__line.size() - 1] == '\r' ) __line.set_size(__line.size() - 1);
73         return true;
74       }
75       if ( __valid > __pos ) __line.append(base + __pos, __valid - __pos);
76       __pos = __valid;
77       max_t r = posix::read(__fd, __chunk.data(), __chunk_sz);
78       if ( r < 0 ) [[unlikely]] {
79         if ( -r == error::interrupted ) continue;
80         __err = static_cast<i32>(r);
81         __line.set_size(0);
82         return false;
83       }
84       if ( r == 0 ) {
85         __eof = true;
86         return !__line.empty();      // final unterminated line, emitted once ('\r' kept)
87       }
88       __pos = 0;
89       __valid = static_cast<usize>(r);
90     }
91   }
92 
93   // valid until the next next(); reused storage
94   const micron::string &
95   line() const noexcept
96   {
97     return __line;
98   }
99 
100   i32
101   error() const noexcept
102   {
103     return __err;
104   }
105 
106   bool
107   at_eof() const noexcept
108   {
109     return __eof;
110   }
111 };

145     {
146       if ( __r && !__r->__cur.next() ) __r = nullptr;
147       return *this;
148     }

152     {
153       return __r == nullptr;
154     }

156     bool
157     operator!=(sentinel) const noexcept
158     {
159       return __r != nullptr;
160     }
161   };

206 template<typename Fn>
207   requires micron::invocable<Fn, const micron::string &>
208 inline usize
209 __split_lines_mem(const byte *p, usize n, Fn &&fn)
210 {
211   const char *s = reinterpret_cast<const char *>(p);
212   micron::string ln;
213   usize count = 0;
214   usize start = 0;
215   for ( usize i = 0; i < n; ++i ) {
216     if ( s[i] != '\n' ) continue;
217     usize len = i - start;
218     if ( len && s[start + len - 1] == '\r' ) --len;
219     ln.set_size(0);
220     if ( len ) ln.append(s + start, len);
221     fn(static_cast<const micron::string &>(ln));
222     ++count;
223     start = i + 1;
224   }
225   if ( start < n ) {      // final unterminated line: '\r' kept
226     ln.set_size(0);
227     ln.append(s + start, n - start);
228     fn(static_cast<const micron::string &>(ln));
229     ++count;
230   }
231   return count;
232 }
