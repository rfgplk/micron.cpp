36 enum class tier : u8 {
37   none,          // ring dead -> every op returns -error::bad_syscall
38   plumbing,      // 5.1-5.5: readv/writev/fsync/fixed only; path-level surface unavailable
39   basic,         // 5.6+: op_read/write/openat/close/statx/fallocate/...; 2-submission path ops
40   fixed          // 5.19+: sparse fixed-file table -> 1-submission whole-file chains
41 };

43 inline constexpr u32 auto_slots = 0xffffffffu;
44 
45 //   bits 63..56  tag   1=oneshot 2=wave 3=chain 0xfe=reclaim(always ignored)
46 //   bits 55..40  gen   wave generation, bumped per wave -- wave N-1 leakage cannot count in wave N
47 //   bits 39..16  slot  per-wave item index (addresses 16M items; waves are ring-depth capped)
48 //   bits 15.. 8  step  0=statx 1=open 2=read 3=write 4=fsync 5=close

57 [[gnu::always_inline]] inline constexpr u64
58 oneshot(u64 seq) noexcept
59 {
60   return (static_cast<u64>(tag_oneshot) << 56) | (seq & 0x00ffffffffffffffull);

65 {
66   return (static_cast<u64>(tag_wave) << 56) | (static_cast<u64>(gen) << 40) | (static_cast<u64>(slot & 0xffffffu) << 16)

331   [[gnu::noinline]] bool
332   __realise_pool() noexcept
333   {
334     if ( __nbufs != 0 ) return true;
335     if ( __pool_want == 0 || __pool_want_sz == 0 || !__r.live() ) return false;
336     const u32 nb = __pool_want;
337     const usize total = static_cast<usize>(nb) * __pool_want_sz;
338     void *p = micron::mmap(nullptr, total, prot_read | prot_write, map_private | map_anonymous, -1, 0);
339     if ( uring::ring::__map_failed(p) ) {
340       __pool_want = 0;
341       return false;
342     }
343     __pool = reinterpret_cast<byte *>(p);
344     __pool_len = total;
345     micron::vector<uring::iovec> iov;
346     iov.reserve(nb);
347     for ( u32 i = 0; i < nb; i++ ) iov.push_back(uring::iovec{ __pool + static_cast<usize>(i) * __pool_want_sz, __pool_want_sz });
348     if ( __r.register_buffers(iov.data(), nb) != 0 ) {
349       __free_pool();
350       __pool_want = 0;
351       return false;
352     }
353     __nbufs = nb;
354     __buf_sz = __pool_want_sz;
355     __buf_free = nb >= 64 ? ~0ull : ((1ull << nb) - 1);
356     return true;
357   }

359   i32
360   acquire_buf(byte *&out, usize &cap, u16 &index) noexcept
361   {
362     if ( __nbufs == 0 ) [[unlikely]] {
363       if ( !__realise_pool() ) return -error::try_again;
364     }
365     if ( __buf_free == 0 ) return -error::try_again;
366     u32 i = static_cast<u32>(__builtin_ctzll(__buf_free));
367     __buf_free &= ~(1ull << i);
368     out = __pool + static_cast<usize>(i) * __buf_sz;
369     cap = __buf_sz;
370     index = static_cast<u16>(i);
371     return 0;
372   }

374   void
375   release_buf(u16 index) noexcept
376   {
377     if ( index < __nbufs ) __buf_free |= 1ull << index;
378   }

388   {

393       return static_cast<i32>(w * 64 + b);
394     }
395     return -error::try_again;
396   }

481 template<bool Write>
482 inline max_t
483 __rw(engine &eng, i32 fd, void *buf, usize cnt, u64 off) noexcept
484 {
485   if ( i32 e = __guard(eng) ) [[unlikely]]
486     return e;
487   byte *p = static_cast<byte *>(buf);
488   usize done = 0;
489   const bool positional = off != static_cast<u64>(-1);
490   while ( done < cnt ) {
491     u32 chunk = static_cast<u32>((cnt - done) < __max_chunk ? (cnt - done) : __max_chunk);
492     uring::sqe *s = eng.sqe();
493     if ( s == nullptr ) [[unlikely]]
494       return done ? static_cast<max_t>(done) : -error::bad_syscall;
495     u64 at = positional ? off + done : static_cast<u64>(-1);
496     if constexpr ( Write )
497       uring::prep_write(s, fd, p + done, chunk, at);
498     else
499       uring::prep_read(s, fd, p + done, chunk, at);
500     u64 ud = eng.next_seq();
501     s->user_data = ud;
502     eng.advance();
503     (void)eng.submit_wait(1);
504     i32 r = eng.await(ud);
505     if ( r < 0 ) [[unlikely]] {
506       if ( r == -error::interrupted ) continue;
507       if ( r == -error::try_again && done ) break;
508       return done ? static_cast<max_t>(done) : r;
509     }
510     if ( r == 0 ) break;
511     done += static_cast<usize>(r);
512   }
513   return static_cast<max_t>(done);
514 }

535 template<typename Dispatch>
536 inline i32
537 __run_wave(engine &eng, u8 tag, u16 gen, u32 expect, Dispatch &&d) noexcept
538 {
539   uring::ring &r = eng.ring();
540   if ( expect == 0 ) return 0;
541 
542   for ( ;; ) {
543     const u32 before = r.to_submit;
544     if ( before == 0 ) break;
545     if ( r.submit() < 0 ) break;
546     if ( r.to_submit >= before ) break;
547   }
548 
549   const u32 left = r.to_submit;
550   const u32 in_flight = left >= expect ? 0u : expect - left;
551   if ( in_flight == 0 ) return 0;
552 
553   u32 got = 0;
554   u32 idle = 0;
555   while ( got < in_flight ) {
556     if ( r.cq_ready() == 0 ) {
557       if ( r.submit_and_wait(1) < 0 ) return -error::bad_syscall;
558     }
559     u32 matched = 0;
560     u32 seen = 0;
561     if ( r.__cqe_mixed ) [[unlikely]] {
562       uring::cqe c{};
563       while ( r.peek_cqe(&c) ) {
564         seen++;
565         if ( __ud::tag_of(c.user_data) == tag && __ud::gen_of(c.user_data) == gen ) {
566           d(__ud::slot_of(c.user_data), __ud::step_of(c.user_data), c.res);
567           matched++;
568         }
569       }
570     } else {
571       seen = r.for_each_cqe([&](const uring::cqe &c) {
572         if ( __ud::tag_of(c.user_data) == tag && __ud::gen_of(c.user_data) == gen ) {
573           d(__ud::slot_of(c.user_data), __ud::step_of(c.user_data), c.res);
574           matched++;
575         }
576       });
577     }
578     got += matched;
579     if ( seen == 0 ) {
580       if ( ++idle > 3 ) break;
581     } else {
582       idle = 0;
583     }
584   }
585   return 0;
586 }

588 template<typename Fill, typename Dispatch>
589 inline u32
590 __stage_and_run(engine &eng, u32 count, u8 step, Fill &&fill, Dispatch &&d) noexcept
591 {
592   const u16 gen = eng.next_gen();
593   const u32 room = eng.sq_room();
594   u32 staged = 0;
595   for ( u32 i = 0; i < count && staged < room; i++ ) {
596     uring::sqe *s = eng.stage(staged);
597     if ( s == nullptr ) break;
598     if ( !fill(s, i) ) continue;
599     s->user_data = __ud::wave(gen, i, step);
600     staged++;
601   }
602   if ( staged == 0 ) return 0;
603   eng.publish(staged);
604   (void)__run_wave(eng, __ud::tag_wave, gen, staged, d);
605   return staged;
606 }

1892   const usize n = specs.size();
1893   out.reserve(n);
1894   if ( n == 0 ) return out;
1895   if ( !available(eng) ) {
1896     for ( usize i = 0; i < n; i++ ) out.push_back(Opt{ io::error_t(-error::bad_syscall) });
1897     return out;
1898   }
1899 
1900   micron::vector<__impl::__bslot> st;
1901   st.reserve(n);
1902   for ( usize i = 0; i < n; i++ ) {
1903     __impl::__bslot b{};
1904     b.want = specs[i].len;
1905     st.push_back(b);
1906   }
1907 
1908   usize base = 0;
1909   while ( base < n ) {
1910     const u32 W = __impl::__batch_window(eng, n - base);
1911 
1912     (void)__impl::__stage_and_run(
1913         eng, W, __ud::st_open,
1914         [&](uring::sqe *s, u32 i) {
1915           uring::prep_openat(s, posix::at_fdcwd, specs[base + i].path.c_str(),
1916                              static_cast<u32>(posix::o_wronly | posix::o_create | posix::o_trunc), posix::mode_file);
1917           return true;
1918         },
1919         [&](u32 i, u8, i32 res) {
1920           if ( i >= W ) return;
1921           if ( res < 0 )
1922             st[base + i].err = res;
1923           else
1924             st[base + i].fd = res;
1925         });
1926 
1927     for ( u32 i = 0; i < W; i++ ) {
1928       __impl::__bslot &s = st[base + i];
1929       if ( s.err == 0 && s.fd < 0 ) s.err = -error::io_error;
1930     }
1931 
1932     for ( u32 round = 0; round < 16; round++ ) {
1933       bool progressed = false;
1934       bool interrupted = false;
1935       const u32 staged = __impl::__stage_and_run(
1936           eng, W, __ud::st_write,
1937           [&](uring::sqe *s_, u32 i) {
1938             __impl::__bslot &s = st[base + i];
1939             if ( s.err != 0 || s.fd < 0 || s.done >= s.want ) return false;
1940             const u64 rem = s.want - s.done;
1941             const u32 chunk = rem > __impl::__max_chunk ? __impl::__max_chunk : static_cast<u32>(rem);
1942             uring::prep_write(s_, s.fd, specs[base + i].data + s.done, chunk, s.done);
1943             return true;
1944           },
1945           [&](u32 i, u8, i32 res) {
1946             if ( i >= W ) return;
1947             __impl::__bslot &s = st[base + i];
1948             if ( res < 0 ) {
1949               if ( res == -error::interrupted )
1950                 interrupted = true;
1951               else
1952                 s.err = res;
1953             } else if ( res == 0 ) {
1954               s.want = s.done;
1955             } else {
1956               s.done += static_cast<u64>(res);
1957               progressed = true;
1958             }
1959           });
1960       if ( staged == 0 || (!progressed && !interrupted) ) break;
1961     }
1962 
1963     for ( u32 i = 0; i < W; i++ ) {
1964       __impl::__bslot &s = st[base + i];
1965       if ( s.err == 0 && s.fd >= 0 && s.done < s.want ) s.err = -error::io_error;
1966     }
1967 
1968     (void)__impl::__stage_and_run(
1969         eng, W, __ud::st_close,
1970         [&](uring::sqe *s_, u32 i) {
1971           if ( st[base + i].fd < 0 ) return false;
1972           uring::prep_close(s_, st[base + i].fd);
1973           return true;
1974         },
1975         [](u32, u8, i32) { });
1976     for ( u32 i = 0; i < W; i++ ) st[base + i].fd = -1;
1977 
1978     base += W;
1979   }
1980 
1981   for ( usize k = 0; k < n; k++ ) {
1982     if ( st[k].err != 0 )
1983       out.push_back(Opt{ io::error_t(st[k].err) });
1984     else
1985       out.push_back(Opt{ static_cast<max_t>(st[k].done) });
1986   }
1987   return out;
1988 }

2002   sx.reserve(n);
2003   sx.set_size(n);
2004 
2005   micron::vector<i32> res;
2006   res.reserve(n);
2007   for ( usize i = 0; i < n; i++ ) res.push_back(-error::io_error);
2008
