45   max_t
46   __write_loop(const void *src, usize len)
47   {
48     const byte *p = static_cast<const byte *>(src);
49     usize done = 0;
50     while ( done < len ) {
51       max_t w = posix::write(__handle.fd, p + done, len - done);
52       if ( w < 0 ) [[unlikely]] {
53         if ( -w == error::interrupted ) continue;
54         if ( -w == error::try_again && !is_nonblock() ) continue;
55         return done ? static_cast<max_t>(done) : w;
56       }
57       if ( w == 0 ) break;
58       done += static_cast<usize>(w);
59     }
60     if ( done ) micron::zero(&sd);      // cached stat (st_size) is stale after a successful write
61     return static_cast<max_t>(done);
62   }

64   max_t
65   __read_loop(void *dst, usize len) const
66   {
67     byte *p = static_cast<byte *>(dst);
68     usize got = 0;
69     while ( got < len ) {
70       max_t r = posix::read(__handle.fd, p + got, len - got);
71       if ( r < 0 ) [[unlikely]] {
72         if ( -r == error::interrupted ) continue;
73         if ( -r == error::try_again && !is_nonblock() ) continue;
74         return got ? static_cast<max_t>(got) : r;
75       }
76       if ( r == 0 ) break;
77       got += static_cast<usize>(r);
78     }
79     return static_cast<max_t>(got);
80   }

92   template<typename T>
93     requires __readable_value<T>
94   max_t
95   __read_value(T &out)
96   {
97     if ( i32 __e = __check() ) [[unlikely]]
98       return __e;
99     if constexpr ( __bulk_value<T> ) {
100       stat<STAT_OVERRIDE>();
101       const usize total = static_cast<usize>(sd.st_size);
102       const posix::off64_t at = tell();
103       const usize remaining = (at >= 0 && total > static_cast<usize>(at)) ? total - static_cast<usize>(at) : 0;
104       if ( remaining == 0 ) {
105         // virtual files (/proc, /sys) fstat to size 0; grow chunk-by-chunk until EOF
106         max_t got = 0;
107         if constexpr ( sizeof(typename T::value_type) == 1 && requires(T t, typename T::value_type v) { t.push_back(v); } ) {
108           byte chunk[4096];
109           for ( ;; ) {
110             max_t r = posix::read(__handle.fd, chunk, sizeof(chunk));
111             if ( r < 0 ) [[unlikely]] {
112               if ( -r == error::interrupted ) continue;
113               return r;
114             }
115             if ( r == 0 ) break;
116             for ( max_t i = 0; i < r; ++i ) out.push_back(static_cast<typename T::value_type>(chunk[i]));
117             got += r;
118           }
119         }
120         return got;
121       }
122       const usize elems = remaining / sizeof(typename T::value_type);
123       if constexpr ( requires(T t, usize n) { t.reserve(n); } ) out.reserve(elems + 1);
124       // a fixed-capacity destination that cannot hold the remainder is an error, not a silent
125       // prefix: modify() reads this count back as the file's old size, so reporting a short
126       // read as a complete one makes it truncate away the tail it never saw.
127       if constexpr ( requires(const T &t) { t.max_size(); } ) {
128         if ( elems > out.max_size() ) [[unlikely]]
129           return -error::file_too_big;
130       }
131       max_t got = __read_loop(out.data(), elems * sizeof(typename T::value_type));
132       if ( got < 0 ) [[unlikely]]
133         return got;
134       if ( got > 0 ) __set_length(out, static_cast<usize>(got) / sizeof(typename T::value_type));
135       return got;
136     } else if constexpr ( __framed_value<T> ) {
137       const posix::off64_t at = tell();
138       if ( at < 0 ) [[unlikely]]
139         return at;
140       stat<STAT_OVERRIDE>();
141       if ( static_cast<usize>(sd.st_size) <= static_cast<usize>(at) ) return 0;
142       return read(out);      // non-throwing framed read
143     } else {
144       // trivially-copyable object: a short read is corruption
145       max_t got = __read_loop(static_cast<void *>(micron::addressof(out)), sizeof(T));
146       if ( got < 0 ) [[unlikely]]
147         return got;
148       if ( static_cast<usize>(got) != sizeof(T) ) [[unlikely]]
149         return -error::io_error;
150       return got;
151     }
152   }

154   // rebind __handle onto fname after a rename swapped a new inode in behind it. F_GETFL
155   // reports the access mode and status flags but never O_CREAT/O_TRUNC/O_EXCL (those are
156   // consumed by open and not retained), so reusing them cannot re-truncate the new file.
157   i32
158   __rebind(void)
159   {
160     const i32 fl = get_status_flags();
161     if ( fl < 0 ) [[unlikely]]
162       return fl;
163     const bool keep_cloexec = is_cloexec();      // an fd flag, not reported by F_GETFL
164     const i32 nfd = static_cast<i32>(posix::openat(posix::at_fdcwd, fname.c_str(), fl | (keep_cloexec ? posix::o_cloexec : 0), 0u));
165     if ( nfd < 0 ) [[unlikely]]
166       return nfd;
167     posix::close(__handle.fd);
168     __handle.fd = nfd;
169     micron::zero(&sd);      // new inode: every cached stat field is stale
170     return 0;
171   }

284   template<typename T>
285     requires((is_string<T> || is_contiguous_container<T>) && micron::is_trivially_copyable_v<typename T::value_type>)
286   T
287   read()
288   {
289     T out{};
290     if ( __check() != 0 ) [[unlikely]]
291       return out;
292     stat<STAT_OVERRIDE>();
293     const usize total = static_cast<usize>(sd.st_size);
294     const posix::off64_t at = tell();
295     const usize remaining = (at >= 0 && total > static_cast<usize>(at)) ? total - static_cast<usize>(at) : 0;
296     if ( remaining == 0 ) {
297       // virtual files (/proc, /sys) fstat to size 0; grow chunk-by-chunk until EOF
298       if constexpr ( sizeof(typename T::value_type) == 1 && requires(T t, typename T::value_type v) { t.push_back(v); } ) {
299         byte chunk[4096];
300         for ( ;; ) {
301           max_t r = posix::read(__handle.fd, chunk, sizeof(chunk));
302           if ( r < 0 && -r == error::interrupted ) continue;
303           if ( r <= 0 ) break;
304           for ( max_t i = 0; i < r; ++i ) out.push_back(static_cast<typename T::value_type>(chunk[i]));
305         }
306       }
307       return out;
308     }
309     const usize elems = remaining / sizeof(typename T::value_type);
310     if constexpr ( requires(T t, usize n) { t.reserve(n); } ) out.reserve(elems + 1);
311     if constexpr ( requires(const T &t) { t.max_size(); } ) {
312       if ( elems > out.max_size() ) [[unlikely]] {
313         exc<except::io_error>("micron::io::file::read: destination cannot hold the file");
314         return out;      // exc<> may return; never read past capacity
315       }
316     }
317     max_t got = __read_loop(out.data(), elems * sizeof(typename T::value_type));
318     if ( got < 0 ) [[unlikely]]      // EIO/EISDIR/EBADF etc
319       exc<except::io_error>("micron::io::file::read: read failed");
320     if ( got > 0 ) __set_length(out, static_cast<usize>(got) / sizeof(typename T::value_type));
321     return out;
322   }

428   // read-modify-write over the whole file
429   //   pure:     T -> T        f.modify([](micron::string s){ return transform(s); });
430   //   in-place: void(T&)      f.modify([](micron::string &s){ s += "\n"; });
431   // in-place rewrite, not crash-atomic
432   template<typename Fn>
433     requires(fn_deducible<Fn> && fn_arity_v<Fn> == 1 && micron::is_default_constructible_v<fn_arg0_t<Fn>> && __readable_value<fn_arg0_t<Fn>>
434              && ((micron::is_void_v<fn_ret_t<Fn>> && micron::same_as<fn_arg0_exact_t<Fn>, fn_arg0_t<Fn> &>)
435                  || micron::same_as<micron::remove_cvref_t<fn_ret_t<Fn>>, fn_arg0_t<Fn>>))
436   max_t
437   modify(Fn &&fn)
438   {
439     using T = fn_arg0_t<Fn>;
440     if ( i32 __e = __check() ) [[unlikely]]
441       return __e;
442     // O_APPEND forces every write to EOF, so the rewrite below would land past the old content
443     // and the trailing truncate would then cut the file down to the new length -- destroying
444     // both versions. There is no way to rewrite in place through an append-only handle.
445     if ( i32 fl = get_status_flags(); fl < 0 ) [[unlikely]]
446       return fl;
447     else if ( fl & posix::o_append ) [[unlikely]]
448       return -error::invalid_arg;
449     if ( posix::off64_t s = seek_to(0); s < 0 ) [[unlikely]]
450       return s;
451     T val{};
452     const max_t old_bytes = __read_value(val);      // empty file: default T, fn still applied
453     if ( old_bytes < 0 ) [[unlikely]]
454       return old_bytes;
455     if constexpr ( micron::is_void_v<fn_ret_t<Fn>> )
456       micron::forward<Fn>(fn)(val);
457     else
458       val = micron::forward<Fn>(fn)(micron::move(val));
459     const max_t need = __marshal_size(val);
460     if ( need < 0 ) [[unlikely]]
461       return need;
462     if ( posix::off64_t s = seek_to(0); s < 0 ) [[unlikely]]
463       return s;
464     max_t n = write(val);
465     if ( n < 0 ) [[unlikely]]
466       return n;
467     if ( n != need ) [[unlikely]]      // short write: old tail may survive past new content
468       return -error::io_error;
469     if ( n < old_bytes )
470       if ( i32 t = truncate(static_cast<posix::off64_t>(n)); t != 0 ) [[unlikely]]
471         return t;
472     return n;
473   }

475   // crash atomic modify via atomic_replace (tmp + fsync + rename + dir-fsync)
476   template<typename Fn>
477     requires(fn_deducible<Fn> && fn_arity_v<Fn> == 1 && micron::is_default_constructible_v<fn_arg0_t<Fn>> && __readable_value<fn_arg0_t<Fn>>
478              && ((micron::is_void_v<fn_ret_t<Fn>> && micron::same_as<fn_arg0_exact_t<Fn>, fn_arg0_t<Fn> &>)
479                  || micron::same_as<micron::remove_cvref_t<fn_ret_t<Fn>>, fn_arg0_t<Fn>>))
480   max_t
481   modify_atomic(Fn &&fn)
482   {
483     using T = fn_arg0_t<Fn>;
484     if ( i32 __e = __check() ) [[unlikely]]
485       return __e;
486     if ( posix::off64_t s = seek_to(0); s < 0 ) [[unlikely]]
487       return s;
488     T val{};
489     if ( max_t r = __read_value(val); r < 0 ) [[unlikely]]
490       return r;
491     if constexpr ( micron::is_void_v<fn_ret_t<Fn>> )
492       micron::forward<Fn>(fn)(val);
493     else
494       val = micron::forward<Fn>(fn)(micron::move(val));
495     if constexpr ( __bulk_value<T> ) {
496       if ( i32 r = atomic_replace(val); r != 0 ) [[unlikely]]
497         return r;
498       return static_cast<max_t>(val.size() * sizeof(typename T::value_type));
499     } else if constexpr ( __framed_value<T> ) {
500       max_t need = serialize::framed_size(val);
501       if ( need < 0 ) [[unlikely]]
502         return need;
503       micron::buffer out(static_cast<usize>(need));
504       max_t n = serialize::frame_into(out.data(), out.size(), val);
505       if ( n < 0 ) [[unlikely]]
506         return n;
507       if ( i32 r = atomic_replace(static_cast<const void *>(out.data()), static_cast<usize>(n)); r != 0 ) [[unlikely]]
508         return r;
509       return n;
510     } else {
511       if ( i32 r = atomic_replace(static_cast<const void *>(micron::addressof(val)), sizeof(T)); r != 0 ) [[unlikely]]
512         return r;
513       return static_cast<max_t>(sizeof(T));
514     }
515   }

521   {

534     }

542   {
543     if ( i32 __e = __check() ) [[unlikely]]

548       usize n = fn(win.data(), fp_window);
549       if ( n == 0 ) break;
550       if ( n > fp_window ) [[unlikely]]
551         return -error::invalid_arg;
552       max_t w = __write_loop(win.data(), n);
553       if ( w < 0 ) [[unlikely]]
554         return w;      // as in read_with: a byte count cannot express "and then it failed"
555       if ( static_cast<usize>(w) != n ) [[unlikely]]
556         return -error::io_error;
557       total += w;
558     }
559     return total;
560   }

562   // apply fn to every line of the file's remainder
563   template<typename Fn>
564     requires(line_fn<Fn> || micron::is_invocable_v<Fn, const micron::string &>)
565   max_t
566   each_line(Fn &&fn)
567   {
568     if ( i32 __e = __check() ) [[unlikely]]
569       return __e;
570     __line_cursor cur(__handle.fd);
571     max_t count = 0;
572     while ( cur.next() ) {
573       if constexpr ( line_fn<Fn> )
574         fn(cur.line().c_str(), cur.line().size());
575       else
576         fn(cur.line());
577       ++count;
578     }
579     if ( i32 e = cur.error() ) [[unlikely]]
580       return e;
581     return count;
582   }

584   // left fold over lines: fn(R, line) -> R with either line shape
585   template<typename R, typename Fn>
586     requires(micron::distinct<R, io::error_t> && (line_fold_raw<Fn, R> || line_fold_str<Fn, R>))
587   auto
588   fold_lines(R init, Fn &&fn) -> micron::option<R, io::error_t>
589   {
590     using Ret = micron::option<R, io::error_t>;
591     if ( i32 __e = __check() ) [[unlikely]]
592       return Ret{ io::error_t(__e) };
593     __line_cursor cur(__handle.fd);
594     while ( cur.next() ) {
595       if constexpr ( line_fold_raw<Fn, R> )
596         init = fn(micron::move(init), cur.line().c_str(), cur.line().size());
597       else
598         init = fn(micron::move(init), cur.line());
599     }
600     if ( i32 e = cur.error() ) [[unlikely]]
601       return Ret{ io::error_t(e) };
602     return Ret{ micron::move(init) };
603   }

653   // atomically replace the files contents
654   // NOTE: filesystem paths are trusted, so this does not defend against an attacker preplanting the temp path
655   i32
656   atomic_replace(const void *new_data, usize new_len)
657   {
658     if ( fname.empty() ) [[unlikely]]
659       return -error::invalid_arg;
660     if ( fname.size() + 16 >= max_name ) [[unlikely]]
661       return -error::name_too_long;
662 
663     micron::sstr<max_name> tmp_name(fname);
664     tmp_name += ".tmp.";
665     // u32 on purpose: linux pids fit, and u64 divmod emits libgcc calls on 32-bit targets
666     u32 pid = static_cast<u32>(posix::getpid());
667     char pid_buf[24];
668     usize pi = 0;
669     if ( pid == 0 ) {
670       pid_buf[pi++] = '0';
671     } else {
672       u32 p = pid;
673       while ( p ) {
674         pid_buf[pi++] = '0' + static_cast<char>(p % 10);
675         p /= 10;
676       }
677     }
678     for ( usize l = 0, r = pi - 1; l < r; ++l, --r ) {
679       char c = pid_buf[l];
680       pid_buf[l] = pid_buf[r];
681       pid_buf[r] = c;
682     }
683     pid_buf[pi] = '\0';
684     tmp_name += pid_buf;
685 
686     // preserve the original files permission bits
687     u32 keep_mode = posix::mode_file;
688     {
689       stat_t orig{};
690       if ( posix::exists(fname.c_str(), orig) ) keep_mode = static_cast<u32>(orig.st_mode) & 0777u;
691     }
692 
693     // create restrictively (0600)
694     posix::fd_t tmp{ (posix::openat(posix::at_fdcwd, tmp_name.c_str(),
695                                     posix::o_wronly | posix::o_create | posix::o_trunc | posix::o_cloexec | posix::o_nofollow, 0600u)) };
696     if ( !tmp ) [[unlikely]]
697       return tmp.fd < 0 ? tmp.fd : -error::io_error;
698 
699     usize written = 0;
700     while ( written < new_len ) {
701       max_t n = posix::write(tmp.fd, static_cast<const byte *>(new_data) + written, new_len - written);
702       if ( n < 0 && -n == error::interrupted ) continue;
703       if ( n <= 0 ) {
704         posix::close(tmp.fd);
705         posix::unlink(tmp_name.c_str());
706         return n < 0 ? static_cast<i32>(n) : -error::io_error;
707       }
708       written += static_cast<usize>(n);
709     }
710 
711     posix::fchmod(tmp.fd, keep_mode);
712 
713     if ( max_t fe = posix::fsync(tmp.fd); fe < 0 ) [[unlikely]] {
714       posix::close(tmp.fd);
715       posix::unlink(tmp_name.c_str());
716       return static_cast<i32>(fe);
717     }
718     if ( max_t ce = posix::close(tmp.fd); ce < 0 ) [[unlikely]] {
719       posix::unlink(tmp_name.c_str());
720       return static_cast<i32>(ce);
721     }
722 
723     if ( i32 r = static_cast<i32>(posix::rename(tmp_name.c_str(), fname.c_str())); r != 0 ) [[unlikely]] {
724       posix::unlink(tmp_name.c_str());
725       return r;
726     }
727 
728     // fsync the parent directory
729     {
730       char dirbuf[max_name];
731       const char *s = fname.c_str();
732       const usize n = fname.size();
733       usize cut = 0;
734       bool have = false;
735       for ( usize k = 0; k < n; ++k )
736         if ( s[k] == '/' ) {
737           cut = k;
738           have = true;
739         }
740       usize dl;
741       if ( !have ) {
742         dirbuf[0] = '.';
743         dl = 1;
744       } else if ( cut == 0 ) {
745         dirbuf[0] = '/';
746         dl = 1;
747       } else {
748         for ( usize k = 0; k < cut; ++k ) dirbuf[k] = s[k];
749         dl = cut;
750       }
751       dirbuf[dl] = '\0';
752       posix::fd_t d{ (posix::openat(posix::at_fdcwd, dirbuf, posix::o_rdonly | posix::o_directory | posix::o_cloexec, 0u)) };
753       if ( d ) {
754         posix::fsync(d.fd);
755         posix::close(d.fd);
756       }
757     }
758 
759     // the contents are on disk, but this handle still refers to the replaced inode
760     return __rebind();
761   }

782   write_encoded(Fn &&fn, const byte *src, usize src_len)
783   {
