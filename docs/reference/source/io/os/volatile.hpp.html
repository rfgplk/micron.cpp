57   inline void
58   __clone_fd(i32 src_fd, usize hint_size = 0)
59   {
60     if ( hint_size == 0 ) {
61       stat_t tmp{};
62       if ( posix::fstat(fd_t{ src_fd }, tmp) == 0 ) hint_size = static_cast<usize>(tmp.st_size);      // fstat returns 0 on success
63     }
64 
65     if ( hint_size > 0 ) posix::ftruncate(__handle, static_cast<posix::off64_t>(hint_size));
66 
67     posix::off64_t src_off = 0;
68     posix::off64_t dst_off = 0;
69     usize remaining = hint_size ? hint_size : static_cast<usize>(-1);
70     usize chunk = (1u << 20);      // 1mb
71 
72     if ( hint_size > 0 ) {
73       while ( remaining > 0 ) {
74         usize want = remaining < chunk ? remaining : chunk;
75         max_t n = micron::copy_file_range(src_fd, &src_off, __handle.fd, &dst_off, want, 0u);
76         if ( n < 0 ) {
77           // cross-fs copy_file_range needs >=5.3
78           if ( -n == 18 /*EXDEV*/ || -n == 38 /*ENOSYS*/ || -n == 22 /*EINVAL*/ ) {
79             byte __buf[16384];
80             while ( remaining > 0 ) {
81               usize __want = remaining < sizeof(__buf) ? remaining : sizeof(__buf);
82               max_t r = posix::pread(src_fd, __buf, __want, src_off);
83               if ( r <= 0 ) break;
84               max_t w = posix::pwrite(__handle.fd, __buf, static_cast<usize>(r), dst_off);
85               if ( w <= 0 ) break;
86               src_off += w;
87               dst_off += w;
88               remaining -= static_cast<usize>(w);
89             }
90           }
91           break;
92         }
93         if ( n == 0 ) break;
94         remaining -= static_cast<usize>(n);
95       }
96     } else {
97       i32 pfd[2] = { -1, -1 };
98       if ( posix::pipe(pfd) >= 0 ) {
99         for ( ;; ) {
100           max_t spliced = micron::splice(src_fd, nullptr, pfd[1], nullptr, chunk, 0u);
101           if ( spliced <= 0 ) {
102             posix::close(pfd[0]);
103             posix::close(pfd[1]);
104             break;
105           }
106           max_t moved = micron::splice(pfd[0], nullptr, __handle.fd, nullptr, static_cast<usize>(spliced), 0u);
107           (void)moved;
108         }
109       }
110     }
111 
112     posix::lseek(__handle, 0, posix::seek_set);
113     micron::zero(&sd);
114   }

197   void
198   close(void)
199   {
200     if ( __map_addr ) {      // do not leak a live mapping when the file closes
201       micron::munmap(reinterpret_cast<addr_t *>(__map_addr), __map_len);
202       __map_addr = nullptr;
203       __map_len = 0;

205     if ( __handle.closed() ) return;
206     posix::close(__handle.fd);
207     __handle.fd = posix::invalid_fd.fd;
208   }

596   void *
597   map_ro(void) const
598   {
599     __alive();
600     stat<STAT_OVERRIDE>();      // force a fresh size: a prior write may have grown the file past the cached sd
601     usize sz = static_cast<usize>(sd.st_size);
602     if ( sz == 0 ) return nullptr;
603     void *p = micron::mmap(nullptr, sz, prot_read, map_shared, __handle.fd, 0);
604     if ( micron::mmap_failed(p) ) return nullptr;
605     if ( __map_addr ) micron::munmap(reinterpret_cast<addr_t *>(__map_addr), __map_len);      // drop a prior mapping (no leak)
606     __map_addr = p;
607     __map_len = sz;
608     return p;
609   }

611   void *
612   map_rw(void)
613   {
614     __alive();
615     stat<STAT_OVERRIDE>();      // force a fresh size (see map_ro)
616     usize sz = static_cast<usize>(sd.st_size);
617     if ( sz == 0 ) return nullptr;
618     void *p = micron::mmap(nullptr, sz, prot_read | prot_write, map_shared, __handle.fd, 0);
619     if ( micron::mmap_failed(p) ) return nullptr;
620     if ( __map_addr ) micron::munmap(reinterpret_cast<addr_t *>(__map_addr), __map_len);
621     __map_addr = p;
622     __map_len = sz;
623     return p;
624   }

626   void
627   unmap(void *addr, usize len)
628   {
629     // if this is the mapping we tracked, use the EXACT length we mapped (a caller-recomputed
630     // size() can differ, and munmap of a too-large len tears down adjacent regions).
631     if ( addr && addr == __map_addr ) {
632       micron::munmap(reinterpret_cast<addr_t *>(__map_addr), __map_len);
633       __map_addr = nullptr;
634       __map_len = 0;
635       return;
636     }
637     micron::munmap(reinterpret_cast<addr_t *>(addr), len);
638   }
