361 inline constexpr i32
362 compose_open_flags(const modes mode, const open_opts &o) noexcept
363 {
364   i32 f = 0;
365   switch ( mode ) {
366   case modes::largeread:
367     f = posix::o_rdonly | posix::o_direct | posix::o_largefile;
368     break;
369   case modes::large:
370     f = posix::o_rdwr | posix::o_create | posix::o_trunc | posix::o_direct | posix::o_largefile;
371     break;
372   case modes::quiet:
373     f = posix::o_rdonly | posix::o_noatime;
374     break;
375   case modes::create:
376     f = posix::o_rdonly | posix::o_create | posix::o_excl;
377     break;
378   case modes::read:
379     f = posix::o_rdonly;
380     break;
381   case modes::readwrite:
382     f = posix::o_rdwr;
383     break;
384   case modes::write:
385     f = posix::o_wronly | posix::o_create | posix::o_trunc;
386     break;
387   case modes::readwritecreate:
388     f = posix::o_rdwr | posix::o_create | posix::o_trunc;
389     break;
390   case modes::append:
391     f = posix::o_wronly | posix::o_create | posix::o_append;
392     break;
393   case modes::appendread:
394     f = posix::o_rdwr | posix::o_create | posix::o_append;
395     break;
396   }
397   if ( o.sync ) f |= posix::o_sync;
398   if ( o.direct ) f |= posix::o_direct;
399   if ( o.nonblock ) f |= posix::o_nonblock;
400   if ( o.cloexec ) f |= posix::o_cloexec;
401   if ( o.noatime ) f |= posix::o_noatime;
402   return f;
403 }

450   template<typename T>
451   inline __attribute__((always_inline)) void
452   __open_linux(const T &str, const modes mode, u32 __accept = __nt_default, const open_opts &opts = {})
453   {
454     if ( !posix::verify(str) ) exc<except::io_error>("error in creating micron::os_file, malformed string.");
455 
456     const bool __read_mode = (mode != modes::append && mode != modes::create && mode != modes::readwritecreate);
457 
458     // No pre-open exists() probe: open() reports ENOENT itself (one fewer syscall, and no
459     // TOCTOU race with the open). This also fixes the create-mode bug — modes::write /
460     // modes::large / modes::appendread carry O_CREAT but were being rejected here on a
461     // missing path; they can now actually create the file.
462     if constexpr ( is_string<T> ) {
463       __handle.fd = static_cast<int>(__syscall_open(str.c_str(), mode, opts));
464     } else {
465       __handle.fd = static_cast<int>(__syscall_open(str, mode, opts));
466     }
467 
468     if ( __handle.has_error() ) exc<except::io_error>("micron::os_file failed to open");
469 
470     micron::zero(&sd);      // default: lazy stat cache (creating modes keep it lazy)
471 
472     // WARNING: validate file type via fstat on the opened fd rather than via the path-based posix::is_file
473     // the path-based variants were misclassified regular files on ARM32 + qemu.
474     // The fstat result also SEEDS the stat cache (sd) so the first size()/metadata access
475     // after a read-mode open is free (no redundant second fstat).
476     if ( __read_mode ) {
477       if ( posix::fstat(__handle, sd) != 0 ) {
478         micron::zero(&sd);
479         posix::close(__handle.fd);
480         __handle.fd = posix::invalid_fd.fd;
481         exc<except::io_error>("micron::os_file fstat failed after open");
482       }
483       const u32 __got = posix::__impl::stat_is_reg(sd)    ? __nt_reg
484                         : posix::__impl::stat_is_chr(sd)  ? __nt_chr
485                         : posix::__impl::stat_is_dir(sd)  ? __nt_dir
486                         : posix::__impl::stat_is_fifo(sd) ? __nt_fifo
487                         : posix::__impl::stat_is_blk(sd)  ? __nt_blk
488                         : posix::__impl::stat_is_sock(sd) ? __nt_sock
489                         : posix::__impl::stat_is_lnk(sd)  ? __nt_lnk
490                                                           : 0u;
491       if ( !(__got & __accept) ) {
492         micron::zero(&sd);      // do not leave a valid stat on a rejected handle
493         posix::close(__handle.fd);
494         __handle.fd = posix::invalid_fd.fd;
495         exc<except::io_error>("micron::os_file file isn't an accepted type (check type)");
496       }
497     }
498 
499     fname = str;
500   }
