51 struct block_info_t {
52   u64 size_bytes;               // BLKGETSIZE64

60   u64 readahead_sectors;        // BLKRAGET

215   max_t
216   fill(posix::off_t dev_offset = 0)
217   {
218     __alive();
219     if ( _buf_dirty && __handle.fd >= 0 )
220       flush_buf();      // skip flush on an invalid fd: flush_buf()->__alive() throws, and this op is noexcept
221     max_t n = posix::pread(__handle.fd, _buf.begin(), _buf.size(), dev_offset);
222     if ( n > 0 ) {
223       _buf_off = dev_offset;
224       _buf_valid = static_cast<usize>(n);
225     } else {
226       _buf_valid = 0;
227     }
228     return n;
229   }

231   max_t
232   flush_buf(void)
233   {
234     if ( !_buf_dirty || _buf_valid == 0 ) return 0;
235     __alive();
236     max_t n = posix::pwrite(__handle.fd, _buf.begin(), _buf_valid, _buf_off);
237     if ( n >= 0 ) _buf_dirty = false;
238     return n;
239   }

241   bool
242   buf_covers(posix::off_t offset, usize size) const noexcept
243   {
244     if ( _buf_valid == 0 ) return false;
245     // overflow-safe: avoid the (offset + size) addition that can wrap and approve an
246     // out-of-bounds cached copy when `size` is near-SIZE_MAX.
247     if ( offset < _buf_off ) return false;
248     const u64 delta = static_cast<u64>(offset) - static_cast<u64>(_buf_off);
249     if ( delta > _buf_valid ) return false;
250     return static_cast<u64>(size) <= static_cast<u64>(_buf_valid) - delta;
251   }
252 
253   max_t
254   write_buf(const void *src, usize size, posix::off_t dev_offset)
255   {
256     if ( !buf_covers(dev_offset, size) ) return -1;
257     usize delta = static_cast<usize>(dev_offset - _buf_off);
258     micron::memcpy(_buf.begin() + delta, static_cast<const byte *>(src), size);
259     _buf_dirty = true;
260     return static_cast<max_t>(size);
261   }

263   max_t
264   read_buffered(void *dst, usize size, posix::off_t offset)
265   {
266     __alive();
267     if ( buf_covers(offset, size) ) {
268       usize delta = static_cast<usize>(offset - _buf_off);
269       micron::memcpy(static_cast<byte *>(dst), _buf.begin() + delta, size);
270       return static_cast<max_t>(size);
271     }
272     max_t n = fill(offset);
273     if ( n <= 0 ) return n;
274     usize copy = static_cast<usize>(n) < size ? static_cast<usize>(n) : size;
275     micron::memcpy(static_cast<byte *>(dst), _buf.begin(), copy);
276     return static_cast<max_t>(copy);
277   }

378   void
379   query_info(void) const
380   {
381     alive_c();
382     u64 sz64 = 0;
383     unsigned long szsec = 0;
384     i32 lbsz = 512;
385     u32 pbsz = 512;
386     u32 iomin = 0;
387     u32 ioopt = 0;
388     i32 alignoff = 0;
389     u16 maxsect = 0;
390     unsigned long ra = 0;
391     i32 ro = 0;
392     u16 rotflag = 0;
393     u32 dzflag = 0;
394 
395     if ( posix::ioctl(__handle.fd, blkgetsize64, &sz64) < 0 ) {      // primary query: without the device size the cache is useless
396       _info_valid = false;
397       return;
398     }
399     posix::ioctl(__handle.fd, blkgetsize, &szsec);
400     posix::ioctl(__handle.fd, blksszget, &lbsz);
401     posix::ioctl(__handle.fd, blkpbszget, &pbsz);
402     posix::ioctl(__handle.fd, blkiomin, &iomin);
403     posix::ioctl(__handle.fd, blkioopt, &ioopt);
404     posix::ioctl(__handle.fd, blkalignoff, &alignoff);
405     posix::ioctl(__handle.fd, blksectget, &maxsect);
406     posix::ioctl(__handle.fd, blkraget, &ra);
407     posix::ioctl(__handle.fd, blkroget, &ro);
408     posix::ioctl(__handle.fd, blkrotational, &rotflag);
409     posix::ioctl(__handle.fd, blkdiscardzero, &dzflag);
410 
411     _info.size_bytes = sz64;
412     _info.size_sectors = szsec;
413     _info.logical_block_size = static_cast<u32>(lbsz > 0 ? lbsz : 512);
414     _info.physical_block_size = pbsz ? pbsz : _info.logical_block_size;
415     _info.min_io_size = iomin ? iomin : _info.logical_block_size;
416     _info.optimal_io_size = ioopt ? ioopt : _info.min_io_size;
417     _info.alignment_offset = alignoff;
418     _info.max_sectors = maxsect;
419     _info.readahead_sectors = ra;
420     _info.read_only = (ro != 0);
421     _info.rotational = (rotflag != 0);
422     _info.discard_zeroes = (dzflag != 0);
423     _info_valid = true;
424   }

691   set_read_only(bool on = true)
692   {
693     __alive();
694     i32 val = on ? 1 : 0;
695     i32 r = static_cast<i32>(posix::ioctl(__handle.fd, blkroset, &val));
696     if ( r == 0 && _info_valid ) _info.read_only = on;
697     return r;
698   }

705   }

712   }

719   }

721   u64
722   current_block(void) const
723   {
724     alive_c();
725     posix::off_t pos = posix::tell(__handle);
726     u32 bsz = __bsz();
727     return bsz ? static_cast<u64>(pos) / static_cast<u64>(bsz) : 0;
728   }

730   max_t
731   advance(void)
732   {
733     __alive();
734     posix::off_t next = _buf_off + static_cast<posix::off_t>(_buf_valid ? _buf_valid : _buf.size());
735     if ( static_cast<u64>(next) >= device_size() ) return 0;
736     return fill(next);
737   }

741   {
742     invalidate_buf();
743     _buf_off = 0;
744   }

747   at_end(void) const
748   {
749     if ( _buf_valid == 0 ) return true;
750     return (static_cast<u64>(_buf_off) + _buf_valid) >= device_size();
751   }

792 private:
793   inline void
794   alive_c(void) const
795   {
796     if ( __handle.fd < 0 ) exc<except::io_error>("micron::block, fd isn't open.");      // any negative fd is invalid, not just -1
797   }
798 
799   u32
800   __bsz(void) const
801   {
802     if ( !_info_valid ) query_info();
803     return _info.logical_block_size ? _info.logical_block_size : 512u;
804   }
805 
806   posix::off_t
807   __block_off(u64 n) const
808   {
809     return static_cast<posix::off_t>(n) * static_cast<posix::off_t>(__bsz());
810   }

812   void
813   __open_block(const char *str, const modes mode)
814   {
815     if ( !posix::verify(str) ) exc<except::io_error>("micron::block, malformed path.");
816     if ( !posix::exists(str) ) exc<except::io_error>("micron::block, path doesn't exist.");
817     if ( !posix::is_block_device(str) ) exc<except::io_error>("micron::block, path is not a block device.");
818 
819     i32 flags = 0;
820     switch ( mode ) {
821     case modes::read:
822     case modes::quiet:
823       flags = posix::o_rdonly | posix::o_cloexec;
824       break;
825     case modes::largeread:
826       flags = posix::o_rdonly | posix::o_direct | posix::o_largefile | posix::o_cloexec;
827       break;
828     case modes::large:
829       flags = posix::o_rdwr | posix::o_direct | posix::o_largefile | posix::o_cloexec;
830       break;
831     case modes::readwrite:
832       flags = posix::o_rdwr | posix::o_sync | posix::o_cloexec;
833       break;
834     default:
835       flags = posix::o_rdonly | posix::o_cloexec;
836       break;
837     }
838 
839     __handle.fd = static_cast<i32>(posix::open(str, flags));
840     if ( __handle.has_error() ) exc<except::io_error>("micron::block, failed to open device.");
841 
842     fname = str;
843     micron::zero(&sd);
844   }
