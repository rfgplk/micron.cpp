23 // frame c and write it at the file's current offset; one allocation, one write
24 template<typename C>
25 max_t
26 frame(io::file &f, const C &c)
27 {
28   max_t need = framed_size(c);
29   if ( need < 0 ) [[unlikely]]
30     return need;
31   micron::buffer out(static_cast<usize>(need));
32   max_t n = frame_into(out.data(), out.size(), c);
33   if ( n < 0 ) [[unlikely]]
34     return n;
35   return f.write(static_cast<const void *>(out.data()), static_cast<usize>(n));
36 }
37 
38 // read the remainder of the file from the current offset and reconstruct out
39 template<typename C>
40 max_t
41 unframe(io::file &f, C &out)
42 {
43   const posix::off64_t at = f.tell();
44   if ( at < 0 ) [[unlikely]]
45     return at;
46   f.refresh_stat();
47   const usize total = static_cast<usize>(f.size());
48   if ( total <= static_cast<usize>(at) ) [[unlikely]]
49     return -error::invalid_arg;
50   const usize len = total - static_cast<usize>(at);
51   micron::buffer in(len);
52   max_t got = f.read(static_cast<void *>(in.data()), len);
53   if ( got < 0 ) [[unlikely]]
54     return got;
55   return unframe_from(in.data(), static_cast<usize>(got), out);
56 }

58 template<typename C>
59 micron::option<C, io::error_t>
60 unframe(io::file &f)
61 {
62   C out{};
63   max_t r = unframe(f, out);
64   if ( r < 0 ) [[unlikely]]
65     return micron::option<C, io::error_t>{ io::error_t(static_cast<i32>(r)) };
66   return micron::option<C, io::error_t>{ micron::move(out) };
67 }

86 template<typename T>
87   requires micron::is_trivially_copyable_v<T>
88 max_t
89 serialize(io::file &f, const T &obj)
90 {
91   return f.write(static_cast<const void *>(&obj), sizeof(T));
92 }
93 
94 template<typename T>
95   requires micron::is_trivially_copyable_v<T>
96 max_t
97 deserialize(io::file &f, T &out)
98 {
99   static_assert(__impl::mfr_pod_safe<T>,
100                 "micron::io::serialize::deserialize: refusing to reconstruct this type from possibly-untrusted bytes "
101                 "(a pointer/bool/enum/opaque-POD field can form an invalid value). Add `using mfr_pod_safe_tag = void;` "
102                 "to the type if EVERY bit pattern is a valid value.");
103   return f.read(static_cast<void *>(&out), sizeof(T));
104 }

106 template<typename T>
107   requires(!micron::is_null_pointer_v<T>)
108 max_t
109 serialize_page(io::file &f, const T &obj)
110 {
111   const byte *b
112       = reinterpret_cast<const byte *>(reinterpret_cast<uintptr_t>(&obj) & ~(static_cast<uintptr_t>(__micron_page_size_default) - 1));
113   return f.write(static_cast<const void *>(b), __micron_page_size_default);
114 }
