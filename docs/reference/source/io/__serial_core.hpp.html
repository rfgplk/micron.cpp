14 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
15 //  micron framed representation
16 //
17 // 8-byte header {
18 //   'M','F','R','1',
19 //   u8 flags (bit0 = little-endian),
20 //   u8 kind (0 seq / 1 map / 2 set),
21 //   u16 reserved }
22 // body grammar (recursive, packed, native-endian):
23 //   trivially-copyable element              -> raw sizeof(E) bytes
24 //   string element                          -> u64 byte_len + raw bytes
25 //   contiguous container of TC elements     -> u64 count + one count*sizeof(vt) blit
26 //   node/nested container                   -> u64 count + frame(element) each
27 //   map                                     -> u64 pair_count + frame(key) frame(value) each

61 concept __kv_rebuildable = requires(T t, K k, V v) { t.insert(micron::move(k), micron::move(v)); }

116 inline bool
117 __get_bytes(const byte *src, usize len, usize &off, void *dst, usize n) noexcept
118 {
119   if ( off > len || n > len - off ) [[unlikely]]
120     return false;
121   micron::bytecpy(static_cast<byte *>(dst), src + off, n);
122   off += n;
123   return true;
124 }

142 template<usize D, typename E>
143 max_t
144 frame_element(byte *dst, usize cap, usize &off, const E &e) noexcept
145 {
146   static_assert(D <= __mfr_max_depth, "micron::io::serialize: MFR1 nesting depth exceeded");
147   using U = micron::remove_cvref_t<E>;
148 
149   if constexpr ( is_string<U> ) {
150     const u64 blen = static_cast<u64>(e.size()) * sizeof(typename U::value_type);
151     if ( dst ) {
152       if ( off + sizeof(u64) + blen > cap ) [[unlikely]]
153         return -error::invalid_arg;
154       __put_u64(dst, off, blen);
155       __put_bytes(dst, off, e.c_str(), static_cast<usize>(blen));
156     } else {
157       off += sizeof(u64) + static_cast<usize>(blen);
158     }
159     return 0;
160   } else if constexpr ( blit_container<U> ) {
161     const u64 cnt = static_cast<u64>(e.size());
162     const usize blen = static_cast<usize>(cnt) * sizeof(typename U::value_type);
163     if ( dst ) {
164       if ( off + sizeof(u64) + blen > cap ) [[unlikely]]
165         return -error::invalid_arg;
166       __put_u64(dst, off, cnt);
167       __put_bytes(dst, off, e.data(), blen);
168     } else {
169       off += sizeof(u64) + blen;
170     }
171     return 0;
172   } else if constexpr ( node_chain<U> ) {
173     u64 cnt = 0;
174     for ( auto *p = e.ibegin(); p != nullptr; p = p->next ) ++cnt;
175     if ( dst ) {
176       if ( off + sizeof(u64) > cap ) [[unlikely]]
177         return -error::invalid_arg;
178       __put_u64(dst, off, cnt);
179     } else {
180       off += sizeof(u64);
181     }
182     for ( auto *p = e.ibegin(); p != nullptr; p = p->next )
183       if ( max_t r = frame_element<D + 1>(dst, cap, off, p->data); r < 0 ) [[unlikely]]
184         return r;
185     return 0;
186   } else if constexpr ( kv_key_iter<U> ) {
187     const u64 cnt = static_cast<u64>(e.size());
188     if ( dst ) {
189       if ( off + sizeof(u64) > cap ) [[unlikely]]
190         return -error::invalid_arg;
191       __put_u64(dst, off, cnt);
192     } else {
193       off += sizeof(u64);
194     }
195     for ( auto itr = e.begin(); itr != e.end(); ++itr ) {
196       if ( !itr->key ) continue;      // skip empty slots (same convention as the print engine)
197       if ( max_t r = frame_element<D + 1>(dst, cap, off, itr->key); r < 0 ) [[unlikely]]
198         return r;
199       if ( max_t r = frame_element<D + 1>(dst, cap, off, itr->value); r < 0 ) [[unlikely]]
200         return r;
201     }
202     return 0;
203   } else if constexpr ( kv_ab_iter<U> ) {
204     const u64 cnt = static_cast<u64>(e.size());
205     if ( dst ) {
206       if ( off + sizeof(u64) > cap ) [[unlikely]]
207         return -error::invalid_arg;
208       __put_u64(dst, off, cnt);
209     } else {
210       off += sizeof(u64);
211     }
212     for ( auto itr = e.begin(); itr != e.end(); ++itr ) {
213       if ( max_t r = frame_element<D + 1>(dst, cap, off, (*itr).a); r < 0 ) [[unlikely]]
214         return r;
215       if ( max_t r = frame_element<D + 1>(dst, cap, off, (*itr).b); r < 0 ) [[unlikely]]
216         return r;
217     }
218     return 0;
219   } else if constexpr ( iterable_generic<U> && !frameable_leaf<U> ) {
220     u64 cnt = 0;
221     for ( auto itr = e.begin(); itr != e.end(); ++itr ) ++cnt;
222     if ( dst ) {
223       if ( off + sizeof(u64) > cap ) [[unlikely]]
224         return -error::invalid_arg;
225       __put_u64(dst, off, cnt);
226     } else {
227       off += sizeof(u64);
228     }
229     for ( auto itr = e.begin(); itr != e.end(); ++itr )
230       if ( max_t r = frame_element<D + 1>(dst, cap, off, *itr); r < 0 ) [[unlikely]]
231         return r;
232     return 0;
233   } else if constexpr ( frameable_leaf<U> ) {
234     if ( dst ) {
235       if ( off + sizeof(U) > cap ) [[unlikely]]
236         return -error::invalid_arg;
237       __put_bytes(dst, off, &e, sizeof(U));
238     } else {
239       off += sizeof(U);
240     }
241     return 0;
242   } else {
243     static_assert(sizeof(U) == 0, "micron::io::serialize: type is not MFR1-frameable");
244     return -error::invalid_arg;
245   }
246 }

254 template<usize D, typename E>
255 max_t
256 unframe_element(const byte *src, usize len, usize &off, E &out) noexcept
257 {
258   static_assert(D <= __mfr_max_depth, "micron::io::serialize: MFR1 nesting depth exceeded");
259   using U = micron::remove_cvref_t<E>;
260 
261   if constexpr ( is_string<U> ) {
262     u64 blen = 0;
263     if ( !__get_u64(src, len, off, blen) ) [[unlikely]]
264       return -error::invalid_arg;
265     if ( blen > len - off ) [[unlikely]]      // overflow safe
266       return -error::invalid_arg;
267     const usize elems = static_cast<usize>(blen) / sizeof(typename U::value_type);
268     if constexpr ( requires(U u, usize n) {
269                      u.reserve(n);
270                      u.data();
271                      u._buf_set_length(n);
272                    } ) {
273       out.reserve(elems + 1);
274       __get_bytes(src, len, off, out.data(), elems * sizeof(typename U::value_type));
275       out._buf_set_length(elems);
276     } else {
277       out = U();
278       if constexpr ( requires(U u, usize n) { u.reserve(n); } ) out.reserve(elems + 1);
279       for ( usize i = 0; i < elems; ++i ) {
280         typename U::value_type ch{};
281         __get_bytes(src, len, off, &ch, sizeof(ch));
282         out.push_back(ch);
283       }
284     }
285     off += static_cast<usize>(blen) - elems * sizeof(typename U::value_type);      // skip any trailing partial element
286     return 0;
287   } else if constexpr ( blit_container<U> ) {
288     static_assert(mfr_pod_safe<typename U::value_type>,
289                   "micron::io::serialize::unframe: refusing to reconstruct a contiguous container of a pointer/bool/"
290                   "enum/opaque-POD element from possibly-untrusted MFR1 bytes. Add using mfr_pod_safe_tag = void; to "
291                   "the element type if every bit pattern is a valid value.");
292     u64 cnt = 0;
293     if ( !__get_u64(src, len, off, cnt) ) [[unlikely]]
294       return -error::invalid_arg;
295     if ( cnt > (len - off) / sizeof(typename U::value_type) ) [[unlikely]]
296       return -error::invalid_arg;
297     const usize nbytes = static_cast<usize>(cnt) * sizeof(typename U::value_type);
298     if constexpr ( (requires(U u, usize n) { u.data(); u.set_size(n); } || requires(U u, usize n) { u.data(); u._buf_set_length(n); })
299                    && requires(U u, usize n) { u.reserve(n); } ) {
300       // contiguous target with a size setter
301       out.reserve(static_cast<usize>(cnt));
302       if ( !__get_bytes(src, len, off, out.data(), nbytes) ) [[unlikely]]
303         return -error::invalid_arg;
304       __blit_set_len(out, static_cast<usize>(cnt));
305     } else if constexpr ( requires(U u, typename U::value_type v) { u.push_back(v); } ) {
306       if constexpr ( requires(U u) { u.clear(); } )
307         out.clear();
308       else
309         out = U();
310       if constexpr ( requires(U u, usize n) { u.reserve(n); } ) out.reserve(static_cast<usize>(cnt));
311       for ( u64 i = 0; i < cnt; ++i ) {
312         typename U::value_type v;
313         __get_bytes(src, len, off, &v, sizeof(v));
314         out.push_back(v);
315       }
316     } else {
317       out = U();
318       usize i = 0;
319       for ( ; i < static_cast<usize>(cnt) && i < out.size(); ++i ) __get_bytes(src, len, off, &out[i], sizeof(typename U::value_type));
320       off += (static_cast<usize>(cnt) - i) * sizeof(typename U::value_type);
321     }
322     return 0;
323   } else if constexpr ( node_chain<U> ) {
324     u64 cnt = 0;
325     if ( !__get_u64(src, len, off, cnt) ) [[unlikely]]
326       return -error::invalid_arg;
327     out = U();
328     for ( u64 i = 0; i < cnt; ++i ) {
329       typename U::value_type v{};
330       if ( max_t r = unframe_element<D + 1>(src, len, off, v); r < 0 ) [[unlikely]]
331         return r;
332       out.push_back(micron::move(v));
333     }
334     return 0;
335   } else if constexpr ( map_like<U> ) {
336     u64 cnt = 0;
337     if ( !__get_u64(src, len, off, cnt) ) [[unlikely]]
338       return -error::invalid_arg;
339     out = U();
340     using K = micron::remove_cvref_t<decltype(__map_key(out))>;
341     using V = micron::remove_cvref_t<decltype(__map_value(out))>;
342     for ( u64 i = 0; i < cnt; ++i ) {
343       K k{};
344       V v{};
345       if ( max_t r = unframe_element<D + 1>(src, len, off, k); r < 0 ) [[unlikely]]
346         return r;
347       if ( max_t r = unframe_element<D + 1>(src, len, off, v); r < 0 ) [[unlikely]]
348         return r;
349       if constexpr ( requires(U u) { u.insert_unhash(k, micron::move(v)); } )
350         out.insert_unhash(k, micron::move(v));      // tagged maps store hashes; do not rehash
351       else
352         out.insert(micron::move(k), micron::move(v));
353     }
354     return 0;
355   } else if constexpr ( iterable_generic<U> && !frameable_leaf<U> ) {
356     u64 cnt = 0;
357     if ( !__get_u64(src, len, off, cnt) ) [[unlikely]]
358       return -error::invalid_arg;
359     if constexpr ( requires(U u, usize n) {
360                      u.data();
361                      u.reserve(n);
362                      u.resize(n);
363                      u[n];
364                      u.push_back(micron::move(u[n]));
365                    } ) {
366       if ( cnt > len - off ) return -error::invalid_arg;
367       for ( usize i = 0; i < static_cast<usize>(cnt); ++i ) {
368         if ( i < out.size() ) {
369           if ( max_t r = unframe_element<D + 1>(src, len, off, out[i]); r < 0 ) [[unlikely]]
370             return r;
371         } else {
372           typename U::value_type value{};
373           if ( max_t r = unframe_element<D + 1>(src, len, off, value); r < 0 ) [[unlikely]]
374             return r;
375           out.push_back(micron::move(value));
376         }
377       }
378       out.resize(static_cast<usize>(cnt));
379       return 0;
380     }
381     out = U();
382     for ( u64 i = 0; i < cnt; ++i ) {
383       typename U::value_type v{};
384       if ( max_t r = unframe_element<D + 1>(src, len, off, v); r < 0 ) [[unlikely]]
385         return r;
386       if constexpr ( requires(U u, typename U::value_type x) { u.push_back(micron::move(x)); } )
387         out.push_back(micron::move(v));
388       else if constexpr ( requires(U u, typename U::value_type x) { u.insert(micron::move(x)); } )
389         out.insert(micron::move(v));
390       else
391         static_assert(sizeof(U) == 0, "micron::io::serialize: no way to insert into this container");
392     }
393     return 0;
394   } else if constexpr ( frameable_leaf<U> ) {
395     static_assert(mfr_pod_safe<U>, "micron::io::serialize::unframe: refusing to reconstruct this type from possibly-untrusted MFR1 "
396                                    "bytes; a pointer/bool/enum/opaque-POD leaf can form an invalid value (wild pointer / UB). Add "
397                                    "using mfr_pod_safe_tag = void; to the type if every bit pattern is a valid value.");
398     if ( !__get_bytes(src, len, off, &out, sizeof(U)) ) [[unlikely]]
399       return -error::invalid_arg;
400     return 0;
401   } else {
402     static_assert(sizeof(U) == 0, "micron::io::serialize: type is not MFR1-unframeable");
403     return -error::invalid_arg;
404   }
405 }

419 template<typename C>
420 max_t
421 framed_size(const C &c) noexcept
422 {
423   usize off = 0;
424   if ( max_t r = __impl::frame_element<0>(nullptr, 0, off, c); r < 0 ) [[unlikely]]
425     return r;
426   return static_cast<max_t>(off + __impl::__mfr_header);

429 // serialize c into [dst, dst+cap)
430 template<typename C>
431 max_t
432 frame_into(byte *dst, usize cap, const C &c) noexcept
433 {
434   if ( dst == nullptr || cap < __impl::__mfr_header ) [[unlikely]]
435     return -error::invalid_arg;
436   usize off = 0;
437   dst[off++] = __impl::__mfr_magic[0];
438   dst[off++] = __impl::__mfr_magic[1];
439   dst[off++] = __impl::__mfr_magic[2];
440   dst[off++] = __impl::__mfr_magic[3];
441   dst[off++] = __impl::__little_endian() ? byte{ 1 } : byte{ 0 };
442   dst[off++] = __impl::__mfr_kind<C>();
443   dst[off++] = 0;
444   dst[off++] = 0;
445   if ( max_t r = __impl::frame_element<0>(dst, cap, off, c); r < 0 ) [[unlikely]]
446     return r;
447   return static_cast<max_t>(off);
448 }

450 template<typename C>
451 max_t
452 unframe_from(const byte *src, usize len, C &out) noexcept
453 {
454   if ( src == nullptr || len < __impl::__mfr_header ) [[unlikely]]
455     return -error::invalid_arg;
456   if ( src[0] != __impl::__mfr_magic[0] || src[1] != __impl::__mfr_magic[1] || src[2] != __impl::__mfr_magic[2]
457        || src[3] != __impl::__mfr_magic[3] ) [[unlikely]]
458     return -error::invalid_arg;
459   if ( (src[4] != 0) != __impl::__little_endian() ) [[unlikely]]
460     return -error::invalid_arg;      // cross-endian streams are detectable, not readable
461   usize off = __impl::__mfr_header;
462   if ( max_t r = __impl::unframe_element<0>(src, len, off, out); r < 0 ) [[unlikely]]
463     return r;
464   return static_cast<max_t>(off);
465 }
