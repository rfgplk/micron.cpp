84 inline void
85 set_color(color c, style s = style::bold)

87   char buf[] = "\033[1;xxm";
88   switch ( c ) {

96     break;

108     break;
109   case color::magenta:

111     buf[5] = ansi_colors::magenta[1];
112     break;
113   case color::cyan:

117   case color::white:

191 template<typename... T>
192 inline void
193 cerror(const T &...str)
194 {
195   set_color(color::red, style::italic);
196   set_color(color::red, style::bold);
197   io::errorln(str...);
198   set_color_reset();
199   (
200       [&] {
201         if constexpr ( requires { except::standard_error(str); } ) exc_silent<except::standard_error>(str);
202       }(),
203       ...);
204 }

206 template<typename... T>
207 inline void
208 __micron_log(const char *FILEMACRO, int line, const T &...str)
209 {
210   set_color(color::cyan, style::italic);
211   set_color(color::cyan, style::bold);
212   io::print(FILEMACRO);
213   io::print(":");
214   set_color_reset();
215   set_color(color::cyan, style::bold);
216   io::print(line);
217   io::print(": ");
218   set_color_reset();
219   set_color(color::yellow, style::bold);
220   io::println(str...);
221   set_color_reset();
222   io::println("\n");
223 }
