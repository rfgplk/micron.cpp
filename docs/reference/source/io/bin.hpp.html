97 inline void
98 kmp_table(const byte *pat, usize pat_len, usize *table)
99 {
100   if ( pat_len == 0 ) return;
101   table[0] = 0;
102   usize k = 0;
103   for ( usize i = 1; i < pat_len; ) {
104     if ( pat[i] == pat[k] ) {
105       table[i] = k + 1;
106       ++i;
107       ++k;
108     } else if ( k ) {
109       k = table[k - 1];
110     } else {
111       table[i] = 0;
112       ++i;
113     }
114   }
115 }

117 inline usize
118 kmp_search(const byte *hay, usize hay_len, const byte *pat, usize pat_len, const usize *table)
119 {
120   if ( pat_len == 0 ) return 0;
121   if ( pat_len > hay_len ) return hay_len;
122 
123   usize k = 0;
124   for ( usize i = 0; i < hay_len; ) {
125     if ( hay[i] == pat[k] ) {
126       ++k;
127       ++i;
128       if ( k == pat_len ) return i - k;
129     } else if ( k ) {
130       k = table[k - 1];
131     } else {
132       ++i;
133     }
134   }
135   return hay_len;
136 }

138 inline usize
139 reverse_search(const byte *hay, usize hay_len, const byte *pat, usize pat_len)
140 {
141   if ( pat_len == 0 || pat_len > hay_len ) return hay_len;
142   usize i = hay_len - pat_len;
143   for ( ;; ) {
144     bool ok = true;
145     for ( usize j = 0; j < pat_len; ++j ) {
146       if ( hay[i + j] != pat[j] ) {
147         ok = false;
148         break;
149       }
150     }
151     if ( ok ) return i;
152     if ( i == 0 ) break;
153     --i;
154   }
155   return hay_len;
156 }

158 inline double
159 fast_log2(double x)
160 {
161   if ( x <= 0.0 ) return 0.0;
162   int exp = 0;
163   while ( x >= 2.0 ) {
164     x *= 0.5;
165     ++exp;
166   }
167   while ( x < 1.0 ) {
168     x *= 2.0;
169     --exp;
170   }
171   double t = x - 1.0;
172   double ln = t;
173   double p = t * t;
174   ln -= p / 2.0;
175   p *= t;
176   ln += p / 3.0;
177   p *= t;
178   ln -= p / 4.0;
179   p *= t;
180   ln += p / 5.0;
181   return static_cast<double>(exp) + ln / 0.693147180559945;
182 }

528   // left fold across the whole file, one window at a time
529   // requires an allocated window buffer
530   template<typename R, typename Fn>
531     requires fn_fold<Fn, R, bin_range_t>
532   micron::option<R, io::error_t>
533   fold(R init, Fn &&fn)
534   {
535     using Ret = micron::option<R, io::error_t>;
536     if ( i32 __e = __check() ) [[unlikely]]
537       return Ret{ io::error_t(__e) };
538     if ( !__buf ) [[unlikely]]
539       return Ret{ io::error_t(-error::invalid_arg) };
540     const usize fsz = static_cast<usize>(size());
541     posix::off_t off = 0;
542     while ( static_cast<usize>(off) < fsz ) {
543       max_t n = fill(off);
544       if ( n < 0 ) [[unlikely]] {
545         if ( -n == error::interrupted ) continue;
546         return Ret{ io::error_t(static_cast<i32>(n)) };
547       }
548       if ( n == 0 ) break;
549       init = fn(micron::move(init), as_range());
550       off += static_cast<posix::off_t>(n);
551     }
552     return Ret{ micron::move(init) };
553   }

555   // visit each window across the whole file; returns total bytes visited or -errno
556   template<typename Fn>
557     requires micron::invocable<Fn, bin_range_t>
558   max_t
559   each_window(Fn &&fn)
560   {
561     if ( i32 __e = __check() ) [[unlikely]]
562       return __e;
563     if ( !__buf ) [[unlikely]]
564       return -error::invalid_arg;
565     const usize fsz = static_cast<usize>(size());
566     posix::off_t off = 0;
567     max_t total = 0;
568     while ( static_cast<usize>(off) < fsz ) {
569       max_t n = fill(off);
570       if ( n < 0 ) [[unlikely]] {
571         if ( -n == error::interrupted ) continue;
572         return total ? total : n;
573       }
574       if ( n == 0 ) break;
575       fn(as_range());
576       total += n;
577       off += static_cast<posix::off_t>(n);
578     }
579     return total;
580   }

783   bin_match_t
784   search_file(const byte *pat, usize pat_len)
785   {
786     __require_buf();
787     if ( pat_len == 0 ) return { true, 0, 0 };
788 
789     micron::buffer __kmp_tbl(pat_len * sizeof(usize));      // heap, not alloca
790     usize *table = reinterpret_cast<usize *>(__kmp_tbl.begin());
791     __bin_impl::kmp_table(pat, pat_len, table);
792 
793     usize overlap = pat_len > 1 ? pat_len - 1 : 0;
794     usize work_sz = __buf_sz + overlap;
795     micron::buffer work(work_sz);
796 
797     posix::off_t file_off = 0;
798     usize file_size = static_cast<usize>(size());
799     usize work_fill = 0;
800 
801     while ( static_cast<usize>(file_off) < file_size ) {
802 
803       if ( overlap && work_fill >= overlap ) micron::memcpy(work.begin(), work.begin() + (work_fill - overlap), overlap);
804 
805       max_t n = posix::pread(__handle.fd, work.begin() + overlap, __buf_sz, file_off);
806       if ( n <= 0 ) break;
807 
808       work_fill = overlap + static_cast<usize>(n);
809 
810       const usize from = (file_off == 0) ? overlap : 0;      // first window: skip the unwritten [0, overlap) carry
811       usize pos = __bin_impl::kmp_search(work.begin() + from, work_fill - from, pat, pat_len, table);
812       if ( pos < work_fill - from ) {
813         const usize w = from + pos;
814         const usize file_pos = static_cast<usize>(file_off) + w - overlap;
815         return { true, file_pos, pat_len };
816       }
817 
818       file_off += static_cast<posix::off_t>(n);
819     }
820 
821     return no_match;
822   }

853     micron::buffer __kmp_tbl(pat_len * sizeof(usize));      // heap, not alloca
854     usize *table = reinterpret_cast<usize *>(__kmp_tbl.begin());
855     __bin_impl::kmp_table(pat, pat_len, table);
856 
857     usize overlap = pat_len > 1 ? pat_len - 1 : 0;

1029   bin_stats_t
1030   analyse(bin_range_t range = {}) const noexcept
1031   {
1032     bin_stats_t st{};
1033     const byte *p;
1034     usize n;
1035     if ( range.ptr && range.len ) {
1036       p = range.ptr;
1037       n = range.len;
1038     } else {
1039       if ( !__buf || __buf_valid == 0 ) return st;
1040       p = __buf->begin();
1041       n = __buf_valid;
1042     }
1043 
1044     st.total_bytes = n;
1045     usize cur_zero = 0, cur_nz = 0;
1046 
1047     for ( usize i = 0; i < n; ++i ) {
1048       byte b = p[i];
1049       st.freq[b]++;
1050       if ( b == 0x00 ) {
1051         st.zero_bytes++;
1052         if ( ++cur_zero > st.longest_zero_run ) st.longest_zero_run = cur_zero;
1053         if ( cur_nz ) {
1054           cur_nz = 0;
1055         }
1056       } else {
1057         if ( ++cur_nz > st.longest_nonzero_run ) st.longest_nonzero_run = cur_nz;
1058         if ( cur_zero ) {
1059           cur_zero = 0;
1060         }
1061       }
1062       if ( b >= 0x20 && b <= 0x7e ) st.printable_bytes++;
1063       if ( b >= 0x80 ) st.high_bytes++;
1064     }
1065 
1066     double H = 0.0;
1067     for ( usize i = 0; i < 256; ++i ) {
1068       if ( st.freq[i] == 0 ) continue;
1069       double pi = static_cast<double>(st.freq[i]) / static_cast<double>(n);
1070       H -= pi * __bin_impl::fast_log2(pi);
1071     }
1072     st.entropy = H;
1073 
1074     return st;
1075   }
