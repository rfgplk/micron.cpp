22 struct node_id {

24   posix::ino64_t ino;
25 };

27 constexpr u32 ftw_max_depth = 100;
28 
29 template<typename Sink>
30 inline bool
31 walk(path &&p, Sink &&sink, node_id *chain, u32 depth, collect what, bool is_root = false)
32 {
33   if ( depth >= ftw_max_depth ) return true;
34 
35   // entries collected at this level
36   // NB: this must stay exception-free. An unreadable sub-directory (EACCES) is skipped, not fatal,
37   // and that cannot be expressed with try/catch: below the eh gate exc<> aborts instead of
38   // throwing, so a catch-based skip would kill the process mid-walk. try_*() report instead.
39   micron::fvector<path_t> coll;
40   if ( const i32 r = (what == collect::files) ? p.try_files(coll) : (what == collect::all ? p.try_all(coll) : p.try_dirs(coll)); r < 0 ) {
41     if ( is_root ) exc_e<except::io_error>(r, "micron::io::ftw, failed to open root directory.");
42     return true;      // unreadable sub-directory -> skip
43   }
44   for ( auto &n : coll ) {
45     if ( n == "." || n == ".." ) continue;
46     if ( !sink(p.join(n.c_str())) ) return false;
47   }
48 
49   // recurse into sub-directories only
50   micron::fvector<path_t> subdirs;
51   if ( const i32 r = p.try_dirs(subdirs); r < 0 ) {
52     if ( is_root ) exc_e<except::io_error>(r, "micron::io::ftw, failed to open root directory.");
53     return true;      // unreadable sub-directory -> skip
54   }
55   for ( auto &n : subdirs ) {
56     if ( n == "." || n == ".." ) continue;
57     path_t child = p.join(n.c_str());
58     // NOTE:: paths are trusted; not racefree against an attacker swapping child for a symlink between the check and the open
59     if ( posix::is_symlink(child.c_str()) ) continue;      // WARNING: do NOT follow symlinks
60 
61     posix::stat_t st{};
62     if ( !posix::__impl::__stat(child.c_str(), st) ) continue;
63     node_id id{ st.st_dev, static_cast<posix::ino64_t>(st.st_ino) };
64     bool seen = false;
65     for ( u32 i = 0; i < depth; ++i )
66       if ( chain[i].dev == id.dev && chain[i].ino == id.ino ) {
67         seen = true;
68         break;
69       }
70     if ( seen ) continue;      // directory cycle -> skip
71 
72     chain[depth] = id;
73     if ( !walk(path(child.c_str()), sink, chain, depth + 1, what) ) return false;
74   }
75   return true;
76 }

78 template<typename Sink>
79 inline bool
80 run_sink(path &&p, Sink &&sink, collect what)
81 {
82   node_id chain[ftw_max_depth];
83   u32 start = 0;
84   posix::stat_t st{};
85   if ( posix::__impl::__stat(p.get().c_str(), st) ) {
86     chain[0] = node_id{ st.st_dev, static_cast<posix::ino64_t>(st.st_ino) };
87     start = 1;
88   }
89   return walk(micron::move(p), sink, chain, start, what, true);
90 }

92 inline micron::fvector<path_t>
93 run(path &&p, collect what)
94 {
95   micron::fvector<path_t> out;
96   run_sink(
97       micron::move(p),
98       [&out](path_t &&x) {
99         out.push_back(micron::move(x));
100         return true;
101       },
102       what);
103   return out;
104 }

106 template<typename Fn>
107 inline usize
108 visit(path &&p, Fn &&fn, collect what)
109 {
110   usize n = 0;
111   run_sink(
112       micron::move(p),
113       [&](path_t &&x) -> bool {
114         ++n;
115         if constexpr ( micron::is_convertible_v<micron::invoke_result_t<Fn, const path_t &>, bool> )
116           return static_cast<bool>(fn(x));
117         else {
118           fn(x);
119           return true;
120         }
121       },
122       what);
123   return n;
124 }

172 // left fold over every visited path
173 template<typename R, typename Fn>
174   requires fn_fold<Fn, R, const path_t &>
175 inline R
176 ftw_fold(path &&p, R init, Fn &&fn)
177 {
178   __ftw::run_sink(
179       micron::move(p),
180       [&](path_t &&x) -> bool {
181         init = fn(micron::move(init), x);
182         return true;
183       },
184       __ftw::collect::all);
185   return init;
