23 // io::path is a lexical, string-backed path handle; holds NO file descriptors and never touches the filesystem on construction
24 // can represent any path (file, dir, missing) cheaply

29   static path_t
30   __parent_of(const path_t &p)
31   {
32     const char *s = p.c_str();
33     usize n = p.size();
34     while ( n > 1 && s[n - 1] == '/' ) --n;
35     usize sep = n;
36     while ( sep > 0 && s[sep - 1] != '/' ) --sep;
37     if ( sep == 0 ) return path_t(".");
38     if ( sep == 1 ) return path_t("/");
39     path_t out;
40     for ( usize i = 0; i < sep - 1; ++i ) out += s[i];
41     return out;
42   }

44   // NOTE: filesystem paths are trusted
45   static path_t
46   __join(const path_t &base, const path_t &rel)
47   {
48     if ( rel.size() == 0 ) return base;
49     if ( rel[0] == '/' ) return rel;
50     if ( base.size() == 0 ) return rel;
51     path_t out(base);
52     if ( out[out.size() - 1] != '/' ) out += '/';
53     out += rel;
54     return prune(micron::move(out));
55   }

76   static linux_permissions
77   __perms_of(const char *p)
78   {
79     stat_t st{};
80     if ( !posix::__impl::__stat(p, st) ) return linux_permissions::none();
81     linux_permissions perms = linux_permissions::from_mode(st.st_mode);
82     perms.setuid = (st.st_mode & posix::s_isuid) != 0;
83     perms.setgid = (st.st_mode & posix::s_isgid) != 0;
84     perms.sticky = (st.st_mode & posix::s_isvtx) != 0;
85     return perms;
86   }

88   template<typename Pred>
89   micron::fvector<path_t>
90   __list(Pred pred) const
91   {
92     micron::fvector<path_t> out;
93     if ( !posix::is_dir(__p.c_str()) ) return out;      // never construct dir on a non-dir (no throw)
94     dir d(__p);                                         // fd opens here, closes when d leaves scope
95     for ( const auto &n : d.get_children() )
96       if ( pred(n) ) out.emplace_back(n.d_name);
97     return out;
98   }
99 
100   // non-throwing sibling of __list: appends to `out` and returns 0, or returns a negative -errno
101   // if the directory could not be opened. Nothing on this path can raise -- dir::try_open adopts
102   // an errno-carrying handle, and get_children()/_read_entries() only run once valid() holds.
103   template<typename Pred>
104   i32
105   __try_list(Pred pred, micron::fvector<path_t> &out) const
106   {
107     dir d = dir::try_open(__p.c_str());
108     if ( !d.valid() ) return d.__handle.fd;      // fd_t carries the negative errno inline
109     for ( const auto &n : d.get_children() )
110       if ( pred(n) ) out.emplace_back(n.d_name);
111     return 0;
112   }

174   basename() const

186   path_t
187   extension() const
188   {
189     path_t base = basename();
190     const char *s = base.c_str();
191     usize l = base.size();
192     if ( l == 0 || s[0] == '.' ) return path_t();
193     usize dot = l;
194     while ( dot > 0 && s[dot - 1] != '.' ) --dot;
195     if ( dot == 0 ) return path_t();
196     path_t out;
197     for ( usize i = dot - 1; i < l; ++i ) out += s[i];
198     return out;
199   }

202   stem() const
203   {
204     path_t base = basename();

219   micron::vector<micron::sstr<posix::name_max + 1>>
220   components() const
221   {
222     micron::vector<micron::sstr<posix::name_max + 1>> out;
223     if ( __p.size() == 0 ) return out;
224     const char *p = __p.c_str();
225     usize n = __p.size();
226     usize start = 0;
227     for ( usize i = 0; i <= n; ++i ) {
228       if ( p[i] == '/' || p[i] == '\0' ) {
229         if ( i > start ) {
230           micron::sstr<posix::name_max + 1> comp;
231           for ( usize j = start; j < i; ++j ) comp += p[j];
232           out.push_back(comp);
233         }
234         start = i + 1;
235       }
236     }
237     return out;
238   }

240   usize
241   depth() const noexcept
242   {
243     if ( __p.size() == 0 ) return 0;
244     const char *p = __p.c_str();
245     usize n = __p.size();
246     usize d = 0;
247     bool in_comp = false;
248     for ( usize i = 0; i < n; ++i ) {
249       if ( p[i] == '/' ) {
250         in_comp = false;
251       } else if ( !in_comp ) {
252         in_comp = true;
253         ++d;
254       }
255     }
256     return d;
257   }

281   path_t
282   relative_to(const char *base) const
283   {
284     const char *p = __p.c_str();
285     usize pl = __p.size();
286     usize bl = 0;
287     while ( base[bl] ) ++bl;
288     usize common = 0;
289     usize i = 0;
290     while ( i < pl && i < bl && p[i] == base[i] ) {
291       if ( p[i] == '/' ) common = i + 1;
292       ++i;
293     }
294     if ( i == bl && (p[i] == '/' || p[i] == '\0') ) common = i + (p[i] == '/');
295     usize up = 0;
296     for ( usize j = common; j < bl; ++j )
297       if ( base[j] == '/' ) ++up;
298     if ( common < bl ) ++up;
299     path_t out;
300     for ( usize j = 0; j < up; ++j ) {
301       if ( out.size() ) out += '/';
302       out += "..";
303     }
304     if ( common < pl ) {
305       if ( out.size() ) out += '/';
306       for ( usize j = common; j < pl; ++j ) out += p[j];
307     }
308     if ( out.size() == 0 ) out += '.';
309     return out;
310   }

843   // non-throwing forms of files()/dirs()/all(): fill `out`, return 0 on success or a negative
844   // -errno if this directory could not be opened. For callers that must tolerate an unreadable
845   // directory in every build -- exc<> aborts rather than throws once __use_exceptions is false,

974   bool
975   operator<(const path &o) const noexcept
976   {
977     const char *a = __p.c_str();
978     const char *b = o.__p.c_str();
979     usize al = __p.size();
980     usize bl = o.__p.size();
981     usize n = al < bl ? al : bl;
982     for ( usize i = 0; i < n; ++i ) {
983       if ( (unsigned char)a[i] < (unsigned char)b[i] ) return true;
984       if ( (unsigned char)a[i] > (unsigned char)b[i] ) return false;
985     }
986     return al < bl;
987   }
