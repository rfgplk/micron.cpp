25 inline auto
26 __parse_device(const char *s, const micron::ustr8 &buf)
27 {
28   auto tk = micron::format::find(buf, s);
29   if ( !tk ) return micron::ustr8{};
30   auto del = micron::format::find(buf, tk, '=');
31   if ( !del ) return micron::ustr8{};
32   auto del_eol = micron::format::find(buf, del, '\n');
33   if ( !del_eol ) del_eol = buf.end();
34   return buf.substr(del + 1, del_eol);
35 }

37 inline usize
38 __parse_hex_words(const micron::ustr8 &buf, i64 *words, usize max_words)
39 {
40   usize n_words = 0;
41   auto cur = buf.cbegin();
42   const auto end = buf.cend();
43   while ( n_words < max_words ) {
44     while ( cur != end && *cur == ' ' ) ++cur;
45     if ( cur == end ) break;
46     auto next = micron::format::find(buf, cur, ' ');
47     if ( next == nullptr ) {
48       words[n_words++] = hex_string_to_int64(buf.substr(cur, end));
49       break;
50     }
51     words[n_words++] = hex_string_to_int64(buf.substr(cur, next));
52     cur = next + 1;
53   }
54   return n_words;
55 }

57 inline bool
58 __key_bit_set(const i64 *words, usize n_words, i32 code)
59 {
60   const usize k = static_cast<usize>(code);
61   const usize word_from_low = k / 64u;
62   const usize bit_in_word = k % 64u;
63   if ( word_from_low >= n_words ) return false;
64   const usize idx = n_words - 1 - word_from_low;
65   return (words[idx] >> bit_in_word) & i64(1);
66 }

68 inline bool
69 __has_key_code(const micron::ustr8 &keybuf, i32 code)
70 {
71   constexpr usize __max_words = 16;
72   i64 words[__max_words] = {};
73   const usize n_words = __parse_hex_words(keybuf, words, __max_words);
74   if ( n_words == 0 ) return false;
75   return __key_bit_set(words, n_words, code);
76 }

78 inline bool
79 __has_alnum_keys(const micron::ustr8 &buf)
80 {
81   constexpr usize __max_words = 16;
82   i64 words[__max_words] = {};
83   const usize n_words = __parse_hex_words(buf, words, __max_words);
84   if ( n_words == 0 ) return false;
85 
86   constexpr byte alnum[] = {
87     key_1, key_2, key_3, key_4, key_5, key_6, key_7, key_8, key_9, key_0, key_q, key_w, key_e, key_r, key_T, key_y, key_u, key_i,
88     key_o, key_p, key_a, key_s, key_d, key_f, key_g, key_h, key_j, key_k, key_l, key_z, key_x, key_c, key_v, key_b, key_n, key_m,
89   };
90   constexpr usize n_alnum = sizeof(alnum) / sizeof(alnum[0]);
91 
92   usize count = 0;
93   for ( usize i = 0; i < n_alnum; ++i )
94     if ( __key_bit_set(words, n_words, static_cast<i32>(alnum[i])) ) ++count;
95   return count >= 30;
96 }

158 inline vector<device_t>
159 get_devices()
160 {
161   micron::vector<device_t> vc;
162   auto ps = micron::io::path("/sys/class/input/").all();
163   for ( auto &n : ps ) {
164     auto loc = format::find(n, "event");
165     if ( loc == nullptr ) continue;
166 
167     // /sys/class/input/[eventX]/device/uevent
168     micron::ustr8 pt = "/sys/class/input/" + n + "/device/uevent";
169     micron::hstring<char> raw;
170     io::open_file(io::path_t(pt.c_str())) >> raw;
171     micron::ustr8 buf(raw);
172 
173     if ( !format::contains(buf, "EV") ) continue;
174     const i64 ev = hex_string_to_int64(__parse_device("EV", buf));
175     const i64 prop = hex_string_to_int64(__parse_device("PROP", buf));
176     const micron::ustr8 keybuf = __parse_device("KEY", buf);
177 
178     // capability bits (EV_*)
179     const bool has_key = ev & (i64(1) << ev_key);
180     const bool has_rel = ev & (i64(1) << ev_rel);
181     const bool has_rep = ev & (i64(1) << ev_rep);
182     const bool has_abs = ev & (i64(1) << ev_abs);
183     // input properties (INPUT_PROP_*)
184     const bool p_direct = prop & (i64(1) << input_prop_direct);
185     const bool p_stick = prop & (i64(1) << input_prop_pointing_stick);
186     // tool / button signatures (BTN_*)
187     const bool k_pen = __has_key_code(keybuf, btn_tool_pen);
188     const bool k_stylus = __has_key_code(keybuf, btn_stylus);
189     const bool k_finger = __has_key_code(keybuf, btn_tool_finger);
190     const bool k_joy = __has_key_code(keybuf, btn_joystick);
191     const bool k_pad = __has_key_code(keybuf, btn_gamepad);
192 
193     // classify by capability
194     // WARNING: full keyboard and a relative pointer (REL axes + buttons) must be decided
195     // before the abs branch, certain gaming mice & combo receivers expose an EV_ABS axis
196     type_t type = type_t::unknown;
197     if ( has_key && has_rep && __has_alnum_keys(keybuf) ) {
198       type = type_t::keyboard;
199     } else if ( has_rel && has_key ) {
200       type = p_stick ? type_t::pointing_stick : type_t::mouse;
201     } else if ( has_abs ) {
202       if ( k_pen || k_stylus )
203         type = type_t::tablet;      // pen digitizer (BTN_TOOL_PEN/STYLUS)
204       else if ( p_direct )
205         type = type_t::touchscreen;      // coords map onto a screen (INPUT_PROP_DIRECT)
206       else if ( k_finger )
207         type = type_t::touchpad;      // indirect finger pad (BTN_TOOL_FINGER)
208       else if ( k_joy || k_pad )
209         type = type_t::joystick;      // stick / gamepad (BTN_TRIGGER/BTN_GAMEPAD)
210     }
211     if ( type == type_t::unknown ) continue;      // power / lid / audio / WMI / etc: correctly ignored
212 
213     auto name = __parse_device("NAME", buf);
214     auto phys = __parse_device("PHYS", buf);
215     vc.emplace_back("/dev/input/" + n, "/sys/class/input/" + n, name, phys, type, -1);
216   }
217   return vc;
218 }

221 // WARNING: need either CAP_SYS_ADMIN or CAP_SYS_RAWIO to be able to read raw character devices.
222 // WILL FAIL OTHERWISE
223 inline int
224 nonblock_bind_device(device_t &dev)
225 {
226   if ( dev.i_path.empty() ) return -1;
227   int fd = static_cast<int>(posix::open(dev.i_path.c_str(), posix::o_nonblock | posix::o_rdonly));
228   if ( fd < 0 ) return fd;
229   if ( dev.bound_fd.open() ) posix::close(dev.bound_fd.fd);      // don't leak a prior binding
230   dev.bound_fd = fd;
231   return fd;
232 }
233 
234 inline int
235 bind_device(device_t &dev)
236 {
237   if ( dev.i_path.empty() ) return -1;
238   int fd = static_cast<int>(posix::open(dev.i_path.c_str(), posix::o_rdonly));
239   if ( fd < 0 ) return fd;
240   if ( dev.bound_fd.open() ) posix::close(dev.bound_fd.fd);      // don't leak a prior binding
241   dev.bound_fd = fd;
242 
243   return fd;
244 }

246 inline void
247 unbind_device(device_t &dev)
248 {
249   // iff the fd is actually open
250   if ( dev.bound_fd.open() ) posix::close(dev.bound_fd.fd);
251   dev.bound_fd = -1;
252 }
