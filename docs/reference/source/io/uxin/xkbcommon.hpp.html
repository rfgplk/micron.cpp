113 class keymap_state
114 {
115   context_t *__ctx = nullptr;
116   keymap_t *__keymap = nullptr;
117   state_t *__state = nullptr;
118 
119 public:
120   ~keymap_state() { reset(); }
121 
122   keymap_state() = default;
123   keymap_state(const keymap_state &) = delete;
124   keymap_state &operator=(const keymap_state &) = delete;
125 
126   // Parse a keymap from a memory-mapped wl_keyboard.keymap fd.
127   void
128   load(const char *keymap_string)
129   {
130     auto &xl = xkb_lib();
131     reset();
132     __ctx = xl.xkb_context_new(XKB_CONTEXT_NO_FLAGS);
133     if ( !__ctx ) exc<except::library_error>("xkbcommon: xkb_context_new failed");
134     __keymap = xl.xkb_keymap_new_from_string(__ctx, keymap_string, XKB_KEYMAP_FORMAT_TEXT_V1, XKB_KEYMAP_COMPILE_NO_FLAGS);
135     if ( !__keymap ) {
136       xl.xkb_context_unref(__ctx);
137       __ctx = nullptr;
138       exc<except::library_error>("xkbcommon: xkb_keymap_new_from_string failed — keymap may be malformed");
139     }
140     __state = xl.xkb_state_new(__keymap);
141     if ( !__state ) {
142       xl.xkb_keymap_unref(__keymap);
143       __keymap = nullptr;
144       xl.xkb_context_unref(__ctx);
145       __ctx = nullptr;
146       exc<except::library_error>("xkbcommon: xkb_state_new failed");
147     }
148   }
149 
150   // Apply the mask from a wl_keyboard.modifiers event.
151   void
152   update_mods(u32 depressed, u32 latched, u32 locked, u32 group) noexcept
153   {
154     if ( !__state ) return;
155     xkb_lib().xkb_state_update_mask(__state, depressed, latched, locked, 0, 0, group);
156   }
157 
158   // Apply a key press/release. evdev_code is the raw KEY_* from wl_keyboard;
159   // xkbcommon expects keycode + 8 (X11 convention).
160   void
161   update_key(u32 evdev_code, bool pressed) noexcept
162   {
163     if ( !__state ) return;
164     xkb_lib().xkb_state_update_key(__state, evdev_code + 8, pressed ? XKB_KEY_DOWN : XKB_KEY_UP);
165   }
166 
167   keysym_t
168   translate_key(u32 evdev_code) const noexcept
169   {
170     if ( !__state ) return XKB_KEY_NoSymbol;
171     return xkb_lib().xkb_state_key_get_one_sym(__state, evdev_code + 8);
172   }
173 
174   u32
175   translate_utf32(u32 evdev_code) const noexcept
176   {
177     if ( !__state ) return 0;
178     return xkb_lib().xkb_state_key_get_utf32(__state, evdev_code + 8);
179   }
180 
181   bool
182   valid() const noexcept
183   {
184     return __state != nullptr;
185   }
186 
187   explicit
188   operator bool() const noexcept
189   {
190     return valid();
191   }
192 
193   void
194   reset() noexcept
195   {
196     if ( !__ctx && !__keymap && !__state ) return;
197     auto &xl = xkb_lib();
198     if ( __state ) xl.xkb_state_unref(__state);
199     if ( __keymap ) xl.xkb_keymap_unref(__keymap);
200     if ( __ctx ) xl.xkb_context_unref(__ctx);
201     __state = nullptr;
202     __keymap = nullptr;
203     __ctx = nullptr;
204   }
205 };
