245 inline auto
246 prepare_generic_keyboard_us(void) -> input_packet_t
247 {
248   const byte __first = key_esc;
249   const byte __end = key_kpdot;
250   const size_t __count = static_cast<size_t>(__end - __first);
251   input_packet_t __input{ ev_key, slice<button_t>(__count), nullptr, nullptr, nullptr };
252 
253   for ( size_t i = 0; i < __count; ++i ) __input.button_mask[i] = static_cast<button_t>(__first + i);
254 
255   return __input;
256 };

272 inline bool
273 read_raw(const device_t &dev, input_event &event)
274 {
275   if ( dev.bound_fd.has_error() ) return false;
276 
277   i64 read_bytes = posix::read(dev.bound_fd.fd, &event, sizeof(input_event));
278   return (read_bytes == sizeof(input_event));
279 };
