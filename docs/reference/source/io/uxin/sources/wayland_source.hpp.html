12 // Compatibility policy for existing users of the system XKB boundary.
13 namespace micron::uxin
14 {
15 struct system_keymap_decoder: xkb::keymap_state {
16   bool
17   prepare() noexcept
18   {
19 #if !defined(__micron_freestanding) || defined(__micron_eh)
20     try {
21       (void)xkb::xkb_lib();
22       return true;
23     } catch ( ... ) {
24       return false;
25     }
26 #else
27     (void)xkb::xkb_lib();
28     return true;
29 #endif
30   }
31 
32   i32
33   load_map(raw_slice<const char> bytes) noexcept
34   {
35     if ( !bytes.len || bytes.ptr[bytes.len - 1] ) return -micron::error::invalid_arg;
36 #if !defined(__micron_freestanding) || defined(__micron_eh)
37     try {
38       load(bytes.ptr);
39       return 0;
40     } catch ( ... ) {
41       return -micron::error::invalid_arg;
42     }
43 #else
44     load(bytes.ptr);
45     return 0;
46 #endif
47   }
48 };
