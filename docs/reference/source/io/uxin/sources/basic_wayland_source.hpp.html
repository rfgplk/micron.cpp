29   static void
30   __on_global(void *data, gfx::platform::wl_registry *registry, u32 name, const char *interface, u32 version) noexcept
31   {
32     auto *self = static_cast<basic_wayland_source *>(data);
33     auto streq = [](const char *a, const char *b) {
34       while ( *a && *a == *b ) {
35         ++a;
36         ++b;
37       }
38       return *a == 0 && *b == 0;
39     };
40     if ( self->on_global ) self->on_global(self->user, registry, name, interface, version);
41     if ( streq(interface, "wl_seat") && !self->__seat ) {
42       const u32 use_ver = version < 7 ? version : 7u;
43       void *p = self->__lib->wl_proxy_marshal_flags(reinterpret_cast<gfx::platform::wl_proxy *>(registry), 0 /*wl_registry.bind*/,
44                                                     &__wl_seat_interface, use_ver, 0, name, "wl_seat", use_ver, nullptr);
45       self->__seat = reinterpret_cast<gfx::platform::wl_seat *>(p);
46       self->__seat_version = use_ver;
47       self->seat_name_ = name;
48 
49       static wl_seat_listener_t listener{ &__on_seat_capabilities, &__on_seat_name };
50       self->__lib->wl_proxy_add_listener(reinterpret_cast<gfx::platform::wl_proxy *>(self->__seat),
51                                          reinterpret_cast<void (**)(void)>(&listener), self);
52     }
53   }

55   static void
56   __on_global_remove(void *data, gfx::platform::wl_registry *, u32 name) noexcept
57   {
58     auto *self = static_cast<basic_wayland_source *>(data);
59     if ( name == self->seat_name_ ) {
60       self->drop_keyboard();
61       self->drop_pointer();
62       if ( self->__seat ) {
63         if ( self->__seat_version < 5 )
64           self->__lib->wl_proxy_destroy(reinterpret_cast<gfx::platform::wl_proxy *>(self->__seat));
65         else
66           self->__lib->wl_proxy_marshal_flags(reinterpret_cast<gfx::platform::wl_proxy *>(self->__seat), wl_seat_ops::wl_seat_release,
67                                               nullptr, self->__seat_version, 1);
68       }
69       self->__seat = nullptr;
70       self->error_ = -error::no_device;
71     }
72     if ( self->on_global_remove ) self->on_global_remove(self->user, name);
73   }

75   static void
76   __on_seat_capabilities(void *data, gfx::platform::wl_seat *seat, u32 caps) noexcept
77   {
78     auto *self = static_cast<basic_wayland_source *>(data);
79     if ( !(caps & wl_seat_capability_pointer) ) self->drop_pointer();
80     if ( !(caps & wl_seat_capability_keyboard) ) self->drop_keyboard();
81     if ( (caps & wl_seat_capability_pointer) && !self->pointer_ ) {
82       void *p = self->__lib->wl_proxy_marshal_flags(reinterpret_cast<gfx::platform::wl_proxy *>(seat), wl_seat_ops::wl_seat_get_pointer,
83                                                     &__wl_pointer_interface, self->__seat_version, 0, nullptr);
84       self->pointer_ = reinterpret_cast<gfx::platform::wl_pointer *>(p);
85       static wl_pointer_listener_t plistener{
86         &__on_p_enter,       &__on_p_leave,     &__on_p_motion,        &__on_p_button,    &__on_p_axis,         &__on_p_frame,
87         &__on_p_axis_source, &__on_p_axis_stop, &__on_p_axis_discrete, &__on_p_axis_v120, &__on_p_axis_rel_dir,
88       };
89       self->__lib->wl_proxy_add_listener(reinterpret_cast<gfx::platform::wl_proxy *>(self->pointer_),
90                                          reinterpret_cast<void (**)(void)>(&plistener), self);
91     }
92     if ( (caps & wl_seat_capability_keyboard) && !self->keyboard_ ) {
93       void *p = self->__lib->wl_proxy_marshal_flags(reinterpret_cast<gfx::platform::wl_proxy *>(seat), wl_seat_ops::wl_seat_get_keyboard,
94                                                     &__wl_keyboard_interface, self->__seat_version, 0, nullptr);
95       self->keyboard_ = reinterpret_cast<gfx::platform::wl_keyboard *>(p);
96       static wl_keyboard_listener_t klistener{
97         &__on_k_keymap, &__on_k_enter, &__on_k_leave, &__on_k_key, &__on_k_modifiers, &__on_k_repeat_info,
98       };
99       self->__lib->wl_proxy_add_listener(reinterpret_cast<gfx::platform::wl_proxy *>(self->keyboard_),
100                                          reinterpret_cast<void (**)(void)>(&klistener), self);
101     }

128   static void
129   __on_p_motion(void *data, gfx::platform::wl_pointer *, u32 time, i32 sx_fixed, i32 sy_fixed) noexcept
130   {
131     auto *self = static_cast<basic_wayland_source *>(data);
132     if ( !self->__pointer_inside ) return;
133     if ( self->on_position ) self->on_position(self->user, float(sx_fixed) / 256.f, float(sy_fixed) / 256.f);
134     if ( !self->on_motion ) return;
135     input_event ev{};
136     ev.time.tv_sec = static_cast<long>(time / 1000);
137     ev.time.tv_usec = static_cast<long>((time % 1000) * 1000);
138     ev.type = ev_rel;
139     ev.code = rel_x;
140     ev.value = sx_fixed >> 8;
141     self->on_motion(self->user, ev);
142     ev.code = rel_y;
143     ev.value = sy_fixed >> 8;
144     self->on_motion(self->user, ev);
145   }

149   {

209   // keyboard callbacks
210   static void
211   __on_k_keymap(void *data, gfx::platform::wl_keyboard *, u32 format, i32 fd, u32 size) noexcept
212   {
213     auto *self = static_cast<basic_wayland_source *>(data);
214     if ( !self->__xkb_available || fd < 0 || size == 0 || size > 4 * 1024 * 1024 || format != 1u ) {
215       if ( fd >= 0 ) posix::close(fd);
216       self->__keymap.reset();
217       self->error_ = size > 4 * 1024 * 1024 ? -error::message_too_long : -error::op_not_supported;
218       return;
219     }
220     // An owned read also makes truncated/raced descriptors an error rather than a SIGBUS mapping fault.
221     try {
222       micron::vector<char> bytes;
223       bytes.resize(size);
224       usize done = 0;
225       self->error_ = 0;
226       while ( done < size ) {
227         const auto count = posix::pread(fd, bytes.data() + done, size - done, static_cast<posix::off64_t>(done));
228         if ( count == -error::interrupted ) continue;
229         if ( count <= 0 ) {
230           self->error_ = count < 0 ? static_cast<i32>(count) : -error::io_error;
231           break;
232         }
233         done += static_cast<usize>(count);
234       }
235       if ( !self->error_ ) self->error_ = self->__keymap.load_map({ bytes.data(), bytes.size() });
236     } catch ( ... ) {
237       self->error_ = -error::out_of_memory;
238     }
239     if ( self->error_ ) self->__keymap.reset();
240     posix::close(fd);
241   }

259   static void
260   __on_k_key(void *data, gfx::platform::wl_keyboard *, u32 serial, u32 time, u32 key, u32 state) noexcept
261   {
262     auto *self = static_cast<basic_wayland_source *>(data);
263     const bool pressed = (state == wl_keyboard_key_state_pressed);
264     if ( !self->__keyboard_inside ) return;
265     if ( self->on_serial ) self->on_serial(self->user, serial);
266 
267     if ( self->on_key ) {
268       input_event ev{};
269       ev.time.tv_sec = static_cast<long>(time / 1000);
270       ev.time.tv_usec = static_cast<long>((time % 1000) * 1000);
271       ev.type = ev_key;
272       ev.code = static_cast<u16>(key);
273       ev.value = pressed ? 1 : 0;
274       self->on_key(self->user, ev);
275     }
276 
277     if ( self->__keymap.valid() ) {
278       self->__keymap.update_key(key, pressed);
279       if ( self->on_keysym ) {
280         u32 sym = self->__keymap.translate_key(key);
281         u32 utf32 = self->__keymap.translate_utf32(key);
282         self->on_keysym(self->user, key, sym, utf32, pressed);
283       }
284     }
285   }

318   drop_keyboard() noexcept
319   {
320     if ( !keyboard_ ) return;
321     if ( __seat_version < 3 )
322       __lib->wl_proxy_destroy(reinterpret_cast<gfx::platform::wl_proxy *>(keyboard_));

328     __keymap.reset();
329     if ( on_keyboard_focus ) on_keyboard_focus(user, false);

354   // NOTE: in terminal sessions that are run by a different user than host, this will fail, without reconfig
355   void
356   bind(gfx::platform::display &dpy, gfx::platform::wl_surface *surface = nullptr, bool synchronous = true)
357   {
358     if ( dpy.backend() != gfx::platform::backend_tag_t::wayland ) {
359       exc<except::logic_error>("basic_wayland_source: gfx::platform::display is not Wayland");
360     }
361     __dpy = dpy.as_wayland();
362     if ( !__dpy || !__dpy->lib() ) exc<except::logic_error>("basic_wayland_source: wayland_display_t missing");
363     __lib = __dpy->lib();
364     __surface = surface;
365     __registry = __lib->wl_display_get_registry(__dpy->display());
366     if ( !__registry ) exc<except::library_error>("basic_wayland_source: registry unavailable");
367 
368     static gfx::platform::wl_registry_listener_t reg_listener{ &__on_global, &__on_global_remove };
369     __lib->wl_proxy_add_listener(reinterpret_cast<gfx::platform::wl_proxy *>(__registry), reinterpret_cast<void (**)(void)>(&reg_listener),
370                                  this);
371     __xkb_available = __keymap.prepare();
372     if ( synchronous ) {
373       __lib->wl_display_roundtrip(__dpy->display());
374       __lib->wl_display_roundtrip(__dpy->display());
375       if ( !__seat ) exc<except::library_error>("wayland_source: compositor advertises no input");
376     }
377   }

412   void
413   release() noexcept
414   {
415     if ( !__lib ) return;
416     drop_keyboard();
417     drop_pointer();
418     if ( __seat ) {
419       if ( __seat_version < 5 )
420         __lib->wl_proxy_destroy(reinterpret_cast<gfx::platform::wl_proxy *>(__seat));
421       else
422         __lib->wl_proxy_marshal_flags(reinterpret_cast<gfx::platform::wl_proxy *>(__seat), wl_seat_ops::wl_seat_release, nullptr,
423                                       __seat_version, 1);
424       __seat = nullptr;
425     }
426     if ( __registry ) __lib->wl_proxy_destroy(reinterpret_cast<gfx::platform::wl_proxy *>(__registry));
427     __registry = nullptr;
428     __lib = nullptr;
429     __keymap.reset();
430     __dpy = nullptr;
431     __surface = nullptr;
432     __pointer_inside = __keyboard_inside = false;
433   }
