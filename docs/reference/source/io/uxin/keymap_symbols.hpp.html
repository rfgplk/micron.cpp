16 constexpr bool
17 same(raw_slice<const char> text, const char *name) noexcept
18 {
19   usize i = 0;
20   for ( ; i < text.len && name[i]; ++i )
21     if ( text.ptr[i] != name[i] ) return false;
22   return i == text.len && !name[i];
23 }

25 constexpr u32
26 unicode_symbol(u32 cp) noexcept
27 {
28   return cp <= 0xff ? cp : cp <= 0x10ffff && !(cp >= 0xd800 && cp <= 0xdfff) ? cp | 0x1000000u : unknown_symbol;
29 }

298 constexpr u32
299 symbol_from_name(raw_slice<const char> name) noexcept
300 {
301   if ( name.len == 1 ) return static_cast<u8>(name.ptr[0]);
302   if ( name.len >= 5 && name.ptr[0] == 'U' ) {
303     u32 cp = 0;
304     bool hexadecimal = true, overflow = false;
305     for ( usize i = 1; i < name.len; ++i ) {
306       const char c = name.ptr[i];
307       const u32 digit = c >= '0' && c <= '9' ? c - '0' : c >= 'a' && c <= 'f' ? c - 'a' + 10 : c >= 'A' && c <= 'F' ? c - 'A' + 10 : 16;
308       if ( digit == 16 ) {
309         hexadecimal = false;
310         break;
311       }
312       if ( cp > (0x10ffffu - digit) / 16 ) {
313         overflow = true;
314         break;
315       }
316       cp = cp * 16 + digit;
317     }
318     if ( overflow ) return unknown_symbol;
319     if ( hexadecimal ) return unicode_symbol(cp);
320   }
321   if ( name.len >= 2 && name.len <= 3 && name.ptr[0] == 'F' ) {
322     u32 n = 0;
323     for ( usize i = 1; i < name.len; ++i ) {
324       if ( name.ptr[i] < '0' || name.ptr[i] > '9' ) return unknown_symbol;
325       n = n * 10 + name.ptr[i] - '0';
326     }
327     if ( n >= 1 && n <= 35 ) return 0xffbd + n;
328   }
329   if ( name.len == 4 && name.ptr[0] == 'K' && name.ptr[1] == 'P' && name.ptr[2] == '_' && name.ptr[3] >= '0' && name.ptr[3] <= '9' )
330     return 0xffb0 + name.ptr[3] - '0';
331   for ( const auto &entry : symbols )
332     if ( same(name, entry.name) ) return entry.value;
333   for ( const auto &entry : latin )
334     if ( same(name, entry.name) ) return entry.symbol;
335   return unknown_symbol;
336 }

338 constexpr u32
339 symbol_unicode(u32 symbol) noexcept
340 {
341   if ( (symbol >= 32 && symbol <= 126) || (symbol >= 0xa0 && symbol <= 0xff) ) return symbol;
342   if ( symbol >= 0x1000100 && symbol <= 0x110ffff ) {
343     const u32 cp = symbol & 0xffffff;
344     return cp >= 0xd800 && cp <= 0xdfff ? 0 : cp;
345   }
346   if ( symbol >= 0xffb0 && symbol <= 0xffb9 ) return '0' + symbol - 0xffb0;
347   if ( symbol == 0xff80 ) return ' ';
348   if ( symbol >= 0xffaa && symbol <= 0xffaf ) {
349     constexpr char operators[] = "*+,-./";
350     return operators[symbol - 0xffaa];
351   }
352   if ( symbol == 0xffbd ) return '=';
353   if ( symbol == 0x20ac ) return symbol;
354   if ( symbol >= 0x13bc && symbol <= 0x13be ) {
355     constexpr u32 extra[] = { 0x152, 0x153, 0x178 };
356     return extra[symbol - 0x13bc];
357   }
358   for ( const auto &entry : latin )
359     if ( entry.symbol == symbol ) return entry.unicode;
360   return 0;
361 }
