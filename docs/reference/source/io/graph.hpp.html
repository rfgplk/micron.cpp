45 struct native_property_codec {
46   template<typename P>
47     requires micron::is_trivially_copyable_v<P>
48   [[nodiscard]] bool
49   encode(const P &property, micron::vector<byte, micron::allocator_serial<>, false> &output) const
50   {
51     const byte *data = reinterpret_cast<const byte *>(micron::addressof(property));
52     output.reserve(output.size() + sizeof(P));
53     for ( usize i = 0; i < sizeof(P); ++i ) output.push_back(data[i]);
54     return true;
55   }
56 
57   template<typename P>
58     requires micron::is_trivially_copyable_v<P>
59   [[nodiscard]] bool
60   decode(const byte *data, usize size, P &property) const noexcept
61   {
62     if ( size != sizeof(P) ) return false;
63     byte *output = reinterpret_cast<byte *>(micron::addressof(property));
64     for ( usize i = 0; i < sizeof(P); ++i ) output[i] = data[i];
65     return true;
66   }
67 
68   template<typename P>
69     requires micron::is_trivially_copyable_v<P>
70   [[nodiscard]] bool
71   decode_construct(const byte *data, usize size, void *output) const noexcept
72   {
73     if ( size != sizeof(P) || output == nullptr ) return false;
74 
75     struct bytes_type {
76       byte value[sizeof(P)];
77     } bytes{};
78 
79     for ( usize i = 0; i < sizeof(P); ++i ) bytes.value[i] = data[i];
80     micron::construct_at(reinterpret_cast<P *>(output), __builtin_bit_cast(P, bytes));
81     return true;
82   }
83 };

107 inline bool
108 next_token(const char *line, usize size, usize &position, token &out) noexcept
109 {
110   while ( position < size && (line[position] == ' ' || line[position] == '\t' || line[position] == '\r') ) ++position;
111   if ( position >= size || line[position] == '#' || line[position] == '%' ) return false;
112   const usize begin = position;
113   while ( position < size && line[position] != ' ' && line[position] != '\t' && line[position] != '\r' && line[position] != '#'
114           && line[position] != '%' )
115     ++position;
116   out = token{ line + begin, position - begin, begin + 1 };
117   return out.size != 0;
118 }

135 template<typename G>
136 bool
137 insert(G &graph, typename G::index_type source, typename G::index_type target, const token *property, parse_status &status)
138 {
139   math::edge_insert_result<typename G::index_type> inserted;
140   if constexpr ( math::graphs::weighted_bundle<typename G::edge_property_type> ) {
141     using weight_type = typename G::edge_property_type::weight_type;
142     if ( property == nullptr ) {
143       status.code = parse_code::missing_endpoint;
144       return false;
145     }
146     weight_type weight{};
147     if ( !parse_number(*property, weight) ) {
148       status.code = parse_code::invalid_token;
149       status.column = property->column;
150       return false;
151     }
152     inserted = graph.add_edge(source, target, weight);
153   } else {
154     inserted = graph.add_edge(source, target);
155   }
156   if ( !inserted.inserted() ) {
157     status.code = insertion_code(inserted.status);
158     status.insertion = inserted.status;
159     return false;
160   }
161   return true;
162 }

209 inline bool
210 get_u16(const byte *data, usize size, usize &position, u16 &value)
211 {
212   if ( size - position < 2 ) return false;
213   value = static_cast<u16>(data[position]) | static_cast<u16>(static_cast<u16>(data[position + 1]) << 8u);
214   position += 2;
215   return true;
216 }

218 inline bool
219 get_u64(const byte *data, usize size, usize &position, u64 &value)
220 {
221   if ( size - position < 8 ) return false;
222   value = 0;
223   for ( usize i = 0; i < 8; ++i ) value |= static_cast<u64>(data[position + i]) << (i * 8u);
224   position += 8;
225   return true;
226 }

249 template<typename Codec, typename P>
250 bool
251 get_property(const byte *data, usize size, usize &position, const Codec &codec, P &property, bool &truncated)
252 {
253   u64 encoded_size{};
254   if ( !get_u64(data, size, position, encoded_size) || encoded_size > static_cast<u64>(size - position) ) {
255     truncated = true;
256     return false;
257   }
258   const bool decoded = codec.decode(data + position, static_cast<usize>(encoded_size), property);
259   position += static_cast<usize>(encoded_size);
260   return decoded;
261 }

283 template<typename Codec, typename P>
284 bool
285 get_property(const byte *data, usize size, usize &position, const Codec &codec, decoded_property<P> &property, bool &truncated)
286 {
287   u64 encoded_size{};
288   if ( !get_u64(data, size, position, encoded_size) || encoded_size > static_cast<u64>(size - position) ) {
289     truncated = true;
290     return false;
291   }
292   bool decoded = false;
293   if constexpr ( requires { codec.template decode_construct<P>(data, usize{}, static_cast<void *>(property.get())); } ) {
294     decoded = codec.template decode_construct<P>(data + position, static_cast<usize>(encoded_size), property.get());
295     property.live = decoded;
296   } else if constexpr ( micron::is_default_constructible_v<P> ) {
297     micron::construct_at(property.get());
298     property.live = true;
299     decoded = codec.decode(data + position, static_cast<usize>(encoded_size), *property.get());
300   }
301   position += static_cast<usize>(encoded_size);
302   return decoded;
303 }

307 template<typename G = math::graph<>>
308 [[nodiscard]] parse_result<G>
309 parse_edge_list(const char *data, usize size)
310 {
311   parse_result<G> result;
312   usize position = 0;
313   usize line_number = 1;
314   while ( position < size ) {
315     const usize line_start = position;
316     while ( position < size && data[position] != '\n' ) ++position;
317     const usize line_size = position - line_start;
318     usize cursor = 0;
319     __impl::token source_token{}, target_token{}, property_token{};
320     if ( __impl::next_token(data + line_start, line_size, cursor, source_token) ) {
321       if ( !__impl::next_token(data + line_start, line_size, cursor, target_token) )
322         return __impl::fail<G>({ parse_code::missing_endpoint, line_start + cursor, line_number, cursor + 1 });
323       typename G::index_type source{}, target{};
324       if ( !__impl::parse_number(source_token, source) )
325         return __impl::fail<G>({ parse_code::invalid_token, line_start + source_token.column - 1, line_number, source_token.column });
326       if ( !__impl::parse_number(target_token, target) )
327         return __impl::fail<G>({ parse_code::invalid_token, line_start + target_token.column - 1, line_number, target_token.column });
328       const bool has_property = __impl::next_token(data + line_start, line_size, cursor, property_token);
329       result.status = { parse_code::ok, line_start, line_number, source_token.column };
330       if ( !__impl::insert(result.value, source, target, has_property ? micron::addressof(property_token) : nullptr, result.status) ) {
331         result.status.byte = line_start + result.status.column - 1;
332         result.status.line = line_number;
333         result.value.clear();
334         return result;
335       }
336     }
337     if ( position < size ) ++position;
338     ++line_number;
339   }
340   return result;
341 }

350 template<typename G = math::graph<>>
351 [[nodiscard]] parse_result<G>
352 parse_adjacency_list(const char *data, usize size)
353 {
354   parse_result<G> result;
355   usize position = 0;
356   usize line_number = 1;
357   while ( position < size ) {
358     const usize line_start = position;
359     while ( position < size && data[position] != '\n' ) ++position;
360     const usize line_size = position - line_start;
361     usize cursor = 0;
362     __impl::token source_token{};
363     if ( __impl::next_token(data + line_start, line_size, cursor, source_token) ) {
364       bool colon = source_token.size && source_token.data[source_token.size - 1] == ':';
365       if ( colon ) --source_token.size;
366       typename G::index_type source{};
367       if ( !__impl::parse_number(source_token, source) )
368         return __impl::fail<G>({ parse_code::invalid_token, line_start + source_token.column - 1, line_number, source_token.column });
369       if ( !result.value.has_vertex(source) ) {
370         if constexpr ( micron::is_default_constructible_v<typename G::vertex_property_type> ) {
371           while ( result.value.vertex_slots() <= static_cast<usize>(source) )
372             if ( !result.value.add_vertex().valid() ) return __impl::fail<G>({ parse_code::index_overflow, line_start, line_number, 1 });
373         }
374       }
375       __impl::token target_token{};
376       while ( __impl::next_token(data + line_start, line_size, cursor, target_token) ) {
377         if ( target_token.size == 1 && target_token.data[0] == ':' ) continue;
378         typename G::index_type target{};
379         if ( !__impl::parse_number(target_token, target) ) {
380           result.value.clear();
381           return __impl::fail<G>({ parse_code::invalid_token, line_start + target_token.column - 1, line_number, target_token.column });
382         }
383         result.status = { parse_code::ok, line_start, line_number, target_token.column };
384         if ( !__impl::insert(result.value, source, target, nullptr, result.status) ) {
385           result.value.clear();
386           return result;
387         }
388       }
389     }
390     if ( position < size ) ++position;
391     ++line_number;
392   }
393   return result;
394 }

403 template<typename G = math::graph<>>
404 [[nodiscard]] parse_result<G>
405 parse_matrix_market(const char *data, usize size)
406 {
407   // Coordinate Matrix Market: banner, optional '%' comments, dimensions,
408   // followed by one-based row/column[/value] records.
409   usize position = 0;
410   usize line = 1;
411   bool dimensions = false;
412   usize expected = 0;
413   usize records = 0;
414   parse_result<G> result;
415   while ( position < size ) {
416     const usize start = position;
417     while ( position < size && data[position] != '\n' ) ++position;
418     const usize length = position - start;
419     if ( length && data[start] != '%' ) {
420       usize cursor = 0;
421       __impl::token a{}, b{}, c{};
422       if ( __impl::next_token(data + start, length, cursor, a) ) {
423         if ( !__impl::next_token(data + start, length, cursor, b) )
424           return __impl::fail<G>({ parse_code::missing_endpoint, start, line, 1 });
425         if ( !dimensions ) {
426           if ( !__impl::next_token(data + start, length, cursor, c) || !__impl::parse_number(c, expected) )
427             return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
428           usize rows{}, columns{};
429           if ( !__impl::parse_number(a, rows) || !__impl::parse_number(b, columns) )
430             return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
431           const usize vertices = rows > columns ? rows : columns;
432           if constexpr ( micron::is_default_constructible_v<typename G::vertex_property_type> ) (void)result.value.add_vertices(vertices);
433           dimensions = true;
434         } else {
435           typename G::index_type row{}, column{};
436           if ( !__impl::parse_number(a, row) || !__impl::parse_number(b, column) || row == 0 || column == 0 )
437             return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
438           --row;
439           --column;
440           const bool has_value = __impl::next_token(data + start, length, cursor, c);
441           result.status = { parse_code::ok, start, line, 1 };
442           if ( !__impl::insert(result.value, row, column, has_value ? micron::addressof(c) : nullptr, result.status) ) {
443             result.value.clear();
444             return result;
445           }
446           ++records;
447         }
448       }
449     }
450     if ( position < size ) ++position;
451     ++line;
452   }
453   if ( !dimensions || records != expected ) return __impl::fail<G>({ parse_code::truncated, position, line, 1 });
454   return result;
455 }

464 template<typename G = math::graph<>>
465 [[nodiscard]] parse_result<G>
466 parse_dimacs(const char *data, usize size)
467 {
468   parse_result<G> result;
469   usize position = 0;
470   usize line = 1;
471   while ( position < size ) {
472     const usize start = position;
473     while ( position < size && data[position] != '\n' ) ++position;
474     const usize length = position - start;
475     usize cursor = 0;
476     __impl::token kind{};
477     if ( __impl::next_token(data + start, length, cursor, kind) && !(kind.size == 1 && kind.data[0] == 'c') ) {
478       if ( kind.size == 1 && kind.data[0] == 'p' ) {
479         __impl::token type{}, vertices{}, edges{};
480         if ( !__impl::next_token(data + start, length, cursor, type) || !__impl::next_token(data + start, length, cursor, vertices)
481              || !__impl::next_token(data + start, length, cursor, edges) )
482           return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
483         usize count{};
484         if ( !__impl::parse_number(vertices, count) ) return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
485         (void)result.value.add_vertices(count);
486       } else if ( kind.size == 1 && (kind.data[0] == 'e' || kind.data[0] == 'a') ) {
487         __impl::token a{}, b{}, weight{};
488         if ( !__impl::next_token(data + start, length, cursor, a) || !__impl::next_token(data + start, length, cursor, b) )
489           return __impl::fail<G>({ parse_code::missing_endpoint, start, line, 1 });
490         typename G::index_type u{}, v{};
491         if ( !__impl::parse_number(a, u) || !__impl::parse_number(b, v) || u == 0 || v == 0 )
492           return __impl::fail<G>({ parse_code::invalid_token, start, line, 1 });
493         --u;
494         --v;
495         const bool has_weight = __impl::next_token(data + start, length, cursor, weight);
496         result.status = { parse_code::ok, start, line, 1 };
497         if ( !__impl::insert(result.value, u, v, has_weight ? micron::addressof(weight) : nullptr, result.status) ) {
498           result.value.clear();
499           return result;
500         }
501       }
502     }
503     if ( position < size ) ++position;
504     ++line;
505   }
506   return result;
507 }

603 [[nodiscard]] micron::vector<byte, micron::allocator_serial<>, false>
604 binary(const G &graph)
605   requires(micron::is_same_v<typename G::vertex_property_type, math::empty_property>
606            && micron::is_same_v<typename G::edge_property_type, math::empty_property>
607            && micron::is_same_v<typename G::graph_property_type, math::empty_property>)
608 {
609   micron::vector<byte, micron::allocator_serial<>, false> output;
610   output.reserve(44 + graph.vertices_count() * 8 + graph.edges_count() * 24);
611   output.push_back('M');
612   output.push_back('C');
613   output.push_back('G');
614   output.push_back('F');
615   __impl::put_u16(output, 3);
616   output.push_back(1);      // little endian
617   output.push_back(static_cast<byte>(sizeof(typename G::index_type)));
618   output.push_back(static_cast<byte>((G::is_directed ? 1u : 0u) | (G::is_simple ? 0u : 2u) | (G::allows_loops ? 4u : 0u)));
619   output.push_back(0);      // no property records
620   output.push_back(0);
621   output.push_back(0);
622   __impl::put_u64(output, graph.vertex_slots());
623   __impl::put_u64(output, graph.edge_slots());
624   __impl::put_u64(output, graph.vertices_count());
625   __impl::put_u64(output, graph.edges_count());
626   for ( usize slot = 0; slot < graph.vertex_slots(); ++slot ) {
627     const typename G::vertex_descriptor vertex(static_cast<typename G::index_type>(slot));
628     if ( graph.has_vertex(vertex) ) __impl::put_u64(output, slot);
629   }
630   for ( usize slot = 0; slot < graph.edge_slots(); ++slot ) {
631     const typename G::edge_descriptor id(static_cast<typename G::index_type>(slot));
632     if ( graph.has_edge(id) ) {
633       __impl::put_u64(output, slot);
634       __impl::put_u64(output, static_cast<u64>(graph.source(id).value));
635       __impl::put_u64(output, static_cast<u64>(graph.target(id).value));
636     }
637   }
638   return output;
639 }

641 template<math::graphs::graph_model G, typename Codec>
642 [[nodiscard]] micron::vector<byte, micron::allocator_serial<>, false>
643 binary(const G &graph, const Codec &codec)
644   requires requires(micron::vector<byte, micron::allocator_serial<>, false> &output,
645                     const typename G::vertex_property_type &vertex_property, const typename G::edge_property_type &edge_property,
646                     const typename G::graph_property_type &graph_property) {
647     { codec.encode(vertex_property, output) } -> micron::convertible_to<bool>;
648     { codec.encode(edge_property, output) } -> micron::convertible_to<bool>;
649     { codec.encode(graph_property, output) } -> micron::convertible_to<bool>;
650   }
651 {
652   micron::vector<byte, micron::allocator_serial<>, false> output;
653   output.reserve(64 + graph.vertices_count() * 16 + graph.edges_count() * 40);
654   output.push_back('M');
655   output.push_back('C');
656   output.push_back('G');
657   output.push_back('F');
658   __impl::put_u16(output, 3);
659   output.push_back(1);
660   output.push_back(static_cast<byte>(sizeof(typename G::index_type)));
661   output.push_back(static_cast<byte>((G::is_directed ? 1u : 0u) | (G::is_simple ? 0u : 2u) | (G::allows_loops ? 4u : 0u)));
662   output.push_back(1);      // typed property records present
663   output.push_back(0);
664   output.push_back(0);
665   __impl::put_u64(output, graph.vertex_slots());
666   __impl::put_u64(output, graph.edge_slots());
667   __impl::put_u64(output, graph.vertices_count());
668   __impl::put_u64(output, graph.edges_count());
669   if ( !__impl::put_property(output, codec, graph.graph_property()) ) return {};
670   for ( usize slot = 0; slot < graph.vertex_slots(); ++slot ) {
671     const typename G::vertex_descriptor vertex(static_cast<typename G::index_type>(slot));
672     if ( !graph.has_vertex(vertex) ) continue;
673     __impl::put_u64(output, slot);
674     if ( !__impl::put_property(output, codec, graph.vertex_property_unchecked(vertex)) ) return {};
675   }
676   for ( usize slot = 0; slot < graph.edge_slots(); ++slot ) {
677     const typename G::edge_descriptor edge(static_cast<typename G::index_type>(slot));
678     if ( !graph.has_edge(edge) ) continue;
679     __impl::put_u64(output, slot);
680     __impl::put_u64(output, static_cast<u64>(graph.source(edge).value));
681     __impl::put_u64(output, static_cast<u64>(graph.target(edge).value));
682     if ( !__impl::put_property(output, codec, graph.edge_property_unchecked(edge)) ) return {};
683   }
684   return output;
685 }

687 template<typename G = math::graph<>>
688 [[nodiscard]] parse_result<G>
689 parse_binary(const byte *data, usize size)
690   requires(micron::is_same_v<typename G::vertex_property_type, math::empty_property>
691            && micron::is_same_v<typename G::edge_property_type, math::empty_property>
692            && micron::is_same_v<typename G::graph_property_type, math::empty_property> && math::graphs::mutable_graph_model<G>)
693 {
694   parse_result<G> result;
695   if ( size < 44 ) return __impl::fail<G>({ parse_code::truncated, size, 1, size + 1 });
696   if ( data[0] != 'M' || data[1] != 'C' || data[2] != 'G' || data[3] != 'F' ) return __impl::fail<G>({ parse_code::corrupt, 0, 1, 1 });
697   usize position = 4;
698   u16 version{};
699   if ( !__impl::get_u16(data, size, position, version) ) return __impl::fail<G>({ parse_code::truncated, position });
700   if ( version != 3 ) return __impl::fail<G>({ parse_code::unknown_version, 4, 1, 5 });
701   const byte endian = data[position++];
702   const byte index_width = data[position++];
703   const byte flags = data[position++];
704   const byte property_flags = data[position++];
705   const byte reserved_b = data[position++];
706   const byte reserved_c = data[position++];
707   if ( (flags & ~byte(7)) != 0 || property_flags != 0 || reserved_b != 0 || reserved_c != 0 )
708     return __impl::fail<G>({ parse_code::corrupt, 9, 1, 10 });
709   const bool valid_index_width = index_width == 1 || index_width == 2 || index_width == 4 || index_width == 8;
710   if ( endian != 1 || !valid_index_width || index_width > sizeof(typename G::index_type) || static_cast<bool>(flags & 1u) != G::is_directed
711        || static_cast<bool>(flags & 2u) == G::is_simple || static_cast<bool>(flags & 4u) != G::allows_loops )
712     return __impl::fail<G>({ parse_code::incompatible_graph, 6, 1, 7 });
713   u64 vertex_slots{}, edge_slots{}, live_vertices{}, live_edges{};
714   if ( !__impl::get_u64(data, size, position, vertex_slots) || !__impl::get_u64(data, size, position, edge_slots)
715        || !__impl::get_u64(data, size, position, live_vertices) || !__impl::get_u64(data, size, position, live_edges) )
716     return __impl::fail<G>({ parse_code::truncated, position });
717   if ( vertex_slots > static_cast<u64>(math::vertex_id<typename G::index_type>::invalid_value())
718        || edge_slots > static_cast<u64>(math::edge_id<typename G::index_type>::invalid_value())
719        || vertex_slots > __impl::index_slot_limit(index_width) || edge_slots > __impl::index_slot_limit(index_width)
720        || vertex_slots > static_cast<u64>(micron::numeric_limits<usize>::max())
721        || edge_slots > static_cast<u64>(micron::numeric_limits<usize>::max()) )
722     return __impl::fail<G>({ parse_code::index_overflow, position });
723   if ( live_vertices > vertex_slots || live_edges > edge_slots ) return __impl::fail<G>({ parse_code::corrupt, position });
724   result.vertex_remap = decltype(result.vertex_remap)(static_cast<usize>(vertex_slots), math::vertex_id<typename G::index_type>::invalid());
725   result.edge_remap = decltype(result.edge_remap)(static_cast<usize>(edge_slots), math::edge_id<typename G::index_type>::invalid());
726   constexpr bool stable_target = micron::is_same_v<typename G::storage_type, math::graphs::stable_adjacency_t>;
727   u64 previous_vertex = micron::numeric_limits<u64>::max();
728   for ( u64 record = 0; record < live_vertices; ++record ) {
729     u64 slot{};
730     if ( !__impl::get_u64(data, size, position, slot) ) return __impl::fail<G>({ parse_code::truncated, position });
731     if ( slot >= vertex_slots || (record != 0 && slot <= previous_vertex) ) return __impl::fail<G>({ parse_code::corrupt, position - 8 });
732     if constexpr ( stable_target )
733       while ( result.value.vertex_slots() < static_cast<usize>(slot) )
734         if ( !result.value.__import_dead_vertex_slot() ) return __impl::fail<G>({ parse_code::index_overflow, position - 8 });
735     const auto inserted = result.value.add_vertex();
736     if ( !inserted.valid() ) return __impl::fail<G>({ parse_code::index_overflow, position - 8 });
737     result.vertex_remap.data()[static_cast<usize>(slot)] = inserted;
738     previous_vertex = slot;
739   }
740   if constexpr ( stable_target )
741     while ( result.value.vertex_slots() < static_cast<usize>(vertex_slots) )
742       if ( !result.value.__import_dead_vertex_slot() ) return __impl::fail<G>({ parse_code::index_overflow, position });
743 
744   u64 previous_edge = micron::numeric_limits<u64>::max();
745   for ( u64 record = 0; record < live_edges; ++record ) {
746     u64 slot{}, source{}, target{};
747     if ( !__impl::get_u64(data, size, position, slot) || !__impl::get_u64(data, size, position, source)
748          || !__impl::get_u64(data, size, position, target) )
749       return __impl::fail<G>({ parse_code::truncated, position });
750     if ( slot >= edge_slots || (record != 0 && slot <= previous_edge) || source >= vertex_slots || target >= vertex_slots
751          || !result.vertex_remap.data()[static_cast<usize>(source)].valid()
752          || !result.vertex_remap.data()[static_cast<usize>(target)].valid() )
753       return __impl::fail<G>({ parse_code::corrupt, position - 24 });
754     if constexpr ( stable_target )
755       while ( result.value.edge_slots() < static_cast<usize>(slot) )
756         if ( !result.value.__import_dead_edge_slot() ) return __impl::fail<G>({ parse_code::index_overflow, position - 24 });
757     auto inserted = result.value.add_edge(result.vertex_remap.data()[static_cast<usize>(source)],
758                                           result.vertex_remap.data()[static_cast<usize>(target)]);
759     if ( !inserted.inserted() ) return __impl::fail<G>({ parse_code::topology_rejected, position - 24 });
760     result.edge_remap.data()[static_cast<usize>(slot)] = inserted.id;
761     previous_edge = slot;
762   }
763   if constexpr ( stable_target )
764     while ( result.value.edge_slots() < static_cast<usize>(edge_slots) )
765       if ( !result.value.__import_dead_edge_slot() ) return __impl::fail<G>({ parse_code::index_overflow, position });
766   if ( position != size || result.value.vertices_count() != live_vertices || result.value.edges_count() != live_edges )
767     return __impl::fail<G>({ parse_code::corrupt, position });
768   return result;
769 }

774   requires(micron::is_same_v<typename G::vertex_property_type, math::empty_property>

799     return finish(math::graphs::freeze_bidirectional(decoded.value));

801     return finish(math::graphs::freeze(decoded.value));
