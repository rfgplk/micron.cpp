96 template<is_iterable_container T>
97   requires(micron::is_fundamental_v<typename T::value_type>)
98 max_t
99 read(i32 fd, T &buf, const usize cnt)
100 {
101   if ( fd < 0 ) return -error::bad_fd;
102   // check alloc'd cap
103   if ( buf.max_size() < cnt ) [[unlikely]]
104     return -error::invalid_arg;
105   max_t r = posix::read(fd, buf.data(), cnt);
106   if constexpr ( has_set_size<T> ) {
107     if ( r >= 0 ) buf.set_size(r);
108   }
109   return r;
110 }

129 template<is_iterable_container T>
130   requires(micron::is_fundamental_v<typename T::value_type>)
131 max_t
132 read(i32 fd, T &buf)
133 {
134   if ( fd < 0 ) return -error::bad_fd;
135   if constexpr ( has_set_size<T> && requires(T t, usize n) { t.reserve(n); } ) {
136     const max_t sz = posix::lseek(fd, 0, posix::seek_end);
137     if ( sz > 0 ) {
138       posix::lseek(fd, 0, posix::seek_set);
139       const usize need = static_cast<usize>(sz);
140       if ( buf.max_size() <= need ) buf.reserve(need + 1);
141       usize got = 0;
142       while ( got < need ) {
143         const max_t r = posix::read(fd, reinterpret_cast<byte *>(buf.data()) + got, need - got);
144         if ( r < 0 ) return r;
145         if ( r == 0 ) break;
146         got += static_cast<usize>(r);
147       }
148       buf.set_size(got / sizeof(typename T::value_type));
149       return static_cast<max_t>(got);
150     }
151   }
152   max_t r = posix::read(fd, buf.data(), buf.max_size());
153   if constexpr ( has_set_size<T> ) {
154     if ( r >= 0 ) buf.set_size(r);
155   }
156   return r;
157 }

276 template<typename T = byte>
277 max_t
278 fwrite(T &ref, usize num, const fd_t &handle)
279 {
280   if ( handle.has_error() or handle.closed() ) return -error::bad_fd;
281   // differentiate depending on handle, maintains cache loc.
282   if ( handle == stdout ) {
283     if ( __global_buffer_stdout->full(num * sizeof(byte)) ) {
284       (*__global_buffer_stdout) >> handle;
285       posix::write(handle.fd, ref, num);
286       return static_cast<max_t>(num);
287     force_flush_out:
288       (*__global_buffer_stdout) >> handle;
289       return static_cast<max_t>(num);
290     }
291     __global_buffer_stdout->append(ref, num);
292     if ( __stdout_line_buffered && simd::any_set_128(ref, num, __global_buffer_flush) ) goto force_flush_out;
293     return static_cast<max_t>(num);
294   } else if ( handle == stderr ) {
295     if ( __global_buffer_stderr->full(num * sizeof(byte)) ) {
296       (*__global_buffer_stderr) >> handle;
297       posix::write(handle.fd, ref, num);
298       return static_cast<max_t>(num);
299     force_flush_err:
300       (*__global_buffer_stderr) >> handle;
301       return static_cast<max_t>(num);
302     }
303     __global_buffer_stderr->append(ref, num);
304     if ( simd::any_set_128(ref, num, __global_buffer_flush) ) goto force_flush_err;
305     return static_cast<max_t>(num);
306   }
307   return posix::write(handle.fd, ref, num);
308 }

310 template<typename T = byte>
311 max_t
312 fwrite(T *ptr, usize num, const fd_t &handle)
313 {
314   if ( ptr == nullptr ) [[unlikely]]
315     return -error::invalid_arg;
316   if ( handle.has_error() or handle.closed() ) return -error::bad_fd;
317   const usize nbytes = num * sizeof(T);
318   if ( handle == stdout ) {
319     if ( __global_buffer_stdout->full(nbytes) ) {
320       (*__global_buffer_stdout) >> handle;
321       if ( max_t __w = posix::write_all(handle, reinterpret_cast<const byte *>(ptr), nbytes); __w < 0 ) return __w;
322       return static_cast<max_t>(num);
323     force_flush_out:
324       (*__global_buffer_stdout) >> handle;
325       return static_cast<max_t>(num);
326     }
327     __global_buffer_stdout->append(ptr, num);
328     if ( __stdout_line_buffered && simd::any_set_128(ptr, nbytes, __global_buffer_flush) ) goto force_flush_out;
329     return static_cast<max_t>(num);
330   } else if ( handle == stderr ) {
331     if ( __global_buffer_stderr->full(nbytes) ) {
332       (*__global_buffer_stderr) >> handle;
333       if ( max_t __w = posix::write_all(handle, reinterpret_cast<const byte *>(ptr), nbytes); __w < 0 ) return __w;
334       return static_cast<max_t>(num);
335     force_flush_err:
336       (*__global_buffer_stderr) >> handle;
337       return static_cast<max_t>(num);
338     }
339     __global_buffer_stderr->append(ptr, num);
340     if ( simd::any_set_128(ptr, nbytes, __global_buffer_flush) ) goto force_flush_err;
341     return static_cast<max_t>(num);
342   }
343   if ( max_t __w = posix::write_all(handle, reinterpret_cast<const byte *>(ptr), nbytes); __w < 0 ) return __w;
344   return static_cast<max_t>(num);
345 }

347 void
348 fflush(const fd_t &handle)
349 {
350   if ( handle.has_error() or handle.closed() ) return;
351   // differentiate depending on handle, maintains cache loc.
352   if ( handle == stdout ) {
353     (*__global_buffer_stdout) >> handle;
354     return;
355   } else if ( handle == stderr ) {
356     (*__global_buffer_stderr) >> handle;
357     return;
358   }
359 }
