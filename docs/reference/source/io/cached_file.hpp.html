156   // load the whole file into the resident buffer
157   max_t
158   load(void)
159   {
160     if ( i32 __e = __check() ) [[unlikely]]
161       return __e;
162     if ( i32 e = stat<STAT_OVERRIDE>(); e != 0 ) [[unlikely]]
163       return e;
164     const usize need = static_cast<usize>(sd.st_size);
165     if ( need == 0 ) {
166       data._buf_set_length(0);
167       return 0;
168     }
169     constexpr usize esz = sizeof(typename T::value_type);
170     data.reserve(need / esz + 1);
171     byte *p = reinterpret_cast<byte *>(data.data());
172     usize got = 0;
173     while ( got < need ) {
174       max_t r = posix::pread(__handle.fd, p + got, need - got, static_cast<posix::off64_t>(got));
175       if ( r < 0 ) [[unlikely]] {
176         if ( -r == error::interrupted ) continue;
177         data._buf_set_length(got / esz);
178         return r;
179       }
180       if ( r == 0 ) break;
181       got += static_cast<usize>(r);
182     }
183     data._buf_set_length(got / esz);
184     return static_cast<max_t>(got);
185   }

187   // growth loop for virtual (/proc, /sys) files that fstat to size 0
188   max_t
189   load_kernel(void)
190   {
191     if ( i32 __e = __check() ) [[unlikely]]
192       return __e;
193     data.clear();
194     byte chunk[4096];
195     usize total = 0;
196     posix::off64_t off = 0;
197     for ( ;; ) {
198       max_t r = posix::pread(__handle.fd, chunk, sizeof(chunk), off);
199       if ( r < 0 ) [[unlikely]] {
200         if ( -r == error::interrupted ) continue;
201         return r;
202       }
203       if ( r == 0 ) break;
204       for ( max_t i = 0; i < r; ++i ) data.push_back(static_cast<typename T::value_type>(chunk[i]));
205       total += static_cast<usize>(r);
206       off += r;
207     }
208     return static_cast<max_t>(total);
209   }

211   max_t
212   read_bytes(usize sz)
213   {
214     if ( i32 __e = __check() ) [[unlikely]]
215       return __e;
216     data.reserve(data.size() + sz + 1);
217     byte chunk[4096];
218     usize total = 0;
219     while ( total < sz ) {
220       const usize want = (sz - total) < sizeof(chunk) ? (sz - total) : sizeof(chunk);
221       max_t r = posix::pread(__handle.fd, chunk, want, static_cast<posix::off64_t>(__cursor));
222       if ( r < 0 ) [[unlikely]] {
223         if ( -r == error::interrupted ) continue;
224         return total ? static_cast<max_t>(total) : r;
225       }
226       if ( r == 0 ) break;
227       for ( max_t i = 0; i < r; ++i ) data.push_back(static_cast<typename T::value_type>(chunk[i]));
228       __cursor += static_cast<usize>(r);
229       total += static_cast<usize>(r);
230     }
231     return static_cast<max_t>(total);
232   }

234   // persist the resident buffer at the logical cursor; content stays resident
235   max_t
236   write(void)
237   {
238     if ( !data ) return 0;
239     if ( i32 __e = __check() ) [[unlikely]]
240       return __e;
241     const byte *p = reinterpret_cast<const byte *>(data.data());
242     const usize len = data.size() * sizeof(typename T::value_type);
243     usize done = 0;
244     while ( done < len ) {
245       max_t w = posix::pwrite(__handle.fd, p + done, len - done, static_cast<posix::off64_t>(__cursor + done));
246       if ( w < 0 ) [[unlikely]] {
247         if ( -w == error::interrupted ) continue;
248         __cursor += done;
249         micron::zero(&sd);
250         return w;
251       }
252       if ( w == 0 ) break;
253       done += static_cast<usize>(w);
254     }
255     __cursor += done;
256     micron::zero(&sd);
257     return static_cast<max_t>(done);
258   }

454   // append the whole contents of another cached_file
455   template<is_string U>
456   max_t
457   append_file(const cached_file<U> &other, usize chunk_sz = 65536u)
458   {
459     if ( i32 __e = __check() ) [[unlikely]]
460       return __e;
461     if ( other.get_fd() < 0 ) [[unlikely]]
462       return -error::bad_fd;
463 
464     posix::off64_t src_off = 0;
465     usize src_sz = static_cast<usize>(const_cast<cached_file<U> &>(other).size());
466     if ( i32 e = stat<STAT_OVERRIDE>(); e != 0 ) [[unlikely]]
467       return e;
468     posix::off64_t dst_off = sd.st_size;
469 
470     micron::buffer win(chunk_sz);
471     max_t total = 0;
472     while ( static_cast<usize>(src_off) < src_sz ) {
473       max_t n = posix::pread(other.get_fd(), win.begin(), chunk_sz, src_off);
474       if ( n < 0 ) [[unlikely]] {
475         if ( -n == error::interrupted ) continue;
476         micron::zero(&sd);
477         return n;      // propagate source read error
478       }
479       if ( n == 0 ) break;
480       // write the n read bytes in full at the advancing destination offset
481       usize wdone = 0;
482       while ( wdone < static_cast<usize>(n) ) {
483         max_t w
484             = posix::pwrite(__handle.fd, win.begin() + wdone, static_cast<usize>(n) - wdone, dst_off + static_cast<posix::off64_t>(wdone));
485         if ( w < 0 ) [[unlikely]] {
486           if ( -w == error::interrupted ) continue;
487           micron::zero(&sd);
488           return w;      // propagate -errno
489         }
490         if ( w == 0 ) break;
491         wdone += static_cast<usize>(w);
492       }
493       src_off += static_cast<posix::off64_t>(wdone);
494       dst_off += static_cast<posix::off64_t>(wdone);
495       total += static_cast<max_t>(wdone);
496       if ( wdone < static_cast<usize>(n) ) break;      // short write (pwrite == 0): stop
497     }
498     micron::zero(&sd);
499     return total;
500   }

502   //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
503   // whole-file helpers
504   max_t
505   copy_to(const char *dest_path, usize chunk_sz = 65536u) const
506   {
507     if ( i32 __e = __check() ) [[unlikely]]
508       return __e;
509     posix::fd_t out{ static_cast<i32>(posix::openat(
510         posix::at_fdcwd, dest_path, posix::o_wronly | posix::o_create | posix::o_trunc | posix::o_cloexec, posix::mode_file)) };
511     if ( !out ) [[unlikely]]
512       return out.fd;
513 
514     usize src_sz = static_cast<usize>(const_cast<cached_file *>(this)->size());
515     posix::off64_t src_off = 0;
516     micron::buffer win(chunk_sz);
517     max_t total = 0;
518 
519     while ( static_cast<usize>(src_off) < src_sz ) {
520       max_t n = posix::pread(__handle.fd, win.begin(), chunk_sz, src_off);
521       if ( n < 0 ) {
522         if ( -n == error::interrupted ) continue;
523         posix::close(out.fd);
524         return n;      // propagate a source read error; never report a truncated copy as success
525       }
526       if ( n == 0 ) break;      // EOF
527       // write all n read bytes to the destination
528       usize wdone = 0;
529       while ( wdone < static_cast<usize>(n) ) {
530         max_t w = posix::write(out.fd, win.begin() + wdone, static_cast<usize>(n) - wdone);
531         if ( w < 0 ) [[unlikely]] {
532           if ( -w == error::interrupted ) continue;
533           posix::close(out.fd);
534           return w;      // never report source size as success
535         }
536         if ( w == 0 ) break;
537         wdone += static_cast<usize>(w);
538       }
539       src_off += static_cast<posix::off64_t>(wdone);
540       total += static_cast<max_t>(wdone);
541       if ( wdone < static_cast<usize>(n) ) break;      // short write (write == 0): stop
542     }
543     posix::close(out.fd);
544     return total;
545   }

554   // three-way byte comparison: -1 / 0 / 1, or negative errno < -1 on failure
555   template<is_string U>
556   i32
557   compare_to(const cached_file<U> &other, usize chunk_sz = 65536u) const
558   {
559     if ( __check() != 0 || other.get_fd() < 0 ) [[unlikely]]
560       return -error::bad_fd;
561 
562     usize a_sz = static_cast<usize>(const_cast<cached_file *>(this)->size());
563     usize b_sz = static_cast<usize>(const_cast<cached_file<U> &>(other).size());
564     usize scan = 0;
565     micron::buffer wa(chunk_sz), wb(chunk_sz);
566 
567     while ( scan < a_sz && scan < b_sz ) {
568       usize take = chunk_sz < (a_sz - scan) ? chunk_sz : (a_sz - scan);
569       take = take < (b_sz - scan) ? take : (b_sz - scan);
570 
571       max_t na = posix::pread(__handle.fd, wa.begin(), take, static_cast<posix::off64_t>(scan));
572       if ( na < 0 ) {
573         if ( -na == error::interrupted ) continue;
574         i32 e = static_cast<i32>(na);
575         return e < -1 ? e : -error::io_error;      // real error -> errno (<-1), never a bogus verdict
576       }
577       max_t nb = posix::pread(other.get_fd(), wb.begin(), take, static_cast<posix::off64_t>(scan));
578       if ( nb < 0 ) {
579         if ( -nb == error::interrupted ) continue;
580         i32 e = static_cast<i32>(nb);
581         return e < -1 ? e : -error::io_error;
582       }
583       if ( na == 0 || nb == 0 ) break;      // EOF on one side (not an error)
584 
585       usize cmp_n = static_cast<usize>(na) < static_cast<usize>(nb) ? static_cast<usize>(na) : static_cast<usize>(nb);
586 
587       for ( usize i = 0; i < cmp_n; ++i ) {
588         if ( wa[i] < wb[i] ) return -1;
589         if ( wa[i] > wb[i] ) return 1;
590       }
591       scan += cmp_n;
592     }
593 
594     if ( a_sz < b_sz ) return -1;
595     if ( a_sz > b_sz ) return 1;
596     return 0;
597   }

599   // load with a tap: fn observes every chunk as it streams into the resident buffer
600   template<io::intercept_fn Fn>
601   max_t
602   load_intercepted(Fn &&fn, usize chunk_sz = 65536u)
603   {
604     if ( i32 __e = __check() ) [[unlikely]]
605       return __e;
606     if ( i32 e = stat<STAT_OVERRIDE>(); e != 0 ) [[unlikely]]
607       return e;
608     usize file_sz = static_cast<usize>(sd.st_size);
609     constexpr usize esz = sizeof(typename T::value_type);
610     data.reserve(file_sz / esz + 1);
611 
612     micron::buffer win(chunk_sz);
613     posix::off64_t off = 0;
614     max_t total = 0;
615 
616     while ( static_cast<usize>(off) < file_sz ) {
617       max_t n = posix::pread(__handle.fd, win.begin(), chunk_sz, off);
618       if ( n < 0 && -n == error::interrupted ) continue;
619       if ( n <= 0 ) break;
620       fn(win.begin(), static_cast<usize>(n));
621       data.append(reinterpret_cast<const typename T::value_type *>(win.begin()), static_cast<usize>(n) / esz);
622       off += n;
623       total += n;
624     }
625     return total;
626   }
