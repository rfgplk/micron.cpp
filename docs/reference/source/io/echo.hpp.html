118 // (b) arbitrary runtime fd
119 template<usize CAP = 1024> struct fd_sink {
120   fd_t __t;
121   usize __len = 0;
122   i32 __err = 0;
123   char __buf[CAP];
124 
125   explicit fd_sink(fd_t t) : __t(t) { }
126 
127   // looped write mirroring io::file::__write_loop
128   max_t
129   __write_all(const char *p, usize n)
130   {
131     usize done = 0;
132     while ( done < n ) {
133       max_t w = posix::write(__t.fd, p + done, n - done);
134       if ( w < 0 ) [[unlikely]] {
135         if ( -w == error::interrupted ) continue;
136         return w;
137       }
138       if ( w == 0 ) break;
139       done += static_cast<usize>(w);
140     }
141     return static_cast<max_t>(done);
142   }
143 
144   max_t
145   put(const char *p, usize n)
146   {
147     if ( __len + n > CAP ) {
148       flush();
149       if ( n >= CAP ) {
150         max_t w = __write_all(p, n);
151         if ( w < 0 && !__err ) [[unlikely]]
152           __err = static_cast<i32>(w);
153         return w;
154       }
155     }
156     micron::memcpy(__buf + __len, p, n);
157     __len += n;
158     return static_cast<max_t>(n);
159   }
160 
161   max_t
162   put(char c)
163   {
164     if ( __len == CAP ) flush();
165     __buf[__len++] = c;
166     return 1;
167   }
168 
169   max_t
170   flush(void)
171   {
172     if ( !__len ) return 0;
173     max_t w = __write_all(__buf, __len);
174     __len = 0;
175     if ( w < 0 && !__err ) [[unlikely]]
176       __err = static_cast<i32>(w);
177     return w;
178   }
179 
180   fd_t
181   target(void) const
182   {
183     return __t;
184   }
185 };

403 // arithmetic
404 template<output_sink S, typename T>
405   requires __printk_arith<T>
406 max_t
407 printk(S &s, const T &x)
408 {
409   char buf[128];
410   usize n = 0;
411 
412   if constexpr ( micron::is_same_v<T, bool> ) {
413     n = __impl::arith_to_buf(buf, 128, x);
414   } else if constexpr ( micron::is_floating_point_v<T> ) {
415     if constexpr ( sizeof(T) <= sizeof(f32) ) {
416       n = __impl::arith_to_buf(buf, 128, static_cast<f32>(x));      // f16 widens exactly
417     } else if constexpr ( sizeof(T) <= sizeof(f64) ) {
418       n = __impl::arith_to_buf(buf, 128, static_cast<f64>(x));
419     } else if constexpr ( micron::is_same_v<T, long double> || micron::is_same_v<T, flong> ) {
420       n = __impl::arith_to_buf(buf, 128, static_cast<long double>(x));
421 #if defined(__micron_f128_distinct)
422     } else if constexpr ( micron::is_same_v<T, f128> ) {
423       n = __impl::arith_to_buf(buf, 128, static_cast<f128>(x));
424 #endif
425     } else {
426       n = __impl::arith_to_buf(buf, 128, static_cast<f64>(x));
427     }
428   } else if constexpr ( micron::is_same_v<T, u128> ) {
429     n = __impl::arith_to_buf(buf, 128, x);
430   } else if constexpr ( micron::is_same_v<T, i128> ) {
431     n = __impl::arith_to_buf(buf, 128, x);
432   } else if constexpr ( micron::is_signed_v<T> ) {
433     n = __impl::arith_to_buf(buf, 128, static_cast<i64>(x));
434   } else {
435     n = __impl::arith_to_buf(buf, 128, static_cast<u64>(x));
436   }
437 
438   return s.put(buf, n);
439 }

514 
515 template<output_sink S, typename T>
516 max_t
517 printkn(S &s, const T &x)
518 {
519   using U = micron::remove_cvref_t<T>;
520   max_t t = 0;
521   if constexpr ( has_cstr<U> && micron::is_class_v<U> && requires { typename U::value_type; } ) {
522     if constexpr ( micron::is_same_v<typename U::value_type, wchar_t> && fd_backed_sink<S> ) {
523       wfput(x.w_str(), s.target());
524     } else if constexpr ( micron::is_same_v<typename U::value_type, char32_t> && fd_backed_sink<S> ) {
525       io::unifput(x.uni_str(), s.target());
526     } else {
527       t += printk(s, x);
528     }
529   } else {
530     t += printk(s, x);
531   }
532   t += s.put('\n');
533   return t;
534 }

560 template<output_sink S, typename... Ts>
561 inline max_t
562 run(S &s, bool newline, const Ts &...args)
563 {
564   max_t total = 0;
565   max_t err = 0;
566   (
567       [&] {
568         max_t r = printk(s, args);
569         if ( r < 0 && err == 0 ) [[unlikely]]
570           err = r;
571         else
572           total += r;
573       }(),
574       ...);
575   if ( newline ) total += s.put('\n');
576   return err != 0 ? err : total;
577 }

641 // redirecting forms
642 template<typename Target, typename... Ts>
643   requires(echo_target<Target> && !micron::any_settling<Ts...>)
644 inline max_t
645 echo(Target &&tgt, const Ts &...args)
646 {
647   using U = micron::remove_cvref_t<Target>;
648   if constexpr ( micron::is_same_v<U, fd_t> ) {
649     if ( tgt.fd < 0 ) [[unlikely]]
650       return -error::bad_fd;
651     fd_sink<> s{ tgt };
652     max_t r = __echo_impl::run(s, true, args...);
653     max_t f = s.flush();
654     if ( s.__err ) [[unlikely]]
655       return s.__err;
656     return f < 0 ? f : r;
657   } else if constexpr ( __echo_impl::is_stream_inst<U>::value ) {
658     stream_sink s{ tgt };
659     return __echo_impl::run(s, true, args...);
660   } else {      // fd-bearing handle (io::file & friends): write through its fd
661     fd_t t = tgt.fd();
662     if ( t.fd < 0 ) [[unlikely]]
663       return -error::bad_fd;
664     fd_sink<> s{ t };
665     max_t r = __echo_impl::run(s, true, args...);
666     max_t f = s.flush();
667     if ( s.__err ) [[unlikely]]
668       return s.__err;
669     return f < 0 ? f : r;
670   }
671 }

682 template<typename Target, typename... Ts>
683   requires(echo_target<Target> && !micron::any_settling<Ts...>)
684 inline max_t
685 echon(Target &&tgt, const Ts &...args)
686 {
687   using U = micron::remove_cvref_t<Target>;
688   if constexpr ( micron::is_same_v<U, fd_t> ) {
689     if ( tgt.fd < 0 ) [[unlikely]]
690       return -error::bad_fd;
691     fd_sink<> s{ tgt };
692     max_t r = __echo_impl::run(s, false, args...);
693     max_t f = s.flush();
694     if ( s.__err ) [[unlikely]]
695       return s.__err;
696     return f < 0 ? f : r;
697   } else if constexpr ( __echo_impl::is_stream_inst<U>::value ) {
698     stream_sink s{ tgt };
699     return __echo_impl::run(s, false, args...);
700   } else {
701     fd_t t = tgt.fd();
702     if ( t.fd < 0 ) [[unlikely]]
703       return -error::bad_fd;

727 template<output_sink S>
728 inline max_t
729 __put_fill_run(S &s, char fill, usize n)
730 {
731   if ( !n ) return 0;
732   char buf[64];
733   const usize k0 = n < sizeof(buf) ? n : sizeof(buf);
734   for ( usize i = 0; i < k0; ++i ) buf[i] = fill;
735   max_t t = 0;
736   while ( n ) {
737     const usize k = n < sizeof(buf) ? n : sizeof(buf);
738     t += s.put(buf, k);
739     n -= k;
740   }
741   return t;
742 }

744 template<output_sink S>
745 inline void
746 apply_padding_sink(S &s, max_t &total, const char *content, usize content_len, const micron::format::__impl::fmt_spec &spec)
747 {
748   if ( spec.width == 0 || content_len >= spec.width ) {
749     total += s.put(content, content_len);
750     return;
751   }
752   usize pad_total = spec.width - content_len;
753   char fill = spec.fill ? spec.fill : ' ';
754   char align = spec.align;
755   if ( align == '\0' ) align = (spec.type == 's' || spec.type == '\0') ? '<' : '>';
756 
757   if ( align == '>' ) {
758     total += __put_fill_run(s, fill, pad_total);
759     total += s.put(content, content_len);
760   } else if ( align == '<' ) {
761     total += s.put(content, content_len);
762     total += __put_fill_run(s, fill, pad_total);
763   } else {
764     usize left = pad_total / 2;
765     usize right = pad_total - left;
766     total += __put_fill_run(s, fill, left);
767     total += s.put(content, content_len);
768     total += __put_fill_run(s, fill, right);
769   }
770 }

779 template<output_sink S, typename T, typename... Rest>
780 inline void
781 format_one_sink(S &s, max_t &total, const char *spec_start, const char *spec_end, usize arg_index, const T &val, const Rest &...rest)
782 {
783   if ( arg_index > 0 ) {
784     format_one_sink(s, total, spec_start, spec_end, arg_index - 1, rest...);
785     return;
786   }
787   micron::format::__impl::fmt_spec spec = micron::format::__impl::parse_spec(spec_start, spec_end);
788   using U = micron::remove_cvref_t<T>;
789   if constexpr ( requires(char *b, const U &v) { micron::format::formatter<U>::write(b, usize{}, v, spec); } ) {
790     // per formatter
791     constexpr usize __bs = micron::format::__impl::__fmt_buf_for<micron::format::formatter<U>>();
792     char buf[__bs];
793     usize n = micron::format::formatter<U>::write(buf, __bs, val, spec);
794     apply_padding_sink(s, total, buf, n, spec);
795   } else if constexpr ( requires(hstring<schar> &o, const U &v) { micron::format::formatter<U>::write_str(o, v, spec); } ) {
796     hstring<schar> out;
797     micron::format::formatter<U>::write_str(out, val, spec);
798     apply_padding_sink(s, total, out.c_str(), out.size(), spec);
799   } else {
800     static_assert(sizeof(U) == 0, "io::echof: no formatter<T> for this argument and it is not an echo-printable container");
801   }
802 }

804 template<output_sink S, typename... Args>
805 inline max_t
806 format_to_sink(S &s, const char *fmt, const Args &...args)
807 {
808   max_t total = 0;
809   if ( fmt == nullptr ) return 0;
810   usize auto_index = 0;
811   const char *p = fmt;
812   while ( *p ) {
813     if ( *p == '{' ) {
814       if ( *(p + 1) == '{' ) {
815         total += s.put('{');
816         p += 2;
817         continue;
818       }
819       ++p;
820       const char *close = p;
821       while ( *close && *close != '}' ) ++close;
822       if ( *close != '}' ) break;
823       const char *colon = p;
824       while ( colon < close && *colon != ':' ) ++colon;
825       usize index = auto_index;
826       bool has_explicit_index = false;
827       if ( colon > p ) {
828         const char *num_p = p;
829         bool all_digits = true;
830         while ( num_p < colon ) {
831           if ( *num_p < '0' || *num_p > '9' ) {
832             all_digits = false;
833             break;
834           }
835           ++num_p;
836         }
837         if ( all_digits ) {
838           index = 0;
839           num_p = p;
840           while ( num_p < colon ) {
841             index = index * 10 + static_cast<usize>(*num_p - '0');
842             ++num_p;
843           }
844           has_explicit_index = true;
845         }
846       }
847       const char *spec_start = (colon < close) ? colon + 1 : close;
848       const char *spec_end = close;
849       format_one_sink(s, total, spec_start, spec_end, index, args...);
850       if ( !has_explicit_index )
851         ++auto_index;
852       else
853         auto_index = index + 1;
854       p = close + 1;
855     } else if ( *p == '}' && *(p + 1) == '}' ) {
856       total += s.put('}');
857       p += 2;
858     } else {
859       const char *run = p;
860       while ( *p && *p != '{' && *p != '}' ) ++p;
861       if ( p != run )
862         total += s.put(run, static_cast<usize>(p - run));
863       else {
864         total += s.put(*p);
865         ++p;
866       }
867     }
868   }
869   return total;
870 }

1171 template<is_printable_container T>
1172 void
1173 bin(const T &data)
1174 {
1175   stdout_sink s;
1176   char buf[4096];
1177   usize cnt = 0;
1178   usize remaining = data.size();
1179   while ( remaining > 0 ) {
1180     usize chunk = remaining > 2047 ? 2047 : remaining;      // 2*chunk+1 must stay within buf[4096]
1181     const auto *src = reinterpret_cast<const u8 *>(&data[cnt]);
1182     for ( usize i = 0; i < chunk; ++i ) {
1183       buf[i * 2] = micron::__impl::__hex_lower[src[i] >> 4];
1184       buf[i * 2 + 1] = micron::__impl::__hex_lower[src[i] & 0xF];
1185     }
1186     s.put(buf, chunk * 2);
1187     cnt += chunk;
1188     remaining -= chunk;
1189   }
1190 }
