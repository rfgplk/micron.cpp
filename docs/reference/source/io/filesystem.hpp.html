19 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
20 // io::basic_filesystem
21 // path-keyed cache of open io::file handles plus the full set of
22 // path-based filesystem operations
23 //   * queries never open descriptors as a side effect
24 //   * only operator[]/open/create/at insert handles into the cache
25 //   * data-path operations return signed max_t (>= 0 count, < 0 negative errno) and never
26 //     throw; implicit opens (operator[]/apply on a missing entry) construct io::file, which
27 //     throws except::io_error on failure
28 //   * lock is a policy (micron::null_lock = single-threaded, zero cost; micron::mutex =
29 //     concurrent)
30 //   * Cap > 0 bounds the cache with LRU eviction of the coldest handle
31

37 // address-stable cache entry; io::file lives in its own heap allocation via a unique_pointer, so heap_swiss_map rehashes (which move its
38 // entries) only relocates the unique_pointer
39 struct __fs_entry {
40   micron::unique_pointer<io::file> fp;
41   io::modes mode = io::modes::read;
42 
43   __fs_entry() = default;
44 
45   __fs_entry(micron::unique_pointer<io::file> &&p, io::modes m) : fp(micron::move(p)), mode(m) { }
46 
47   __fs_entry(__fs_entry &&) noexcept = default;
48   __fs_entry &operator=(__fs_entry &&) noexcept = default;
49 
50   // copy duplicates the descriptor (io::file copy dup()s the fd)
51   // WARNING: present ONLY to satisfy heap_swiss_maps is_copy_constructible_v
52   __fs_entry(const __fs_entry &o)
53       : fp(o.fp ? micron::unique_pointer<io::file>(*o.fp) : micron::unique_pointer<io::file>()), mode(o.mode) { }
54 
55   __fs_entry &
56   operator=(const __fs_entry &o)
57   {
58     fp = o.fp ? micron::unique_pointer<io::file>(*o.fp) : micron::unique_pointer<io::file>();
59     mode = o.mode;
60     return *this;
61   }
62 };

108 inline constexpr bool
109 __mode_serviceable(const io::modes cached, const io::modes want) noexcept
110 {
111   // O_TRUNC must empty the file and O_EXCL must fail when it exists
112   if ( __mode_has_trunc(want) || __mode_has_excl(want) ) return false;
113   if ( __mode_is_direct(want) != __mode_is_direct(cached) ) return false;
114   if ( __mode_is_noatime(want) && !__mode_is_noatime(cached) ) return false;
115   if ( __mode_is_append(want) ) return __mode_can_write(cached);                                   // append
116   if ( __mode_can_write(want) ) return __mode_can_write(cached) && !__mode_is_append(cached);      // positioned write
117   return __mode_can_read(cached);                                                                  // read
118 }

143   void
144   __touch(const micron::string &k)
145   {
146     if constexpr ( Cap > 0 ) {
147       for ( usize i = 0; i < __recency.size(); i++ ) {
148         if ( __recency[i] == k ) {
149           for ( usize j = i; j + 1 < __recency.size(); j++ ) __recency[j] = micron::move(__recency[j + 1]);
150           __recency.pop_back();
151           break;
152         }
153       }
154       __recency.push_back(k);
155       if ( __recency.size() > Cap ) {
156         // flush the coldest handle before auto-evicting it, matching close()/evict() and the dtor
157         if ( __fs_entry *e = __open.find(__recency[0]); e != nullptr && e->fp ) e->fp->flush_data();
158         __open.erase(__recency[0]);
159         for ( usize j = 0; j + 1 < __recency.size(); j++ ) __recency[j] = micron::move(__recency[j + 1]);
160         __recency.pop_back();
161       }
162     }
163   }
164 
165   void
166   __drop_recency(const micron::string &k)
167   {
168     if constexpr ( Cap > 0 ) {
169       for ( usize i = 0; i < __recency.size(); i++ ) {
170         if ( __recency[i] == k ) {
171           for ( usize j = i; j + 1 < __recency.size(); j++ ) __recency[j] = micron::move(__recency[j + 1]);
172           __recency.pop_back();
173           return;
174         }
175       }
176     }
177   }

179   io::file &
180   __get_or_open(const micron::string &k, io::modes m)
181   {
182     if ( __fs_entry *e = __open.find(k); e != nullptr ) {
183       // cache hit
184       if ( !__mode_serviceable(e->mode, m) ) {
185         *e->fp = io::file(k.c_str(), m);
186         e->mode = m;
187       }
188       __touch(k);
189       return *e->fp;
190     }
191     auto ins = __open.emplace(k, __fs_entry{ micron::unique_pointer<io::file>(k.c_str(), m), m });
192     __touch(k);
193     return *ins.b->fp;
194   }

347   template<typename C>
348   max_t
349   read(const io::path_t &p, C &out)
350   {
351     __scope g(__mtx);
352     io::file &f = __get_or_open(__key(p), DefaultMode);
353     if ( max_t r = f.seek_to(0); r < 0 ) return r;
354     if constexpr ( (is_string<C> || is_contiguous_container<C>) && micron::is_trivially_copyable_v<typename C::value_type> ) {
355       out = f.template read<C>();
356       return static_cast<max_t>(out.size() * sizeof(typename C::value_type));
357     } else {
358       return f.read(out);      // framed tiers already consume the whole remainder
359     }
360   }

364   write(const io::path_t &p, const C &c)

628   max_t
629   move(const io::path_t &from, const io::path_t &to)
630   {
631     max_t r = rename(from, to);
632     if ( r == 0 || -r != error::cross_device_link ) return r;
633     // EXDEV: byte copy then unlink the source (non-atomic across filesystems)
634     if ( max_t c = copy(from, to); c < 0 ) return c;
635     return unlink(from);

638   // copy_file_range -> sendfile -> pread/pwrite loop
639   max_t
640   copy(const io::path_t &from, const io::path_t &to)
641   {
642     __scope g(__mtx);
643     {
644       micron::string kt = __key(to);
645       __drop_recency(kt);
646       __open.erase(kt);
647     }
648     posix::fd_t src{ static_cast<i32>(posix::open(from.c_str(), posix::o_rdonly | posix::o_cloexec)) };
649     if ( !src ) return src.fd;
650     stat_t st{};
651     if ( posix::fstat(src, st) < 0 ) {      // was ignored -> dst could be created with mode 0
652       posix::close(src.fd);
653       return -error::io_error;
654     }
655     {
656       stat_t dst_st{};      // refuse same-file copy: O_TRUNC would empty the shared inode before the read
657       if ( posix::stat(to.c_str(), dst_st) == 0 && dst_st.st_dev == st.st_dev && dst_st.st_ino == st.st_ino ) {
658         posix::close(src.fd);
659         return -error::invalid_arg;
660       }
661     }
662     posix::fd_t dst{ static_cast<i32>(
663         posix::open(to.c_str(), posix::o_wronly | posix::o_create | posix::o_trunc | posix::o_cloexec, st.st_mode & 0777u)) };
664     if ( !dst ) {
665       posix::close(src.fd);
666       return dst.fd;
667     }
668 
669     max_t total = 0;
670     bool cfr_ok = true, sf_ok = true;
671     for ( ;; ) {
672       max_t n = -1;
673       if ( cfr_ok ) {
674         n = posix::copy_file_range(src.fd, nullptr, dst.fd, nullptr, 1u << 20, 0u);
675         if ( n < 0 && (-n == error::cross_device_link || -n == error::invalid_arg || -n == error::bad_syscall) ) {
676           cfr_ok = false;
677           continue;
678         }
679       } else if ( sf_ok ) {
680         n = posix::sendfile(dst, src, nullptr, 1u << 20);
681         if ( n < 0 && (-n == error::invalid_arg || -n == error::bad_syscall) ) {
682           sf_ok = false;
683           continue;
684         }
685       } else {
686         byte buf[65536];
687         n = posix::read(src.fd, buf, sizeof(buf));
688         if ( n > 0 ) {
689           const usize want = static_cast<usize>(n);
690           usize off = 0;
691           while ( off < want ) {
692             max_t w = posix::write(dst.fd, buf + off, want - off);
693             if ( w < 0 ) {
694               if ( -w == error::interrupted ) continue;
695               n = w;
696               break;
697             }
698             if ( w == 0 ) {      // no progress on a > 0 request: treat as I/O failure, never silent truncation
699               n = -error::io_error;
700               break;
701             }
702             off += static_cast<usize>(w);
703           }
704         }
705       }
706       if ( n < 0 && -n == error::interrupted ) continue;
707       if ( n < 0 ) {
708         posix::close(src.fd);
709         posix::close(dst.fd);
710         return n;
711       }
712       if ( n == 0 ) break;
713       total += n;
714     }
715     posix::fsync(dst.fd);
716     posix::close(src.fd);
717     posix::close(dst.fd);
718     return total;
719   }

724   {
725     max_t total = 0;

727     for ( max_t r : rs ) {

861 // io::rooted_filesystem
862 // every operation resolves relative to one pinned directory fd via the *at syscall family
863 
864 class rooted_filesystem
865 {
866   micron::heap_swiss_map<io::path_t, __fs_entry> __open;
867   fd_t __root;
868   io::path_t __root_path;
869 
870   io::file &
871   __get_or_open(const io::path_t &k, io::modes m, const open_opts &opts = {})
872   {
873     if ( __fs_entry *e = __open.find(k); e != nullptr ) {
874       if ( !__mode_serviceable(e->mode, m) ) {
875         i32 fd = static_cast<i32>(posix::openat(__root, k.c_str(), compose_open_flags(m, opts), opts.perms));
876         if ( fd < 0 )      // reopen failed: surface it rather than silently returning a stale wrong-mode handle
877           exc<except::filesystem_error>("micron::io::rooted_filesystem: reopen for requested mode failed");
878         *e->fp = io::file(fd_t{ fd }, k.c_str());
879         e->mode = m;
880       }
881       return *e->fp;
882     }
883     i32 fd = static_cast<i32>(posix::openat(__root, k.c_str(), compose_open_flags(m, opts), opts.perms));
884     if ( fd < 0 )      // do NOT cache a failed open: a poisoned entry denies the path permanently even after it appears
885       exc<except::filesystem_error>("micron::io::rooted_filesystem: openat failed");
886     auto ins = __open.emplace(k, __fs_entry{ micron::unique_pointer<io::file>(fd_t{ fd }, k.c_str()), m });
887     return *ins.b->fp;
888   }
889 
890 public:
891   ~rooted_filesystem()
892   {
893     __open.for_each([](const io::path_t &, __fs_entry &e) {
894       if ( e.fp ) e.fp->flush_data();
895     });
896     if ( __root.fd >= 0 ) posix::close(__root.fd);
897   }
898 
899   rooted_filesystem() = delete;
900 
901   explicit rooted_filesystem(const io::path_t &root) : __open(), __root(posix::invalid_fd), __root_path(io::prune(root))
902   {
903     i32 fd = static_cast<i32>(posix::open(__root_path.c_str(), posix::o_path | posix::o_directory | posix::o_cloexec));
904     if ( fd < 0 ) exc<except::filesystem_error>("micron::io::rooted_filesystem: cannot open root directory");
905     __root.fd = fd;
906   }

908   rooted_filesystem(const rooted_filesystem &) = delete;
909   rooted_filesystem &operator=(const rooted_filesystem &) = delete;

911   rooted_filesystem(rooted_filesystem &&o) noexcept
912       : __open(micron::move(o.__open)), __root(o.__root), __root_path(micron::move(o.__root_path))
913   {
914     o.__root = posix::invalid_fd;
915   }

917   rooted_filesystem &
918   operator=(rooted_filesystem &&o) noexcept
919   {
920     if ( this == &o ) return *this;
921     __open.clear();
922     if ( __root.fd >= 0 ) posix::close(__root.fd);
923     __open = micron::move(o.__open);
924     __root = o.__root;
925     __root_path = micron::move(o.__root_path);
926     o.__root = posix::invalid_fd;
