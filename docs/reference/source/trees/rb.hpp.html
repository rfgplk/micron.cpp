258   void
259   rotate_left(node *x) noexcept
260   {
261     node *y = x->right;
262     x->right = y->left;
263     if ( y->left ) y->left->parent = x;
264     y->parent = x->parent;
265     if ( !x->parent )
266       root_ = y;
267     else if ( x == x->parent->left )
268       x->parent->left = y;
269     else
270       x->parent->right = y;
271     y->left = x;
272     x->parent = y;
273   }

277   {
278     node *y = x->left;
279     x->left = y->right;
280     if ( y->right ) y->right->parent = x;
281     y->parent = x->parent;
282     if ( !x->parent )
283       root_ = y;
284     else if ( x == x->parent->right )
285       x->parent->right = y;
286     else
287       x->parent->left = y;
288     y->right = x;
289     x->parent = y;

292   void
293   insert_fixup(node *z) noexcept
294   {
295     while ( z->parent && z->parent->color == RBColor::RED ) {
296       if ( z->parent == z->parent->parent->left ) {
297         node *y = z->parent->parent->right;
298         if ( y && y->color == RBColor::RED ) {
299           z->parent->color = RBColor::BLACK;
300           y->color = RBColor::BLACK;
301           z->parent->parent->color = RBColor::RED;
302           z = z->parent->parent;
303         } else {
304           if ( z == z->parent->right ) {
305             z = z->parent;
306             rotate_left(z);
307           }
308           z->parent->color = RBColor::BLACK;
309           z->parent->parent->color = RBColor::RED;
310           rotate_right(z->parent->parent);
311         }
312       } else {
313         node *y = z->parent->parent->left;
314         if ( y && y->color == RBColor::RED ) {
315           z->parent->color = RBColor::BLACK;
316           y->color = RBColor::BLACK;
317           z->parent->parent->color = RBColor::RED;
318           z = z->parent->parent;
319         } else {
320           if ( z == z->parent->left ) {
321             z = z->parent;
322             rotate_right(z);
323           }
324           z->parent->color = RBColor::BLACK;
325           z->parent->parent->color = RBColor::RED;
326           rotate_left(z->parent->parent);
327         }
328       }
329     }
330     root_->color = RBColor::BLACK;
331   }

333   void

335   {

345   void
346   erase_fixup(node *x, node *x_parent) noexcept
347   {
348     while ( (x != root_) && (!x || x->color == RBColor::BLACK) ) {
349       if ( x == (x_parent ? x_parent->left : nullptr) ) {
350         node *w = x_parent ? x_parent->right : nullptr;
351         if ( w && w->color == RBColor::RED ) {
352           w->color = RBColor::BLACK;
353           x_parent->color = RBColor::RED;
354           rotate_left(x_parent);
355           w = x_parent->right;
356         }
357         if ( (!w || (!w->left || w->left->color == RBColor::BLACK)) && (!w || (!w->right || w->right->color == RBColor::BLACK)) ) {
358           if ( w ) w->color = RBColor::RED;
359           x = x_parent;
360           x_parent = x ? x->parent : nullptr;
361         } else {
362           if ( !w || !w->right || w->right->color == RBColor::BLACK ) {
363             if ( w && w->left ) w->left->color = RBColor::BLACK;
364             if ( w ) w->color = RBColor::RED;
365             if ( x_parent ) rotate_right(w);
366             w = x_parent ? x_parent->right : nullptr;
367           }
368           if ( w ) w->color = x_parent ? x_parent->color : RBColor::BLACK;
369           if ( x_parent ) x_parent->color = RBColor::BLACK;
370           if ( w && w->right ) w->right->color = RBColor::BLACK;
371           if ( x_parent ) rotate_left(x_parent);
372           x = root_;
373           x_parent = nullptr;
374         }
375       } else {
376         node *w = x_parent ? x_parent->left : nullptr;
377         if ( w && w->color == RBColor::RED ) {
378           w->color = RBColor::BLACK;
379           x_parent->color = RBColor::RED;
380           rotate_right(x_parent);
381           w = x_parent->left;
382         }
383         if ( (!w || (!w->right || w->right->color == RBColor::BLACK)) && (!w || (!w->left || w->left->color == RBColor::BLACK)) ) {
384           if ( w ) w->color = RBColor::RED;
385           x = x_parent;
386           x_parent = x ? x->parent : nullptr;
387         } else {
388           if ( !w || !w->left || w->left->color == RBColor::BLACK ) {
389             if ( w && w->right ) w->right->color = RBColor::BLACK;
390             if ( w ) w->color = RBColor::RED;
391             if ( x_parent ) rotate_left(w);
392             w = x_parent ? x_parent->left : nullptr;
393           }
394           if ( w ) w->color = x_parent ? x_parent->color : RBColor::BLACK;
395           if ( x_parent ) x_parent->color = RBColor::BLACK;
396           if ( w && w->left ) w->left->color = RBColor::BLACK;
397           if ( x_parent ) rotate_right(x_parent);
398           x = root_;
399           x_parent = nullptr;
400         }
401       }
402     }
403     if ( x ) x->color = RBColor::BLACK;
404   }

466   clone_subtree(node *src, node *parent)
467   {
468     if ( !src ) return nullptr;

470     n->color = src->color;

489   void
490   destroy_subtree(node *x) noexcept
491   {
492     while ( x ) {
493       if ( x->left ) {
494         node *l = x->left;
495         x->left = l->right;
496         l->right = x;
497         x = l;
498       } else {
499         node *r = x->right;
500         destroy_node(x);
501         x = r;
502       }
503     }
504   }

506   void
507   __erase_node(node *z) noexcept
508   {
509     node *y = z;
510     RBColor y_original_color = y->color;
511     node *x = nullptr;
512     node *x_parent = nullptr;
513 
514     if ( !z->left ) {
515       x = z->right;
516       x_parent = z->parent;
517       transplant(z, z->right);
518     } else if ( !z->right ) {
519       x = z->left;
520       x_parent = z->parent;
521       transplant(z, z->left);
522     } else {
523       y = minimum(z->right);
524       y_original_color = y->color;
525       x = y->right;
526       if ( y->parent == z ) {
527         if ( x ) x->parent = y;
528         x_parent = y;
529       } else {
530         transplant(y, y->right);
531         y->right = z->right;
532         if ( y->right ) y->right->parent = y;
533         x_parent = y->parent;
534       }
535       transplant(z, y);
536       y->left = z->left;
537       if ( y->left ) y->left->parent = y;
538       y->color = z->color;
539     }
540 
541     destroy_node(z);
542     --size_;
543 
544     if ( y_original_color == RBColor::BLACK ) erase_fixup(x, x_parent);
545   }

567   template<traversal_order O, class Fn>
568   static walk_ctl
569   __traverse_rec(const node *x, Fn &fn)
570   {
571     if ( !x ) return walk_ctl::continue_;
572     if constexpr ( O == traversal_order::preorder ) {
573       walk_ctl c = micron::__impl::invoke_walk(fn, x->data);
574       if ( c == walk_ctl::stop ) return walk_ctl::stop;
575       if ( c == walk_ctl::skip_children ) return walk_ctl::continue_;
576       if ( __traverse_rec<O>(x->left, fn) == walk_ctl::stop ) return walk_ctl::stop;
577       return __traverse_rec<O>(x->right, fn);
578     } else if constexpr ( O == traversal_order::postorder ) {
579       if ( __traverse_rec<O>(x->left, fn) == walk_ctl::stop ) return walk_ctl::stop;
580       if ( __traverse_rec<O>(x->right, fn) == walk_ctl::stop ) return walk_ctl::stop;
581       return micron::__impl::invoke_walk(fn, x->data) == walk_ctl::stop ? walk_ctl::stop : walk_ctl::continue_;
582     } else {      // inorder
583       if ( __traverse_rec<O>(x->left, fn) == walk_ctl::stop ) return walk_ctl::stop;
584       if ( micron::__impl::invoke_walk(fn, x->data) == walk_ctl::stop ) return walk_ctl::stop;
585       return __traverse_rec<O>(x->right, fn);
586     }
587   }

589   template<class Fn>
590   static walk_ctl
591   __traverse_level(const node *root, Fn &fn)
592   {
593     if ( !root ) return walk_ctl::continue_;
594     micron::fvector<const node *> q;
595     q.push_back(root);
596     for ( usize head = 0; head < q.size(); ++head ) {
597       const node *x = q[head];
598       walk_ctl c = micron::__impl::invoke_walk(fn, x->data);
599       if ( c == walk_ctl::stop ) return walk_ctl::stop;
600       if ( c != walk_ctl::skip_children ) {
601         if ( x->left ) q.push_back(x->left);
602         if ( x->right ) q.push_back(x->right);
603       }
604     }
605     return walk_ctl::continue_;
606   }

782   template<class U>
783   T &
784   insert_or_assign(U &&value)
785   {
786     node *y = nullptr;
787     node *x = root_;
788     while ( x ) {
789       y = x;
790       if ( Less::lt(value, x->data) )
791         x = x->left;
792       else if ( Less::lt(x->data, value) )
793         x = x->right;
794       else {
795         x->data = micron::forward<U>(value);
796         return x->data;
797       }
798     }
799     node *z = make_node(micron::forward<U>(value));
800     z->parent = y;
801     z->left = nullptr;
802     z->right = nullptr;
803     z->color = RBColor::RED;
804     z->kind = node::kind_t::leaf;
805     if ( !y )
806       root_ = z;
807     else if ( Less::lt(z->data, y->data) )
808       y->left = z;
809     else
810       y->right = z;
811     insert_fixup(z);
812     ++size_;
813     return z->data;
814   }

902   template<bool Const> class __cursor
903   {
904     using __node = micron::conditional_t<Const, const node, node>;
905     __node *__n;
906 
907     static __node *
908     __succ(__node *__x) noexcept
909     {
910       if ( __x->right ) {
911         __x = __x->right;
912         while ( __x->left ) __x = __x->left;
913         return __x;
914       }
915       __node *__p = __x->parent;
916       while ( __p && __x == __p->right ) {
917         __x = __p;
918         __p = __p->parent;
919       }
920       return __p;
921     }
922 
923   public:
924     using value_type = T;
925     using reference = const T &;
926     using pointer = const T *;
927     using difference_type = ssize_t;
928 
929     constexpr __cursor() noexcept : __n(nullptr) { }
930 
931     constexpr explicit __cursor(__node *__p) noexcept : __n(__p) { }
932 
933     constexpr __cursor(const __cursor<!Const> &__o) noexcept
934       requires(Const)
935         : __n(__o.node_ptr())
936     {
937     }
938 
939     const T &
940     operator*() const
941     {
942       return __n->data;
943     }
944 
945     const T *
946     operator->() const
947     {
948       return micron::addressof(__n->data);
949     }
950 
951     __cursor &
952     operator++()
953     {
954       __n = __succ(__n);
955       return *this;
956     }
957 
958     __cursor
959     operator++(int)
960     {
961       __cursor __t = *this;
962       __n = __succ(__n);
963       return __t;
964     }
965 
966     bool
967     operator==(const __cursor &__o) const noexcept
968     {
969       return __n == __o.__n;
970     }
971 
972     bool
973     operator!=(const __cursor &__o) const noexcept
974     {
975       return __n != __o.__n;
976     }
977 
978     __node *
979     node_ptr() const noexcept
980     {
981       return __n;
982     }
983   };
