105 template<typename K> inline constexpr u16 lanes_for = (sizeof(K) == 4) ? lanes32 : ((sizeof(K) == 8) ? lanes64 : 0);

107 template<typename K, bool Strict> struct probe {
108   static constexpr u16 W = lanes_for<K>;
109   static constexpr bool sgn = micron::is_signed_v<K>;
110 
111 #if defined(__micron_x86_sse2)
112   __m128i tv;
113   __m128i bias;
114 
115   [[gnu::always_inline]] explicit probe(const K &key) noexcept
116   {
117     if constexpr ( sizeof(K) == 8 ) {
118       bias = simd::sse::splat_i64(sgn ? 0LL : static_cast<long long>(0x8000000000000000ULL));
119       tv = simd::sse::xor_i128(simd::sse::splat_i64(static_cast<long long>(key)), bias);
120     } else {
121       bias = simd::sse::splat_i32(sgn ? 0 : static_cast<int>(0x80000000u));
122       tv = simd::sse::xor_i128(simd::sse::splat_i32(static_cast<int>(key)), bias);
123     }
124   }
125 
126   [[nodiscard, gnu::always_inline]] u16
127   count(const K *p) const noexcept
128   {
129     const __m128i kv = simd::sse::xor_i128(simd::sse::loadu_i128(reinterpret_cast<const __m128i_u *>(p)), bias);
130     if constexpr ( sizeof(K) == 8 ) {
131       const __m128i cmp = Strict ? simd::sse::gt_i64(tv, kv) : simd::sse::gt_i64(kv, tv);
132       u32 m = static_cast<u32>(simd::sse::movemask_f64(simd::sse::cast_i128_to_f64(cmp))) & 0x3u;
133       if constexpr ( !Strict ) m = (~m) & 0x3u;
134       return static_cast<u16>(m == 0x3u ? 2 : __builtin_ctz(~m));
135     } else {
136       const __m128i cmp = Strict ? simd::sse::gt_i32(tv, kv) : simd::sse::gt_i32(kv, tv);
137       u32 m = static_cast<u32>(simd::sse::movemask_f32(simd::sse::cast_i128_to_f32(cmp))) & 0xFu;
138       if constexpr ( !Strict ) m = (~m) & 0xFu;
139       return static_cast<u16>(m == 0xFu ? 4 : __builtin_ctz(~m));
140     }
141   }
142 
143 #else
144   [[gnu::always_inline]] explicit probe(const K &) noexcept { }
145 
146   [[nodiscard, gnu::always_inline]] u16
147   count(const K *) const noexcept
148   {
149     return 0;
150   }
151 #endif

158 #if defined(MICRON_TREE_VECTOR_SCAN)
159 template<typename K, typename Compare, bool Strict>
160 [[nodiscard, gnu::always_inline]] inline u16
161 __vector_rank(const K *keys, u16 n, const K &key) noexcept
162 {
163   constexpr u16 W = __scan::lanes_for<K>;
164   const __scan::probe<K, Strict> pr(key);
165   u16 i = 0;
166   while ( static_cast<u16>(i + W) <= n ) {
167     const u16 got = pr.count(keys + i);
168     i = static_cast<u16>(i + got);
169     if ( got != W ) return i;
170   }
171   if constexpr ( Strict ) {
172     while ( i < n && Compare::lt(keys[i], key) ) ++i;
173   } else {
174     while ( i < n && !Compare::lt(key, keys[i]) ) ++i;
175   }
176   return i;
177 }
178 #endif      // MICRON_TREE_VECTOR_SCAN

180 // log2(n) comparisons, both updates conditional moves
181 template<typename K, typename Compare, bool Strict>
182 [[nodiscard, gnu::always_inline]] inline u16
183 __ladder_rank(const K *keys, u16 n, const K &key) noexcept
184 {
185   u16 lo = 0;
186   u16 len = n;
187   while ( len > 0 ) {
188     const u16 half = static_cast<u16>(len >> 1);
189     const u16 mid = static_cast<u16>(lo + half);
190     const bool go_right = Strict ? Compare::lt(keys[mid], key) : !Compare::lt(key, keys[mid]);
191     lo = go_right ? static_cast<u16>(mid + 1) : lo;
192     len = go_right ? static_cast<u16>(len - half - 1) : half;
193   }
194   return lo;
195 }

197 template<typename K, typename Compare, bool Strict>
198 [[nodiscard, gnu::always_inline]] inline u16
199 __linear_rank(const K *keys, u16 n, const K &key) noexcept
200 {
201   u16 i = 0;
202   if constexpr ( Strict ) {
203     while ( i < n && Compare::lt(keys[i], key) ) ++i;
204   } else {
205     while ( i < n && !Compare::lt(key, keys[i]) ) ++i;
206   }
207   return i;
208 }

210 // count of i in [0,n) with lt(keys[i], key)
211 template<typename K, typename Compare>
212 [[nodiscard, gnu::always_inline]] inline u16
213 lower_bound_scan(const K *keys, u16 n, const K &key) noexcept
214 {
215 #if defined(MICRON_TREE_VECTOR_SCAN)
216   if constexpr ( vector_rankable_v<K, Compare> && __scan::lanes_for<K> > 0 )
217     return __vector_rank<K, Compare, true>(keys, n, key);
218   else
219 #endif
220       if constexpr ( cheap_compare_v<K, Compare> )
221     return __linear_rank<K, Compare, true>(keys, n, key);
222   else

245 template<typename T>
246 inline constexpr bool is_byte_relocatable_v = micron::is_trivially_copyable_v<T> && micron::is_trivially_destructible_v<T>;
247 
248 template<typename T>
249 __attribute__((always_inline)) inline void
250 shift_right(T *base, usize first, usize last) noexcept
251 {
252   if ( first >= last ) return;
253   if constexpr ( is_byte_relocatable_v<T> ) {
254     const usize cnt = last - first;
255     using __ab = unsigned char __attribute__((may_alias));
256     __ab *d = reinterpret_cast<__ab *>(base + first + 1);
257     const __ab *s = reinterpret_cast<const __ab *>(base + first);
258     for ( usize i = cnt * sizeof(T); i > 0; --i ) d[i - 1] = s[i - 1];
259   } else {
260     for ( usize i = last; i > first; --i ) {
261       new (micron::addr(base[i])) T(micron::move(base[i - 1]));
262       base[i - 1].~T();
263     }
264   }
265 }

267 template<typename T>
268 __attribute__((always_inline)) inline void
269 erase_at(T *base, usize first, usize last) noexcept
270 {
271   if ( first >= last ) return;
272   if constexpr ( is_byte_relocatable_v<T> ) {
273     const usize cnt = (first + 1 < last) ? (last - first - 1) : 0;
274     if ( cnt ) {
275       using __ab = unsigned char __attribute__((may_alias));
276       __ab *d = reinterpret_cast<__ab *>(base + first);
277       const __ab *s = reinterpret_cast<const __ab *>(base + first + 1);
278       for ( usize i = 0; i < cnt * sizeof(T); ++i ) d[i] = s[i];
279     }
280   } else {
281     base[first].~T();
282     for ( usize i = first + 1; i < last; ++i ) {
283       new (micron::addr(base[i - 1])) T(micron::move(base[i]));
284       base[i].~T();
285     }
286   }
287 }

310 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
311 // node_arena
312 //
313 // NOTE: grow() reallocates the slab, and because raw_slot is trivially
314 // copyable that goes through __owned_memory_resource's realloc path, meaning the whole slab is moveable
315 // .. every leaf_node&/internal_node&/K*/V* derived from raw() is invalidated by any allocate() that grows
316 // .. K and V are byte-relocated even when they are not trivially copyable
317 
318 template<usize B, usize Al = 64, class Alloc = allocator_serial<>> class node_arena
319 {
320   using slot_t = raw_slot<B, Al>;
321   static_assert(B >= sizeof(node_idx), "slot too small to hold the free-list link");
322 
323   typedef node_idx __attribute__((may_alias)) __alias_idx;
324 
325   __immutable_memory_resource<slot_t, Alloc> mem_;
326   node_idx free_head_;
327 
328   [[gnu::always_inline]] static node_idx
329   link_load(const byte *p) noexcept
330   {
331     return *reinterpret_cast<const __alias_idx *>(p);
332   }
333 
334   [[gnu::always_inline]] static void
335   link_store(byte *p, node_idx v) noexcept
336   {
337     *reinterpret_cast<__alias_idx *>(p) = v;
338   }
339 
340   void
341   grow()
342   {
343     const usize old = mem_.capacity;
344     const usize minimum = old == static_cast<usize>(-1) ? old : old + 1;
345     if ( minimum == old ) exc<except::length_error>("node_arena: capacity overflow");
346     mem_.expand(mem_.recommended_capacity(old, minimum));
347   }
348 
349 public:
350   node_arena() : mem_(), free_head_(nil) { }
351 
352   node_arena(const node_arena &) = delete;
353   node_arena &operator=(const node_arena &) = delete;
354 
355   node_arena(node_arena &&o) noexcept : mem_(micron::move(o.mem_)), free_head_(o.free_head_) { o.free_head_ = nil; }
356 
357   node_arena &
358   operator=(node_arena &&o) noexcept
359   {
360     if ( this != &o ) {
361       mem_ = micron::move(o.mem_);
362       free_head_ = o.free_head_;
363       o.free_head_ = nil;
364     }
365     return *this;
366   }
367 
368   [[gnu::always_inline]] byte *
369   raw(node_idx i) noexcept
370   {
371     return mem_.memory[i].raw;
372   }
373 
374   [[gnu::always_inline]] const byte *
375   raw(node_idx i) const noexcept
376   {
377     return mem_.memory[i].raw;
378   }
379 
380   node_idx
381   allocate()
382   {
383     if ( free_head_ != nil ) {
384       const node_idx i = free_head_;
385       free_head_ = link_load(mem_.memory[i].raw);
386       return i;
387     }
388     if ( mem_.length >= mem_.capacity ) grow();
389     if ( mem_.length >= static_cast<usize>(max_slots) ) exc<except::length_error>("node_arena: node index space exhausted");
390     return static_cast<node_idx>(mem_.length++);
391   }
392 
393   void
394   deallocate(node_idx i) noexcept
395   {
396     link_store(mem_.memory[i].raw, free_head_);
397     free_head_ = i;
398   }
399 
400   void
401   reset() noexcept
402   {
403     mem_.length = 0;
404     free_head_ = nil;
405   }
406 
407   void
408   swap(node_arena &o) noexcept
409   {
410     auto tm = micron::move(mem_);
411     mem_ = micron::move(o.mem_);
412     o.mem_ = micron::move(tm);
413     const node_idx th = free_head_;
414     free_head_ = o.free_head_;
415     o.free_head_ = th;
416   }
417 
418   void
419   reserve(usize n_slots)
420   {
421     if ( mem_.capacity < n_slots ) mem_.expand(n_slots);
422   }
423 
424   [[nodiscard, gnu::always_inline]] usize
425   slots_used() const noexcept
426   {
427     return mem_.length;
428   }
429 
430   [[nodiscard]] usize
431   slots_live() const noexcept
432   {
433     usize freed = 0;
434     for ( node_idx i = free_head_; i != nil; ++freed ) i = link_load(mem_.memory[i].raw);
435     return mem_.length - freed;
436   }
437 
438   [[nodiscard, gnu::always_inline]] usize
439   slots_reserved() const noexcept
440   {
441     return mem_.capacity;
442   }
443 
444   [[nodiscard, gnu::always_inline]] static constexpr usize
445   slot_bytes() noexcept
446   {
447     return sizeof(slot_t);
448   }
449 };

454 template<typename T, class Alloc = allocator_serial<>, usize MinBytes = 4096, usize MaxBytes = 4096u * 256u> class block_pool
455 {
456   typedef T *__attribute__((may_alias)) __alias_ptr;
457 
458   static_assert(sizeof(T) >= sizeof(T *), "a pool slot must be able to hold a chain link");
459 
460   static constexpr usize __min_slots = 8;
461 
462   T *blocks_;        // block chain; slot 0 of each block is the next-block link, never handed out
463   T *bump_;          // next unused slot in the newest block
464   T *bump_end_;      // one past the newest block's last slot
465   T *free_;          // free list head
466   usize block_bytes_;
467 
468   [[gnu::always_inline]] static T *
469   link_load(const T *slot) noexcept
470   {
471     return *reinterpret_cast<const __alias_ptr *>(slot);
472   }
473 
474   [[gnu::always_inline]] static void
475   link_store(T *slot, T *v) noexcept
476   {
477     *reinterpret_cast<__alias_ptr *>(slot) = v;
478   }
479 
480   void
481   add_block()
482   {
483     usize slots = block_bytes_ / sizeof(T);
484     if ( slots < __min_slots ) slots = __min_slots;
485     T *blk = __allocator_allocate_array<Alloc, T>(slots);
486     link_store(blk, blocks_);
487     blocks_ = blk;
488     bump_ = blk + 1;
489     bump_end_ = blk + slots;
490     if ( block_bytes_ < MaxBytes ) block_bytes_ <<= 1;
491   }
492 
493 public:
494   block_pool() noexcept : blocks_(nullptr), bump_(nullptr), bump_end_(nullptr), free_(nullptr), block_bytes_(MinBytes) { }
495 
496   block_pool(const block_pool &) = delete;
497   block_pool &operator=(const block_pool &) = delete;
498 
499   block_pool(block_pool &&o) noexcept
500       : blocks_(o.blocks_), bump_(o.bump_), bump_end_(o.bump_end_), free_(o.free_), block_bytes_(o.block_bytes_)
501   {
502     o.blocks_ = nullptr;
503     o.bump_ = nullptr;
504     o.bump_end_ = nullptr;
505     o.free_ = nullptr;
506     o.block_bytes_ = MinBytes;
507   }
508 
509   block_pool &
510   operator=(block_pool &&o) noexcept
511   {
512     if ( this != &o ) {
513       release();
514       blocks_ = o.blocks_;
515       bump_ = o.bump_;
516       bump_end_ = o.bump_end_;
517       free_ = o.free_;
518       block_bytes_ = o.block_bytes_;
519       o.blocks_ = nullptr;
520       o.bump_ = nullptr;
521       o.bump_end_ = nullptr;
522       o.free_ = nullptr;
523       o.block_bytes_ = MinBytes;
524     }
525     return *this;
526   }
527 
528   ~block_pool() { release(); }
529 
530   [[nodiscard, gnu::always_inline]] T *
531   take()
532   {
533     if ( free_ ) {
534       T *p = free_;
535       free_ = link_load(p);
536       return p;
537     }
538     if ( bump_ == bump_end_ ) [[unlikely]]
539       add_block();
540     return bump_++;
541   }
542 
543   [[gnu::always_inline]] void
544   give(T *p) noexcept
545   {
546     link_store(p, free_);
547     free_ = p;
548   }
549 
550   void
551   release() noexcept
552   {
553     T *b = blocks_;
554     while ( b ) {
555       T *nx = link_load(b);
556       __allocator_deallocate_array<Alloc>(b);
557       b = nx;
558     }
559     blocks_ = nullptr;
560     bump_ = nullptr;
561     bump_end_ = nullptr;
562     free_ = nullptr;
563     block_bytes_ = MinBytes;
564   }
565 
566   // carve one run big enough for n more objects, so a build of known size stays contiguous
567   void
568   reserve(usize n)
569   {
570     const usize have = bump_end_ ? static_cast<usize>(bump_end_ - bump_) : 0;
571     if ( n <= have ) return;
572     const usize want = (n - have + 1) * sizeof(T);
573     while ( block_bytes_ < want && block_bytes_ < MaxBytes ) block_bytes_ <<= 1;
574     if ( block_bytes_ < want ) block_bytes_ = want;
575     add_block();
576     (void)0;
577   }
578 
579   void
580   swap(block_pool &o) noexcept
581   {
582     micron::swap(blocks_, o.blocks_);
583     micron::swap(bump_, o.bump_);
584     micron::swap(bump_end_, o.bump_end_);
585     micron::swap(free_, o.free_);
586     micron::swap(block_bytes_, o.block_bytes_);
587   }
588 
589   [[nodiscard]] usize
590   blocks() const noexcept
591   {
592     usize n = 0;
593     for ( const T *b = blocks_; b; b = link_load(b) ) ++n;
594     return n;
595   }
596 
597   [[nodiscard]] usize
598   bytes_held() const noexcept
599   {
600     usize total = 0;
601     usize bs = MinBytes;
602     for ( usize i = 0, k = blocks(); i < k; ++i ) {
603       total += bs;
604       if ( bs < MaxBytes ) bs <<= 1;
605     }
606     return total;
607   }
608 };
