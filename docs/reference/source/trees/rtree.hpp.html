103   void
104   full_rec(node_idx n, Fn &fn) noexcept
105   {
106     if ( is_leaf(n) ) {
107       leaf_node &L = lnode(n);

121   void
122   knn_rec(node_idx n, const point_type &p, usize k, fvector<knn_item> &best) noexcept
123   {
124     if ( is_leaf(n) ) {
125       leaf_node &L = lnode(n);
126       for ( u16 i = 0; i < L.count; ++i ) {
127         F d = mindist2(p, L.mbr[i]);
128         if ( best.size() < k || d < best[best.size() - 1].d ) knn_insert(best, k, knn_item{ d, L.mbr[i], micron::addr(L.vals()[i]) });
129       }
130       return;
131     }
132     internal_node &I = inode(n);
133     u16 order[__cap];
134     F od[__cap];
135     for ( u16 i = 0; i < I.count; ++i ) {
136       order[i] = i;
137       od[i] = mindist2(p, I.mbr[i]);
138     }
139     for ( u16 i = 1; i < I.count; ++i ) {
140       u16 c = order[i];
141       F key = od[c];
142       int j = static_cast<int>(i) - 1;
143       while ( j >= 0 && od[order[j]] > key ) {
144         order[j + 1] = order[j];
145         --j;
146       }
147       order[j + 1] = c;
148     }
149     for ( u16 t = 0; t < I.count; ++t ) {
150       F md = od[order[t]];
151       if ( best.size() >= k && md >= best[best.size() - 1].d ) break;      // prune remaining (sorted)
152       knn_rec(I.kids[order[t]], p, k, best);
153     }
154   }

156   static void
157   knn_insert(fvector<knn_item> &best, usize k, const knn_item &it)
158   {
159     usize pos = best.size();
160     best.push_back(it);
161     while ( pos > 0 && best[pos - 1].d > best[pos].d ) {
162       knn_item tmp = best[pos - 1];
163       best[pos - 1] = best[pos];
164       best[pos] = tmp;
165       --pos;
166     }
167     if ( best.size() > k ) best.pop_back();
168   }

200   static bool
201   same_box(const box_type &a, const box_type &b) noexcept
202   {
203     for ( usize d = 0; d < Dim; ++d )
204       if ( a.min_corner.data[d] != b.min_corner.data[d] || a.max_corner.data[d] != b.max_corner.data[d] ) return false;
205     return true;
206   }

208   static F
209   area(const box_type &b) noexcept
210   {
211     F a = F(1);
212     F lo = F(0);
213     for ( usize d = 0; d < Dim; ++d ) {
214       const F e = b.max_corner.data[d] - b.min_corner.data[d];
215       a *= e;
216       lo = e < lo ? e : lo;
217     }
218     return lo < F(0) ? F(0) : a;
219   }

221   static F
222   margin(const box_type &b) noexcept
223   {
224     F s = F(0);
225     for ( usize d = 0; d < Dim; ++d ) s += (b.max_corner.data[d] - b.min_corner.data[d]);
226     return s;
227   }

229   static F
230   overlap(const box_type &a, const box_type &b) noexcept
231   {
232     F o = F(1);
233     for ( usize d = 0; d < Dim; ++d ) {
234       F lo = a.min_corner.data[d] > b.min_corner.data[d] ? a.min_corner.data[d] : b.min_corner.data[d];
235       F hi = a.max_corner.data[d] < b.max_corner.data[d] ? a.max_corner.data[d] : b.max_corner.data[d];
236       F e = hi - lo;
237       if ( e <= F(0) ) return F(0);
238       o *= e;
239     }
240     return o;
241   }

251   static F
252   enlargement(const box_type &e, const box_type &b) noexcept
253   {
254     F united = F(1);
255     F own = F(1);
256     for ( usize d = 0; d < Dim; ++d ) {
257       const F lo = e.min_corner.data[d] < b.min_corner.data[d] ? e.min_corner.data[d] : b.min_corner.data[d];
258       const F hi = e.max_corner.data[d] > b.max_corner.data[d] ? e.max_corner.data[d] : b.max_corner.data[d];
259       united *= (hi - lo);
260       own *= (e.max_corner.data[d] - e.min_corner.data[d]);
261     }
262     return united - own;
263   }

265   static F
266   mindist2(const point_type &p, const box_type &b) noexcept
267   {
268     F s = F(0);
269     for ( usize d = 0; d < Dim; ++d ) {
270       const F v = p.data[d];
271       const F a = b.min_corner.data[d] - v;
272       const F c = v - b.max_corner.data[d];
273       F dd = a > c ? a : c;
274       dd = dd > F(0) ? dd : F(0);
275       s += dd * dd;
276     }
277     return s;
278   }

330   u16
331   choose_subtree(internal_node &I, const box_type &b) noexcept
332   {
333     const bool kids_are_leaves = is_leaf(I.kids[0]);
334     u16 best = 0;
335     if ( kids_are_leaves ) {
336       F best_ov = F(0), best_en = F(0), best_ar = F(0);
337       for ( u16 i = 0; i < I.count; ++i ) {
338         box_type comb = combine(I.mbr[i], b);
339         F ov = F(0);
340         for ( u16 j = 0; j < I.count; ++j )
341           if ( j != i ) ov += overlap(comb, I.mbr[j]) - overlap(I.mbr[i], I.mbr[j]);
342         F en = area(comb) - area(I.mbr[i]);
343         F ar = area(I.mbr[i]);
344         if ( i == 0 || ov < best_ov || (ov == best_ov && en < best_en) || (ov == best_ov && en == best_en && ar < best_ar) ) {
345           best = i;
346           best_ov = ov;
347           best_en = en;
348           best_ar = ar;
349         }
350       }
351     } else {
352       F best_en = F(0), best_ar = F(0);
353       for ( u16 i = 0; i < I.count; ++i ) {
354         F en = enlargement(I.mbr[i], b);
355         F ar = area(I.mbr[i]);
356         if ( i == 0 || en < best_en || (en == best_en && ar < best_ar) ) {
357           best = i;
358           best_en = en;
359           best_ar = ar;
360         }
361       }
362     }
363     return best;
364   }

366   static void
367   sort_idx(u16 *idx, const box_type *mbr, u16 n, usize axis, bool by_max) noexcept
368   {
369     for ( u16 i = 1; i < n; ++i ) {
370       u16 cur = idx[i];
371       F key = by_max ? mbr[cur].max_corner.data[axis] : mbr[cur].min_corner.data[axis];
372       int j = static_cast<int>(i) - 1;
373       while ( j >= 0 ) {
374         F kj = by_max ? mbr[idx[j]].max_corner.data[axis] : mbr[idx[j]].min_corner.data[axis];
375         if ( kj <= key ) break;
376         idx[j + 1] = idx[j];
377         --j;
378       }
379       idx[j + 1] = cur;
380     }
381   }

383   void
384   choose_split(const box_type *mbr, u16 *out_perm, u16 &out_k) noexcept
385   {
386     const u16 n = __cap;
387     u16 idx[__cap];
388     usize best_axis = 0;
389     F best_axis_margin = F(0);
390     bool first_axis = true;
391     for ( usize axis = 0; axis < Dim; ++axis ) {
392       F margin_sum = F(0);
393       for ( int mode = 0; mode < 2; ++mode ) {
394         for ( u16 i = 0; i < n; ++i ) idx[i] = i;
395         sort_idx(idx, mbr, n, axis, mode == 1);
396         for ( u16 k = __minf; k <= n - __minf; ++k ) {
397           box_type b1 = box_type::empty();
398           box_type b2 = box_type::empty();
399           for ( u16 t = 0; t < k; ++t ) b1.extend(mbr[idx[t]]);
400           for ( u16 t = k; t < n; ++t ) b2.extend(mbr[idx[t]]);
401           margin_sum += margin(b1) + margin(b2);
402         }
403       }
404       if ( first_axis || margin_sum < best_axis_margin ) {
405         best_axis = axis;
406         best_axis_margin = margin_sum;
407         first_axis = false;
408       }
409     }
410     F best_ov = F(0), best_ar = F(0);
411     bool first = true;
412     for ( int mode = 0; mode < 2; ++mode ) {
413       for ( u16 i = 0; i < n; ++i ) idx[i] = i;
414       sort_idx(idx, mbr, n, best_axis, mode == 1);
415       for ( u16 k = __minf; k <= n - __minf; ++k ) {
416         box_type b1 = box_type::empty();
417         box_type b2 = box_type::empty();
418         for ( u16 t = 0; t < k; ++t ) b1.extend(mbr[idx[t]]);
419         for ( u16 t = k; t < n; ++t ) b2.extend(mbr[idx[t]]);
420         F ov = overlap(b1, b2);
421         F ar = area(b1) + area(b2);
422         if ( first || ov < best_ov || (ov == best_ov && ar < best_ar) ) {
423           best_ov = ov;
424           best_ar = ar;
425           out_k = k;
426           for ( u16 t = 0; t < n; ++t ) out_perm[t] = idx[t];
427           first = false;
428         }
429       }
430     }
431   }

433   void
434   split_leaf(node_idx n, node_idx &out_right, box_type &out_rmbr)
435   {
436     u16 perm[__cap];
437     u16 k = __minf;
438     choose_split(lnode(n).mbr, perm, k);
439 
440     node_idx ridx = alloc_leaf();      // may move slab; re-fetch below
441     leaf_node &L = lnode(n);
442     leaf_node &R = lnode(ridx);
443 
444     box_type sbox[__cap];
445     alignas(Value) byte sraw[sizeof(Value) * __cap];
446     Value *sv = static_cast<Value *>(static_cast<void *>(sraw));
447     for ( u16 t = 0; t < __cap; ++t ) {
448       sbox[t] = L.mbr[perm[t]];
449       new (micron::addr(sv[t])) Value(micron::move(L.vals()[perm[t]]));
450     }
451     for ( u16 i = 0; i < __cap; ++i ) L.vals()[i].~Value();
452 
453     for ( u16 t = 0; t < k; ++t ) {
454       L.mbr[t] = sbox[t];
455       new (micron::addr(L.vals()[t])) Value(micron::move(sv[t]));
456     }
457     L.count = k;
458     for ( u16 t = k; t < __cap; ++t ) {
459       R.mbr[t - k] = sbox[t];
460       new (micron::addr(R.vals()[t - k])) Value(micron::move(sv[t]));
461     }
462     R.count = static_cast<u16>(__cap - k);
463     for ( u16 t = 0; t < __cap; ++t ) sv[t].~Value();
464 
465     out_right = ridx;
466     out_rmbr = subtree_mbr(ridx);
467   }

501   void
502   reinsert_leaf(node_idx n)
503   {
504     leaf_node &L = lnode(n);
505     box_type bb = subtree_mbr(n);
506     point_type c = bb.center();
507 
508     u16 idx[__cap];
509     F dist[__cap];
510     for ( u16 i = 0; i < __cap; ++i ) {
511       idx[i] = i;
512       point_type ec = L.mbr[i].center();
513       F dd = F(0);
514       for ( usize d = 0; d < Dim; ++d ) {
515         F df = ec.data[d] - c.data[d];
516         dd += df * df;
517       }
518       dist[i] = dd;
519     }
520     // sort idx by dist ascending (insertion sort)
521     for ( u16 i = 1; i < __cap; ++i ) {
522       u16 cur = idx[i];
523       F key = dist[cur];
524       int j = static_cast<int>(i) - 1;
525       while ( j >= 0 && dist[idx[j]] > key ) {
526         idx[j + 1] = idx[j];
527         --j;
528       }
529       idx[j + 1] = cur;
530     }
531     const u16 keep = static_cast<u16>(__cap - __prei);
532 
533     box_type sbox[__cap];
534     alignas(Value) byte sraw[sizeof(Value) * __cap];
535     Value *sv = static_cast<Value *>(static_cast<void *>(sraw));
536     for ( u16 t = 0; t < __cap; ++t ) {
537       sbox[t] = L.mbr[idx[t]];
538       new (micron::addr(sv[t])) Value(micron::move(L.vals()[idx[t]]));
539     }
540     for ( u16 i = 0; i < __cap; ++i ) L.vals()[i].~Value();
541 
542     for ( u16 t = 0; t < keep; ++t ) {
543       L.mbr[t] = sbox[t];
544       new (micron::addr(L.vals()[t])) Value(micron::move(sv[t]));
545     }
546     L.count = keep;
547     for ( u16 t = keep; t < __cap; ++t ) __reins.push_back(pending{ sbox[t], sv[t] });
548     for ( u16 t = 0; t < __cap; ++t ) sv[t].~Value();
549   }

551   bool
552   insert_descend(node_idx n, const box_type &b, const Value &v, node_idx &out_right, box_type &out_rmbr)
553   {
554     if ( is_leaf(n) ) {
555       leaf_node &L = lnode(n);
556       L.mbr[L.count] = b;
557       new (micron::addr(L.vals()[L.count])) Value(v);
558       ++L.count;
559       if ( L.count <= M ) return false;
560       if ( !__did_reinsert && n != __root ) {
561         __did_reinsert = true;
562         reinsert_leaf(n);
563         return false;
564       }
565       split_leaf(n, out_right, out_rmbr);
566       return true;
567     }
568     u16 ci = choose_subtree(inode(n), b);
569     node_idx child = inode(n).kids[ci];
570     node_idx cr;
571     box_type crm;
572     bool csplit = insert_descend(child, b, v, cr, crm);
573     inode(n).mbr[ci].extend(b);
574     if ( !csplit ) {
575       // child may have shrunk via reinsert; its mbr stays a valid superset (correct).
576       return false;
577     }
578     internal_node &I = inode(n);
579     I.mbr[ci] = subtree_mbr(I.kids[ci]);
580     I.mbr[I.count] = crm;
581     I.kids[I.count] = cr;
582     ++I.count;
583     if ( I.count <= M ) return false;
584     split_internal(n, out_right, out_rmbr);
585     return true;
586   }

588   void
589   insert_from_root(const box_type &b, const Value &v)
590   {
591     node_idx rr;
592     box_type rm;
593     bool s = insert_descend(__root, b, v, rr, rm);
594     if ( s ) {
595       node_idx nr = alloc_internal();      // may move slab; re-fetch
596       internal_node &NR = inode(nr);
597       NR.count = 2;
598       NR.kids[0] = __root;
599       NR.kids[1] = rr;
600       NR.mbr[1] = rm;
601       node_idx oldroot = __root;
602       __root = nr;
603       inode(nr).mbr[0] = subtree_mbr(oldroot);
604     }
605   }

607   template<typename Fn>
608   void
609   query_rec(node_idx n, const box_type &q, Fn &fn) noexcept
610   {
611     if ( is_leaf(n) ) {
612       leaf_node &L = lnode(n);
613       for ( u16 i = 0; i < L.count; ++i )
614         if ( L.mbr[i].intersects(q) ) fn(static_cast<const box_type &>(L.mbr[i]), L.vals()[i]);
615       return;
616     }
617     internal_node &I = inode(n);
618     for ( u16 i = 0; i < I.count; ++i )
619       if ( I.mbr[i].intersects(q) ) query_rec(I.kids[i], q, fn);
620   }

622   void
623   collect_data(node_idx n)
624   {
625     if ( is_leaf(n) ) {
626       leaf_node &L = lnode(n);
627       for ( u16 i = 0; i < L.count; ++i ) __reins.push_back(pending{ L.mbr[i], L.vals()[i] });
628       return;
629     }
630     internal_node &I = inode(n);
631     u16 c = I.count;
632     node_idx kids[__cap];
633     for ( u16 i = 0; i < c; ++i ) kids[i] = I.kids[i];
634     for ( u16 i = 0; i < c; ++i ) collect_data(kids[i]);
635   }
636 
637   static void
638   remove_internal_entry(internal_node &I, u16 i) noexcept
639   {
640     for ( u16 j = i; j + 1 < I.count; ++j ) {
641       I.mbr[j] = I.mbr[j + 1];
642       I.kids[j] = I.kids[j + 1];
643     }
644     --I.count;
645   }
646 
647   void
648   remove_leaf_entry(leaf_node &L, u16 i) noexcept
649   {
650     L.vals()[i].~Value();
651     for ( u16 j = i; j + 1 < L.count; ++j ) {
652       L.mbr[j] = L.mbr[j + 1];
653       new (micron::addr(L.vals()[j])) Value(micron::move(L.vals()[j + 1]));
654       L.vals()[j + 1].~Value();
655     }
656     --L.count;
657   }

659   bool
660   erase_rec(node_idx n, const box_type &b, const Value &v)
661   {
662     if ( is_leaf(n) ) {
663       leaf_node &L = lnode(n);
664       for ( u16 i = 0; i < L.count; ++i ) {
665         if ( same_box(L.mbr[i], b) && L.vals()[i] == v ) {
666           remove_leaf_entry(L, i);
667           return true;
668         }
669       }
670       return false;
671     }
672     internal_node &I = inode(n);
673     for ( u16 ci = 0; ci < I.count; ++ci ) {
674       if ( !I.mbr[ci].intersects(b) ) continue;
675       if ( !erase_rec(I.kids[ci], b, v) ) continue;
676       node_idx child = inode(n).kids[ci];
677       if ( entries(child) < __minf ) {
678         collect_data(child);      // gather orphaned data entries for reinsertion
679         free_subtree(child);
680         remove_internal_entry(inode(n), ci);
681       } else {
682         inode(n).mbr[ci] = subtree_mbr(child);
683       }
684       return true;
685     }
686     return false;
687   }

785   void
786   insert(const box_type &b, const Value &v)
787   {
788     if ( __root == nil ) __root = alloc_leaf();
789     __did_reinsert = false;
790     __reins.clear();
791     insert_from_root(b, v);
792     ++__size;
793     for ( usize i = 0; i < __reins.size(); ++i ) insert_from_root(__reins[i].b, __reins[i].v);

795   }

806   bool
807   erase(const box_type &b, const Value &v)
808   {
809     if ( __root == nil ) return false;
810     __reins.clear();
811     bool removed = erase_rec(__root, b, v);
812     if ( !removed ) return false;
813     --__size;
814     if ( __root != nil ) {
815       if ( !is_leaf(__root) && inode(__root).count == 1 ) {
816         node_idx old = __root;
817         __root = inode(__root).kids[0];
818         inode(old).count = 0;      // detach so destroy_node doesn't recurse
819         destroy_node(old);
820       } else if ( is_leaf(__root) && lnode(__root).count == 0 ) {
821         destroy_node(__root);
822         __root = nil;
823       }
824     }
825     if ( !__reins.empty() ) {
826       __did_reinsert = true;      // orphans must not re-trigger forced reinsert cascades
827       fvector<pending> orphans(micron::move(__reins));
828       __reins.clear();
829       for ( usize i = 0; i < orphans.size(); ++i ) {
830         if ( __root == nil ) __root = alloc_leaf();
831         insert_from_root(orphans[i].b, orphans[i].v);
832       }
833     }
834     return removed;
835   }

892   template<class Acc, class LeafElemFn, class InnerFn>
893   Acc
894   __cata_rec(node_idx n, const Acc &empty, LeafElemFn &leaf_elem, InnerFn &inner)
895   {
896     if ( is_leaf(n) ) {
897       leaf_node &L = lnode(n);
898       Acc acc = empty;
899       for ( u16 i = 0; i < L.count; ++i )
900         acc = leaf_elem(micron::move(acc), static_cast<const box_type &>(L.mbr[i]), static_cast<const Value &>(L.vals()[i]));
901       return acc;
902     }
903     internal_node &I = inode(n);
904     Acc kids[__cap];
905     for ( u16 i = 0; i < I.count; ++i ) kids[i] = __cata_rec<Acc>(I.kids[i], empty, leaf_elem, inner);
906     return inner(kids, I.count);
907   }

909   template<class Fn>
910   walk_ctl
911   traverse(Fn fn) const
912   {
913     rtree *self = const_cast<rtree *>(this);
914     if ( self->__root == nil ) return walk_ctl::continue_;
915     return self->__traverse_rec(self->__root, fn);
916   }

918   template<class Fn>
919   walk_ctl
920   __traverse_rec(node_idx n, Fn &fn)
921   {
922     if ( is_leaf(n) ) {
923       leaf_node &L = lnode(n);
924       for ( u16 i = 0; i < L.count; ++i )
925         if ( micron::__impl::invoke_walk(fn, static_cast<const box_type &>(L.mbr[i]), static_cast<const Value &>(L.vals()[i]))
926              == walk_ctl::stop )
927           return walk_ctl::stop;
928       return walk_ctl::continue_;
929     }
930     internal_node &I = inode(n);
931     for ( u16 i = 0; i < I.count; ++i )
932       if ( __traverse_rec(I.kids[i], fn) == walk_ctl::stop ) return walk_ctl::stop;
933     return walk_ctl::continue_;
934   }

938   nearest(const point_type &p, usize k, Fn &&fn) noexcept
939   {
940     if ( __root == nil || k == 0 ) return;

945   }
