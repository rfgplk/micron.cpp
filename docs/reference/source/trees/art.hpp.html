153   void
154   __destroy_leaves(__node_base *n) noexcept
155   {
156     if ( !n ) return;
157     if ( n->kind == __node_kind::leaf ) {
158       __leaf *cur = static_cast<__leaf *>(n);
159       while ( cur ) {
160         __leaf *next = cur->next;
161         cur->~__leaf();
162         cur = next;
163       }
164       return;
165     }
166     switch ( n->kind ) {
167     case __node_kind::n4: {
168       auto *p = static_cast<__n4 *>(n);
169       for ( u8 i = 0; i < p->num_children; ++i ) __destroy_leaves(p->children[i]);
170       break;
171     }
172     case __node_kind::n16: {
173       auto *p = static_cast<__n16 *>(n);
174       for ( u8 i = 0; i < p->num_children; ++i ) __destroy_leaves(p->children[i]);
175       break;
176     }
177     case __node_kind::n48: {
178       auto *p = static_cast<__n48 *>(n);
179       for ( u16 i = 0; i < 48; ++i )
180         if ( p->children[i] ) __destroy_leaves(p->children[i]);
181       break;
182     }
183     case __node_kind::n256: {
184       auto *p = static_cast<__n256 *>(n);
185       for ( u16 i = 0; i < 256; ++i ) __destroy_leaves(p->children[i]);
186       break;
187     }
188     default:
189       break;
190     }

216   static __node_base *
217   __find_child(__node_base *n, u8 b) noexcept
218   {
219     if ( !n || n->kind == __node_kind::leaf ) return nullptr;
220     switch ( n->kind ) {
221     case __node_kind::n4: {
222       auto *p = static_cast<__n4 *>(n);
223       for ( u8 i = 0; i < p->num_children; ++i )
224         if ( p->keys[i] == b ) return p->children[i];
225       return nullptr;
226     }
227     case __node_kind::n16: {
228       auto *p = static_cast<__n16 *>(n);
229       for ( u8 i = 0; i < p->num_children; ++i )

232     }
233     case __node_kind::n48: {
234       auto *p = static_cast<__n48 *>(n);
235       u8 ix = p->idx[b];
236       return ix == 0 ? nullptr : p->children[ix - 1];
237     }
238     case __node_kind::n256: {
239       auto *p = static_cast<__n256 *>(n);

278   __node_base *
279   __grow(__node_base *n)
280   {
281     switch ( n->kind ) {
282     case __node_kind::n4: {
283       auto *p = static_cast<__n4 *>(n);
284       auto *q = __mk(__p16);
285       q->num_children = p->num_children;
286       for ( u8 i = 0; i < 4u; ++i ) {
287         q->keys[i] = p->keys[i];
288         q->children[i] = p->children[i];
289       }
290       __rm(__p4, p);
291       return q;
292     }
293     case __node_kind::n16: {
294       auto *p = static_cast<__n16 *>(n);
295       auto *q = __mk(__p48);
296       q->num_children = p->num_children;
297       for ( u8 i = 0; i < 16u; ++i ) {
298         q->idx[p->keys[i]] = i + 1;
299         q->children[i] = p->children[i];
300       }
301       __rm(__p16, p);
302       return q;
303     }
304     case __node_kind::n48: {
305       auto *p = static_cast<__n48 *>(n);
306       auto *q = __mk(__p256);
307       q->num_children = p->num_children;
308       for ( int b = 0; b < 256; ++b ) {
309         if ( p->idx[b] != 0 ) q->children[b] = p->children[p->idx[b] - 1];
310       }
311       __rm(__p48, p);
312       return q;
313     }
314     default:
315       return n;
316     }

319   __node_base *
320   __add_child(__node_base *n, u8 b, __node_base *child)
321   {
322     switch ( n->kind ) {
323     case __node_kind::n4: {
324       auto *p = static_cast<__n4 *>(n);
325       if ( p->num_children == 4 ) {
326         n = __grow(n);
327         return __add_child(n, b, child);
328       }
329       p->keys[p->num_children] = b;
330       p->children[p->num_children] = child;
331       ++p->num_children;
332       return n;
333     }
334     case __node_kind::n16: {
335       auto *p = static_cast<__n16 *>(n);
336       if ( p->num_children == 16 ) {
337         n = __grow(n);
338         return __add_child(n, b, child);
339       }
340       p->keys[p->num_children] = b;
341       p->children[p->num_children] = child;
342       ++p->num_children;
343       return n;
344     }
345     case __node_kind::n48: {
346       auto *p = static_cast<__n48 *>(n);
347       if ( p->num_children == 48 ) {
348         n = __grow(n);
349         return __add_child(n, b, child);
350       }
351       p->idx[b] = p->num_children + 1;
352       p->children[p->num_children] = child;
353       ++p->num_children;
354       return n;
355     }
356     case __node_kind::n256: {
357       auto *p = static_cast<__n256 *>(n);
358       if ( !p->children[b] ) ++p->num_children;
359       p->children[b] = child;
360       return n;
361     }
362     default:
363       return n;
364     }

368   __shrink(__node_base *n)
369   {
370     switch ( n->kind ) {
371     case __node_kind::n16: {
372       auto *p = static_cast<__n16 *>(n);
373       auto *q = __mk(__p4);
374       q->num_children = p->num_children;
375       for ( u8 i = 0; i < p->num_children; ++i ) {
376         q->keys[i] = p->keys[i];
377         q->children[i] = p->children[i];
378       }
379       __rm(__p16, p);
380       return q;
381     }
382     case __node_kind::n48: {
383       auto *p = static_cast<__n48 *>(n);
384       auto *q = __mk(__p16);
385       u8 m = 0;
386       for ( int bb = 0; bb < 256; ++bb )
387         if ( p->idx[bb] != 0 ) {
388           q->keys[m] = static_cast<u8>(bb);
389           q->children[m] = p->children[p->idx[bb] - 1];
390           ++m;
391         }
392       q->num_children = m;
393       __rm(__p48, p);
394       return q;
395     }
396     case __node_kind::n256: {
397       auto *p = static_cast<__n256 *>(n);
398       auto *q = __mk(__p48);
399       u8 m = 0;
400       for ( int bb = 0; bb < 256; ++bb )
401         if ( p->children[bb] ) {
402           q->children[m] = p->children[bb];
403           q->idx[bb] = static_cast<u8>(m + 1);
404           ++m;
405         }
406       q->num_children = m;
407       __rm(__p256, p);
408       return q;
409     }
410     default:
411       return n;

415   __node_base *
416   __remove_child(__node_base *n, u8 b)
417   {
418     switch ( n->kind ) {
419     case __node_kind::n4: {
420       auto *p = static_cast<__n4 *>(n);
421       for ( u8 i = 0; i < p->num_children; ++i ) {
422         if ( p->keys[i] == b ) {
423           for ( u8 j = i; j + 1 < p->num_children; ++j ) {
424             p->keys[j] = p->keys[j + 1];
425             p->children[j] = p->children[j + 1];
426           }
427           --p->num_children;
428           if ( p->num_children == 0 ) {
429             __rm(__p4, p);
430             return nullptr;
431           }
432           return n;
433         }
434       }
435       return n;
436     }
437     case __node_kind::n16: {
438       auto *p = static_cast<__n16 *>(n);
439       for ( u8 i = 0; i < p->num_children; ++i ) {
440         if ( p->keys[i] == b ) {
441           for ( u8 j = i; j + 1 < p->num_children; ++j ) {
442             p->keys[j] = p->keys[j + 1];
443             p->children[j] = p->children[j + 1];
444           }
445           --p->num_children;
446           if ( p->num_children == 0 ) {
447             __rm(__p16, p);
448             return nullptr;
449           }
450           if ( p->num_children <= 3 ) return __shrink(n);
451           return n;
452         }
453       }
454       return n;
455     }
456     case __node_kind::n48: {
457       auto *p = static_cast<__n48 *>(n);
458       u8 ix = p->idx[b];
459       if ( ix == 0 ) return n;
460       u16 slot = static_cast<u16>(ix - 1);
461       u16 last = static_cast<u16>(p->num_children - 1);
462       if ( slot != last ) {
463         p->children[slot] = p->children[last];
464         for ( int bb = 0; bb < 256; ++bb ) {
465           if ( p->idx[bb] == last + 1 ) {
466             p->idx[bb] = static_cast<u8>(slot + 1);
467             break;
468           }
469         }
470       }
471       p->children[last] = nullptr;
472       p->idx[b] = 0;
473       --p->num_children;
474       if ( p->num_children == 0 ) {
475         __rm(__p48, p);
476         return nullptr;
477       }
478       if ( p->num_children <= 12 ) return __shrink(n);
479       return n;
480     }
481     case __node_kind::n256: {
482       auto *p = static_cast<__n256 *>(n);
483       if ( p->children[b] ) {
484         p->children[b] = nullptr;
485         --p->num_children;
486       }
487       if ( p->num_children == 0 ) {
488         __rm(__p256, p);
489         return nullptr;
490       }
491       if ( p->num_children <= 37 ) return __shrink(n);
492       return n;
493     }
494     default:
495       return n;
496     }
497   }

500   __release(__node_base *n)
501   {
502     if ( !n ) return;
503     if ( n->kind == __node_kind::leaf ) {
504       __leaf *cur = static_cast<__leaf *>(n);
505       while ( cur ) {
506         __leaf *nx = cur->next;
507         __rm(__pleaf, cur);
508         cur = nx;
509       }
510       return;
511     }
512     switch ( n->kind ) {
513     case __node_kind::n4: {
514       auto *p = static_cast<__n4 *>(n);
515       for ( u8 i = 0; i < p->num_children; ++i ) __release(p->children[i]);
516       __rm(__p4, p);
517       break;
518     }
519     case __node_kind::n16: {
520       auto *p = static_cast<__n16 *>(n);
521       for ( u8 i = 0; i < p->num_children; ++i ) __release(p->children[i]);
522       __rm(__p16, p);
523       break;
524     }
525     case __node_kind::n48: {
526       auto *p = static_cast<__n48 *>(n);
527       for ( u8 i = 0; i < 48; ++i )
528         if ( p->children[i] ) __release(p->children[i]);
529       __rm(__p48, p);
530       break;
531     }
532     case __node_kind::n256: {
533       auto *p = static_cast<__n256 *>(n);
534       for ( int i = 0; i < 256; ++i ) __release(p->children[i]);
535       __rm(__p256, p);
536       break;
537     }
538     default:
539       break;

543   __node_base *
544   __insert(__node_base *n, hash64_t h, const K &k, V v, usize depth, bool &inserted)
545   {
546     if ( !n ) {
547       inserted = true;
548       return __mk(__pleaf, h, k, micron::move(v));
549     }
550     if ( n->kind == __node_kind::leaf ) {
551       auto *lf = static_cast<__leaf *>(n);
552       if ( lf->hash == h ) {
553 
554         __leaf *cur = lf;
555         while ( true ) {
556           if ( cur->key == k ) {
557             cur->value = micron::move(v);
558             inserted = false;
559             return n;
560           }
561           if ( !cur->next ) {
562             cur->next = __mk(__pleaf, h, k, micron::move(v));
563             inserted = true;
564             return n;
565           }
566           cur = cur->next;
567         }
568       }
569 
570       if ( depth >= __hash_bytes ) [[unlikely]] {
571         __leaf *cur = lf;
572         while ( cur->next ) cur = cur->next;
573         cur->next = __mk(__pleaf, h, k, micron::move(v));
574         inserted = true;
575         return n;
576       }
577 
578       u8 b_old = __byte_at(lf->hash, depth);
579       u8 b_new = __byte_at(h, depth);
580       auto *inner = __mk(__p4);
581       __node_base *new_root = inner;
582       if ( b_old == b_new ) {
583 
584         __node_base *deeper = __split_descend(lf, h, k, micron::move(v), depth + 1);
585         new_root = __add_child(new_root, b_old, deeper);
586         inserted = true;
587         return new_root;
588       }
589       __node_base *new_leaf = __mk(__pleaf, h, k, micron::move(v));
590       new_root = __add_child(new_root, b_old, lf);
591       new_root = __add_child(new_root, b_new, new_leaf);
592       inserted = true;
593       return new_root;
594     }
595 
596     u8 b = __byte_at(h, depth);
597     __node_base **slot = __find_child_slot(n, b);
598     if ( slot ) {
599       *slot = __insert(*slot, h, k, micron::move(v), depth + 1, inserted);
600       return n;
601     }
602 
603     auto *new_leaf = __mk(__pleaf, h, k, micron::move(v));
604     inserted = true;
605     return __add_child(n, b, new_leaf);
606   }

608   __node_base *
609   __split_descend(__leaf *lf, hash64_t h, const K &k, V v, usize depth)
610   {
611     if ( depth >= __hash_bytes ) [[unlikely]] {
612       __leaf *cur = lf;
613       while ( cur->next ) cur = cur->next;
614       cur->next = __mk(__pleaf, h, k, micron::move(v));
615       return lf;
616     }
617     u8 b_old = __byte_at(lf->hash, depth);
618     u8 b_new = __byte_at(h, depth);
619     if ( b_old != b_new ) {
620       __node_base *new_root = __mk(__p4);
621       __node_base *new_leaf = __mk(__pleaf, h, k, micron::move(v));
622       new_root = __add_child(new_root, b_old, lf);
623       new_root = __add_child(new_root, b_new, new_leaf);
624       return new_root;
625     }
626     auto *inner = __mk(__p4);
627     __node_base *deeper = __split_descend(lf, h, k, micron::move(v), depth + 1);
628     return __add_child(inner, b_old, deeper);
629   }

631   bool
632   __erase(__node_base *&n, hash64_t h, const K &k, usize depth)
633   {
634     if ( !n ) return false;
635     if ( n->kind == __node_kind::leaf ) {
636       __leaf *head = static_cast<__leaf *>(n);
637 
638       __leaf *prev = nullptr;
639       for ( __leaf *cur = head; cur; cur = cur->next ) {
640         if ( cur->hash == h && cur->key == k ) {
641           if ( !prev ) {
642 
643             if ( cur->next ) {
644               n = cur->next;
645               cur->next = nullptr;
646               __rm(__pleaf, cur);
647               return true;
648             }
649             __rm(__pleaf, cur);
650             n = nullptr;
651             return true;
652           }
653           prev->next = cur->next;
654           cur->next = nullptr;
655           __rm(__pleaf, cur);
656           return true;
657         }
658         prev = cur;
659       }
660       return false;
661     }
662     u8 b = __byte_at(h, depth);
663     __node_base **slot = __find_child_slot(n, b);
664     if ( !slot ) return false;
665     __node_base *child = *slot;
666     bool e = __erase(child, h, k, depth + 1);
667     if ( !e ) return false;
668     if ( child == nullptr ) {
669       n = __remove_child(n, b);
670     } else {
671       *slot = child;
672     }
673     return true;
674   }

676   __leaf *
677   __find_leaf(__node_base *n, hash64_t h, const K &k, usize depth)
678   {
679     while ( n ) {
680       if ( n->kind == __node_kind::leaf ) {
681         for ( __leaf *lf = static_cast<__leaf *>(n); lf; lf = lf->next ) {
682           if ( lf->hash == h && lf->key == k ) return lf;
683         }
684         return nullptr;
685       }
686       u8 b = __byte_at(h, depth);
687       n = __find_child(n, b);
688       ++depth;
689     }
690     return nullptr;
691   }

693   template<typename Fn>
694   void
695   __walk(__node_base *n, Fn &&fn)
696   {
697     if ( !n ) return;
698     if ( n->kind == __node_kind::leaf ) {
699       for ( auto *lf = static_cast<__leaf *>(n); lf; lf = lf->next ) fn(lf->key, lf->value);
700       return;
701     }
702     switch ( n->kind ) {
703     case __node_kind::n4: {
704       auto *p = static_cast<__n4 *>(n);
705       for ( u8 i = 0; i < p->num_children; ++i ) __walk(p->children[i], micron::forward<Fn>(fn));
706       break;
707     }
708     case __node_kind::n16: {
709       auto *p = static_cast<__n16 *>(n);
710       for ( u8 i = 0; i < p->num_children; ++i ) __walk(p->children[i], micron::forward<Fn>(fn));
711       break;
712     }
713     case __node_kind::n48: {
714       auto *p = static_cast<__n48 *>(n);
715       for ( u8 i = 0; i < 48; ++i )
716         if ( p->children[i] ) __walk(p->children[i], micron::forward<Fn>(fn));
717       break;
718     }
719     case __node_kind::n256: {
720       auto *p = static_cast<__n256 *>(n);
721       for ( int i = 0; i < 256; ++i )
722         if ( p->children[i] ) __walk(p->children[i], micron::forward<Fn>(fn));
723       break;
724     }
725     default:
726       break;
727     }
728   }

730   static usize
731   __bytes_rec(__node_base *n) noexcept
732   {
733     if ( !n ) return 0;
734     if ( n->kind == __node_kind::leaf ) {
735       usize t = 0;
736       for ( __leaf *c = static_cast<__leaf *>(n); c; c = c->next ) t += sizeof(__leaf);
737       return t;
738     }
739     switch ( n->kind ) {
740     case __node_kind::n4: {
741       auto *p = static_cast<__n4 *>(n);
742       usize t = sizeof(__n4);
743       for ( u8 i = 0; i < p->num_children; ++i ) t += __bytes_rec(p->children[i]);
744       return t;
745     }
746     case __node_kind::n16: {
747       auto *p = static_cast<__n16 *>(n);
748       usize t = sizeof(__n16);
749       for ( u8 i = 0; i < p->num_children; ++i ) t += __bytes_rec(p->children[i]);
750       return t;
751     }
752     case __node_kind::n48: {
753       auto *p = static_cast<__n48 *>(n);
754       usize t = sizeof(__n48);
755       for ( u16 i = 0; i < 48; ++i )
756         if ( p->children[i] ) t += __bytes_rec(p->children[i]);
757       return t;
758     }
759     case __node_kind::n256: {
760       auto *p = static_cast<__n256 *>(n);
761       usize t = sizeof(__n256);
762       for ( u16 i = 0; i < 256; ++i ) t += __bytes_rec(p->children[i]);
763       return t;
764     }
765     default:
766       return 0;
767     }
768   }

963   // explicit-stack cursor
964   class const_iterator
965   {
966     struct __frame {
967       const __node_base *n;
968       u16 i;      // next child slot to try
969     };
970 
971     __frame __st[__hash_bytes + 2]{};
972     i32 __top = -1;
973     const __leaf *__lf = nullptr;
974 
975     static const __node_base *
976     __child_at(const __node_base *n, u16 i) noexcept
977     {
978       switch ( n->kind ) {
979       case __node_kind::n4: {
980         auto *p = static_cast<const __n4 *>(n);
981         return i < p->num_children ? p->children[i] : nullptr;
982       }
983       case __node_kind::n16: {
984         auto *p = static_cast<const __n16 *>(n);
985         return i < p->num_children ? p->children[i] : nullptr;
986       }
987       case __node_kind::n48: {
988         auto *p = static_cast<const __n48 *>(n);
989         return i < 48u ? p->children[i] : nullptr;
990       }
991       case __node_kind::n256: {
992         auto *p = static_cast<const __n256 *>(n);
993         return i < 256u ? p->children[i] : nullptr;
994       }
995       default:
996         return nullptr;
997       }
998     }
999 
1000     static u16
1001     __slot_count(const __node_base *n) noexcept
1002     {
1003       switch ( n->kind ) {
1004       case __node_kind::n4:
1005         return static_cast<const __n4 *>(n)->num_children;
1006       case __node_kind::n16:
1007         return static_cast<const __n16 *>(n)->num_children;
1008       case __node_kind::n48:
1009         return 48u;
1010       case __node_kind::n256:
1011         return 256u;
1012       default:
1013         return 0u;
1014       }
1015     }
1016 
1017     // descend into n, pushing frames, until a leaf is reached
1018     void
1019     __descend(const __node_base *n)
1020     {
1021       while ( n ) {
1022         if ( n->kind == __node_kind::leaf ) {
1023           __lf = static_cast<const __leaf *>(n);
1024           return;
1025         }
1026         __st[++__top] = __frame{ n, 0 };
1027         n = __step_top();
1028       }
1029       __unwind();
1030     }
1031 
1032     const __node_base *
1033     __step_top() noexcept
1034     {
1035       __frame &f = __st[__top];
1036       const u16 cnt = __slot_count(f.n);
1037       while ( f.i < cnt ) {
1038         const __node_base *c = __child_at(f.n, f.i++);
1039         if ( c ) return c;
1040       }
1041       return nullptr;
1042     }
1043 
1044     // pop exhausted frames and resume the first one that still has children
1045     void
1046     __unwind()
1047     {
1048       while ( __top >= 0 ) {
1049         const __node_base *c = __step_top();
1050         if ( c ) {
1051           __descend(c);
1052           return;
1053         }
1054         --__top;
1055       }
1056       __lf = nullptr;
1057     }

1060     using value_type = micron::pair<const K &, const V &>;
1061     using reference = micron::pair<const K &, const V &>;
1062     using difference_type = ssize_t;

1064     constexpr const_iterator() = default;

1066     explicit const_iterator(const __node_base *root)
1067     {
1068       if ( root ) __descend(root);
1069     }

1072     operator*() const
1073     {

1078     operator++()
1079     {
1080       if ( __lf && __lf->next ) {
1081         __lf = __lf->next;
1082         return *this;
1083       }
1084       __lf = nullptr;
1085       __unwind();
1086       return *this;
1087     }
