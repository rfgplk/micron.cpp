45   static constexpr usize __kv = sizeof(K) + sizeof(V);
46   static constexpr usize __auto_t = (__kv <= 16) ? 16u : ((__kv <= 64) ? 8u : 4u);
47   static constexpr usize T = (DegreeOverride > 0) ? DegreeOverride : __auto_t;
48   static_assert(T >= 2, "b_tree minimum degree must be >= 2");

50   static constexpr u16 MAXK = static_cast<u16>(2 * T - 1);      // max keys / leaf entries
51   static constexpr u16 MAXC = static_cast<u16>(2 * T);          // max children
52   static constexpr u16 MINK = static_cast<u16>(T - 1);          // min keys (non-root)

54   struct internal_node {
55     u8 leaf;      // 0

61     internal_node() noexcept : leaf(0), _pad(0), nkeys(0) { }

64     {

83     u8 leaf;      // 1
84     u8 _pad;

190   void
191   free_subtree(node_idx i) noexcept
192   {
193     if ( i == nil ) return;
194     if ( !is_leaf(i) ) {
195       internal_node &I = inode(i);
196       u16 nk = I.nkeys;
197       node_idx kids[MAXC];
198       for ( u16 c = 0; c <= nk; ++c ) kids[c] = I.kids[c];
199       for ( u16 c = 0; c <= nk; ++c ) free_subtree(kids[c]);
200     }
201     destroy_node(i);
202   }

222   void
223   split_child(node_idx parent, u16 ci)
224   {
225     node_idx cidx = inode(parent).kids[ci];
226     const bool child_leaf = is_leaf(cidx);
227     node_idx ridx = child_leaf ? alloc_leaf() : alloc_internal();      // may move the slab
228 
229     if ( child_leaf ) {
230       leaf_node &C = lnode(cidx);
231       leaf_node &R = lnode(ridx);
232       const u16 right_n = static_cast<u16>(MAXK - T);
233       __tree_store::relocate_n(R.keys(), C.keys() + T, right_n);
234       __tree_store::relocate_n(R.vals(), C.vals() + T, right_n);
235       C.count = static_cast<u16>(T);
236       R.count = right_n;
237       // splice R after C in the leaf list
238       R.next = C.next;
239       R.prev = cidx;
240       if ( C.next != nil ) lnode(C.next).prev = ridx;
241       C.next = ridx;
242       internal_node &P = inode(parent);
243       __tree_store::shift_right(P.keys(), ci, P.nkeys);
244       __tree_store::shift_right(P.kids, static_cast<u16>(ci + 1), static_cast<u16>(P.nkeys + 1));
245       new (micron::addr(P.keys()[ci])) K(lnode(ridx).keys()[0]);
246       P.kids[ci + 1] = ridx;
247       ++P.nkeys;
248     } else {
249       internal_node &C = inode(cidx);
250       internal_node &R = inode(ridx);
251       const u16 right_k = static_cast<u16>(MAXK - T);      // = T-1
252       K median(micron::move(C.keys()[T - 1]));
253       C.keys()[T - 1].~K();
254       __tree_store::relocate_n(R.keys(), C.keys() + T, right_k);
255       for ( u16 i = 0; i <= right_k; ++i ) R.kids[i] = C.kids[T + i];
256       R.nkeys = right_k;
257       C.nkeys = static_cast<u16>(T - 1);
258       internal_node &P = inode(parent);
259       __tree_store::shift_right(P.keys(), ci, P.nkeys);
260       __tree_store::shift_right(P.kids, static_cast<u16>(ci + 1), static_cast<u16>(P.nkeys + 1));
261       new (micron::addr(P.keys()[ci])) K(micron::move(median));
262       P.kids[ci + 1] = ridx;
263       ++P.nkeys;
264     }
265   }

267   template<class VV>
268   V *
269   put(const K &key, VV &&val, bool overwrite, bool &inserted)
270   {
271     inserted = false;
272     if ( __root == nil ) __root = alloc_leaf();
273     if ( entries(__root) == MAXK ) {
274       node_idx nr = alloc_internal();
275       inode(nr).kids[0] = __root;
276       __root = nr;
277       split_child(__root, 0);
278     }
279     node_idx cur = __root;
280     while ( !is_leaf(cur) ) {
281       u16 ci = child_index(inode(cur), key);
282       if ( entries(inode(cur).kids[ci]) == MAXK ) {
283         split_child(cur, ci);
284         ci = child_index(inode(cur), key);
285       }
286       cur = inode(cur).kids[ci];
287     }
288     leaf_node &L = lnode(cur);
289     u16 pos = leaf_lb(L, key);
290     if ( pos < L.count && eq_at_lb(L.keys()[pos], key) ) {
291       if ( overwrite ) {
292         L.vals()[pos].~V();
293         new (micron::addr(L.vals()[pos])) V(micron::forward<VV>(val));
294       }
295       return micron::addr(L.vals()[pos]);
296     }
297     __tree_store::shift_right(L.keys(), pos, L.count);
298     __tree_store::shift_right(L.vals(), pos, L.count);
299     new (micron::addr(L.keys()[pos])) K(key);
300     new (micron::addr(L.vals()[pos])) V(micron::forward<VV>(val));
301     ++L.count;
302     ++__size;
303     inserted = true;
304     return micron::addr(L.vals()[pos]);
305   }

307   bool
308   erase_descend(node_idx cur, const K &key)
309   {
310     while ( !is_leaf(cur) ) {
311       u16 ci = child_index(inode(cur), key);
312       if ( entries(inode(cur).kids[ci]) == MINK ) {
313         fill_child(cur, ci);
314         ci = child_index(inode(cur), key);
315       }
316       cur = inode(cur).kids[ci];
317     }
318     leaf_node &L = lnode(cur);
319     u16 pos = leaf_lb(L, key);
320     if ( pos < L.count && eq_at_lb(L.keys()[pos], key) ) {
321       __tree_store::erase_at(L.keys(), pos, L.count);
322       __tree_store::erase_at(L.vals(), pos, L.count);
323       --L.count;
324       return true;
325     }
326     return false;
327   }

329   void
330   fill_child(node_idx parent, u16 ci)
331   {
332     internal_node &P = inode(parent);
333     if ( ci > 0 && entries(P.kids[ci - 1]) > MINK )
334       borrow_prev(parent, ci);
335     else if ( ci < P.nkeys && entries(P.kids[ci + 1]) > MINK )
336       borrow_next(parent, ci);
337     else if ( ci < P.nkeys )
338       merge(parent, ci);
339     else
340       merge(parent, static_cast<u16>(ci - 1));
341   }

343   void
344   borrow_prev(node_idx parent, u16 ci)
345   {
346     internal_node &P = inode(parent);
347     node_idx lft = P.kids[ci - 1];
348     node_idx cur = P.kids[ci];
349     if ( is_leaf(cur) ) {
350       leaf_node &Lf = lnode(lft);
351       leaf_node &C = lnode(cur);
352       __tree_store::shift_right(C.keys(), 0, C.count);
353       __tree_store::shift_right(C.vals(), 0, C.count);
354       new (micron::addr(C.keys()[0])) K(micron::move(Lf.keys()[Lf.count - 1]));
355       new (micron::addr(C.vals()[0])) V(micron::move(Lf.vals()[Lf.count - 1]));
356       Lf.keys()[Lf.count - 1].~K();
357       Lf.vals()[Lf.count - 1].~V();
358       --Lf.count;
359       ++C.count;
360       P.keys()[ci - 1].~K();
361       new (micron::addr(P.keys()[ci - 1])) K(C.keys()[0]);
362     } else {
363       internal_node &Lf = inode(lft);
364       internal_node &C = inode(cur);
365       __tree_store::shift_right(C.keys(), 0, C.nkeys);
366       __tree_store::shift_right(C.kids, 0, static_cast<u16>(C.nkeys + 1));
367       new (micron::addr(C.keys()[0])) K(micron::move(P.keys()[ci - 1]));      // separator down
368       C.kids[0] = Lf.kids[Lf.nkeys];
369       ++C.nkeys;
370       P.keys()[ci - 1].~K();
371       new (micron::addr(P.keys()[ci - 1])) K(micron::move(Lf.keys()[Lf.nkeys - 1]));      // left's last key up
372       Lf.keys()[Lf.nkeys - 1].~K();
373       --Lf.nkeys;
374     }
375   }

377   void
378   borrow_next(node_idx parent, u16 ci)
379   {
380     internal_node &P = inode(parent);
381     node_idx cur = P.kids[ci];
382     node_idx rgt = P.kids[ci + 1];
383     if ( is_leaf(cur) ) {
384       leaf_node &C = lnode(cur);
385       leaf_node &Rf = lnode(rgt);
386       new (micron::addr(C.keys()[C.count])) K(micron::move(Rf.keys()[0]));
387       new (micron::addr(C.vals()[C.count])) V(micron::move(Rf.vals()[0]));
388       ++C.count;
389       __tree_store::erase_at(Rf.keys(), 0, Rf.count);
390       __tree_store::erase_at(Rf.vals(), 0, Rf.count);
391       --Rf.count;
392       P.keys()[ci].~K();
393       new (micron::addr(P.keys()[ci])) K(Rf.keys()[0]);
394     } else {
395       internal_node &C = inode(cur);
396       internal_node &Rf = inode(rgt);
397       new (micron::addr(C.keys()[C.nkeys])) K(micron::move(P.keys()[ci]));      // separator down
398       C.kids[C.nkeys + 1] = Rf.kids[0];
399       ++C.nkeys;
400       P.keys()[ci].~K();
401       new (micron::addr(P.keys()[ci])) K(micron::move(Rf.keys()[0]));      // right's first key up
402       __tree_store::erase_at(Rf.keys(), 0, Rf.nkeys);
403       for ( u16 i = 0; i < Rf.nkeys; ++i ) Rf.kids[i] = Rf.kids[i + 1];
404       --Rf.nkeys;
405     }
406   }

408   void
409   merge(node_idx parent, u16 m)
410   {
411     internal_node &P = inode(parent);
412     node_idx lft = P.kids[m];
413     node_idx rgt = P.kids[m + 1];
414     if ( is_leaf(lft) ) {
415       leaf_node &L = lnode(lft);
416       leaf_node &R = lnode(rgt);
417       __tree_store::relocate_n(L.keys() + L.count, R.keys(), R.count);
418       __tree_store::relocate_n(L.vals() + L.count, R.vals(), R.count);
419       L.count = static_cast<u16>(L.count + R.count);
420       R.count = 0;
421       L.next = R.next;
422       if ( R.next != nil ) lnode(R.next).prev = lft;
423     } else {
424       internal_node &L = inode(lft);
425       internal_node &R = inode(rgt);
426       const u16 ln = L.nkeys;
427       new (micron::addr(L.keys()[ln])) K(micron::move(P.keys()[m]));      // separator down
428       __tree_store::relocate_n(L.keys() + ln + 1, R.keys(), R.nkeys);
429       for ( u16 i = 0; i <= R.nkeys; ++i ) L.kids[ln + 1 + i] = R.kids[i];
430       L.nkeys = static_cast<u16>(ln + 1 + R.nkeys);
431       R.nkeys = 0;
432     }
433     destroy_node(rgt);
434     // drop separator m and child pointer (m+1) from the parent
435     __tree_store::erase_at(P.keys(), m, P.nkeys);
436     for ( u16 i = static_cast<u16>(m + 1); i < P.nkeys; ++i ) P.kids[i] = P.kids[i + 1];
437     --P.nkeys;
438   }

514   // pre-size the slab. Worth doing for more than the one saved allocation: growth REALLOCATES and
515   // byte-copies the whole arena, and it invalidates every V* handed out by find()/at()/operator[]
516   void
517   reserve(usize n_entries)
518   {
519     // one leaf per MAXK entries at perfect fill, plus the internal spine; 2x covers realistic fill

542   V *
543   find(const K &key) noexcept
544   {
545     node_idx cur = __root;
546     while ( cur != nil ) {
547       if ( is_leaf(cur) ) {
548         leaf_node &L = lnode(cur);
549         u16 pos = leaf_lb(L, key);
550         if ( pos < L.count && eq_at_lb(L.keys()[pos], key) ) return micron::addr(L.vals()[pos]);
551         return nullptr;
552       }
553       cur = inode(cur).kids[child_index(inode(cur), key)];
554     }
555     return nullptr;
556   }

651   bool
652   erase(const K &key)
653   {
654     if ( __root == nil ) return false;
655     bool removed = erase_descend(__root, key);
656     if ( removed ) --__size;
657     if ( __root != nil ) {
658       if ( !is_leaf(__root) && inode(__root).nkeys == 0 ) {
659         node_idx old = __root;
660         __root = inode(__root).kids[0];
661         destroy_node(old);
662       } else if ( is_leaf(__root) && lnode(__root).count == 0 ) {
663         destroy_node(__root);
664         __root = nil;
665       }
666     }
667     return removed;
668   }

684   class iterator
685   {
686     b_tree *t;
687     node_idx leaf;
688     u16 pos;
689 
690   public:
691     iterator(b_tree *tp, node_idx l, u16 p) noexcept : t(tp), leaf(l), pos(p) { }
692 
693     kv_ref
694     operator*() const noexcept
695     {
696       leaf_node &L = t->lnode(leaf);
697       return kv_ref{ L.keys()[pos], L.vals()[pos] };
698     }
699 
700     iterator &
701     operator++() noexcept
702     {
703       leaf_node &L = t->lnode(leaf);
704       ++pos;
705       if ( pos >= L.count ) {
706         leaf = L.next;
707         pos = 0;
708       }
709       return *this;
710     }
711 
712     bool
713     operator==(const iterator &o) const noexcept
714     {
715       return leaf == o.leaf && pos == o.pos;
716     }
717 
718     bool
719     operator!=(const iterator &o) const noexcept
720     {
721       return !(*this == o);
722     }
723   };

726   begin() noexcept
727   {
728     return iterator(this, leftmost_leaf(__root), 0);

737   iterator
738   lower_bound(const K &key) noexcept
739   {
740     node_idx cur = __root;
741     while ( cur != nil && !is_leaf(cur) ) cur = inode(cur).kids[child_index(inode(cur), key)];
742     if ( cur == nil ) return end();
743     leaf_node &L = lnode(cur);
744     u16 pos = leaf_lb(L, key);
745     if ( pos >= L.count ) {
746       node_idx nx = L.next;
747       return nx == nil ? end() : iterator(this, nx, 0);
748     }
749     return iterator(this, cur, pos);
750   }

752   iterator
753   upper_bound(const K &key) noexcept
754   {
755     iterator it = lower_bound(key);
756     if ( it != end() ) {
757       kv_ref kv = *it;
758       if ( eq(kv.key, key) ) ++it;
759     }
760     return it;
761   }

763   template<typename Fn>
764   void
765   for_each(Fn &&fn)
766   {
767     node_idx l = leftmost_leaf(__root);
768     while ( l != nil ) {
769       leaf_node &L = lnode(l);
770       for ( u16 i = 0; i < L.count; ++i ) fn(static_cast<const K &>(L.keys()[i]), L.vals()[i]);
771       l = L.next;
772     }
773   }

782   template<class Fn>
783   auto
784   map(Fn fn) const
785   {
786     if constexpr ( micron::is_invocable_v<Fn, const K &, const V &> ) {
787       using V2 = micron::remove_cvref_t<micron::invoke_result_t<Fn, const K &, const V &>>;
788       b_tree<K, V2, Compare, DegreeOverride> out;
789       for_each([&](const K &k, const V &v) { out.insert(k, fn(k, v)); });
790       return out;
791     } else {
792       using V2 = micron::remove_cvref_t<micron::invoke_result_t<Fn, const V &>>;
793       b_tree<K, V2, Compare, DegreeOverride> out;
794       for_each([&](const K &k, const V &v) { out.insert(k, fn(v)); });
795       return out;
796     }
797   }

799   template<class LeafFn, class InnerFn>
800   auto
801   cata(LeafFn leaf, InnerFn inner) const
802   {
803     using Acc = micron::remove_cvref_t<micron::invoke_result_t<LeafFn, const K *, const V *, u16>>;
804     if ( __root == nil ) return Acc{};
805     return const_cast<b_tree *>(this)->template __cata_rec<Acc>(__root, leaf, inner);
806   }
807 
808   template<class Acc, class LeafFn, class InnerFn>
809   Acc
810   __cata_rec(node_idx n, LeafFn &leaf, InnerFn &inner)
811   {
812     if ( is_leaf(n) ) {
813       leaf_node &L = lnode(n);
814       return leaf(L.keys(), L.vals(), L.count);
815     }
816     internal_node &I = inode(n);
817     const u16 nk = I.nkeys;
818     const u16 nkids = static_cast<u16>(nk + 1);
819     Acc kids[MAXC];
820     for ( u16 i = 0; i < nkids; ++i ) kids[i] = __cata_rec<Acc>(I.kids[i], leaf, inner);
821     return inner(I.keys(), nk, kids, nkids);
822   }

824   template<class Fn>
825   walk_ctl
826   traverse(Fn fn) const
827   {
828     b_tree *self = const_cast<b_tree *>(this);
829     node_idx l = self->leftmost_leaf(self->__root);
830     while ( l != nil ) {
831       leaf_node &L = self->lnode(l);
832       for ( u16 i = 0; i < L.count; ++i )
833         if ( micron::__impl::invoke_walk(fn, static_cast<const K &>(L.keys()[i]), static_cast<const V &>(L.vals()[i])) == walk_ctl::stop )
834           return walk_ctl::stop;
835       l = L.next;
836     }
837     return walk_ctl::continue_;
838   }
