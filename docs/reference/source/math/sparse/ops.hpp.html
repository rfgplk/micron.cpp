70 template<ieee754_floating T, micron::integral I>
71 [[nodiscard]] inline T
72 dot(const sparse_vec<T, I> &a, const sparse_vec<T, I> &b) noexcept
73 {
74   T s = T(0);
75   usize ia = 0, ib = 0;
76   const usize na = a.nnz();
77   const usize nb = b.nnz();
78   while ( ia < na && ib < nb ) {
79     const I ka = a.idx.data()[ia];
80     const I kb = b.idx.data()[ib];
81     if ( ka == kb ) {
82       s = math::fma<T>(a.values.data()[ia], b.values.data()[ib], s);
83       ++ia;
84       ++ib;
85     } else if ( ka < kb ) {
86       ++ia;
87     } else {
88       ++ib;
89     }
90   }
91   return s;
92 }

94 template<ieee754_floating T, micron::integral I>
95 [[nodiscard]] inline T
96 dot(const sparse_vec<T, I> &a, const dynvec<T> &b) noexcept
97 {
98   T s = T(0);
99   const usize na = a.nnz();
100   for ( usize k = 0; k < na; ++k ) s = math::fma<T>(a.values.data()[k], b[a.idx.data()[k]], s);
101   return s;
102 }
