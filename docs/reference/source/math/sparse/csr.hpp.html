9 // compressed sparse row matrix
10 //
11 //   outer  = length rows+1; outer[i] = start in inner/values for row i
12 //   inner  = length nnz; inner[k] = column index of values[k]
13 //   values = length nnz

28 template<arith_scalar T, micron::integral I = u32> struct csr {
29   using value_type = T;
30   using index_type = I;
31   using vec_i = micron::vector<I, micron::allocator_serial<>, false>;
32   using vec_v = micron::vector<T, micron::allocator_serial<>, false>;
33 
34   usize rows{ 0 };
35   usize cols{ 0 };
36   vec_i outer;       // size rows+1
37   vec_i inner;       // size nnz
38   vec_v values;      // size nnz
39 
40   csr() noexcept = default;
41 
42   csr(usize r, usize c) : rows(r), cols(c), outer(r + 1, I(0)) { }
43 
44   csr(const csr &) = default;
45   csr(csr &&) noexcept = default;
46   csr &operator=(const csr &) = default;
47   csr &operator=(csr &&) noexcept = default;
48 
49   [[nodiscard, gnu::always_inline]] usize
50   nnz() const noexcept
51   {
52     return values.size();
53   }
54 
55   [[nodiscard, gnu::always_inline]] I
56   row_begin(usize i) const noexcept
57   {
58     return outer.data()[i];
59   }
60 
61   [[nodiscard, gnu::always_inline]] I
62   row_end(usize i) const noexcept
63   {
64     return outer.data()[i + 1];
65   }
66 
67   [[nodiscard, gnu::always_inline]] usize
68   row_nnz(usize i) const noexcept
69   {
70     const I e = outer.data()[i + 1];
71     const I s = outer.data()[i];
72     // guard non-monotone outer
73     return (e >= s) ? static_cast<usize>(e - s) : usize(0);
74   }
75 };

77 template<arith_scalar T, micron::integral I>
78 [[nodiscard]] inline csc<T, I>
79 to_csc(const csr<T, I> &A) noexcept
80 {
81   csc<T, I> out(A.rows, A.cols);
82   if ( A.nnz() == 0 ) return out;
83   micron::vector<usize, micron::allocator_serial<>, false> counts(A.cols, usize(0));
84   for ( usize k = 0; k < A.nnz(); ++k ) {
85     const usize j = static_cast<usize>(A.inner.data()[k]);
86     if ( j < A.cols ) counts.data()[j] += 1;
87   }
88   out.outer.data()[0] = I(0);
89   for ( usize j = 0; j < A.cols; ++j ) out.outer.data()[j + 1] = out.outer.data()[j] + static_cast<I>(counts.data()[j]);
90   const usize total = A.nnz();
91   out.inner = typename csc<T, I>::vec_i(total, I(0));
92   out.values = typename csc<T, I>::vec_v(total, T(0));
93   micron::vector<usize, micron::allocator_serial<>, false> pos(A.cols, usize(0));
94   for ( usize j = 0; j < A.cols; ++j ) pos.data()[j] = static_cast<usize>(out.outer.data()[j]);
95   for ( usize i = 0; i < A.rows; ++i ) {
96     const usize a = static_cast<usize>(A.outer.data()[i]);
97     const usize b = static_cast<usize>(A.outer.data()[i + 1]);
98     for ( usize k = a; k < b; ++k ) {
99       const usize j = static_cast<usize>(A.inner.data()[k]);
100       if ( j >= A.cols ) continue;
101       const usize p = pos.data()[j];
102       out.inner.data()[p] = static_cast<I>(i);
103       out.values.data()[p] = A.values.data()[k];
104       pos.data()[j] = p + 1;
105     }
106   }
107   return out;
108 }

110 template<arith_scalar T, micron::integral I>
111 [[nodiscard]] inline csr<T, I>
112 to_csr(const csc<T, I> &A) noexcept
113 {
114   csr<T, I> out(A.rows, A.cols);
115   if ( A.nnz() == 0 ) return out;
116   micron::vector<usize, micron::allocator_serial<>, false> counts(A.rows, usize(0));
117   for ( usize k = 0; k < A.nnz(); ++k ) {
118     const usize i = static_cast<usize>(A.inner.data()[k]);
119     if ( i < A.rows ) counts.data()[i] += 1;
120   }
121   out.outer.data()[0] = I(0);
122   for ( usize i = 0; i < A.rows; ++i ) out.outer.data()[i + 1] = out.outer.data()[i] + static_cast<I>(counts.data()[i]);
123   const usize total = A.nnz();
124   out.inner = typename csr<T, I>::vec_i(total, I(0));
125   out.values = typename csr<T, I>::vec_v(total, T(0));
126   micron::vector<usize, micron::allocator_serial<>, false> pos(A.rows, usize(0));
127   for ( usize i = 0; i < A.rows; ++i ) pos.data()[i] = static_cast<usize>(out.outer.data()[i]);
128   for ( usize j = 0; j < A.cols; ++j ) {
129     const usize a = static_cast<usize>(A.outer.data()[j]);
130     const usize b = static_cast<usize>(A.outer.data()[j + 1]);
131     for ( usize k = a; k < b; ++k ) {
132       const usize i = static_cast<usize>(A.inner.data()[k]);
133       if ( i >= A.rows ) continue;
134       const usize p = pos.data()[i];
135       out.inner.data()[p] = static_cast<I>(j);
136       out.values.data()[p] = A.values.data()[k];
137       pos.data()[i] = p + 1;
138     }
139   }
140   return out;
141 }
