27 struct __convolution_geometry {
28   buffer<usize> shape, output, crop;
29   usize input_size{ 1 }, kernel_size{ 1 }, output_size{ 1 }, volume{ 1 };
30   bool empty{};
31 
32   explicit __convolution_geometry(const convolution_descriptor &d)
33   {
34     usize rank = d.input_shape.size();
35     if ( !rank || rank != d.kernel_shape.size() || unsigned(d.mode) > unsigned(convolution_mode::circular) )
36       exc<except::invalid_argument>("fft: invalid convolution shape");
37     if ( d.mode == convolution_mode::circular && d.period.size() != rank )
38       exc<except::invalid_argument>("fft: circular convolution requires a period per axis");
39     shape.resize(rank);
40     output.resize(rank);
41     crop.resize(rank);
42     for ( usize a = 0; a < rank; ++a )
43       if ( !d.input_shape[a] || !d.kernel_shape[a] ) {
44         empty = true;
45         break;
46       }
47     if ( empty ) {
48       for ( usize a = 0; a < rank; ++a ) {
49         usize n = d.input_shape[a], m = d.kernel_shape[a];
50         if ( d.mode == convolution_mode::circular ) {
51           usize p = d.period[a];
52           if ( !p || n > p || m > p ) exc<except::invalid_argument>("fft: inputs exceed circular period");
53         } else if ( d.mode == convolution_mode::valid && n && m && n < m )
54           exc<except::invalid_argument>("fft: valid convolution requires input >= kernel");
55         shape[a] = 1;
56         output[a] = 0;
57       }
58       input_size = kernel_size = output_size = 0;
59       return;
60     }
61     for ( usize a = 0; a < rank; ++a ) {
62       usize n = d.input_shape[a], m = d.kernel_shape[a];
63       input_size = __mul(input_size, n);
64       kernel_size = __mul(kernel_size, m);
65       if ( d.mode == convolution_mode::circular ) {
66         usize p = d.period[a];
67         if ( !p || n > p || m > p ) exc<except::invalid_argument>("fft: inputs exceed circular period");
68         shape[a] = output[a] = p;
69       } else {
70         usize full = n && m ? __add(n, m) - 1 : 0;
71         shape[a] = next_fast_len(full);
72         switch ( d.mode ) {
73         case convolution_mode::full:
74           output[a] = full;
75           break;
76         case convolution_mode::same:
77           output[a] = n;
78           crop[a] = m ? (m - 1) / 2 : 0;
79           break;
80         case convolution_mode::valid:
81           if ( n && m && n < m ) exc<except::invalid_argument>("fft: valid convolution requires input >= kernel");
82           output[a] = n && m ? n - m + 1 : 0;
83           crop[a] = m ? m - 1 : 0;
84           break;
85         default:
86           break;
87         }
88       }
89       volume = __mul(volume, shape[a]);
90       output_size = __mul(output_size, output[a]);
91     }
92   }
93 };
94 
95 inline usize
96 __embed_index(usize index, const buffer<usize> &source, const buffer<usize> &target, bool reverse = false) noexcept
97 {
98   usize result = 0, step = 1;
99   for ( usize a = source.size(); a-- > 0; ) {
100     usize c = index % source[a];
101     index /= source[a];
102     result += (reverse ? source[a] - 1 - c : c) * step;
103     step *= target[a];
104   }
105   return result;
106 }

111 template<fft_scalar F> class convolution_plan
112 {
113   convolution_descriptor __desc;
114   __impl::__convolution_geometry __geometry;
115   plan<F> __fft;
116   buffer<complex<F>> __kernel;
117 
118   template<typename Kernel>
119   void
120   __prepare(const Kernel *kernel)
121   {
122     if ( __geometry.empty ) return;
123     if ( !kernel ) exc<except::invalid_argument>("fft: null convolution kernel");
124     buffer<complex<F>> time(__geometry.volume);
125     bool reverse = __desc.correlation && __desc.mode != convolution_mode::circular;
126     for ( usize k = 0; k < __geometry.kernel_size; ++k ) {
127       complex<F> z{ kernel[k] };
128       if ( reverse ) z = conj(z);
129       time[__impl::__embed_index(k, __desc.kernel_shape, __geometry.shape, reverse)] = z;
130     }
131     workspace<F> work(__fft);
132     __fft.execute(time.data(), __kernel.data(), work.view());
133     if ( __desc.correlation && __desc.mode == convolution_mode::circular )
134       for ( auto &z : __kernel ) z = conj(z);
135   }
136 
137 public:
138   template<typename Input, typename Output, typename Runner>
139   max_t
140   __execute(const Input *input, Output *output, workspace_view<F> work, Runner &runner) const noexcept
141   {
142     if ( __geometry.empty ) return 0;
143     if ( !input || !output ) return -22;
144     usize base = 3 * __geometry.volume, per_worker = __fft.workspace_size() - __geometry.volume;
145     if ( !runner.workers || runner.workers > (__impl::__limit - base) / per_worker || !work.data
146          || work.size < base + runner.workers * per_worker )
147       return -105;
148     usize n = __geometry.volume;
149     auto *time = work.data, *freq = time + n;
150     workspace_view<F> scratch{ freq + n, work.size - 2 * n };
151     for ( usize k = 0; k < n; ++k ) time[k] = {};
152     for ( usize k = 0; k < __geometry.input_size; ++k )
153       time[__impl::__embed_index(k, __desc.input_shape, __geometry.shape)] = complex<F>{ input[k] };
154     __fft.__execute([&](ssize_t i) { return time[i]; }, [&](ssize_t i, complex<F> z) { freq[i] = z; }, scratch, direction::forward, runner);
155     __impl::__multiply(freq, __kernel.data(), n);
156     __fft.__execute([&](ssize_t i) { return freq[i]; }, [&](ssize_t i, complex<F> z) { time[i] = z; }, scratch, direction::backward,
157                     runner);
158     for ( usize k = 0; k < __geometry.output_size; ++k ) {
159       usize index = k, source = 0, step = 1;
160       for ( usize a = __geometry.output.size(); a-- > 0; ) {
161         source += (index % __geometry.output[a] + __geometry.crop[a]) * step;
162         index /= __geometry.output[a];
163         step *= __geometry.shape[a];
164       }
165       if constexpr ( micron::is_same_v<Output, F> )
166         output[k] = time[source].real;
167       else
168         output[k] = time[source];
169     }
170     return max_t(__geometry.output_size * sizeof(Output));
171   }
172 
173 public:
174   template<typename Kernel>
175   convolution_plan(const convolution_descriptor &d, const Kernel *kernel)
176       : __desc(d), __geometry(d), __fft(descriptor<F>(__geometry.shape)), __kernel(__geometry.volume)
177   {
178     __impl::__mul(__geometry.output_size, sizeof(complex<F>));
179     __impl::__mul(workspace_size(), sizeof(complex<F>));
180     __prepare(kernel);
181   }

184   output_shape() const noexcept
185   {
186     return __geometry.output;
187   }

190   output_size() const noexcept

195   usize
196   workspace_size(usize workers = 1) const
197   {
198     return __geometry.empty ? 0 : __impl::__add(2 * __geometry.volume, __fft.workspace_size(workers));
199   }
200 
201   usize
202   workspace_bytes(usize workers = 1) const
203   {
204     return __impl::__mul(workspace_size(workers), sizeof(complex<F>));
205   }

207   max_t
208   execute(const complex<F> *input, complex<F> *output, workspace_view<F> work) const noexcept
209   {
210     __impl::__serial<F> runner;
211     return __execute(input, output, work, runner);
212   }

230 template<typename T>
231 buffer<T>
232 convolve(const T *input, const T *kernel, const convolution_descriptor &d)
233 {
234   using F = typename __convolution_scalar<T>::type;
235   convolution_plan<F> p(d, kernel);
236   workspace<F> work(p);
237   buffer<T> out(p.output_size());
238   if ( p.execute(input, out.data(), work.view()) < 0 ) exc<except::invalid_argument>("fft: invalid convolution input");
239   return out;
240 }
