23 inline usize
24 __complete_layout(layout &l, const buffer<usize> &shape, usize batches)
25 {
26   usize volume = __product(shape);
27   if ( l.strides.empty() ) {
28     l.strides.resize(shape.size());
29     usize stride = 1;
30     for ( usize i = shape.size(); i-- > 0; ) {
31       l.strides[i] = ssize_t(stride);
32       stride = __mul(stride, shape[i]);
33     }
34   }
35   if ( l.strides.size() != shape.size() ) exc<except::invalid_argument>("fft: stride rank mismatch");
36   if ( !l.distance ) l.distance = ssize_t(volume);
37   usize extent = 0;
38   for ( usize i = 0; i <= shape.size(); ++i ) {
39     ssize_t s = i == shape.size() ? l.distance : l.strides[i];
40     usize n = i == shape.size() ? batches : shape[i];
41     if ( s == -ssize_t(__limit) - 1 || (!s && n > 1) ) exc<except::invalid_argument>("fft: invalid stride");
42     usize magnitude = usize(s < 0 ? -s : s);
43     extent = __add(extent, __mul(magnitude, n ? n - 1 : 0));
44   }
45   if ( l.offset == -ssize_t(__limit) - 1 ) exc<except::length_error>("fft: offset overflow");
46   return __add(extent, usize(l.offset < 0 ? -l.offset : l.offset));
47 }

49 inline ssize_t
50 __offset(usize index, const buffer<usize> &shape, const layout &l, usize batch) noexcept
51 {
52   ssize_t offset = l.offset + ssize_t(batch) * l.distance;
53   if ( shape.size() == 1 ) return offset + ssize_t(index) * l.strides[0];
54   for ( usize j = shape.size(); j-- > 0; ) {
55     offset += ssize_t(index % shape[j]) * l.strides[j];
56     index /= shape[j];
57   }
58   return offset;
59 }

61 inline bool
62 __dense(const buffer<usize> &shape, const layout &l) noexcept
63 {
64   usize stride = 1;
65   for ( usize j = shape.size(); j-- > 0; ) {
66     if ( l.strides[j] != ssize_t(stride) ) return false;
67     stride *= shape[j];
68   }
69   return true;
70 }

84 template<fft_scalar F> class plan

86   descriptor<F> __desc;
87   buffer<usize> __input_shape, __output_shape, __steps;
88   buffer<__impl::__engine<F>> __engines;
89   buffer<complex<F>> __real_twiddles;
90   usize __volume{}, __input_volume{}, __output_volume{}, __transform_volume{ 1 };
91   usize __line_size{}, __engine_scratch{}, __data_size{}, __scratch_size{}, __packed_axis{};
92   bool __input_dense{}, __output_dense{};

94   usize
95   __packed_index(usize index, bool &mirror) const noexcept
96   {
97     usize coordinate = (index / __steps[__packed_axis]) % __desc.shape[__packed_axis];
98     mirror = coordinate >= __input_shape[__packed_axis];
99     usize result = 0, step = 1;
100     for ( usize a = __desc.shape.size(); a-- > 0; ) {
101       usize c = (index / __steps[a]) % __desc.shape[a];
102       if ( mirror )
103         for ( usize axis : __desc.axes )
104           if ( a == axis && c ) c = __desc.shape[a] - c;
105       result += c * step;
106       step *= __input_shape[a];
107     }
108     return result;
109   }

121   explicit plan(const descriptor<F> &d) : __desc(d), __input_shape(d.shape), __output_shape(d.shape)
122   {
123     if ( unsigned(d.kind) > unsigned(domain::real_backward) || unsigned(d.norm) > unsigned(normalization::ortho) )
124       exc<except::invalid_argument>("fft: invalid transform configuration");
125     if ( d.shape.empty() ) exc<except::invalid_argument>("fft: empty rank");
126     __volume = __impl::__product(d.shape);
127     if ( __desc.axes.empty() )
128       for ( usize i = 0; i < d.shape.size(); ++i ) __desc.axes.push_back(i);
129     __packed_axis = __desc.axes[__desc.axes.size() - 1];
130     for ( usize i = 0; i < __desc.axes.size(); ++i ) {
131       usize axis = __desc.axes[i];
132       if ( axis >= d.shape.size() ) exc<except::invalid_argument>("fft: axis out of range");
133       for ( usize j = 0; j < i; ++j )
134         if ( __desc.axes[j] == axis ) exc<except::invalid_argument>("fft: duplicate axis");
135       usize n = d.shape[axis];
136       __transform_volume = __impl::__mul(__transform_volume, n);
137       bool half = d.kind != domain::complex && axis == __packed_axis && n % 2 == 0;
138       __engines.push_back(__impl::__engine<F>(half ? n / 2 : n));
139       if ( half ) {
140         __real_twiddles.resize(n / 2 + 1);
141         for ( usize k = 0; k <= n / 2; ++k ) __real_twiddles[k] = __impl::__root<F>(k, n);
142       }
143       if ( n > __line_size ) __line_size = n;
144       usize scratch = __engines[i].scratch_size();
145       if ( scratch > __engine_scratch ) __engine_scratch = scratch;
146     }
147     __packed_axis = __desc.axes[__desc.axes.size() - 1];
148     if ( d.kind == domain::real_forward ) __output_shape[__packed_axis] = d.shape[__packed_axis] / 2 + 1;
149     if ( d.kind == domain::real_backward ) __input_shape[__packed_axis] = d.shape[__packed_axis] / 2 + 1;
150     __input_volume = __impl::__product(__input_shape);
151     __output_volume = __impl::__product(__output_shape);
152     __steps.resize(d.shape.size());
153     usize step = 1;
154     for ( usize i = d.shape.size(); i-- > 0; ) {
155       __steps[i] = step;
156       step *= d.shape[i];
157     }
158     __impl::__mul(__impl::__add(__impl::__complete_layout(__desc.input, __input_shape, d.batches), 1),
159                   d.kind == domain::real_forward ? sizeof(F) : sizeof(complex<F>));
160     __impl::__mul(__impl::__add(__impl::__complete_layout(__desc.output, __output_shape, d.batches), 1),
161                   d.kind == domain::real_backward ? sizeof(F) : sizeof(complex<F>));
162     __input_dense = __impl::__dense(__input_shape, __desc.input);
163     __output_dense = __impl::__dense(__output_shape, __desc.output);
164     __data_size = __impl::__mul(__volume, d.batches);
165     __scratch_size = __impl::__add(__impl::__mul(5, __line_size), __engine_scratch);
166     __impl::__mul(workspace_size(), sizeof(complex<F>));
167     __impl::__mul(__impl::__mul(__output_volume, d.batches), d.kind == domain::real_backward ? sizeof(F) : sizeof(complex<F>));
168   }

213   // Internal runner seam: one barrier per separable axis, one scratch slice per worker.
214   template<typename Load, typename Store, typename Runner>
215   max_t
216   __execute(Load load, Store store, workspace_view<F> work, direction dir, Runner &runner) const noexcept
217   {
218     if ( dir != direction::forward && dir != direction::backward ) return -22;
219     if ( !__desc.batches ) return 0;
220     if ( !runner.workers || runner.workers > (__impl::__limit - __data_size) / __scratch_size || !work.data
221          || work.size < __data_size + runner.workers * __scratch_size )
222       return -105;
223     auto *data = work.data;
224     for ( usize b = 0; b < __desc.batches; ++b )
225       for ( usize k = 0; k < __volume; ++k ) {
226         bool mirror = false;
227         usize source = __desc.kind == domain::real_backward ? __packed_index(k, mirror) : k;
228         ssize_t index = __input_dense ? __desc.input.offset + ssize_t(b) * __desc.input.distance + ssize_t(source)
229                                       : __impl::__offset(source, __input_shape, __desc.input, b);
230         auto z = load(index);
231         data[b * __volume + k] = mirror ? conj(z) : z;
232       }
233     for ( usize pass = 0; pass < __desc.axes.size(); ++pass ) {
234       usize a = __desc.kind == domain::real_forward ? (pass ? pass - 1 : __desc.axes.size() - 1) : pass;
235       usize axis = __desc.axes[a], n = __desc.shape[axis], stride = __steps[axis];
236       usize lines = __data_size / n;
237       usize tile = stride >= 4 ? 4 : 1;
238       runner((lines + tile - 1) / tile, data + __data_size, __scratch_size, [&](usize group, complex<F> *scratch) noexcept {
239         if ( __desc.kind == domain::complex && stride == 1 ) {
240           auto *line = data + group * n;
241           auto *temp = scratch + __line_size;
242           if ( lines == 1 )
243             __engines[a].execute_parallel(line, scratch, temp, dir == direction::backward, runner);
244           else
245             __engines[a].execute(line, scratch, temp, dir == direction::backward);
246           for ( usize j = 0; j < n; ++j ) line[j] = scratch[j];
247           return;
248         }
249         usize count = lines - group * tile < tile ? lines - group * tile : tile;
250         usize bases[4];
251         for ( usize lane = 0; lane < count; ++lane ) {
252           usize line = group * tile + lane;
253           bases[lane] = (line / stride) * (stride * n) + line % stride;
254         }
255         for ( usize j = 0; j < n; ++j )
256           for ( usize lane = 0; lane < count; ++lane ) scratch[lane * __line_size + j] = data[bases[lane] + j * stride];
257         auto *out = scratch + 4 * __line_size, *temp = out + __line_size;
258         for ( usize lane = 0; lane < count; ++lane ) {
259           auto *in = scratch + lane * __line_size;
260           usize base = bases[lane];
261           if ( !__real_twiddles.empty() && axis == __packed_axis ) {
262             usize h = n / 2;
263             if ( dir == direction::forward ) {
264               for ( usize j = 0; j < h; ++j ) in[j] = { in[2 * j].real, in[2 * j + 1].real };
265               if ( lines == 1 )
266                 __engines[a].execute_parallel(in, out, temp, false, runner);
267               else
268                 __engines[a].execute(in, out, temp, false);
269               for ( usize k = 0; k <= h; ++k ) {
270                 auto x = out[k % h], y = conj(out[(h - k) % h]);
271                 auto difference = (x - y) * __real_twiddles[k];
272                 auto z = (x + y + complex<F>{ difference.imag, -difference.real }) * F(0.5);
273                 in[k] = z;
274                 if ( k && k < h ) in[n - k] = conj(z);
275               }
276             } else {
277               for ( usize k = 0; k < h; ++k ) {
278                 auto x = data[base + k * stride], y = conj(data[base + (h - k) * stride]);
279                 auto difference = (x - y) * conj(__real_twiddles[k]);
280                 in[k] = (x + y + complex<F>{ -difference.imag, difference.real }) * F(0.5);
281               }
282               if ( lines == 1 )
283                 __engines[a].execute_parallel(in, out, temp, true, runner);
284               else
285                 __engines[a].execute(in, out, temp, true);
286               for ( usize k = 0; k < h; ++k ) {
287                 in[2 * k] = { F(2) * out[k].real };
288                 in[2 * k + 1] = { F(2) * out[k].imag };
289               }
290             }
291           } else {
292             if ( lines == 1 )
293               __engines[a].execute_parallel(in, out, temp, dir == direction::backward, runner);
294             else
295               __engines[a].execute(in, out, temp, dir == direction::backward);
296             for ( usize j = 0; j < n; ++j ) in[j] = out[j];
297           }
298         }
299         for ( usize j = 0; j < n; ++j )
300           for ( usize lane = 0; lane < count; ++lane ) data[bases[lane] + j * stride] = scratch[lane * __line_size + j];
301       });
302     }
303 
304     F scale = __impl::__scale<F>(__desc.norm, dir, __transform_volume)
305               * (dir == direction::forward ? __desc.forward_scale : __desc.backward_scale);
306     for ( usize b = 0; b < __desc.batches; ++b )
307       for ( usize k = 0; k < __output_volume; ++k ) {
308         usize source = k;
309         if ( __desc.kind == domain::real_forward ) {
310           usize t = k;
311           source = 0;
312           for ( usize j = __desc.shape.size(); j-- > 0; ) {
313             source += (t % __output_shape[j]) * __steps[j];
314             t /= __output_shape[j];
315           }
316         }
317         ssize_t index = __output_dense ? __desc.output.offset + ssize_t(b) * __desc.output.distance + ssize_t(k)
318                                        : __impl::__offset(k, __output_shape, __desc.output, b);
319         store(index, data[b * __volume + source] * scale);
320       }
321     return max_t(output_size() * (__desc.kind == domain::real_backward ? sizeof(F) : sizeof(complex<F>)));
322   }

324   max_t
325   execute(const complex<F> *in, complex<F> *out, workspace_view<F> work, direction dir = direction::forward) const noexcept
326   {
327     if ( __desc.kind != domain::complex || (__desc.batches && (!in || !out)) ) return -22;
328     if ( dir != direction::forward && dir != direction::backward ) return -22;
329     if ( __desc.shape.size() == 1 && __desc.batches == 1 && __desc.input.offset == 0 && __desc.output.offset == 0
330          && __desc.input.strides[0] == 1 && __desc.output.strides[0] == 1 ) {
331       if ( !work.data || work.size < __data_size + __scratch_size ) return -105;
332       const complex<F> *source = in;
333       if ( in == out ) {
334         for ( usize k = 0; k < __volume; ++k ) work.data[k] = in[k];
335         source = work.data;
336       }
337       __engines[0].execute(source, out, work.data + __volume, dir == direction::backward);
338       F scale = __impl::__scale<F>(__desc.norm, dir, __transform_volume)
339                 * (dir == direction::forward ? __desc.forward_scale : __desc.backward_scale);
340       if ( scale != F(1) )
341         for ( usize k = 0; k < __volume; ++k ) out[k] *= scale;
342       return max_t(__volume * sizeof(complex<F>));
343     }
344     __impl::__serial<F> runner;
345     return __execute([&](ssize_t i) { return in[i]; }, [&](ssize_t i, complex<F> z) { out[i] = z; }, work, dir, runner);
346   }

361   max_t
362   execute(const F *in, complex<F> *out, workspace_view<F> work) const noexcept
363   {
364     if ( __desc.kind != domain::real_forward || (__desc.batches && (!in || !out)) ) return -22;
365     if ( __desc.shape.size() == 1 && __desc.batches == 1 && __input_dense && __output_dense ) {
366       if ( !work.data || work.size < __data_size + __scratch_size ) return -105;
367       in += __desc.input.offset;
368       out += __desc.output.offset;
369       auto *packed = work.data, *spectrum = packed + __volume, *temp = spectrum + __volume;
370       usize n = __volume, h = n / 2;
371       F scale = __impl::__scale<F>(__desc.norm, direction::forward, n) * __desc.forward_scale;
372       if ( n % 2 == 0 ) {
373         for ( usize j = 0; j < h; ++j ) packed[j] = { in[2 * j], in[2 * j + 1] };
374         __engines[0].execute(packed, spectrum, temp, false);
375         for ( usize k = 0; k <= h; ++k ) {
376           auto x = spectrum[k == h ? 0 : k], y = conj(spectrum[k ? h - k : 0]);
377           auto difference = (x - y) * __real_twiddles[k];
378           out[k] = (x + y + complex<F>{ difference.imag, -difference.real }) * (F(0.5) * scale);
379         }
380       } else {
381         for ( usize j = 0; j < n; ++j ) packed[j] = { in[j] };
382         __engines[0].execute(packed, spectrum, temp, false);
383         for ( usize k = 0; k <= h; ++k ) out[k] = spectrum[k] * scale;
384       }
385       return max_t(__output_volume * sizeof(complex<F>));
386     }
387     __impl::__serial<F> runner;
388     return __execute([&](ssize_t i) { return complex<F>{ in[i] }; }, [&](ssize_t i, complex<F> z) { out[i] = z; }, work, direction::forward,
389                      runner);
390   }

392   max_t
393   execute(const complex<F> *in, F *out, workspace_view<F> work) const noexcept
394   {
395     if ( __desc.kind != domain::real_backward || (__desc.batches && (!in || !out)) ) return -22;
396     if ( __desc.shape.size() == 1 && __desc.batches == 1 && __input_dense && __output_dense ) {
397       if ( !work.data || work.size < __data_size + __scratch_size ) return -105;
398       in += __desc.input.offset;
399       out += __desc.output.offset;
400       auto *packed = work.data, *time = packed + __volume, *temp = time + __volume;
401       usize n = __volume, h = n / 2;
402       F scale = __impl::__scale<F>(__desc.norm, direction::backward, n) * __desc.backward_scale;
403       if ( n % 2 == 0 ) {
404         for ( usize k = 0; k < h; ++k ) {
405           auto x = in[k], y = conj(in[h - k]);
406           auto difference = (x - y) * conj(__real_twiddles[k]);
407           packed[k] = (x + y + complex<F>{ -difference.imag, difference.real }) * F(0.5);
408         }
409         __engines[0].execute(packed, time, temp, true);
410         for ( usize k = 0; k < h; ++k ) {
411           out[2 * k] = time[k].real * (F(2) * scale);
412           out[2 * k + 1] = time[k].imag * (F(2) * scale);
413         }
414       } else {
415         for ( usize k = 0; k <= h; ++k ) packed[k] = in[k];
416         for ( usize k = h + 1; k < n; ++k ) packed[k] = conj(in[n - k]);
417         __engines[0].execute(packed, time, temp, true);
418         for ( usize k = 0; k < n; ++k ) out[k] = time[k].real * scale;
419       }
420       return max_t(__output_volume * sizeof(F));
421     }
422     __impl::__serial<F> runner;
423     return __execute([&](ssize_t i) { return in[i]; }, [&](ssize_t i, complex<F> z) { out[i] = z.real; }, work, direction::backward,
424                      runner);
425   }

469 buffer<complex<F>>
470 transform(const complex<F> *input, usize n, direction dir = direction::forward, normalization norm = normalization::backward)
471 {
472   plan<F> p(n, norm);
473   workspace<F> work(p);
474   buffer<complex<F>> result(n);
475   if ( p.execute(input, result.data(), work.view(), dir) < 0 ) exc<except::invalid_argument>("fft: invalid input");
476   return result;

479 template<fft_scalar F>
480 buffer<complex<F>>
481 rfft(const F *input, usize n, normalization norm = normalization::backward)
482 {
483   descriptor<F> d(n);
484   d.kind = domain::real_forward;
485   d.norm = norm;
486   plan<F> p(d);
487   workspace<F> work(p);
488   buffer<complex<F>> result(p.output_size());
489   if ( p.execute(input, result.data(), work.view()) < 0 ) exc<except::invalid_argument>("fft: invalid input");
490   return result;
491 }

497   descriptor<F> d(n);

499   d.norm = norm;

504   return result;
