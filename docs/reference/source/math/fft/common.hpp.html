27 inline usize

29 {

36 {

41 inline usize
42 __product(const buffer<usize> &shape)
43 {
44   usize n = 1;
45   for ( usize v : shape ) {
46     if ( !v ) exc<except::invalid_argument>("fft: zero dimension");
47     n = __mul(n, v);
48   }
49   return n;

71 template<fft_scalar F>
72 F
73 __sqrt(F x) noexcept
74 {
75   if ( x == F(0) ) return x;
76   F scale = 1;
77   while ( x > F(4) ) {
78     x /= F(4);
79     scale *= F(2);
80   }
81   while ( x < F(1) ) {
82     x *= F(4);
83     scale /= F(2);
84   }
85   F y = (x + F(1)) / F(2);
86   for ( unsigned i = 0; i < 12; ++i ) y = (y + x / y) / F(2);
87   return y * scale;
88 }

90 // exp(-2*pi*i*k/n), with integer quadrant reduction before floating conversion.
91 template<fft_scalar F>
92 complex<F>
93 __root(usize k, usize n) noexcept
94 {
95   k %= n;
96   unsigned quadrant = 0;
97   // k*4 is bounded by the plan's checked phase denominator.
98   usize scaled = k * 4;
99   quadrant = unsigned(scaled / n);
100   usize residue = scaled % n;
101   bool reflect = residue > n / 2;
102   usize reduced = reflect ? n - residue : residue;
103   F x = (F(reduced) / F(n)) * F(1.570796326794896619231321691639751442098584699687552910487L);
104   F xx = x * x, s = x, c = 1, st = x, ct = 1;
105   for ( unsigned j = 1; j <= 20; ++j ) {
106     st *= -xx / F((2 * j) * (2 * j + 1));
107     s += st;
108     ct *= -xx / F((2 * j - 1) * (2 * j));
109     c += ct;
110   }
111   if ( reflect ) {
112     F tmp = s;
113     s = c;
114     c = tmp;
115   }
116   switch ( quadrant ) {
117   case 0:
118     return { c, -s };
119   case 1:
120     return { -s, -c };
121   case 2:
122     return { -c, s };
123   default:
124     return { s, c };
125   }
126 }

128 template<fft_scalar F>
129 F
130 __scale(normalization mode, direction dir, usize n) noexcept
131 {
132   if ( mode == normalization::ortho ) return F(1) / __sqrt(F(n));
133   if ( (mode == normalization::backward && dir == direction::backward) || (mode == normalization::forward && dir == direction::forward) )
134     return F(1) / F(n);
135   return F(1);
136 }

139 // Smallest 2/3/5/7-smooth length >= n; no linear scan through candidate lengths.
140 inline usize
141 next_fast_len(usize n)
142 {
143   if ( n <= 1 ) return 1;
144   if ( n > __impl::__limit / 8 ) exc<except::length_error>("fft: length too large");
145   usize best = 1;
146   while ( best < n ) best *= 2;
147   for ( usize a = 1; a <= best; ) {
148     for ( usize b = a; b <= best; ) {
149       for ( usize c = b; c <= best; ) {
150         usize d = c;
151         while ( d < n && d <= best / 2 ) d *= 2;
152         if ( d >= n && d < best ) best = d;
153         if ( c > best / 3 ) break;
154         c *= 3;
155       }
156       if ( b > best / 5 ) break;
157       b *= 5;
158     }
159     if ( a > best / 7 ) break;
160     a *= 7;
161   }
162   return best;
163 }
