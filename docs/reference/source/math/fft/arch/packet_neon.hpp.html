12 // GCC scalarizes generic vector<float> arithmetic on ARMv7 at strict -O3.
13 // Explicit NEON operations avoid that expansion and its register spills. They
14 // have ARMv7 NEON's native denormal behavior; no FPSCR state is changed.
15 #if defined(__micron_arch_arm32)
16 [[gnu::always_inline]] inline float32x4_t
17 __neon_add(float32x4_t a, float32x4_t b) noexcept
18 {
19   float32x4_t r;
20   asm("vadd.f32 %q0, %q1, %q2" : "=w"(r) : "w"(a), "w"(b));
21   return r;
22 }
23 
24 [[gnu::always_inline]] inline float32x4_t
25 __neon_sub(float32x4_t a, float32x4_t b) noexcept
26 {
27   float32x4_t r;
28   asm("vsub.f32 %q0, %q1, %q2" : "=w"(r) : "w"(a), "w"(b));
29   return r;
30 }
31 
32 [[gnu::always_inline]] inline float32x4_t
33 __neon_mul(float32x4_t a, float32x4_t b) noexcept
34 {
35   float32x4_t r;
36   asm("vmul.f32 %q0, %q1, %q2" : "=w"(r) : "w"(a), "w"(b));
37   return r;
38 }
39 
40 [[gnu::always_inline]] inline float32x4_t
41 __neon_mla(float32x4_t acc, float32x4_t a, float32x4_t b) noexcept
42 {
43   asm("vmla.f32 %q0, %q1, %q2" : "+w"(acc) : "w"(a), "w"(b));
44   return acc;
45 }
46 
47 [[gnu::always_inline]] inline float32x4_t
48 __neon_mls(float32x4_t acc, float32x4_t a, float32x4_t b) noexcept
49 {
50   asm("vmls.f32 %q0, %q1, %q2" : "+w"(acc) : "w"(a), "w"(b));
51   return acc;
52 }
53 #endif

55 template<> struct __complex_packet<float> {
56   struct value {
57     float32x4_t real, imag;
58   };
59 
60   static constexpr usize width = 4;
61 
62   static value
63   load(const complex<float> *p) noexcept
64   {
65     auto lo = simd::neon::load_f32(reinterpret_cast<const float *>(p));
66     auto hi = simd::neon::load_f32(reinterpret_cast<const float *>(p + 2));
67     return { simd::neon::unzip_lo_f32(lo, hi), simd::neon::unzip_hi_f32(lo, hi) };
68   }
69 
70   static void
71   store(complex<float> *p, value v) noexcept
72   {
73     auto lo = simd::neon::zip_lo_f32(v.real, v.imag), hi = simd::neon::zip_hi_f32(v.real, v.imag);
74     simd::neon::store_f32(reinterpret_cast<float *>(p), lo);
75     simd::neon::store_f32(reinterpret_cast<float *>(p + 2), hi);
76   }
77 
78   static value
79   add(value a, value b) noexcept
80   {
81 #if defined(__micron_arch_arm32)
82     return { __neon_add(a.real, b.real), __neon_add(a.imag, b.imag) };
83 #else
84     return { simd::neon::add(a.real, b.real), simd::neon::add(a.imag, b.imag) };
85 #endif
86   }
87 
88   static value
89   sub(value a, value b) noexcept
90   {
91 #if defined(__micron_arch_arm32)
92     return { __neon_sub(a.real, b.real), __neon_sub(a.imag, b.imag) };
93 #else
94     return { simd::neon::sub(a.real, b.real), simd::neon::sub(a.imag, b.imag) };
95 #endif
96   }
97 
98   static value
99   scale(value a, float b) noexcept
100   {
101 #if defined(__micron_arch_arm32)
102     return { __neon_mul(a.real, simd::neon::splat_f32(b)), __neon_mul(a.imag, simd::neon::splat_f32(b)) };
103 #else
104     return { simd::neon::mul(a.real, simd::neon::splat_f32(b)), simd::neon::mul(a.imag, simd::neon::splat_f32(b)) };
105 #endif
106   }
107 
108   static value
109   conjugate(value a) noexcept
110   {
111     return { a.real, simd::neon::neg(a.imag) };
112   }
113 
114   static value
115   mul(value a, value b) noexcept
116   {
117 #if defined(__micron_arch_arm32)
118     return { __neon_mls(__neon_mul(a.real, b.real), a.imag, b.imag), __neon_mla(__neon_mul(a.real, b.imag), a.imag, b.real) };
119 #elif defined(__micron_arm_fma)
120     return { simd::neon::fms(simd::neon::mul(a.real, b.real), a.imag, b.imag),
121              simd::neon::fma(simd::neon::mul(a.real, b.imag), a.imag, b.real) };
122 #else
123     return { simd::neon::sub(simd::neon::mul(a.real, b.real), simd::neon::mul(a.imag, b.imag)),
124              simd::neon::add(simd::neon::mul(a.real, b.imag), simd::neon::mul(a.imag, b.real)) };
125 #endif
126   }
127 
128   template<bool Inverse>
129   static value
130   rotate(value a) noexcept
131   {
132     if constexpr ( Inverse )
133       return { simd::neon::neg(a.imag), a.real };
134     else
135       return { a.imag, simd::neon::neg(a.real) };
136   }
137 };
