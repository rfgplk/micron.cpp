11 template<fft_scalar F> struct __scalar_packet {
12   using value = complex<F>;
13   static constexpr usize width = 1;
14 
15   static value
16   load(const complex<F> *p) noexcept
17   {
18     return *p;
19   }
20 
21   static void
22   store(complex<F> *p, value v) noexcept
23   {
24     *p = v;
25   }
26 
27   static value
28   add(value a, value b) noexcept
29   {
30     return a + b;
31   }
32 
33   static value
34   sub(value a, value b) noexcept
35   {
36     return a - b;
37   }
38 
39   static value
40   mul(value a, value b) noexcept
41   {
42     return a * b;
43   }
44 
45   static value
46   scale(value a, F b) noexcept
47   {
48     return a * b;
49   }
50 
51   static value
52   conjugate(value a) noexcept
53   {
54     return conj(a);
55   }
56 
57   template<bool Inverse>
58   static value
59   rotate(value a) noexcept
60   {
61     if constexpr ( Inverse )
62       return { -a.imag, a.real };
63     else
64       return { a.imag, -a.real };
65   }
66 };
