16 // matches math::compute's executor contract
17 // a false submission must not retain the job
18 template<fft_scalar F> class parallel_workspace
19 {
20   struct __job {
21     void *context{};
22     void (*function)(void *, usize, complex<F> *) noexcept {};
23     complex<F> *scratch{};
24     usize begin{}, end{};
25     u32 *remaining{};
26 
27     static void
28     __entry(void *opaque) noexcept
29     {
30       auto &j = *static_cast<__job *>(opaque);
31       for ( usize i = j.begin; i < j.end; ++i ) j.function(j.context, i, j.scratch);
32       auto *remaining = j.remaining;
33       if ( micron::atom::fetch_sub(remaining, u32(1), micron::atomic_acq_rel) == 1 ) micron::wake_futex(remaining, 0x7fffffff);
34     }
35   };
36 
37   static usize
38   __checked_workers(usize n)
39   {
40     if ( !n || n > 0x7fffffff ) exc<except::invalid_argument>("fft: invalid worker count");
41     return n;
42   }
43 
44   workspace<F> __work;
45   buffer<__job> __jobs;
46   alignas(4) u32 __remaining{};
47 
48   template<typename Executor> struct __runner {
49     parallel_workspace &owner;
50     Executor &executor;
51     usize workers;
52 
53     template<typename Function>
54     void
55     operator()(usize count, complex<F> *scratch, usize scratch_size, Function &&fn) noexcept
56     {
57       usize jobs = count < workers ? count : workers;
58       if ( !jobs ) return;
59       if ( jobs == 1 ) {
60         for ( usize i = 0; i < count; ++i ) fn(i, scratch);
61         return;
62       }
63       micron::atom::store(&owner.__remaining, u32(jobs), micron::atomic_release);
64       for ( usize w = 0; w < jobs; ++w ) {
65         auto &job = owner.__jobs[w];
66         job.context = &fn;
67         job.function
68             = [](void *context, usize i, complex<F> *s) noexcept { (*static_cast<micron::remove_reference_t<Function> *>(context))(i, s); };
69         job.scratch = scratch + w * scratch_size;
70         usize portion = count / jobs, remainder = count % jobs;
71         job.begin = w * portion + (w < remainder ? w : remainder);
72         job.end = job.begin + portion + (w < remainder ? 1 : 0);
73         job.remaining = &owner.__remaining;
74         if ( w + 1 == jobs || !executor.submit(&__job::__entry, &job) ) __job::__entry(&job);
75       }
76       while ( u32 n = micron::atom::load(&owner.__remaining, micron::atomic_acquire) ) (void)micron::try_wait_futex(&owner.__remaining, n);
77     }
78   };

81   template<typename Plan>
82   parallel_workspace(const Plan &p, usize workers) : __work(p.workspace_size(__checked_workers(workers))), __jobs(workers)
83   {
84     if ( !workers || workers > 0x7fffffff ) exc<except::invalid_argument>("fft: invalid worker count");
85   }

96   template<typename Executor, typename Input, typename Output>
97   max_t
98   execute(const plan<F> &p, const Input *in, Output *out, Executor &executor, direction dir = direction::forward) noexcept
99   {
100     static_assert(noexcept(executor.submit(static_cast<void (*)(void *) noexcept>(nullptr), nullptr)),
101                   "FFT executor submit must be noexcept");
102     static_assert((micron::is_same_v<Input, F> && micron::is_same_v<Output, complex<F>>)
103                       || (micron::is_same_v<Input, complex<F>> && (micron::is_same_v<Output, F> || micron::is_same_v<Output, complex<F>>)),
104                   "FFT input/output types must match the transform domains");
105     const auto &d = p.configuration();
106     if ( d.batches && (!in || !out) ) return -22;
107     if constexpr ( micron::is_same_v<Input, F> ) {
108       if ( d.kind != domain::real_forward ) return -22;
109       dir = direction::forward;
110     } else if constexpr ( micron::is_same_v<Output, F> ) {
111       if ( d.kind != domain::real_backward ) return -22;
112       dir = direction::backward;
113     } else if ( d.kind != domain::complex )
114       return -22;
115     __runner<Executor> runner{ *this, executor, workers() };
116     return p.__execute([&](ssize_t i) { return complex<F>{ in[i] }; },
117                        [&](ssize_t i, complex<F> z) {
118                          if constexpr ( micron::is_same_v<Output, F> )
119                            out[i] = z.real;
120                          else
121                            out[i] = z;
122                        },
123                        __work.view(), dir, runner);
124   }

129   {

136   template<typename Executor>
137   max_t
138   execute(const plan<F> &p, split_input<F> input, split_output<F> output, Executor &executor, direction dir = direction::forward) noexcept
139   {
140     static_assert(noexcept(executor.submit(static_cast<void (*)(void *) noexcept>(nullptr), nullptr)),
141                   "FFT executor submit must be noexcept");
142     if ( p.configuration().kind != domain::complex
143          || (p.configuration().batches && (!input.real || !input.imag || !output.real || !output.imag)) )
144       return -22;
145     __runner<Executor> runner{ *this, executor, workers() };
146     return p.__execute([&](ssize_t i) { return complex<F>{ input.real[i], input.imag[i] }; },
147                        [&](ssize_t i, complex<F> z) {
148                          output.real[i] = z.real;
149                          output.imag[i] = z.imag;
150                        },
151                        __work.view(), dir, runner);
152   }

154   template<typename Executor>
155   max_t
156   execute_inplace(const plan<F> &p, F *storage, Executor &executor) noexcept
157   {
158     static_assert(noexcept(executor.submit(static_cast<void (*)(void *) noexcept>(nullptr), nullptr)),
159                   "FFT executor submit must be noexcept");
160     auto kind = p.configuration().kind;
161     if ( kind == domain::complex || (p.configuration().batches && !storage) ) return -22;
162     __runner<Executor> runner{ *this, executor, workers() };
163     if ( kind == domain::real_forward )
164       return p.__execute([&](ssize_t i) { return complex<F>{ storage[i] }; },
165                          [&](ssize_t i, complex<F> z) {
166                            storage[2 * i] = z.real;
167                            storage[2 * i + 1] = z.imag;
168                          },
169                          __work.view(), direction::forward, runner);
170     return p.__execute([&](ssize_t i) { return complex<F>{ storage[2 * i], storage[2 * i + 1] }; },
171                        [&](ssize_t i, complex<F> z) { storage[i] = z.real; }, __work.view(), direction::backward, runner);
172   }
