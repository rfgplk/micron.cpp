19   static usize
20   __radix(usize n) noexcept
21   {
22     if ( n == 8 ) return 8;
23     for ( usize r : { usize(4), usize(2), usize(3), usize(5), usize(7) } )
24       if ( n % r == 0 ) return r;
25     return n;
26   }

28   template<bool Inverse>
29   void
30   __combine_stage(complex<F> *out, usize stage, usize begin, usize end) const noexcept
31   {
32     usize m = __sizes[stage + 1];
33     const auto *tw = __twiddles.data() + __twiddle_offsets[stage];
34     switch ( __radices[stage] ) {
35     case 2:
36       __stage<2, Inverse>(out, tw, m, begin, end);
37       break;
38     case 3:
39       __stage<3, Inverse>(out, tw, m, begin, end);
40       break;
41     case 4:
42       __stage<4, Inverse>(out, tw, m, begin, end);
43       break;
44     case 5:
45       __stage<5, Inverse>(out, tw, m, begin, end);
46       break;
47     case 7:
48       __stage<7, Inverse>(out, tw, m, begin, end);
49       break;
50     }
51   }

53   template<bool Inverse>
54   void
55   __smooth_stage(const complex<F> *in, complex<F> *out, usize stage, usize stride) const noexcept
56   {
57     if ( stage == __radices.size() ) {
58       *out = *in;
59       return;
60     }
61     usize r = __radices[stage], m = __sizes[stage + 1];
62     if ( m == 1 ) {
63       switch ( r ) {
64       case 2:
65         __leaf<2, Inverse>(in, out, stride);
66         break;
67       case 3:
68         __leaf<3, Inverse>(in, out, stride);
69         break;
70       case 4:
71         __leaf<4, Inverse>(in, out, stride);
72         break;
73       case 5:
74         __leaf<5, Inverse>(in, out, stride);
75         break;
76       case 7:
77         __leaf<7, Inverse>(in, out, stride);
78         break;
79       case 8:
80         __leaf<8, Inverse>(in, out, stride);
81         break;
82       }
83       return;
84     }
85     for ( usize j = 0; j < r; ++j ) __smooth_stage<Inverse>(in + j * stride, out + j * m, stage + 1, stride * r);
86     __combine_stage<Inverse>(out, stage, 0, m);
87   }

89   void
90   __smooth(const complex<F> *in, complex<F> *out, usize n, usize stride, bool inverse) const noexcept
91   {
92     usize stage = 0;
93     while ( __sizes[stage] != n ) ++stage;
94     if ( inverse )
95       __smooth_stage<true>(in, out, stage, stride);
96     else
97       __smooth_stage<false>(in, out, stage, stride);
98   }

100   void
101   __combine(complex<F> *out, usize, bool inverse, usize begin, usize end) const noexcept
102   {
103     if ( inverse )
104       __combine_stage<true>(out, 0, begin, end);
105     else
106       __combine_stage<false>(out, 0, begin, end);
107   }

109   template<typename Runner>
110   void
111   __parallel_smooth(const complex<F> *in, complex<F> *out, bool inverse, Runner &runner) const noexcept
112   {
113     if ( __m < 1024 || runner.workers == 1 ) {
114       __smooth(in, out, __m, 1, inverse);
115       return;
116     }
117     usize r = __radix(__m), m = __m / r;
118     runner(r, out, 0, [&](usize j, complex<F> *) noexcept { __smooth(in + j, out + j * m, m, r, inverse); });
119     constexpr usize block = 64;
120     runner((m + block - 1) / block, out, 0, [&](usize j, complex<F> *) noexcept {
121       usize end = (j + 1) * block < m ? (j + 1) * block : m;
122       __combine(out, __m, inverse, j * block, end);
123     });
124   }

126 public:
127   explicit __engine(usize n) : __n(n)
128   {
129     if ( !n || n > __limit / (16 * sizeof(complex<F>)) ) exc<except::length_error>("fft: unsupported length");
130     usize remaining = n;
131     for ( usize r : { usize(2), usize(3), usize(5), usize(7) } )
132       while ( remaining % r == 0 ) remaining /= r;
133     __m = remaining == 1 ? n : next_fast_len(__add(n, n) - 1);
134     usize offset = 0;
135     for ( usize size = __m; size > 1; ) {
136       usize radix = __radix(size);
137       __sizes.push_back(size);
138       __radices.push_back(radix);
139       __twiddle_offsets.push_back(offset);
140       offset += (radix - 1) * (size / radix);
141       size /= radix;
142     }
143     __sizes.push_back(1);
144     __twiddles.resize(offset);
145     for ( usize stage = 0; stage < __radices.size(); ++stage ) {
146       usize r = __radices[stage], m = __sizes[stage + 1], step = __m / __sizes[stage];
147       auto *tw = __twiddles.data() + __twiddle_offsets[stage];
148       for ( usize j = 1; j < r; ++j )
149         for ( usize k = 0; k < m; ++k ) {
150           auto w = __root<F>(j * k * step, __m);
151           tw[(j - 1) * m + k] = w;
152         }
153     }
154     if ( __m == __n ) return;
155     __chirp.resize(n);
156     __kernel.resize(__m);
157     buffer<complex<F>> time(__m);
158     usize square = 0, delta = 1, modulus = 2 * n;
159     for ( usize k = 0; k < n; ++k ) {
160       __chirp[k] = __root<F>(square, modulus);
161       time[k] = conj(__chirp[k]);
162       if ( k ) time[__m - k] = time[k];
163       square = (square + delta) % modulus;
164       delta = (delta + 2) % modulus;
165     }
166     __smooth(time.data(), __kernel.data(), __m, 1, false);
167   }

187   template<typename Runner>
188   void
189   execute_parallel(const complex<F> *in, complex<F> *out, complex<F> *scratch, bool inverse, Runner &runner) const noexcept
190   {
191     if ( __m == __n ) {
192       __parallel_smooth(in, out, inverse, runner);
193       return;
194     }
195     auto *a = scratch, *b = scratch + __m;
196     for ( usize k = 0; k < __m; ++k ) a[k] = k < __n ? (inverse ? conj(in[k]) : in[k]) * __chirp[k] : complex<F>{};
197     __parallel_smooth(a, b, false, runner);
198     constexpr usize block = 256;
199     runner((__m + block - 1) / block, b, 0, [&](usize j, complex<F> *) noexcept {
200       usize begin = j * block, count = __m - begin < block ? __m - begin : block;
201       __multiply(b + begin, __kernel.data() + begin, count);
202     });
203     __parallel_smooth(b, a, true, runner);
204     for ( usize k = 0; k < __n; ++k ) {
205       auto z = (a[k] * __chirp[k]) / F(__m);
206       out[k] = inverse ? conj(z) : z;
207     }
208   }

210   void
211   execute(const complex<F> *in, complex<F> *out, complex<F> *scratch, bool inverse) const noexcept
212   {
213     if ( __m == __n ) {
214       __smooth(in, out, __n, 1, inverse);
215       return;
216     }
217     auto *a = scratch, *b = scratch + __m;
218     for ( usize k = 0; k < __m; ++k ) a[k] = k < __n ? (inverse ? conj(in[k]) : in[k]) * __chirp[k] : complex<F>{};
219     __smooth(a, b, __m, 1, false);
220     __multiply(b, __kernel.data(), __m);
221     __smooth(b, a, __m, 1, true);
222     for ( usize k = 0; k < __n; ++k ) {
223       auto z = (a[k] * __chirp[k]) / F(__m);
224       out[k] = inverse ? conj(z) : z;
225     }
226   }
