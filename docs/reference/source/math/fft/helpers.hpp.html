15 template<fft_scalar F, bool Packed>
16 buffer<F>
17 __frequency_bins(usize n, F spacing)
18 {
19   if ( !n || n > __limit || !(spacing > F(0)) ) exc<except::invalid_argument>("fft: invalid sampling interval or length");
20   buffer<F> out(Packed ? n / 2 + 1 : n);
21   constexpr F minimum = [] {
22     if constexpr ( sizeof(F) == sizeof(float) )
23       return F(micron::numeric_limits<float>::min());
24     else if constexpr ( sizeof(F) == sizeof(double) )
25       return F(micron::numeric_limits<double>::min());
26     else
27       return F(micron::numeric_limits<long double>::min());
28   }();
29   constexpr F maximum = [] {
30     if constexpr ( sizeof(F) == sizeof(float) )
31       return F(micron::numeric_limits<float>::max());
32     else if constexpr ( sizeof(F) == sizeof(double) )
33       return F(micron::numeric_limits<double>::max());
34     else
35       return F(micron::numeric_limits<long double>::max());
36   }();
37   // These spacing bounds keep the reciprocal normal for every supported length.
38   if ( spacing >= F(2) / maximum && spacing <= (F(1) / F(__limit)) / minimum ) [[likely]] {
39     F step = F(1) / (F(n) * spacing);
40     for ( usize k = 0; k < out.size(); ++k ) out[k] = (Packed || k < (n + 1) / 2 ? F(k) : -F(n - k)) * step;
41   } else {
42     out[0] = F(0);
43     for ( usize k = 1; k < out.size(); ++k ) out[k] = ((Packed || k < (n + 1) / 2 ? F(k) : -F(n - k)) / F(n)) / spacing;
44   }
45   return out;
46 }
47 }      // namespace __impl

78 template<typename T>
79 buffer<T>
80 shift(const T *input, const buffer<usize> &shape, const buffer<usize> &axes = {}, bool inverse = false)
81 {
82   usize n = __impl::__product(shape);
83   if ( !input || shape.empty() ) exc<except::invalid_argument>("fft: invalid shift input");
84   buffer<usize> selected = axes;
85   if ( selected.empty() )
86     for ( usize i = 0; i < shape.size(); ++i ) selected.push_back(i);
87   for ( usize i = 0; i < selected.size(); ++i ) {
88     if ( selected[i] >= shape.size() ) exc<except::invalid_argument>("fft: invalid shift axis");
89     for ( usize j = 0; j < i; ++j )
90       if ( selected[i] == selected[j] ) exc<except::invalid_argument>("fft: duplicate shift axis");
91   }
92   buffer<T> out(n);
93   for ( usize i = 0; i < n; ++i ) {
94     usize index = i, source = 0, step = 1;
95     for ( usize a = shape.size(); a-- > 0; ) {
96       usize coordinate = index % shape[a];
97       index /= shape[a];
98       for ( usize axis : selected )
99         if ( axis == a ) coordinate = (coordinate + (inverse ? shape[a] / 2 : (shape[a] + 1) / 2)) % shape[a];
100       source += coordinate * step;
101       step *= shape[a];
102     }
103     out[i] = input[source];
104   }
105   return out;
106 }
