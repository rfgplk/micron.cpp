11 // each packet lane is an ind. butterfly
12 // NOTE: no horizontal reductions or materialized real/imaginary arrays are needed between loads and stores
13 template<usize R, bool Inverse, typename P, fft_scalar F>
14 [[gnu::always_inline]] inline void
15 __codelet(const typename P::value *v, complex<F> *out, usize stride) noexcept
16 {
17   if constexpr ( R == 2 ) {
18     P::store(out, P::add(v[0], v[1]));
19     P::store(out + stride, P::sub(v[0], v[1]));
20   } else if constexpr ( R == 3 ) {
21     auto sum = P::add(v[1], v[2]);
22     auto center = P::sub(v[0], P::scale(sum, F(0.5)));
23     auto difference
24         = P::template rotate<Inverse>(P::scale(P::sub(v[1], v[2]), F(0.866025403784438646763723170752936183471402626905190314L)));
25     P::store(out, P::add(v[0], sum));
26     P::store(out + stride, P::add(center, difference));
27     P::store(out + 2 * stride, P::sub(center, difference));
28   } else if constexpr ( R == 4 ) {
29     auto a = P::add(v[0], v[2]), b = P::sub(v[0], v[2]);
30     auto c = P::add(v[1], v[3]), d = P::template rotate<Inverse>(P::sub(v[1], v[3]));
31     P::store(out, P::add(a, c));
32     P::store(out + 2 * stride, P::sub(a, c));
33     P::store(out + stride, P::add(b, d));
34     P::store(out + 3 * stride, P::sub(b, d));
35   } else if constexpr ( R == 5 ) {
36     constexpr F c1 = F(0.309016994374947424102293417182819058860154589902881431L);
37     constexpr F c2 = F(-0.809016994374947424102293417182819058860154589902881431L);
38     constexpr F s1 = F(0.951056516295153572116439333379382143405698634125750222L);
39     constexpr F s2 = F(0.587785252292473129168705954639072768597652437643145991L);
40     auto p = P::add(v[1], v[4]), q = P::add(v[2], v[3]);
41     auto d = P::sub(v[1], v[4]), e = P::sub(v[2], v[3]);
42     P::store(out, P::add(P::add(v[0], p), q));
43     auto a = P::add(v[0], P::add(P::scale(p, c1), P::scale(q, c2)));
44     auto c = P::template rotate<Inverse>(P::add(P::scale(d, s1), P::scale(e, s2)));
45     P::store(out + stride, P::add(a, c));
46     P::store(out + 4 * stride, P::sub(a, c));
47     auto b = P::add(v[0], P::add(P::scale(p, c2), P::scale(q, c1)));
48     auto f = P::template rotate<Inverse>(P::sub(P::scale(d, s2), P::scale(e, s1)));
49     P::store(out + 2 * stride, P::add(b, f));
50     P::store(out + 3 * stride, P::sub(b, f));
51   } else if constexpr ( R == 8 ) {
52     constexpr F h = F(0.707106781186547524400844362104849039284835937688474037L);
53     auto a = P::add(v[0], v[4]), b = P::sub(v[0], v[4]);
54     auto c = P::add(v[2], v[6]), d = P::template rotate<Inverse>(P::sub(v[2], v[6]));
55     auto e0 = P::add(a, c), e2 = P::sub(a, c), e1 = P::add(b, d), e3 = P::sub(b, d);
56     a = P::add(v[1], v[5]);
57     b = P::sub(v[1], v[5]);
58     c = P::add(v[3], v[7]);
59     d = P::template rotate<Inverse>(P::sub(v[3], v[7]));
60     auto t = P::add(a, c);
61     P::store(out, P::add(e0, t));
62     P::store(out + 4 * stride, P::sub(e0, t));
63     t = P::template rotate<Inverse>(P::sub(a, c));
64     P::store(out + 2 * stride, P::add(e2, t));
65     P::store(out + 6 * stride, P::sub(e2, t));
66     t = P::add(b, d);
67     t = P::scale(P::add(t, P::template rotate<Inverse>(t)), h);
68     P::store(out + stride, P::add(e1, t));
69     P::store(out + 5 * stride, P::sub(e1, t));
70     t = P::sub(b, d);
71     t = P::scale(P::sub(P::template rotate<Inverse>(t), t), h);
72     P::store(out + 3 * stride, P::add(e3, t));
73     P::store(out + 7 * stride, P::sub(e3, t));
74   } else {
75     static_assert(R == 7);
76     constexpr F c[3]
77         = { F(0.623489801858733530525004884004239810632274730896402106L), F(-0.222520933956314404288902564496794759466355568764544955L),
78             F(-0.900968867902419126236102319507445051165919162131857151L) };
79     constexpr F s[3]
80         = { F(0.781831482468029808708444526674057750232334518708687529L), F(0.974927912181823607018131682993931217232785800619997438L),
81             F(0.433883739117558120475768332848358754609990727787459876L) };
82     typename P::value sums[3], differences[3];
83     auto dc = v[0];
84     for ( usize j = 0; j < 3; ++j ) {
85       sums[j] = P::add(v[j + 1], v[6 - j]);
86       differences[j] = P::sub(v[j + 1], v[6 - j]);
87       dc = P::add(dc, sums[j]);
88     }
89     P::store(out, dc);
90     for ( usize l = 1; l <= 3; ++l ) {
91       auto a = P::add(v[0], P::scale(sums[0], c[l - 1]));
92       auto b = P::scale(differences[0], s[l - 1]);
93       for ( usize j = 1; j < 3; ++j ) {
94         usize phase = (l * (j + 1)) % 7, index = phase > 3 ? 6 - phase : phase - 1;
95         a = P::add(a, P::scale(sums[j], c[index]));
96         b = P::add(b, P::scale(differences[j], phase > 3 ? -s[index] : s[index]));
97       }
98       b = P::template rotate<Inverse>(b);
99       P::store(out + l * stride, P::add(a, b));
100       P::store(out + (7 - l) * stride, P::sub(a, b));
101     }
102   }
103 }

105 template<usize R, bool Inverse, typename P, fft_scalar F>
106 [[gnu::always_inline]] inline void
107 __stage_packet(complex<F> *out, const complex<F> *twiddles, usize m, usize k) noexcept
108 {
109   typename P::value v[R];
110   v[0] = P::load(out + k);
111   for ( usize j = 1; j < R; ++j ) {
112     auto w = P::load(twiddles + (j - 1) * m + k);
113     if constexpr ( Inverse ) w = P::conjugate(w);
114     v[j] = P::mul(P::load(out + j * m + k), w);
115   }
116   __codelet<R, Inverse, P>(v, out + k, m);
117 }
118 
119 template<usize R, bool Inverse, fft_scalar F>
120 void
121 __stage(complex<F> *out, const complex<F> *twiddles, usize m, usize begin, usize end) noexcept
122 {
123   using P = __complex_packet<F>;
124   if ( begin >= end ) return;
125   const usize vector_end = end - (end - begin) % P::width;
126   usize k = begin;
127   for ( ; k < vector_end; k += P::width ) __stage_packet<R, Inverse, P>(out, twiddles, m, k);
128   if constexpr ( P::width > 1 )
129     for ( usize tail = 0; tail < (end - begin) % P::width; ++tail )
130       __stage_packet<R, Inverse, __scalar_packet<F>>(out, twiddles, m, vector_end + tail);
131 }

133 template<usize R, bool Inverse, fft_scalar F>
134 [[gnu::noinline]] void
135 __leaf_large(const complex<F> *in, complex<F> *out, usize stride) noexcept
136 {
137   using P = __scalar_packet<F>;
138   typename P::value v[R];
139   for ( usize j = 0; j < R; ++j ) v[j] = in[j * stride];
140   __codelet<R, Inverse, P>(v, out, 1);
141 }
142 
143 template<usize R, bool Inverse, fft_scalar F>
144 [[gnu::always_inline]] inline void
145 __leaf(const complex<F> *in, complex<F> *out, usize stride) noexcept
146 {
147   if constexpr ( R > 4 ) {
148     // NOTE: keep the larger codelets live ranges and stack frame out of the recursive radix-2/4 dispatcher
149     // they are reached only at the leaves
150     __leaf_large<R, Inverse>(in, out, stride);
151   } else {
152     using P = __scalar_packet<F>;
153     typename P::value v[R] = { in[0], in[stride] };
154     if constexpr ( R >= 3 ) v[2] = in[2 * stride];
155     if constexpr ( R == 4 ) v[3] = in[3 * stride];
156     __codelet<R, Inverse, P>(v, out, 1);
157   }
158 }
