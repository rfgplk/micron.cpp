13 enum class r2r_kind { dct_i, dct_ii, dct_iii, dct_iv, dst_i, dst_ii, dst_iii, dst_iv };

17 inline r2r_kind
18 __inverse_kind(r2r_kind kind) noexcept
19 {
20   switch ( kind ) {
21   case r2r_kind::dct_ii:
22     return r2r_kind::dct_iii;
23   case r2r_kind::dct_iii:
24     return r2r_kind::dct_ii;
25   case r2r_kind::dst_ii:
26     return r2r_kind::dst_iii;
27   case r2r_kind::dst_iii:
28     return r2r_kind::dst_ii;
29   default:
30     return kind;
31   }
32 }

34 inline usize
35 __logical_size(usize n, r2r_kind kind)
36 {
37   if ( !n || (kind == r2r_kind::dct_i && n < 2) ) exc<except::invalid_argument>("fft: invalid real transform length");
38   if ( kind == r2r_kind::dct_i ) return __mul(2, n - 1);
39   if ( kind == r2r_kind::dst_i ) return __mul(2, __add(n, 1));
40   return __mul(2, n);
41 }
42 
43 inline usize
44 __embedding_size(usize n, r2r_kind kind)
45 {
46   usize logical = __logical_size(n, kind);
47   return kind == r2r_kind::dct_iv || kind == r2r_kind::dst_iv ? __mul(4, logical) : logical;
48 }

50 template<fft_scalar F> class __real_engine
51 {
52   usize __n, __logical, __length;
53   r2r_kind __kind;
54   __engine<F> __fft;
55   buffer<complex<F>> __phase;
56 
57 public:
58   __real_engine(usize n, r2r_kind kind)
59       : __n(n), __logical(__logical_size(n, kind)), __length(__embedding_size(n, kind)), __kind(kind), __fft(__length), __phase(n + 1)
60   {
61     if ( unsigned(kind) > unsigned(r2r_kind::dst_iv) ) exc<except::invalid_argument>("fft: invalid real transform kind");
62     for ( usize k = 0; k <= n; ++k ) __phase[k] = __root<F>(k, 4 * n);
63   }
64 
65   usize
66   logical_size() const noexcept
67   {
68     return __logical;
69   }

71   usize
72   scratch_size() const noexcept
73   {
74     return 2 * __length + __fft.scratch_size();
75   }

77   template<typename Runner>
78   void
79   execute(const complex<F> *input, complex<F> *output, complex<F> *scratch, bool inverse, bool ortho, Runner &runner,
80           bool parallel) const noexcept
81   {
82     r2r_kind kind = inverse ? __inverse_kind(__kind) : __kind;
83     auto *time = scratch, *freq = time + __length, *temp = freq + __length;
84     for ( usize i = 0; i < __length; ++i ) time[i] = {};
85     F root2 = __sqrt(F(2));
86     for ( usize j = 0; j < __n; ++j ) {
87       F x = input[j].real;
88       if ( ortho
89            && ((kind == r2r_kind::dct_i && (j == 0 || j + 1 == __n)) || (kind == r2r_kind::dct_iii && j == 0)
90                || (kind == r2r_kind::dst_iii && j + 1 == __n)) )
91         x *= root2;
92       switch ( kind ) {
93       case r2r_kind::dct_i:
94         time[j] = { x };
95         if ( j && j + 1 < __n ) time[__length - j] = { x };
96         break;
97       case r2r_kind::dct_ii:
98         time[j] = { x };
99         time[__length - 1 - j] = { x };
100         break;
101       case r2r_kind::dct_iii:
102         time[j] = conj(__phase[j]) * x;
103         if ( j ) time[__length - j] = conj(time[j]);
104         break;
105       case r2r_kind::dct_iv:
106         time[2 * j + 1] = { x };
107         time[__length - (2 * j + 1)] = { x };
108         break;
109       case r2r_kind::dst_i:
110         time[j + 1] = { x };
111         time[__length - j - 1] = { -x };
112         break;
113       case r2r_kind::dst_ii:
114         time[j] = { x };
115         time[__length - 1 - j] = { -x };
116         break;
117       case r2r_kind::dst_iii:
118         time[j + 1] = complex<F>{ 0, -x } * conj(__phase[j + 1]);
119         if ( j + 1 < __n ) time[__length - j - 1] = conj(time[j + 1]);
120         break;
121       case r2r_kind::dst_iv:
122         time[2 * j + 1] = { x };
123         time[__length - (2 * j + 1)] = { -x };
124         break;
125       }
126     }
127     if ( parallel )
128       __fft.execute_parallel(time, freq, temp, kind == r2r_kind::dct_iii || kind == r2r_kind::dst_iii, runner);
129     else
130       __fft.execute(time, freq, temp, kind == r2r_kind::dct_iii || kind == r2r_kind::dst_iii);
131     for ( usize k = 0; k < __n; ++k ) {
132       F x{};
133       switch ( kind ) {
134       case r2r_kind::dct_i:
135       case r2r_kind::dct_iii:
136       case r2r_kind::dst_iii:
137         x = freq[k].real;
138         break;
139       case r2r_kind::dct_ii:
140         x = (freq[k] * __phase[k]).real;
141         break;
142       case r2r_kind::dct_iv:
143         x = freq[2 * k + 1].real;
144         break;
145       case r2r_kind::dst_i:
146         x = -freq[k + 1].imag;
147         break;
148       case r2r_kind::dst_ii:
149         x = -(freq[k + 1] * __phase[k + 1]).imag;
150         break;
151       case r2r_kind::dst_iv:
152         x = -freq[2 * k + 1].imag;
153         break;
154       }
155       if ( ortho
156            && ((kind == r2r_kind::dct_i && (k == 0 || k + 1 == __n)) || (kind == r2r_kind::dct_ii && k == 0)
157                || (kind == r2r_kind::dst_ii && k + 1 == __n)) )
158         x /= root2;
159       output[k] = { x };
160     }
161   }

165 template<fft_scalar F> class r2r_plan
166 {
167   descriptor<F> __desc;
168   buffer<__impl::__real_engine<F>> __engines;
169   buffer<usize> __steps;
170   usize __volume{}, __logical{ 1 }, __line_size{}, __scratch_size{}, __data_size{};
171 
172 public:
173   r2r_plan(usize n, r2r_kind kind, normalization norm = normalization::backward)
174       : r2r_plan(
175             [&] {
176               descriptor<F> d(n);
177               d.norm = norm;
178               return d;
179             }(),
180             buffer<r2r_kind>{ kind })
181   {
182   }
183 
184   r2r_plan(const descriptor<F> &d, const buffer<r2r_kind> &kinds) : __desc(d)
185   {
186     if ( unsigned(d.norm) > unsigned(normalization::ortho) ) exc<except::invalid_argument>("fft: invalid normalization");
187     if ( d.shape.empty() || d.kind != domain::complex ) exc<except::invalid_argument>("fft: invalid real transform descriptor");
188     __volume = __impl::__product(d.shape);
189     if ( __desc.axes.empty() )
190       for ( usize i = 0; i < d.shape.size(); ++i ) __desc.axes.push_back(i);
191     if ( kinds.size() != __desc.axes.size() ) exc<except::invalid_argument>("fft: one real transform kind per axis required");
192     usize engine_scratch = 0;
193     for ( usize i = 0; i < __desc.axes.size(); ++i ) {
194       usize axis = __desc.axes[i];
195       if ( axis >= d.shape.size() ) exc<except::invalid_argument>("fft: axis out of range");
196       for ( usize j = 0; j < i; ++j )
197         if ( __desc.axes[j] == axis ) exc<except::invalid_argument>("fft: duplicate axis");
198       __engines.push_back(__impl::__real_engine<F>(d.shape[axis], kinds[i]));
199       __logical = __impl::__mul(__logical, __engines[i].logical_size());
200       if ( __engines[i].scratch_size() > engine_scratch ) engine_scratch = __engines[i].scratch_size();
201       if ( d.shape[axis] > __line_size ) __line_size = d.shape[axis];
202     }
203     __scratch_size = __impl::__add(__impl::__mul(2, __line_size), engine_scratch);
204     __data_size = __impl::__mul(__volume, d.batches);
205     __steps.resize(d.shape.size());
206     usize step = 1;
207     for ( usize i = d.shape.size(); i-- > 0; ) {
208       __steps[i] = step;
209       step *= d.shape[i];
210     }
211     __impl::__mul(__impl::__add(__impl::__complete_layout(__desc.input, d.shape, d.batches), 1), sizeof(F));
212     __impl::__mul(__impl::__add(__impl::__complete_layout(__desc.output, d.shape, d.batches), 1), sizeof(F));
213     __impl::__mul(workspace_size(), sizeof(complex<F>));
214   }

241   template<typename Runner>
242   max_t
243   __execute(const F *input, F *output, workspace_view<F> work, direction dir, Runner &runner) const noexcept
244   {
245     if ( dir != direction::forward && dir != direction::backward ) return -22;
246     if ( !__desc.batches ) return 0;
247     if ( !input || !output ) return -22;
248     if ( !runner.workers || runner.workers > (__impl::__limit - __data_size) / __scratch_size || !work.data
249          || work.size < __data_size + runner.workers * __scratch_size )
250       return -105;
251     auto *data = work.data;
252     for ( usize b = 0; b < __desc.batches; ++b )
253       for ( usize k = 0; k < __volume; ++k ) data[b * __volume + k] = { input[__impl::__offset(k, __desc.shape, __desc.input, b)] };
254     for ( usize a = 0; a < __desc.axes.size(); ++a ) {
255       usize axis = __desc.axes[a], n = __desc.shape[axis], stride = __steps[axis];
256       runner(__data_size / n, data + __data_size, __scratch_size, [&](usize line, complex<F> *scratch) noexcept {
257         usize base = (line / stride) * (stride * n) + line % stride;
258         auto *in = scratch, *out = scratch + __line_size, *temp = out + __line_size;
259         for ( usize j = 0; j < n; ++j ) in[j] = data[base + j * stride];
260         __engines[a].execute(in, out, temp, dir == direction::backward, __desc.norm == normalization::ortho, runner, __data_size / n == 1);
261         for ( usize j = 0; j < n; ++j ) data[base + j * stride] = out[j];
262       });
263     }
264     F scale = __impl::__scale<F>(__desc.norm, dir, __logical) * (dir == direction::forward ? __desc.forward_scale : __desc.backward_scale);
265     for ( usize b = 0; b < __desc.batches; ++b )
266       for ( usize k = 0; k < __volume; ++k )
267         output[__impl::__offset(k, __desc.shape, __desc.output, b)] = data[b * __volume + k].real * scale;
268     return max_t(__data_size * sizeof(F));
269   }

271   max_t
272   execute(const F *input, F *output, workspace_view<F> work, direction dir = direction::forward) const noexcept
273   {
274     __impl::__serial<F> runner;

279 template<fft_scalar F>
280 buffer<F>
281 real_transform(const F *input, usize n, r2r_kind kind, direction dir = direction::forward, normalization norm = normalization::backward)
282 {
283   r2r_plan<F> p(n, kind, norm);
284   workspace<F> work(p);
285   buffer<F> out(n);
286   if ( p.execute(input, out.data(), work.view(), dir) < 0 ) exc<except::invalid_argument>("fft: invalid real transform input");
287   return out;
288 }

290 template<fft_scalar F>
291 buffer<F>
292 dct(const F *input, usize n, unsigned type = 2, normalization norm = normalization::backward, direction dir = direction::forward)
293 {
294   if ( type < 1 || type > 4 ) exc<except::invalid_argument>("fft: DCT type must be 1..4");
295   return real_transform(input, n, r2r_kind(type - 1), dir, norm);
296 }

299 buffer<F>
300 dst(const F *input, usize n, unsigned type = 2, normalization norm = normalization::backward, direction dir = direction::forward)
301 {
302   if ( type < 1 || type > 4 ) exc<except::invalid_argument>("fft: DST type must be 1..4");
303   return real_transform(input, n, r2r_kind(type + 3), dir, norm);
