166 struct host_domains {
167   using domain_type = u16;
168   static constexpr domain_type host = 0;
169 
170   [[nodiscard]] static constexpr bool
171   valid(domain_type domain) noexcept
172   {
173     return domain == host;
174   }
175 
176   [[nodiscard]] static constexpr usize
177   minimum_alignment(domain_type) noexcept
178   {
179     return alignof(max_align_t);
180   }
181 
182   [[nodiscard]] static void *
183   allocate(domain_type domain, usize bytes, usize alignment) noexcept
184   {
185     if ( !valid(domain) || bytes == 0 || alignment == 0 || (alignment & (alignment - 1)) != 0 ) return nullptr;
186     usize padded{};
187     if ( __builtin_add_overflow(bytes, alignment - 1, &padded) ) return nullptr;
188     padded &= ~(alignment - 1);
189     return micron::__alloc_aligned(alignment, padded);
190   }
191 
192   static void
193   release(domain_type, void *pointer, usize, usize alignment) noexcept
194   {
195     if ( !pointer ) return;
196     micron::__free_aligned(pointer, alignment);
197   }
198 
199   [[nodiscard]] static compute_status
200   copy(domain_type destination_domain, void *destination, domain_type source_domain, const void *source, usize bytes) noexcept
201   {
202     if ( !valid(destination_domain) || !valid(source_domain) || (!destination && bytes) || (!source && bytes) )
203       return compute_status::domain_failure;
204     if ( bytes ) micron::memcpy(reinterpret_cast<byte *>(destination), reinterpret_cast<const byte *>(source), bytes);
205     return compute_status::ok;
206   }
207 
208   [[nodiscard]] static constexpr compute_status
209   synchronize(domain_type) noexcept
210   {
211     return compute_status::ok;
212   }
213 };

246 template<domain_provider Domains = host_domains> class basic_tensor_storage
247 {
248   using control_type = __impl::__tensor_storage_control<Domains>;
249   control_type *__control{};
250 
251   explicit basic_tensor_storage(control_type *control) noexcept : __control(control) { }
252 
253   void
254   __release() noexcept
255   {
256     if ( !__control ) return;
257     if ( __control->references.fetch_sub(usize(1), micron::memory_order_acq_rel) == 1 ) {
258       Domains::release(__control->domain, __control->pointer, __control->bytes, __control->alignment);
259       delete __control;
260     }
261     __control = nullptr;
262   }
263 
264 public:
265   using domain_type = typename Domains::domain_type;
266 
267   basic_tensor_storage() noexcept = default;
268 
269   ~basic_tensor_storage() { __release(); }
270 
271   basic_tensor_storage(const basic_tensor_storage &other) noexcept : __control(other.__control)
272   {
273     if ( __control ) __control->references.fetch_add(usize(1), micron::memory_order_relaxed);
274   }
275 
276   basic_tensor_storage(basic_tensor_storage &&other) noexcept : __control(other.__control) { other.__control = nullptr; }
277 
278   basic_tensor_storage &
279   operator=(const basic_tensor_storage &other) noexcept
280   {
281     if ( this == micron::addressof(other) ) return *this;
282     if ( other.__control ) other.__control->references.fetch_add(usize(1), micron::memory_order_relaxed);
283     __release();
284     __control = other.__control;
285     return *this;
286   }
287 
288   basic_tensor_storage &
289   operator=(basic_tensor_storage &&other) noexcept
290   {
291     if ( this == micron::addressof(other) ) return *this;
292     __release();
293     __control = other.__control;
294     other.__control = nullptr;
295     return *this;
296   }
297 
298   [[nodiscard]] static basic_tensor_storage
299   allocate(domain_type domain, usize bytes, usize alignment)
300   {
301     if ( bytes == 0 ) return basic_tensor_storage(new control_type(nullptr, 0, alignment, domain));
302     void *pointer = Domains::allocate(domain, bytes, alignment);
303     if ( !pointer ) return {};
304 #if !defined(__micron_freestanding) || defined(__micron_eh)
305     try {
306       return basic_tensor_storage(new control_type(pointer, bytes, alignment, domain));
307     } catch ( ... ) {
308       Domains::release(domain, pointer, bytes, alignment);
309       throw;
310     }
311 #else
312     return basic_tensor_storage(new control_type(pointer, bytes, alignment, domain));
313 #endif
314   }

446 template<typename Ops, domain_provider Domains> class compute_session
447 {
448   static_assert(__compute_session_registry<Ops>, "compute_session requires a compile-time operation registry");
449 
450 public:
451   using plan_type = compute_plan<Ops, Domains>;
452   using operation_type = typename Ops::operation_type;
453   using dtype_type = typename Ops::dtype_type;
454   using context_type = typename Ops::context_type;
455   using descriptor_type = tensor_descriptor<dtype_type>;
456   using storage_type = basic_tensor_storage<Domains>;
457   using shared_tensor_type = shared_tensor<dtype_type, Domains>;
458 
459 private:
460   using runtime_tensor = __impl::__runtime_tensor<Domains>;
461   using job_type = __impl::__compute_job<compute_session>;
462 
463   const plan_type *__plan{};
464   context_type __context;
465   __impl::__compute_vec<storage_type> __slot_storage;
466   __impl::__compute_vec<runtime_tensor> __runtime;
467   __impl::__compute_vec<tensor_view<dtype_type>> __input_views;
468   __impl::__compute_vec<tensor_view<dtype_type>> __output_views;
469   __impl::__compute_vec<job_type> __jobs;
470   __impl::__compute_vec<micron::atomic_token<u32>> __pending;
471   __impl::__compute_vec<micron::atomic_token<u8>> __failed_dependency;
472   __impl::__compute_vec<compute_status> __operation_status;
473   micron::atomic_token<u32> __remaining{ 0 };
474   micron::atomic_token<u32> __earliest_failure{ operation_id::invalid_value() };
475   void *__executor{};
476   bool (*__submit)(void *, void (*)(void *) noexcept, void *) noexcept {};
477   bool __executing{};

479   [[nodiscard]] bool
480   __valid_tensor(tensor_id tensor) const noexcept
481   {
482     return __plan && tensor.valid() && tensor.value < __plan->__tensors.size();
483   }
484 
485   [[nodiscard]] static usize
486   __logical_origin(const descriptor_type &descriptor, ssize_t minimum) noexcept
487   {
488     (void)descriptor;
489     return minimum < 0 ? static_cast<usize>(-(minimum + 1)) + 1 : 0;
490   }

492   void
493   __refresh_runtime_pointers() noexcept
494   {
495     for ( usize tensor = 0; tensor < __plan->__tensors.size(); ++tensor ) {
496       const auto &meta = __plan->__tensors.data()[tensor];
497       if ( meta.root.value == tensor ) continue;
498       auto &runtime = __runtime.data()[tensor];
499       const auto &root = __runtime.data()[meta.root.value];
500       runtime.storage = root.storage;
501       runtime.borrowed = root.borrowed;
502       runtime.pointer = root.pointer ? reinterpret_cast<byte *>(root.pointer) + meta.root_offset : nullptr;
503     }
504   }
505 
506   void
507   __initialize()
508   {
509     static_assert(noexcept(Ops::execute(micron::declval<const operation_type &>(), micron::declval<context_type &>(),
510                                         micron::declval<micron::raw_slice<const tensor_view<dtype_type>>>(),
511                                         micron::declval<micron::raw_slice<tensor_view<dtype_type>>>())),
512                   "Ops::execute must be noexcept");
513     if ( !__plan ) __impl::__compute_plan_error("compute_session: null plan");
514     __slot_storage.reserve(__plan->__slots.size());
515     for ( const auto &slot : __plan->__slots ) {
516       auto storage = storage_type::allocate(slot.domain, slot.bytes, slot.alignment);
517       if ( !storage ) micron::exc<micron::except::memory_error>("compute_session: arena allocation failed");
518       if ( slot.bytes ) micron::memset(storage.data(), byte(0), slot.bytes);
519       __slot_storage.push_back(micron::move(storage));
520     }
521 
522     __runtime.resize(__plan->__tensors.size());
523     for ( usize tensor = 0; tensor < __plan->__tensors.size(); ++tensor ) {
524       const auto &meta = __plan->__tensors.data()[tensor];
525       if ( meta.root.value != tensor || meta.external || meta.slot == micron::numeric_limits<u32>::max() ) continue;
526       auto &runtime = __runtime.data()[tensor];
527       runtime.storage = __slot_storage.data()[meta.slot];
528       runtime.pointer = runtime.storage.data()
529                             ? reinterpret_cast<byte *>(runtime.storage.data()) + __logical_origin(meta.descriptor, meta.minimum_offset)
530                             : nullptr;
531       runtime.valid = meta.state;
532     }
533     __refresh_runtime_pointers();
534 
535     __input_views.resize(__plan->__inputs.size());
536     __output_views.resize(__plan->__outputs.size());
537     __jobs.resize(__plan->__operations.size());
538     __pending.resize(__plan->__operations.size(), micron::atomic_token<u32>(0));
539     __failed_dependency.resize(__plan->__operations.size(), micron::atomic_token<u8>(0));
540     __operation_status.resize(__plan->__operations.size(), compute_status::ok);
541     for ( usize operation = 0; operation < __jobs.size(); ++operation ) __jobs.data()[operation] = { this, static_cast<u32>(operation) };
542   }
543 
544   void
545   __publish_failure(u32 operation, compute_status status) noexcept
546   {
547     __operation_status.data()[operation] = status;
548     u32 current = __earliest_failure.get(micron::memory_order_acquire);
549     while ( operation < current
550             && !__earliest_failure.compare_exchange_weak(current, operation, micron::memory_order_acq_rel, micron::memory_order_acquire) ) {
551     }
552   }

554   void __schedule(u32 operation) noexcept;
555 
556   void
557   __complete(u32 operation, compute_status status, bool dependency_failure) noexcept
558   {
559     const auto &node = __plan->__operations.data()[operation];
560     const bool failed = status != compute_status::ok;
561     if ( failed ) {
562       if ( !dependency_failure ) __publish_failure(operation, status);
563     } else {
564       __operation_status.data()[operation] = compute_status::ok;
565       for ( usize output = 0; output < node.output_count; ++output )
566         __runtime.data()[__plan->__outputs.data()[node.output_begin + output].value].valid = true;
567     }
568     for ( usize i = 0; i < node.dependent_count; ++i ) {
569       const u32 dependent = __plan->__dependents.data()[node.dependent_begin + i];
570       if ( failed ) __failed_dependency.data()[dependent].store(u8(1), micron::memory_order_release);
571       if ( __pending.data()[dependent].fetch_sub(u32(1), micron::memory_order_acq_rel) == 1 ) __schedule(dependent);
572     }
573     if ( __remaining.fetch_sub(u32(1), micron::memory_order_acq_rel) == 1 ) micron::wake_futex(__remaining.ptr(), 0x7fffffff);
574   }
575 
576   void
577   __run_operation(u32 operation) noexcept
578   {
579     if ( __failed_dependency.data()[operation].get(micron::memory_order_acquire) != 0 ) {
580       __complete(operation, compute_status::dependency_failed, true);
581       return;
582     }
583     const auto &node = __plan->__operations.data()[operation];
584     const auto status
585         = Ops::execute(node.operation, __context,
586                        micron::raw_slice<const tensor_view<dtype_type>>(__input_views.data() + node.input_begin, node.input_count),
587                        micron::raw_slice<tensor_view<dtype_type>>(__output_views.data() + node.output_begin, node.output_count));
588     __complete(operation, status, false);
589   }

615   void
616   __prepare_execution() noexcept
617   {
618     for ( usize tensor = 0; tensor < __plan->__tensors.size(); ++tensor ) {
619       const auto &meta = __plan->__tensors.data()[tensor];
620       if ( !meta.external && !meta.state ) __runtime.data()[tensor].valid = false;
621     }
622     __refresh_runtime_pointers();
623     for ( usize index = 0; index < __plan->__inputs.size(); ++index ) {
624       const auto tensor = __plan->__inputs.data()[index];
625       const auto &meta = __plan->__tensors.data()[tensor.value];
626       __input_views.data()[index] = { __runtime.data()[tensor.value].pointer, micron::addressof(meta.descriptor), meta.descriptor.access };
627     }
628     for ( usize index = 0; index < __plan->__outputs.size(); ++index ) {
629       const auto tensor = __plan->__outputs.data()[index];
630       const auto &meta = __plan->__tensors.data()[tensor.value];
631       __output_views.data()[index] = { __runtime.data()[tensor.value].pointer, micron::addressof(meta.descriptor), meta.descriptor.access };
632     }
633     for ( usize operation = 0; operation < __plan->__operations.size(); ++operation ) {
634       __pending.data()[operation].store(__plan->__operations.data()[operation].dependency_count, micron::memory_order_relaxed);
635       __failed_dependency.data()[operation].store(u8(0), micron::memory_order_relaxed);
636       __operation_status.data()[operation] = compute_status::ok;
637     }
638     __earliest_failure.store(operation_id::invalid_value(), micron::memory_order_relaxed);
639     __remaining.store(static_cast<u32>(__plan->__operations.size()), micron::memory_order_release);
640   }

678   [[nodiscard]] compute_status
679   bind_borrowed(tensor_id tensor, void *pointer, const descriptor_type &descriptor) noexcept
680   {
681     if ( __executing || !__valid_tensor(tensor) ) return compute_status::invalid_tensor;
682     const auto &meta = __plan->__tensors.data()[tensor.value];
683     descriptor_type normalized = descriptor;
684     ssize_t minimum{};
685     ssize_t maximum{};
686     if ( !__impl::__normalize_descriptor<Ops, Domains>(normalized, minimum, maximum) || !meta.external || meta.root != tensor
687          || !__impl::__descriptor_compatible(meta.descriptor, normalized) )
688       return meta.external ? compute_status::incompatible_binding : compute_status::invalid_binding;
689     if ( meta.descriptor.byte_size && !pointer ) return compute_status::invalid_binding;
690     if ( pointer && (reinterpret_cast<uintptr_t>(pointer) & (meta.descriptor.alignment - 1)) != 0 )
691       return compute_status::incompatible_binding;
692     auto &runtime = __runtime.data()[tensor.value];
693     runtime.storage = {};
694     runtime.pointer = pointer;
695     runtime.borrowed = true;
696     runtime.valid = true;
697     __refresh_runtime_pointers();
698     return compute_status::ok;
699   }

703   {
704     if ( __valid_tensor(tensor) && __plan->__tensors.data()[tensor.value].descriptor.access == tensor_access::read_write )
705       return compute_status::invalid_binding;
706     return bind_borrowed(tensor, const_cast<void *>(pointer), descriptor);
707   }

709   [[nodiscard]] compute_status
710   bind_shared(tensor_id tensor, const shared_tensor_type &binding) noexcept
711   {
712     if ( __executing || !__valid_tensor(tensor) || !binding ) return compute_status::invalid_binding;
713     const auto &meta = __plan->__tensors.data()[tensor.value];
714     descriptor_type normalized = binding.descriptor;
715     ssize_t minimum{};
716     ssize_t maximum{};
717     if ( meta.root != tensor || (!meta.external && !meta.state) ) return compute_status::invalid_binding;
718     if ( !__impl::__normalize_descriptor<Ops, Domains>(normalized, minimum, maximum)
719          || !__impl::__descriptor_compatible(meta.descriptor, normalized)
720          || binding.storage.domain() != static_cast<typename Domains::domain_type>(meta.descriptor.domain)
721          || binding.offset_bytes > binding.storage.size() )
722       return compute_status::incompatible_binding;
723     const usize origin = __logical_origin(normalized, minimum);
724     if ( origin > binding.offset_bytes || meta.descriptor.byte_size > binding.storage.size() - (binding.offset_bytes - origin) )
725       return compute_status::incompatible_binding;
726     void *pointer = const_cast<void *>(binding.data());
727     if ( pointer && (reinterpret_cast<uintptr_t>(pointer) & (meta.descriptor.alignment - 1)) != 0 )
728       return compute_status::incompatible_binding;
729     auto &runtime = __runtime.data()[tensor.value];
730     runtime.storage = binding.storage;
731     runtime.pointer = pointer;
732     runtime.borrowed = false;
733     runtime.valid = true;
734     __refresh_runtime_pointers();
735     return compute_status::ok;
736   }

742     const auto &meta = __plan->__tensors.data()[tensor.value];

744     return Domains::copy(static_cast<typename Domains::domain_type>(meta.descriptor.domain), __runtime.data()[tensor.value].pointer,
745                          source_domain, source, bytes);
746   }

769   [[nodiscard]] shared_tensor_type
770   export_tensor(tensor_id tensor) const noexcept
771   {
772     if ( __executing || !__valid_tensor(tensor) || !__runtime.data()[tensor.value].valid ) return {};
773     const auto &meta = __plan->__tensors.data()[tensor.value];
774     const auto &root_meta = __plan->__tensors.data()[meta.root.value];
775     const auto &runtime = __runtime.data()[tensor.value];
776     if ( runtime.borrowed || !runtime.storage || (!meta.retained && !root_meta.state) ) return {};
777     if ( meta.descriptor.byte_size == 0 ) return { runtime.storage, meta.descriptor, 0 };
778     const auto *base = reinterpret_cast<const byte *>(runtime.storage.data());
779     const auto *pointer = reinterpret_cast<const byte *>(runtime.pointer);
780     if ( pointer < base || static_cast<usize>(pointer - base) > runtime.storage.size() ) return {};
781     const usize offset = static_cast<usize>(pointer - base);
782     const usize origin = __logical_origin(meta.descriptor, meta.minimum_offset);
783     if ( origin > offset || meta.descriptor.byte_size > runtime.storage.size() - (offset - origin) ) return {};
784     return { runtime.storage, meta.descriptor, offset };
785   }

787   void
788   reset() noexcept
789   {
790     if ( __executing ) return;
791     for ( usize tensor = 0; tensor < __plan->__tensors.size(); ++tensor ) {
792       const auto &meta = __plan->__tensors.data()[tensor];
793       if ( !meta.external && !meta.state ) __runtime.data()[tensor].valid = false;
794     }
795   }
796 
797   void
798   hard_reset() noexcept
799   {
800     if ( __executing ) return;
801     for ( usize slot = 0; slot < __slot_storage.size(); ++slot )
802       if ( __slot_storage.data()[slot].size() )
803         micron::memset(__slot_storage.data()[slot].data(), byte(0), __slot_storage.data()[slot].size());
804     for ( usize tensor = 0; tensor < __plan->__tensors.size(); ++tensor ) {
805       const auto &meta = __plan->__tensors.data()[tensor];
806       auto &runtime = __runtime.data()[tensor];
807       if ( meta.root.value != tensor ) continue;
808       if ( meta.external ) {
809         runtime = {};
810       } else if ( meta.slot != micron::numeric_limits<u32>::max() ) {
811         runtime.storage = __slot_storage.data()[meta.slot];
812         runtime.pointer = runtime.storage.data()
813                               ? reinterpret_cast<byte *>(runtime.storage.data()) + __logical_origin(meta.descriptor, meta.minimum_offset)
814                               : nullptr;
815         runtime.borrowed = false;
816         runtime.valid = meta.state;
817       }
818     }
819     __refresh_runtime_pointers();
820     reset();
821   }

823   template<typename Executor>
824   [[nodiscard]] compute_result
825   execute(Executor &executor) noexcept
826     requires requires(Executor value, void (*function)(void *) noexcept, void *argument) {
827       { value.submit(function, argument) } -> micron::convertible_to<bool>;
828     }
829   {
830     static_assert(noexcept(executor.submit(static_cast<void (*)(void *) noexcept>(nullptr), nullptr)),
831                   "compute executors must provide a noexcept submit function");
832     if ( __executing ) return { compute_status::invalid_plan, operation_id::invalid() };
833     if ( !__bindings_ready() ) return { compute_status::missing_binding, operation_id::invalid() };
834     for ( const auto &slot : __plan->__slots )
835       if ( Domains::synchronize(slot.domain) != compute_status::ok ) return { compute_status::domain_failure, operation_id::invalid() };
836     __prepare_execution();
837     if ( __plan->__operations.empty() ) return {};
838     __executor = micron::addressof(executor);
839     __submit = &__submit_adapter<Executor>;
840     __executing = true;
841     for ( usize operation = 0; operation < __plan->__operations.size(); ++operation )
842       if ( __plan->__operations.data()[operation].dependency_count == 0 ) __schedule(static_cast<u32>(operation));
843     while ( __remaining.get(micron::memory_order_acquire) != 0 ) {
844       const u32 expected = __remaining.get(micron::memory_order_relaxed);
845       if ( expected != 0 ) micron::wait_futex(__remaining.ptr(), expected);
846     }
847     __executing = false;
848     __executor = nullptr;
849     __submit = nullptr;
850     const u32 failure = __earliest_failure.get(micron::memory_order_acquire);
851     if ( failure == operation_id::invalid_value() ) return {};
852     return { __operation_status.data()[failure], operation_id{ failure } };
853   }
854 
855   [[nodiscard]] compute_result
856   execute() noexcept
857   {
858     inline_executor executor;
859     return execute(executor);
860   }

936   add_input(const descriptor_type &descriptor)

939     const tensor_id id{ static_cast<u32>(__tensors.size()) };
940     __tensors.push_back({ descriptor, operation_id::invalid(), true, false, false });

950     const tensor_id id{ static_cast<u32>(__tensors.size()) };

955   [[nodiscard]] __impl::__compute_vec<tensor_id>
956   add_operation(operation_type operation, micron::raw_slice<const tensor_id> inputs, usize output_count)
957   {
958     if ( __operations.size() >= operation_id::invalid_value() ) __impl::__compute_plan_error("compute_graph: operation id overflow");
959     if ( output_count > tensor_id::invalid_value() - __tensors.size() ) __impl::__compute_plan_error("compute_graph: tensor id overflow");
960     for ( auto input : inputs )
961       if ( !__valid_tensor(input) ) __impl::__compute_plan_error("compute_graph: invalid operation input");
962     const operation_id producer{ static_cast<u32>(__operations.size()) };
963     builder_operation node{ micron::move(operation), {}, {} };
964     node.inputs.reserve(inputs.size());
965     for ( auto input : inputs ) node.inputs.push_back(input);
966     node.outputs.reserve(output_count);
967     for ( usize output = 0; output < output_count; ++output ) {
968       const tensor_id id{ static_cast<u32>(__tensors.size()) };
969       __tensors.push_back({ descriptor_type{}, producer, false, false, false });
970       node.outputs.push_back(id);
971     }
972     auto result = node.outputs;
973     __operations.push_back(micron::move(node));
974     return result;
975   }

1012   [[nodiscard]] plan_type
1013   finalize()
1014   {
1015     using operation_meta = __impl::__compute_operation_meta<operation_type>;
1016     using slot_meta = __impl::__compute_slot_meta<typename Domains::domain_type>;
1017 
1018     plan_type plan;
1019     plan.__tensors.resize(__tensors.size());
1020     for ( usize tensor = 0; tensor < __tensors.size(); ++tensor ) {
1021       const auto &source = __tensors.data()[tensor];
1022       auto &target = plan.__tensors.data()[tensor];
1023       target.descriptor = source.descriptor;
1024       target.producer = source.producer;
1025       target.root = tensor_id{ static_cast<u32>(tensor) };
1026       target.external = source.external;
1027       target.state = source.state;
1028       target.retained = source.retained;
1029       if ( !source.producer.valid()
1030            && !__impl::__normalize_descriptor<Ops, Domains>(target.descriptor, target.minimum_offset, target.maximum_offset) )
1031         __impl::__compute_plan_error("compute_graph: invalid input or state tensor descriptor");
1032     }
1033 
1034     __impl::__compute_vec<__impl::__compute_vec<u32>> consumers(__tensors.size());
1035     __impl::__compute_vec<__impl::__compute_vec<u32>> dependencies(__operations.size());
1036     __impl::__compute_vec<alias_declaration> aliases;
1037     __impl::__compute_vec<descriptor_type> input_descriptors;
1038     __impl::__compute_vec<descriptor_type> output_descriptors;
1039     plan.__operations.reserve(__operations.size());
1040 
1041     for ( usize operation = 0; operation < __operations.size(); ++operation ) {
1042       auto &builder = __operations.data()[operation];
1043       input_descriptors.clear();
1044       output_descriptors.clear();
1045       aliases.clear();
1046       input_descriptors.reserve(builder.inputs.size());
1047       output_descriptors.reserve(builder.outputs.size());
1048       aliases.resize(builder.outputs.size(), alias_declaration{});
1049       for ( auto input : builder.inputs ) {
1050         input_descriptors.push_back(plan.__tensors.data()[input.value].descriptor);
1051         consumers.data()[input.value].push_back(static_cast<u32>(operation));
1052         const auto producer = plan.__tensors.data()[input.value].producer;
1053         if ( producer.valid() ) __impl::__add_dependency(dependencies, operation, producer.value);
1054       }
1055       for ( auto output : builder.outputs ) output_descriptors.push_back(plan.__tensors.data()[output.value].descriptor);
1056 
1057       const auto input_slice = micron::raw_slice<const descriptor_type>(input_descriptors.data(), input_descriptors.size());
1058       auto output_slice = micron::raw_slice<descriptor_type>(output_descriptors.data(), output_descriptors.size());
1059       if ( Ops::infer(builder.operation, input_slice, output_slice) != compute_status::ok )
1060         __impl::__compute_plan_error("compute_graph: operation shape/type inference failed");
1061       for ( usize output = 0; output < builder.outputs.size(); ++output ) {
1062         auto &meta = plan.__tensors.data()[builder.outputs.data()[output].value];
1063         meta.descriptor = output_descriptors.data()[output];
1064         if ( !__impl::__normalize_descriptor<Ops, Domains>(meta.descriptor, meta.minimum_offset, meta.maximum_offset) )
1065           __impl::__compute_plan_error("compute_graph: invalid inferred output descriptor");
1066         output_descriptors.data()[output] = meta.descriptor;
1067       }
1068       if ( Ops::validate(builder.operation, input_slice,
1069                          micron::raw_slice<descriptor_type>(output_descriptors.data(), output_descriptors.size()))
1070            != compute_status::ok )
1071         __impl::__compute_plan_error("compute_graph: operation validation failed");
1072       if constexpr ( requires {
1073                        {
1074                          Ops::aliases(builder.operation, input_slice,
1075                                       micron::raw_slice<const descriptor_type>(output_descriptors.data(), output_descriptors.size()),
1076                                       micron::raw_slice<alias_declaration>(aliases.data(), aliases.size()))
1077                        } -> micron::same_as<compute_status>;
1078                      } ) {
1079         if ( Ops::aliases(builder.operation, input_slice,
1080                           micron::raw_slice<const descriptor_type>(output_descriptors.data(), output_descriptors.size()),
1081                           micron::raw_slice<alias_declaration>(aliases.data(), aliases.size()))
1082              != compute_status::ok )
1083           __impl::__compute_plan_error("compute_graph: operation alias validation failed");
1084       }
1085 
1086       const usize input_begin = plan.__inputs.size();
1087       const usize output_begin = plan.__outputs.size();
1088       for ( auto input : builder.inputs ) plan.__inputs.push_back(input);
1089       for ( usize output = 0; output < builder.outputs.size(); ++output ) {
1090         const auto id = builder.outputs.data()[output];
1091         auto &meta = plan.__tensors.data()[id.value];
1092         const auto declaration = aliases.data()[output];
1093         if ( declaration.mode != alias_mode::none ) {
1094           if ( declaration.input_index >= builder.inputs.size()
1095                || declaration.offset_bytes > static_cast<usize>(micron::numeric_limits<ssize_t>::max()) )
1096             __impl::__compute_plan_error("compute_graph: invalid alias declaration");
1097           const auto input_id = builder.inputs.data()[declaration.input_index];
1098           const auto &input = plan.__tensors.data()[input_id.value];
1099           ssize_t alias_min{};
1100           ssize_t alias_max{};
1101           const ssize_t offset = static_cast<ssize_t>(declaration.offset_bytes);
1102           if ( __builtin_add_overflow(offset, meta.minimum_offset, &alias_min)
1103                || __builtin_add_overflow(offset, meta.maximum_offset, &alias_max) || alias_min < input.minimum_offset
1104                || alias_max > input.maximum_offset )
1105             __impl::__compute_plan_error("compute_graph: alias view exceeds its input storage");
1106           if ( declaration.offset_bytes % meta.descriptor.alignment != 0 )
1107             __impl::__compute_plan_error("compute_graph: alias offset violates output alignment");
1108           if ( declaration.mode == alias_mode::destructive_in_place
1109                && (input.descriptor.access != tensor_access::read_write || meta.descriptor.access != tensor_access::read_write) )
1110             __impl::__compute_plan_error("compute_graph: destructive alias requires read-write tensors");
1111           if ( declaration.mode == alias_mode::read_only_view ) meta.descriptor.access = tensor_access::read_only;
1112           meta.alias = declaration.mode;
1113           meta.alias_input = declaration.input_index;
1114           meta.root = input.root;
1115           if ( __builtin_add_overflow(input.root_offset, declaration.offset_bytes, &meta.root_offset) )
1116             __impl::__compute_plan_error("compute_graph: alias offset overflow");
1117         }
1118         plan.__outputs.push_back(id);
1119       }
1120       plan.__operations.push_back(operation_meta{ micron::move(builder.operation), input_begin, builder.inputs.size(), output_begin,
1121                                                   builder.outputs.size(), 0, 0, 0 });
1122     }
1123 
1124     __impl::__compute_vec<__impl::__compute_vec<u32>> root_consumers(__tensors.size());
1125     __impl::__compute_vec<usize> root_last(__tensors.size(), usize(0));
1126     __impl::__compute_vec<u8> root_retained(__tensors.size(), u8(0));
1127     for ( usize tensor = 0; tensor < plan.__tensors.size(); ++tensor ) {
1128       auto &meta = plan.__tensors.data()[tensor];
1129       const usize root = meta.root.value;
1130       if ( meta.retained || meta.state ) root_retained.data()[root] = 1;
1131       if ( meta.producer.valid() && root_last.data()[root] < meta.producer.value ) root_last.data()[root] = meta.producer.value;
1132       for ( u32 consumer : consumers.data()[tensor] ) {
1133         bool duplicate = false;
1134         for ( u32 existing : root_consumers.data()[root] ) duplicate |= existing == consumer;
1135         if ( !duplicate ) root_consumers.data()[root].push_back(consumer);
1136         if ( root_last.data()[root] < consumer ) root_last.data()[root] = consumer;
1137       }
1138     }
1139 
1140     for ( usize operation = 0; operation < plan.__operations.size(); ++operation ) {
1141       const auto &node = plan.__operations.data()[operation];
1142       for ( usize output = 0; output < node.output_count; ++output ) {
1143         const auto output_id = plan.__outputs.data()[node.output_begin + output];
1144         const auto &output_meta = plan.__tensors.data()[output_id.value];
1145         if ( output_meta.alias != alias_mode::destructive_in_place ) continue;
1146         const auto input_id = plan.__inputs.data()[node.input_begin + output_meta.alias_input];
1147         const usize root = plan.__tensors.data()[input_id.value].root.value;
1148         for ( u32 consumer : root_consumers.data()[root] ) {
1149           if ( consumer == operation ) continue;
1150           __impl::__compute_vec<u8> seen(plan.__operations.size(), u8(0));
1151           __impl::__compute_vec<u32> stack;
1152           stack.push_back(consumer);
1153           bool follows_write = false;
1154           while ( !stack.empty() && !follows_write ) {
1155             const u32 current = stack.data()[stack.size() - 1];
1156             stack.pop_back();
1157             if ( current == operation ) {
1158               follows_write = true;
1159               break;
1160             }
1161             if ( seen.data()[current] ) continue;
1162             seen.data()[current] = 1;
1163             for ( u32 dependency : dependencies.data()[current] ) stack.push_back(dependency);
1164           }
1165           if ( !follows_write ) __impl::__add_dependency(dependencies, operation, consumer);
1166         }
1167       }
1168     }
1169 
1170     for ( usize tensor = 0; tensor < plan.__tensors.size(); ++tensor ) {
1171       auto &meta = plan.__tensors.data()[tensor];
1172       if ( meta.root.value != tensor || meta.external ) continue;
1173       const bool dedicated = meta.state || root_retained.data()[tensor] != 0;
1174       const usize producer = meta.producer.valid() ? meta.producer.value : 0;
1175       const usize available = root_last.data()[tensor] < producer ? producer : root_last.data()[tensor];
1176       u32 selected = micron::numeric_limits<u32>::max();
1177       if ( !dedicated && meta.producer.valid() ) {
1178         for ( usize slot = 0; slot < plan.__slots.size(); ++slot ) {
1179           auto &candidate = plan.__slots.data()[slot];
1180           if ( candidate.dedicated || candidate.domain != static_cast<typename Domains::domain_type>(meta.descriptor.domain)
1181                || candidate.available_after >= producer )
1182             continue;
1183           selected = static_cast<u32>(slot);
1184           const usize previous_root = candidate.last_root.value;
1185           for ( u32 consumer : root_consumers.data()[previous_root] ) __impl::__add_dependency(dependencies, producer, consumer);
1186           const auto previous_producer = plan.__tensors.data()[previous_root].producer;
1187           if ( root_consumers.data()[previous_root].empty() && previous_producer.valid() )
1188             __impl::__add_dependency(dependencies, producer, previous_producer.value);
1189           if ( candidate.bytes < meta.descriptor.byte_size ) candidate.bytes = meta.descriptor.byte_size;
1190           if ( candidate.alignment < meta.descriptor.alignment ) candidate.alignment = meta.descriptor.alignment;
1191           candidate.available_after = available;
1192           candidate.last_root = tensor_id{ static_cast<u32>(tensor) };
1193           break;
1194         }
1195       }
1196       if ( selected == micron::numeric_limits<u32>::max() ) {
1197         if ( plan.__slots.size() >= micron::numeric_limits<u32>::max() )
1198           __impl::__compute_plan_error("compute_graph: arena slot id overflow");
1199         selected = static_cast<u32>(plan.__slots.size());
1200         plan.__slots.push_back(slot_meta{ static_cast<typename Domains::domain_type>(meta.descriptor.domain), meta.descriptor.byte_size,
1201                                           meta.descriptor.alignment, dedicated, tensor_id{ static_cast<u32>(tensor) }, available });
1202       }
1203       meta.slot = selected;
1204     }
1205     for ( usize tensor = 0; tensor < plan.__tensors.size(); ++tensor ) {
1206       auto &meta = plan.__tensors.data()[tensor];
1207       meta.retained = meta.retained || root_retained.data()[meta.root.value] != 0;
1208       if ( meta.root.value != tensor ) meta.slot = plan.__tensors.data()[meta.root.value].slot;
1209     }
1210 
1211     __impl::__compute_vec<__impl::__compute_vec<u32>> dependents(plan.__operations.size());
1212     __impl::__compute_vec<u32> indegree(plan.__operations.size(), u32(0));
1213     for ( usize operation = 0; operation < dependencies.size(); ++operation ) {
1214       indegree.data()[operation] = static_cast<u32>(dependencies.data()[operation].size());
1215       for ( u32 dependency : dependencies.data()[operation] ) dependents.data()[dependency].push_back(static_cast<u32>(operation));
1216     }
1217     __impl::__compute_vec<u32> queue;
1218     for ( usize operation = 0; operation < indegree.size(); ++operation )
1219       if ( indegree.data()[operation] == 0 ) queue.push_back(static_cast<u32>(operation));
1220     usize visited = 0;
1221     for ( usize head = 0; head < queue.size(); ++head ) {
1222       const u32 operation = queue.data()[head];
1223       ++visited;
1224       for ( u32 dependent : dependents.data()[operation] )
1225         if ( --indegree.data()[dependent] == 0 ) queue.push_back(dependent);
1226     }
1227     if ( visited != plan.__operations.size() ) __impl::__compute_plan_error("compute_graph: data or memory anti-dependencies form a cycle");
1228 
1229     for ( usize operation = 0; operation < plan.__operations.size(); ++operation ) {
1230       auto &meta = plan.__operations.data()[operation];
1231       meta.dependency_count = static_cast<u32>(dependencies.data()[operation].size());
1232       meta.dependent_begin = plan.__dependents.size();
1233       meta.dependent_count = dependents.data()[operation].size();
1234       for ( u32 dependent : dependents.data()[operation] ) plan.__dependents.push_back(dependent);
1235     }
1236     for ( const auto &slot : plan.__slots ) {
1237       if ( __builtin_add_overflow(plan.__peak_bytes, slot.bytes, &plan.__peak_bytes) )
1238         __impl::__compute_plan_error("compute_graph: arena peak byte count overflow");
1239     }
1240     return plan;
1241   }
1242 };

1272        || expected.alignment != actual.alignment || expected.dimensions.size() != actual.dimensions.size()
1273        || expected.strides.size() != actual.strides.size() )

1276     if ( expected.dimensions.data()[i] != actual.dimensions.data()[i] || expected.strides.data()[i] != actual.strides.data()[i] )

1281 template<typename Ops, domain_provider Domains>
1282 [[nodiscard]] bool
1283 __normalize_descriptor(tensor_descriptor<typename Ops::dtype_type> &descriptor, ssize_t &minimum, ssize_t &maximum) noexcept
1284 {
1285   static_assert(operation_registry<Ops>);
1286   using domain_type = typename Domains::domain_type;
1287   const domain_type domain = static_cast<domain_type>(descriptor.domain);
1288   if ( !Domains::valid(domain) ) return false;
1289   const usize element = static_cast<usize>(Ops::dtype_size(descriptor.dtype));
1290   if ( element == 0 || element > static_cast<usize>(micron::numeric_limits<ssize_t>::max()) ) return false;
1291   usize alignment = descriptor.alignment;
1292   const usize domain_alignment = Domains::minimum_alignment(domain);
1293   if ( alignment == 0 ) alignment = domain_alignment;
1294   if ( !__power_of_two(alignment) ) return false;
1295   descriptor.alignment = alignment;
1296 
1297   if ( descriptor.strides.empty() ) {
1298     descriptor.strides.resize(descriptor.dimensions.size(), ssize_t(0));
1299     usize stride = element;
1300     for ( usize i = descriptor.dimensions.size(); i != 0; --i ) {
1301       if ( stride > static_cast<usize>(micron::numeric_limits<ssize_t>::max()) ) return false;
1302       descriptor.strides.data()[i - 1] = static_cast<ssize_t>(stride);
1303       if ( descriptor.dimensions.data()[i - 1] != 0 && __builtin_mul_overflow(stride, descriptor.dimensions.data()[i - 1], &stride) )
1304         return false;
1305     }
1306   } else if ( descriptor.strides.size() != descriptor.dimensions.size() ) {
1307     return false;
1308   }
1309 
1310   minimum = 0;
1311   maximum = 0;
1312   bool empty = false;
1313   for ( usize i = 0; i < descriptor.dimensions.size(); ++i ) {
1314     const usize dimension = descriptor.dimensions.data()[i];
1315     if ( dimension == 0 ) {
1316       empty = true;
1317       continue;
1318     }
1319     if ( dimension - 1 > static_cast<usize>(micron::numeric_limits<ssize_t>::max()) ) return false;
1320     ssize_t delta{};
1321     if ( __builtin_mul_overflow(static_cast<ssize_t>(dimension - 1), descriptor.strides.data()[i], &delta) ) return false;
1322     if ( delta < 0 ) {
1323       if ( __builtin_add_overflow(minimum, delta, &minimum) ) return false;
1324     } else if ( __builtin_add_overflow(maximum, delta, &maximum) ) {
1325       return false;
1326     }
1327   }
1328   usize extent = 0;
1329   if ( !empty ) {
1330     ssize_t span{};
1331     if ( __builtin_sub_overflow(maximum, minimum, &span) || __builtin_add_overflow(span, static_cast<ssize_t>(element), &span) || span < 0 )
1332       return false;
1333     extent = static_cast<usize>(span);
1334   }
1335   if ( descriptor.byte_size != 0 && descriptor.byte_size != extent ) return false;
1336   descriptor.byte_size = extent;
1337   return true;
1338 }
