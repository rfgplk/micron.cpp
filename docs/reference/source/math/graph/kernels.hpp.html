64 inline void
65 expand_frontier(u64 *out, const u64 *frontier, const u64 *adjacency_rows, usize vertices) noexcept
66 {
67   const usize words = (vertices + 63u) / 64u;
68   for ( usize word = 0; word < words; ++word ) out[word] = 0;
69   for ( usize vertex = 0; vertex < vertices; ++vertex ) {
70     if ( (frontier[vertex >> 6u] & (u64(1) << (vertex & 63u))) == 0 ) continue;
71     const u64 *row = adjacency_rows + vertex * words;
72     set_union(out, out, row, words);
73   }
74 }

76 template<micron::integral I>
77 [[nodiscard]] inline usize
78 masked_slot_scan(const u64 *mask, usize slots, I *out, usize capacity) noexcept
79 {
80   usize count = 0;
81   const usize words = (slots + 63u) / 64u;
82   for ( usize word = 0; word < words; ++word ) {
83     u64 bits = mask[word];
84     while ( bits ) {
85       const usize bit = static_cast<usize>(__builtin_ctzll(bits));
86       const usize slot = word * 64u + bit;
87       if ( slot < slots && count < capacity ) out[count++] = static_cast<I>(slot);
88       bits &= bits - 1;
89     }
90   }
91   return count;
92 }

94 inline void
95 dense_boolean_product(const u8 *a, const u8 *b, u8 *out, usize n) noexcept
96 {
97   for ( usize i = 0; i < n; ++i ) {
98     for ( usize j = 0; j < n; ++j ) {
99       u8 value = 0;
100       for ( usize k = 0; k < n; ++k ) value |= static_cast<u8>(a[i * n + k] && b[k * n + j]);
101       out[i * n + j] = static_cast<u8>(value != 0);
102     }
103   }
104 }

106 template<micron::integral I>
107 [[nodiscard]] inline bool
108 csr_prefix_sum(const I *counts, I *outer, usize rows) noexcept
109 {
110   outer[0] = I(0);
111   for ( usize row = 0; row < rows; ++row ) {
112     I next{};
113     if ( __builtin_add_overflow(outer[row], counts[row], &next) ) return false;
114     outer[row + 1] = next;
115   }
116   return true;
117 }
