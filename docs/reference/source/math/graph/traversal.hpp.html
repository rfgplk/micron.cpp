19 template<micron::integral I> struct traversal_workspace {
20   using vertex_descriptor = vertex_id<I>;
21 
22   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> frontier;
23   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> from;
24   micron::vector<edge_id<I>, micron::allocator_serial<>, false> incoming;
25   micron::vector<u8, micron::allocator_serial<>, false> phase;
26   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> order;
27   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> parent;
28   micron::vector<usize, micron::allocator_serial<>, false> depth;
29   micron::vector<u8, micron::allocator_serial<>, false> discovered;
30 
31   void
32   reserve(usize slots, usize stack_entries = 0)
33   {
34     const usize entries = stack_entries > slots ? stack_entries : slots;
35     frontier.reserve(entries);
36     if ( stack_entries ) {
37       from.reserve(entries);
38       incoming.reserve(entries);
39       phase.reserve(entries);
40     }
41     order.reserve(slots);
42     parent.reserve(slots);
43     depth.reserve(slots);
44     discovered.reserve(slots);
45   }
46 
47   void
48   reset(usize slots)
49   {
50     frontier.clear();
51     from.clear();
52     incoming.clear();
53     phase.clear();
54     order.clear();
55     parent.resize(slots, vertex_descriptor::invalid());
56     parent.fill(vertex_descriptor::invalid());
57     depth.resize(slots, usize(0));
58     depth.fill(usize(0));
59     discovered.resize(slots, u8(0));
60     discovered.fill(u8(0));
61   }
62 };

64 template<micron::integral I> struct traversal_result {
65   algorithm_status status{ algorithm_status::ok };
66   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> order;
67   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> parent;
68   micron::vector<usize, micron::allocator_serial<>, false> depth;
69   micron::vector<u8, micron::allocator_serial<>, false> reached;
70 
71   [[nodiscard]] bool
72   contains(vertex_id<I> vertex) const noexcept
73   {
74     const usize slot = static_cast<usize>(vertex.value);
75     return vertex.valid() && slot < reached.size() && reached.data()[slot] != 0;
76   }
77 };

117 template<graph_model G, typename Visitor = null_visitor>
118 algorithm_status
119 bfs_into(const G &graph, typename G::vertex_descriptor source, traversal_workspace<typename G::index_type> &workspace, Visitor visitor = {})
120 {
121   using vertex_descriptor = typename G::vertex_descriptor;
122   if ( !graph.has_vertex(source) ) return algorithm_status::invalid_vertex;
123 
124   workspace.reset(graph.vertex_slots());
125   workspace.frontier.push_back(source);
126   workspace.discovered.data()[static_cast<usize>(source.value)] = 1;
127   workspace.parent.data()[static_cast<usize>(source.value)] = source;
128   if ( !__impl::discover(visitor, graph, source) ) return algorithm_status::ok;
129 
130   usize head = 0;
131   while ( head < workspace.frontier.size() ) {
132     const vertex_descriptor u = workspace.frontier.data()[head++];
133     workspace.order.push_back(u);
134     for ( auto edge : graph.out_edges(u) ) {
135       if ( !__impl::examine_edge(visitor, graph, edge) ) return algorithm_status::ok;
136     }
137     for ( auto v : graph.out_neighbors(u) ) {
138       const usize slot = static_cast<usize>(v.value);
139       if ( workspace.discovered.data()[slot] ) continue;
140       workspace.discovered.data()[slot] = 1;
141       workspace.parent.data()[slot] = u;
142       workspace.depth.data()[slot] = workspace.depth.data()[static_cast<usize>(u.value)] + 1;
143       workspace.frontier.push_back(v);
144       if ( !__impl::discover(visitor, graph, v) ) return algorithm_status::ok;
145     }
146     __impl::finish(visitor, graph, u);
147   }
148   return algorithm_status::ok;
149 }

151 template<graph_model G, typename Visitor = null_visitor>
152 [[nodiscard]] traversal_result<typename G::index_type>
153 bfs(const G &graph, typename G::vertex_descriptor source, Visitor visitor = {})
154 {
155   traversal_workspace<typename G::index_type> workspace;
156   workspace.reserve(graph.vertex_slots());
157   const algorithm_status status = bfs_into(graph, source, workspace, micron::move(visitor));
158   return { status, micron::move(workspace.order), micron::move(workspace.parent), micron::move(workspace.depth),
159            micron::move(workspace.discovered) };
160 }

169 template<graph_model G, typename Visitor = null_visitor>
170 algorithm_status
171 dfs_into(const G &graph, typename G::vertex_descriptor source, traversal_workspace<typename G::index_type> &workspace, Visitor visitor = {})
172 {
173   using vertex_descriptor = typename G::vertex_descriptor;
174   if ( !graph.has_vertex(source) ) return algorithm_status::invalid_vertex;
175   workspace.reset(graph.vertex_slots());
176 
177   workspace.frontier.push_back(source);
178   workspace.from.push_back(source);
179   workspace.incoming.push_back(edge_id<typename G::index_type>::invalid());
180   workspace.phase.push_back(0);
181 
182   while ( !workspace.frontier.empty() ) {
183     const usize top = workspace.frontier.size() - 1;
184     const vertex_descriptor u = workspace.frontier.data()[top];
185     const vertex_descriptor from = workspace.from.data()[top];
186     const auto incoming = workspace.incoming.data()[top];
187     const u8 phase = workspace.phase.data()[top];
188     workspace.frontier.pop_back();
189     workspace.from.pop_back();
190     workspace.incoming.pop_back();
191     workspace.phase.pop_back();
192 
193     if ( phase ) {
194       __impl::finish(visitor, graph, u);
195       continue;
196     }
197 
198     if ( incoming.valid() && !__impl::examine_edge(visitor, graph, incoming) ) return algorithm_status::ok;
199     const usize slot = static_cast<usize>(u.value);
200     if ( workspace.discovered.data()[slot] ) continue;
201     workspace.discovered.data()[slot] = 1;
202     workspace.parent.data()[slot] = from;
203     if ( u != source ) workspace.depth.data()[slot] = workspace.depth.data()[static_cast<usize>(from.value)] + 1;
204     if ( !__impl::discover(visitor, graph, u) ) return algorithm_status::ok;
205     workspace.order.push_back(u);
206 
207     workspace.frontier.push_back(u);
208     workspace.from.push_back(from);
209     workspace.incoming.push_back(edge_id<typename G::index_type>::invalid());
210     workspace.phase.push_back(1);
211     for ( auto edge : graph.out_edges(u) ) {
212       const vertex_descriptor v = G::is_directed ? graph.target(edge) : graph.opposite(edge, u);
213       workspace.frontier.push_back(v);
214       workspace.from.push_back(u);
215       workspace.incoming.push_back(edge);
216       workspace.phase.push_back(0);
217     }
218   }
219   return algorithm_status::ok;
220 }

222 template<graph_model G, typename Visitor = null_visitor>
223 [[nodiscard]] traversal_result<typename G::index_type>
224 dfs(const G &graph, typename G::vertex_descriptor source, Visitor visitor = {})
225 {
226   traversal_workspace<typename G::index_type> workspace;
227   workspace.reserve(graph.vertex_slots(), graph.edges_count() * (G::is_directed ? 1 : 2) + 1);
228   const algorithm_status status = dfs_into(graph, source, workspace, micron::move(visitor));
229   return { status, micron::move(workspace.order), micron::move(workspace.parent), micron::move(workspace.depth),
230            micron::move(workspace.discovered) };
231 }

240 template<graph_model G>
241 [[nodiscard]] bool
242 reachable(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target)
243 {
244   if ( !graph.has_vertex(source) || !graph.has_vertex(target) ) return false;
245   if ( source == target ) return true;
246   auto result = bfs(graph, source);
247   return result.contains(target);
248 }

258 template<graph_model G>
259 [[nodiscard]] auto
260 diffusion(const G &graph, typename G::vertex_descriptor source, usize rounds)
261 {
262   auto result = bfs(graph, source);
263   usize keep = 0;
264   while ( keep < result.order.size() && result.depth.data()[static_cast<usize>(result.order.data()[keep].value)] <= rounds ) ++keep;
265   result.order.resize(keep);
266   return result;
267 }

269 template<graph_model G, typename Rng>
270 [[nodiscard]] micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false>
271 random_walk(const G &graph, typename G::vertex_descriptor source, usize steps, Rng &&rng)
272 {
273   using vertex_descriptor = typename G::vertex_descriptor;
274   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> walk;
275   if ( !graph.has_vertex(source) ) return walk;
276   walk.reserve(steps + 1);
277   walk.push_back(source);
278   vertex_descriptor current = source;
279   for ( usize step = 0; step < steps; ++step ) {
280     const usize degree = graph.out_degree(current);
281     if ( degree == 0 ) break;
282     const usize selected = static_cast<usize>(micron::invoke(rng)) % degree;
283     usize at = 0;
284     vertex_descriptor next = current;
285     for ( auto neighbor : graph.out_neighbors(current) ) {
286       if ( at++ == selected ) {
287         next = neighbor;
288         break;
289       }
290     }
291     walk.push_back(next);
292     current = next;
293   }
294   return walk;
295 }
