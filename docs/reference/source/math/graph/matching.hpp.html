14 template<micron::integral I> struct matching_result {
15   algorithm_status status{ algorithm_status::ok };
16   usize cardinality{};
17   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> mate;
18   micron::vector<edge_id<I>, micron::allocator_serial<>, false> edges;
19 };

21 template<graph_model G>
22 [[nodiscard]] matching_result<typename G::index_type>
23 hopcroft_karp(const G &graph)
24 {
25   using vertex_descriptor = typename G::vertex_descriptor;
26   using I = typename G::index_type;
27   matching_result<I> result{ algorithm_status::ok,
28                              0,
29                              micron::vector<vertex_descriptor, micron::allocator_serial<>, false>(graph.vertex_slots(),
30                                                                                                   vertex_descriptor::invalid()),
31                              {} };
32   auto partition = bipartite_test(graph);
33   if ( !partition.bipartite ) {
34     result.status = algorithm_status::invalid_graph;
35     return result;
36   }
37   micron::vector<usize, micron::allocator_serial<>, false> distance(graph.vertex_slots(), usize(0));
38   micron::vector<u8, micron::allocator_serial<>, false> active(graph.vertex_slots(), u8(0));
39   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> queue;
40   queue.reserve(graph.vertices_count());
41 
42   auto each_neighbor = [&](vertex_descriptor u, auto &&fn) {
43     for ( auto v : graph.out_neighbors(u) ) fn(v);
44     if constexpr ( G::is_directed )
45       for ( auto v : graph.in_neighbors(u) ) fn(v);
46   };
47 
48   auto build_layers = [&]() {
49     queue.clear();
50     active.fill(u8(0));
51     for ( auto vertex : graph.vertices() ) {
52       const usize slot = static_cast<usize>(vertex.value);
53       if ( partition.color.data()[slot] == 0 && !result.mate.data()[slot].valid() ) {
54         active.data()[slot] = 1;
55         distance.data()[slot] = 0;
56         queue.push_back(vertex);
57       }
58     }
59     bool augmenting = false;
60     usize head = 0;
61     while ( head < queue.size() ) {
62       const vertex_descriptor u = queue.data()[head++];
63       each_neighbor(u, [&](vertex_descriptor v) {
64         const vertex_descriptor mate = result.mate.data()[static_cast<usize>(v.value)];
65         if ( !mate.valid() ) {
66           augmenting = true;
67         } else {
68           const usize ms = static_cast<usize>(mate.value);
69           if ( !active.data()[ms] ) {
70             active.data()[ms] = 1;
71             distance.data()[ms] = distance.data()[static_cast<usize>(u.value)] + 1;
72             queue.push_back(mate);
73           }
74         }
75       });
76     }
77     return augmenting;
78   };
79 
80   auto augment = [&](auto &&self, vertex_descriptor u) -> bool {
81     bool success = false;
82     each_neighbor(u, [&](vertex_descriptor v) {
83       if ( success ) return;
84       const vertex_descriptor mate = result.mate.data()[static_cast<usize>(v.value)];
85       if ( !mate.valid()
86            || (distance.data()[static_cast<usize>(mate.value)] == distance.data()[static_cast<usize>(u.value)] + 1 && self(self, mate)) ) {
87         result.mate.data()[static_cast<usize>(u.value)] = v;
88         result.mate.data()[static_cast<usize>(v.value)] = u;
89         success = true;
90       }
91     });
92     if ( !success ) active.data()[static_cast<usize>(u.value)] = 0;
93     return success;
94   };
95 
96   while ( build_layers() ) {
97     for ( auto vertex : graph.vertices() ) {
98       const usize slot = static_cast<usize>(vertex.value);
99       if ( partition.color.data()[slot] == 0 && !result.mate.data()[slot].valid() && augment(augment, vertex) ) ++result.cardinality;
100     }
101   }
102   result.edges.reserve(result.cardinality);
103   for ( auto vertex : graph.vertices() ) {
104     const usize slot = static_cast<usize>(vertex.value);
105     if ( partition.color.data()[slot] != 0 || !result.mate.data()[slot].valid() ) continue;
106     auto edge = graph.find_edge(vertex, result.mate.data()[slot]);
107     if ( !edge.valid() && G::is_directed ) edge = graph.find_edge(result.mate.data()[slot], vertex);
108     if ( edge.valid() ) result.edges.push_back(edge);
109   }
110   return result;
111 }

113 template<graph_model G>
114 [[nodiscard]] matching_result<typename G::index_type>
115 greedy_maximal_matching(const G &graph)
116 {
117   using vertex_descriptor = typename G::vertex_descriptor;
118   matching_result<typename G::index_type> result{ algorithm_status::ok,
119                                                   0,
120                                                   micron::vector<vertex_descriptor, micron::allocator_serial<>, false>(
121                                                       graph.vertex_slots(), vertex_descriptor::invalid()),
122                                                   {} };
123   for ( auto edge : graph.edges() ) {
124     const usize u = static_cast<usize>(edge.source.value);
125     const usize v = static_cast<usize>(edge.target.value);
126     if ( edge.source == edge.target || result.mate.data()[u].valid() || result.mate.data()[v].valid() ) continue;
127     result.mate.data()[u] = edge.target;
128     result.mate.data()[v] = edge.source;
129     result.edges.push_back(edge.id);
130     ++result.cardinality;
131   }
132   return result;
133 }

135 template<graph_model G>
136 [[nodiscard]] bool
137 validate_matching(const G &graph, const matching_result<typename G::index_type> &matching)
138 {
139   micron::vector<u8, micron::allocator_serial<>, false> used(graph.vertex_slots(), u8(0));
140   for ( auto edge : matching.edges ) {
141     if ( !graph.has_edge(edge) ) return false;
142     const auto u = graph.source(edge);
143     const auto v = graph.target(edge);
144     if ( u == v || used.data()[static_cast<usize>(u.value)] || used.data()[static_cast<usize>(v.value)] ) return false;
145     used.data()[static_cast<usize>(u.value)] = 1;
146     used.data()[static_cast<usize>(v.value)] = 1;
147   }
148   return matching.edges.size() == matching.cardinality;
149 }

151 template<graph_model G>
152 [[nodiscard]] micron::vector<typename G::edge_descriptor, micron::allocator_serial<>, false>
153 minimum_edge_cover(const G &graph)
154 {
155   auto matching = hopcroft_karp(graph);
156   micron::vector<typename G::edge_descriptor, micron::allocator_serial<>, false> result = micron::move(matching.edges);
157   micron::vector<u8, micron::allocator_serial<>, false> covered(graph.vertex_slots(), u8(0));
158   for ( auto edge : result ) {
159     covered.data()[static_cast<usize>(graph.source(edge).value)] = 1;
160     covered.data()[static_cast<usize>(graph.target(edge).value)] = 1;
161   }
162   for ( auto vertex : graph.vertices() ) {
163     const usize slot = static_cast<usize>(vertex.value);
164     if ( covered.data()[slot] ) continue;
165     auto range = graph.out_edges(vertex);
166     auto iterator = range.begin();
167     if ( iterator == range.end() && G::is_directed ) {
168       auto incoming = graph.in_edges(vertex);
169       if ( incoming.begin() != incoming.end() ) {
170         const auto edge = *incoming.begin();
171         result.push_back(edge);
172         covered.data()[slot] = 1;
173         covered.data()[static_cast<usize>(graph.source(edge).value)] = 1;
174       }
175     } else if ( iterator != range.end() ) {
176       const auto edge = *iterator;
177       result.push_back(edge);
178       covered.data()[slot] = 1;
179       covered.data()[static_cast<usize>(graph.opposite(edge, vertex).value)] = 1;
180     }
181   }
182   return result;
183 }
