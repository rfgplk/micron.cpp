16 template<graph_model G>
17 [[nodiscard]] bool
18 has_cycle(const G &graph)
19 {
20   using vertex_descriptor = typename G::vertex_descriptor;
21   using edge_descriptor = typename G::edge_descriptor;
22   micron::vector<u8, micron::allocator_serial<>, false> color(graph.vertex_slots(), u8(0));
23   auto visit_directed = [&](auto &&self, vertex_descriptor u) -> bool {
24     color.data()[static_cast<usize>(u.value)] = 1;
25     for ( auto v : graph.out_neighbors(u) ) {
26       const u8 state = color.data()[static_cast<usize>(v.value)];
27       if ( state == 1 || (state == 0 && self(self, v)) ) return true;
28     }
29     color.data()[static_cast<usize>(u.value)] = 2;
30     return false;
31   };
32   auto visit_undirected = [&](auto &&self, vertex_descriptor u, edge_descriptor parent) -> bool {
33     color.data()[static_cast<usize>(u.value)] = 1;
34     for ( auto edge : graph.out_edges(u) ) {
35       if ( edge == parent ) continue;
36       const vertex_descriptor v = graph.opposite(edge, u);
37       if ( color.data()[static_cast<usize>(v.value)] || self(self, v, edge) ) return true;
38     }
39     return false;
40   };
41   for ( auto root : graph.vertices() ) {
42     if ( color.data()[static_cast<usize>(root.value)] ) continue;
43     if constexpr ( G::is_directed ) {
44       if ( visit_directed(visit_directed, root) ) return true;
45     } else if ( visit_undirected(visit_undirected, root, edge_descriptor::invalid()) ) {
46       return true;
47     }
48   }
49   return false;
50 }

59 template<graph_model G>
60 [[nodiscard]] euler_result<typename G::index_type>
61 eulerian_path(const G &graph)
62 {
63   using vertex_descriptor = typename G::vertex_descriptor;
64   using edge_descriptor = typename G::edge_descriptor;
65   euler_result<typename G::index_type> result;
66   if ( graph.edges_count() == 0 ) {
67     auto vertices = graph.vertices();
68     if ( vertices.begin() != vertices.end() ) result.vertices.push_back(*vertices.begin());
69     result.circuit = true;
70     return result;
71   }
72 
73   vertex_descriptor start = vertex_descriptor::invalid();
74   if constexpr ( G::is_directed ) {
75     usize plus = 0;
76     usize minus = 0;
77     for ( auto vertex : graph.vertices() ) {
78       const usize out = graph.out_degree(vertex);
79       const usize in = graph.in_degree(vertex);
80       if ( out == in + 1 ) {
81         ++plus;
82         start = vertex;
83       } else if ( in == out + 1 ) {
84         ++minus;
85       } else if ( in != out ) {
86         result.status = algorithm_status::invalid_graph;
87         return result;
88       }
89       if ( !start.valid() && out ) start = vertex;
90     }
91     if ( !((plus == 0 && minus == 0) || (plus == 1 && minus == 1)) ) {
92       result.status = algorithm_status::invalid_graph;
93       return result;
94     }
95     result.circuit = plus == 0;
96   } else {
97     usize odd = 0;
98     for ( auto vertex : graph.vertices() ) {
99       if ( graph.degree(vertex) & 1u ) {
100         ++odd;
101         start = vertex;
102       } else if ( !start.valid() && graph.degree(vertex) ) {
103         start = vertex;
104       }
105     }
106     if ( odd != 0 && odd != 2 ) {
107       result.status = algorithm_status::invalid_graph;
108       return result;
109     }
110     result.circuit = odd == 0;
111   }
112 
113   micron::vector<u8, micron::allocator_serial<>, false> used(graph.edge_slots(), u8(0));
114   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> vertex_stack;
115   micron::vector<edge_descriptor, micron::allocator_serial<>, false> edge_stack;
116   vertex_stack.push_back(start);
117   while ( !vertex_stack.empty() ) {
118     const vertex_descriptor u = vertex_stack.data()[vertex_stack.size() - 1];
119     edge_descriptor next_edge = edge_descriptor::invalid();
120     for ( auto edge : graph.out_edges(u) )
121       if ( !used.data()[static_cast<usize>(edge.value)] ) {
122         next_edge = edge;
123         break;
124       }
125     if ( next_edge.valid() ) {
126       used.data()[static_cast<usize>(next_edge.value)] = 1;
127       const vertex_descriptor v = G::is_directed ? graph.target(next_edge) : graph.opposite(next_edge, u);
128       vertex_stack.push_back(v);
129       edge_stack.push_back(next_edge);
130     } else {
131       result.vertices.push_back(u);
132       vertex_stack.pop_back();
133       if ( !edge_stack.empty() ) {
134         result.edges.push_back(edge_stack.data()[edge_stack.size() - 1]);
135         edge_stack.pop_back();
136       }
137     }
138   }
139   if ( result.edges.size() != graph.edges_count() ) {
140     result.status = algorithm_status::disconnected;
141     result.vertices.clear();
142     result.edges.clear();
143     return result;
144   }
145   for ( usize a = 0, b = result.vertices.size() ? result.vertices.size() - 1 : 0; a < b; ++a, --b )
146     micron::swap(result.vertices.data()[a], result.vertices.data()[b]);
147   for ( usize a = 0, b = result.edges.size() ? result.edges.size() - 1 : 0; a < b; ++a, --b )
148     micron::swap(result.edges.data()[a], result.edges.data()[b]);
149   return result;
150 }

154 is_eulerian(const G &graph)

156   auto result = eulerian_path(graph);
157   return result.status == algorithm_status::ok && result.circuit;

160 template<graph_model G>
161 [[nodiscard]] topology_graph_t<G>
162 transitive_closure(const G &graph)
163 {
164   auto result = topology_vertices(graph);
165   for ( auto source : graph.vertices() ) {
166     auto reached = bfs(graph, source);
167     for ( auto target : reached.order ) {
168       if ( source == target && !topology_graph_t<G>::allows_loops ) continue;
169       (void)result.add_edge(source, target);
170     }
171   }
172   return result;
173 }

175 template<graph_model G>
176 [[nodiscard]] topology_graph_t<G>
177 transitive_reduction(const G &graph, algorithm_status *status = nullptr)
178 {
179   auto result = topology_vertices(graph);
180   auto order = topological_sort(graph);
181   if ( order.status != algorithm_status::ok ) {
182     if ( status ) *status = algorithm_status::not_a_dag;
183     return result;
184   }
185   micron::vector<u8, micron::allocator_serial<>, false> seen(graph.vertex_slots(), u8(0));
186   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> queue;
187   for ( auto candidate : graph.edges() ) {
188     seen.fill(u8(0));
189     queue.clear();
190     queue.push_back(candidate.source);
191     seen.data()[static_cast<usize>(candidate.source.value)] = 1;
192     usize head = 0;
193     bool alternate = false;
194     while ( head < queue.size() && !alternate ) {
195       const auto u = queue.data()[head++];
196       for ( auto edge : graph.out_edges(u) ) {
197         if ( edge == candidate.id ) continue;
198         const auto v = graph.target(edge);
199         if ( v == candidate.target ) {
200           alternate = true;
201           break;
202         }
203         const usize slot = static_cast<usize>(v.value);
204         if ( !seen.data()[slot] ) {
205           seen.data()[slot] = 1;
206           queue.push_back(v);
207         }
208       }
209     }
210     if ( !alternate ) (void)result.add_edge(candidate.source, candidate.target);
211   }
212   if ( status ) *status = algorithm_status::ok;
213   return result;
214 }

216 template<graph_model G>
217 [[nodiscard]] bool
218 is_chordal(const G &graph)
219 {
220   using vertex_descriptor = typename G::vertex_descriptor;
221   if constexpr ( G::is_directed ) return false;
222   const usize n = graph.vertices_count();
223   micron::vector<usize, micron::allocator_serial<>, false> label(graph.vertex_slots(), usize(0));
224   micron::vector<usize, micron::allocator_serial<>, false> number(graph.vertex_slots(), usize(0));
225   micron::vector<u8, micron::allocator_serial<>, false> selected(graph.vertex_slots(), u8(0));
226   for ( usize step = n; step > 0; --step ) {
227     vertex_descriptor best = vertex_descriptor::invalid();
228     for ( auto vertex : graph.vertices() ) {
229       const usize slot = static_cast<usize>(vertex.value);
230       if ( selected.data()[slot] ) continue;
231       if ( !best.valid() || label.data()[static_cast<usize>(best.value)] < label.data()[slot] ) best = vertex;
232     }
233     const usize bs = static_cast<usize>(best.value);
234     selected.data()[bs] = 1;
235     number.data()[bs] = step;
236     for ( auto neighbor : graph.out_neighbors(best) )
237       if ( !selected.data()[static_cast<usize>(neighbor.value)] ) ++label.data()[static_cast<usize>(neighbor.value)];
238   }
239   for ( auto vertex : graph.vertices() ) {
240     vertex_descriptor parent = vertex_descriptor::invalid();
241     for ( auto neighbor : graph.out_neighbors(vertex) ) {
242       if ( number.data()[static_cast<usize>(neighbor.value)] <= number.data()[static_cast<usize>(vertex.value)] ) continue;
243       if ( !parent.valid() || number.data()[static_cast<usize>(neighbor.value)] < number.data()[static_cast<usize>(parent.value)] )
244         parent = neighbor;
245     }
246     if ( !parent.valid() ) continue;
247     for ( auto neighbor : graph.out_neighbors(vertex) ) {
248       if ( neighbor == parent || number.data()[static_cast<usize>(neighbor.value)] <= number.data()[static_cast<usize>(vertex.value)] )
249         continue;
250       if ( !graph.has_edge(parent, neighbor) ) return false;
251     }
252   }
253   return true;
254 }

256 template<typename Range>
257 [[nodiscard]] bool
258 is_graphical_degree_sequence(const Range &input)
259 {
260   micron::vector<usize, micron::allocator_serial<>, false> degrees;
261   for ( auto value : input ) degrees.push_back(static_cast<usize>(value));
262 
263   struct descending {
264     usize value;
265 
266     [[nodiscard]] bool
267     operator<(const descending &other) const
268     {
269       return value > other.value;
270     }
271   };
272 
273   while ( !degrees.empty() ) {
274     micron::vector<descending, micron::allocator_serial<>, false> sorted;
275     sorted.reserve(degrees.size());
276     for ( usize value : degrees )
277       if ( value ) sorted.push_back({ value });
278     if ( sorted.empty() ) return true;
279     sorted.sort();
280     const usize degree = sorted.data()[0].value;
281     if ( degree >= sorted.size() ) return false;
282     degrees.clear();
283     for ( usize i = 1; i < sorted.size(); ++i ) {
284       usize value = sorted.data()[i].value;
285       if ( i <= degree ) {
286         if ( value == 0 ) return false;
287         --value;
288       }
289       degrees.push_back(value);
290     }
291   }
292   return true;
293 }

297 [[nodiscard]] constexpr u64
298 mix_color(u64 value) noexcept
299 {
300   value ^= value >> 30;
301   value *= 0xbf58476d1ce4e5b9ull;
302   value ^= value >> 27;
303   value *= 0x94d049bb133111ebull;
304   return value ^ (value >> 31);

308 template<graph_model G>
309 [[nodiscard]] u64
310 weisfeiler_lehman_hash(const G &graph, usize iterations = 3)
311 {
312   micron::vector<u64, micron::allocator_serial<>, false> color(graph.vertex_slots(), u64(0));
313   micron::vector<u64, micron::allocator_serial<>, false> next(graph.vertex_slots(), u64(0));
314   for ( auto vertex : graph.vertices() ) color.data()[static_cast<usize>(vertex.value)] = __impl::mix_color(graph.degree(vertex) + 1);
315   for ( usize iteration = 0; iteration < iterations; ++iteration ) {
316     for ( auto vertex : graph.vertices() ) {
317       micron::vector<u64, micron::allocator_serial<>, false> neighbors;
318       neighbors.reserve(graph.degree(vertex));
319       for ( auto neighbor : graph.out_neighbors(vertex) ) neighbors.push_back(color.data()[static_cast<usize>(neighbor.value)]);
320       if constexpr ( G::is_directed )
321         for ( auto neighbor : graph.in_neighbors(vertex) )
322           neighbors.push_back(__impl::mix_color(color.data()[static_cast<usize>(neighbor.value)] ^ 0x9e3779b97f4a7c15ull));
323       neighbors.sort();
324       u64 hash = __impl::mix_color(color.data()[static_cast<usize>(vertex.value)] ^ static_cast<u64>(neighbors.size()));
325       for ( u64 value : neighbors ) hash = __impl::mix_color(hash ^ value);
326       next.data()[static_cast<usize>(vertex.value)] = hash;
327     }
328     color.swap(next);
329   }
330   micron::vector<u64, micron::allocator_serial<>, false> sorted;
331   sorted.reserve(graph.vertices_count());
332   for ( auto vertex : graph.vertices() ) sorted.push_back(color.data()[static_cast<usize>(vertex.value)]);
333   sorted.sort();
334   u64 result = __impl::mix_color(static_cast<u64>(graph.vertices_count()) ^ (static_cast<u64>(graph.edges_count()) << 32u));
335   for ( u64 value : sorted ) result = __impl::mix_color(result ^ value);
336   return result;
337 }

339 template<graph_model A, graph_model B>
340 [[nodiscard]] bool
341 trees_isomorphic(const A &a, const B &b)
342 {
343   if ( a.vertices_count() != b.vertices_count() || a.edges_count() != b.edges_count() ) return false;
344   if ( a.vertices_count() && (a.edges_count() + 1 != a.vertices_count() || !is_connected(a) || !is_connected(b)) ) return false;
345   return weisfeiler_lehman_hash(a, a.vertices_count()) == weisfeiler_lehman_hash(b, b.vertices_count());
346 }

348 template<micron::integral I> struct dominator_result {
349   algorithm_status status{ algorithm_status::ok };
350   usize words{};
351   micron::vector<u64, micron::allocator_serial<>, false> sets;
352   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> immediate;
353 
354   [[nodiscard]] bool
355   dominates(vertex_id<I> dominator, vertex_id<I> vertex) const noexcept
356   {
357     return (sets.data()[static_cast<usize>(vertex.value) * words + (static_cast<usize>(dominator.value) >> 6u)]
358             & (u64(1) << (static_cast<usize>(dominator.value) & 63u)))
359            != 0;
360   }
361 };

363 template<graph_model G>
364 [[nodiscard]] dominator_result<typename G::index_type>
365 dominators(const G &graph, typename G::vertex_descriptor start)
366 {
367   using vertex_descriptor = typename G::vertex_descriptor;
368   dominator_result<typename G::index_type> result;
369   if ( !graph.has_vertex(start) ) {
370     result.status = algorithm_status::invalid_vertex;
371     return result;
372   }
373   result.words = (graph.vertex_slots() + 63u) / 64u;
374   result.sets = micron::vector<u64, micron::allocator_serial<>, false>(graph.vertex_slots() * result.words, u64(0));
375   result.immediate
376       = micron::vector<vertex_descriptor, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_descriptor::invalid());
377   micron::vector<u64, micron::allocator_serial<>, false> universe(result.words, u64(0));
378   for ( auto vertex : graph.vertices() )
379     universe.data()[static_cast<usize>(vertex.value) >> 6u] |= u64(1) << (static_cast<usize>(vertex.value) & 63u);
380   for ( auto vertex : graph.vertices() ) {
381     u64 *set = result.sets.data() + static_cast<usize>(vertex.value) * result.words;
382     if ( vertex == start )
383       set[static_cast<usize>(start.value) >> 6u] |= u64(1) << (static_cast<usize>(start.value) & 63u);
384     else
385       for ( usize word = 0; word < result.words; ++word ) set[word] = universe.data()[word];
386   }
387   bool changed = true;
388   while ( changed ) {
389     changed = false;
390     for ( auto vertex : graph.vertices() ) {
391       if ( vertex == start ) continue;
392       micron::vector<u64, micron::allocator_serial<>, false> intersection = universe;
393       bool have_predecessor = false;
394       for ( auto predecessor : graph.in_neighbors(vertex) ) {
395         have_predecessor = true;
396         const u64 *set = result.sets.data() + static_cast<usize>(predecessor.value) * result.words;
397         for ( usize word = 0; word < result.words; ++word ) intersection.data()[word] &= set[word];
398       }
399       if ( !have_predecessor ) intersection.fill(u64(0));
400       intersection.data()[static_cast<usize>(vertex.value) >> 6u] |= u64(1) << (static_cast<usize>(vertex.value) & 63u);
401       u64 *current = result.sets.data() + static_cast<usize>(vertex.value) * result.words;
402       for ( usize word = 0; word < result.words; ++word )
403         if ( current[word] != intersection.data()[word] ) {
404           current[word] = intersection.data()[word];
405           changed = true;
406         }
407     }
408   }
409   result.immediate.data()[static_cast<usize>(start.value)] = start;
410   for ( auto vertex : graph.vertices() ) {
411     if ( vertex == start ) continue;
412     vertex_descriptor best = vertex_descriptor::invalid();
413     for ( auto candidate : graph.vertices() ) {
414       if ( candidate == vertex || !result.dominates(candidate, vertex) ) continue;
415       if ( !best.valid() || result.dominates(best, candidate) ) best = candidate;
416     }
417     result.immediate.data()[static_cast<usize>(vertex.value)] = best;
418   }
419   return result;
420 }
