62 template<typename Weight> struct blossom_score {
63   using number_type = micron::conditional_t<micron::is_integral_v<Weight>, int128_t, long double>;
64   int128_t cardinal{};
65   number_type weight{};
66 
67   friend constexpr bool operator==(const blossom_score &, const blossom_score &) noexcept = default;
68 
69   friend constexpr bool
70   operator<(const blossom_score &a, const blossom_score &b) noexcept
71   {
72     return a.cardinal < b.cardinal || (a.cardinal == b.cardinal && a.weight < b.weight);
73   }
74 
75   friend constexpr bool
76   operator>(const blossom_score &a, const blossom_score &b) noexcept
77   {
78     return b < a;
79   }
80 
81   friend constexpr bool
82   operator<=(const blossom_score &a, const blossom_score &b) noexcept
83   {
84     return !(b < a);
85   }
86 
87   friend constexpr blossom_score
88   operator+(const blossom_score &a, const blossom_score &b) noexcept
89   {
90     return { a.cardinal + b.cardinal, a.weight + b.weight };
91   }
92 
93   friend constexpr blossom_score
94   operator-(const blossom_score &a, const blossom_score &b) noexcept
95   {
96     return { a.cardinal - b.cardinal, a.weight - b.weight };
97   }
98 
99   friend constexpr blossom_score
100   operator*(const blossom_score &a, int value) noexcept
101   {
102     return { a.cardinal * value, a.weight * static_cast<number_type>(value) };
103   }
104 
105   friend constexpr blossom_score
106   operator/(const blossom_score &a, int value) noexcept
107   {
108     return { a.cardinal / value, a.weight / static_cast<number_type>(value) };
109   }
110 
111   blossom_score &
112   operator+=(const blossom_score &other) noexcept
113   {
114     return *this = *this + other;
115   }
116 
117   blossom_score &
118   operator-=(const blossom_score &other) noexcept
119   {
120     return *this = *this - other;
121   }
122 };

193   void
194   update_slack(int u, int x)
195   {
196     if ( __slack.data()[x] == 0
197          || delta(g(static_cast<usize>(u), static_cast<usize>(x)))
198                 < delta(g(static_cast<usize>(__slack.data()[x]), static_cast<usize>(x))) )
199       __slack.data()[x] = u;
200   }

202   void
203   set_slack(int x)
204   {
205     __slack.data()[x] = 0;
206     for ( usize u = 1; u <= __n; ++u )
207       if ( edge_present(u, static_cast<usize>(x)) && __base.data()[u] != x && __state.data()[__base.data()[u]] == 0 )
208         update_slack(static_cast<int>(u), x);
209   }

211   void
212   queue_push(int x)
213   {
214     if ( x <= static_cast<int>(__n) ) {
215       __queue.push_back(x);
216       return;
217     }
218     for ( int vertex : __flower.data()[x] ) queue_push(vertex);

221   void
222   set_base(int x, int base)
223   {
224     __base.data()[x] = base;

226       for ( int vertex : __flower.data()[x] ) set_base(vertex, base);

229   int
230   flower_position(int blossom, int root)
231   {
232     auto &cycle = __flower.data()[blossom];
233     usize position = 0;
234     while ( position < cycle.size() && cycle.data()[position] != root ) ++position;
235     if ( position == cycle.size() ) return 0;
236     if ( position & 1u ) {
237       for ( usize left = 1, right = cycle.size() - 1; left < right; ++left, --right ) micron::swap(cycle.data()[left], cycle.data()[right]);
238       return static_cast<int>(cycle.size() - position);
239     }
240     return static_cast<int>(position);
241   }

243   void
244   rotate_flower(int blossom, usize position)
245   {
246     auto &cycle = __flower.data()[blossom];
247     if ( position == 0 || position >= cycle.size() ) return;
248     micron::vector<int, micron::allocator_serial<>, false> rotated;
249     rotated.reserve(cycle.size());
250     for ( usize i = position; i < cycle.size(); ++i ) rotated.push_back(cycle.data()[i]);
251     for ( usize i = 0; i < position; ++i ) rotated.push_back(cycle.data()[i]);
252     cycle = micron::move(rotated);
253   }

255   void
256   set_match(int u, int v)
257   {
258     const auto edge = g(static_cast<usize>(u), static_cast<usize>(v));
259     __match.data()[u] = edge_present(static_cast<usize>(u), static_cast<usize>(v)) ? edge.target : 0;
260     if ( u <= static_cast<int>(__n) ) return;
261     const int root = from(static_cast<usize>(u), static_cast<usize>(edge.source));
262     const int position = flower_position(u, root);
263     for ( int i = 0; i < position; ++i )
264       set_match(__flower.data()[u].data()[static_cast<usize>(i)], __flower.data()[u].data()[static_cast<usize>(i ^ 1)]);
265     set_match(root, v);
266     rotate_flower(u, static_cast<usize>(position));
267   }

269   void
270   augment(int u, int v)
271   {
272     for ( ;; ) {
273       const int next = __base.data()[__match.data()[u]];
274       set_match(u, v);
275       if ( next == 0 ) return;
276       set_match(next, __base.data()[__parent.data()[next]]);
277       u = __base.data()[__parent.data()[next]];
278       v = next;
279     }
280   }

282   int
283   lca(int u, int v)
284   {
285     ++__visit_token;
286     while ( u != 0 || v != 0 ) {
287       if ( u != 0 ) {
288         if ( __visited.data()[u] == __visit_token ) return u;
289         __visited.data()[u] = __visit_token;
290         u = __base.data()[__match.data()[u]];
291         if ( u != 0 ) u = __base.data()[__parent.data()[u]];
292       }
293       micron::swap(u, v);
294     }
295     return 0;
296   }

298   void
299   add_blossom(int u, int ancestor, int v)
300   {
301     int blossom = static_cast<int>(__n + 1);
302     while ( blossom <= static_cast<int>(__nodes) && __base.data()[blossom] != 0 ) ++blossom;
303     if ( blossom > static_cast<int>(__nodes) ) __nodes = static_cast<usize>(blossom);
304     __label.data()[blossom] = zero();
305     __state.data()[blossom] = 0;
306     __match.data()[blossom] = __match.data()[ancestor];
307     __flower.data()[blossom].clear();
308     __flower.data()[blossom].push_back(ancestor);
309     for ( int x = u, y = 0; x != ancestor; x = __base.data()[__parent.data()[y]] ) {
310       __flower.data()[blossom].push_back(x);
311       y = __base.data()[__match.data()[x]];
312       __flower.data()[blossom].push_back(y);
313       queue_push(y);
314     }
315     auto &cycle = __flower.data()[blossom];
316     for ( usize left = 1, right = cycle.size() - 1; left < right; ++left, --right ) micron::swap(cycle.data()[left], cycle.data()[right]);
317     for ( int x = v, y = 0; x != ancestor; x = __base.data()[__parent.data()[y]] ) {
318       cycle.push_back(x);
319       y = __base.data()[__match.data()[x]];
320       cycle.push_back(y);
321       queue_push(y);
322     }
323     set_base(blossom, blossom);
324     for ( usize x = 1; x < __stride; ++x ) {
325       g(static_cast<usize>(blossom), x) = { blossom, static_cast<int>(x), zero() };
326       g(x, static_cast<usize>(blossom)) = { static_cast<int>(x), blossom, zero() };
327       from(static_cast<usize>(blossom), x) = 0;
328     }
329     for ( int part : cycle ) {
330       for ( usize x = 1; x <= __nodes; ++x )
331         if ( !edge_present(static_cast<usize>(blossom), x)
332              || delta(g(static_cast<usize>(part), x)) < delta(g(static_cast<usize>(blossom), x)) ) {
333           g(static_cast<usize>(blossom), x) = g(static_cast<usize>(part), x);
334           g(x, static_cast<usize>(blossom)) = g(x, static_cast<usize>(part));
335         }
336       for ( usize x = 1; x <= __n; ++x )
337         if ( from(static_cast<usize>(part), x) != 0 ) from(static_cast<usize>(blossom), x) = part;
338     }
339     set_slack(blossom);
340   }

342   void
343   expand_blossom(int blossom)
344   {
345     for ( int part : __flower.data()[blossom] ) set_base(part, part);
346     const auto parent_edge = g(static_cast<usize>(blossom), static_cast<usize>(__parent.data()[blossom]));
347     const int root = from(static_cast<usize>(blossom), static_cast<usize>(parent_edge.source));
348     const int position = flower_position(blossom, root);
349     for ( int i = 0; i < position; i += 2 ) {
350       const int outer = __flower.data()[blossom].data()[static_cast<usize>(i)];
351       const int inner = __flower.data()[blossom].data()[static_cast<usize>(i + 1)];
352       __parent.data()[outer] = g(static_cast<usize>(inner), static_cast<usize>(outer)).source;
353       __state.data()[outer] = 1;
354       __state.data()[inner] = 0;
355       __slack.data()[outer] = 0;
356       set_slack(inner);
357       queue_push(inner);
358     }
359     __state.data()[root] = 1;
360     __parent.data()[root] = __parent.data()[blossom];
361     for ( usize i = static_cast<usize>(position + 1); i < __flower.data()[blossom].size(); ++i ) {
362       const int part = __flower.data()[blossom].data()[i];
363       __state.data()[part] = -1;
364       set_slack(part);
365     }
366     __base.data()[blossom] = 0;
367   }

369   bool
370   found_edge(const blossom_edge &edge)
371   {
372     const int u = __base.data()[edge.source];
373     const int v = __base.data()[edge.target];
374     if ( __state.data()[v] == -1 ) {
375       __parent.data()[v] = edge.source;
376       __state.data()[v] = 1;
377       const int next = __base.data()[__match.data()[v]];
378       __slack.data()[v] = 0;
379       __slack.data()[next] = 0;
380       __state.data()[next] = 0;
381       queue_push(next);
382     } else if ( __state.data()[v] == 0 ) {
383       const int ancestor = lca(u, v);
384       if ( ancestor == 0 ) {
385         augment(u, v);
386         augment(v, u);
387         return true;
388       }
389       add_blossom(u, ancestor, v);
390     }
391     return false;
392   }

394   bool
395   matching_step()
396   {
397     __state.fill(-1);
398     __slack.fill(0);
399     __queue.clear();
400     __queue_head = 0;
401     for ( usize x = 1; x <= __nodes; ++x )
402       if ( __base.data()[x] == static_cast<int>(x) && __match.data()[x] == 0 ) {
403         __parent.data()[x] = 0;
404         __state.data()[x] = 0;
405         queue_push(static_cast<int>(x));
406       }
407     if ( __queue.empty() ) return false;
408     for ( ;; ) {
409       while ( __queue_head < __queue.size() ) {
410         const int u = __queue.data()[__queue_head++];
411         if ( __state.data()[__base.data()[u]] == 1 ) continue;
412         for ( usize v = 1; v <= __n; ++v ) {
413           if ( !edge_present(static_cast<usize>(u), v) || __base.data()[u] == __base.data()[v] ) continue;
414           if ( delta(g(static_cast<usize>(u), v)) == zero() ) {
415             if ( found_edge(g(static_cast<usize>(u), v)) ) return true;
416           } else {
417             update_slack(u, __base.data()[v]);
418           }
419         }
420       }
421 
422       score adjustment{};
423       bool have_adjustment = false;
424       auto consider = [&](const score &candidate) {
425         if ( !have_adjustment || candidate < adjustment ) {
426           adjustment = candidate;
427           have_adjustment = true;
428         }
429       };
430       for ( usize blossom = __n + 1; blossom <= __nodes; ++blossom )
431         if ( __base.data()[blossom] == static_cast<int>(blossom) && __state.data()[blossom] == 1 ) consider(__label.data()[blossom] / 2);
432       for ( usize x = 1; x <= __nodes; ++x )
433         if ( __base.data()[x] == static_cast<int>(x) && __slack.data()[x] != 0 ) {
434           const score candidate = delta(g(static_cast<usize>(__slack.data()[x]), x));
435           if ( __state.data()[x] == -1 )
436             consider(candidate);
437           else if ( __state.data()[x] == 0 )
438             consider(candidate / 2);
439         }
440       if ( !have_adjustment ) return false;
441       for ( usize u = 1; u <= __n; ++u ) {
442         if ( __state.data()[__base.data()[u]] == 0 ) {
443           if ( __label.data()[u] <= adjustment ) return false;
444           __label.data()[u] -= adjustment;
445         } else if ( __state.data()[__base.data()[u]] == 1 ) {
446           __label.data()[u] += adjustment;
447         }
448       }
449       for ( usize blossom = __n + 1; blossom <= __nodes; ++blossom )
450         if ( __base.data()[blossom] == static_cast<int>(blossom) ) {
451           if ( __state.data()[blossom] == 0 )
452             __label.data()[blossom] += adjustment * 2;
453           else if ( __state.data()[blossom] == 1 )
454             __label.data()[blossom] -= adjustment * 2;
455         }
456       __queue.clear();
457       __queue_head = 0;
458       for ( usize x = 1; x <= __nodes; ++x )
459         if ( __base.data()[x] == static_cast<int>(x) && __slack.data()[x] != 0 && __base.data()[__slack.data()[x]] != static_cast<int>(x)
460              && delta(g(static_cast<usize>(__slack.data()[x]), x)) == zero() )
461           if ( found_edge(g(static_cast<usize>(__slack.data()[x]), x)) ) return true;
462       for ( usize blossom = __n + 1; blossom <= __nodes; ++blossom )
463         if ( __base.data()[blossom] == static_cast<int>(blossom) && __state.data()[blossom] == 1 && __label.data()[blossom] == zero() )
464           expand_blossom(static_cast<int>(blossom));
465     }
466   }

468 public:
469   explicit weighted_blossom_engine(usize vertices)
470       : __n(vertices), __nodes(vertices), __stride(vertices * 2 + 1), __edges(__stride * __stride, blossom_edge{}),
471         __label(__stride, zero()), __match(__stride, 0), __slack(__stride, 0), __base(__stride, 0), __parent(__stride, 0),
472         __state(__stride, -1), __visited(__stride, usize(0)), __flower(__stride), __flower_from(__stride * __stride, 0)
473   {
474     for ( usize u = 0; u < __stride; ++u )
475       for ( usize v = 0; v < __stride; ++v ) g(u, v) = { static_cast<int>(u), static_cast<int>(v), zero() };
476     for ( usize vertex = 1; vertex <= __n; ++vertex ) {
477       __base.data()[vertex] = static_cast<int>(vertex);
478       from(vertex, vertex) = static_cast<int>(vertex);
479     }
480   }

482   void
483   add_edge(usize u, usize v, const score &value)
484   {
485     if ( g(u, v).weight < value ) {
486       g(u, v) = { static_cast<int>(u), static_cast<int>(v), value };
487       g(v, u) = { static_cast<int>(v), static_cast<int>(u), value };
488     }
489   }

491   [[nodiscard]] const micron::vector<int, micron::allocator_serial<>, false> &
492   solve()
493   {
494     score maximum{};
495     for ( usize u = 1; u <= __n; ++u )
496       for ( usize v = 1; v <= __n; ++v )
497         if ( maximum < g(u, v).weight ) maximum = g(u, v).weight;
498     for ( usize u = 1; u <= __n; ++u ) __label.data()[u] = maximum;
499     while ( matching_step() ) {
500     }
501     return __match;
502   }

507 template<graph_model G, typename WeightMap>
508 [[nodiscard]] auto
509 maximum_weighted_matching(
510     const G &graph, WeightMap weight_map,
511     weighted_matching_workspace<typename G::index_type,
512                                 micron::remove_cvref_t<decltype(__impl::weight(micron::declval<WeightMap &>(), micron::declval<const G &>(),
513                                                                                micron::declval<typename G::edge_descriptor>()))>>
514         &workspace,
515     matching_objective objective = matching_objective::maximum_weight)
516 {
517   using I = typename G::index_type;
518   using W = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
519   weighted_matching_result<I, W> result{
520     algorithm_status::ok,
521     micron::vector<vertex_id<I>, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_id<I>::invalid()),
522     micron::vector<edge_id<I>, micron::allocator_serial<>, false>(graph.vertex_slots(), edge_id<I>::invalid()),
523     {},
524     0,
525     W{},
526     graph.vertices_count()
527   };
528   if constexpr ( G::is_directed ) {
529     result.status = algorithm_status::invalid_graph;
530     return result;
531   }
532   workspace.reserve(graph.vertex_slots(), graph.vertices_count());
533   workspace.dense_to_vertex.clear();
534   workspace.vertex_to_dense.resize(graph.vertex_slots(), vertex_id<I>::invalid_value());
535   workspace.vertex_to_dense.fill(vertex_id<I>::invalid_value());
536   for ( auto vertex : graph.vertices() ) {
537     workspace.vertex_to_dense.data()[static_cast<usize>(vertex.value)] = static_cast<I>(workspace.dense_to_vertex.size());
538     workspace.dense_to_vertex.push_back(vertex);
539   }
540   const usize n = workspace.dense_to_vertex.size();
541   __impl::weighted_blossom_engine<W> engine(n);
542   using score = __impl::blossom_score<W>;
543   workspace.chosen.resize(n * n, edge_id<I>::invalid());
544   workspace.chosen.fill(edge_id<I>::invalid());
545   workspace.chosen_weight.resize(n * n, W{});
546   workspace.chosen_weight.fill(W{});
547   auto &chosen = workspace.chosen;
548   auto &chosen_weight = workspace.chosen_weight;
549   for ( auto edge : graph.edges() ) {
550     const W weight = static_cast<W>(__impl::weight(weight_map, graph, edge.id));
551     if ( __impl::nonfinite_weight(weight) ) {
552       result.status = algorithm_status::invalid_weight;
553       return result;
554     }
555     if ( edge.source == edge.target ) continue;
556     const usize u = static_cast<usize>(workspace.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
557     const usize v = static_cast<usize>(workspace.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
558     const usize position = u * n + v;
559     if ( !chosen.data()[position].valid() || chosen_weight.data()[position] < weight
560          || (!(weight < chosen_weight.data()[position]) && !(chosen_weight.data()[position] < weight)
561              && edge.id < chosen.data()[position]) ) {
562       chosen.data()[position] = chosen.data()[v * n + u] = edge.id;
563       chosen_weight.data()[position] = chosen_weight.data()[v * n + u] = weight;
564     }
565   }
566   typename score::number_type cardinality_bonus{};
567   if ( objective == matching_objective::maximum_cardinality_then_weight ) {
568     typename score::number_type largest_magnitude{};
569     for ( usize u = 0; u < n; ++u )
570       for ( usize v = u + 1; v < n; ++v ) {
571         if ( !chosen.data()[u * n + v].valid() ) continue;
572         typename score::number_type value = static_cast<typename score::number_type>(chosen_weight.data()[u * n + v]);
573         if ( value < typename score::number_type{} ) value = -value;
574         if ( largest_magnitude < value ) largest_magnitude = value;
575       }
576     cardinality_bonus = largest_magnitude * static_cast<typename score::number_type>(n + 1) + static_cast<typename score::number_type>(1);
577   }
578   for ( usize u = 0; u < n; ++u )
579     for ( usize v = u + 1; v < n; ++v ) {
580       if ( !chosen.data()[u * n + v].valid() ) continue;
581       const W weight = chosen_weight.data()[u * n + v];
582       score value{};
583       if constexpr ( micron::is_integral_v<W> )
584         value.weight = (static_cast<int128_t>(weight) + cardinality_bonus) * 2;
585       else
586         value.weight = (static_cast<long double>(weight) + cardinality_bonus) * 2.0L;
587       if ( objective == matching_objective::maximum_weight && !(score{} < value) ) continue;
588       engine.add_edge(u + 1, v + 1, value);
589     }
590   const auto &match = engine.solve();
591   result.edges.reserve(n / 2);
592   for ( usize u = 1; u <= n; ++u ) {
593     const int raw_v = match.data()[u];
594     if ( raw_v <= 0 || raw_v > static_cast<int>(n) ) continue;
595     const usize v = static_cast<usize>(raw_v - 1);
596     const auto original_u = workspace.dense_to_vertex.data()[u - 1];
597     const auto original_v = workspace.dense_to_vertex.data()[v];
598     const edge_id<I> edge = chosen.data()[(u - 1) * n + v];
599     result.mate.data()[static_cast<usize>(original_u.value)] = original_v;
600     result.mate_edge.data()[static_cast<usize>(original_u.value)] = edge;
601     if ( u - 1 < v ) {
602       result.edges.push_back(edge);
603       ++result.cardinality;
604       W total{};
605       if ( !__impl::add_distance(result.total_weight, chosen_weight.data()[(u - 1) * n + v], total) ) {
606         result.status = algorithm_status::overflow;
607         return result;
608       }
609       result.total_weight = total;
610     }
611   }
612   return result;
613 }
