21 template<graph_model G>
22 [[nodiscard]] partition_result<typename G::index_type>
23 label_propagation(const G &graph, usize max_iterations = 100)
24 {
25   using I = typename G::index_type;
26   partition_result<I> result{ algorithm_status::ok, graph.vertices_count(), 0,
27                               micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_id<I>::invalid_value()) };
28   for ( auto vertex : graph.vertices() ) result.community.data()[static_cast<usize>(vertex.value)] = vertex.value;
29   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
30     bool changed = false;
31     for ( auto vertex : graph.vertices() ) {
32       I best = result.community.data()[static_cast<usize>(vertex.value)];
33       usize best_count = 0;
34       for ( auto candidate : graph.out_neighbors(vertex) ) {
35         const I label = result.community.data()[static_cast<usize>(candidate.value)];
36         usize count = 0;
37         for ( auto neighbor : graph.out_neighbors(vertex) )
38           if ( result.community.data()[static_cast<usize>(neighbor.value)] == label ) ++count;
39         if constexpr ( G::is_directed )
40           for ( auto neighbor : graph.in_neighbors(vertex) )
41             if ( result.community.data()[static_cast<usize>(neighbor.value)] == label ) ++count;
42         if ( count > best_count || (count == best_count && label < best) ) {
43           best = label;
44           best_count = count;
45         }
46       }
47       if constexpr ( G::is_directed ) {
48         for ( auto candidate : graph.in_neighbors(vertex) ) {
49           const I label = result.community.data()[static_cast<usize>(candidate.value)];
50           usize count = 0;
51           for ( auto neighbor : graph.out_neighbors(vertex) )
52             if ( result.community.data()[static_cast<usize>(neighbor.value)] == label ) ++count;
53           for ( auto neighbor : graph.in_neighbors(vertex) )
54             if ( result.community.data()[static_cast<usize>(neighbor.value)] == label ) ++count;
55           if ( count > best_count || (count == best_count && label < best) ) {
56             best = label;
57             best_count = count;
58           }
59         }
60       }
61       I &current = result.community.data()[static_cast<usize>(vertex.value)];
62       if ( current != best ) {
63         current = best;
64         changed = true;
65       }
66     }
67     result.iterations = iteration + 1;
68     if ( !changed ) break;
69     if ( iteration + 1 == max_iterations ) result.status = algorithm_status::non_convergent;
70   }
71   // Canonicalize sparse labels to [0,k).
72   micron::vector<I, micron::allocator_serial<>, false> labels;
73   for ( auto vertex : graph.vertices() ) {
74     const I label = result.community.data()[static_cast<usize>(vertex.value)];
75     bool found = false;
76     for ( I existing : labels )
77       if ( existing == label ) {
78         found = true;
79         break;
80       }
81     if ( !found ) labels.push_back(label);
82   }
83   for ( auto vertex : graph.vertices() ) {
84     I &label = result.community.data()[static_cast<usize>(vertex.value)];
85     for ( usize i = 0; i < labels.size(); ++i )
86       if ( labels.data()[i] == label ) {
87         label = static_cast<I>(i);
88         break;
89       }
90   }
91   result.communities = labels.size();
92   return result;
93 }

95 template<graph_model G, typename Partition>
96 [[nodiscard]] f64
97 modularity(const G &graph, const Partition &community, f64 resolution = 1.0)
98 {
99   const f64 m = static_cast<f64>(graph.edges_count());
100   if ( m == 0 ) return f64(0);
101   f64 score = 0;
102   for ( auto u : graph.vertices() ) {
103     for ( auto v : graph.vertices() ) {
104       if ( community[static_cast<usize>(u.value)] != community[static_cast<usize>(v.value)] ) continue;
105       const f64 adjacency = graph.has_edge(u, v) ? f64(1) : f64(0);
106       if constexpr ( G::is_directed )
107         score += adjacency - resolution * static_cast<f64>(graph.out_degree(u) * graph.in_degree(v)) / m;
108       else
109         score += adjacency - resolution * static_cast<f64>(graph.degree(u) * graph.degree(v)) / (f64(2) * m);
110     }
111   }
112   return score / (G::is_directed ? m : f64(2) * m);
113 }

115 template<graph_model G>
116 [[nodiscard]] auto
117 louvain(const G &graph, usize max_iterations = 100)
118 {
119   using I = typename G::index_type;
120   partition_result<I> result{ algorithm_status::ok, graph.vertices_count(), 0,
121                               micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_id<I>::invalid_value()) };
122   for ( auto vertex : graph.vertices() ) result.community.data()[static_cast<usize>(vertex.value)] = vertex.value;
123   if ( graph.edges_count() == 0 ) return result;
124   const f64 normalization = static_cast<f64>(graph.edges_count());
125   micron::vector<f64, micron::allocator_serial<>, false> out_total(graph.vertex_slots(), f64(0));
126   micron::vector<f64, micron::allocator_serial<>, false> in_total(graph.vertex_slots(), f64(0));
127   for ( auto vertex : graph.vertices() ) {
128     const usize slot = static_cast<usize>(vertex.value);
129     out_total.data()[slot] = static_cast<f64>(G::is_directed ? graph.out_degree(vertex) : graph.degree(vertex));
130     in_total.data()[slot] = static_cast<f64>(G::is_directed ? graph.in_degree(vertex) : graph.degree(vertex));
131   }
132 
133   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
134     bool changed = false;
135     for ( auto vertex : graph.vertices() ) {
136       const usize slot = static_cast<usize>(vertex.value);
137       const I original = result.community.data()[slot];
138       micron::vector<I, micron::allocator_serial<>, false> candidates;
139       micron::vector<f64, micron::allocator_serial<>, false> outgoing;
140       micron::vector<f64, micron::allocator_serial<>, false> incoming;
141       candidates.push_back(original);
142       outgoing.push_back(0);
143       incoming.push_back(0);
144       auto consider = [&](auto neighbor, bool is_outgoing) {
145         const I label = result.community.data()[static_cast<usize>(neighbor.value)];
146         usize index = 0;
147         while ( index < candidates.size() && candidates.data()[index] != label ) ++index;
148         if ( index == candidates.size() ) {
149           candidates.push_back(label);
150           outgoing.push_back(0);
151           incoming.push_back(0);
152         }
153         if ( is_outgoing )
154           outgoing.data()[index] += 1;
155         else
156           incoming.data()[index] += 1;
157       };
158       for ( auto neighbor : graph.out_neighbors(vertex) ) consider(neighbor, true);
159       if constexpr ( G::is_directed ) {
160         for ( auto neighbor : graph.in_neighbors(vertex) ) consider(neighbor, false);
161       } else {
162         for ( usize i = 0; i < outgoing.size(); ++i ) incoming.data()[i] = outgoing.data()[i];
163       }
164 
165       const f64 out_degree = static_cast<f64>(graph.out_degree(vertex));
166       const f64 in_degree = static_cast<f64>(G::is_directed ? graph.in_degree(vertex) : graph.degree(vertex));
167       out_total.data()[static_cast<usize>(original)] -= G::is_directed ? out_degree : in_degree;
168       in_total.data()[static_cast<usize>(original)] -= in_degree;
169 
170       I best = original;
171       f64 best_gain = -1.0;
172       bool have_gain = false;
173       for ( usize i = 0; i < candidates.size(); ++i ) {
174         const I candidate = candidates.data()[i];
175         f64 gain{};
176         if constexpr ( G::is_directed ) {
177           gain = (outgoing.data()[i] + incoming.data()[i]) / normalization
178                  - (out_degree * in_total.data()[static_cast<usize>(candidate)]
179                     + in_degree * out_total.data()[static_cast<usize>(candidate)])
180                        / (normalization * normalization);
181         } else {
182           gain = outgoing.data()[i] / normalization
183                  - in_degree * in_total.data()[static_cast<usize>(candidate)] / (f64(2) * normalization * normalization);
184         }
185         if ( !have_gain || gain > best_gain + 1e-12 || (gain >= best_gain - 1e-12 && candidate < best) ) {
186           have_gain = true;
187           best = candidate;
188           best_gain = gain;
189         }
190       }
191       result.community.data()[slot] = best;
192       out_total.data()[static_cast<usize>(best)] += G::is_directed ? out_degree : in_degree;
193       in_total.data()[static_cast<usize>(best)] += in_degree;
194       if ( best != original ) changed = true;
195     }
196     result.iterations = iteration + 1;
197     if ( !changed ) break;
198     if ( iteration + 1 == max_iterations ) result.status = algorithm_status::non_convergent;
199   }
200 
201   micron::vector<I, micron::allocator_serial<>, false> labels;
202   for ( auto vertex : graph.vertices() ) {
203     const I label = result.community.data()[static_cast<usize>(vertex.value)];
204     bool found = false;
205     for ( I existing : labels )
206       if ( existing == label ) {
207         found = true;
208         break;
209       }
210     if ( !found ) labels.push_back(label);
211   }
212   for ( auto vertex : graph.vertices() ) {
213     I &label = result.community.data()[static_cast<usize>(vertex.value)];
214     for ( usize i = 0; i < labels.size(); ++i )
215       if ( labels.data()[i] == label ) {
216         label = static_cast<I>(i);
217         break;
218       }
219   }
220   result.communities = labels.size();
221   return result;
222 }

224 template<graph_model G>
225 [[nodiscard]] auto
226 leiden(const G &graph, usize max_iterations = 100)
227 {
228   using I = typename G::index_type;
229   auto result = louvain(graph, max_iterations);
230   micron::vector<I, micron::allocator_serial<>, false> refined(graph.vertex_slots(), vertex_id<I>::invalid_value());
231   micron::vector<u8, micron::allocator_serial<>, false> seen(graph.vertex_slots(), u8(0));
232   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> queue;
233   queue.reserve(graph.vertices_count());
234   I next_label = 0;
235   for ( auto root : graph.vertices() ) {
236     const usize root_slot = static_cast<usize>(root.value);
237     if ( seen.data()[root_slot] ) continue;
238     const I old_label = result.community.data()[root_slot];
239     queue.clear();
240     queue.push_back(root);
241     seen.data()[root_slot] = 1;
242     refined.data()[root_slot] = next_label;
243     usize head = 0;
244     while ( head < queue.size() ) {
245       const auto vertex = queue.data()[head++];
246       auto inspect = [&](auto neighbor) {
247         const usize slot = static_cast<usize>(neighbor.value);
248         if ( seen.data()[slot] || result.community.data()[slot] != old_label ) return;
249         seen.data()[slot] = 1;
250         refined.data()[slot] = next_label;
251         queue.push_back(neighbor);
252       };
253       for ( auto neighbor : graph.out_neighbors(vertex) ) inspect(neighbor);
254       if constexpr ( G::is_directed )
255         for ( auto neighbor : graph.in_neighbors(vertex) ) inspect(neighbor);
256     }
257     ++next_label;
258   }
259   result.community = micron::move(refined);
260   result.communities = static_cast<usize>(next_label);
261   return result;
262 }

264 template<graph_model G>
265 [[nodiscard]] auto
266 greedy_modularity_communities(const G &graph, usize max_iterations = 100)
267 {
268   using I = typename G::index_type;
269   partition_result<I> result{ algorithm_status::ok, graph.vertices_count(), 0,
270                               micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_id<I>::invalid_value()) };
271   for ( auto vertex : graph.vertices() ) result.community.data()[static_cast<usize>(vertex.value)] = vertex.value;
272   if ( graph.edges_count() == 0 ) return result;
273   const f64 normalization = static_cast<f64>(graph.edges_count());
274 
275   for ( usize iteration = 0; iteration < max_iterations && result.communities > 1; ++iteration ) {
276     micron::vector<I, micron::allocator_serial<>, false> labels;
277     for ( auto vertex : graph.vertices() ) {
278       const I label = result.community.data()[static_cast<usize>(vertex.value)];
279       bool found = false;
280       for ( I existing : labels )
281         if ( existing == label ) {
282           found = true;
283           break;
284         }
285       if ( !found ) labels.push_back(label);
286     }
287 
288     const usize count = labels.size();
289     micron::vector<usize, micron::allocator_serial<>, false> label_index(graph.vertex_slots(), usize(-1));
290     for ( usize i = 0; i < count; ++i ) label_index.data()[static_cast<usize>(labels.data()[i])] = i;
291     micron::vector<f64, micron::allocator_serial<>, false> outgoing(count, f64(0));
292     micron::vector<f64, micron::allocator_serial<>, false> incoming(count, f64(0));
293     micron::vector<f64, micron::allocator_serial<>, false> between(count * count, f64(0));
294     for ( auto vertex : graph.vertices() ) {
295       const usize community = label_index.data()[static_cast<usize>(result.community.data()[static_cast<usize>(vertex.value)])];
296       outgoing.data()[community] += static_cast<f64>(G::is_directed ? graph.out_degree(vertex) : graph.degree(vertex));
297       incoming.data()[community] += static_cast<f64>(G::is_directed ? graph.in_degree(vertex) : graph.degree(vertex));
298     }
299     for ( auto edge : graph.edges() ) {
300       const usize a = label_index.data()[static_cast<usize>(result.community.data()[static_cast<usize>(edge.source.value)])];
301       const usize b = label_index.data()[static_cast<usize>(result.community.data()[static_cast<usize>(edge.target.value)])];
302       if ( a == b ) continue;
303       between.data()[a * count + b] += 1;
304       if constexpr ( !G::is_directed ) between.data()[b * count + a] += 1;
305     }
306 
307     bool improved = false;
308     I best_a{};
309     I best_b{};
310     f64 best_delta = 0;
311     for ( usize a = 0; a < count; ++a ) {
312       for ( usize b = a + 1; b < count; ++b ) {
313         f64 delta{};
314         if constexpr ( G::is_directed ) {
315           delta = (between.data()[a * count + b] + between.data()[b * count + a]) / normalization
316                   - (outgoing.data()[a] * incoming.data()[b] + outgoing.data()[b] * incoming.data()[a]) / (normalization * normalization);
317         } else {
318           delta = between.data()[a * count + b] / normalization
319                   - outgoing.data()[a] * outgoing.data()[b] / (f64(2) * normalization * normalization);
320         }
321         if ( delta > best_delta + 1e-12 ) {
322           improved = true;
323           best_delta = delta;
324           best_a = labels.data()[a];
325           best_b = labels.data()[b];
326         }
327       }
328     }
329     if ( !improved ) break;
330     for ( auto vertex : graph.vertices() ) {
331       I &label = result.community.data()[static_cast<usize>(vertex.value)];
332       if ( label == best_b ) label = best_a;
333     }
334     --result.communities;
335     result.iterations = iteration + 1;
336     if ( iteration + 1 == max_iterations && result.communities > 1 ) result.status = algorithm_status::non_convergent;
337   }
338   micron::vector<I, micron::allocator_serial<>, false> labels;
339   for ( auto vertex : graph.vertices() ) {
340     const I label = result.community.data()[static_cast<usize>(vertex.value)];
341     bool found = false;
342     for ( I existing : labels )
343       if ( existing == label ) {
344         found = true;
345         break;
346       }
347     if ( !found ) labels.push_back(label);
348   }
349   for ( auto vertex : graph.vertices() ) {
350     I &label = result.community.data()[static_cast<usize>(vertex.value)];
351     for ( usize i = 0; i < labels.size(); ++i )
352       if ( labels.data()[i] == label ) {
353         label = static_cast<I>(i);
354         break;
355       }
356   }
357   result.communities = labels.size();
358   return result;
359 }

361 template<graph_model G, typename Partition>
362 [[nodiscard]] f64
363 partition_coverage(const G &graph, const Partition &community)
364 {
365   if ( graph.edges_count() == 0 ) return f64(0);
366   usize internal = 0;
367   for ( auto edge : graph.edges() )
368     if ( community[static_cast<usize>(edge.source.value)] == community[static_cast<usize>(edge.target.value)] ) ++internal;
369   return static_cast<f64>(internal) / static_cast<f64>(graph.edges_count());
370 }
