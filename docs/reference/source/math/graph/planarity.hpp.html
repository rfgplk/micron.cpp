71 template<typename T, typename Key>
72 void
73 __stable_count_sort(__planar_vec<T> &values, __planar_vec<T> &temporary, usize key_count, Key key)
74 {
75   if ( values.size() < 2 ) return;
76   __planar_vec<usize> offsets(key_count + 1, usize(0));
77   for ( const auto &value : values ) ++offsets.data()[key(value) + 1];
78   for ( usize i = 1; i < offsets.size(); ++i ) offsets.data()[i] += offsets.data()[i - 1];
79   temporary.resize(values.size());
80   for ( const auto &value : values ) temporary.data()[offsets.data()[key(value)]++] = value;
81   values = micron::move(temporary);
82   temporary.clear();
83 }

85 template<graph_model G>
86 [[nodiscard]] auto
87 __make_planar_core(const G &graph)
88 {
89   using I = typename G::index_type;
90   auto mapping = __dense_vertex_mapping(graph);
91   __planar_core<I> core;
92   core.vertices = mapping.dense_to_vertex.size();
93   core.dense_to_vertex = micron::move(mapping.dense_to_vertex);
94   core.loops.resize(core.vertices);
95 
96   __planar_vec<__planar_input_edge<I>> input;
97   input.reserve(graph.edges_count());
98   for ( auto edge : graph.edges() ) {
99     const usize u = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
100     const usize v = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
101     if ( u == v ) {
102       core.loops.data()[u].push_back(edge.id);
103       continue;
104     }
105     input.push_back({ u < v ? u : v, u < v ? v : u, edge.id });
106   }
107   __planar_vec<__planar_input_edge<I>> temporary;
108   __stable_count_sort(input, temporary, core.vertices, [](const auto &edge) { return edge.target; });
109   __stable_count_sort(input, temporary, core.vertices, [](const auto &edge) { return edge.source; });
110 
111   for ( usize first = 0; first < input.size(); ) {
112     usize last = first + 1;
113     while ( last < input.size() && input.data()[last].source == input.data()[first].source
114             && input.data()[last].target == input.data()[first].target )
115       ++last;
116     const usize begin = core.originals.size();
117     for ( usize i = first; i < last; ++i ) core.originals.push_back(input.data()[i].original);
118     core.edges.push_back({ input.data()[first].source, input.data()[first].target, begin, last - first });
119     first = last;
120   }
121   return core;
122 }

174   void
175   __finish_orientation_edge(int edge)
176   {
177     auto &current = __edge.data()[static_cast<usize>(edge)];
178     current.nesting = current.lowpoint * 2 + (current.second_lowpoint < __height.data()[current.source] ? 1 : 0);
179     const int parent = __parent_edge.data()[current.source];
180     if ( parent == -1 ) return;
181     auto &up = __edge.data()[static_cast<usize>(parent)];
182     if ( current.lowpoint < up.lowpoint ) {
183       up.second_lowpoint = up.lowpoint < current.second_lowpoint ? up.lowpoint : current.second_lowpoint;
184       up.lowpoint = current.lowpoint;
185     } else if ( current.lowpoint > up.lowpoint ) {
186       if ( current.lowpoint < up.second_lowpoint ) up.second_lowpoint = current.lowpoint;
187     } else if ( current.second_lowpoint < up.second_lowpoint ) {
188       up.second_lowpoint = current.second_lowpoint;
189     }
190   }

192   void
193   __orient()
194   {
195     struct frame {
196       int vertex{};
197       usize next{};
198     };
199 
200     __planar_vec<frame> stack;
201     for ( usize root = 0; root < __n; ++root ) {
202       if ( __height.data()[root] != -1 ) continue;
203       __height.data()[root] = 0;
204       __roots.push_back(static_cast<int>(root));
205       stack.push_back({ static_cast<int>(root), 0 });
206       while ( !stack.empty() ) {
207         auto &top = stack.data()[stack.size() - 1];
208         const usize vertex = static_cast<usize>(top.vertex);
209         if ( top.next == __adjacency.data()[vertex].size() ) {
210           const int parent = __parent_edge.data()[vertex];
211           stack.pop_back();
212           if ( parent != -1 ) __finish_orientation_edge(parent);
213           continue;
214         }
215         const int core_edge = __adjacency.data()[vertex].data()[top.next++];
216         if ( __oriented_core.data()[static_cast<usize>(core_edge)] != -1 ) continue;
217         const usize a = __core_source.data()[static_cast<usize>(core_edge)];
218         const usize b = __core_target.data()[static_cast<usize>(core_edge)];
219         const usize target = a == vertex ? b : a;
220         const int oriented = static_cast<int>(__edge.size());
221         __oriented_core.data()[static_cast<usize>(core_edge)] = oriented;
222         __edge.push_back(
223             { static_cast<int>(vertex), static_cast<int>(target), __height.data()[vertex], __height.data()[vertex], 0, -1, 1, core_edge });
224         if ( __height.data()[target] == -1 ) {
225           __height.data()[target] = __height.data()[vertex] + 1;
226           __parent_edge.data()[target] = oriented;
227           stack.push_back({ static_cast<int>(target), 0 });
228         } else {
229           __edge.data()[static_cast<usize>(oriented)].lowpoint = __height.data()[target];
230           __finish_orientation_edge(oriented);
231         }
232       }
233     }

236   void
237   __build_ordered(bool signed_depth)
238   {
239     for ( auto &row : __ordered ) row.clear();
240     if ( __edge.empty() ) return;
241     const usize range = signed_depth ? __n * 4 + 3 : __n * 2 + 2;
242     const int offset = signed_depth ? static_cast<int>(__n * 2 + 1) : 0;
243     __planar_vec<int> order;
244     __planar_vec<int> temporary;
245     order.reserve(__edge.size());
246     for ( usize i = 0; i < __edge.size(); ++i ) order.push_back(static_cast<int>(i));
247     __stable_count_sort(order, temporary, range, [&](int edge) {
248       const auto &value = __edge.data()[static_cast<usize>(edge)];
249       const int depth = signed_depth ? value.side * value.nesting : value.nesting;
250       return static_cast<usize>(depth + offset);
251     });
252     __stable_count_sort(order, temporary, __n,
253                         [&](int edge) { return static_cast<usize>(__edge.data()[static_cast<usize>(edge)].source); });
254     for ( int edge : order ) __ordered.data()[static_cast<usize>(__edge.data()[static_cast<usize>(edge)].source)].push_back(edge);
255   }

264   [[nodiscard]] int
265   __lowest(const __lr_conflict_pair &pair) const noexcept
266   {
267     if ( pair.left.empty() ) return __edge.data()[static_cast<usize>(pair.right.low)].lowpoint;
268     if ( pair.right.empty() ) return __edge.data()[static_cast<usize>(pair.left.low)].lowpoint;
269     const int left = __edge.data()[static_cast<usize>(pair.left.low)].lowpoint;
270     const int right = __edge.data()[static_cast<usize>(pair.right.low)].lowpoint;
271     return left < right ? left : right;
272   }

274   [[nodiscard]] bool
275   __add_constraints(int current, int parent)
276   {
277     __lr_conflict_pair merged;
278     const usize bottom = __stack_bottom.data()[static_cast<usize>(current)];
279     while ( __conflicts.size() > bottom ) {
280       auto part = __conflicts.data()[__conflicts.size() - 1];
281       __conflicts.pop_back();
282       if ( !part.left.empty() ) micron::swap(part.left, part.right);
283       if ( !part.left.empty() ) return false;
284       if ( __edge.data()[static_cast<usize>(part.right.low)].lowpoint > __edge.data()[static_cast<usize>(parent)].lowpoint ) {
285         if ( merged.right.empty() )
286           merged.right = part.right;
287         else
288           __edge.data()[static_cast<usize>(merged.right.low)].reference = part.right.high;
289         merged.right.low = part.right.low;
290       } else {
291         __edge.data()[static_cast<usize>(part.right.low)].reference = __lowpoint_edge.data()[static_cast<usize>(parent)];
292       }
293     }
294     while ( !__conflicts.empty()
295             && (__conflicting(__conflicts.data()[__conflicts.size() - 1].left, current)
296                 || __conflicting(__conflicts.data()[__conflicts.size() - 1].right, current)) ) {
297       auto part = __conflicts.data()[__conflicts.size() - 1];
298       __conflicts.pop_back();
299       if ( __conflicting(part.right, current) ) micron::swap(part.left, part.right);
300       if ( __conflicting(part.right, current) ) return false;
301       if ( !merged.right.empty() ) __edge.data()[static_cast<usize>(merged.right.low)].reference = part.right.high;
302       if ( !part.right.empty() ) merged.right.low = part.right.low;
303       if ( merged.left.empty() )
304         merged.left = part.left;
305       else if ( !merged.left.empty() )
306         __edge.data()[static_cast<usize>(merged.left.low)].reference = part.left.high;
307       merged.left.low = part.left.low;
308     }
309     if ( !merged.left.empty() || !merged.right.empty() ) __conflicts.push_back(merged);
310     return true;
311   }

313   void
314   __remove_back_edges(int parent)
315   {
316     const int vertex = __edge.data()[static_cast<usize>(parent)].source;
317     while ( !__conflicts.empty() && __lowest(__conflicts.data()[__conflicts.size() - 1]) == __height.data()[vertex] ) {
318       auto part = __conflicts.data()[__conflicts.size() - 1];
319       __conflicts.pop_back();
320       if ( !part.left.empty() ) __edge.data()[static_cast<usize>(part.left.low)].side = -1;
321     }
322     if ( !__conflicts.empty() ) {
323       auto part = __conflicts.data()[__conflicts.size() - 1];
324       __conflicts.pop_back();
325       while ( part.left.high != -1 && __edge.data()[static_cast<usize>(part.left.high)].target == vertex )
326         part.left.high = __edge.data()[static_cast<usize>(part.left.high)].reference;
327       if ( part.left.high == -1 && !part.left.empty() ) {
328         __edge.data()[static_cast<usize>(part.left.low)].reference = part.right.low;
329         __edge.data()[static_cast<usize>(part.left.low)].side = -1;
330         part.left.low = -1;
331       }
332       while ( part.right.high != -1 && __edge.data()[static_cast<usize>(part.right.high)].target == vertex )
333         part.right.high = __edge.data()[static_cast<usize>(part.right.high)].reference;
334       if ( part.right.high == -1 && !part.right.empty() ) {
335         __edge.data()[static_cast<usize>(part.right.low)].reference = part.left.low;
336         __edge.data()[static_cast<usize>(part.right.low)].side = -1;
337         part.right.low = -1;
338       }
339       __conflicts.push_back(part);
340     }
341     if ( __edge.data()[static_cast<usize>(parent)].lowpoint < __height.data()[vertex] ) {
342       const auto &part = __conflicts.data()[__conflicts.size() - 1];
343       const int left = part.left.high;
344       const int right = part.right.high;
345       if ( left != -1
346            && (right == -1 || __edge.data()[static_cast<usize>(left)].lowpoint > __edge.data()[static_cast<usize>(right)].lowpoint) )
347         __edge.data()[static_cast<usize>(parent)].reference = left;
348       else
349         __edge.data()[static_cast<usize>(parent)].reference = right;
350     }
351   }

353   [[nodiscard]] bool
354   __test()
355   {
356     struct frame {
357       int vertex{};
358       usize next{};
359       bool resumed{};
360     };
361 
362     __planar_vec<frame> stack;
363     __planar_vec<u8> started(__edge.size(), u8(0));
364     for ( int root : __roots ) {
365       stack.push_back({ root, 0, false });
366       while ( !stack.empty() ) {
367         auto &top = stack.data()[stack.size() - 1];
368         auto &row = __ordered.data()[static_cast<usize>(top.vertex)];
369         if ( top.next == row.size() ) {
370           const int parent = __parent_edge.data()[static_cast<usize>(top.vertex)];
371           stack.pop_back();
372           if ( parent != -1 ) __remove_back_edges(parent);
373           continue;
374         }
375         const int current = row.data()[top.next];
376         const auto &current_edge = __edge.data()[static_cast<usize>(current)];
377         if ( !started.data()[static_cast<usize>(current)] ) {
378           started.data()[static_cast<usize>(current)] = 1;
379           __stack_bottom.data()[static_cast<usize>(current)] = __conflicts.size();
380           if ( __parent_edge.data()[static_cast<usize>(current_edge.target)] == current ) {
381             stack.push_back({ current_edge.target, 0, false });
382             continue;
383           }
384           __lowpoint_edge.data()[static_cast<usize>(current)] = current;
385           __conflicts.push_back({ {}, { current, current } });
386         }
387         if ( current_edge.lowpoint < __height.data()[static_cast<usize>(top.vertex)] ) {
388           const int parent = __parent_edge.data()[static_cast<usize>(top.vertex)];
389           if ( top.next == 0 ) {
390             if ( parent != -1 ) __lowpoint_edge.data()[static_cast<usize>(parent)] = __lowpoint_edge.data()[static_cast<usize>(current)];
391           } else if ( !__add_constraints(current, parent) ) {
392             return false;
393           }
394         }
395         ++top.next;
396       }
397     }
398     return true;
399   }

401   void
402   __resolve_sides()
403   {
404     __planar_vec<int> path;
405     for ( usize first = 0; first < __edge.size(); ++first ) {
406       path.clear();
407       int current = static_cast<int>(first);
408       while ( __edge.data()[static_cast<usize>(current)].reference != -1 ) {
409         path.push_back(current);
410         current = __edge.data()[static_cast<usize>(current)].reference;
411       }
412       int absolute = __edge.data()[static_cast<usize>(current)].side;
413       while ( !path.empty() ) {
414         const int edge = path.data()[path.size() - 1];
415         path.pop_back();
416         __edge.data()[static_cast<usize>(edge)].side *= absolute;
417         absolute = __edge.data()[static_cast<usize>(edge)].side;
418         __edge.data()[static_cast<usize>(edge)].reference = -1;
419       }
420     }
421   }

423   void
424   __insert_first(int vertex, int dart)
425   {
426     int &head = __rotation_head.data()[static_cast<usize>(vertex)];
427     if ( head == -1 ) {
428       head = dart;
429       __dart_next.data()[static_cast<usize>(dart)] = dart;
430       __dart_previous.data()[static_cast<usize>(dart)] = dart;
431       return;
432     }
433     const int before = __dart_previous.data()[static_cast<usize>(head)];
434     __dart_next.data()[static_cast<usize>(before)] = dart;
435     __dart_previous.data()[static_cast<usize>(dart)] = before;
436     __dart_next.data()[static_cast<usize>(dart)] = head;
437     __dart_previous.data()[static_cast<usize>(head)] = dart;
438     head = dart;
439   }

458   void
459   __embed()
460   {
461     __dart_next = __planar_vec<int>(__edge.size() * 2, int(-1));
462     __dart_previous = __planar_vec<int>(__edge.size() * 2, int(-1));
463     __rotation_head = __planar_vec<int>(__n, int(-1));
464     for ( usize vertex = 0; vertex < __n; ++vertex ) {
465       int previous = -1;
466       for ( int edge : __ordered.data()[vertex] ) {
467         const int dart = edge * 2;
468         if ( previous == -1 )
469           __insert_first(static_cast<int>(vertex), dart);
470         else
471           __insert_after(dart, previous);
472         previous = dart;
473       }
474     }
475 
476     struct frame {
477       int vertex{};
478       usize next{};
479     };
480 
481     __planar_vec<frame> stack;
482     for ( int root : __roots ) {
483       stack.push_back({ root, 0 });
484       while ( !stack.empty() ) {
485         auto &top = stack.data()[stack.size() - 1];
486         auto &row = __ordered.data()[static_cast<usize>(top.vertex)];
487         if ( top.next == row.size() ) {
488           stack.pop_back();
489           continue;
490         }
491         const int edge = row.data()[top.next++];
492         const auto &value = __edge.data()[static_cast<usize>(edge)];
493         if ( __parent_edge.data()[static_cast<usize>(value.target)] == edge ) {
494           __insert_first(value.target, edge * 2 + 1);
495           __left_reference.data()[static_cast<usize>(value.source)] = edge * 2;
496           __right_reference.data()[static_cast<usize>(value.source)] = edge * 2;
497           stack.push_back({ value.target, 0 });
498         } else if ( value.side == 1 ) {
499           __insert_after(edge * 2 + 1, __right_reference.data()[static_cast<usize>(value.target)]);
500         } else {
501           __insert_before(edge * 2 + 1, __left_reference.data()[static_cast<usize>(value.target)]);
502           __left_reference.data()[static_cast<usize>(value.target)] = edge * 2 + 1;
503         }
504       }
505     }
506   }

508 public:
509   template<micron::integral I>
510   __lr_planarity(usize vertices, const __planar_vec<__planar_core_edge<I>> &edges, const u8 *active)
511       : __n(vertices), __active(active), __adjacency(vertices), __oriented_core(edges.size(), int(-1)), __height(vertices, int(-1)),
512         __parent_edge(vertices, int(-1)), __ordered(vertices), __left_reference(vertices, int(-1)), __right_reference(vertices, int(-1))
513   {
514     __core_source.reserve(edges.size());
515     __core_target.reserve(edges.size());
516     for ( usize i = 0; i < edges.size(); ++i ) {
517       __core_source.push_back(edges.data()[i].source);
518       __core_target.push_back(edges.data()[i].target);
519       if ( active && active[i] == 0 ) continue;
520       __adjacency.data()[edges.data()[i].source].push_back(static_cast<int>(i));
521       __adjacency.data()[edges.data()[i].target].push_back(static_cast<int>(i));
522     }
523   }

525   [[nodiscard]] bool
526   run(bool embedding)
527   {
528     usize active_edges = 0;
529     for ( usize i = 0; i < __core_source.size(); ++i )
530       if ( !__active || __active[i] ) ++active_edges;
531     if ( __n > 2 && active_edges > __n * 3 - 6 ) return false;
532     __edge.reserve(active_edges);
533     __orient();
534     __stack_bottom = __planar_vec<usize>(__edge.size(), usize(0));
535     __lowpoint_edge = __planar_vec<int>(__edge.size(), int(-1));
536     __build_ordered(false);
537     if ( !__test() ) return false;
538     if ( embedding ) {
539       __resolve_sides();
540       __build_ordered(true);
541       __embed();
542     }
543     return true;
544   }

547   void
548   make_rotation(const __planar_core<I> &core,
549                 micron::vector<micron::vector<edge_id<I>, micron::allocator_serial<>, false>, micron::allocator_serial<>, false> &out,
550                 usize vertex_slots) const
551   {
552     out.clear();
553     out.resize(vertex_slots);
554     for ( usize dense = 0; dense < __n; ++dense ) {
555       auto &row = out.data()[static_cast<usize>(core.dense_to_vertex.data()[dense].value)];
556       const int head = __rotation_head.data()[dense];
557       if ( head != -1 ) {
558         int dart = head;
559         do {
560           const int oriented = dart / 2;
561           const auto &simple = core.edges.data()[static_cast<usize>(__edge.data()[static_cast<usize>(oriented)].core_edge)];
562           if ( dense == simple.source ) {
563             for ( usize i = 0; i < simple.originals_count; ++i ) row.push_back(core.originals.data()[simple.originals_begin + i]);
564           } else {
565             for ( usize i = simple.originals_count; i != 0; --i ) row.push_back(core.originals.data()[simple.originals_begin + i - 1]);
566           }
567           dart = __dart_next.data()[static_cast<usize>(dart)];
568         } while ( dart != head );
569       }
570       for ( auto loop : core.loops.data()[dense] ) {
571         row.push_back(loop);
572         row.push_back(loop);
573       }
574     }
575   }
576 };

586 template<micron::integral I>
587 void
588 __extract_kuratowski(const __planar_core<I> &core, __planar_vec<edge_id<I>> &witness)
589 {
590   __planar_vec<u8> active(core.edges.size(), u8(1));
591   for ( usize edge = 0; edge < core.edges.size(); ++edge ) {
592     active.data()[edge] = 0;
593     if ( __core_is_planar(core, active.data()) ) active.data()[edge] = 1;
594   }
595   for ( usize edge = 0; edge < core.edges.size(); ++edge )
596     if ( active.data()[edge] ) witness.push_back(core.originals.data()[core.edges.data()[edge].originals_begin]);
597 }

601 template<graph_model G>
602 [[nodiscard]] auto
603 boyer_myrvold_planarity(const G &graph, planarity_workspace<typename G::index_type> &workspace)
604 {
605   using I = typename G::index_type;
606   planarity_result<I> result;
607   result.rotation.resize(graph.vertex_slots());
608   if constexpr ( G::is_directed ) {
609     result.status = algorithm_status::invalid_graph;
610     result.planar = false;
611     return result;
612   }
613   workspace.reserve(graph.vertices_count(), graph.edges_count());
614   auto core = __impl::__make_planar_core(graph);
615   __impl::__lr_planarity test(core.vertices, core.edges, nullptr);
616   result.planar = test.run(true);
617   if ( result.planar )
618     test.make_rotation(core, result.rotation, graph.vertex_slots());
619   else
620     __impl::__extract_kuratowski(core, result.kuratowski_edges);
621   return result;
622 }

646 template<graph_model G>
647 [[nodiscard]] bool
648 validate_rotation_system(const G &graph, const planarity_result<typename G::index_type> &embedding)
649 {
650   using I = typename G::index_type;
651   if constexpr ( G::is_directed ) return false;
652   if ( !embedding.is_planar() || embedding.rotation.size() != graph.vertex_slots() ) return false;
653   const usize edge_slots = graph.edge_slots();
654   micron::vector<u8, micron::allocator_serial<>, false> incidence_count(edge_slots, u8(0));
655   micron::vector<usize, micron::allocator_serial<>, false> position_a(edge_slots, usize(-1));
656   micron::vector<usize, micron::allocator_serial<>, false> position_b(edge_slots, usize(-1));
657   for ( auto vertex : graph.vertices() ) {
658     const usize raw = static_cast<usize>(vertex.value);
659     const auto &row = embedding.rotation.data()[raw];
660     for ( usize position = 0; position < row.size(); ++position ) {
661       const auto edge = row.data()[position];
662       if ( !graph.has_edge(edge) ) return false;
663       const auto source = graph.source(edge);
664       const auto target = graph.target(edge);
665       if ( source != vertex && target != vertex ) return false;
666       auto &count = incidence_count.data()[static_cast<usize>(edge.value)];
667       if ( source == target ) {
668         if ( count == 0 )
669           position_a.data()[static_cast<usize>(edge.value)] = position;
670         else if ( count == 1 )
671           position_b.data()[static_cast<usize>(edge.value)] = position;
672         else
673           return false;
674       } else if ( vertex == source ) {
675         if ( position_a.data()[static_cast<usize>(edge.value)] != usize(-1) ) return false;
676         position_a.data()[static_cast<usize>(edge.value)] = position;
677       } else {
678         if ( position_b.data()[static_cast<usize>(edge.value)] != usize(-1) ) return false;
679         position_b.data()[static_cast<usize>(edge.value)] = position;
680       }
681       ++count;
682     }
683   }
684   for ( auto edge : graph.edges() )
685     if ( incidence_count.data()[static_cast<usize>(edge.id.value)] != 2 ) return false;
686 
687   micron::vector<u8, micron::allocator_serial<>, false> visited(edge_slots * 2, u8(0));
688   micron::vector<usize, micron::allocator_serial<>, false> component(graph.vertex_slots(), usize(-1));
689   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> queue;
690   micron::vector<usize, micron::allocator_serial<>, false> component_vertices;
691   micron::vector<usize, micron::allocator_serial<>, false> component_edges;
692   usize components = 0;
693   for ( auto root : graph.vertices() ) {
694     if ( component.data()[static_cast<usize>(root.value)] != usize(-1) ) continue;
695     usize vertices = 0;
696     usize degree_sum = 0;
697     queue.clear();
698     queue.push_back(root);
699     component.data()[static_cast<usize>(root.value)] = components;
700     for ( usize head = 0; head < queue.size(); ++head ) {
701       const auto vertex = queue.data()[head];
702       ++vertices;
703       degree_sum += embedding.rotation.data()[static_cast<usize>(vertex.value)].size();
704       for ( auto edge : graph.out_edges(vertex) ) {
705         const auto next = graph.opposite(edge, vertex);
706         if ( component.data()[static_cast<usize>(next.value)] == usize(-1) ) {
707           component.data()[static_cast<usize>(next.value)] = components;
708           queue.push_back(next);
709         }
710       }
711     }
712     component_vertices.push_back(vertices);
713     component_edges.push_back(degree_sum / 2);
714     ++components;
715   }
716   micron::vector<usize, micron::allocator_serial<>, false> faces(components, usize(0));
717   for ( auto edge : graph.edges() )
718     for ( usize side = 0; side < 2; ++side ) {
719       usize dart = static_cast<usize>(edge.id.value) * 2 + side;
720       if ( visited.data()[dart] ) continue;
721       const usize owner_component = component.data()[static_cast<usize>((side == 0 ? edge.source : edge.target).value)];
722       ++faces.data()[owner_component];
723       usize steps = 0;
724       while ( !visited.data()[dart] ) {
725         if ( ++steps > graph.edges_count() * 2 + 1 ) return false;
726         visited.data()[dart] = 1;
727         const edge_id<I> current(static_cast<I>(dart / 2));
728         const usize current_side = dart & 1u;
729         const auto source = graph.source(current);
730         const auto target = graph.target(current);
731         const auto arrival = current_side == 0 ? target : source;
732         const usize twin_position = current_side == 0 ? position_b.data()[static_cast<usize>(current.value)]
733                                                       : position_a.data()[static_cast<usize>(current.value)];
734         const auto &row = embedding.rotation.data()[static_cast<usize>(arrival.value)];
735         if ( row.empty() || twin_position >= row.size() ) return false;
736         const auto next_edge = row.data()[(twin_position + 1) % row.size()];
737         const auto next_source = graph.source(next_edge);
738         const auto next_target = graph.target(next_edge);
739         usize next_side = 0;
740         if ( next_source == next_target ) {
741           const usize next_position = (twin_position + 1) % row.size();
742           next_side = position_a.data()[static_cast<usize>(next_edge.value)] == next_position ? 0 : 1;
743         } else {
744           next_side = next_source == arrival ? 0 : 1;
745         }
746         dart = static_cast<usize>(next_edge.value) * 2 + next_side;
747       }
748     }
749   for ( usize component_id = 0; component_id < components; ++component_id ) {
750     if ( component_edges.data()[component_id] == 0 ) continue;
751     if ( component_vertices.data()[component_id] + faces.data()[component_id] != component_edges.data()[component_id] + 2 ) return false;
752   }
753   return true;
754 }

763 template<graph_model G, typename Range>
764 [[nodiscard]] kuratowski_kind
765 reduce_kuratowski_witness(const G &graph, const Range &witness)
766 {
767   using I = typename G::index_type;
768   if constexpr ( G::is_directed ) return kuratowski_kind::none;
769   micron::vector<edge_id<I>, micron::allocator_serial<>, false> edges;
770   micron::vector<u8, micron::allocator_serial<>, false> selected(graph.edge_slots(), u8(0));
771   for ( auto edge : witness ) {
772     if ( !graph.has_edge(edge) || selected.data()[static_cast<usize>(edge.value)] || graph.source(edge) == graph.target(edge) )
773       return kuratowski_kind::none;
774     selected.data()[static_cast<usize>(edge.value)] = 1;
775     edges.push_back(edge);
776   }
777   if ( edges.empty() ) return kuratowski_kind::none;
778   micron::vector<usize, micron::allocator_serial<>, false> degree(graph.vertex_slots(), usize(0));
779   for ( auto edge : edges ) {
780     ++degree.data()[static_cast<usize>(graph.source(edge).value)];
781     ++degree.data()[static_cast<usize>(graph.target(edge).value)];
782   }
783   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> branch;
784   for ( auto vertex : graph.vertices() ) {
785     const usize value = degree.data()[static_cast<usize>(vertex.value)];
786     if ( value == 1 || value > 4 ) return kuratowski_kind::none;
787     if ( value != 0 && value != 2 ) branch.push_back(vertex);
788   }
789   const bool maybe_k5 = branch.size() == 5;
790   const bool maybe_k33 = branch.size() == 6;
791   if ( !maybe_k5 && !maybe_k33 ) return kuratowski_kind::none;
792   micron::vector<usize, micron::allocator_serial<>, false> branch_index(graph.vertex_slots(), usize(-1));
793   for ( usize i = 0; i < branch.size(); ++i ) {
794     const usize expected = maybe_k5 ? 4 : 3;
795     if ( degree.data()[static_cast<usize>(branch.data()[i].value)] != expected ) return kuratowski_kind::none;
796     branch_index.data()[static_cast<usize>(branch.data()[i].value)] = i;
797   }
798   micron::vector<u8, micron::allocator_serial<>, false> consumed(graph.edge_slots(), u8(0));
799   micron::vector<u8, micron::allocator_serial<>, false> adjacency(branch.size() * branch.size(), u8(0));
800   for ( usize start_index = 0; start_index < branch.size(); ++start_index ) {
801     const auto start = branch.data()[start_index];
802     for ( auto first : graph.out_edges(start) ) {
803       if ( !selected.data()[static_cast<usize>(first.value)] || consumed.data()[static_cast<usize>(first.value)] ) continue;
804       consumed.data()[static_cast<usize>(first.value)] = 1;
805       auto current = graph.opposite(first, start);
806       usize walked = 1;
807       while ( branch_index.data()[static_cast<usize>(current.value)] == usize(-1) ) {
808         edge_id<I> next = edge_id<I>::invalid();
809         for ( auto candidate : graph.out_edges(current) )
810           if ( selected.data()[static_cast<usize>(candidate.value)] && !consumed.data()[static_cast<usize>(candidate.value)] ) {
811             next = candidate;
812             break;
813           }
814         if ( !next.valid() || ++walked > edges.size() ) return kuratowski_kind::none;
815         consumed.data()[static_cast<usize>(next.value)] = 1;
816         current = graph.opposite(next, current);
817       }
818       const usize target_index = branch_index.data()[static_cast<usize>(current.value)];
819       if ( target_index == start_index || adjacency.data()[start_index * branch.size() + target_index] ) return kuratowski_kind::none;
820       adjacency.data()[start_index * branch.size() + target_index] = adjacency.data()[target_index * branch.size() + start_index] = 1;
821     }
822   }
823   for ( auto edge : edges )
824     if ( !consumed.data()[static_cast<usize>(edge.value)] ) return kuratowski_kind::none;
825   if ( maybe_k5 ) {
826     for ( usize u = 0; u < 5; ++u )
827       for ( usize v = 0; v < 5; ++v )
828         if ( (u != v) != static_cast<bool>(adjacency.data()[u * 5 + v]) ) return kuratowski_kind::none;
829     return kuratowski_kind::k5;
830   }
831   micron::vector<int, micron::allocator_serial<>, false> color(6, int(-1));
832   color.data()[0] = 0;
833   micron::vector<usize, micron::allocator_serial<>, false> queue;
834   queue.push_back(0);
835   for ( usize head = 0; head < queue.size(); ++head ) {
836     const usize u = queue.data()[head];
837     for ( usize v = 0; v < 6; ++v ) {
838       if ( !adjacency.data()[u * 6 + v] ) continue;
839       if ( color.data()[v] == -1 ) {
840         color.data()[v] = 1 - color.data()[u];
841         queue.push_back(v);
842       } else if ( color.data()[v] == color.data()[u] ) {
843         return kuratowski_kind::none;
844       }
845     }
846   }
847   usize left = 0;
848   for ( int value : color ) left += value == 0;
849   return left == 3 ? kuratowski_kind::k33 : kuratowski_kind::none;
850 }

852 template<graph_model G, typename Range>
853 [[nodiscard]] bool
854 validate_kuratowski_witness(const G &graph, const Range &witness)
855 {
856   return reduce_kuratowski_witness(graph, witness) != kuratowski_kind::none;
857 }
