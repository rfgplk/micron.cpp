62 template<graph_model G, typename WeightMap, typename D>
63 algorithm_status
64 yen_potentials(const G &graph, WeightMap &weight_map, micron::vector<D, micron::allocator_serial<>, false> &potential)
65 {
66   potential.resize(graph.vertex_slots(), D{});
67   potential.fill(D{});
68   for ( auto edge : graph.edges() )
69     if ( __impl::nonfinite_weight(static_cast<D>(__impl::weight(weight_map, graph, edge.id))) ) return algorithm_status::invalid_weight;
70 
71   auto relax = [&](typename G::vertex_descriptor u, typename G::vertex_descriptor v, const D &weight) {
72     D candidate{};
73     if ( !__impl::add_distance(potential.data()[static_cast<usize>(u.value)], weight, candidate) ) return -1;
74     D &target = potential.data()[static_cast<usize>(v.value)];
75     if ( candidate < target ) {
76       target = candidate;
77       return 1;
78     }
79     return 0;
80   };
81   for ( usize pass = 1; pass < graph.vertices_count(); ++pass ) {
82     bool changed = false;
83     for ( auto edge : graph.edges() ) {
84       const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge.id));
85       const int forward = relax(edge.source, edge.target, weight);
86       if ( forward < 0 ) return algorithm_status::overflow;
87       changed |= forward != 0;
88       if constexpr ( !G::is_directed ) {
89         const int reverse = relax(edge.target, edge.source, weight);
90         if ( reverse < 0 ) return algorithm_status::overflow;
91         changed |= reverse != 0;
92       }
93     }
94     if ( !changed ) break;
95   }
96   for ( auto edge : graph.edges() ) {
97     const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge.id));
98     const int forward = relax(edge.source, edge.target, weight);
99     if ( forward < 0 ) return algorithm_status::overflow;
100     if ( forward > 0 ) return algorithm_status::negative_cycle;
101     if constexpr ( !G::is_directed ) {
102       const int reverse = relax(edge.target, edge.source, weight);
103       if ( reverse < 0 ) return algorithm_status::overflow;
104       if ( reverse > 0 ) return algorithm_status::negative_cycle;
105     }
106   }
107   return algorithm_status::ok;
108 }

110 template<typename I, typename D>
111 [[nodiscard]] bool
112 same_weighted_path(const weighted_path<I, D> &a, const weighted_path<I, D> &b) noexcept
113 {
114   if ( a.edges.size() != b.edges.size() || a.vertices.size() != b.vertices.size() ) return false;
115   for ( usize i = 0; i < a.edges.size(); ++i )
116     if ( a.edges.data()[i] != b.edges.data()[i] ) return false;
117   for ( usize i = 0; i < a.vertices.size(); ++i )
118     if ( a.vertices.data()[i] != b.vertices.data()[i] ) return false;
119   return true;
120 }

122 template<typename I, typename D>
123 [[nodiscard]] bool
124 weighted_path_less(const weighted_path<I, D> &a, const weighted_path<I, D> &b) noexcept
125 {
126   if ( a.cost < b.cost ) return true;
127   if ( b.cost < a.cost ) return false;
128   const usize vertices = a.vertices.size() < b.vertices.size() ? a.vertices.size() : b.vertices.size();
129   for ( usize i = 0; i < vertices; ++i ) {
130     if ( a.vertices.data()[i] < b.vertices.data()[i] ) return true;
131     if ( b.vertices.data()[i] < a.vertices.data()[i] ) return false;
132   }
133   if ( a.vertices.size() != b.vertices.size() ) return a.vertices.size() < b.vertices.size();
134   for ( usize i = 0; i < a.edges.size(); ++i ) {
135     if ( a.edges.data()[i] < b.edges.data()[i] ) return true;
136     if ( b.edges.data()[i] < a.edges.data()[i] ) return false;
137   }
138   return false;
139 }

143 template<graph_model G, typename WeightMap>
144 [[nodiscard]] auto
145 yen(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target, usize count,
146     yen_workspace<typename G::index_type,
147                   micron::remove_cvref_t<decltype(__impl::weight(micron::declval<WeightMap &>(), micron::declval<const G &>(),
148                                                                  micron::declval<typename G::edge_descriptor>()))>> &workspace,
149     WeightMap weight_map)
150 {
151   using I = typename G::index_type;
152   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
153   using vertex_descriptor = typename G::vertex_descriptor;
154   using edge_descriptor = typename G::edge_descriptor;
155   yen_result<I, D> result;
156   if ( count == 0 ) return result;
157   if ( !graph.has_vertex(source) || !graph.has_vertex(target) ) {
158     result.status = algorithm_status::invalid_vertex;
159     return result;
160   }
161   workspace.reserve(graph.vertex_slots(), graph.edge_slots());
162   result.status = __impl::yen_potentials(graph, weight_map, workspace.potential);
163   if ( result.status != algorithm_status::ok ) return result;
164   workspace.candidates.clear();
165   workspace.excluded_vertex.resize(graph.vertex_slots(), u8(0));
166   workspace.excluded_edge.resize(graph.edge_slots(), u8(0));
167 
168   auto shortest_filtered = [&](vertex_descriptor start, vertex_descriptor finish, weighted_path<I, D> &path) {
169     const usize slots = graph.vertex_slots();
170     workspace.distance.resize(slots, D{});
171     workspace.distance.fill(D{});
172     workspace.parent.resize(slots, vertex_descriptor::invalid());
173     workspace.parent.fill(vertex_descriptor::invalid());
174     workspace.parent_edge.resize(slots, edge_descriptor::invalid());
175     workspace.parent_edge.fill(edge_descriptor::invalid());
176     workspace.reached.resize(slots, u8(0));
177     workspace.reached.fill(u8(0));
178     workspace.settled.resize(slots, u8(0));
179     workspace.settled.fill(u8(0));
180     workspace.heap_vertex.clear();
181     workspace.heap_distance.clear();
182     if ( workspace.excluded_vertex.data()[static_cast<usize>(start.value)]
183          || workspace.excluded_vertex.data()[static_cast<usize>(finish.value)] )
184       return algorithm_status::unreachable;
185     workspace.reached.data()[static_cast<usize>(start.value)] = 1;
186     workspace.parent.data()[static_cast<usize>(start.value)] = start;
187 
188     auto less = [](const D &ad, vertex_descriptor av, const D &bd, vertex_descriptor bv) { return ad < bd || (!(bd < ad) && av < bv); };
189     auto push = [&](vertex_descriptor vertex, const D &distance) {
190       workspace.heap_vertex.push_back(vertex);
191       workspace.heap_distance.push_back(distance);
192       usize child = workspace.heap_vertex.size() - 1;
193       while ( child ) {
194         const usize parent = (child - 1) / 2;
195         if ( !less(workspace.heap_distance.data()[child], workspace.heap_vertex.data()[child], workspace.heap_distance.data()[parent],
196                    workspace.heap_vertex.data()[parent]) )
197           break;
198         micron::swap(workspace.heap_vertex.data()[child], workspace.heap_vertex.data()[parent]);
199         micron::swap(workspace.heap_distance.data()[child], workspace.heap_distance.data()[parent]);
200         child = parent;
201       }
202     };
203     push(start, D{});
204     while ( !workspace.heap_vertex.empty() ) {
205       const vertex_descriptor selected = workspace.heap_vertex.data()[0];
206       const D selected_distance = workspace.heap_distance.data()[0];
207       const usize last = workspace.heap_vertex.size() - 1;
208       workspace.heap_vertex.data()[0] = workspace.heap_vertex.data()[last];
209       workspace.heap_distance.data()[0] = workspace.heap_distance.data()[last];
210       workspace.heap_vertex.pop_back();
211       workspace.heap_distance.pop_back();
212       usize parent = 0;
213       while ( parent < workspace.heap_vertex.size() ) {
214         const usize left = parent * 2 + 1;
215         if ( left >= workspace.heap_vertex.size() ) break;
216         const usize right = left + 1;
217         usize child = left;
218         if ( right < workspace.heap_vertex.size()
219              && less(workspace.heap_distance.data()[right], workspace.heap_vertex.data()[right], workspace.heap_distance.data()[left],
220                      workspace.heap_vertex.data()[left]) )
221           child = right;
222         if ( !less(workspace.heap_distance.data()[child], workspace.heap_vertex.data()[child], workspace.heap_distance.data()[parent],
223                    workspace.heap_vertex.data()[parent]) )
224           break;
225         micron::swap(workspace.heap_vertex.data()[child], workspace.heap_vertex.data()[parent]);
226         micron::swap(workspace.heap_distance.data()[child], workspace.heap_distance.data()[parent]);
227         parent = child;
228       }
229       const usize us = static_cast<usize>(selected.value);
230       if ( workspace.settled.data()[us] || selected_distance != workspace.distance.data()[us] ) continue;
231       workspace.settled.data()[us] = 1;
232       if ( selected == finish ) break;
233       for ( auto edge : graph.out_edges(selected) ) {
234         if ( workspace.excluded_edge.data()[static_cast<usize>(edge.value)] ) continue;
235         const vertex_descriptor neighbor = __impl::edge_neighbor(graph, edge, selected);
236         const usize vs = static_cast<usize>(neighbor.value);
237         if ( workspace.excluded_vertex.data()[vs] || workspace.settled.data()[vs] ) continue;
238         const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge));
239         D partial{}, reduced{}, candidate{};
240         if ( !__impl::add_distance(weight, workspace.potential.data()[us], partial)
241              || !__impl::sub_distance(partial, workspace.potential.data()[vs], reduced)
242              || !__impl::add_distance(workspace.distance.data()[us], reduced, candidate) )
243           return algorithm_status::overflow;
244         const bool equal
245             = workspace.reached.data()[vs] && !(candidate < workspace.distance.data()[vs]) && !(workspace.distance.data()[vs] < candidate);
246         if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs]
247              || (equal
248                  && (selected < workspace.parent.data()[vs]
249                      || (selected == workspace.parent.data()[vs] && edge < workspace.parent_edge.data()[vs]))) ) {
250           workspace.reached.data()[vs] = 1;
251           workspace.distance.data()[vs] = candidate;
252           workspace.parent.data()[vs] = selected;
253           workspace.parent_edge.data()[vs] = edge;
254           push(neighbor, candidate);
255         }
256       }
257     }
258     if ( !workspace.reached.data()[static_cast<usize>(finish.value)] ) return algorithm_status::unreachable;
259     path.vertices.clear();
260     path.edges.clear();
261     workspace.path_vertex.resize(slots, u8(0));
262     workspace.path_vertex.fill(u8(0));
263     workspace.reach_seen.resize(slots, u8(0));
264     vertex_descriptor current = start;
265     path.vertices.push_back(current);
266     workspace.path_vertex.data()[static_cast<usize>(current.value)] = 1;
267     bool path_overflow = false;
268 
269     auto tight = [&](vertex_descriptor u, edge_descriptor edge, vertex_descriptor v) {
270       const usize us = static_cast<usize>(u.value);
271       const usize vs = static_cast<usize>(v.value);
272       const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge));
273       D partial{}, reduced{}, candidate{};
274       if ( !__impl::add_distance(weight, workspace.potential.data()[us], partial)
275            || !__impl::sub_distance(partial, workspace.potential.data()[vs], reduced)
276            || !__impl::add_distance(workspace.distance.data()[us], reduced, candidate) ) {
277         path_overflow = true;
278         return false;
279       }
280       return workspace.reached.data()[us] && workspace.reached.data()[vs] && !(candidate < workspace.distance.data()[vs])
281              && !(workspace.distance.data()[vs] < candidate);
282     };
283 
284     auto reaches_finish = [&](vertex_descriptor first) {
285       workspace.reach_seen.fill(u8(0));
286       workspace.reach_stack.clear();
287       workspace.reach_stack.push_back(first);
288       workspace.reach_seen.data()[static_cast<usize>(first.value)] = 1;
289       while ( !workspace.reach_stack.empty() ) {
290         const vertex_descriptor u = workspace.reach_stack.data()[workspace.reach_stack.size() - 1];
291         workspace.reach_stack.pop_back();
292         if ( u == finish ) return true;
293         for ( auto edge : graph.out_edges(u) ) {
294           if ( workspace.excluded_edge.data()[static_cast<usize>(edge.value)] ) continue;
295           const vertex_descriptor v = __impl::edge_neighbor(graph, edge, u);
296           const usize vs = static_cast<usize>(v.value);
297           if ( workspace.excluded_vertex.data()[vs] || workspace.path_vertex.data()[vs] || workspace.reach_seen.data()[vs]
298                || !tight(u, edge, v) )
299             continue;
300           workspace.reach_seen.data()[vs] = 1;
301           workspace.reach_stack.push_back(v);
302         }
303       }
304       return false;
305     };
306 
307     for ( usize guard = 0; current != finish && guard < graph.vertices_count(); ++guard ) {
308       vertex_descriptor selected_vertex = vertex_descriptor::invalid();
309       edge_descriptor selected_edge = edge_descriptor::invalid();
310       for ( auto edge : graph.out_edges(current) ) {
311         if ( workspace.excluded_edge.data()[static_cast<usize>(edge.value)] ) continue;
312         const vertex_descriptor neighbor = __impl::edge_neighbor(graph, edge, current);
313         const usize slot = static_cast<usize>(neighbor.value);
314         if ( workspace.excluded_vertex.data()[slot] || workspace.path_vertex.data()[slot] || !tight(current, edge, neighbor) ) continue;
315         if ( selected_vertex.valid() && (selected_vertex < neighbor || (selected_vertex == neighbor && selected_edge < edge)) ) continue;
316         if ( reaches_finish(neighbor) ) {
317           selected_vertex = neighbor;
318           selected_edge = edge;
319         }
320       }
321       if ( path_overflow ) return algorithm_status::overflow;
322       if ( !selected_vertex.valid() ) return algorithm_status::invalid_graph;
323       path.edges.push_back(selected_edge);
324       path.vertices.push_back(selected_vertex);
325       workspace.path_vertex.data()[static_cast<usize>(selected_vertex.value)] = 1;
326       current = selected_vertex;
327     }
328     if ( current != finish ) return algorithm_status::invalid_graph;
329     D partial{};
330     if ( !__impl::sub_distance(workspace.distance.data()[static_cast<usize>(finish.value)],
331                                workspace.potential.data()[static_cast<usize>(start.value)], partial)
332          || !__impl::add_distance(partial, workspace.potential.data()[static_cast<usize>(finish.value)], path.cost) )
333       return algorithm_status::overflow;
334     return algorithm_status::ok;
335   };
336 
337   workspace.excluded_vertex.fill(u8(0));
338   workspace.excluded_edge.fill(u8(0));
339   weighted_path<I, D> first;
340   result.status = shortest_filtered(source, target, first);
341   if ( result.status != algorithm_status::ok ) return result;
342   result.paths.push_back(micron::move(first));
343 
344   while ( result.paths.size() < count ) {
345     const weighted_path<I, D> &previous = result.paths.data()[result.paths.size() - 1];
346     D root_cost{};
347     for ( usize spur_index = 0; spur_index < previous.edges.size(); ++spur_index ) {
348       workspace.excluded_vertex.fill(u8(0));
349       workspace.excluded_edge.fill(u8(0));
350       for ( usize i = 0; i < spur_index; ++i ) workspace.excluded_vertex.data()[static_cast<usize>(previous.vertices.data()[i].value)] = 1;
351       for ( const auto &accepted : result.paths ) {
352         if ( accepted.edges.size() <= spur_index ) continue;
353         bool same_root = accepted.vertices.data()[spur_index] == previous.vertices.data()[spur_index];
354         for ( usize i = 0; same_root && i < spur_index; ++i )
355           same_root = accepted.edges.data()[i] == previous.edges.data()[i] && accepted.vertices.data()[i] == previous.vertices.data()[i];
356         if ( same_root ) workspace.excluded_edge.data()[static_cast<usize>(accepted.edges.data()[spur_index].value)] = 1;
357       }
358 
359       weighted_path<I, D> spur;
360       const algorithm_status spur_status = shortest_filtered(previous.vertices.data()[spur_index], target, spur);
361       if ( spur_status == algorithm_status::overflow ) {
362         result.status = spur_status;
363         return result;
364       }
365       if ( spur_status == algorithm_status::ok ) {
366         weighted_path<I, D> candidate;
367         candidate.vertices.reserve(spur_index + spur.vertices.size());
368         candidate.edges.reserve(spur_index + spur.edges.size());
369         for ( usize i = 0; i < spur_index; ++i ) {
370           candidate.vertices.push_back(previous.vertices.data()[i]);
371           candidate.edges.push_back(previous.edges.data()[i]);
372         }
373         for ( auto vertex : spur.vertices ) candidate.vertices.push_back(vertex);
374         for ( auto edge : spur.edges ) candidate.edges.push_back(edge);
375         if ( !__impl::add_distance(root_cost, spur.cost, candidate.cost) ) {
376           result.status = algorithm_status::overflow;
377           return result;
378         }
379         bool duplicate = false;
380         for ( const auto &accepted : result.paths ) duplicate |= __impl::same_weighted_path(accepted, candidate);
381         for ( const auto &queued : workspace.candidates ) duplicate |= __impl::same_weighted_path(queued, candidate);
382         if ( !duplicate ) workspace.candidates.push_back(micron::move(candidate));
383       }
384       D next_cost{};
385       const D edge_weight = static_cast<D>(__impl::weight(weight_map, graph, previous.edges.data()[spur_index]));
386       if ( !__impl::add_distance(root_cost, edge_weight, next_cost) ) {
387         result.status = algorithm_status::overflow;
388         return result;
389       }
390       root_cost = next_cost;
391     }
392     if ( workspace.candidates.empty() ) break;
393     usize best = 0;
394     for ( usize i = 1; i < workspace.candidates.size(); ++i )
395       if ( __impl::weighted_path_less(workspace.candidates.data()[i], workspace.candidates.data()[best]) ) best = i;
396     result.paths.push_back(micron::move(workspace.candidates.data()[best]));
397     workspace.candidates.erase(best);
398   }
399   result.status = algorithm_status::ok;
400   return result;
401 }

403 template<graph_model G, typename WeightMap>
404 [[nodiscard]] auto
405 yen(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target, usize count, WeightMap weight_map)
406 {
407   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
408   yen_workspace<typename G::index_type, D> workspace;
409   return yen(graph, source, target, count, workspace, weight_map);
410 }
