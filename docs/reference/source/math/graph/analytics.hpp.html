14 template<graph_model G>
15 [[nodiscard]] f64
16 density(const G &graph) noexcept
17 {
18   const usize n = graph.vertices_count();
19   if ( n < 2 && !G::allows_loops ) return f64(0);
20   usize possible{};
21   if constexpr ( G::allows_loops )
22     possible = G::is_directed ? n * n : n * (n + 1) / 2;
23   else
24     possible = G::is_directed ? n * (n - 1) : n * (n - 1) / 2;
25   return possible ? static_cast<f64>(graph.edges_count()) / static_cast<f64>(possible) : f64(0);
26 }

40 template<graph_model G>
41 [[nodiscard]] usize
42 common_neighbors(const G &graph, typename G::vertex_descriptor a, typename G::vertex_descriptor b)
43 {
44   if ( !graph.has_vertex(a) || !graph.has_vertex(b) ) return 0;
45   if constexpr ( bitset_neighbor_graph<G> ) {
46     const auto left = graph.neighbor_words(a);
47     const auto right = graph.neighbor_words(b);
48     const usize words = left.words < right.words ? left.words : right.words;
49     usize result = 0;
50     for ( usize word = 0; word < words; ++word ) result += static_cast<usize>(__builtin_popcountll(left.data[word] & right.data[word]));
51     return result;
52   }
53   usize result = 0;
54   if constexpr ( matrix_adjacency_graph<G> ) {
55     for ( auto neighbor : graph.out_neighbors(a) )
56       if ( graph.matrix_has_edge(b, neighbor) || (!G::is_directed && graph.matrix_has_edge(neighbor, b)) ) ++result;
57   } else {
58     for ( auto neighbor : graph.out_neighbors(a) )
59       if ( graph.has_edge(b, neighbor) || (!G::is_directed && graph.has_edge(neighbor, b)) ) ++result;
60   }
61   return result;
62 }

64 template<graph_model G>
65 [[nodiscard]] usize
66 triangles(const G &graph)
67 {
68   usize count = 0;
69   for ( auto u : graph.vertices() ) {
70     for ( auto v : graph.out_neighbors(u) ) {
71       if ( v.value <= u.value ) continue;
72       for ( auto w : graph.out_neighbors(v) ) {
73         if ( w.value <= v.value ) continue;
74         if ( graph.has_edge(u, w) ) ++count;
75       }
76     }
77   }
78   return count;
79 }

81 template<graph_model G>
82 [[nodiscard]] f64
83 clustering(const G &graph, typename G::vertex_descriptor vertex)
84 {
85   const usize degree = graph.out_degree(vertex);
86   if ( degree < 2 ) return f64(0);
87   usize links = 0;
88   for ( auto a : graph.out_neighbors(vertex) )
89     for ( auto b : graph.out_neighbors(vertex) )
90       if ( a.value < b.value && (graph.has_edge(a, b) || graph.has_edge(b, a)) ) ++links;
91   return (f64(2) * static_cast<f64>(links)) / static_cast<f64>(degree * (degree - 1));
92 }

103 template<graph_model G>
104 [[nodiscard]] f64
105 transitivity(const G &graph)
106 {
107   usize triples = 0;
108   for ( auto vertex : graph.vertices() ) {
109     const usize degree = graph.out_degree(vertex);
110     triples += degree * (degree - 1) / 2;
111   }
112   return triples ? f64(3) * static_cast<f64>(triangles(graph)) / static_cast<f64>(triples) : f64(0);
113 }

115 template<graph_model G>
116 [[nodiscard]] f64
117 reciprocity(const G &graph)
118 {
119   if constexpr ( !G::is_directed ) return graph.edges_count() ? f64(1) : f64(0);
120   if ( graph.edges_count() == 0 ) return f64(0);
121   usize reciprocal = 0;
122   for ( auto edge : graph.edges() )
123     if ( graph.has_edge(edge.target, edge.source) ) ++reciprocal;
124   return static_cast<f64>(reciprocal) / static_cast<f64>(graph.edges_count());
125 }

134 template<graph_model G>
135 [[nodiscard]] iterative_scores_result<f64>
136 pagerank(const G &graph, f64 damping = 0.85, f64 tolerance = 1e-10, usize max_iterations = 100)
137 {
138   iterative_scores_result<f64> result;
139   const usize n = graph.vertices_count();
140   result.score = micron::vector<f64, micron::allocator_serial<>, false>(graph.vertex_slots(), f64(0));
141   if ( n == 0 ) return result;
142   const f64 initial = f64(1) / static_cast<f64>(n);
143   for ( auto vertex : graph.vertices() ) result.score.data()[static_cast<usize>(vertex.value)] = initial;
144   micron::vector<f64, micron::allocator_serial<>, false> next(graph.vertex_slots(), f64(0));
145   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
146     f64 dangling = 0;
147     for ( auto vertex : graph.vertices() )
148       if ( graph.out_degree(vertex) == 0 ) dangling += result.score.data()[static_cast<usize>(vertex.value)];
149     const f64 base = (f64(1) - damping) / static_cast<f64>(n) + damping * dangling / static_cast<f64>(n);
150     next.fill(f64(0));
151     for ( auto vertex : graph.vertices() ) next.data()[static_cast<usize>(vertex.value)] = base;
152     for ( auto u : graph.vertices() ) {
153       const usize degree = graph.out_degree(u);
154       if ( degree == 0 ) continue;
155       const f64 share = damping * result.score.data()[static_cast<usize>(u.value)] / static_cast<f64>(degree);
156       for ( auto v : graph.out_neighbors(u) ) next.data()[static_cast<usize>(v.value)] += share;
157     }
158     f64 residual = 0;
159     for ( auto vertex : graph.vertices() ) {
160       const usize slot = static_cast<usize>(vertex.value);
161       f64 delta = next.data()[slot] - result.score.data()[slot];
162       if ( delta < 0 ) delta = -delta;
163       residual += delta;
164       result.score.data()[slot] = next.data()[slot];
165     }
166     result.iterations = iteration + 1;
167     result.residual = residual;
168     if ( residual <= tolerance ) return result;
169   }
170   result.status = algorithm_status::non_convergent;
171   return result;
172 }

182 template<graph_model G>
183 [[nodiscard]] hits_result<f64>
184 hits(const G &graph, f64 tolerance = 1e-10, usize max_iterations = 100)
185 {
186   hits_result<f64> result;
187   const usize n = graph.vertices_count();
188   result.hubs = micron::vector<f64, micron::allocator_serial<>, false>(graph.vertex_slots(), f64(0));
189   result.authorities = micron::vector<f64, micron::allocator_serial<>, false>(graph.vertex_slots(), f64(0));
190   if ( n == 0 ) return result;
191   for ( auto vertex : graph.vertices() ) {
192     result.hubs.data()[static_cast<usize>(vertex.value)] = f64(1);
193     result.authorities.data()[static_cast<usize>(vertex.value)] = f64(1);
194   }
195   micron::vector<f64, micron::allocator_serial<>, false> next_hub(graph.vertex_slots(), f64(0));
196   micron::vector<f64, micron::allocator_serial<>, false> next_auth(graph.vertex_slots(), f64(0));
197   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
198     next_hub.fill(f64(0));
199     next_auth.fill(f64(0));
200     for ( auto edge : graph.edges() ) {
201       next_auth.data()[static_cast<usize>(edge.target.value)] += result.hubs.data()[static_cast<usize>(edge.source.value)];
202       next_hub.data()[static_cast<usize>(edge.source.value)] += result.authorities.data()[static_cast<usize>(edge.target.value)];
203       if constexpr ( !G::is_directed ) {
204         next_auth.data()[static_cast<usize>(edge.source.value)] += result.hubs.data()[static_cast<usize>(edge.target.value)];
205         next_hub.data()[static_cast<usize>(edge.target.value)] += result.authorities.data()[static_cast<usize>(edge.source.value)];
206       }
207     }
208     f64 hub_norm = 0;
209     f64 auth_norm = 0;
210     for ( auto vertex : graph.vertices() ) {
211       const usize slot = static_cast<usize>(vertex.value);
212       hub_norm += next_hub.data()[slot] * next_hub.data()[slot];
213       auth_norm += next_auth.data()[slot] * next_auth.data()[slot];
214     }
215     hub_norm = __builtin_sqrt(hub_norm);
216     auth_norm = __builtin_sqrt(auth_norm);
217     if ( hub_norm == 0 ) hub_norm = 1;
218     if ( auth_norm == 0 ) auth_norm = 1;
219     f64 residual = 0;
220     for ( auto vertex : graph.vertices() ) {
221       const usize slot = static_cast<usize>(vertex.value);
222       next_hub.data()[slot] /= hub_norm;
223       next_auth.data()[slot] /= auth_norm;
224       f64 dh = next_hub.data()[slot] - result.hubs.data()[slot];
225       f64 da = next_auth.data()[slot] - result.authorities.data()[slot];
226       if ( dh < 0 ) dh = -dh;
227       if ( da < 0 ) da = -da;
228       residual += dh + da;
229       result.hubs.data()[slot] = next_hub.data()[slot];
230       result.authorities.data()[slot] = next_auth.data()[slot];
231     }
232     result.iterations = iteration + 1;
233     result.residual = residual;
234     if ( residual <= tolerance ) return result;
235   }
236   result.status = algorithm_status::non_convergent;
237   return result;
238 }

254 closeness_centrality(const G &graph)

257   for ( auto source : graph.vertices() ) {

289 template<graph_model G>
290 [[nodiscard]] micron::vector<f64, micron::allocator_serial<>, false>
291 betweenness_centrality(const G &graph, bool normalized = true)
292 {
293   using vertex_descriptor = typename G::vertex_descriptor;
294   const usize slots = graph.vertex_slots();
295   micron::vector<f64, micron::allocator_serial<>, false> centrality(slots, f64(0));
296   using predecessor_vector = micron::vector<vertex_descriptor, micron::allocator_serial<>, false>;
297   for ( auto source : graph.vertices() ) {
298     micron::vector<predecessor_vector, micron::allocator_serial<>, false> predecessors(slots);
299     micron::vector<f64, micron::allocator_serial<>, false> sigma(slots, f64(0));
300     micron::vector<max_t, micron::allocator_serial<>, false> distance(slots, max_t(-1));
301     micron::vector<vertex_descriptor, micron::allocator_serial<>, false> queue;
302     micron::vector<vertex_descriptor, micron::allocator_serial<>, false> stack;
303     sigma.data()[static_cast<usize>(source.value)] = f64(1);
304     distance.data()[static_cast<usize>(source.value)] = 0;
305     queue.push_back(source);
306     usize head = 0;
307     while ( head < queue.size() ) {
308       const vertex_descriptor u = queue.data()[head++];
309       stack.push_back(u);
310       for ( auto v : graph.out_neighbors(u) ) {
311         const usize vs = static_cast<usize>(v.value);
312         if ( distance.data()[vs] < 0 ) {
313           distance.data()[vs] = distance.data()[static_cast<usize>(u.value)] + 1;
314           queue.push_back(v);
315         }
316         if ( distance.data()[vs] == distance.data()[static_cast<usize>(u.value)] + 1 ) {
317           sigma.data()[vs] += sigma.data()[static_cast<usize>(u.value)];
318           predecessors.data()[vs].push_back(u);
319         }
320       }
321     }
322     micron::vector<f64, micron::allocator_serial<>, false> dependency(slots, f64(0));
323     while ( !stack.empty() ) {
324       const vertex_descriptor w = stack.data()[stack.size() - 1];
325       stack.pop_back();
326       const usize ws = static_cast<usize>(w.value);
327       for ( auto v : predecessors.data()[ws] ) {
328         const usize vs = static_cast<usize>(v.value);
329         if ( sigma.data()[ws] != 0 ) dependency.data()[vs] += sigma.data()[vs] / sigma.data()[ws] * (f64(1) + dependency.data()[ws]);
330       }
331       if ( w != source ) centrality.data()[ws] += dependency.data()[ws];
332     }
333   }
334   if constexpr ( !G::is_directed )
335     for ( auto vertex : graph.vertices() ) centrality.data()[static_cast<usize>(vertex.value)] *= f64(0.5);
336   if ( normalized && graph.vertices_count() > 2 ) {
337     const f64 scale = G::is_directed ? f64(1) / static_cast<f64>((graph.vertices_count() - 1) * (graph.vertices_count() - 2))
338                                      : f64(2) / static_cast<f64>((graph.vertices_count() - 1) * (graph.vertices_count() - 2));
339     for ( auto vertex : graph.vertices() ) centrality.data()[static_cast<usize>(vertex.value)] *= scale;
340   }
341   return centrality;
342 }

352 template<graph_model G>
353 [[nodiscard]] distance_extrema_result<typename G::index_type>
354 distance_extrema(const G &graph)
355 {
356   distance_extrema_result<typename G::index_type> result;
357   result.eccentricity = micron::vector<usize, micron::allocator_serial<>, false>(graph.vertex_slots(), usize(0));
358   bool first = true;
359   for ( auto source : graph.vertices() ) {
360     auto paths = unweighted_shortest_paths(graph, source);
361     usize eccentricity = 0;
362     for ( auto target : graph.vertices() ) {
363       if ( !paths.contains(target) ) {
364         result.status = algorithm_status::disconnected;
365         return result;
366       }
367       const usize distance = paths.distance.data()[static_cast<usize>(target.value)];
368       if ( distance > eccentricity ) eccentricity = distance;
369     }
370     result.eccentricity.data()[static_cast<usize>(source.value)] = eccentricity;
371     if ( first || eccentricity < result.radius ) result.radius = eccentricity;
372     if ( first || eccentricity > result.diameter ) result.diameter = eccentricity;
373     first = false;
374   }
375   for ( auto vertex : graph.vertices() )
376     if ( result.eccentricity.data()[static_cast<usize>(vertex.value)] == result.radius ) result.centers.push_back(vertex);
377   return result;
378 }

380 template<graph_model G>
381 [[nodiscard]] micron::vector<usize, micron::allocator_serial<>, false>
382 core_numbers(const G &graph)
383 {
384   micron::vector<usize, micron::allocator_serial<>, false> core(graph.vertex_slots(), usize(0));
385   micron::vector<usize, micron::allocator_serial<>, false> degree(graph.vertex_slots(), usize(0));
386   micron::vector<u8, micron::allocator_serial<>, false> active(graph.vertex_slots(), u8(0));
387   for ( auto vertex : graph.vertices() ) {
388     const usize slot = static_cast<usize>(vertex.value);
389     active.data()[slot] = 1;
390     degree.data()[slot] = graph.out_degree(vertex);
391   }
392   usize current_core = 0;
393   for ( usize removed = 0; removed < graph.vertices_count(); ++removed ) {
394     using vertex_descriptor = typename G::vertex_descriptor;
395     vertex_descriptor selected = vertex_descriptor::invalid();
396     for ( auto vertex : graph.vertices() ) {
397       const usize slot = static_cast<usize>(vertex.value);
398       if ( !active.data()[slot] ) continue;
399       if ( !selected.valid() || degree.data()[slot] < degree.data()[static_cast<usize>(selected.value)] ) selected = vertex;
400     }
401     if ( !selected.valid() ) break;
402     const usize ss = static_cast<usize>(selected.value);
403     if ( degree.data()[ss] > current_core ) current_core = degree.data()[ss];
404     core.data()[ss] = current_core;
405     active.data()[ss] = 0;
406     for ( auto neighbor : graph.out_neighbors(selected) ) {
407       usize &d = degree.data()[static_cast<usize>(neighbor.value)];
408       if ( active.data()[static_cast<usize>(neighbor.value)] && d > current_core ) --d;
409     }
410   }
411   return core;
412 }

423 template<graph_model G>
424 [[nodiscard]] f64
425 global_efficiency(const G &graph)
426 {
427   const usize n = graph.vertices_count();
428   if ( n < 2 ) return f64(0);
429   f64 sum = 0;
430   for ( auto source : graph.vertices() ) {
431     auto paths = unweighted_shortest_paths(graph, source);
432     for ( auto target : graph.vertices() ) {
433       if ( target == source || !paths.contains(target) ) continue;
434       const usize distance = paths.distance.data()[static_cast<usize>(target.value)];
435       if ( distance ) sum += f64(1) / static_cast<f64>(distance);
436     }
437   }
438   return sum / static_cast<f64>(n * (n - 1));
439 }

441 template<graph_model G>
442 [[nodiscard]] f64
443 degree_assortativity(const G &graph)
444 {
445   f64 sum_x = 0;
446   f64 sum_y = 0;
447   f64 sum_xx = 0;
448   f64 sum_yy = 0;
449   f64 sum_xy = 0;
450   usize samples = 0;
451   auto account = [&](f64 x, f64 y) {
452     sum_x += x;
453     sum_y += y;
454     sum_xx += x * x;
455     sum_yy += y * y;
456     sum_xy += x * y;
457     ++samples;
458   };
459   for ( auto edge : graph.edges() ) {
460     const f64 source_degree = static_cast<f64>(G::is_directed ? graph.out_degree(edge.source) : graph.degree(edge.source));
461     const f64 target_degree = static_cast<f64>(G::is_directed ? graph.in_degree(edge.target) : graph.degree(edge.target));
462     account(source_degree, target_degree);
463     if constexpr ( !G::is_directed ) account(target_degree, source_degree);
464   }
465   if ( samples == 0 ) return f64(0);
466   const f64 count = static_cast<f64>(samples);
467   const f64 covariance = count * sum_xy - sum_x * sum_y;
468   const f64 variance_x = count * sum_xx - sum_x * sum_x;
469   const f64 variance_y = count * sum_yy - sum_y * sum_y;
470   const f64 denominator = __builtin_sqrt(variance_x * variance_y);
471   return denominator == f64(0) ? f64(0) : covariance / denominator;
472 }

484 template<graph_model G>
485 [[nodiscard]] iterative_scores_result<f64>
486 eigenvector_centrality(const G &graph, f64 tolerance = 1e-10, usize max_iterations = 100)
487 {
488   iterative_scores_result<f64> result;
489   result.score = micron::vector<f64, micron::allocator_serial<>, false>(graph.vertex_slots(), f64(0));
490   if ( graph.vertices_count() == 0 ) return result;
491   const f64 initial = f64(1) / __builtin_sqrt(static_cast<f64>(graph.vertices_count()));
492   for ( auto vertex : graph.vertices() ) result.score.data()[static_cast<usize>(vertex.value)] = initial;
493   micron::vector<f64, micron::allocator_serial<>, false> next(graph.vertex_slots(), f64(0));
494   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
495     next.fill(f64(0));
496     for ( auto edge : graph.edges() ) {
497       next.data()[static_cast<usize>(edge.target.value)] += result.score.data()[static_cast<usize>(edge.source.value)];
498       if constexpr ( !G::is_directed )
499         next.data()[static_cast<usize>(edge.source.value)] += result.score.data()[static_cast<usize>(edge.target.value)];
500     }
501     f64 norm = 0;
502     for ( auto vertex : graph.vertices() ) {
503       const f64 value = next.data()[static_cast<usize>(vertex.value)];
504       norm += value * value;
505     }
506     norm = __builtin_sqrt(norm);
507     if ( norm == f64(0) ) return result;
508     f64 residual = 0;
509     for ( auto vertex : graph.vertices() ) {
510       const usize slot = static_cast<usize>(vertex.value);
511       next.data()[slot] /= norm;
512       f64 difference = next.data()[slot] - result.score.data()[slot];
513       if ( difference < 0 ) difference = -difference;
514       residual += difference;
515       result.score.data()[slot] = next.data()[slot];
516     }
517     result.iterations = iteration + 1;
518     result.residual = residual;
519     if ( residual <= tolerance ) return result;
520   }
521   result.status = algorithm_status::non_convergent;
522   return result;
523 }

525 template<graph_model G>
526 [[nodiscard]] iterative_scores_result<f64>
527 katz_centrality(const G &graph, f64 alpha = 0.1, f64 beta = 1.0, f64 tolerance = 1e-10, usize max_iterations = 100)
528 {
529   iterative_scores_result<f64> result;
530   result.score = micron::vector<f64, micron::allocator_serial<>, false>(graph.vertex_slots(), f64(0));
531   if ( __impl::nonfinite_weight(alpha) || __impl::nonfinite_weight(beta) || alpha < f64(0) ) {
532     result.status = algorithm_status::invalid_weight;
533     return result;
534   }
535   for ( auto vertex : graph.vertices() ) result.score.data()[static_cast<usize>(vertex.value)] = beta;
536   micron::vector<f64, micron::allocator_serial<>, false> next(graph.vertex_slots(), f64(0));
537   for ( usize iteration = 0; iteration < max_iterations; ++iteration ) {
538     next.fill(f64(0));
539     for ( auto vertex : graph.vertices() ) next.data()[static_cast<usize>(vertex.value)] = beta;
540     for ( auto edge : graph.edges() ) {
541       next.data()[static_cast<usize>(edge.target.value)] += alpha * result.score.data()[static_cast<usize>(edge.source.value)];
542       if constexpr ( !G::is_directed )
543         next.data()[static_cast<usize>(edge.source.value)] += alpha * result.score.data()[static_cast<usize>(edge.target.value)];
544     }
545     f64 residual = 0;
546     for ( auto vertex : graph.vertices() ) {
547       const usize slot = static_cast<usize>(vertex.value);
548       if ( __impl::nonfinite_weight(next.data()[slot]) ) {
549         result.status = algorithm_status::overflow;
550         return result;
551       }
552       f64 difference = next.data()[slot] - result.score.data()[slot];
553       if ( difference < 0 ) difference = -difference;
554       residual += difference;
555       result.score.data()[slot] = next.data()[slot];
556     }
557     result.iterations = iteration + 1;
558     result.residual = residual;
559     if ( residual <= tolerance ) return result;
560   }
561   result.status = algorithm_status::non_convergent;
562   return result;
563 }

587 template<graph_model G>
588 [[nodiscard]] f64
589 rich_club_coefficient(const G &graph, usize degree_threshold)
590 {
591   micron::vector<u8, micron::allocator_serial<>, false> selected(graph.vertex_slots(), u8(0));
592   usize vertices = 0;
593   for ( auto vertex : graph.vertices() )
594     if ( graph.degree(vertex) > degree_threshold ) {
595       selected.data()[static_cast<usize>(vertex.value)] = 1;
596       ++vertices;
597     }
598   if ( vertices < 2 ) return f64(0);
599   usize edges = 0;
600   for ( auto edge : graph.edges() )
601     if ( selected.data()[static_cast<usize>(edge.source.value)] && selected.data()[static_cast<usize>(edge.target.value)] ) ++edges;
602   const usize possible = G::is_directed ? vertices * (vertices - 1) : vertices * (vertices - 1) / 2;
603   return possible ? static_cast<f64>(edges) / static_cast<f64>(possible) : f64(0);
604 }

606 template<graph_model G>
607 [[nodiscard]] f64
608 adamic_adar_index(const G &graph, typename G::vertex_descriptor a, typename G::vertex_descriptor b)
609 {
610   if ( !graph.has_vertex(a) || !graph.has_vertex(b) ) return f64(0);
611   f64 result = 0;
612   for ( auto neighbor : graph.out_neighbors(a) ) {
613     if ( !graph.has_edge(b, neighbor) && !graph.has_edge(neighbor, b) ) continue;
614     const usize degree = graph.degree(neighbor);
615     if ( degree > 1 ) result += f64(1) / __builtin_log(static_cast<f64>(degree));
616   }
617   return result;
618 }
