369   [[nodiscard]] edge_descriptor
370   __find_edge(usize u, usize v) const noexcept
371   {
372     if ( !__live_vertex(u) || !__live_vertex(v) ) return edge_descriptor::invalid();
373     if constexpr ( __edge_list ) {
374       for ( usize i = 0; i < __edges.size(); ++i ) {
375         const edge_slot &slot = __edges.data()[i];
376         if ( !slot.live ) continue;
377         if ( static_cast<usize>(slot.source) == u && static_cast<usize>(slot.target) == v ) return edge_descriptor(static_cast<Index>(i));
378         if constexpr ( is_undirected )
379           if ( static_cast<usize>(slot.source) == v && static_cast<usize>(slot.target) == u ) return edge_descriptor(static_cast<Index>(i));
380       }
381       return edge_descriptor::invalid();
382     }
383     const auto &arcs = __vertices.data()[u].out;
384     for ( usize i = 0; i < arcs.size(); ++i ) {
385       const arc_type &arc = arcs.data()[i];
386       if ( static_cast<usize>(arc.neighbor) == v && __live_edge(static_cast<usize>(arc.edge)) ) return edge_descriptor(arc.edge);

394   {
395     const usize needed = values.size() + count;
396     if ( needed <= values.max_size() ) return;

398     usize grown = capacity < 4 ? 4 : capacity;

458   template<typename P>
459   edge_insert_result<Index>
460   __insert_edge(Index ui, Index vi, P &&property, bool auto_create)
461   {
462     const usize u = static_cast<usize>(ui);
463     const usize v = static_cast<usize>(vi);
464     const usize high = u > v ? u : v;
465 
466     if ( !__slot_fits(high) || !__slot_fits(__edges.size()) )
467       return { graphs::edge_insert_status::index_overflow, edge_descriptor::invalid() };
468     if constexpr ( !allows_loops ) {
469       if ( u == v ) return { graphs::edge_insert_status::self_loop, edge_descriptor::invalid() };
470     }
471 
472     if ( u < __vertices.size() && !__vertices.data()[u].live )
473       return { graphs::edge_insert_status::invalid_vertex, edge_descriptor::invalid() };
474     if ( v < __vertices.size() && !__vertices.data()[v].live )
475       return { graphs::edge_insert_status::invalid_vertex, edge_descriptor::invalid() };
476     if ( high >= __vertices.size() && !auto_create ) return { graphs::edge_insert_status::invalid_vertex, edge_descriptor::invalid() };
477     if ( high >= __vertices.size() ) {
478       if constexpr ( !micron::is_default_constructible_v<VertexProperty> )
479         return { graphs::edge_insert_status::property_required, edge_descriptor::invalid() };
480     }
481 
482     if constexpr ( is_simple ) {
483       if ( u < __vertices.size() && v < __vertices.size() ) {
484         const edge_descriptor duplicate = __find_edge(u, v);
485         if ( duplicate.valid() ) return { graphs::edge_insert_status::duplicate, duplicate };
486       }
487     }
488 
489     if ( high >= __vertices.size() ) {
490       if constexpr ( micron::is_default_constructible_v<VertexProperty> ) {
491         __vertices.reserve(high + 1);
492         while ( __vertices.size() <= high ) {
493           __vertices.emplace_back();
494           ++__live_vertices;
495         }
496       } else {
497         return { graphs::edge_insert_status::property_required, edge_descriptor::invalid() };
498       }
499     }
500 
501     __reserve_append(__edges, 1);
502     if constexpr ( !__edge_list ) {
503       if constexpr ( is_directed ) {
504         __reserve_append(__vertices.data()[u].out, 1);
505         __reserve_append(__vertices.data()[v].in, 1);
506       } else if ( u == v ) {
507         __reserve_append(__vertices.data()[u].out, 2);
508       } else {
509         __reserve_append(__vertices.data()[u].out, 1);
510         __reserve_append(__vertices.data()[v].out, 1);
511       }
512     }
513 
514     const Index raw_id = static_cast<Index>(__edges.size());
515     __edges.emplace_back(ui, vi, micron::forward<P>(property));
516     if constexpr ( !__edge_list ) {
517       if constexpr ( is_directed ) {
518         __vertices.data()[u].out.push_back(arc_type{ raw_id, vi });
519         __vertices.data()[v].in.push_back(arc_type{ raw_id, ui });
520       } else {
521         __vertices.data()[u].out.push_back(arc_type{ raw_id, vi });
522         __vertices.data()[v].out.push_back(arc_type{ raw_id, ui });
523       }
524     }
525     ++__live_edges;
526     return { graphs::edge_insert_status::inserted, edge_descriptor(raw_id) };
527   }

651   class __neighbor_iterator
652   {
653     const __basic_adjacency_graph *__graph{};
654     usize __vertex{};
655     usize __arc{};
656     __arc_direction __direction{};
657     bool __repeat_loop{};
658 
659     [[nodiscard]] const auto &
660     __arcs() const noexcept
661     {
662       if constexpr ( is_directed ) {
663         if ( __direction == __arc_direction::in ) return __graph->__vertices.data()[__vertex].in;
664       }
665       return __graph->__vertices.data()[__vertex].out;
666     }
667 
668     void
669     __skip() noexcept
670     {
671       if constexpr ( __edge_list ) {
672         __repeat_loop = false;
673         while ( __arc < __graph->__edges.size() ) {
674           const edge_slot &slot = __graph->__edges.data()[__arc];
675           if ( slot.live ) {
676             const bool match
677                 = __direction == __arc_direction::in && is_directed
678                       ? static_cast<usize>(slot.target) == __vertex
679                       : static_cast<usize>(slot.source) == __vertex || (is_undirected && static_cast<usize>(slot.target) == __vertex);
680             if ( match ) {
681               if constexpr ( is_undirected && allows_loops ) __repeat_loop = slot.source == slot.target;
682               break;
683             }
684           }
685           ++__arc;
686         }
687         return;
688       }
689       const auto &arcs = __arcs();
690       while ( __arc < arcs.size() ) {
691         const arc_type &a = arcs.data()[__arc];
692         if ( __graph->__live_edge(static_cast<usize>(a.edge)) && __graph->__live_vertex(static_cast<usize>(a.neighbor)) ) break;
693         ++__arc;
694       }
695     }

726         }

1282       for ( usize i = 0; i < __edges.size(); ++i ) {
1283         edge_slot &slot = __edges.data()[i];
1284         if ( slot.live && (slot.source == raw || slot.target == raw) ) {
1285           slot.make_dead();
1286           --__live_edges;
1287         }
1288       }
1289       __vertices.data()[static_cast<usize>(raw)].make_dead();
1290       --__live_vertices;
1291       return true;
1292     }
1293 
1294     for ( usize i = 0; i < __edges.size(); ) {
1295       const edge_slot &slot = __edges.data()[i];
1296       if ( slot.source == raw || slot.target == raw )
1297         (void)remove_edge(edge_descriptor(static_cast<Index>(i)));
1298       else
1299         ++i;
1300     }
1301 
1302     const usize erased = static_cast<usize>(raw);
1303     const usize last = __vertices.size() - 1;
1304     if ( erased != last ) {
1305       const Index old_vertex = static_cast<Index>(last);
1306       __vertices.data()[erased] = micron::move(__vertices.data()[last]);
1307       for ( usize i = 0; i < __edges.size(); ++i ) {
1308         edge_slot &slot = __edges.data()[i];
1309         if ( slot.source == old_vertex ) slot.source = raw;
1310         if ( slot.target == old_vertex ) slot.target = raw;
1311       }
1312       if constexpr ( !__edge_list ) {
1313         for ( usize i = 0; i < __vertices.size(); ++i ) {
1314           auto &out = __vertices.data()[i].out;
1315           for ( usize j = 0; j < out.size(); ++j )
1316             if ( out.data()[j].neighbor == old_vertex ) out.data()[j].neighbor = raw;
1317           if constexpr ( is_directed ) {
1318             auto &in = __vertices.data()[i].in;
1319             for ( usize j = 0; j < in.size(); ++j )
1320               if ( in.data()[j].neighbor == old_vertex ) in.data()[j].neighbor = raw;
1321           }
1322         }
1323       }
1324     }
1325     __vertices.pop_back();
1326     --__live_vertices;
1327     return true;
1328   }

1618   [[nodiscard]] compact_result<Index>
1619   compact()
1620   {
1621     compact_result<Index> remap{
1622       micron::vector<vertex_descriptor, micron::allocator_serial<>, false>(__vertices.size(), vertex_descriptor::invalid()),
1623       micron::vector<edge_descriptor, micron::allocator_serial<>, false>(__edges.size(), edge_descriptor::invalid())
1624     };
1625 
1626     __basic_adjacency_graph replacement(micron::move(__property));
1627     replacement.reserve_vertices(__live_vertices);
1628     replacement.reserve_edges(__live_edges);
1629     for ( usize i = 0; i < __vertices.size(); ++i ) {
1630       if ( !__vertices.data()[i].live ) continue;
1631       remap.vertex_remap.data()[i] = replacement.add_vertex(micron::move(__vertices.data()[i].property));
1632     }
1633     for ( usize i = 0; i < __edges.size(); ++i ) {
1634       edge_slot &slot = __edges.data()[i];
1635       if ( !slot.live ) continue;
1636       const vertex_descriptor u = remap.vertex_remap.data()[static_cast<usize>(slot.source)];
1637       const vertex_descriptor v = remap.vertex_remap.data()[static_cast<usize>(slot.target)];
1638       auto inserted = replacement.add_edge(u, v, micron::move(slot.property));
1639       remap.edge_remap.data()[i] = inserted.id;
1640     }
1641     *this = micron::move(replacement);
1642     return remap;
1643   }
