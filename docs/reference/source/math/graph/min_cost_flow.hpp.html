124 template<typename Cost>
125 [[nodiscard]] bool
126 negate_cost(const Cost &value, Cost &result) noexcept
127 {
128   if constexpr ( micron::is_integral_v<Cost> ) {
129     return !__builtin_sub_overflow(Cost{}, value, &result);
130   } else {
131     result = -value;
132     return !nonfinite_weight(result);
133   }
134 }

136 template<typename Capacity, typename Cost>
137 [[nodiscard]] bool
138 flow_cost(const Capacity &flow, const Cost &cost, Cost &result) noexcept
139 {
140   if constexpr ( micron::is_integral_v<Cost> ) {
141     return !__builtin_mul_overflow(flow, cost, &result);
142   } else {
143     result = static_cast<Cost>(flow) * cost;
144     return !nonfinite_weight(result);
145   }
146 }

148 template<micron::integral Capacity, micron::integral Supply>
149 [[nodiscard]] bool
150 supply_capacity(Supply value, Capacity &result) noexcept
151 {
152   if constexpr ( micron::is_signed_v<Supply> )
153     if ( value < Supply{} ) return false;
154   if ( static_cast<u64>(value) > static_cast<u64>(micron::numeric_limits<Capacity>::max()) ) return false;
155   result = static_cast<Capacity>(value);
156   return true;

159 template<graph_model G, typename CapacityMap, typename CostMap, typename C, typename W>
160 algorithm_status
161 build_cost_residual(const G &graph, CapacityMap &capacity_map, CostMap &cost_map,
162                     min_cost_flow_workspace<typename G::index_type, C, W> &workspace, usize extra_vertices = 0)
163 {
164   using I = typename G::index_type;
165   using arc_type = min_cost_residual_arc<I, C, W>;
166   using raw_capacity_type
167       = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
168   static_assert(micron::is_integral_v<raw_capacity_type>, "min-cost flow capacities must be integral");
169   const usize vertex_slots = graph.vertex_slots();
170   const usize edge_slots = graph.edge_slots();
171   if ( extra_vertices > micron::numeric_limits<usize>::max() - vertex_slots ) return algorithm_status::overflow;
172   const usize vertices = graph.vertex_slots() + extra_vertices;
173   if ( (vertices != 0 && vertices - 1 > static_cast<usize>(micron::numeric_limits<I>::max()))
174        || edge_slots > micron::numeric_limits<usize>::max() - vertices )
175     return algorithm_status::overflow;
176   workspace.reserve(vertices, edge_slots);
177   workspace.residual.clear();
178   workspace.residual.resize(vertices);
179   workspace.location.resize(edge_slots);
180   workspace.original_capacity.resize(edge_slots, C{});
181   workspace.original_capacity.fill(C{});
182   workspace.original_cost.resize(edge_slots, W{});
183   workspace.original_cost.fill(W{});
184   for ( auto edge : graph.edges() ) {
185     const raw_capacity_type raw_capacity = __impl::weight(capacity_map, graph, edge.id);
186     if ( __impl::invalid_weight(raw_capacity) ) return algorithm_status::invalid_weight;
187     if ( static_cast<u64>(raw_capacity) > static_cast<u64>(micron::numeric_limits<C>::max()) ) return algorithm_status::overflow;
188     const C capacity = static_cast<C>(raw_capacity);
189     const W cost = static_cast<W>(__impl::weight(cost_map, graph, edge.id));
190     if ( __impl::nonfinite_weight(cost) ) return algorithm_status::invalid_weight;
191     W reverse_cost{};
192     if ( !negate_cost(cost, reverse_cost) ) return algorithm_status::overflow;
193     const usize u = static_cast<usize>(edge.source.value);
194     const usize v = static_cast<usize>(edge.target.value);
195     const usize forward = workspace.residual.data()[u].size();
196     const usize reverse = u == v ? forward + 1 : workspace.residual.data()[v].size();
197     workspace.residual.data()[u].push_back(arc_type{ edge.target.value, reverse, capacity, cost, edge.id, true, false });
198     workspace.residual.data()[v].push_back(arc_type{ edge.source.value, forward, C{}, reverse_cost, edge.id, false, false });
199     workspace.location.data()[static_cast<usize>(edge.id.value)] = { edge.source.value, forward };
200     workspace.original_capacity.data()[static_cast<usize>(edge.id.value)] = capacity;
201     workspace.original_cost.data()[static_cast<usize>(edge.id.value)] = cost;
202   }
203   return algorithm_status::ok;
204 }

206 template<typename I, typename C, typename W>
207 algorithm_status
208 add_residual_arc(min_cost_flow_workspace<I, C, W> &workspace, usize source, usize target, C capacity, W cost, bool auxiliary)
209 {
210   using arc_type = min_cost_residual_arc<I, C, W>;
211   W reverse_cost{};
212   if ( !negate_cost(cost, reverse_cost) ) return algorithm_status::overflow;
213   const usize forward = workspace.residual.data()[source].size();
214   const usize reverse = source == target ? forward + 1 : workspace.residual.data()[target].size();
215   workspace.residual.data()[source].push_back(
216       arc_type{ static_cast<I>(target), reverse, capacity, cost, edge_id<I>::invalid(), false, auxiliary });
217   workspace.residual.data()[target].push_back(
218       arc_type{ static_cast<I>(source), forward, C{}, reverse_cost, edge_id<I>::invalid(), false, auxiliary });
219   return algorithm_status::ok;
220 }

222 template<typename I, typename C, typename W>
223 algorithm_status
224 dinic_residual(min_cost_flow_workspace<I, C, W> &workspace, usize source, usize sink, C limit, bool limited, C &achieved)
225 {
226   const usize vertices = workspace.residual.size();
227   workspace.level.resize(vertices, max_t(-1));
228   workspace.next.resize(vertices, usize(0));
229   auto levels = [&]() {
230     workspace.level.fill(max_t(-1));
231     workspace.queue.clear();
232     workspace.queue.push_back(static_cast<I>(source));
233     workspace.level.data()[source] = 0;
234     for ( usize head = 0; head < workspace.queue.size(); ++head ) {
235       const usize u = static_cast<usize>(workspace.queue.data()[head]);
236       for ( const auto &arc : workspace.residual.data()[u] ) {
237         const usize v = static_cast<usize>(arc.target);
238         if ( arc.capacity > C{} && workspace.level.data()[v] < 0 ) {
239           workspace.level.data()[v] = workspace.level.data()[u] + 1;
240           workspace.queue.push_back(arc.target);
241         }
242       }
243     }
244     return workspace.level.data()[sink] >= 0;
245   };
246   bool overflow = false;
247   auto send = [&](auto &&self, usize u, C offered) -> C {
248     if ( u == sink || offered == C{} ) return offered;
249     auto &arcs = workspace.residual.data()[u];
250     for ( usize &at = workspace.next.data()[u]; at < arcs.size(); ++at ) {
251       auto &arc = arcs.data()[at];
252       const usize v = static_cast<usize>(arc.target);
253       if ( arc.capacity <= C{} || workspace.level.data()[v] != workspace.level.data()[u] + 1 ) continue;
254       const C delivered = self(self, v, offered < arc.capacity ? offered : arc.capacity);
255       if ( delivered == C{} ) continue;
256       C reverse{};
257       if ( !__impl::add_distance(workspace.residual.data()[v].data()[arc.reverse].capacity, delivered, reverse) ) {
258         overflow = true;
259         return C{};
260       }
261       arc.capacity -= delivered;
262       workspace.residual.data()[v].data()[arc.reverse].capacity = reverse;
263       return delivered;
264     }
265     return C{};
266   };
267   C source_bound{};
268   for ( const auto &arc : workspace.residual.data()[source] ) {
269     C sum{};
270     if ( !__impl::add_distance(source_bound, arc.capacity, sum) ) return algorithm_status::overflow;
271     source_bound = sum;
272   }
273   while ( (!limited || achieved < limit) && levels() ) {
274     workspace.next.fill(usize(0));
275     for ( ;; ) {
276       C offered = source_bound;
277       if ( limited ) offered = limit - achieved < offered ? limit - achieved : offered;
278       if ( offered == C{} ) break;
279       const C pushed = send(send, source, offered);
280       if ( overflow ) return algorithm_status::overflow;
281       if ( pushed == C{} ) break;
282       C total{};
283       if ( !__impl::add_distance(achieved, pushed, total) ) return algorithm_status::overflow;
284       achieved = total;
285     }
286   }
287   return algorithm_status::ok;
288 }

290 template<typename I, typename C, typename W>
291 algorithm_status
292 reduced_cost_augment(min_cost_flow_workspace<I, C, W> &workspace, usize source, usize sink, C limit, bool limited, C &achieved,
293                      bool &applicable)
294 {
295   const usize vertices = workspace.residual.size();
296   applicable = false;
297   workspace.potential.resize(vertices, W{});
298   workspace.potential.fill(W{});
299   for ( usize pass = 0; pass < vertices; ++pass ) {
300     bool changed = false;
301     for ( usize u = 0; u < vertices; ++u )
302       for ( const auto &arc : workspace.residual.data()[u] ) {
303         const usize v = static_cast<usize>(arc.target);
304         if ( v >= vertices || arc.capacity <= C{} || arc.auxiliary ) continue;
305         W candidate{};
306         if ( !__impl::add_distance(workspace.potential.data()[u], arc.cost, candidate) ) return algorithm_status::overflow;
307         if ( candidate < workspace.potential.data()[v] ) {
308           workspace.potential.data()[v] = candidate;
309           changed = true;
310           if ( pass + 1 == vertices ) return algorithm_status::ok;
311         }
312       }
313     if ( !changed ) break;
314   }
315   applicable = true;
316 
317   workspace.distance.resize(vertices, W{});
318   workspace.parent_vertex.resize(vertices, vertex_id<I>::invalid_value());
319   workspace.parent_arc.resize(vertices, usize(0));
320   workspace.reached.resize(vertices, u8(0));
321   workspace.settled.resize(vertices, u8(0));
322 
323   auto push_heap = [&](I vertex, W distance) {
324     workspace.heap_vertex.push_back(vertex);
325     workspace.heap_distance.push_back(distance);
326     usize child = workspace.heap_vertex.size() - 1;
327     while ( child ) {
328       const usize parent = (child - 1) / 2;
329       if ( !(workspace.heap_distance.data()[child] < workspace.heap_distance.data()[parent]) ) break;
330       micron::swap(workspace.heap_vertex.data()[child], workspace.heap_vertex.data()[parent]);
331       micron::swap(workspace.heap_distance.data()[child], workspace.heap_distance.data()[parent]);
332       child = parent;
333     }
334   };
335   auto pop_heap = [&]() {
336     const I vertex = workspace.heap_vertex.data()[0];
337     const W distance = workspace.heap_distance.data()[0];
338     const usize last = workspace.heap_vertex.size() - 1;
339     workspace.heap_vertex.data()[0] = workspace.heap_vertex.data()[last];
340     workspace.heap_distance.data()[0] = workspace.heap_distance.data()[last];
341     workspace.heap_vertex.pop_back();
342     workspace.heap_distance.pop_back();
343     usize parent = 0;
344     while ( parent < workspace.heap_vertex.size() ) {
345       const usize left = parent * 2 + 1;
346       if ( left >= workspace.heap_vertex.size() ) break;
347       const usize right = left + 1;
348       const usize child
349           = right < workspace.heap_vertex.size() && workspace.heap_distance.data()[right] < workspace.heap_distance.data()[left] ? right
350                                                                                                                                  : left;
351       if ( !(workspace.heap_distance.data()[child] < workspace.heap_distance.data()[parent]) ) break;
352       micron::swap(workspace.heap_vertex.data()[child], workspace.heap_vertex.data()[parent]);
353       micron::swap(workspace.heap_distance.data()[child], workspace.heap_distance.data()[parent]);
354       parent = child;
355     }
356     return micron::pair<I, W>{ vertex, distance };
357   };
358 
359   while ( !limited || achieved < limit ) {
360     workspace.distance.fill(W{});
361     workspace.parent_vertex.fill(vertex_id<I>::invalid_value());
362     workspace.reached.fill(u8(0));
363     workspace.settled.fill(u8(0));
364     workspace.heap_vertex.clear();
365     workspace.heap_distance.clear();
366     workspace.reached.data()[source] = 1;
367     push_heap(static_cast<I>(source), W{});
368     while ( !workspace.heap_vertex.empty() ) {
369       const auto selected = pop_heap();
370       const usize u = static_cast<usize>(selected.a);
371       if ( workspace.settled.data()[u] || selected.b != workspace.distance.data()[u] ) continue;
372       workspace.settled.data()[u] = 1;
373       if ( u == sink ) break;
374       const auto &arcs = workspace.residual.data()[u];
375       for ( usize at = 0; at < arcs.size(); ++at ) {
376         const auto &arc = arcs.data()[at];
377         const usize v = static_cast<usize>(arc.target);
378         if ( v >= vertices || arc.capacity <= C{} || arc.auxiliary ) continue;
379         W partial{}, reduced{};
380         if ( !__impl::add_distance(arc.cost, workspace.potential.data()[u], partial)
381              || !__impl::sub_distance(partial, workspace.potential.data()[v], reduced) )
382           return algorithm_status::overflow;
383         if ( reduced < W{} ) {
384           applicable = false;
385           return algorithm_status::ok;
386         }
387         W candidate{};
388         if ( !__impl::add_distance(workspace.distance.data()[u], reduced, candidate) ) return algorithm_status::overflow;
389         if ( !workspace.reached.data()[v] || candidate < workspace.distance.data()[v] ) {
390           workspace.reached.data()[v] = 1;
391           workspace.distance.data()[v] = candidate;
392           workspace.parent_vertex.data()[v] = static_cast<I>(u);
393           workspace.parent_arc.data()[v] = at;
394           push_heap(static_cast<I>(v), candidate);
395         }
396       }
397     }
398     if ( !workspace.reached.data()[sink] ) return algorithm_status::ok;
399     for ( usize vertex = 0; vertex < vertices; ++vertex ) {
400       if ( !workspace.reached.data()[vertex] ) continue;
401       W updated{};
402       if ( !__impl::add_distance(workspace.potential.data()[vertex], workspace.distance.data()[vertex], updated) )
403         return algorithm_status::overflow;
404       workspace.potential.data()[vertex] = updated;
405     }
406 
407     C pushed{};
408     bool have = false;
409     usize current = sink;
410     while ( current != source ) {
411       const I parent_value = workspace.parent_vertex.data()[current];
412       if ( parent_value == vertex_id<I>::invalid_value() ) return algorithm_status::invalid_graph;
413       const usize parent = static_cast<usize>(parent_value);
414       const auto &arc = workspace.residual.data()[parent].data()[workspace.parent_arc.data()[current]];
415       if ( !have || arc.capacity < pushed ) {
416         pushed = arc.capacity;
417         have = true;
418       }
419       current = parent;
420     }
421     if ( limited ) {
422       const C remaining = limit - achieved;
423       if ( remaining < pushed ) pushed = remaining;
424     }
425     if ( !have || pushed == C{} ) return algorithm_status::invalid_graph;
426     current = sink;
427     while ( current != source ) {
428       const usize parent = static_cast<usize>(workspace.parent_vertex.data()[current]);
429       auto &arc = workspace.residual.data()[parent].data()[workspace.parent_arc.data()[current]];
430       C reverse{};
431       if ( !__impl::add_distance(workspace.residual.data()[current].data()[arc.reverse].capacity, pushed, reverse) )
432         return algorithm_status::overflow;
433       arc.capacity -= pushed;
434       workspace.residual.data()[current].data()[arc.reverse].capacity = reverse;
435       current = parent;
436     }
437     C total{};
438     if ( !__impl::add_distance(achieved, pushed, total) ) return algorithm_status::overflow;
439     achieved = total;
440   }
441   return algorithm_status::ok;
442 }

444 template<typename I, typename C, typename W>
445 algorithm_status
446 cancel_negative_cycles(min_cost_flow_workspace<I, C, W> &workspace, usize vertices)
447 {
448   workspace.distance.resize(vertices, W{});
449   workspace.parent_vertex.resize(vertices, vertex_id<I>::invalid_value());
450   workspace.parent_arc.resize(vertices, usize(0));
451   for ( ;; ) {
452     workspace.distance.fill(W{});
453     workspace.parent_vertex.fill(vertex_id<I>::invalid_value());
454     I changed = vertex_id<I>::invalid_value();
455     for ( usize pass = 0; pass < vertices; ++pass ) {
456       changed = vertex_id<I>::invalid_value();
457       for ( usize u = 0; u < vertices; ++u ) {
458         const auto &arcs = workspace.residual.data()[u];
459         for ( usize at = 0; at < arcs.size(); ++at ) {
460           const auto &arc = arcs.data()[at];
461           const usize v = static_cast<usize>(arc.target);
462           if ( v >= vertices || arc.capacity <= C{} || arc.auxiliary ) continue;
463           W candidate{};
464           if ( !__impl::add_distance(workspace.distance.data()[u], arc.cost, candidate) ) return algorithm_status::overflow;
465           if ( candidate < workspace.distance.data()[v] ) {
466             workspace.distance.data()[v] = candidate;
467             workspace.parent_vertex.data()[v] = static_cast<I>(u);
468             workspace.parent_arc.data()[v] = at;
469             changed = static_cast<I>(v);
470           }
471         }
472       }
473       if ( changed == vertex_id<I>::invalid_value() ) break;
474     }
475     if ( changed == vertex_id<I>::invalid_value() ) return algorithm_status::ok;
476     usize cycle = static_cast<usize>(changed);
477     for ( usize i = 0; i < vertices; ++i ) {
478       const I parent = workspace.parent_vertex.data()[cycle];
479       if ( parent == vertex_id<I>::invalid_value() ) return algorithm_status::invalid_graph;
480       cycle = static_cast<usize>(parent);
481     }
482     C bottleneck{};
483     bool have = false;
484     usize current = cycle;
485     do {
486       const usize parent = static_cast<usize>(workspace.parent_vertex.data()[current]);
487       const auto &arc = workspace.residual.data()[parent].data()[workspace.parent_arc.data()[current]];
488       if ( !have || arc.capacity < bottleneck ) {
489         bottleneck = arc.capacity;
490         have = true;
491       }
492       current = parent;
493     } while ( current != cycle );
494     if ( !have || bottleneck == C{} ) return algorithm_status::invalid_graph;
495     current = cycle;
496     do {
497       const usize parent = static_cast<usize>(workspace.parent_vertex.data()[current]);
498       auto &arc = workspace.residual.data()[parent].data()[workspace.parent_arc.data()[current]];
499       C reverse{};
500       if ( !__impl::add_distance(workspace.residual.data()[current].data()[arc.reverse].capacity, bottleneck, reverse) )
501         return algorithm_status::overflow;
502       arc.capacity -= bottleneck;
503       workspace.residual.data()[current].data()[arc.reverse].capacity = reverse;
504       current = parent;
505     } while ( current != cycle );
506   }
507 }

509 template<typename I, typename C, typename W>
510 algorithm_status
511 negative_cycle_status(min_cost_flow_workspace<I, C, W> &workspace, usize vertices)
512 {
513   workspace.distance.resize(vertices, W{});
514   workspace.distance.fill(W{});
515   for ( usize pass = 0; pass < vertices; ++pass ) {
516     bool changed = false;
517     for ( usize u = 0; u < vertices; ++u )
518       for ( const auto &arc : workspace.residual.data()[u] ) {
519         const usize v = static_cast<usize>(arc.target);
520         if ( v >= vertices || arc.capacity <= C{} || arc.auxiliary ) continue;
521         W candidate{};
522         if ( !__impl::add_distance(workspace.distance.data()[u], arc.cost, candidate) ) return algorithm_status::overflow;
523         if ( candidate < workspace.distance.data()[v] ) {
524           workspace.distance.data()[v] = candidate;
525           changed = true;
526           if ( pass + 1 == vertices ) return algorithm_status::negative_cycle;
527         }
528       }
529     if ( !changed ) break;
530   }
531   return algorithm_status::ok;
532 }

534 template<graph_model G, typename C, typename W>
535 algorithm_status
536 finish_cost_result(const G &graph, min_cost_flow_workspace<typename G::index_type, C, W> &workspace,
537                    min_cost_flow_result<typename G::index_type, C, W> &result)
538 {
539   result.flow.resize(graph.edge_slots(), C{});
540   result.flow.fill(C{});
541   result.incoming.resize(graph.vertex_slots(), C{});
542   result.incoming.fill(C{});
543   result.outgoing.resize(graph.vertex_slots(), C{});
544   result.outgoing.fill(C{});
545   result.total_cost = W{};
546   for ( auto edge : graph.edges() ) {
547     const usize slot = static_cast<usize>(edge.id.value);
548     const auto location = workspace.location.data()[slot];
549     const C remaining = workspace.residual.data()[static_cast<usize>(location.source)].data()[location.arc].capacity;
550     const C flow = workspace.original_capacity.data()[slot] - remaining;
551     result.flow.data()[slot] = flow;
552     C sum{};
553     const usize u = static_cast<usize>(edge.source.value);
554     const usize v = static_cast<usize>(edge.target.value);
555     if ( !__impl::add_distance(result.outgoing.data()[u], flow, sum) ) return algorithm_status::overflow;
556     result.outgoing.data()[u] = sum;
557     if ( !__impl::add_distance(result.incoming.data()[v], flow, sum) ) return algorithm_status::overflow;
558     result.incoming.data()[v] = sum;
559     W term{}, total{};
560     if ( !flow_cost(flow, workspace.original_cost.data()[slot], term) || !__impl::add_distance(result.total_cost, term, total) )
561       return algorithm_status::overflow;
562     result.total_cost = total;
563   }
564   return algorithm_status::ok;
565 }

569 template<graph_model G, typename Capacity, typename CapacityMap, typename CostMap>
570 [[nodiscard]] auto
571 min_cost_flow(
572     const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink, Capacity requested, CapacityMap capacity_map,
573     CostMap cost_map,
574     min_cost_flow_workspace<typename G::index_type, Capacity,
575                             micron::remove_cvref_t<decltype(__impl::weight(micron::declval<CostMap &>(), micron::declval<const G &>(),
576                                                                            micron::declval<typename G::edge_descriptor>()))>> &workspace)
577   requires micron::integral<Capacity>
578 {
579   using I = typename G::index_type;
580   using W = micron::remove_cvref_t<decltype(__impl::weight(cost_map, graph, micron::declval<typename G::edge_descriptor>()))>;
581   min_cost_flow_result<I, Capacity, W> result;
582   if constexpr ( !G::is_directed ) {
583     result.status = algorithm_status::invalid_graph;
584     return result;
585   }
586   if ( !graph.has_vertex(source) || !graph.has_vertex(sink) || source == sink ) {
587     result.status = algorithm_status::invalid_vertex;
588     return result;
589   }
590   if constexpr ( micron::is_signed_v<Capacity> )
591     if ( requested < Capacity{} ) {
592       result.status = algorithm_status::invalid_weight;
593       return result;
594     }
595   result.status = __impl::build_cost_residual<G, CapacityMap, CostMap, Capacity, W>(graph, capacity_map, cost_map, workspace);
596   if ( result.status != algorithm_status::ok ) return result;
597   bool reduced_costs = false;
598   result.status = __impl::reduced_cost_augment(workspace, static_cast<usize>(source.value), static_cast<usize>(sink.value), requested, true,
599                                                result.achieved_flow, reduced_costs);
600   if ( result.status == algorithm_status::ok && !reduced_costs )
601     result.status = __impl::dinic_residual(workspace, static_cast<usize>(source.value), static_cast<usize>(sink.value), requested, true,
602                                            result.achieved_flow);
603   if ( result.status != algorithm_status::ok ) return result;
604   if ( result.achieved_flow != requested ) {
605     result.status = algorithm_status::infeasible;
606     return result;
607   }
608   result.status = __impl::cancel_negative_cycles(workspace, graph.vertex_slots());
609   if ( result.status == algorithm_status::ok ) result.status = __impl::finish_cost_result(graph, workspace, result);
610   return result;
611 }

684 template<graph_model G, typename Supplies, typename CapacityMap, typename CostMap>
685 [[nodiscard]] auto
686 min_cost_circulation(
687     const G &graph, const Supplies &supplies, CapacityMap capacity_map, CostMap cost_map,
688     min_cost_flow_workspace<typename G::index_type,
689                             micron::remove_cvref_t<decltype(__impl::weight(micron::declval<CapacityMap &>(), micron::declval<const G &>(),
690                                                                            micron::declval<typename G::edge_descriptor>()))>,
691                             micron::remove_cvref_t<decltype(__impl::weight(micron::declval<CostMap &>(), micron::declval<const G &>(),
692                                                                            micron::declval<typename G::edge_descriptor>()))>> &workspace)
693 {
694   using I = typename G::index_type;
695   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
696   using W = micron::remove_cvref_t<decltype(__impl::weight(cost_map, graph, micron::declval<typename G::edge_descriptor>()))>;
697   using S = micron::remove_cvref_t<decltype(supplies[usize(0)])>;
698   static_assert(micron::integral<S>, "min_cost_circulation supplies must be integral");
699   min_cost_flow_result<I, C, W> result;
700   if constexpr ( !G::is_directed ) {
701     result.status = algorithm_status::invalid_graph;
702     return result;
703   }
704   const usize vertices = graph.vertex_slots();
705   if ( supplies.size() < vertices ) {
706     result.status = algorithm_status::invalid_graph;
707     return result;
708   }
709   result.status = __impl::build_cost_residual<G, CapacityMap, CostMap, C, W>(graph, capacity_map, cost_map, workspace, 2);
710   if ( result.status != algorithm_status::ok ) return result;
711   const usize super_source = vertices;
712   const usize super_sink = vertices + 1;
713   C total_supply{};
714   C total_demand{};
715   for ( auto vertex : graph.vertices() ) {
716     const usize slot = static_cast<usize>(vertex.value);
717     const auto value = supplies[slot];
718     if ( value > decltype(value){} ) {
719       C amount{};
720       if ( !__impl::supply_capacity(value, amount) ) {
721         result.status = algorithm_status::overflow;
722         return result;
723       }
724       C total{};
725       if ( !__impl::add_distance(total_supply, amount, total) ) {
726         result.status = algorithm_status::overflow;
727         return result;
728       }
729       total_supply = total;
730       result.status = __impl::add_residual_arc(workspace, super_source, slot, amount, W{}, true);
731     } else if ( value < decltype(value){} ) {
732       S positive{};
733       if ( __builtin_sub_overflow(S{}, value, &positive) ) {
734         result.status = algorithm_status::overflow;
735         return result;
736       }
737       C amount{};
738       if ( !__impl::supply_capacity(positive, amount) ) {
739         result.status = algorithm_status::overflow;
740         return result;
741       }
742       C total{};
743       if ( !__impl::add_distance(total_demand, amount, total) ) {
744         result.status = algorithm_status::overflow;
745         return result;
746       }
747       total_demand = total;
748       result.status = __impl::add_residual_arc(workspace, slot, super_sink, amount, W{}, true);
749     }
750     if ( result.status != algorithm_status::ok ) return result;
751   }
752   if ( total_supply != total_demand ) {
753     result.status = algorithm_status::infeasible;
754     return result;
755   }
756   C sent{};
757   result.status = __impl::dinic_residual(workspace, super_source, super_sink, total_supply, true, sent);
758   if ( result.status != algorithm_status::ok ) return result;
759   if ( sent != total_supply ) {
760     result.status = algorithm_status::infeasible;
761     return result;
762   }
763   result.achieved_flow = sent;
764   for ( usize u = 0; u < workspace.residual.size(); ++u )
765     for ( auto &arc : workspace.residual.data()[u] )
766       if ( arc.auxiliary ) arc.capacity = C{};
767   result.status = __impl::cancel_negative_cycles(workspace, vertices);
768   if ( result.status == algorithm_status::ok ) result.status = __impl::finish_cost_result(graph, workspace, result);
769   return result;
770 }

789 template<graph_model G, typename C, typename W, typename CapacityMap, typename CostMap>
790 [[nodiscard]] bool
791 verify_min_cost_flow(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink,
792                      const min_cost_flow_result<typename G::index_type, C, W> &result, CapacityMap capacity_map, CostMap cost_map)
793 {
794   if ( result.status != algorithm_status::ok || result.flow.size() < graph.edge_slots() ) return false;
795   micron::vector<C, micron::allocator_serial<>, false> incoming(graph.vertex_slots(), C{});
796   micron::vector<C, micron::allocator_serial<>, false> outgoing(graph.vertex_slots(), C{});
797   min_cost_flow_workspace<typename G::index_type, C, W> workspace;
798   if ( __impl::build_cost_residual<G, CapacityMap, CostMap, C, W>(graph, capacity_map, cost_map, workspace) != algorithm_status::ok )
799     return false;
800   W total_cost{};
801   for ( auto edge : graph.edges() ) {
802     const usize slot = static_cast<usize>(edge.id.value);
803     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
804     const C flow = result.flow.data()[slot];
805     if constexpr ( micron::is_signed_v<C> )
806       if ( flow < C{} ) return false;
807     if ( flow > capacity ) return false;
808     C capacity_sum{};
809     const usize u = static_cast<usize>(edge.source.value);
810     const usize v = static_cast<usize>(edge.target.value);
811     if ( !__impl::add_distance(outgoing.data()[u], flow, capacity_sum) ) return false;
812     outgoing.data()[u] = capacity_sum;
813     if ( !__impl::add_distance(incoming.data()[v], flow, capacity_sum) ) return false;
814     incoming.data()[v] = capacity_sum;
815     W term{}, cost_sum{};
816     if ( !__impl::flow_cost(flow, static_cast<W>(__impl::weight(cost_map, graph, edge.id)), term)
817          || !__impl::add_distance(total_cost, term, cost_sum) )
818       return false;
819     total_cost = cost_sum;
820     const auto location = workspace.location.data()[slot];
821     auto &forward = workspace.residual.data()[static_cast<usize>(location.source)].data()[location.arc];
822     auto &reverse = workspace.residual.data()[static_cast<usize>(forward.target)].data()[forward.reverse];
823     forward.capacity = capacity - flow;
824     reverse.capacity = flow;
825   }
826   if ( total_cost != result.total_cost ) return false;
827   for ( auto vertex : graph.vertices() ) {
828     const usize slot = static_cast<usize>(vertex.value);
829     if ( vertex != source && vertex != sink && incoming.data()[slot] != outgoing.data()[slot] ) return false;
830   }
831   const usize ss = static_cast<usize>(source.value);
832   const usize ts = static_cast<usize>(sink.value);
833   if ( !(outgoing.data()[ss] >= incoming.data()[ss] && outgoing.data()[ss] - incoming.data()[ss] == result.achieved_flow
834          && incoming.data()[ts] >= outgoing.data()[ts] && incoming.data()[ts] - outgoing.data()[ts] == result.achieved_flow) )
835     return false;
836   if ( result.incoming.size() != result.outgoing.size() ) return false;
837   if ( result.incoming.size() == graph.vertex_slots() )
838     for ( usize i = 0; i < graph.vertex_slots(); ++i )
839       if ( result.incoming.data()[i] != incoming.data()[i] || result.outgoing.data()[i] != outgoing.data()[i] ) return false;
840   return __impl::negative_cycle_status(workspace, graph.vertex_slots()) == algorithm_status::ok;
841 }

843 template<graph_model G, typename Supplies, typename C, typename W, typename CapacityMap, typename CostMap>
844 [[nodiscard]] bool
845 verify_min_cost_flow(const G &graph, const Supplies &supplies, const min_cost_flow_result<typename G::index_type, C, W> &result,
846                      CapacityMap capacity_map, CostMap cost_map)
847 {
848   using S = micron::remove_cvref_t<decltype(supplies[usize(0)])>;
849   static_assert(micron::integral<S>, "min-cost circulation supplies must be integral");
850   if ( result.status != algorithm_status::ok || supplies.size() < graph.vertex_slots() || result.flow.size() < graph.edge_slots() )
851     return false;
852   micron::vector<C, micron::allocator_serial<>, false> incoming(graph.vertex_slots(), C{});
853   micron::vector<C, micron::allocator_serial<>, false> outgoing(graph.vertex_slots(), C{});
854   min_cost_flow_workspace<typename G::index_type, C, W> workspace;
855   if ( __impl::build_cost_residual<G, CapacityMap, CostMap, C, W>(graph, capacity_map, cost_map, workspace) != algorithm_status::ok )
856     return false;
857   W total_cost{};
858   for ( auto edge : graph.edges() ) {
859     const usize slot = static_cast<usize>(edge.id.value);
860     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
861     const C flow = result.flow.data()[slot];
862     if constexpr ( micron::is_signed_v<C> )
863       if ( flow < C{} ) return false;
864     if ( flow > capacity ) return false;
865     const usize u = static_cast<usize>(edge.source.value);
866     const usize v = static_cast<usize>(edge.target.value);
867     C sum{};
868     if ( !__impl::add_distance(outgoing.data()[u], flow, sum) ) return false;
869     outgoing.data()[u] = sum;
870     if ( !__impl::add_distance(incoming.data()[v], flow, sum) ) return false;
871     incoming.data()[v] = sum;
872     W term{}, cost_sum{};
873     if ( !__impl::flow_cost(flow, static_cast<W>(__impl::weight(cost_map, graph, edge.id)), term)
874          || !__impl::add_distance(total_cost, term, cost_sum) )
875       return false;
876     total_cost = cost_sum;
877     const auto location = workspace.location.data()[slot];
878     auto &forward = workspace.residual.data()[static_cast<usize>(location.source)].data()[location.arc];
879     auto &reverse = workspace.residual.data()[static_cast<usize>(forward.target)].data()[forward.reverse];
880     forward.capacity = capacity - flow;
881     reverse.capacity = flow;
882   }
883   if ( total_cost != result.total_cost ) return false;
884   C total_supply{};
885   for ( usize slot = 0; slot < graph.vertex_slots(); ++slot ) {
886     const auto vertex = typename G::vertex_descriptor(static_cast<typename G::index_type>(slot));
887     if ( !graph.has_vertex(vertex) ) {
888       if ( supplies[slot] != S{} ) return false;
889       continue;
890     }
891     const S supply = supplies[slot];
892     C amount{};
893     if ( supply > S{} ) {
894       if ( !__impl::supply_capacity(supply, amount) || outgoing[slot] < incoming[slot] || outgoing[slot] - incoming[slot] != amount )
895         return false;
896       C sum{};
897       if ( !__impl::add_distance(total_supply, amount, sum) ) return false;
898       total_supply = sum;
899     } else if ( supply < S{} ) {
900       S positive{};
901       if ( __builtin_sub_overflow(S{}, supply, &positive) || !__impl::supply_capacity(positive, amount) || incoming[slot] < outgoing[slot]
902            || incoming[slot] - outgoing[slot] != amount )
903         return false;
904     } else if ( incoming[slot] != outgoing[slot] ) {
905       return false;
906     }
907   }
908   if ( result.achieved_flow != total_supply ) return false;
909   if ( result.incoming.size() != graph.vertex_slots() || result.outgoing.size() != graph.vertex_slots() ) return false;
910   for ( usize slot = 0; slot < graph.vertex_slots(); ++slot )
911     if ( result.incoming[slot] != incoming[slot] || result.outgoing[slot] != outgoing[slot] ) return false;
912   return __impl::negative_cycle_status(workspace, graph.vertex_slots()) == algorithm_status::ok;
913 }
