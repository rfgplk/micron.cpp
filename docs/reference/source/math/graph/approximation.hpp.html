22 template<graph_model G, typename Ordering>
23 [[nodiscard]] coloring_result<typename G::index_type>
24 greedy_coloring(const G &graph, const Ordering &ordering)
25 {
26   using I = typename G::index_type;
27   const I none = vertex_id<I>::invalid_value();
28   coloring_result<I> result{ algorithm_status::ok, 0, micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), none) };
29   micron::vector<u8, micron::allocator_serial<>, false> forbidden(graph.vertices_count() + 1, u8(0));
30   for ( auto vertex : ordering ) {
31     if ( !graph.has_vertex(vertex) ) continue;
32     forbidden.fill(u8(0));
33     for ( auto neighbor : graph.out_neighbors(vertex) ) {
34       const I color = result.color.data()[static_cast<usize>(neighbor.value)];
35       if ( color != none && static_cast<usize>(color) < forbidden.size() ) forbidden.data()[static_cast<usize>(color)] = 1;
36     }
37     if constexpr ( G::is_directed )
38       for ( auto neighbor : graph.in_neighbors(vertex) ) {
39         const I color = result.color.data()[static_cast<usize>(neighbor.value)];
40         if ( color != none && static_cast<usize>(color) < forbidden.size() ) forbidden.data()[static_cast<usize>(color)] = 1;
41       }
42     usize color = 0;
43     while ( color < forbidden.size() && forbidden.data()[color] ) ++color;
44     result.color.data()[static_cast<usize>(vertex.value)] = static_cast<I>(color);
45     if ( color + 1 > result.colors ) result.colors = color + 1;
46   }
47   return result;
48 }

58 template<graph_model G>
59 [[nodiscard]] micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false>
60 maximal_independent_set(const G &graph)
61 {
62   auto ordering = smallest_last_ordering(graph);
63   micron::vector<u8, micron::allocator_serial<>, false> blocked(graph.vertex_slots(), u8(0));
64   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> result;
65   for ( auto vertex : ordering ) {
66     if ( blocked.data()[static_cast<usize>(vertex.value)] ) continue;
67     result.push_back(vertex);
68     blocked.data()[static_cast<usize>(vertex.value)] = 1;
69     for ( auto neighbor : graph.out_neighbors(vertex) ) blocked.data()[static_cast<usize>(neighbor.value)] = 1;
70     if constexpr ( G::is_directed )
71       for ( auto neighbor : graph.in_neighbors(vertex) ) blocked.data()[static_cast<usize>(neighbor.value)] = 1;
72   }
73   return result;
74 }

76 template<graph_model G>
77 [[nodiscard]] micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false>
78 greedy_dominating_set(const G &graph)
79 {
80   using vertex_descriptor = typename G::vertex_descriptor;
81   micron::vector<u8, micron::allocator_serial<>, false> covered(graph.vertex_slots(), u8(0));
82   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> result;
83   usize covered_count = 0;
84   while ( covered_count < graph.vertices_count() ) {
85     vertex_descriptor best = vertex_descriptor::invalid();
86     usize best_gain = 0;
87     for ( auto vertex : graph.vertices() ) {
88       usize gain = covered.data()[static_cast<usize>(vertex.value)] ? 0 : 1;
89       for ( auto neighbor : graph.out_neighbors(vertex) )
90         if ( !covered.data()[static_cast<usize>(neighbor.value)] ) ++gain;
91       if ( !best.valid() || gain > best_gain ) {
92         best = vertex;
93         best_gain = gain;
94       }
95     }
96     if ( !best.valid() ) break;
97     result.push_back(best);
98     const usize bs = static_cast<usize>(best.value);
99     if ( !covered.data()[bs] ) {
100       covered.data()[bs] = 1;
101       ++covered_count;
102     }
103     for ( auto neighbor : graph.out_neighbors(best) ) {
104       const usize slot = static_cast<usize>(neighbor.value);
105       if ( !covered.data()[slot] ) {
106         covered.data()[slot] = 1;
107         ++covered_count;
108       }
109     }
110   }
111   return result;
112 }

114 template<graph_model G>
115 [[nodiscard]] micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false>
116 approximate_vertex_cover(const G &graph)
117 {
118   auto matching = greedy_maximal_matching(graph);
119   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> result;
120   result.reserve(matching.cardinality * 2);
121   for ( auto edge : matching.edges ) {
122     result.push_back(graph.source(edge));
123     result.push_back(graph.target(edge));
124   }
125   return result;
126 }

133 template<graph_model G>
134 [[nodiscard]] cut_heuristic_result<typename G::index_type>
135 max_cut(const G &graph)
136 {
137   cut_heuristic_result<typename G::index_type> result{ 0,
138                                                        micron::vector<u8, micron::allocator_serial<>, false>(graph.vertex_slots(), u8(0)) };
139   usize ordinal = 0;
140   for ( auto vertex : graph.vertices() ) result.side.data()[static_cast<usize>(vertex.value)] = static_cast<u8>(ordinal++ & 1u);
141   bool improved = true;
142   while ( improved ) {
143     improved = false;
144     for ( auto vertex : graph.vertices() ) {
145       usize same = 0;
146       usize different = 0;
147       for ( auto neighbor : graph.out_neighbors(vertex) ) {
148         if ( result.side.data()[static_cast<usize>(neighbor.value)] == result.side.data()[static_cast<usize>(vertex.value)] )
149           ++same;
150         else
151           ++different;
152       }
153       if ( same > different ) {
154         result.side.data()[static_cast<usize>(vertex.value)] ^= 1u;
155         improved = true;
156       }
157     }
158   }
159   for ( auto edge : graph.edges() )
160     if ( result.side.data()[static_cast<usize>(edge.source.value)] != result.side.data()[static_cast<usize>(edge.target.value)] )
161       ++result.cut_edges;
162   return result;
163 }

165 template<graph_model G>
166 [[nodiscard]] micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false>
167 maximum_clique_heuristic(const G &graph)
168 {
169   auto order = smallest_last_ordering(graph);
170   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> clique;
171   for ( auto candidate : order ) {
172     bool adjacent = true;
173     for ( auto member : clique )
174       if ( !graph.has_edge(candidate, member) && !graph.has_edge(member, candidate) ) {
175         adjacent = false;
176         break;
177       }
178     if ( adjacent ) clique.push_back(candidate);
179   }
180   return clique;
181 }

183 template<graph_model G>
184 [[nodiscard]] usize
185 treewidth_min_degree_upper_bound(const G &graph)
186 {
187   using I = typename G::index_type;
188   using vertex_descriptor = typename G::vertex_descriptor;
189   using slot_vector = micron::vector<I, micron::allocator_serial<>, false>;
190   const usize slots = graph.vertex_slots();
191   micron::vector<u8, micron::allocator_serial<>, false> active(slots, u8(0));
192   micron::vector<u8, micron::allocator_serial<>, false> seen(slots, u8(0));
193   micron::vector<slot_vector, micron::allocator_serial<>, false> adjacency(slots);
194   for ( auto vertex : graph.vertices() ) active.data()[static_cast<usize>(vertex.value)] = 1;
195   for ( auto vertex : graph.vertices() ) {
196     const usize slot = static_cast<usize>(vertex.value);
197     auto record = [&](vertex_descriptor neighbor) {
198       const usize other = static_cast<usize>(neighbor.value);
199       if ( other == slot || seen.data()[other] ) return;
200       seen.data()[other] = 1;
201       adjacency.data()[slot].push_back(neighbor.value);
202     };
203     for ( auto neighbor : graph.out_neighbors(vertex) ) record(neighbor);
204     if constexpr ( G::is_directed )
205       for ( auto neighbor : graph.in_neighbors(vertex) ) record(neighbor);
206     for ( I id : adjacency.data()[slot] ) seen.data()[static_cast<usize>(id)] = 0;
207   }
208 
209   slot_vector clique;
210   usize bound = 0;
211   for ( usize removed = 0; removed < graph.vertices_count(); ++removed ) {
212     vertex_descriptor selected = vertex_descriptor::invalid();
213     usize selected_degree = 0;
214     for ( auto vertex : graph.vertices() ) {
215       const usize slot = static_cast<usize>(vertex.value);
216       if ( !active.data()[slot] ) continue;
217       auto &incident = adjacency.data()[slot];
218       usize degree = 0;
219       for ( usize i = 0; i < incident.size(); ++i ) {
220         const usize other = static_cast<usize>(incident.data()[i]);
221         if ( !active.data()[other] || seen.data()[other] ) continue;
222         seen.data()[other] = 1;
223         incident.data()[degree++] = incident.data()[i];
224       }
225       incident.resize(degree);
226       for ( I id : incident ) seen.data()[static_cast<usize>(id)] = 0;
227       if ( !selected.valid() || degree < selected_degree ) {
228         selected = vertex;
229         selected_degree = degree;
230       }
231     }
232     if ( !selected.valid() ) break;
233     if ( selected_degree > bound ) bound = selected_degree;
234     const usize slot = static_cast<usize>(selected.value);
235     active.data()[slot] = 0;
236     clique.clear();
237     for ( I id : adjacency.data()[slot] ) {
238       const usize other = static_cast<usize>(id);
239       if ( !active.data()[other] || seen.data()[other] ) continue;
240       seen.data()[other] = 1;
241       clique.push_back(id);
242     }
243     for ( I id : clique ) seen.data()[static_cast<usize>(id)] = 0;
244     for ( I id : clique ) {
245       const usize source = static_cast<usize>(id);
246       seen.data()[source] = 1;
247       for ( I other : adjacency.data()[source] ) seen.data()[static_cast<usize>(other)] = 1;
248       for ( I other : clique )
249         if ( !seen.data()[static_cast<usize>(other)] ) {
250           seen.data()[static_cast<usize>(other)] = 1;
251           adjacency.data()[source].push_back(other);
252         }
253       seen.data()[source] = 0;
254       for ( I other : adjacency.data()[source] ) seen.data()[static_cast<usize>(other)] = 0;
255     }
256   }
257   return bound;
258 }
