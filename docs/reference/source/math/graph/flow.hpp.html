22 template<graph_model G, typename CapacityMap>
23 [[nodiscard]] auto
24 dinic(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink, CapacityMap capacity_map)
25 {
26   using I = typename G::index_type;
27   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
28 
29   struct residual_arc {
30     I target{};
31     usize reverse{};
32     C capacity{};
33   };
34 
35   struct edge_location {
36     I source{ vertex_id<I>::invalid_value() };
37     usize arc{};
38   };
39 
40   max_flow_result<I, C> result{ algorithm_status::ok, C{}, micron::vector<C, micron::allocator_serial<>, false>(graph.edge_slots(), C{}),
41                                 micron::vector<u8, micron::allocator_serial<>, false>(graph.vertex_slots(), u8(0)) };
42   if constexpr ( !G::is_directed ) {
43     result.status = algorithm_status::invalid_graph;
44     return result;
45   }
46   if ( !graph.has_vertex(source) || !graph.has_vertex(sink) || source == sink ) {
47     result.status = algorithm_status::invalid_vertex;
48     return result;
49   }
50 
51   using arc_vector = micron::vector<residual_arc, micron::allocator_serial<>, false>;
52   micron::vector<arc_vector, micron::allocator_serial<>, false> residual(graph.vertex_slots());
53   micron::vector<edge_location, micron::allocator_serial<>, false> location(graph.edge_slots());
54   for ( auto edge : graph.edges() ) {
55     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
56     if ( __impl::invalid_weight(capacity) ) {
57       result.status = algorithm_status::invalid_weight;
58       return result;
59     }
60     const usize u = static_cast<usize>(edge.source.value);
61     const usize v = static_cast<usize>(edge.target.value);
62     const usize forward = residual.data()[u].size();
63     const usize reverse = residual.data()[v].size();
64     residual.data()[u].push_back(residual_arc{ edge.target.value, reverse, capacity });
65     residual.data()[v].push_back(residual_arc{ edge.source.value, forward, C{} });
66     location.data()[static_cast<usize>(edge.id.value)] = edge_location{ edge.source.value, forward };
67   }
68 
69   micron::vector<max_t, micron::allocator_serial<>, false> level(graph.vertex_slots(), max_t(-1));
70   micron::vector<usize, micron::allocator_serial<>, false> next(graph.vertex_slots(), usize(0));
71   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> queue;
72   queue.reserve(graph.vertices_count());
73 
74   auto build_levels = [&]() {
75     level.fill(max_t(-1));
76     queue.clear();
77     queue.push_back(source);
78     level.data()[static_cast<usize>(source.value)] = 0;
79     usize head = 0;
80     while ( head < queue.size() ) {
81       const auto u = queue.data()[head++];
82       for ( const residual_arc &arc : residual.data()[static_cast<usize>(u.value)] ) {
83         const usize v = static_cast<usize>(arc.target);
84         if ( arc.capacity > C{} && level.data()[v] < 0 ) {
85           level.data()[v] = level.data()[static_cast<usize>(u.value)] + 1;
86           queue.push_back(typename G::vertex_descriptor(arc.target));
87         }
88       }
89     }
90     return level.data()[static_cast<usize>(sink.value)] >= 0;
91   };
92 
93   auto send = [&](auto &&self, I raw_u, C pushed) -> C {
94     if ( raw_u == sink.value || pushed == C{} ) return pushed;
95     const usize u = static_cast<usize>(raw_u);
96     arc_vector &arcs = residual.data()[u];
97     for ( usize &at = next.data()[u]; at < arcs.size(); ++at ) {
98       residual_arc &arc = arcs.data()[at];
99       const usize v = static_cast<usize>(arc.target);
100       if ( arc.capacity <= C{} || level.data()[v] != level.data()[u] + 1 ) continue;
101       const C offered = pushed < arc.capacity ? pushed : arc.capacity;
102       const C delivered = self(self, arc.target, offered);
103       if ( delivered == C{} ) continue;
104       arc.capacity -= delivered;
105       residual.data()[v].data()[arc.reverse].capacity += delivered;
106       return delivered;
107     }
108     return C{};
109   };
110 
111   // The source's outgoing residual sum is a finite, input-derived upper bound;
112   // it replaces an infinity sentinel and remains sound under -Ofast.
113   C source_bound{};
114   for ( const residual_arc &arc : residual.data()[static_cast<usize>(source.value)] ) {
115     C next_bound{};
116     if ( !__impl::add_distance(source_bound, arc.capacity, next_bound) ) {
117       result.status = algorithm_status::overflow;
118       return result;
119     }
120     source_bound = next_bound;
121   }
122 
123   while ( build_levels() ) {
124     next.fill(usize(0));
125     for ( ;; ) {
126       const C pushed = send(send, source.value, source_bound);
127       if ( pushed == C{} ) break;
128       C total{};
129       if ( !__impl::add_distance(result.value, pushed, total) ) {
130         result.status = algorithm_status::overflow;
131         return result;
132       }
133       result.value = total;
134     }
135   }
136 
137   for ( auto edge : graph.edges() ) {
138     const edge_location place = location.data()[static_cast<usize>(edge.id.value)];
139     const C original = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
140     const C remaining = residual.data()[static_cast<usize>(place.source)].data()[place.arc].capacity;
141     result.flow.data()[static_cast<usize>(edge.id.value)] = original - remaining;
142   }
143 
144   // Final residual reachability is the min-cut certificate.
145   queue.clear();
146   queue.push_back(source);
147   result.source_side.data()[static_cast<usize>(source.value)] = 1;
148   usize head = 0;
149   while ( head < queue.size() ) {
150     const auto u = queue.data()[head++];
151     for ( const residual_arc &arc : residual.data()[static_cast<usize>(u.value)] ) {
152       const usize v = static_cast<usize>(arc.target);
153       if ( arc.capacity > C{} && !result.source_side.data()[v] ) {
154         result.source_side.data()[v] = 1;
155         queue.push_back(typename G::vertex_descriptor(arc.target));
156       }
157     }
158   }
159   return result;
160 }

169 template<graph_model G, typename CapacityMap>
170 [[nodiscard]] auto
171 edmonds_karp(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink, CapacityMap capacity_map)
172 {
173   using I = typename G::index_type;
174   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
175 
176   struct residual_arc {
177     I target{};
178     usize reverse{};
179     C capacity{};
180   };
181 
182   struct edge_location {
183     I source{ vertex_id<I>::invalid_value() };
184     usize arc{};
185   };
186 
187   max_flow_result<I, C> result{ algorithm_status::ok, C{}, micron::vector<C, micron::allocator_serial<>, false>(graph.edge_slots(), C{}),
188                                 micron::vector<u8, micron::allocator_serial<>, false>(graph.vertex_slots(), u8(0)) };
189   if constexpr ( !G::is_directed ) {
190     result.status = algorithm_status::invalid_graph;
191     return result;
192   }
193   if ( !graph.has_vertex(source) || !graph.has_vertex(sink) || source == sink ) {
194     result.status = algorithm_status::invalid_vertex;
195     return result;
196   }
197 
198   using arc_vector = micron::vector<residual_arc, micron::allocator_serial<>, false>;
199   micron::vector<arc_vector, micron::allocator_serial<>, false> residual(graph.vertex_slots());
200   micron::vector<edge_location, micron::allocator_serial<>, false> location(graph.edge_slots());
201   for ( auto edge : graph.edges() ) {
202     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
203     if ( __impl::invalid_weight(capacity) ) {
204       result.status = algorithm_status::invalid_weight;
205       return result;
206     }
207     const usize u = static_cast<usize>(edge.source.value);
208     const usize v = static_cast<usize>(edge.target.value);
209     const usize forward = residual.data()[u].size();
210     const usize reverse = residual.data()[v].size();
211     residual.data()[u].push_back(residual_arc{ edge.target.value, reverse, capacity });
212     residual.data()[v].push_back(residual_arc{ edge.source.value, forward, C{} });
213     location.data()[static_cast<usize>(edge.id.value)] = edge_location{ edge.source.value, forward };
214   }
215 
216   const usize slots = graph.vertex_slots();
217   micron::vector<I, micron::allocator_serial<>, false> parent(slots, vertex_id<I>::invalid_value());
218   micron::vector<usize, micron::allocator_serial<>, false> parent_arc(slots, usize(0));
219   micron::vector<u8, micron::allocator_serial<>, false> reached(slots, u8(0));
220   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> queue;
221   queue.reserve(graph.vertices_count());
222 
223   for ( ;; ) {
224     parent.fill(vertex_id<I>::invalid_value());
225     reached.fill(u8(0));
226     queue.clear();
227     queue.push_back(source);
228     reached.data()[static_cast<usize>(source.value)] = 1;
229     usize head = 0;
230     while ( head < queue.size() && !reached.data()[static_cast<usize>(sink.value)] ) {
231       const auto u = queue.data()[head++];
232       const auto &arcs = residual.data()[static_cast<usize>(u.value)];
233       for ( usize arc_index = 0; arc_index < arcs.size(); ++arc_index ) {
234         const residual_arc &arc = arcs.data()[arc_index];
235         const usize v = static_cast<usize>(arc.target);
236         if ( arc.capacity <= C{} || reached.data()[v] ) continue;
237         reached.data()[v] = 1;
238         parent.data()[v] = u.value;
239         parent_arc.data()[v] = arc_index;
240         queue.push_back(typename G::vertex_descriptor(arc.target));
241         if ( arc.target == sink.value ) break;
242       }
243     }
244     if ( !reached.data()[static_cast<usize>(sink.value)] ) {
245       result.source_side = micron::move(reached);
246       break;
247     }
248 
249     C bottleneck{};
250     bool have_bottleneck = false;
251     I current = sink.value;
252     while ( current != source.value ) {
253       const I previous = parent.data()[static_cast<usize>(current)];
254       const residual_arc &arc = residual.data()[static_cast<usize>(previous)].data()[parent_arc.data()[static_cast<usize>(current)]];
255       if ( !have_bottleneck || arc.capacity < bottleneck ) {
256         bottleneck = arc.capacity;
257         have_bottleneck = true;
258       }
259       current = previous;
260     }
261 
262     current = sink.value;
263     while ( current != source.value ) {
264       const I previous = parent.data()[static_cast<usize>(current)];
265       residual_arc &arc = residual.data()[static_cast<usize>(previous)].data()[parent_arc.data()[static_cast<usize>(current)]];
266       arc.capacity -= bottleneck;
267       residual.data()[static_cast<usize>(current)].data()[arc.reverse].capacity += bottleneck;
268       current = previous;
269     }
270     C total{};
271     if ( !__impl::add_distance(result.value, bottleneck, total) ) {
272       result.status = algorithm_status::overflow;
273       return result;
274     }
275     result.value = total;
276   }
277 
278   for ( auto edge : graph.edges() ) {
279     const edge_location place = location.data()[static_cast<usize>(edge.id.value)];
280     const C original = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
281     const C remaining = residual.data()[static_cast<usize>(place.source)].data()[place.arc].capacity;
282     result.flow.data()[static_cast<usize>(edge.id.value)] = original - remaining;
283   }
284   return result;
285 }

287 template<graph_model G, typename CapacityMap>
288 [[nodiscard]] auto
289 push_relabel(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink, CapacityMap capacity_map)
290 {
291   using I = typename G::index_type;
292   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
293 
294   struct residual_arc {
295     I target{};
296     usize reverse{};
297     C capacity{};
298   };
299 
300   struct edge_location {
301     I source{ vertex_id<I>::invalid_value() };
302     usize arc{};
303   };
304 
305   max_flow_result<I, C> result{ algorithm_status::ok, C{}, micron::vector<C, micron::allocator_serial<>, false>(graph.edge_slots(), C{}),
306                                 micron::vector<u8, micron::allocator_serial<>, false>(graph.vertex_slots(), u8(0)) };
307   if constexpr ( !G::is_directed ) {
308     result.status = algorithm_status::invalid_graph;
309     return result;
310   }
311   if ( !graph.has_vertex(source) || !graph.has_vertex(sink) || source == sink ) {
312     result.status = algorithm_status::invalid_vertex;
313     return result;
314   }
315 
316   using arc_vector = micron::vector<residual_arc, micron::allocator_serial<>, false>;
317   micron::vector<arc_vector, micron::allocator_serial<>, false> residual(graph.vertex_slots());
318   micron::vector<edge_location, micron::allocator_serial<>, false> location(graph.edge_slots());
319   for ( auto edge : graph.edges() ) {
320     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
321     if ( __impl::invalid_weight(capacity) ) {
322       result.status = algorithm_status::invalid_weight;
323       return result;
324     }
325     const usize u = static_cast<usize>(edge.source.value);
326     const usize v = static_cast<usize>(edge.target.value);
327     const usize forward = residual.data()[u].size();
328     const usize reverse = residual.data()[v].size();
329     residual.data()[u].push_back(residual_arc{ edge.target.value, reverse, capacity });
330     residual.data()[v].push_back(residual_arc{ edge.source.value, forward, C{} });
331     location.data()[static_cast<usize>(edge.id.value)] = edge_location{ edge.source.value, forward };
332   }
333 
334   const usize slots = graph.vertex_slots();
335   micron::vector<usize, micron::allocator_serial<>, false> height(slots, usize(0));
336   micron::vector<usize, micron::allocator_serial<>, false> current(slots, usize(0));
337   micron::vector<C, micron::allocator_serial<>, false> excess(slots, C{});
338   micron::vector<u8, micron::allocator_serial<>, false> queued(slots, u8(0));
339   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> active;
340   active.reserve(graph.edges_count() + graph.vertices_count());
341   height.data()[static_cast<usize>(source.value)] = graph.vertices_count();
342 
343   auto enqueue = [&](I raw_vertex) {
344     if ( raw_vertex == source.value || raw_vertex == sink.value ) return;
345     const usize slot = static_cast<usize>(raw_vertex);
346     if ( excess.data()[slot] <= C{} || queued.data()[slot] ) return;
347     queued.data()[slot] = 1;
348     active.push_back(typename G::vertex_descriptor(raw_vertex));
349   };
350 
351   arc_vector &source_arcs = residual.data()[static_cast<usize>(source.value)];
352   for ( usize arc_index = 0; arc_index < source_arcs.size(); ++arc_index ) {
353     residual_arc &arc = source_arcs.data()[arc_index];
354     if ( arc.capacity <= C{} ) continue;
355     const C amount = arc.capacity;
356     arc.capacity = C{};
357     residual.data()[static_cast<usize>(arc.target)].data()[arc.reverse].capacity += amount;
358     if ( arc.target != source.value ) {
359       C total{};
360       if ( !__impl::add_distance(excess.data()[static_cast<usize>(arc.target)], amount, total) ) {
361         result.status = algorithm_status::overflow;
362         return result;
363       }
364       excess.data()[static_cast<usize>(arc.target)] = total;
365       enqueue(arc.target);
366     }
367   }
368 
369   usize head = 0;
370   while ( head < active.size() ) {
371     const auto vertex = active.data()[head++];
372     const usize u = static_cast<usize>(vertex.value);
373     queued.data()[u] = 0;
374     arc_vector &arcs = residual.data()[u];
375 
376     while ( excess.data()[u] > C{} ) {
377       if ( current.data()[u] == arcs.size() ) {
378         bool found = false;
379         usize minimum_height = 0;
380         for ( const residual_arc &arc : arcs ) {
381           if ( arc.capacity <= C{} ) continue;
382           const usize candidate = height.data()[static_cast<usize>(arc.target)];
383           if ( !found || candidate < minimum_height ) {
384             found = true;
385             minimum_height = candidate;
386           }
387         }
388         if ( !found ) {
389           result.status = algorithm_status::invalid_graph;
390           return result;
391         }
392         height.data()[u] = minimum_height + 1;
393         current.data()[u] = 0;
394         continue;
395       }
396 
397       residual_arc &arc = arcs.data()[current.data()[u]];
398       const usize v = static_cast<usize>(arc.target);
399       if ( arc.capacity <= C{} || height.data()[u] != height.data()[v] + 1 ) {
400         ++current.data()[u];
401         continue;
402       }
403 
404       const C amount = excess.data()[u] < arc.capacity ? excess.data()[u] : arc.capacity;
405       const C previous_excess = excess.data()[v];
406       arc.capacity -= amount;
407       residual.data()[v].data()[arc.reverse].capacity += amount;
408       excess.data()[u] -= amount;
409       if ( arc.target != source.value ) {
410         C total{};
411         if ( !__impl::add_distance(excess.data()[v], amount, total) ) {
412           result.status = algorithm_status::overflow;
413           return result;
414         }
415         excess.data()[v] = total;
416         if ( previous_excess == C{} ) enqueue(arc.target);
417       }
418     }
419   }
420 
421   result.value = excess.data()[static_cast<usize>(sink.value)];
422   for ( auto edge : graph.edges() ) {
423     const edge_location place = location.data()[static_cast<usize>(edge.id.value)];
424     const C original = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
425     const C remaining = residual.data()[static_cast<usize>(place.source)].data()[place.arc].capacity;
426     result.flow.data()[static_cast<usize>(edge.id.value)] = original - remaining;
427   }
428 
429   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> queue;
430   queue.reserve(graph.vertices_count());
431   queue.push_back(source);
432   result.source_side.data()[static_cast<usize>(source.value)] = 1;
433   head = 0;
434   while ( head < queue.size() ) {
435     const auto u = queue.data()[head++];
436     for ( const residual_arc &arc : residual.data()[static_cast<usize>(u.value)] ) {
437       const usize v = static_cast<usize>(arc.target);
438       if ( arc.capacity > C{} && !result.source_side.data()[v] ) {
439         result.source_side.data()[v] = 1;
440         queue.push_back(typename G::vertex_descriptor(arc.target));
441       }
442     }
443   }
444   return result;
445 }

468 template<graph_model G, typename CapacityMap>
469 [[nodiscard]] bool
470 verify_flow(
471     const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor sink,
472     const max_flow_result<typename G::index_type,
473                           micron::remove_cvref_t<decltype(__impl::weight(micron::declval<CapacityMap &>(), micron::declval<const G &>(),
474                                                                          micron::declval<typename G::edge_descriptor>()))>> &flow,
475     CapacityMap capacity_map)
476 {
477   using C = typename decltype(flow.flow)::value_type;
478   if ( flow.status != algorithm_status::ok ) return false;
479   micron::vector<C, micron::allocator_serial<>, false> incoming(graph.vertex_slots(), C{});
480   micron::vector<C, micron::allocator_serial<>, false> outgoing(graph.vertex_slots(), C{});
481   for ( auto edge : graph.edges() ) {
482     const C value = flow.flow.data()[static_cast<usize>(edge.id.value)];
483     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
484     if ( value < C{} || capacity < value ) return false;
485     outgoing.data()[static_cast<usize>(edge.source.value)] += value;
486     incoming.data()[static_cast<usize>(edge.target.value)] += value;
487   }
488   for ( auto vertex : graph.vertices() ) {
489     if ( vertex == source || vertex == sink ) continue;
490     const usize slot = static_cast<usize>(vertex.value);
491     if ( incoming.data()[slot] != outgoing.data()[slot] ) return false;
492   }
493   const usize source_slot = static_cast<usize>(source.value);
494   const usize sink_slot = static_cast<usize>(sink.value);
495   return outgoing.data()[source_slot] >= incoming.data()[source_slot]
496          && outgoing.data()[source_slot] - incoming.data()[source_slot] == flow.value
497          && incoming.data()[sink_slot] >= outgoing.data()[sink_slot]
498          && incoming.data()[sink_slot] - outgoing.data()[sink_slot] == flow.value;
499 }

507 template<graph_model G, typename WeightMap>
508 [[nodiscard]] auto
509 stoer_wagner(const G &graph, WeightMap weight_map)
510 {
511   using I = typename G::index_type;
512   using W = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
513   global_cut_result<I, W> result{ algorithm_status::ok, W{},
514                                   micron::vector<u8, micron::allocator_serial<>, false>(graph.vertex_slots(), u8(0)) };
515   if constexpr ( G::is_directed ) {
516     result.status = algorithm_status::invalid_graph;
517     return result;
518   }
519   const usize n = graph.vertices_count();
520   if ( n < 2 ) return result;
521   auto mapping = __dense_vertex_mapping(graph);
522   micron::vector<W, micron::allocator_serial<>, false> weights(n * n, W{});
523   for ( auto edge : graph.edges() ) {
524     const usize u = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
525     const usize v = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
526     const W weight = static_cast<W>(__impl::weight(weight_map, graph, edge.id));
527     if ( __impl::invalid_weight(weight) ) {
528       result.status = algorithm_status::invalid_weight;
529       return result;
530     }
531     weights.data()[u * n + v] += weight;
532     weights.data()[v * n + u] += weight;
533   }
534   micron::vector<usize, micron::allocator_serial<>, false> vertices(n);
535   micron::vector<micron::vector<usize, micron::allocator_serial<>, false>, micron::allocator_serial<>, false> groups(n);
536   for ( usize i = 0; i < n; ++i ) {
537     vertices.data()[i] = i;
538     groups.data()[i].push_back(i);
539   }
540   bool have_cut = false;
541   usize active = n;
542   while ( active > 1 ) {
543     micron::vector<W, micron::allocator_serial<>, false> connectivity(n, W{});
544     micron::vector<u8, micron::allocator_serial<>, false> added(n, u8(0));
545     usize previous = vertices.data()[0];
546     usize selected = previous;
547     for ( usize phase = 0; phase < active; ++phase ) {
548       selected = vertices.data()[0];
549       bool found = false;
550       for ( usize j = 0; j < active; ++j ) {
551         const usize candidate = vertices.data()[j];
552         if ( added.data()[candidate] ) continue;
553         if ( !found || connectivity.data()[selected] < connectivity.data()[candidate] ) {
554           selected = candidate;
555           found = true;
556         }
557       }
558       if ( phase + 1 == active ) {
559         const W cut = connectivity.data()[selected];
560         if ( !have_cut || cut < result.value ) {
561           have_cut = true;
562           result.value = cut;
563           result.side.fill(u8(0));
564           for ( usize dense : groups.data()[selected] )
565             result.side.data()[static_cast<usize>(mapping.dense_to_vertex.data()[dense].value)] = 1;
566         }
567         for ( usize j = 0; j < n; ++j ) {
568           weights.data()[previous * n + j] += weights.data()[selected * n + j];
569           weights.data()[j * n + previous] = weights.data()[previous * n + j];
570         }
571         for ( usize dense : groups.data()[selected] ) groups.data()[previous].push_back(dense);
572         for ( usize j = 0; j < active; ++j )
573           if ( vertices.data()[j] == selected ) {
574             vertices.data()[j] = vertices.data()[active - 1];
575             break;
576           }
577         --active;
578         break;
579       }
580       added.data()[selected] = 1;
581       previous = selected;
582       for ( usize j = 0; j < active; ++j ) {
583         const usize candidate = vertices.data()[j];
584         if ( !added.data()[candidate] ) connectivity.data()[candidate] += weights.data()[selected * n + candidate];
585       }
586     }
587   }
588   return result;
589 }

598 template<graph_model G>
599 [[nodiscard]] usize
600 local_edge_connectivity(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target)
601 {
602   if ( !graph.has_vertex(source) || !graph.has_vertex(target) || source == target ) return 0;
603   auto mapping = __dense_vertex_mapping(graph);
604   using network_type = weighted_digraph<usize, empty_property, empty_property, empty_property, typename G::index_type, parallel_t>;
605   network_type network;
606   (void)network.add_vertices(graph.vertices_count());
607   for ( auto edge : graph.edges() ) {
608     const auto u = mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)];
609     const auto v = mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)];
610     if ( u == v ) continue;
611     (void)network.add_edge(u, v, usize(1));
612     if constexpr ( !G::is_directed ) (void)network.add_edge(v, u, usize(1));
613   }
614   const auto dense_source = mapping.vertex_to_dense.data()[static_cast<usize>(source.value)];
615   const auto dense_target = mapping.vertex_to_dense.data()[static_cast<usize>(target.value)];
616   auto flow
617       = dinic(network, typename network_type::vertex_descriptor(dense_source), typename network_type::vertex_descriptor(dense_target));
618   return flow.status == algorithm_status::ok ? flow.value : 0;
619 }

621 template<graph_model G>
622 [[nodiscard]] usize
623 edge_connectivity(const G &graph)
624 {
625   if ( graph.vertices_count() < 2 ) return 0;
626   auto vertices = graph.vertices();
627   const auto first = *vertices.begin();
628   usize best = graph.edges_count();
629   if constexpr ( G::is_directed ) {
630     for ( auto source : graph.vertices() )
631       for ( auto target : graph.vertices() )
632         if ( source != target ) {
633           const usize value = local_edge_connectivity(graph, source, target);
634           if ( value < best ) best = value;
635         }
636   } else {
637     for ( auto target : graph.vertices() )
638       if ( target != first ) {
639         const usize value = local_edge_connectivity(graph, first, target);
640         if ( value < best ) best = value;
641       }
642   }
643   return best;
644 }

646 template<graph_model G>
647 [[nodiscard]] usize
648 local_node_connectivity(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target)
649 {
650   if ( !graph.has_vertex(source) || !graph.has_vertex(target) || source == target ) return 0;
651   auto mapping = __dense_vertex_mapping(graph);
652   using network_type = weighted_digraph<usize, empty_property, empty_property, empty_property, usize, parallel_t>;
653   network_type network;
654   const usize count = graph.vertices_count();
655   (void)network.add_vertices(count * 2);
656   const usize dense_source = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(source.value)]);
657   const usize dense_target = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(target.value)]);
658   for ( usize vertex = 0; vertex < count; ++vertex )
659     (void)network.add_edge(vertex * 2, vertex * 2 + 1, vertex == dense_source || vertex == dense_target ? count : usize(1));
660   for ( auto edge : graph.edges() ) {
661     const usize u = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
662     const usize v = static_cast<usize>(mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
663     if ( u == v ) continue;
664     (void)network.add_edge(u * 2 + 1, v * 2, usize(1));
665     if constexpr ( !G::is_directed ) (void)network.add_edge(v * 2 + 1, u * 2, usize(1));
666   }
667   auto flow = dinic(network, typename network_type::vertex_descriptor(dense_source * 2 + 1),
668                     typename network_type::vertex_descriptor(dense_target * 2));
669   return flow.status == algorithm_status::ok ? flow.value : 0;
670 }

672 template<graph_model G>
673 [[nodiscard]] usize
674 node_connectivity(const G &graph)
675 {
676   if ( graph.vertices_count() < 2 ) return 0;
677   auto vertices = graph.vertices();
678   const auto first = *vertices.begin();
679   usize best = graph.vertices_count() - 1;
680   if constexpr ( G::is_directed ) {
681     for ( auto source : graph.vertices() )
682       for ( auto target : graph.vertices() )
683         if ( source != target ) {
684           const usize value = local_node_connectivity(graph, source, target);
685           if ( value < best ) best = value;
686         }
687   } else {
688     for ( auto target : graph.vertices() )
689       if ( target != first ) {
690         const usize value = local_node_connectivity(graph, first, target);
691         if ( value < best ) best = value;
692       }
693   }
694   return best;
695 }
