13 template<micron::integral I, typename Capacity> struct gomory_hu_workspace {
14   micron::vector<Capacity, micron::allocator_serial<>, false> base;
15   micron::vector<Capacity, micron::allocator_serial<>, false> residual;
16   micron::vector<max_t, micron::allocator_serial<>, false> level;
17   micron::vector<usize, micron::allocator_serial<>, false> next;
18   micron::vector<usize, micron::allocator_serial<>, false> queue;
19   micron::vector<u8, micron::allocator_serial<>, false> side;
20 
21   void
22   reserve(usize vertices)
23   {
24     if ( vertices == 0 || vertices <= micron::numeric_limits<usize>::max() / vertices ) {
25       base.reserve(vertices * vertices);
26       residual.reserve(vertices * vertices);
27     }
28     level.reserve(vertices);
29     next.reserve(vertices);
30     queue.reserve(vertices);
31     side.reserve(vertices);
32   }
33 };

35 template<micron::integral I, typename Capacity> struct gomory_hu_result {
36   algorithm_status status{ algorithm_status::ok };
37   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> parent;
38   micron::vector<Capacity, micron::allocator_serial<>, false> cut;
39   weighted_graph<Capacity, empty_property, empty_property, empty_property, I> tree;
40   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> dense_to_vertex;
41   micron::vector<I, micron::allocator_serial<>, false> vertex_to_dense;
42   micron::vector<micron::vector<u64, micron::allocator_serial<>, false>, micron::allocator_serial<>, false> cut_side;
43 
44   [[nodiscard]] bool
45   side_contains(usize tree_child, usize dense_vertex) const noexcept
46   {
47     return tree_child < cut_side.size() && dense_vertex / 64 < cut_side.data()[tree_child].size()
48            && (cut_side.data()[tree_child].data()[dense_vertex / 64] & (u64(1) << (dense_vertex & 63))) != 0;
49   }
50 };

52 template<graph_model G, typename CapacityMap>
53 [[nodiscard]] auto
54 gomory_hu(const G &graph, CapacityMap capacity_map,
55           gomory_hu_workspace<typename G::index_type,
56                               micron::remove_cvref_t<decltype(__impl::weight(micron::declval<CapacityMap &>(), micron::declval<const G &>(),
57                                                                              micron::declval<typename G::edge_descriptor>()))>> &workspace)
58 {
59   using I = typename G::index_type;
60   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
61   gomory_hu_result<I, C> result;
62   if constexpr ( G::is_directed ) {
63     result.status = algorithm_status::invalid_graph;
64     return result;
65   }
66   auto mapping = __dense_vertex_mapping(graph);
67   result.dense_to_vertex = micron::move(mapping.dense_to_vertex);
68   result.vertex_to_dense = micron::move(mapping.vertex_to_dense);
69   const usize n = result.dense_to_vertex.size();
70   if ( n != 0 && n > micron::numeric_limits<usize>::max() / n ) {
71     result.status = algorithm_status::overflow;
72     return result;
73   }
74   workspace.reserve(n);
75   workspace.base.resize(n * n, C{});
76   workspace.base.fill(C{});
77   for ( auto edge : graph.edges() ) {
78     const C capacity = static_cast<C>(__impl::weight(capacity_map, graph, edge.id));
79     if ( __impl::invalid_weight(capacity) ) {
80       result.status = algorithm_status::invalid_weight;
81       return result;
82     }
83     const usize u = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
84     const usize v = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
85     if ( u == v ) continue;
86     C sum{};
87     if ( !__impl::add_distance(workspace.base.data()[u * n + v], capacity, sum) ) {
88       result.status = algorithm_status::overflow;
89       return result;
90     }
91     workspace.base.data()[u * n + v] = sum;
92     if ( !__impl::add_distance(workspace.base.data()[v * n + u], capacity, sum) ) {
93       result.status = algorithm_status::overflow;
94       return result;
95     }
96     workspace.base.data()[v * n + u] = sum;
97   }
98 
99   result.parent.resize(n, vertex_id<I>::invalid());
100   result.cut.resize(n, C{});
101   for ( usize vertex = 1; vertex < n; ++vertex ) result.parent.data()[vertex] = vertex_id<I>(I(0));
102   workspace.level.resize(n, max_t(-1));
103   workspace.next.resize(n, usize(0));
104   workspace.side.resize(n, u8(0));
105 
106   auto max_flow = [&](usize source, usize sink, C &value) {
107     workspace.residual = workspace.base;
108     value = C{};
109     auto build_levels = [&]() {
110       workspace.level.fill(max_t(-1));
111       workspace.queue.clear();
112       workspace.queue.push_back(source);
113       workspace.level.data()[source] = 0;
114       usize head = 0;
115       while ( head < workspace.queue.size() ) {
116         const usize u = workspace.queue.data()[head++];
117         for ( usize v = 0; v < n; ++v )
118           if ( workspace.residual.data()[u * n + v] > C{} && workspace.level.data()[v] < 0 ) {
119             workspace.level.data()[v] = workspace.level.data()[u] + 1;
120             workspace.queue.push_back(v);
121           }
122       }
123       return workspace.level.data()[sink] >= 0;
124     };
125     bool overflow = false;
126     auto send = [&](auto &&self, usize u, C offered) -> C {
127       if ( u == sink || offered == C{} ) return offered;
128       for ( usize &v = workspace.next.data()[u]; v < n; ++v ) {
129         C &capacity = workspace.residual.data()[u * n + v];
130         if ( capacity <= C{} || workspace.level.data()[v] != workspace.level.data()[u] + 1 ) continue;
131         const C pushed = self(self, v, offered < capacity ? offered : capacity);
132         if ( pushed == C{} ) continue;
133         C reverse{};
134         if ( !__impl::add_distance(workspace.residual.data()[v * n + u], pushed, reverse) ) {
135           overflow = true;
136           return C{};
137         }
138         capacity -= pushed;
139         workspace.residual.data()[v * n + u] = reverse;
140         return pushed;
141       }
142       return C{};
143     };
144     C bound{};
145     for ( usize v = 0; v < n; ++v ) {
146       C sum{};
147       if ( !__impl::add_distance(bound, workspace.base.data()[source * n + v], sum) ) return algorithm_status::overflow;
148       bound = sum;
149     }
150     while ( build_levels() ) {
151       workspace.next.fill(usize(0));
152       for ( ;; ) {
153         const C pushed = send(send, source, bound);
154         if ( overflow ) return algorithm_status::overflow;
155         if ( pushed == C{} ) break;
156         C total{};
157         if ( !__impl::add_distance(value, pushed, total) ) return algorithm_status::overflow;
158         value = total;
159       }
160     }
161     workspace.side.fill(u8(0));
162     workspace.queue.clear();
163     workspace.queue.push_back(source);
164     workspace.side.data()[source] = 1;
165     for ( usize head = 0; head < workspace.queue.size(); ++head ) {
166       const usize u = workspace.queue.data()[head];
167       for ( usize v = 0; v < n; ++v )
168         if ( workspace.residual.data()[u * n + v] > C{} && !workspace.side.data()[v] ) {
169           workspace.side.data()[v] = 1;
170           workspace.queue.push_back(v);
171         }
172     }
173     return algorithm_status::ok;
174   };
175 
176   for ( usize source = 1; source < n; ++source ) {
177     const usize sink = static_cast<usize>(result.parent.data()[source].value);
178     C value{};
179     result.status = max_flow(source, sink, value);
180     if ( result.status != algorithm_status::ok ) return result;
181     for ( usize vertex = source + 1; vertex < n; ++vertex )
182       if ( result.parent.data()[vertex].value == static_cast<I>(sink) && workspace.side.data()[vertex] )
183         result.parent.data()[vertex] = vertex_id<I>(static_cast<I>(source));
184     if ( sink != 0 ) {
185       const usize sink_parent = static_cast<usize>(result.parent.data()[sink].value);
186       if ( workspace.side.data()[sink_parent] ) {
187         result.parent.data()[source] = result.parent.data()[sink];
188         result.parent.data()[sink] = vertex_id<I>(static_cast<I>(source));
189         result.cut.data()[source] = result.cut.data()[sink];
190         result.cut.data()[sink] = value;
191         continue;
192       }
193     }
194     result.cut.data()[source] = value;
195   }
196 
197   (void)result.tree.add_vertices(n);
198   for ( usize child = 1; child < n; ++child )
199     (void)result.tree.add_edge(static_cast<I>(child), result.parent.data()[child].value, result.cut.data()[child]);
200 
201   const usize words = (n + 63) / 64;
202   result.cut_side.resize(n);
203   for ( usize child = 1; child < n; ++child ) {
204     auto &bits = result.cut_side.data()[child];
205     bits.resize(words, u64(0));
206     workspace.side.fill(u8(0));
207     workspace.queue.clear();
208     workspace.queue.push_back(child);
209     workspace.side.data()[child] = 1;
210     for ( usize head = 0; head < workspace.queue.size(); ++head ) {
211       const usize u = workspace.queue.data()[head];
212       bits.data()[u / 64] |= u64(1) << (u & 63);
213       if ( u != 0 ) {
214         const usize p = static_cast<usize>(result.parent.data()[u].value);
215         if ( !((u == child && p == static_cast<usize>(result.parent.data()[child].value))
216                || (p == child && u == static_cast<usize>(result.parent.data()[child].value)))
217              && !workspace.side.data()[p] ) {
218           workspace.side.data()[p] = 1;
219           workspace.queue.push_back(p);
220         }
221       }
222       for ( usize v = 1; v < n; ++v ) {
223         const usize p = static_cast<usize>(result.parent.data()[v].value);
224         if ( p != u || (v == child && u == static_cast<usize>(result.parent.data()[child].value)) || workspace.side.data()[v] ) continue;
225         workspace.side.data()[v] = 1;
226         workspace.queue.push_back(v);
227       }
228     }
229   }
230   return result;
231 }

233 template<graph_model G, typename CapacityMap>
234 [[nodiscard]] auto
235 gomory_hu(const G &graph, CapacityMap capacity_map)
236 {
237   using C = micron::remove_cvref_t<decltype(__impl::weight(capacity_map, graph, micron::declval<typename G::edge_descriptor>()))>;
238   gomory_hu_workspace<typename G::index_type, C> workspace;
239   return gomory_hu(graph, capacity_map, workspace);
240 }
