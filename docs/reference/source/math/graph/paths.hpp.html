47 template<typename W>
48 [[nodiscard]] constexpr bool
49 invalid_weight(const W &weight) noexcept
50 {
51   if constexpr ( micron::is_floating_point_v<W> )
52     if ( !micron::math::ieee::is_finite(weight) ) return true;
53   if constexpr ( micron::is_signed_v<W> ) return weight < W(0);
54   return false;

65 template<typename D, typename W>
66 [[nodiscard]] constexpr bool
67 add_distance(const D &a, const W &b, D &result) noexcept
68 {
69   if constexpr ( micron::is_integral_v<D> && micron::is_integral_v<W> ) {
70     return !__builtin_add_overflow(a, static_cast<D>(b), &result);
71   } else {
72     result = static_cast<D>(a + static_cast<D>(b));
73     if constexpr ( micron::is_floating_point_v<D> ) return micron::math::ieee::is_finite(result);
74     return true;
75   }
76 }

79 [[nodiscard]] constexpr bool
80 sub_distance(const D &a, const W &b, D &result) noexcept
81 {
82   if constexpr ( micron::is_integral_v<D> && micron::is_integral_v<W> ) {
83     return !__builtin_sub_overflow(a, static_cast<D>(b), &result);

85     result = static_cast<D>(a - static_cast<D>(b));

101 template<micron::integral I, typename D> struct shortest_path_workspace {
102   micron::vector<D, micron::allocator_serial<>, false> distance;
103   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> parent;
104   micron::vector<u8, micron::allocator_serial<>, false> reached;
105   micron::vector<u8, micron::allocator_serial<>, false> settled;
106   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> queue;
107   micron::vector<D, micron::allocator_serial<>, false> queue_distance;
108 
109   void
110   reserve(usize slots, usize queue_entries = 0)
111   {
112     distance.reserve(slots);
113     parent.reserve(slots);
114     reached.reserve(slots);
115     settled.reserve(slots);
116     queue.reserve(queue_entries > slots ? queue_entries : slots);
117     queue_distance.reserve(queue_entries > slots ? queue_entries : slots);
118   }
119 
120   void
121   reset(usize slots)
122   {
123     distance.resize(slots, D{});
124     distance.fill(D{});
125     parent.resize(slots, vertex_id<I>::invalid());
126     parent.fill(vertex_id<I>::invalid());
127     reached.resize(slots, u8(0));
128     reached.fill(u8(0));
129     settled.resize(slots, u8(0));
130     settled.fill(u8(0));
131     queue.clear();
132     queue_distance.clear();
133   }
134 };

136 template<micron::integral I, typename D> struct shortest_paths_result {
137   using distance_type = D;
138   algorithm_status status{ algorithm_status::ok };
139   vertex_id<I> source{};
140   micron::vector<D, micron::allocator_serial<>, false> distance;
141   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> parent;
142   micron::vector<u8, micron::allocator_serial<>, false> reached;
143 
144   [[nodiscard]] bool
145   contains(vertex_id<I> vertex) const noexcept
146   {
147     const usize slot = static_cast<usize>(vertex.value);
148     return vertex.valid() && slot < reached.size() && reached.data()[slot] != 0;
149   }
150 
151   [[nodiscard]] const D *
152   try_distance(vertex_id<I> vertex) const noexcept
153   {
154     return contains(vertex) ? distance.data() + static_cast<usize>(vertex.value) : nullptr;
155   }
156 
157   [[nodiscard]] micron::vector<vertex_id<I>, micron::allocator_serial<>, false>
158   path_to(vertex_id<I> target) const
159   {
160     micron::vector<vertex_id<I>, micron::allocator_serial<>, false> path;
161     if ( !contains(target) ) return path;
162     vertex_id<I> current = target;
163     for ( usize guard = 0; guard <= parent.size(); ++guard ) {
164       path.push_back(current);
165       if ( current == source ) break;
166       const usize slot = static_cast<usize>(current.value);
167       if ( slot >= parent.size() || !parent.data()[slot].valid() ) {
168         path.clear();
169         return path;
170       }
171       current = parent.data()[slot];
172     }
173     for ( usize a = 0, b = path.size() ? path.size() - 1 : 0; a < b; ++a, --b ) micron::swap(path.data()[a], path.data()[b]);
174     return path;
175   }
176 };

178 template<graph_model G, typename WeightMap>
179 algorithm_status
180 dijkstra_into(
181     const G &graph, typename G::vertex_descriptor source,
182     shortest_path_workspace<typename G::index_type,
183                             micron::remove_cvref_t<decltype(__impl::weight(micron::declval<WeightMap &>(), micron::declval<const G &>(),
184                                                                            micron::declval<typename G::edge_descriptor>()))>> &workspace,
185     WeightMap weight_map)
186 {
187   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
188   using vertex_descriptor = typename G::vertex_descriptor;
189   if ( !graph.has_vertex(source) ) return algorithm_status::invalid_vertex;
190   workspace.reset(graph.vertex_slots());
191   const usize source_slot = static_cast<usize>(source.value);
192   workspace.reached.data()[source_slot] = 1;
193   workspace.distance.data()[source_slot] = D{};
194   workspace.parent.data()[source_slot] = source;
195   workspace.queue.push_back(source);
196   workspace.queue_distance.push_back(D{});
197 
198   auto push_heap = [&](vertex_descriptor vertex) {
199     workspace.queue.push_back(vertex);
200     workspace.queue_distance.push_back(workspace.distance.data()[static_cast<usize>(vertex.value)]);
201     usize child = workspace.queue.size() - 1;
202     while ( child ) {
203       const usize parent = (child - 1) / 2;
204       if ( !(workspace.queue_distance.data()[child] < workspace.queue_distance.data()[parent]) ) break;
205       micron::swap(workspace.queue.data()[child], workspace.queue.data()[parent]);
206       micron::swap(workspace.queue_distance.data()[child], workspace.queue_distance.data()[parent]);
207       child = parent;
208     }
209   };
210   auto pop_heap = [&]() {
211     const vertex_descriptor result = workspace.queue.data()[0];
212     const vertex_descriptor tail = workspace.queue.data()[workspace.queue.size() - 1];
213     const D tail_distance = workspace.queue_distance.data()[workspace.queue_distance.size() - 1];
214     workspace.queue.pop_back();
215     workspace.queue_distance.pop_back();
216     if ( !workspace.queue.empty() ) {
217       workspace.queue.data()[0] = tail;
218       workspace.queue_distance.data()[0] = tail_distance;
219       usize parent = 0;
220       for ( ;; ) {
221         const usize left = parent * 2 + 1;
222         if ( left >= workspace.queue.size() ) break;
223         const usize right = left + 1;
224         usize child = left;
225         if ( right < workspace.queue.size() && workspace.queue_distance.data()[right] < workspace.queue_distance.data()[left] )
226           child = right;
227         if ( !(workspace.queue_distance.data()[child] < workspace.queue_distance.data()[parent]) ) break;
228         micron::swap(workspace.queue.data()[child], workspace.queue.data()[parent]);
229         micron::swap(workspace.queue_distance.data()[child], workspace.queue_distance.data()[parent]);
230         parent = child;
231       }
232     }
233     return result;
234   };
235 
236   while ( !workspace.queue.empty() ) {
237     const vertex_descriptor selected = pop_heap();
238     const usize selected_slot = static_cast<usize>(selected.value);
239     if ( workspace.settled.data()[selected_slot] ) continue;
240     workspace.settled.data()[selected_slot] = 1;
241 
242     for ( auto edge : graph.out_edges(selected) ) {
243       const auto raw_weight = __impl::weight(weight_map, graph, edge);
244       if ( __impl::invalid_weight(raw_weight) ) return algorithm_status::invalid_weight;
245       const vertex_descriptor neighbor = __impl::edge_neighbor(graph, edge, selected);
246       const usize neighbor_slot = static_cast<usize>(neighbor.value);
247       D candidate{};
248       if ( !__impl::add_distance(workspace.distance.data()[selected_slot], raw_weight, candidate) ) return algorithm_status::overflow;
249       if ( !workspace.reached.data()[neighbor_slot] || candidate < workspace.distance.data()[neighbor_slot] ) {
250         workspace.reached.data()[neighbor_slot] = 1;
251         workspace.distance.data()[neighbor_slot] = candidate;
252         workspace.parent.data()[neighbor_slot] = selected;
253         push_heap(neighbor);
254       }
255     }
256   }
257   return algorithm_status::ok;
258 }

260 template<graph_model G, typename WeightMap>
261 [[nodiscard]] auto
262 dijkstra(const G &graph, typename G::vertex_descriptor source, WeightMap weight_map)
263 {

269                                                            micron::move(workspace.reached) };

309 [[nodiscard]] auto
310 shortest_path(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target, WeightMap weight_map)
311 {
312   auto paths = dijkstra(graph, source, weight_map);
313   using D = typename decltype(paths)::distance_type;
314   if ( paths.status != algorithm_status::ok ) return path_result<typename G::index_type, D>{ paths.status, D{}, false, {} };
315   if ( !paths.contains(target) ) return path_result<typename G::index_type, D>{ algorithm_status::unreachable, D{}, false, {} };
316   return path_result<typename G::index_type, D>{ algorithm_status::ok, paths.distance.data()[static_cast<usize>(target.value)], true,
317                                                  paths.path_to(target) };
318 }

342 template<graph_model G, typename Heuristic, typename WeightMap = intrinsic_edge_weight>
343 [[nodiscard]] auto
344 astar(const G &graph, typename G::vertex_descriptor source, typename G::vertex_descriptor target, Heuristic heuristic,
345       WeightMap weight_map = {})
346 {
347   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
348   using vertex_descriptor = typename G::vertex_descriptor;
349   shortest_path_workspace<typename G::index_type, D> workspace;
350   workspace.reserve(graph.vertex_slots());
351   workspace.reset(graph.vertex_slots());
352   if ( !graph.has_vertex(source) || !graph.has_vertex(target) )
353     return path_result<typename G::index_type, D>{ algorithm_status::invalid_vertex, D{}, false, {} };
354   const usize ss = static_cast<usize>(source.value);
355   workspace.reached.data()[ss] = 1;
356   workspace.parent.data()[ss] = source;
357   algorithm_status status = algorithm_status::ok;
358 
359   for ( usize iteration = 0; iteration < graph.vertices_count(); ++iteration ) {
360     vertex_descriptor selected = vertex_descriptor::invalid();
361     D selected_score{};
362     for ( auto vertex : graph.vertices() ) {
363       const usize slot = static_cast<usize>(vertex.value);
364       if ( !workspace.reached.data()[slot] || workspace.settled.data()[slot] ) continue;
365       const auto h = __impl::heuristic(heuristic, graph, vertex, target);
366       if ( __impl::invalid_weight(h) ) {
367         status = algorithm_status::invalid_weight;
368         break;
369       }
370       D score{};
371       if ( !__impl::add_distance(workspace.distance.data()[slot], h, score) ) {
372         status = algorithm_status::overflow;
373         break;
374       }
375       if ( !selected.valid() || score < selected_score ) {
376         selected = vertex;
377         selected_score = score;
378       }
379     }
380     if ( status != algorithm_status::ok || !selected.valid() ) break;
381     if ( selected == target ) break;
382     const usize us = static_cast<usize>(selected.value);
383     workspace.settled.data()[us] = 1;
384     for ( auto edge : graph.out_edges(selected) ) {
385       const auto weight = __impl::weight(weight_map, graph, edge);
386       if ( __impl::invalid_weight(weight) ) {
387         status = algorithm_status::invalid_weight;
388         break;
389       }
390       const vertex_descriptor v = __impl::edge_neighbor(graph, edge, selected);
391       const usize vs = static_cast<usize>(v.value);
392       D candidate{};
393       if ( !__impl::add_distance(workspace.distance.data()[us], weight, candidate) ) {
394         status = algorithm_status::overflow;
395         break;
396       }
397       if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs] ) {
398         workspace.reached.data()[vs] = 1;
399         workspace.distance.data()[vs] = candidate;
400         workspace.parent.data()[vs] = selected;
401       }
402     }
403     if ( status != algorithm_status::ok ) break;
404   }
405   if ( status != algorithm_status::ok ) return path_result<typename G::index_type, D>{ status, D{}, false, {} };
406   if ( !workspace.reached.data()[static_cast<usize>(target.value)] )
407     return path_result<typename G::index_type, D>{ algorithm_status::unreachable, D{}, false, {} };
408   shortest_paths_result<typename G::index_type, D> paths{ algorithm_status::ok, source, micron::move(workspace.distance),
409                                                           micron::move(workspace.parent), micron::move(workspace.reached) };
410   return path_result<typename G::index_type, D>{ algorithm_status::ok, paths.distance.data()[static_cast<usize>(target.value)], true,
411                                                  paths.path_to(target) };
412 }

414 template<graph_model G, typename WeightMap>
415 [[nodiscard]] auto
416 bellman_ford(const G &graph, typename G::vertex_descriptor source, WeightMap weight_map)
417 {
418   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
419   using vertex_descriptor = typename G::vertex_descriptor;
420   shortest_path_workspace<typename G::index_type, D> workspace;
421   workspace.reserve(graph.vertex_slots());
422   workspace.reset(graph.vertex_slots());
423   if ( !graph.has_vertex(source) )
424     return shortest_paths_result<typename G::index_type, D>{ algorithm_status::invalid_vertex, source, micron::move(workspace.distance),
425                                                              micron::move(workspace.parent), micron::move(workspace.reached) };
426   const usize ss = static_cast<usize>(source.value);
427   workspace.reached.data()[ss] = 1;
428   workspace.parent.data()[ss] = source;
429 
430   algorithm_status status = algorithm_status::ok;
431   auto relax = [&](vertex_descriptor u, vertex_descriptor v, const auto &raw_weight) -> bool {
432     const usize us = static_cast<usize>(u.value);
433     const usize vs = static_cast<usize>(v.value);
434     if ( !workspace.reached.data()[us] ) return false;
435     D candidate{};
436     if ( !__impl::add_distance(workspace.distance.data()[us], raw_weight, candidate) ) {
437       status = algorithm_status::overflow;
438       return false;
439     }
440     if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs] ) {
441       workspace.reached.data()[vs] = 1;
442       workspace.distance.data()[vs] = candidate;
443       workspace.parent.data()[vs] = u;
444       return true;
445     }
446     return false;
447   };
448 
449   for ( usize pass = 1; pass < graph.vertices_count(); ++pass ) {
450     bool changed = false;
451     for ( auto edge : graph.edges() ) {
452       const auto raw_weight = __impl::weight(weight_map, graph, edge.id);
453       if constexpr ( micron::is_floating_point_v<decltype(raw_weight)> ) {
454         if ( __impl::nonfinite_weight(raw_weight) ) status = algorithm_status::invalid_weight;
455       }
456       changed |= relax(edge.source, edge.target, raw_weight);
457       if constexpr ( !G::is_directed ) changed |= relax(edge.target, edge.source, raw_weight);
458       if ( status != algorithm_status::ok ) break;
459     }
460     if ( status != algorithm_status::ok || !changed ) break;
461   }
462 
463   if ( status == algorithm_status::ok ) {
464     for ( auto edge : graph.edges() ) {
465       const auto raw_weight = __impl::weight(weight_map, graph, edge.id);
466       const usize us = static_cast<usize>(edge.source.value);
467       const usize vs = static_cast<usize>(edge.target.value);
468       D candidate{};
469       if ( workspace.reached.data()[us] && __impl::add_distance(workspace.distance.data()[us], raw_weight, candidate)
470            && (!workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs]) ) {
471         status = algorithm_status::negative_cycle;
472         break;
473       }
474       if constexpr ( !G::is_directed ) {
475         if ( workspace.reached.data()[vs] && __impl::add_distance(workspace.distance.data()[vs], raw_weight, candidate)
476              && candidate < workspace.distance.data()[us] ) {
477           status = algorithm_status::negative_cycle;
478           break;
479         }
480       }
481     }
482   }
483   return shortest_paths_result<typename G::index_type, D>{ status, source, micron::move(workspace.distance), micron::move(workspace.parent),
484                                                            micron::move(workspace.reached) };
485 }

494 template<graph_model G, typename WeightMap>
495 [[nodiscard]] auto
496 spfa(const G &graph, typename G::vertex_descriptor source, WeightMap weight_map)
497 {
498   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
499   using vertex_descriptor = typename G::vertex_descriptor;
500   shortest_path_workspace<typename G::index_type, D> workspace;
501   workspace.reserve(graph.vertex_slots());
502   workspace.reset(graph.vertex_slots());
503   if ( !graph.has_vertex(source) )
504     return shortest_paths_result<typename G::index_type, D>{ algorithm_status::invalid_vertex, source, micron::move(workspace.distance),
505                                                              micron::move(workspace.parent), micron::move(workspace.reached) };
506 
507   micron::vector<u8, micron::allocator_serial<>, false> queued(graph.vertex_slots(), u8(0));
508   micron::vector<usize, micron::allocator_serial<>, false> relaxations(graph.vertex_slots(), usize(0));
509   workspace.queue.push_back(source);
510   queued.data()[static_cast<usize>(source.value)] = 1;
511   workspace.reached.data()[static_cast<usize>(source.value)] = 1;
512   workspace.parent.data()[static_cast<usize>(source.value)] = source;
513   usize head = 0;
514   algorithm_status status = algorithm_status::ok;
515   while ( head < workspace.queue.size() && status == algorithm_status::ok ) {
516     const vertex_descriptor u = workspace.queue.data()[head++];
517     const usize us = static_cast<usize>(u.value);
518     queued.data()[us] = 0;
519     for ( auto edge : graph.out_edges(u) ) {
520       const auto raw_weight = __impl::weight(weight_map, graph, edge);
521       if constexpr ( micron::is_floating_point_v<decltype(raw_weight)> )
522         if ( __impl::nonfinite_weight(raw_weight) ) {
523           status = algorithm_status::invalid_weight;
524           break;
525         }
526       const vertex_descriptor v = __impl::edge_neighbor(graph, edge, u);
527       const usize vs = static_cast<usize>(v.value);
528       D candidate{};
529       if ( !__impl::add_distance(workspace.distance.data()[us], raw_weight, candidate) ) {
530         status = algorithm_status::overflow;
531         break;
532       }
533       if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs] ) {
534         workspace.reached.data()[vs] = 1;
535         workspace.distance.data()[vs] = candidate;
536         workspace.parent.data()[vs] = u;
537         if ( ++relaxations.data()[vs] >= graph.vertices_count() ) {
538           status = algorithm_status::negative_cycle;
539           break;
540         }
541         if ( !queued.data()[vs] ) {
542           queued.data()[vs] = 1;
543           workspace.queue.push_back(v);
544         }
545       }
546     }
547   }
548   return shortest_paths_result<typename G::index_type, D>{ status, source, micron::move(workspace.distance), micron::move(workspace.parent),
549                                                            micron::move(workspace.reached) };
550 }

559 template<micron::integral I> struct topological_result {
560   algorithm_status status{ algorithm_status::ok };
561   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> order;
562 };

564 template<graph_model G>
565 [[nodiscard]] topological_result<typename G::index_type>
566 topological_sort(const G &graph)
567 {
568   using vertex_descriptor = typename G::vertex_descriptor;
569   topological_result<typename G::index_type> result;
570   if constexpr ( !G::is_directed ) {
571     if ( graph.edges_count() != 0 ) {
572       result.status = algorithm_status::not_a_dag;
573       return result;
574     }
575   }
576   micron::vector<usize, micron::allocator_serial<>, false> indegree(graph.vertex_slots(), usize(0));
577   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> queue;
578   queue.reserve(graph.vertices_count());
579   result.order.reserve(graph.vertices_count());
580   for ( auto vertex : graph.vertices() ) {
581     indegree.data()[static_cast<usize>(vertex.value)] = graph.in_degree(vertex);
582     if ( indegree.data()[static_cast<usize>(vertex.value)] == 0 ) queue.push_back(vertex);
583   }
584   usize head = 0;
585   while ( head < queue.size() ) {
586     const vertex_descriptor u = queue.data()[head++];
587     result.order.push_back(u);
588     for ( auto v : graph.out_neighbors(u) ) {
589       usize &degree = indegree.data()[static_cast<usize>(v.value)];
590       if ( degree > 0 && --degree == 0 ) queue.push_back(v);
591     }
592   }
593   if ( result.order.size() != graph.vertices_count() ) result.status = algorithm_status::not_a_dag;
594   return result;
595 }

597 template<graph_model G>
598 [[nodiscard]] auto
599 topological_generations(const G &graph)
600 {
601   using vertex_descriptor = typename G::vertex_descriptor;
602   micron::vector<micron::vector<vertex_descriptor, micron::allocator_serial<>, false>, micron::allocator_serial<>, false> result;
603   auto sorted = topological_sort(graph);
604   if ( sorted.status != algorithm_status::ok ) return result;
605   micron::vector<usize, micron::allocator_serial<>, false> level(graph.vertex_slots(), usize(0));
606   usize max_level = 0;
607   for ( auto u : sorted.order ) {
608     const usize next = level.data()[static_cast<usize>(u.value)] + 1;
609     for ( auto v : graph.out_neighbors(u) ) {
610       usize &target_level = level.data()[static_cast<usize>(v.value)];
611       if ( next > target_level ) target_level = next;
612       if ( target_level > max_level ) max_level = target_level;
613     }
614   }
615   result.resize(max_level + 1);
616   for ( auto vertex : sorted.order ) result.data()[level.data()[static_cast<usize>(vertex.value)]].push_back(vertex);
617   return result;
618 }

620 template<graph_model G, typename WeightMap>
621 [[nodiscard]] auto
622 dag_shortest_paths(const G &graph, typename G::vertex_descriptor source, WeightMap weight_map)
623 {
624   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
625   shortest_path_workspace<typename G::index_type, D> workspace;
626   workspace.reserve(graph.vertex_slots());
627   workspace.reset(graph.vertex_slots());
628   auto sorted = topological_sort(graph);
629   if ( sorted.status != algorithm_status::ok || !graph.has_vertex(source) )
630     return shortest_paths_result<typename G::index_type, D>{ sorted.status == algorithm_status::ok ? algorithm_status::invalid_vertex
631                                                                                                    : sorted.status,
632                                                              source, micron::move(workspace.distance), micron::move(workspace.parent),
633                                                              micron::move(workspace.reached) };
634   const usize ss = static_cast<usize>(source.value);
635   workspace.reached.data()[ss] = 1;
636   workspace.parent.data()[ss] = source;
637   algorithm_status status = algorithm_status::ok;
638   for ( auto u : sorted.order ) {
639     const usize us = static_cast<usize>(u.value);
640     if ( !workspace.reached.data()[us] ) continue;
641     for ( auto edge : graph.out_edges(u) ) {
642       const auto raw_weight = __impl::weight(weight_map, graph, edge);
643       if ( __impl::nonfinite_weight(raw_weight) ) {
644         status = algorithm_status::invalid_weight;
645         break;
646       }
647       const auto v = graph.target(edge);
648       const usize vs = static_cast<usize>(v.value);
649       D candidate{};
650       if ( !__impl::add_distance(workspace.distance.data()[us], raw_weight, candidate) ) {
651         status = algorithm_status::overflow;
652         break;
653       }
654       if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs] ) {
655         workspace.reached.data()[vs] = 1;
656         workspace.distance.data()[vs] = candidate;
657         workspace.parent.data()[vs] = u;
658       }
659     }
660     if ( status != algorithm_status::ok ) break;
661   }
662   return shortest_paths_result<typename G::index_type, D>{ status, source, micron::move(workspace.distance), micron::move(workspace.parent),
663                                                            micron::move(workspace.reached) };
664 }

726 template<micron::integral I, typename D> struct all_pairs_shortest_paths_result {
727   algorithm_status status{ algorithm_status::ok };
728   usize slots{};
729   micron::vector<D, micron::allocator_serial<>, false> distance;
730   micron::vector<u8, micron::allocator_serial<>, false> reached;
731 
732   [[nodiscard]] bool
733   contains(vertex_id<I> u, vertex_id<I> v) const noexcept
734   {
735     return reached.data()[static_cast<usize>(u.value) * slots + static_cast<usize>(v.value)] != 0;
736   }
737 
738   [[nodiscard]] const D *
739   try_distance(vertex_id<I> u, vertex_id<I> v) const noexcept
740   {
741     return contains(u, v) ? distance.data() + static_cast<usize>(u.value) * slots + static_cast<usize>(v.value) : nullptr;
742   }
743 };

745 template<graph_model G, typename WeightMap>
746 [[nodiscard]] auto
747 floyd_warshall(const G &graph, WeightMap weight_map)
748 {
749   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
750   const usize n = graph.vertex_slots();
751   all_pairs_shortest_paths_result<typename G::index_type, D> result{ algorithm_status::ok, n,
752                                                                      micron::vector<D, micron::allocator_serial<>, false>(n * n, D{}),
753                                                                      micron::vector<u8, micron::allocator_serial<>, false>(n * n, u8(0)) };
754   for ( auto vertex : graph.vertices() ) {
755     const usize i = static_cast<usize>(vertex.value);
756     result.reached.data()[i * n + i] = 1;
757   }
758   for ( auto edge : graph.edges() ) {
759     const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge.id));
760     if ( __impl::nonfinite_weight(weight) ) {
761       result.status = algorithm_status::invalid_weight;
762       return result;
763     }
764     const usize u = static_cast<usize>(edge.source.value);
765     const usize v = static_cast<usize>(edge.target.value);
766     if ( !result.reached.data()[u * n + v] || weight < result.distance.data()[u * n + v] ) {
767       result.reached.data()[u * n + v] = 1;
768       result.distance.data()[u * n + v] = weight;
769     }
770     if constexpr ( !G::is_directed ) {
771       result.reached.data()[v * n + u] = 1;
772       result.distance.data()[v * n + u] = weight;
773     }
774   }
775   for ( auto k_id : graph.vertices() ) {
776     const usize k = static_cast<usize>(k_id.value);
777     for ( auto i_id : graph.vertices() ) {
778       const usize i = static_cast<usize>(i_id.value);
779       if ( !result.reached.data()[i * n + k] ) continue;
780       for ( auto j_id : graph.vertices() ) {
781         const usize j = static_cast<usize>(j_id.value);
782         if ( !result.reached.data()[k * n + j] ) continue;
783         D candidate{};
784         if ( !__impl::add_distance(result.distance.data()[i * n + k], result.distance.data()[k * n + j], candidate) ) {
785           result.status = algorithm_status::overflow;
786           return result;
787         }
788         if ( !result.reached.data()[i * n + j] || candidate < result.distance.data()[i * n + j] ) {
789           result.reached.data()[i * n + j] = 1;
790           result.distance.data()[i * n + j] = candidate;
791         }
792       }
793     }
794   }
795   if constexpr ( micron::is_signed_v<D> || micron::is_floating_point_v<D> ) {
796     for ( auto vertex : graph.vertices() ) {
797       const usize i = static_cast<usize>(vertex.value);
798       if ( result.distance.data()[i * n + i] < D(0) ) result.status = algorithm_status::negative_cycle;
799     }
800   }
801   return result;
802 }

811 template<micron::integral I, typename D> struct johnson_workspace {
812   micron::vector<D, micron::allocator_serial<>, false> potential;
813   shortest_path_workspace<I, D> shortest;
814 
815   void
816   reserve(usize vertex_slots, usize edge_entries)
817   {
818     potential.reserve(vertex_slots);
819     shortest.reserve(vertex_slots, edge_entries);
820   }
821 };

823 template<graph_model G, typename WeightMap>
824 [[nodiscard]] auto
825 johnson(const G &graph, WeightMap weight_map,
826         johnson_workspace<typename G::index_type,
827                           micron::remove_cvref_t<decltype(__impl::weight(micron::declval<WeightMap &>(), micron::declval<const G &>(),
828                                                                          micron::declval<typename G::edge_descriptor>()))>> &storage)
829 {
830   using D = micron::remove_cvref_t<decltype(__impl::weight(weight_map, graph, micron::declval<typename G::edge_descriptor>()))>;
831   using I = typename G::index_type;
832   using vertex_descriptor = typename G::vertex_descriptor;
833   const usize slots = graph.vertex_slots();
834   all_pairs_shortest_paths_result<I, D> result{ algorithm_status::ok, slots,
835                                                 micron::vector<D, micron::allocator_serial<>, false>(slots * slots, D{}),
836                                                 micron::vector<u8, micron::allocator_serial<>, false>(slots * slots, u8(0)) };
837   storage.reserve(slots, graph.edges_count() + 1);
838   storage.potential.resize(slots, D{});
839   storage.potential.fill(D{});
840   auto &potential = storage.potential;
841 
842   for ( auto edge : graph.edges() )
843     if ( __impl::nonfinite_weight(static_cast<D>(__impl::weight(weight_map, graph, edge.id))) ) {
844       result.status = algorithm_status::invalid_weight;
845       return result;
846     }
847 
848   auto relax_potential = [&](vertex_descriptor u, vertex_descriptor v, const D &weight) -> bool {
849     D candidate{};
850     if ( !__impl::add_distance(potential.data()[static_cast<usize>(u.value)], weight, candidate) ) {
851       result.status = algorithm_status::overflow;
852       return false;
853     }
854     D &target = potential.data()[static_cast<usize>(v.value)];
855     if ( candidate < target ) {
856       target = candidate;
857       return true;
858     }
859     return false;
860   };
861 
862   for ( usize pass = 1; pass < graph.vertices_count(); ++pass ) {
863     bool changed = false;
864     for ( auto edge : graph.edges() ) {
865       const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge.id));
866       changed |= relax_potential(edge.source, edge.target, weight);
867       if constexpr ( !G::is_directed ) changed |= relax_potential(edge.target, edge.source, weight);
868       if ( result.status != algorithm_status::ok ) return result;
869     }
870     if ( !changed ) break;
871   }
872 
873   for ( auto edge : graph.edges() ) {
874     const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge.id));
875     auto can_relax = [&](vertex_descriptor u, vertex_descriptor v) {
876       D candidate{};
877       if ( !__impl::add_distance(potential.data()[static_cast<usize>(u.value)], weight, candidate) ) {
878         result.status = algorithm_status::overflow;
879         return false;
880       }
881       return candidate < potential.data()[static_cast<usize>(v.value)];
882     };
883     if ( can_relax(edge.source, edge.target) || (!G::is_directed && can_relax(edge.target, edge.source)) ) {
884       if ( result.status == algorithm_status::ok ) result.status = algorithm_status::negative_cycle;
885       return result;
886     }
887   }
888 
889   auto &workspace = storage.shortest;
890   for ( auto source : graph.vertices() ) {
891     workspace.reset(slots);
892     const usize source_slot = static_cast<usize>(source.value);
893     workspace.reached.data()[source_slot] = 1;
894     workspace.parent.data()[source_slot] = source;
895     workspace.queue.push_back(source);
896     workspace.queue_distance.push_back(D{});
897 
898     auto push_heap = [&](vertex_descriptor vertex, const D &distance) {
899       workspace.queue.push_back(vertex);
900       workspace.queue_distance.push_back(distance);
901       usize child = workspace.queue.size() - 1;
902       while ( child ) {
903         const usize parent = (child - 1) / 2;
904         if ( !(workspace.queue_distance.data()[child] < workspace.queue_distance.data()[parent]) ) break;
905         micron::swap(workspace.queue.data()[child], workspace.queue.data()[parent]);
906         micron::swap(workspace.queue_distance.data()[child], workspace.queue_distance.data()[parent]);
907         child = parent;
908       }
909     };
910     auto pop_heap = [&]() {
911       const vertex_descriptor selected = workspace.queue.data()[0];
912       const D selected_distance = workspace.queue_distance.data()[0];
913       const usize last = workspace.queue.size() - 1;
914       workspace.queue.data()[0] = workspace.queue.data()[last];
915       workspace.queue_distance.data()[0] = workspace.queue_distance.data()[last];
916       workspace.queue.pop_back();
917       workspace.queue_distance.pop_back();
918       usize parent = 0;
919       while ( parent < workspace.queue.size() ) {
920         const usize left = parent * 2 + 1;
921         if ( left >= workspace.queue.size() ) break;
922         const usize right = left + 1;
923         const usize child = right < workspace.queue.size() && workspace.queue_distance.data()[right] < workspace.queue_distance.data()[left]
924                                 ? right
925                                 : left;
926         if ( !(workspace.queue_distance.data()[child] < workspace.queue_distance.data()[parent]) ) break;
927         micron::swap(workspace.queue.data()[child], workspace.queue.data()[parent]);
928         micron::swap(workspace.queue_distance.data()[child], workspace.queue_distance.data()[parent]);
929         parent = child;
930       }
931       return micron::pair<vertex_descriptor, D>{ selected, selected_distance };
932     };
933 
934     while ( !workspace.queue.empty() ) {
935       const auto selected_pair = pop_heap();
936       const vertex_descriptor u = selected_pair.a;
937       const usize us = static_cast<usize>(u.value);
938       if ( workspace.settled.data()[us] || selected_pair.b != workspace.distance.data()[us] ) continue;
939       workspace.settled.data()[us] = 1;
940       for ( auto edge : graph.out_edges(u) ) {
941         const vertex_descriptor v = __impl::edge_neighbor(graph, edge, u);
942         const usize vs = static_cast<usize>(v.value);
943         const D weight = static_cast<D>(__impl::weight(weight_map, graph, edge));
944         D reduced{}, partial{};
945         if ( !__impl::add_distance(weight, potential.data()[us], partial)
946              || !__impl::sub_distance(partial, potential.data()[vs], reduced) ) {
947           result.status = algorithm_status::overflow;
948           return result;
949         }
950         D candidate{};
951         if ( !__impl::add_distance(workspace.distance.data()[us], reduced, candidate) ) {
952           result.status = algorithm_status::overflow;
953           return result;
954         }
955         if ( !workspace.reached.data()[vs] || candidate < workspace.distance.data()[vs] ) {
956           workspace.reached.data()[vs] = 1;
957           workspace.distance.data()[vs] = candidate;
958           workspace.parent.data()[vs] = u;
959           push_heap(v, candidate);
960         }
961       }
962     }
963 
964     for ( auto target : graph.vertices() ) {
965       const usize target_slot = static_cast<usize>(target.value);
966       if ( !workspace.reached.data()[target_slot] ) continue;
967       D partial{}, original{};
968       if ( !__impl::sub_distance(workspace.distance.data()[target_slot], potential.data()[source_slot], partial)
969            || !__impl::add_distance(partial, potential.data()[target_slot], original) ) {
970         result.status = algorithm_status::overflow;
971         return result;
972       }
973       result.reached.data()[source_slot * slots + target_slot] = 1;
974       result.distance.data()[source_slot * slots + target_slot] = original;
975     }
976   }
977   return result;
978 }
