49   template<graphs::graph_model G>
50   void
51   __build(const G &source, mapping_vertices &dense_to_vertex, mapping_vertices &vertex_remap, mapping_edges &edge_remap)
52   {
53     dense_to_vertex.clear();
54     dense_to_vertex.reserve(source.vertices_count());
55     vertex_remap = mapping_vertices(source.vertex_slots(), vertex_descriptor::invalid());
56     edge_remap = mapping_edges(source.edge_slots(), edge_descriptor::invalid());
57     __vertices.reserve(source.vertices_count());
58     __edges.reserve(source.edges_count());
59 
60     for ( auto vertex : source.vertices() ) {
61       const vertex_descriptor dense(static_cast<Index>(__vertices.size()));
62       vertex_remap.data()[static_cast<usize>(vertex.value)] = dense;
63       dense_to_vertex.push_back(vertex_descriptor(static_cast<Index>(vertex.value)));
64       __vertices.push_back(source.vertex_property_unchecked(vertex));
65     }
66     for ( auto edge : source.edges() ) {
67       const vertex_descriptor u = vertex_remap.data()[static_cast<usize>(edge.source.value)];
68       const vertex_descriptor v = vertex_remap.data()[static_cast<usize>(edge.target.value)];
69       const edge_descriptor dense(static_cast<Index>(__edges.size()));
70       edge_remap.data()[static_cast<usize>(edge.id.value)] = dense;
71       __edges.emplace_back(u.value, v.value, edge.property);
72     }
73 
74     const usize vertices = __vertices.size();
75     __out_outer.resize(vertices + 1, usize(0));
76     if constexpr ( __has_in_index ) __in_outer.resize(vertices + 1, usize(0));
77     for ( usize edge = 0; edge < __edges.size(); ++edge ) {
78       const edge_slot &slot = __edges.data()[edge];
79       ++__out_outer.data()[static_cast<usize>(slot.source) + 1];
80       if constexpr ( __has_in_index )
81         ++__in_outer.data()[static_cast<usize>(slot.target) + 1];
82       else if constexpr ( is_undirected )
83         ++__out_outer.data()[static_cast<usize>(slot.target) + 1];
84     }
85     for ( usize vertex = 1; vertex <= vertices; ++vertex ) __out_outer.data()[vertex] += __out_outer.data()[vertex - 1];
86     __out_arcs.resize(__out_outer.data()[vertices]);
87     micron::vector<usize, Alloc, false> out_cursor(vertices, usize(0));
88     for ( usize vertex = 0; vertex < vertices; ++vertex ) out_cursor.data()[vertex] = __out_outer.data()[vertex];
89 
90     micron::vector<usize, Alloc, false> in_cursor;
91     if constexpr ( __has_in_index ) {
92       for ( usize vertex = 1; vertex <= vertices; ++vertex ) __in_outer.data()[vertex] += __in_outer.data()[vertex - 1];
93       __in_arcs.resize(__in_outer.data()[vertices]);
94       in_cursor.resize(vertices, usize(0));
95       for ( usize vertex = 0; vertex < vertices; ++vertex ) in_cursor.data()[vertex] = __in_outer.data()[vertex];
96     }
97 
98     for ( usize edge = 0; edge < __edges.size(); ++edge ) {
99       const edge_slot &slot = __edges.data()[edge];
100       const usize u = static_cast<usize>(slot.source);
101       const usize v = static_cast<usize>(slot.target);
102       __out_arcs.data()[out_cursor.data()[u]++] = arc{ static_cast<Index>(edge), slot.target };
103       if constexpr ( __has_in_index )
104         __in_arcs.data()[in_cursor.data()[v]++] = arc{ static_cast<Index>(edge), slot.source };
105       else if constexpr ( is_undirected )
106         __out_arcs.data()[out_cursor.data()[v]++] = arc{ static_cast<Index>(edge), slot.source };
107     }
108 
109     auto sort_rows = [](auto &arcs, const auto &outer, usize row_count) {
110       for ( usize row = 0; row < row_count; ++row ) {
111         const usize begin = outer.data()[row];
112         const usize end = outer.data()[row + 1];
113         const usize count = end - begin;
114         auto less = [](const arc &left, const arc &right) {
115           if ( left.neighbor != right.neighbor ) return left.neighbor < right.neighbor;
116           return static_cast<Index>(left.edge) < static_cast<Index>(right.edge);
117         };
118         auto sift = [&](usize root, usize length) {
119           for ( ;; ) {
120             usize child = root * 2 + 1;
121             if ( child >= length ) break;
122             if ( child + 1 < length && less(arcs.data()[begin + child], arcs.data()[begin + child + 1]) ) ++child;
123             if ( !less(arcs.data()[begin + root], arcs.data()[begin + child]) ) break;
124             micron::swap(arcs.data()[begin + root], arcs.data()[begin + child]);
125             root = child;
126           }
127         };
128         for ( usize start = count / 2; start-- > 0; ) sift(start, count);
129         for ( usize length = count; length > 1; ) {
130           --length;
131           micron::swap(arcs.data()[begin], arcs.data()[begin + length]);
132           sift(0, length);
133         }
134       }
135     };
136     sort_rows(__out_arcs, __out_outer, vertices);
137     if constexpr ( __has_in_index ) sort_rows(__in_arcs, __in_outer, vertices);
138   }

249   [[nodiscard]] edge_descriptor
250   find_edge(vertex_descriptor u, vertex_descriptor v) const noexcept
251   {
252     if ( !has_vertex(u) || !has_vertex(v) ) return edge_descriptor::invalid();
253     usize begin = __out_outer.data()[static_cast<usize>(u.value)];
254     usize end = __out_outer.data()[static_cast<usize>(u.value) + 1];
255     while ( begin < end ) {
256       const usize middle = begin + (end - begin) / 2;
257       if ( __out_arcs.data()[middle].neighbor < v.value )
258         begin = middle + 1;
259       else
260         end = middle;
261     }
262     const usize row_end = __out_outer.data()[static_cast<usize>(u.value) + 1];
263     if ( begin < row_end && __out_arcs.data()[begin].neighbor == v.value ) return edge_descriptor(__out_arcs.data()[begin].edge);
264     return edge_descriptor::invalid();
265   }

673   [[nodiscard]] usize
674   out_degree(vertex_descriptor vertex) const noexcept
675   {
676     if ( !has_vertex(vertex) ) return 0;
677     const usize slot = static_cast<usize>(vertex.value);
678     return __out_outer.data()[slot + 1] - __out_outer.data()[slot];
679   }
680 
681   [[nodiscard]] usize
682   in_degree(vertex_descriptor vertex) const noexcept
683   {
684     if constexpr ( is_undirected ) return out_degree(vertex);
685     if ( !has_vertex(vertex) ) return 0;
686     if constexpr ( has_in_index ) {
687       const usize slot = static_cast<usize>(vertex.value);
688       return __in_outer.data()[slot + 1] - __in_outer.data()[slot];
689     } else {
690       usize result = 0;
691       for ( auto edge : in_edges(vertex) ) {
692         (void)edge;
693         ++result;
694       }
695       return result;
696     }
697   }
698 
699   [[nodiscard]] usize
700   degree(vertex_descriptor vertex) const noexcept
701   {
702     if constexpr ( is_directed ) return out_degree(vertex) + in_degree(vertex);
703     return out_degree(vertex);
704   }

707   thaw() const
708   {
709     return graphs::thaw(*this);
710   }

712   [[nodiscard]] auto
713   thaw_stable() const
714   {
715     return graphs::thaw_stable(*this);

718   template<typename Out>
719   void
720   __micron_print(Out &out) const
721   {
722     char mode = '\0';
723     if constexpr ( requires { out.mode(); } ) mode = out.mode();
724     auto print_vertex = [&](vertex_descriptor vertex) {
725       if constexpr ( graphs::labeled_bundle<VertexProperty> ) {
726         out.elem(vertex_property_unchecked(vertex).label);
727         out.raw(" @ ", 3);
728       }
729       out.elem(vertex);
730     };
731     if ( mode == 's' ) {
732       out.raw("graph{ directed: ", 17);
733       out.raw(is_directed ? "true" : "false", is_directed ? 4 : 5);
734       out.raw(", simple: ", 10);
735       out.raw(is_simple ? "true" : "false", is_simple ? 4 : 5);
736       out.raw(", vertices: ", 12);
737       out.num(static_cast<u64>(vertices_count()));
738       out.raw(", edges: ", 9);
739       out.num(static_cast<u64>(edges_count()));
740       out.raw(" }", 2);
741       return;
742     }
743     if ( mode == 'a' ) {
744       out.raw("graph{ ", 7);
745       bool first_vertex = true;
746       for ( auto vertex : vertices() ) {
747         if ( !first_vertex ) out.raw(", ", 2);
748         first_vertex = false;
749         print_vertex(vertex);
750         out.raw(": { ", 4);
751         bool first_neighbor = true;
752         for ( auto neighbor : out_neighbors(vertex) ) {
753           if ( !first_neighbor ) out.raw(", ", 2);
754           first_neighbor = false;
755           print_vertex(neighbor);
756         }
757         out.raw(" }", 2);
758       }
759       out.raw(" }", 2);
760       return;
761     }
762     out.raw("graph{ directed: ", 17);
763     out.raw(is_directed ? "true" : "false", is_directed ? 4 : 5);
764     out.raw(", simple: ", 10);
765     out.raw(is_simple ? "true" : "false", is_simple ? 4 : 5);
766     if ( mode != 'e' ) {
767       out.raw(", vertices: { ", 14);
768       bool first_vertex = true;
769       for ( auto vertex : vertices() ) {
770         if ( !first_vertex ) out.raw(", ", 2);
771         first_vertex = false;
772         print_vertex(vertex);
773         out.raw(": ", 2);
774         if constexpr ( graphs::labeled_bundle<VertexProperty> )
775           out.elem(vertex_property_unchecked(vertex).property);
776         else
777           out.elem(vertex_property_unchecked(vertex));
778       }
779       out.raw(" }", 2);
780     }
781     out.raw(", edges: { ", 11);
782     bool first_edge = true;
783     for ( auto edge : edges() ) {
784       if ( !first_edge ) out.raw(", ", 2);
785       first_edge = false;
786       print_vertex(edge.source);
787       out.raw(is_directed ? " -> " : " -- ", 4);
788       print_vertex(edge.target);
789       out.raw(": ", 2);
790       out.elem(edge.property);
791     }
792     out.raw(" } }", 4);
793   }
