36 template<typename Out, micron::integral I> struct representation_result {
37   Out value;
38   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> dense_to_vertex;
39   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> vertex_remap;
40   micron::vector<edge_id<I>, micron::allocator_serial<>, false> edge_remap;
41   algorithm_status status{ algorithm_status::ok };
42 
43   [[nodiscard]] bool
44   succeeded() const noexcept
45   {
46     return status == algorithm_status::ok;
47   }
48 
49   [[nodiscard]] explicit
50   operator bool() const noexcept
51   {
52     return succeeded();
53   }
54 
55   [[nodiscard]] Out *
56   operator->() noexcept
57   {
58     return micron::addressof(value);
59   }
60 
61   [[nodiscard]] const Out *
62   operator->() const noexcept
63   {
64     return micron::addressof(value);
65   }
66 
67   [[nodiscard]] Out &
68   operator*() noexcept
69   {
70     return value;
71   }
72 
73   [[nodiscard]] const Out &
74   operator*() const noexcept
75   {
76     return value;
77   }

101 template<typename G>
102 [[nodiscard]] auto
103 __copy_representation(const G &source)
104 {
105   using I = typename G::index_type;
106   using Out = thawed_graph_t<G>;
107   representation_result<Out, I> result{
108     Out(source.graph_property()),
109     {},
110     micron::vector<vertex_id<I>, micron::allocator_serial<>, false>(source.vertex_slots(), vertex_id<I>::invalid()),
111     micron::vector<edge_id<I>, micron::allocator_serial<>, false>(source.edge_slots(), edge_id<I>::invalid())
112   };
113   result.value.reserve_vertices(source.vertices_count());
114   result.value.reserve_edges(source.edges_count());
115   result.dense_to_vertex.reserve(source.vertices_count());
116   for ( auto vertex : source.vertices() ) {
117     const vertex_id<I> dense = result.value.add_vertex(source.vertex_property_unchecked(vertex));
118     if ( !dense.valid() ) {
119       result.status = algorithm_status::overflow;
120       return result;
121     }
122     result.vertex_remap.data()[static_cast<usize>(vertex.value)] = dense;
123     result.dense_to_vertex.push_back(vertex);
124   }
125   for ( auto edge : source.edges() ) {
126     const auto u = result.vertex_remap.data()[static_cast<usize>(edge.source.value)];
127     const auto v = result.vertex_remap.data()[static_cast<usize>(edge.target.value)];
128     auto inserted = result.value.add_edge(u, v, edge.property);
129     if ( !inserted.inserted() ) {
130       result.status = inserted.status == edge_insert_status::index_overflow ? algorithm_status::overflow : algorithm_status::invalid_graph;
131       return result;
132     }
133     result.edge_remap.data()[static_cast<usize>(edge.id.value)] = inserted.id;
134   }
135   return result;
136 }

173 template<typename G>
174 [[nodiscard]] auto
175 thaw_stable(const G &source)
176 {
177   using I = typename G::index_type;
178   using Out = stable_thawed_graph_t<G>;
179   representation_result<Out, I> result{
180     Out(source.graph_property()),
181     {},
182     micron::vector<vertex_id<I>, micron::allocator_serial<>, false>(source.vertex_slots(), vertex_id<I>::invalid()),
183     micron::vector<edge_id<I>, micron::allocator_serial<>, false>(source.edge_slots(), edge_id<I>::invalid())
184   };
185   result.value.reserve_vertices(source.vertex_slots());
186   result.value.reserve_edges(source.edge_slots());
187   result.dense_to_vertex.reserve(source.vertices_count());
188   for ( usize slot = 0; slot < source.vertex_slots(); ++slot ) {
189     const vertex_id<I> original(static_cast<I>(slot));
190     if ( source.has_vertex(original) ) {
191       const vertex_id<I> inserted = result.value.add_vertex(source.vertex_property_unchecked(original));
192       if ( !inserted.valid() ) {
193         result.status = algorithm_status::overflow;
194         return result;
195       }
196       result.vertex_remap.data()[slot] = inserted;
197       result.dense_to_vertex.push_back(original);
198     } else {
199       if ( !result.value.__import_dead_vertex_slot() ) {
200         result.status = algorithm_status::overflow;
201         return result;
202       }
203     }
204   }
205   for ( usize slot = 0; slot < source.edge_slots(); ++slot ) {
206     const edge_id<I> original(static_cast<I>(slot));
207     if ( source.has_edge(original) ) {
208       const auto u = result.vertex_remap.data()[static_cast<usize>(source.source(original).value)];
209       const auto v = result.vertex_remap.data()[static_cast<usize>(source.target(original).value)];
210       auto inserted = result.value.add_edge(u, v, source.edge_property_unchecked(original));
211       if ( !inserted.inserted() ) {
212         result.status
213             = inserted.status == edge_insert_status::index_overflow ? algorithm_status::overflow : algorithm_status::invalid_graph;
214         return result;
215       }
216       result.edge_remap.data()[slot] = inserted.id;
217     } else {
218       if ( !result.value.__import_dead_edge_slot() ) {
219         result.status = algorithm_status::overflow;
220         return result;
221       }
222     }
223   }
224   return result;
225 }

227 template<typename G>
228 [[nodiscard]] auto
229 to_edge_list(const G &source)
230 {
231   using I = typename G::index_type;
232   using Out = edge_list_graph<typename G::vertex_property_type, typename G::edge_property_type, typename G::graph_property_type, I,
233                               typename G::direction_type, typename G::multiplicity_type, typename G::loop_type, typename G::allocator_type>;
234   auto stable = __copy_representation(source);
235   representation_result<Out, I> result{ Out(stable.value.graph_property()), micron::move(stable.dense_to_vertex),
236                                         micron::move(stable.vertex_remap), micron::move(stable.edge_remap), stable.status };
237   if ( !stable ) return result;
238   result.value.reserve_vertices(stable.value.vertices_count());
239   result.value.reserve_edges(stable.value.edges_count());
240   for ( auto v : stable.value.vertices() )
241     if ( !result.value.add_vertex(stable.value.vertex_property_unchecked(v)).valid() ) {
242       result.status = algorithm_status::overflow;
243       return result;
244     }
245   for ( auto e : stable.value.edges() ) {
246     const auto inserted = result.value.add_edge(e.source, e.target, e.property);
247     if ( !inserted.inserted() ) {
248       result.status = inserted.status == edge_insert_status::index_overflow ? algorithm_status::overflow : algorithm_status::invalid_graph;
249       return result;
250     }
251   }
252   return result;
253 }

255 template<typename Out, typename G>
256 [[nodiscard]] auto
257 __convert_packed_representation(const G &source)
258 {
259   using I = typename G::index_type;
260   representation_result<Out, I> result{
261     Out(source.graph_property()),
262     {},
263     micron::vector<vertex_id<I>, micron::allocator_serial<>, false>(source.vertex_slots(), vertex_id<I>::invalid()),
264     micron::vector<edge_id<I>, micron::allocator_serial<>, false>(source.edge_slots(), edge_id<I>::invalid())
265   };
266   result.dense_to_vertex.reserve(source.vertices_count());
267   for ( auto vertex : source.vertices() ) {
268     const auto dense = result.value.add_vertex(source.vertex_property_unchecked(vertex));
269     if ( !dense.valid() ) {
270       result.status = algorithm_status::overflow;
271       return result;
272     }
273     result.vertex_remap.data()[static_cast<usize>(vertex.value)] = dense;
274     result.dense_to_vertex.push_back(vertex);
275   }
276   for ( auto edge : source.edges() ) {
277     const auto u = result.vertex_remap.data()[static_cast<usize>(edge.source.value)];
278     const auto v = result.vertex_remap.data()[static_cast<usize>(edge.target.value)];
279     auto inserted = result.value.add_edge(u, v, edge.property);
280     if ( !inserted.inserted() ) {
281       result.status = inserted.status == edge_insert_status::index_overflow ? algorithm_status::overflow : algorithm_status::invalid_graph;
282       return result;
283     }
284     result.edge_remap.data()[static_cast<usize>(edge.id.value)] = inserted.id;
285   }
286   return result;
