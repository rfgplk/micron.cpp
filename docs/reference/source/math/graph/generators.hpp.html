46 [[nodiscard]] G
47 complete_graph(usize vertices)
48 {
49   G result;
50   (void)result.add_vertices(vertices);
51   for ( usize i = 0; i < vertices; ++i ) {
52     const usize begin = G::is_directed ? 0 : i + 1;
53     for ( usize j = begin; j < vertices; ++j ) {
54       if ( i == j && !G::allows_loops ) continue;
55       (void)result.add_edge(static_cast<typename G::index_type>(i), static_cast<typename G::index_type>(j));
56     }
57   }
58   return result;

73 wheel_graph(usize vertices)
74 {
75   if ( vertices < 2 ) return empty_graph<G>(vertices);
76   G result;
77   (void)result.add_vertices(vertices);
78   for ( usize i = 1; i < vertices; ++i ) {
79     (void)result.add_edge(typename G::index_type(0), static_cast<typename G::index_type>(i));

82   }

86 template<typename G = graph<>>
87 [[nodiscard]] G
88 grid_graph(usize rows, usize columns, bool periodic_rows = false, bool periodic_columns = false)
89 {
90   G result;
91   if ( rows == 0 || columns == 0 ) return result;
92   (void)result.add_vertices(rows * columns);
93   auto id = [columns](usize row, usize column) { return static_cast<typename G::index_type>(row * columns + column); };
94   for ( usize row = 0; row < rows; ++row ) {
95     for ( usize column = 0; column < columns; ++column ) {
96       if ( column + 1 < columns )
97         (void)result.add_edge(id(row, column), id(row, column + 1));
98       else if ( periodic_columns && columns > 2 )
99         (void)result.add_edge(id(row, column), id(row, 0));
100       if ( row + 1 < rows )
101         (void)result.add_edge(id(row, column), id(row + 1, column));
102       else if ( periodic_rows && rows > 2 )
103         (void)result.add_edge(id(row, column), id(0, column));
104     }
105   }
106   return result;
107 }

109 template<typename G = graph<>>
110 [[nodiscard]] G
111 balanced_tree(usize branching, usize height)
112 {
113   G result;
114   (void)result.add_vertex();
115   usize level_begin = 0;
116   usize level_size = 1;
117   for ( usize depth = 0; depth < height; ++depth ) {
118     const usize child_begin = result.vertices_count();
119     (void)result.add_vertices(level_size * branching);
120     usize child = child_begin;
121     for ( usize parent = level_begin; parent < level_begin + level_size; ++parent )
122       for ( usize branch = 0; branch < branching; ++branch )
123         (void)result.add_edge(static_cast<typename G::index_type>(parent), static_cast<typename G::index_type>(child++));
124     level_begin = child_begin;
125     level_size *= branching;
126   }
127   return result;
128 }

133 [[nodiscard]] inline f64
134 unit_random(Rng &rng)
135 {
136   const u64 raw = static_cast<u64>(micron::invoke(rng));
137   return static_cast<f64>(raw >> 11u) * static_cast<f64>(1.0 / 9007199254740992.0);

141 template<typename G = graph<>, typename Rng>
142 [[nodiscard]] G
143 erdos_renyi(usize vertices, f64 probability, Rng &&rng)
144 {
145   G result;
146   (void)result.add_vertices(vertices);
147   if ( probability <= f64(0) ) return result;
148   if ( probability > f64(1) ) probability = f64(1);
149   for ( usize i = 0; i < vertices; ++i ) {
150     const usize begin = G::is_directed ? 0 : i + 1;
151     for ( usize j = begin; j < vertices; ++j ) {
152       if ( i == j && !G::allows_loops ) continue;
153       if ( __impl::unit_random(rng) < probability )
154         (void)result.add_edge(static_cast<typename G::index_type>(i), static_cast<typename G::index_type>(j));
155     }
156   }
157   return result;
158 }

160 template<typename G = graph<>, typename Rng>
161 [[nodiscard]] G
162 barabasi_albert(usize vertices, usize attachments, Rng &&rng)
163 {
164   G result;
165   if ( vertices == 0 || attachments == 0 ) return result;
166   const usize initial = attachments + 1 < vertices ? attachments + 1 : vertices;
167   result = complete_graph<G>(initial);
168   micron::vector<usize, micron::allocator_serial<>, false> degree(vertices, usize(0));
169   for ( usize i = 0; i < initial; ++i )
170     degree.data()[i] = result.degree(typename G::vertex_descriptor(static_cast<typename G::index_type>(i)));
171   for ( usize vertex = initial; vertex < vertices; ++vertex ) {
172     (void)result.add_vertex();
173     usize added = 0;
174     usize attempts = 0;
175     while ( added < attachments && attempts++ < attachments * vertices * 4 ) {
176       usize total = 0;
177       for ( usize i = 0; i < vertex; ++i ) total += degree.data()[i] + 1;
178       usize pick = static_cast<usize>(micron::invoke(rng)) % total;
179       usize target = 0;
180       for ( ; target < vertex; ++target ) {
181         const usize weight = degree.data()[target] + 1;
182         if ( pick < weight ) break;
183         pick -= weight;
184       }
185       auto inserted = result.add_edge(static_cast<typename G::index_type>(vertex), static_cast<typename G::index_type>(target));
186       if ( inserted.inserted() ) {
187         ++degree.data()[vertex];
188         ++degree.data()[target];
189         ++added;
190       }
191     }
192   }
193   return result;
194 }

196 template<typename G = graph<>, typename Rng>
197 [[nodiscard]] G
198 watts_strogatz(usize vertices, usize neighbors, f64 rewire_probability, Rng &&rng)
199 {
200   G result;
201   (void)result.add_vertices(vertices);
202   if ( vertices < 2 ) return result;
203   if ( neighbors >= vertices ) neighbors = vertices - 1;
204   if ( neighbors & 1u ) --neighbors;
205   const usize half = neighbors / 2;
206   for ( usize u = 0; u < vertices; ++u ) {
207     for ( usize offset = 1; offset <= half; ++offset ) {
208       usize v = (u + offset) % vertices;
209       if ( __impl::unit_random(rng) < rewire_probability ) {
210         for ( usize attempt = 0; attempt < vertices * 2; ++attempt ) {
211           const usize candidate = static_cast<usize>(micron::invoke(rng)) % vertices;
212           if ( candidate != u
213                && !result.has_edge(static_cast<typename G::index_type>(u), static_cast<typename G::index_type>(candidate)) ) {
214             v = candidate;
215             break;
216           }
217         }
218       }
219       (void)result.add_edge(static_cast<typename G::index_type>(u), static_cast<typename G::index_type>(v));
220     }
221   }
222   return result;
223 }

225 template<typename G = graph<>, typename Rng>
226 [[nodiscard]] G
227 rmat(usize scale, usize edges, Rng &&rng, f64 a = 0.57, f64 b = 0.19, f64 c = 0.19)
228 {
229   const usize vertices = usize(1) << scale;
230   G result;
231   (void)result.add_vertices(vertices);
232   for ( usize e = 0; e < edges; ++e ) {
233     usize u = 0;
234     usize v = 0;
235     for ( usize bit = scale; bit > 0; --bit ) {
236       const f64 p = __impl::unit_random(rng);
237       const usize mask = usize(1) << (bit - 1);
238       if ( p < a ) {
239       } else if ( p < a + b ) {
240         v |= mask;
241       } else if ( p < a + b + c ) {
242         u |= mask;
243       } else {
244         u |= mask;
245         v |= mask;
246       }
247     }
248     (void)result.add_edge(static_cast<typename G::index_type>(u), static_cast<typename G::index_type>(v));
249   }
250   return result;
251 }

259 topology_vertices(const G &source)
260 {
261   topology_graph_t<G> result;

281 template<graph_model A, graph_model B>
282 [[nodiscard]] topology_graph_t<A>
283 graph_union(const A &a, const B &b)
284 {
285   topology_graph_t<A> result;
286   const usize slots = a.vertex_slots() > b.vertex_slots() ? a.vertex_slots() : b.vertex_slots();
287   for ( usize slot = 0; slot < slots; ++slot ) {
288     const bool in_a = slot < a.vertex_slots() && a.has_vertex(typename A::vertex_descriptor(static_cast<typename A::index_type>(slot)));
289     const bool in_b = slot < b.vertex_slots() && b.has_vertex(typename B::vertex_descriptor(static_cast<typename B::index_type>(slot)));
290     if ( in_a || in_b )
291       (void)result.add_vertex();
292     else
293       (void)result.__import_dead_vertex_slot();
294   }
295   for ( auto edge : a.edges() ) (void)result.add_edge(edge.source, edge.target);
296   for ( auto edge : b.edges() ) (void)result.add_edge(edge.source, edge.target);
297   return result;
298 }

336 template<graph_model G>
337 [[nodiscard]] topology_graph_t<G>
338 complement(const G &graph)
339 {
340   topology_graph_t<G> result;
341   for ( usize slot = 0; slot < graph.vertex_slots(); ++slot ) {
342     const typename G::vertex_descriptor vertex(static_cast<typename G::index_type>(slot));
343     if ( graph.has_vertex(vertex) )
344       (void)result.add_vertex();
345     else
346       (void)result.__import_dead_vertex_slot();
347   }
348   for ( auto u : graph.vertices() ) {
349     for ( auto v : graph.vertices() ) {
350       if ( u == v && !topology_graph_t<G>::allows_loops ) continue;
351       if constexpr ( !G::is_directed )
352         if ( v.value <= u.value ) continue;
353       if ( !graph.has_edge(u, v) ) (void)result.add_edge(u, v);
354     }
355   }
356   return result;
357 }

366 template<graph_model G>
367 [[nodiscard]] graph<edge_id<typename G::index_type>, empty_property, empty_property, typename G::index_type>
368 line_graph(const G &source)
369 {
370   using I = typename G::index_type;
371   graph<edge_id<I>, empty_property, empty_property, I> result;
372   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> edge_vertex(source.edge_slots(), vertex_id<I>::invalid());
373   for ( auto edge : source.edges() ) edge_vertex.data()[static_cast<usize>(edge.id.value)] = result.add_vertex(edge.id);
374   for ( auto a : source.edges() ) {
375     for ( auto b : source.edges() ) {
376       if ( b.id.value <= a.id.value ) continue;
377       if ( a.source == b.source || a.source == b.target || a.target == b.source || a.target == b.target )
378         (void)result.add_edge(edge_vertex.data()[static_cast<usize>(a.id.value)], edge_vertex.data()[static_cast<usize>(b.id.value)]);
379     }
380   }
381   return result;
382 }

384 template<graph_model G>
385 [[nodiscard]] topology_graph_t<G>
386 ego_graph(const G &source, typename G::vertex_descriptor center, usize radius = 1)
387 {
388   auto reached = diffusion(source, center, radius);
389   micron::vector<u8, micron::allocator_serial<>, false> include(source.vertex_slots(), u8(0));
390   micron::vector<typename G::vertex_descriptor, micron::allocator_serial<>, false> remap(source.vertex_slots(),
391                                                                                          G::vertex_descriptor::invalid());
392   topology_graph_t<G> result;
393   for ( auto vertex : reached.order ) {
394     const usize slot = static_cast<usize>(vertex.value);
395     include.data()[slot] = 1;
396     remap.data()[slot] = result.add_vertex();
397   }
398   for ( auto edge : source.edges() )
399     if ( include.data()[static_cast<usize>(edge.source.value)] && include.data()[static_cast<usize>(edge.target.value)] )
400       (void)result.add_edge(remap.data()[static_cast<usize>(edge.source.value)], remap.data()[static_cast<usize>(edge.target.value)]);
401   return result;
402 }

408 template<graph_model A, graph_model B>
409   requires micron::is_same_v<typename A::direction_type, typename B::direction_type>
410 [[nodiscard]] product_graph_t<A, B>
411 cartesian_product(const A &a, const B &b)
412 {
413   product_graph_t<A, B> result;
414   auto a_mapping = __dense_vertex_mapping(a);
415   auto b_mapping = __dense_vertex_mapping(b);
416   const usize an = a.vertices_count();
417   const usize bn = b.vertices_count();
418   (void)result.add_vertices(an * bn);
419   auto id = [bn](usize av, usize bv) { return av * bn + bv; };
420   for ( auto edge : a.edges() ) {
421     const usize u = static_cast<usize>(a_mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
422     const usize v = static_cast<usize>(a_mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
423     for ( usize bv = 0; bv < bn; ++bv ) (void)result.add_edge(id(u, bv), id(v, bv));
424   }
425   for ( auto edge : b.edges() ) {
426     const usize u = static_cast<usize>(b_mapping.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
427     const usize v = static_cast<usize>(b_mapping.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
428     for ( usize av = 0; av < an; ++av ) (void)result.add_edge(id(av, u), id(av, v));
429   }
430   return result;
431 }

433 template<graph_model A, graph_model B>
434   requires micron::is_same_v<typename A::direction_type, typename B::direction_type>
435 [[nodiscard]] product_graph_t<A, B>
436 tensor_product(const A &a, const B &b)
437 {
438   product_graph_t<A, B> result;
439   auto a_mapping = __dense_vertex_mapping(a);
440   auto b_mapping = __dense_vertex_mapping(b);
441   const usize bn = b.vertices_count();
442   (void)result.add_vertices(a.vertices_count() * bn);
443   auto id = [bn](usize av, usize bv) { return av * bn + bv; };
444   for ( auto ae : a.edges() ) {
445     const usize au = static_cast<usize>(a_mapping.vertex_to_dense.data()[static_cast<usize>(ae.source.value)]);
446     const usize av = static_cast<usize>(a_mapping.vertex_to_dense.data()[static_cast<usize>(ae.target.value)]);
447     for ( auto be : b.edges() ) {
448       const usize bu = static_cast<usize>(b_mapping.vertex_to_dense.data()[static_cast<usize>(be.source.value)]);
449       const usize bv = static_cast<usize>(b_mapping.vertex_to_dense.data()[static_cast<usize>(be.target.value)]);
450       (void)result.add_edge(id(au, bu), id(av, bv));
451       if constexpr ( !A::is_directed ) (void)result.add_edge(id(au, bv), id(av, bu));
452     }
453   }
454   return result;
455 }

457 template<graph_model A, graph_model B>
458   requires micron::is_same_v<typename A::direction_type, typename B::direction_type>
459 [[nodiscard]] product_graph_t<A, B>
460 strong_product(const A &a, const B &b)
461 {
462   auto result = cartesian_product(a, b);
463   auto diagonal = tensor_product(a, b);
464   for ( auto edge : diagonal.edges() ) (void)result.add_edge(edge.source, edge.target);
465   return result;
