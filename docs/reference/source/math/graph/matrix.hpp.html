18 template<typename Matrix, micron::integral I> struct matrix_conversion_result {
19   Matrix value;
20   micron::vector<vertex_id<I>, micron::allocator_serial<>, false> dense_to_vertex;
21   micron::vector<I, micron::allocator_serial<>, false> vertex_to_dense;
22   micron::vector<edge_id<I>, micron::allocator_serial<>, false> dense_to_edge;
23 };

25 template<graph_model G>
26 [[nodiscard]] auto
27 __dense_vertex_mapping(const G &graph)
28 {
29   using I = typename G::index_type;
30 
31   struct mapping {
32     micron::vector<vertex_id<I>, micron::allocator_serial<>, false> dense_to_vertex;
33     micron::vector<I, micron::allocator_serial<>, false> vertex_to_dense;
34   } result{ {}, micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), vertex_id<I>::invalid_value()) };
35 
36   result.dense_to_vertex.reserve(graph.vertices_count());
37   for ( auto vertex : graph.vertices() ) {
38     const I dense = static_cast<I>(result.dense_to_vertex.size());
39     result.vertex_to_dense.data()[static_cast<usize>(vertex.value)] = dense;
40     result.dense_to_vertex.push_back(vertex);
41   }
42   return result;
43 }

45 template<arith_scalar T = u8, graph_model G>
46 [[nodiscard]] matrix_conversion_result<dynmat<T>, typename G::index_type>
47 adjacency_matrix(const G &graph, T absent = T(0), T present = T(1))
48 {
49   using I = typename G::index_type;
50   auto mapping = __dense_vertex_mapping(graph);
51   const usize n = graph.vertices_count();
52   matrix_conversion_result<dynmat<T>, I> result{
53     dynmat<T>(n, n, absent), micron::move(mapping.dense_to_vertex), micron::move(mapping.vertex_to_dense), {}
54   };
55   for ( auto edge : graph.edges() ) {
56     const usize u = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
57     const usize v = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
58     result.value.at(u, v) = present;
59     if constexpr ( !G::is_directed ) result.value.at(v, u) = present;
60   }
61   return result;
62 }

92 template<arith_scalar T = usize, graph_model G>
93 [[nodiscard]] matrix_conversion_result<dynmat<T>, typename G::index_type>
94 degree_matrix(const G &graph)
95 {
96   using I = typename G::index_type;
97   auto mapping = __dense_vertex_mapping(graph);
98   const usize n = graph.vertices_count();
99   matrix_conversion_result<dynmat<T>, I> result{
100     dynmat<T>(n, n, T(0)), micron::move(mapping.dense_to_vertex), micron::move(mapping.vertex_to_dense), {}
101   };
102   for ( usize i = 0; i < n; ++i ) result.value.at(i, i) = static_cast<T>(graph.degree(result.dense_to_vertex.data()[i]));
103   return result;

106 template<arith_scalar T = i64, graph_model G>
107 [[nodiscard]] matrix_conversion_result<dynmat<T>, typename G::index_type>
108 laplacian_matrix(const G &graph)
109 {
110   auto result = adjacency_matrix<T>(graph, T(0), T(1));
111   const usize n = result.value.rows;
112   for ( usize i = 0; i < n; ++i ) {
113     T degree{};
114     for ( usize j = 0; j < n; ++j ) degree += result.value.at(i, j);
115     for ( usize j = 0; j < n; ++j ) result.value.at(i, j) = -result.value.at(i, j);
116     result.value.at(i, i) += degree;
117   }
118   return result;
119 }

121 template<arith_scalar T = i64, graph_model G>
122 [[nodiscard]] matrix_conversion_result<dynmat<T>, typename G::index_type>
123 incidence_matrix(const G &graph)
124 {
125   using I = typename G::index_type;
126   auto mapping = __dense_vertex_mapping(graph);
127   const usize n = graph.vertices_count();
128   const usize m = graph.edges_count();
129   matrix_conversion_result<dynmat<T>, I> result{
130     dynmat<T>(n, m, T(0)), micron::move(mapping.dense_to_vertex), micron::move(mapping.vertex_to_dense), {}
131   };
132   result.dense_to_edge.reserve(m);
133   usize column = 0;
134   for ( auto edge : graph.edges() ) {
135     const usize u = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
136     const usize v = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
137     if constexpr ( G::is_directed ) {
138       result.value.at(u, column) -= T(1);
139       result.value.at(v, column) += T(1);
140     } else {
141       result.value.at(u, column) += T(1);
142       result.value.at(v, column) += T(1);
143     }
144     result.dense_to_edge.push_back(edge.id);
145     ++column;
146   }
147   return result;
148 }

150 template<arith_scalar T = u8, graph_model G>
151 [[nodiscard]] matrix_conversion_result<sparse::csr<T, typename G::index_type>, typename G::index_type>
152 sparse_adjacency_matrix(const G &graph, T present = T(1))
153 {
154   using I = typename G::index_type;
155   auto mapping = __dense_vertex_mapping(graph);
156   const usize n = graph.vertices_count();
157   matrix_conversion_result<sparse::csr<T, I>, I> result{
158     sparse::csr<T, I>(n, n), micron::move(mapping.dense_to_vertex), micron::move(mapping.vertex_to_dense), {}
159   };
160   micron::vector<usize, micron::allocator_serial<>, false> count(n, usize(0));
161   for ( auto edge : graph.edges() ) {
162     const usize u = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)]);
163     const usize v = static_cast<usize>(result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)]);
164     ++count.data()[u];
165     if constexpr ( !G::is_directed )
166       if ( u != v ) ++count.data()[v];
167   }
168   for ( usize row = 0; row < n; ++row )
169     result.value.outer.data()[row + 1] = result.value.outer.data()[row] + static_cast<I>(count.data()[row]);
170   result.value.inner.resize(static_cast<usize>(result.value.outer.data()[n]), I{});
171   result.value.values.resize(result.value.inner.size(), T{});
172   micron::vector<usize, micron::allocator_serial<>, false> cursor(n, usize(0));
173   for ( usize row = 0; row < n; ++row ) cursor.data()[row] = static_cast<usize>(result.value.outer.data()[row]);
174   for ( auto edge : graph.edges() ) {
175     const I u = result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)];
176     const I v = result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)];
177     usize position = cursor.data()[static_cast<usize>(u)]++;
178     result.value.inner.data()[position] = v;
179     result.value.values.data()[position] = present;
180     if constexpr ( !G::is_directed ) {
181       if ( u != v ) {
182         position = cursor.data()[static_cast<usize>(v)]++;
183         result.value.inner.data()[position] = u;
184         result.value.values.data()[position] = present;
185       }
186     }
187   }
188   return result;
189 }

191 template<arith_scalar T, graph_model G, typename WeightMap = intrinsic_edge_weight>
192 [[nodiscard]] matrix_conversion_result<sparse::csr<T, typename G::index_type>, typename G::index_type>
193 sparse_weighted_adjacency_matrix(const G &graph, WeightMap weight_map = {})
194 {
195   auto result = sparse_adjacency_matrix<T>(graph, T{});
196   micron::vector<usize, micron::allocator_serial<>, false> cursor(result.value.rows, usize(0));
197   for ( usize row = 0; row < result.value.rows; ++row ) cursor.data()[row] = static_cast<usize>(result.value.outer.data()[row]);
198   for ( auto edge : graph.edges() ) {
199     const auto u = result.vertex_to_dense.data()[static_cast<usize>(edge.source.value)];
200     const auto v = result.vertex_to_dense.data()[static_cast<usize>(edge.target.value)];
201     const T weight = static_cast<T>(__impl::weight(weight_map, graph, edge.id));
202     result.value.values.data()[cursor.data()[static_cast<usize>(u)]++] = weight;
203     if constexpr ( !G::is_directed )
204       if ( u != v ) result.value.values.data()[cursor.data()[static_cast<usize>(v)]++] = weight;
205   }
206   return result;
207 }

218 template<typename G = graph<>, typename Matrix>
219 [[nodiscard]] G
220 from_adjacency_matrix(const Matrix &matrix)
221 {
222   G result;
223   const usize n = static_cast<usize>(matrix.rows < matrix.cols ? matrix.rows : matrix.cols);
224   (void)result.add_vertices(n);
225   for ( usize i = 0; i < n; ++i ) {
226     const usize begin = G::is_directed ? 0 : i;
227     for ( usize j = begin; j < n; ++j ) {
228       const auto &entry = matrix.at(i, j);
229       using entry_type = micron::remove_cvref_t<decltype(entry)>;
230       if ( entry == entry_type(0) ) continue;
231       if constexpr ( weighted_bundle<typename G::edge_property_type> )
232         (void)result.add_edge(static_cast<typename G::index_type>(i), static_cast<typename G::index_type>(j), entry);
233       else
234         (void)result.add_edge(static_cast<typename G::index_type>(i), static_cast<typename G::index_type>(j));
235     }
236   }
237   return result;
238 }

240 template<typename G = graph<>, arith_scalar T, micron::integral I>
241 [[nodiscard]] G
242 from_sparse_adjacency_matrix(const sparse::csr<T, I> &matrix)
243 {
244   G result;
245   const usize n = matrix.rows < matrix.cols ? matrix.rows : matrix.cols;
246   (void)result.add_vertices(n);
247   if ( matrix.outer.size() < matrix.rows + 1 || matrix.inner.size() != matrix.values.size() ) return result;
248   for ( usize row = 0; row < n; ++row ) {
249     usize begin = static_cast<usize>(matrix.outer.data()[row]);
250     usize end = static_cast<usize>(matrix.outer.data()[row + 1]);
251     if ( begin > end || end > matrix.inner.size() ) return G{};
252     for ( usize position = begin; position < end; ++position ) {
253       const usize column = static_cast<usize>(matrix.inner.data()[position]);
254       if ( column >= n || matrix.values.data()[position] == T{} ) continue;
255       if constexpr ( !G::is_directed )
256         if ( column < row ) continue;
257       if constexpr ( weighted_bundle<typename G::edge_property_type> )
258         (void)result.add_edge(static_cast<typename G::index_type>(row), static_cast<typename G::index_type>(column),
259                               matrix.values.data()[position]);
260       else
261         (void)result.add_edge(static_cast<typename G::index_type>(row), static_cast<typename G::index_type>(column));
262     }
263   }
264   return result;
265 }
