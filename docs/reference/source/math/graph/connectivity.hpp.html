28 template<graph_model G>
29 [[nodiscard]] components_result<typename G::index_type>
30 connected_components(const G &graph)
31 {
32   using I = typename G::index_type;
33   using vertex_descriptor = typename G::vertex_descriptor;
34   const I none = micron::numeric_limits<I>::max();
35   components_result<I> result{ algorithm_status::ok, 0, micron::vector<I, micron::allocator_serial<>, false>(graph.vertex_slots(), none) };
36   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> queue;
37   queue.reserve(graph.vertices_count());
38 
39   for ( auto root : graph.vertices() ) {
40     if ( result.component.data()[static_cast<usize>(root.value)] != none ) continue;
41     const I component_id = static_cast<I>(result.count++);
42     queue.clear();
43     queue.push_back(root);
44     result.component.data()[static_cast<usize>(root.value)] = component_id;
45     usize head = 0;
46     while ( head < queue.size() ) {
47       const vertex_descriptor u = queue.data()[head++];
48       for ( auto v : graph.out_neighbors(u) ) {
49         const usize slot = static_cast<usize>(v.value);
50         if ( result.component.data()[slot] != none ) continue;
51         result.component.data()[slot] = component_id;
52         queue.push_back(v);
53       }
54       if constexpr ( G::is_directed ) {
55         for ( auto v : graph.in_neighbors(u) ) {
56           const usize slot = static_cast<usize>(v.value);
57           if ( result.component.data()[slot] != none ) continue;
58           result.component.data()[slot] = component_id;
59           queue.push_back(v);
60         }
61       }
62     }
63   }
64   return result;
65 }

74 template<graph_model G>
75 [[nodiscard]] components_result<typename G::index_type>
76 strongly_connected_components(const G &graph)
77 {
78   using I = typename G::index_type;
79   using vertex_descriptor = typename G::vertex_descriptor;
80   if constexpr ( !G::is_directed ) return connected_components(graph);
81 
82   const usize slots = graph.vertex_slots();
83   micron::vector<u8, micron::allocator_serial<>, false> seen(slots, u8(0));
84   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> finish;
85   finish.reserve(graph.vertices_count());
86 
87   auto visit_out = [&](auto &&self, vertex_descriptor u) -> void {
88     seen.data()[static_cast<usize>(u.value)] = 1;
89     for ( auto v : graph.out_neighbors(u) )
90       if ( !seen.data()[static_cast<usize>(v.value)] ) self(self, v);
91     finish.push_back(u);
92   };
93   for ( auto vertex : graph.vertices() )
94     if ( !seen.data()[static_cast<usize>(vertex.value)] ) visit_out(visit_out, vertex);
95 
96   const I none = micron::numeric_limits<I>::max();
97   components_result<I> result{ algorithm_status::ok, 0, micron::vector<I, micron::allocator_serial<>, false>(slots, none) };
98   auto visit_in = [&](auto &&self, vertex_descriptor u, I component_id) -> void {
99     result.component.data()[static_cast<usize>(u.value)] = component_id;
100     for ( auto v : graph.in_neighbors(u) )
101       if ( result.component.data()[static_cast<usize>(v.value)] == none ) self(self, v, component_id);
102   };
103   for ( usize i = finish.size(); i > 0; --i ) {
104     const vertex_descriptor root = finish.data()[i - 1];
105     if ( result.component.data()[static_cast<usize>(root.value)] != none ) continue;
106     const I component_id = static_cast<I>(result.count++);
107     visit_in(visit_in, root, component_id);
108   }
109   return result;
110 }

119 template<graph_model G>
120 [[nodiscard]] auto
121 tarjan_scc(const G &graph)
122 {
123   using I = typename G::index_type;
124   using vertex_descriptor = typename G::vertex_descriptor;
125   if constexpr ( !G::is_directed ) return connected_components(graph);
126 
127   const usize slots = graph.vertex_slots();
128   const I none = micron::numeric_limits<I>::max();
129   components_result<I> result{ algorithm_status::ok, 0, micron::vector<I, micron::allocator_serial<>, false>(slots, none) };
130   micron::vector<usize, micron::allocator_serial<>, false> discovery(slots, usize(0));
131   micron::vector<usize, micron::allocator_serial<>, false> low(slots, usize(0));
132   micron::vector<u8, micron::allocator_serial<>, false> on_stack(slots, u8(0));
133   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> stack;
134   stack.reserve(graph.vertices_count());
135   usize time = 0;
136 
137   auto visit = [&](auto &&self, vertex_descriptor u) -> void {
138     const usize us = static_cast<usize>(u.value);
139     discovery.data()[us] = low.data()[us] = ++time;
140     stack.push_back(u);
141     on_stack.data()[us] = 1;
142 
143     for ( auto v : graph.out_neighbors(u) ) {
144       const usize vs = static_cast<usize>(v.value);
145       if ( discovery.data()[vs] == 0 ) {
146         self(self, v);
147         if ( low.data()[vs] < low.data()[us] ) low.data()[us] = low.data()[vs];
148       } else if ( on_stack.data()[vs] && discovery.data()[vs] < low.data()[us] ) {
149         low.data()[us] = discovery.data()[vs];
150       }
151     }
152 
153     if ( low.data()[us] != discovery.data()[us] ) return;
154     const I component_id = static_cast<I>(result.count++);
155     for ( ;; ) {
156       const vertex_descriptor member = stack.back();
157       stack.pop_back();
158       const usize slot = static_cast<usize>(member.value);
159       on_stack.data()[slot] = 0;
160       result.component.data()[slot] = component_id;
161       if ( member == u ) break;
162     }
163   };
164 
165   for ( auto vertex : graph.vertices() )
166     if ( discovery.data()[static_cast<usize>(vertex.value)] == 0 ) visit(visit, vertex);
167   return result;
168 }

192 template<graph_model G>
193 [[nodiscard]] condensation_result<G>
194 condensation(const G &graph)
195 {
196   using I = typename G::index_type;
197   condensation_result<G> result;
198   result.components = strongly_connected_components(graph);
199   (void)result.value.add_vertices(result.components.count);
200   for ( auto edge : graph.edges() ) {
201     const I a = result.components.component.data()[static_cast<usize>(edge.source.value)];
202     const I b = result.components.component.data()[static_cast<usize>(edge.target.value)];
203     if ( a != b ) (void)result.value.add_edge(a, b);
204   }
205   return result;
206 }

208 template<graph_model G>
209 [[nodiscard]] micron::vector<typename G::index_type, micron::allocator_serial<>, false>
210 attracting_components(const G &graph)
211 {
212   using I = typename G::index_type;
213   auto scc = strongly_connected_components(graph);
214   micron::vector<u8, micron::allocator_serial<>, false> attracting(scc.count, u8(1));
215   for ( auto edge : graph.edges() ) {
216     const I a = scc.component.data()[static_cast<usize>(edge.source.value)];
217     const I b = scc.component.data()[static_cast<usize>(edge.target.value)];
218     if ( a != b ) attracting.data()[static_cast<usize>(a)] = 0;
219   }
220   micron::vector<I, micron::allocator_serial<>, false> result;
221   for ( usize i = 0; i < attracting.size(); ++i )
222     if ( attracting.data()[i] ) result.push_back(static_cast<I>(i));
223   return result;
224 }

234 template<graph_model G>
235 [[nodiscard]] bipartite_result<typename G::index_type>
236 bipartite_test(const G &graph)
237 {
238   using vertex_descriptor = typename G::vertex_descriptor;
239   bipartite_result<typename G::index_type> result;
240   result.color = micron::vector<i8, micron::allocator_serial<>, false>(graph.vertex_slots(), i8(-1));
241   micron::vector<vertex_descriptor, micron::allocator_serial<>, false> queue;
242   queue.reserve(graph.vertices_count());
243   for ( auto root : graph.vertices() ) {
244     if ( result.color.data()[static_cast<usize>(root.value)] != -1 ) continue;
245     result.color.data()[static_cast<usize>(root.value)] = 0;
246     queue.clear();
247     queue.push_back(root);
248     usize head = 0;
249     while ( head < queue.size() ) {
250       const vertex_descriptor u = queue.data()[head++];
251       const i8 next_color = static_cast<i8>(1 - result.color.data()[static_cast<usize>(u.value)]);
252       auto inspect = [&](vertex_descriptor v) {
253         i8 &color = result.color.data()[static_cast<usize>(v.value)];
254         if ( color == -1 ) {
255           color = next_color;
256           queue.push_back(v);
257         } else if ( color != next_color ) {
258           result.bipartite = false;
259           result.conflict_a = u;
260           result.conflict_b = v;
261         }
262       };
263       for ( auto v : graph.out_neighbors(u) ) inspect(v);
264       if constexpr ( G::is_directed )
265         for ( auto v : graph.in_neighbors(u) ) inspect(v);
266       if ( !result.bipartite ) return result;
267     }
268   }
269   return result;
270 }

279 template<micron::integral I = u32> class union_find
280 {
281   micron::vector<I, micron::allocator_serial<>, false> __parent;
282   micron::vector<u8, micron::allocator_serial<>, false> __rank;
283   usize __sets{};
284 
285 public:
286   union_find() = default;
287 
288   explicit union_find(usize size) : __parent(size), __rank(size, u8(0)), __sets(size)
289   {
290     for ( usize i = 0; i < size; ++i ) __parent.data()[i] = static_cast<I>(i);
291   }
292 
293   [[nodiscard]] usize
294   size() const noexcept
295   {
296     return __parent.size();
297   }
298 
299   [[nodiscard]] usize
300   sets_count() const noexcept
301   {
302     return __sets;
303   }
304 
305   I
306   find(I item) noexcept
307   {
308     I root = item;
309     while ( __parent.data()[static_cast<usize>(root)] != root ) root = __parent.data()[static_cast<usize>(root)];
310     while ( __parent.data()[static_cast<usize>(item)] != item ) {
311       const I next = __parent.data()[static_cast<usize>(item)];
312       __parent.data()[static_cast<usize>(item)] = root;
313       item = next;
314     }
315     return root;
316   }
317 
318   bool
319   unite(I a, I b) noexcept
320   {
321     a = find(a);
322     b = find(b);
323     if ( a == b ) return false;
324     u8 &rank_a = __rank.data()[static_cast<usize>(a)];
325     u8 &rank_b = __rank.data()[static_cast<usize>(b)];
326     if ( rank_a < rank_b )
327       __parent.data()[static_cast<usize>(a)] = b;
328     else {
329       __parent.data()[static_cast<usize>(b)] = a;
330       if ( rank_a == rank_b ) ++rank_a;
331     }
332     --__sets;
333     return true;
334   }
335 
336   [[nodiscard]] bool
337   connected(I a, I b) noexcept
338   {
339     return find(a) == find(b);
340   }
341 };

348 template<graph_model G>
349 [[nodiscard]] cut_structure_result<typename G::index_type>
350 cut_structure(const G &graph)
351 {
352   using vertex_descriptor = typename G::vertex_descriptor;
353   using edge_descriptor = typename G::edge_descriptor;
354   const usize slots = graph.vertex_slots();
355   cut_structure_result<typename G::index_type> result;
356   micron::vector<usize, micron::allocator_serial<>, false> discovery(slots, usize(0));
357   micron::vector<usize, micron::allocator_serial<>, false> low(slots, usize(0));
358   micron::vector<u8, micron::allocator_serial<>, false> articulation(slots, u8(0));
359   usize time = 0;
360 
361   auto visit = [&](auto &&self, vertex_descriptor u, edge_descriptor parent_edge) -> void {
362     const usize us = static_cast<usize>(u.value);
363     discovery.data()[us] = low.data()[us] = ++time;
364     usize children = 0;
365     for ( auto edge : graph.out_edges(u) ) {
366       if ( edge == parent_edge ) continue;
367       const vertex_descriptor v = G::is_directed ? graph.target(edge) : graph.opposite(edge, u);
368       const usize vs = static_cast<usize>(v.value);
369       if ( discovery.data()[vs] == 0 ) {
370         ++children;
371         self(self, v, edge);
372         if ( low.data()[vs] < low.data()[us] ) low.data()[us] = low.data()[vs];
373         if ( parent_edge.valid() && low.data()[vs] >= discovery.data()[us] ) articulation.data()[us] = 1;
374         if ( low.data()[vs] > discovery.data()[us] ) result.bridges.push_back(edge);
375       } else if ( discovery.data()[vs] < low.data()[us] ) {
376         low.data()[us] = discovery.data()[vs];
377       }
378     }
379     if ( !parent_edge.valid() && children > 1 ) articulation.data()[us] = 1;
380   };
381 
382   for ( auto root : graph.vertices() )
383     if ( discovery.data()[static_cast<usize>(root.value)] == 0 ) visit(visit, root, edge_descriptor::invalid());
384   for ( auto vertex : graph.vertices() )
385     if ( articulation.data()[static_cast<usize>(vertex.value)] ) result.articulation_points.push_back(vertex);
386   return result;
387 }

409 template<graph_model G>
410 [[nodiscard]] biconnected_result<typename G::index_type>
411 biconnected_components(const G &graph)
412 {
413   using vertex_descriptor = typename G::vertex_descriptor;
414   using edge_descriptor = typename G::edge_descriptor;
415   biconnected_result<typename G::index_type> result;
416   if constexpr ( G::is_directed ) {
417     result.status = algorithm_status::invalid_graph;
418     return result;
419   }
420   micron::vector<usize, micron::allocator_serial<>, false> discovery(graph.vertex_slots(), usize(0));
421   micron::vector<usize, micron::allocator_serial<>, false> low(graph.vertex_slots(), usize(0));
422   micron::vector<edge_descriptor, micron::allocator_serial<>, false> stack;
423   micron::vector<u8, micron::allocator_serial<>, false> loop_seen(graph.edge_slots(), u8(0));
424   usize time = 0;
425   auto visit = [&](auto &&self, vertex_descriptor vertex, edge_descriptor parent) -> void {
426     const usize slot = static_cast<usize>(vertex.value);
427     discovery.data()[slot] = low.data()[slot] = ++time;
428     for ( auto edge : graph.out_edges(vertex) ) {
429       if ( graph.source(edge) == graph.target(edge) ) {
430         const usize es = static_cast<usize>(edge.value);
431         if ( !loop_seen.data()[es] ) {
432           loop_seen.data()[es] = 1;
433           micron::vector<edge_descriptor, micron::allocator_serial<>, false> block;
434           block.push_back(edge);
435           result.blocks.push_back(micron::move(block));
436         }
437         continue;
438       }
439       if ( edge == parent ) continue;
440       const vertex_descriptor neighbor = graph.opposite(edge, vertex);
441       const usize ns = static_cast<usize>(neighbor.value);
442       if ( discovery.data()[ns] == 0 ) {
443         stack.push_back(edge);
444         self(self, neighbor, edge);
445         if ( low.data()[ns] < low.data()[slot] ) low.data()[slot] = low.data()[ns];
446         if ( low.data()[ns] >= discovery.data()[slot] ) {
447           micron::vector<edge_descriptor, micron::allocator_serial<>, false> block;
448           for ( ;; ) {
449             const edge_descriptor member = stack.data()[stack.size() - 1];
450             stack.pop_back();
451             block.push_back(member);
452             if ( member == edge ) break;
453           }
454           result.blocks.push_back(micron::move(block));
455         }
456       } else if ( discovery.data()[ns] < discovery.data()[slot] ) {
457         stack.push_back(edge);
458         if ( discovery.data()[ns] < low.data()[slot] ) low.data()[slot] = discovery.data()[ns];
459       }
460     }
461   };
462   for ( auto root : graph.vertices() )
463     if ( discovery.data()[static_cast<usize>(root.value)] == 0 ) visit(visit, root, edge_descriptor::invalid());
464   result.articulation = articulation_points(graph);
465   return result;
466 }
