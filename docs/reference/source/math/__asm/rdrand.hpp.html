18 [[nodiscard, gnu::always_inline]] inline bool
19 rdrand64(u64 &out) noexcept
20 {
21 #if defined(__micron_x86_rdrnd) && defined(__micron_arch_amd64)
22   unsigned char ok = 0;
23   for ( int retry = 0; retry < 10; ++retry ) {
24     asm volatile("rdrand %0; setc %1" : "=r"(out), "=qm"(ok)::"cc");
25     if ( ok ) return true;
26   }
27   return false;
28 #elif defined(__micron_arch_arm64) && defined(__micron_arm_rng)
29   u64 v = 0;
30   for ( int retry = 0; retry < 10; ++retry ) {
31     u32 nzcv;
32     asm volatile("mrs %0, s3_3_c2_c4_0\n\tmrs %1, nzcv" : "=r"(v), "=r"(nzcv));
33     if ( ((nzcv >> 30) & 1u) == 0u ) {
34       out = v;
35       return true;
36     }
37   }
38   return false;
39 #else
40   (void)out;
41   return false;
42 #endif
43 }

45 [[nodiscard, gnu::always_inline]] inline bool
46 rdseed64(u64 &out) noexcept
47 {
48 #if defined(__micron_x86_rdseed) && defined(__micron_arch_amd64)
49   unsigned char ok = 0;
50   for ( int retry = 0; retry < 64; ++retry ) {
51     asm volatile("rdseed %0; setc %1" : "=r"(out), "=qm"(ok)::"cc");
52     if ( ok ) return true;
53   }
54   return false;
55 #elif defined(__micron_arch_arm64) && defined(__micron_arm_rng)
56   u64 v = 0;
57   for ( int retry = 0; retry < 64; ++retry ) {
58     u32 nzcv;
59     asm volatile("mrs %0, s3_3_c2_c4_1\n\tmrs %1, nzcv" : "=r"(v), "=r"(nzcv));
60     if ( ((nzcv >> 30) & 1u) == 0u ) {
61       out = v;
62       return true;
63     }
64   }
65   return false;
66 #else
67   (void)out;
68   return false;
69 #endif
70 }

81 [[nodiscard, gnu::always_inline]] inline u64
82 rdtsc64() noexcept
83 {
84 #if defined(__micron_arch_x86_any)
85   u32 lo = 0, hi = 0;
86   asm volatile("rdtsc" : "=a"(lo), "=d"(hi));
87   return (u64(hi) << 32) | u64(lo);
88 #elif defined(__micron_arch_arm64)
89   // NOTE: CNTVCT_EL0 is a fixed-frequency virtual counter (rate = CNTFRQ_EL0), NOT a per-cycle count
90   u64 v;
91   asm volatile("mrs %0, cntvct_el0" : "=r"(v));
92   return v;
93 #elif defined(__micron_arch_arm32) && defined(MICRON_CHRONO_ARM32_CNTVCT)
94   u32 lo, hi;
95   asm volatile("mrrc p15, 1, %0, %1, c14" : "=r"(lo), "=r"(hi));
96   return (u64(hi) << 32) | u64(lo);
97 #else
98   return 0;
99 #endif
100 }
