126 inline float
127 fsqrt_simd(const float x)
128 {
129 #if defined(__micron_arch_x86_any)
130   return simd::sse::extract_low_f32(simd::sse::sqrt_scalar_f32(simd::sse::set_scalar_f32(x)));
131 #elif defined(__micron_arch_arm64) && defined(__micron_arm_neon)
132   return simd::neon::get_lane_f32<0>(simd::neon::sqrt(simd::neon::splat_f32(x)));
133 #elif defined(__micron_arch_arm32) && defined(__micron_arm_neon)
134   return float(hw::sqrt_ss(f32(x)));
135 #else
136   return float(hw::sqrt_ss(f32(x)));
137 #endif
138 }

154 // exact 1/sqrt(x) on every arch
155 inline float
156 ss_rsqrt(const float x)
157 {
158 #if defined(__micron_arch_x86_any)
159   return 1.0f / simd::sse::extract_low_f32(simd::sse::sqrt_scalar_f32(simd::sse::set_scalar_f32(x)));
160 #elif defined(__micron_arch_arm64) && defined(__micron_arm_neon)
161   return 1.0f / simd::neon::get_lane_f32<0>(simd::neon::sqrt(simd::neon::splat_f32(x)));
162 #else
163   return 1.0f / float(hw::sqrt_ss(f32(x)));
164 #endif
165 }

214 __attribute__((always_inline)) inline simd::f128
215 vsqrt(simd::f128 v) noexcept
216 {
217 #if defined(__micron_arch_arm64)
218   return simd::neon::sqrt(v);
219 #else
220   float tmp[4];
221   simd::neon::store_f32(tmp, v);
222   tmp[0] = float(hw::sqrt_ss(f32(tmp[0])));
223   tmp[1] = float(hw::sqrt_ss(f32(tmp[1])));
224   tmp[2] = float(hw::sqrt_ss(f32(tmp[2])));
225   tmp[3] = float(hw::sqrt_ss(f32(tmp[3])));
226   return simd::neon::load_f32(tmp);
227 #endif
228 }
