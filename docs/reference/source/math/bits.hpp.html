23 template<typename T>
24   requires(micron::is_integral_v<T>)
25 [[nodiscard, gnu::always_inline]] inline constexpr T
26 rol(T x, int r) noexcept
27 {
28   using U = micron::make_unsigned_t<T>;
29   constexpr int w = sizeof(T) * 8;
30   const int s = r & (w - 1);
31   if ( s == 0 ) return x;
32   const U u = static_cast<U>(x);
33   return static_cast<T>(static_cast<U>(U(u << s) | U(u >> (w - s))));
34 }

132 // WARNING: raw bits; -Ofast implies -fno-signed-zeros
133 template<typename F>
134   requires(micron::is_floating_point_v<F>)
135 [[nodiscard, gnu::always_inline]] inline constexpr int
136 sign_bit(F x) noexcept
137 {
138   if constexpr ( sizeof(F) == 4 )
139     return (__builtin_bit_cast(u32, x) >> 31) != 0;
140   else if constexpr ( sizeof(F) == 8 )
141     return (__builtin_bit_cast(u64, x) >> 63) != 0;
142   else
143     return __builtin_signbit(x);
144 }

149 template<typename To, typename From>
150 [[nodiscard, gnu::always_inline]] inline constexpr To
151 bit_cast(const From &from) noexcept
152 {
153   static_assert(sizeof(To) == sizeof(From), "math::bits::bit_cast size mismatch");
154   static_assert(micron::is_trivially_copyable_v<To>, "bit_cast destination must be trivially copyable");
155   static_assert(micron::is_trivially_copyable_v<From>, "bit_cast source must be trivially copyable");
156   return __builtin_bit_cast(To, from);
157 }

180 template<typename T>
181   requires(micron::is_integral_v<T>)
182 [[nodiscard, gnu::always_inline]] inline constexpr T
183 byteswap(T x) noexcept
184 {
185   if constexpr ( sizeof(T) == 1 )
186     return x;
187   else if constexpr ( sizeof(T) == 2 )
188     return static_cast<T>(__builtin_bswap16(static_cast<u16>(x)));
189   else if constexpr ( sizeof(T) == 4 )
190     return static_cast<T>(__builtin_bswap32(static_cast<u32>(x)));
191   else if constexpr ( sizeof(T) == 8 )
192     return static_cast<T>(__builtin_bswap64(static_cast<u64>(x)));
193 }

197 template<typename T>
198   requires(micron::is_integral_v<T> && micron::is_unsigned_v<T>)
199 [[nodiscard, gnu::always_inline]] inline constexpr T
200 floor_pow2(T x) noexcept
201 {
202   if ( x == 0 ) return 0;
203   return T(1) << (sizeof(T) * 8 - 1 - clz(x));
204 }

206 template<typename T>
207   requires(micron::is_integral_v<T> && micron::is_unsigned_v<T>)
208 [[nodiscard, gnu::always_inline]] inline constexpr T
209 ceil_pow2(T x) noexcept
210 {
211   if ( x <= 1 ) return 1;
212   // saturate through an all-ones mask
213   return static_cast<T>(static_cast<T>(static_cast<T>(~T(0)) >> clz(static_cast<T>(x - 1))) + T(1));
214 }
