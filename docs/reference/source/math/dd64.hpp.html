17 struct dd64 {
18   f64 hi;
19   f64 lo;
20 
21   constexpr dd64() noexcept : hi(0.0), lo(0.0) { }
22 
23   constexpr dd64(f64 h, f64 l) noexcept : hi(h), lo(l) { }
24 
25   constexpr dd64(f64 h) noexcept : hi(h), lo(0.0) { }
26 
27   [[nodiscard, gnu::always_inline]] constexpr f64
28   to_double() const noexcept
29   {
30     return hi + lo;
31   }
32 
33   explicit constexpr
34   operator f64() const noexcept
35   {
36     return hi + lo;
37   }
38 };

50 [[nodiscard, gnu::always_inline]] inline constexpr f64
51 __keep(f64 x) noexcept
52 {
53 #if defined(__has_builtin) && __has_builtin(__builtin_assoc_barrier)
54   return __builtin_assoc_barrier(x);
55 #else
56   return x;
57 #endif
58 }

60 // dekker's split
61 [[nodiscard, gnu::always_inline]] inline constexpr dd64
62 __split(f64 a) noexcept
63 {
64   constexpr f64 c = 0x1.0p27 + 1.0;
65   const f64 t = c * a;
66   const f64 h = t - (t - a);
67   return { h, a - h };
68 }
69 
70 // Knuth's two sum (exact when a + b == s + e)
71 [[nodiscard, gnu::always_inline]] inline constexpr dd64
72 two_sum(f64 a, f64 b) noexcept
73 {
74   f64 s = __keep(a + b);
75   f64 bp = __keep(s - a);
76   f64 e = __keep(__keep(a - __keep(s - bp)) + __keep(b - bp));
77   return { s, e };
78 }
79 
80 // dekker's fast two sum (exact when abs(a) >= abs(b))
81 [[nodiscard, gnu::always_inline]] inline constexpr dd64
82 fast_two_sum(f64 a, f64 b) noexcept
83 {
84   f64 s = __keep(a + b);
85   f64 e = __keep(b - __keep(s - a));
86   return { s, e };
87 }
88 
89 // two product via FMA (exact when a * b == p + e)
90 [[nodiscard, gnu::always_inline]] inline constexpr dd64
91 two_prod(f64 a, f64 b) noexcept
92 {
93   f64 p = __keep(a * b);
94   // clang doesn't support __builtin_fma in a constexpr somehow?!?!?
95   if consteval {
96 #if defined(__micron_compiler_gcc)
97     f64 e = __builtin_fma(a, b, -p);
98 #else
99     f64 sa = a, sb = b, k = 1.0;
100     if ( sa > 0x1.0p996 || sa < -0x1.0p996 ) {
101       sa *= 0x1.0p-64;
102       k *= 0x1.0p64;
103     }
104     if ( sb > 0x1.0p996 || sb < -0x1.0p996 ) {
105       sb *= 0x1.0p-64;
106       k *= 0x1.0p64;
107     }
108     const dd64 ah = __split(sa);
109     const dd64 bh = __split(sb);
110     const f64 sp = p / k;
111     f64 e = (((ah.hi * bh.hi - sp) + ah.hi * bh.lo + ah.lo * bh.hi) + ah.lo * bh.lo) * k;
112 #endif
113     return { p, e };
114   }
115   f64 e = __keep(__builtin_fma(a, b, -p));
116   return { p, e };
117 }

119 [[nodiscard, gnu::always_inline]] inline constexpr dd64
120 add(dd64 a, dd64 b) noexcept
121 {
122   dd64 s = two_sum(a.hi, b.hi);
123   dd64 t = two_sum(a.lo, b.lo);
124   f64 c = s.lo + t.hi;
125   dd64 r1 = two_sum(s.hi, c);
126   f64 d = r1.lo + t.lo;
127   return two_sum(r1.hi, d);
128 }
129 
130 [[nodiscard, gnu::always_inline]] inline constexpr dd64
131 add(dd64 a, f64 b) noexcept
132 {
133   dd64 s = two_sum(a.hi, b);
134   return two_sum(s.hi, s.lo + a.lo);
135 }
136 
137 [[nodiscard, gnu::always_inline]] inline constexpr dd64
138 sub(dd64 a, dd64 b) noexcept
139 {
140   return add(a, dd64{ -b.hi, -b.lo });
141 }
142 
143 [[nodiscard, gnu::always_inline]] inline constexpr dd64
144 mul(dd64 a, dd64 b) noexcept
145 {
146   dd64 p = two_prod(a.hi, b.hi);
147   f64 t = a.hi * b.lo + a.lo * b.hi;
148   return fast_two_sum(p.hi, p.lo + t);
149 }
150 
151 [[nodiscard, gnu::always_inline]] inline constexpr dd64
152 mul(dd64 a, f64 b) noexcept
153 {
154   dd64 p = two_prod(a.hi, b);
155   return fast_two_sum(p.hi, p.lo + a.lo * b);
156 }
157 
158 [[nodiscard, gnu::always_inline]] inline constexpr dd64
159 div(dd64 a, dd64 b) noexcept
160 {
161   f64 q1 = a.hi / b.hi;
162   dd64 r = sub(a, mul(b, dd64{ q1, 0.0 }));
163   f64 q2 = r.hi / b.hi;
164   return fast_two_sum(q1, q2);
165 }
