30 // SplitMix64
31 struct splitmix64 {
32   using result_type = u64;
33   static constexpr u32 result_bits = 64;
34   static constexpr u64 increment = 0x9E3779B97F4A7C15ULL;
35 
36   u64 s;
37 
38   constexpr splitmix64() noexcept : s(0) { }
39 
40   constexpr explicit splitmix64(u64 seed) noexcept : s(seed) { }
41 
42   [[nodiscard]] static constexpr splitmix64
43   from_seed(u64 seed) noexcept
44   {
45     return splitmix64(seed);
46   }
47 
48   [[nodiscard, gnu::always_inline]] static constexpr u64
49   __mix(u64 z) noexcept
50   {
51     z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
52     z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
53     return z ^ (z >> 31);
54   }
55 
56   [[nodiscard, gnu::always_inline]] constexpr u64
57   next() noexcept
58   {
59     s += increment;
60     return __mix(s);
61   }
62 
63   [[nodiscard, gnu::always_inline]] constexpr u64
64   next64() noexcept
65   {
66     return next();
67   }
68 
69   inline constexpr void
70   generate(u64 *__restrict__ out, usize n) noexcept
71   {
72     u64 state = s;
73     constexpr u64 i2 = increment * 2ULL;
74     constexpr u64 i3 = increment * 3ULL;
75     constexpr u64 i4 = increment * 4ULL;
76     constexpr u64 i5 = increment * 5ULL;
77     constexpr u64 i6 = increment * 6ULL;
78     constexpr u64 i7 = increment * 7ULL;
79     constexpr u64 i8 = increment * 8ULL;
80     while ( n >= 8 ) {
81       out[0] = __mix(state + increment);
82       out[1] = __mix(state + i2);
83       out[2] = __mix(state + i3);
84       out[3] = __mix(state + i4);
85       out[4] = __mix(state + i5);
86       out[5] = __mix(state + i6);
87       out[6] = __mix(state + i7);
88       out[7] = __mix(state + i8);
89       state += i8;
90       out += 8;
91       n -= 8;
92     }
93     while ( n-- != 0 ) {
94       state += increment;
95       *out++ = __mix(state);
96     }
97     s = state;
98   }
99 };

112

125   [[nodiscard, gnu::always_inline]] static constexpr u64
126   __step(u64 &s0, u64 &s1, u64 &s2, u64 &s3) noexcept
127   {
128     const u64 result = bits::rol64(s1 * 5ULL, 7) * 9ULL;
129     const u64 t = s1 << 17;
130     s2 ^= s0;
131     s3 ^= s1;
132     s1 ^= s2;
133     s0 ^= s3;
134     s2 ^= t;
135     s3 = bits::rol64(s3, 45);
136     return result;
137   }

151   inline constexpr void
152   generate(u64 *__restrict__ out, usize n) noexcept
153   {
154     u64 s0 = s[0], s1 = s[1], s2 = s[2], s3 = s[3];
155     while ( n >= 4 ) {
156       out[0] = __step(s0, s1, s2, s3);
157       out[1] = __step(s0, s1, s2, s3);
158       out[2] = __step(s0, s1, s2, s3);
159       out[3] = __step(s0, s1, s2, s3);
160       out += 4;
161       n -= 4;
162     }
163     while ( n-- != 0 ) *out++ = __step(s0, s1, s2, s3);
164     s[0] = s0;
165     s[1] = s1;
166     s[2] = s2;
167     s[3] = s3;
168   }

171   jump() noexcept
172   {
173     static constexpr u64 J[] = { 0x180EC6D33CFD0ABAULL, 0xD5A61266F0C9392CULL, 0xA9582618E03FC9AAULL, 0x39ABDC4529B1661CULL };
174     u64 s0 = 0, s1 = 0, s2 = 0, s3 = 0;
175     for ( int i = 0; i < 4; ++i ) {
176       for ( int b = 0; b < 64; ++b ) {
177         const u64 mask = u64(0) - u64((J[i] >> b) & 1ULL);
178         s0 ^= s[0] & mask;
179         s1 ^= s[1] & mask;
180         s2 ^= s[2] & mask;
181         s3 ^= s[3] & mask;

183       }
184     }

186     s[1] = s1;
187     s[2] = s2;
188     s[3] = s3;
189   }

336 // PCG XSL-RR 128/64
337 struct pcg64 {
338   using result_type = u64;
339   static constexpr u32 result_bits = 64;
340 
341   u128 state;
342   u128 inc;
343 
344   [[nodiscard, gnu::always_inline]] static constexpr u128
345   __multiplier() noexcept
346   {
347     return (u128(0x2360ED051FC65DA4ULL) << 64) | u128(0x4385DF649FCCF645ULL);
348   }
349 
350   [[nodiscard, gnu::always_inline]] static constexpr u64
351   __output(u128 old) noexcept
352   {
353     const u64 xsl = u64(old >> 64) ^ u64(old);
354     const u32 rot = u32(u64(old >> 122));
355     return bits::ror64(xsl, int(rot));
356   }
357 
358   constexpr pcg64() noexcept : state(0), inc(3)
359   {
360     state = state * __multiplier() + inc;
361     state = state * __multiplier() + inc;
362   }
363 
364   constexpr pcg64(u128 s, u128 i) noexcept : state(s), inc(i | u128(1)) { }
365 
366   [[nodiscard]] static constexpr pcg64
367   make(u64 seed, u64 stream = 1) noexcept
368   {
369     pcg64 r{ u128(0), (u128(stream) << 1) | u128(1) };
370     r.state = r.state * __multiplier() + r.inc;
371     r.state += u128(seed);
372     r.state = r.state * __multiplier() + r.inc;
373     return r;
374   }
375 
376   [[nodiscard, gnu::always_inline]] constexpr u64
377   next() noexcept
378   {
379     const u128 old = state;
380     state = old * __multiplier() + inc;
381     return __output(old);
382   }
383 
384   [[nodiscard, gnu::always_inline]] constexpr u64
385   next64() noexcept
386   {
387     return next();
388   }
389 
390   inline constexpr void
391   generate(u64 *__restrict__ out, usize n) noexcept
392   {
393     u128 st = state;
394     const u128 increment_value = inc;
395     const u128 a1 = __multiplier();
396     const u128 a2 = a1 * a1;
397     const u128 a3 = a2 * a1;
398     const u128 a4 = a2 * a2;
399     const u128 c2 = increment_value * (a1 + u128(1));
400     const u128 c3 = increment_value * (a2 + a1 + u128(1));
401     const u128 c4 = increment_value * (a3 + a2 + a1 + u128(1));
402     while ( n >= 4 ) {
403       const u128 s1 = st * a1 + increment_value;
404       const u128 s2 = st * a2 + c2;
405       const u128 s3 = st * a3 + c3;
406       out[0] = __output(st);
407       out[1] = __output(s1);
408       out[2] = __output(s2);
409       out[3] = __output(s3);
410       st = st * a4 + c4;
411       out += 4;
412       n -= 4;
413     }
414     while ( n-- != 0 ) {
415       const u128 old = st;
416       st = old * a1 + increment_value;
417       *out++ = __output(old);
418     }
419     state = st;
420   }
421 };

426 // MT19937
427 struct mt19937 {
428   using result_type = u32;
429   static constexpr u32 result_bits = 32;
430   static constexpr usize N = 624;
431   static constexpr usize M = 397;
432   static constexpr u32 MATRIX_A = 0x9908B0DFu;
433   static constexpr u32 UPPER_MASK = 0x80000000u;
434   static constexpr u32 LOWER_MASK = 0x7FFFFFFFu;
435 
436   u32 mt[N];
437   u32 idx;
438 
439   constexpr mt19937() noexcept : mt{}, idx(0) { seed_with(5489u); }
440 
441   constexpr explicit mt19937(u32 seed) noexcept : mt{}, idx(0) { seed_with(seed); }
442 
443   [[nodiscard]] static constexpr mt19937
444   from_seed(u64 seed) noexcept
445   {
446     return mt19937(u32(seed));
447   }
448 
449   constexpr void
450   seed_with(u32 seed) noexcept
451   {
452     mt[0] = seed;
453     for ( usize i = 1; i < N; ++i ) mt[i] = u32(1812433253u * (mt[i - 1] ^ (mt[i - 1] >> 30)) + u32(i));
454     idx = N;
455   }
456 
457   [[nodiscard, gnu::always_inline]] static constexpr u32
458   __twist_word(u32 upper, u32 lower, u32 mix) noexcept
459   {
460     const u32 y = (upper & UPPER_MASK) | (lower & LOWER_MASK);
461     return mix ^ (y >> 1) ^ ((u32(0) - (y & 1u)) & MATRIX_A);
462   }
463 
464   constexpr void
465   twist() noexcept
466   {
467     for ( usize i = 0; i < N - M; ++i ) mt[i] = __twist_word(mt[i], mt[i + 1], mt[i + M]);
468     for ( usize i = N - M; i < N - 1; ++i ) mt[i] = __twist_word(mt[i], mt[i + 1], mt[i + (M - N)]);
469     mt[N - 1] = __twist_word(mt[N - 1], mt[0], mt[M - 1]);
470     idx = 0;
471   }
472 
473   constexpr void
474   generate() noexcept
475   {
476     twist();
477   }
478 
479   [[nodiscard, gnu::always_inline]] static constexpr u32
480   __temper(u32 y) noexcept
481   {
482     y ^= y >> 11;
483     y ^= (y << 7) & 0x9D2C5680u;
484     y ^= (y << 15) & 0xEFC60000u;
485     return y ^ (y >> 18);
486   }
487 
488   [[nodiscard, gnu::always_inline]] constexpr u32
489   next() noexcept
490   {
491     if ( idx >= N ) [[unlikely]]
492       twist();
493     return __temper(mt[idx++]);
494   }
495 
496   [[nodiscard, gnu::always_inline]] constexpr u32
497   next32() noexcept
498   {
499     return next();
500   }
501 
502   [[nodiscard, gnu::always_inline]] constexpr u64
503   next64() noexcept
504   {
505     const u64 lo = next();
506     const u64 hi = next();
507     return lo | (hi << 32);
508   }
509 
510   inline constexpr void
511   generate(u32 *__restrict__ out, usize n) noexcept
512   {
513     while ( n != 0 ) {
514       if ( idx >= N ) twist();
515       const usize available = N - idx;
516       const usize count = n < available ? n : available;
517       const u32 *input = mt + idx;
518       for ( usize i = 0; i < count; ++i ) out[i] = __temper(input[i]);
519       idx += u32(count);
520       out += count;
521       n -= count;
522     }
523   }
524 };

528 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
529 // multiply-with-carry
530 struct mwc64 {
531   using result_type = u64;
532   static constexpr u32 result_bits = 64;
533   static constexpr u64 A_HI = 4294957665ULL;
534   static constexpr u64 A_LO = 4294963023ULL;
535 
536   u64 hi;
537   u64 lo;
538 
539   constexpr mwc64() noexcept : hi(0xCAFEBABEDEADBEEFULL), lo(0xFEEDFACEC0FFEE00ULL) { }
540 
541   constexpr mwc64(u64 a, u64 b) noexcept : hi(a), lo(b) { }
542 
543   [[nodiscard, gnu::always_inline]] static constexpr u64
544   __normalize(u64 state, u64 multiplier) noexcept
545   {
546     u32 value = u32(state);
547     const u32 carry = u32(state >> 32) % u32(multiplier);
548     if ( value == 0 && carry == 0 ) value = 1;
549     return (u64(carry) << 32) | value;
550   }
551 
552   [[nodiscard]] static constexpr mwc64
553   from_seed(u64 seed) noexcept
554   {
555     splitmix64 sm{ seed };
556     return mwc64{ __normalize(sm.next(), A_HI), __normalize(sm.next(), A_LO) };
557   }
558 
559   [[nodiscard, gnu::always_inline]] static constexpr u64
560   __step(u64 &high, u64 &low) noexcept
561   {
562     high = A_HI * u64(u32(high)) + (high >> 32);
563     low = A_LO * u64(u32(low)) + (low >> 32);
564     return (u64(u32(high)) << 32) | u64(u32(low));
565   }
566 
567   [[nodiscard, gnu::always_inline]] constexpr u64
568   next() noexcept
569   {
570     return __step(hi, lo);
571   }
572 
573   [[nodiscard, gnu::always_inline]] constexpr u64
574   next64() noexcept
575   {
576     return next();
577   }
578 
579   inline constexpr void
580   generate(u64 *__restrict__ out, usize n) noexcept
581   {
582     u64 high = hi, low = lo;
583     while ( n >= 4 ) {
584       out[0] = __step(high, low);
585       out[1] = __step(high, low);
586       out[2] = __step(high, low);
587       out[3] = __step(high, low);
588       out += 4;
589       n -= 4;
590     }
591     while ( n-- != 0 ) *out++ = __step(high, low);
592     hi = high;
593     lo = low;
594   }
595 };

599 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
600 // full-period 64-bit LCG
601 struct lcg64 {
602   using result_type = u64;
603   static constexpr u32 result_bits = 64;
604   static constexpr u64 A = 6364136223846793005ULL;
605   static constexpr u64 C = 1442695040888963407ULL;
606   static constexpr u64 A2 = A * A;
607   static constexpr u64 C2 = A * C + C;
608   static constexpr u64 A3 = A2 * A;
609   static constexpr u64 C3 = A * C2 + C;
610   static constexpr u64 A4 = A2 * A2;
611   static constexpr u64 C4 = A * C3 + C;
612 
613   u64 s;
614 
615   constexpr lcg64() noexcept : s(0) { }
616 
617   constexpr explicit lcg64(u64 seed) noexcept : s(seed) { }
618 
619   [[nodiscard]] static constexpr lcg64
620   from_seed(u64 seed) noexcept
621   {
622     return lcg64{ seed };
623   }
624 
625   [[nodiscard, gnu::always_inline]] constexpr u64
626   next() noexcept
627   {
628     s = A * s + C;
629     return s;
630   }
631 
632   [[nodiscard, gnu::always_inline]] constexpr u64
633   next64() noexcept
634   {
635     return next();
636   }
637 
638   [[nodiscard, gnu::always_inline]] constexpr u32
639   next32() noexcept
640   {
641     return u32(next() >> 32);
642   }
643 
644   inline constexpr void
645   generate(u64 *__restrict__ out, usize n) noexcept
646   {
647     u64 state = s;
648     while ( n >= 4 ) {
649       out[0] = A * state + C;
650       out[1] = A2 * state + C2;
651       out[2] = A3 * state + C3;
652       out[3] = A4 * state + C4;
653       state = out[3];
654       out += 4;
655       n -= 4;
656     }
657     while ( n-- != 0 ) {
658       state = A * state + C;
659       *out++ = state;
660     }
661     s = state;
662   }
663 };
664 
665 static_assert(sizeof(lcg64) == 8, "lcg64 state must be 8 bytes");

668 // generic width-aware draws
669 template<typename T>
670 concept rng_concept = requires(T t) {
671   { t.next() } -> micron::convertible_to<u64>;
672 };
673 
674 namespace __impl
675 {
676 
677 template<rng_concept Rng>
678 [[nodiscard, gnu::always_inline]] inline constexpr u32
679 next32(Rng &g) noexcept
680 {
681   if constexpr ( requires { g.next32(); } )
682     return u32(g.next32());
683   else
684     return u32(g.next());
685 }
686 
687 template<rng_concept Rng>
688 [[nodiscard, gnu::always_inline]] inline constexpr u64
689 next64(Rng &g) noexcept
690 {
691   if constexpr ( requires { g.next64(); } ) {
692     return u64(g.next64());
693   } else if constexpr ( sizeof(decltype(g.next())) <= sizeof(u32) ) {
694     const u64 lo = u32(g.next());
695     const u64 hi = u32(g.next());
696     return lo | (hi << 32);
697   } else {
698     return u64(g.next());
699   }
700 }
