8 // 256-layer ziggurat tables generated at compile time
9 //   .. region 0 is the base strip + tail (boundary x = R, density floor f[0]=1);
10 //   .. regions 1..254 are the equal-area rectangles, x increasing toward 0;
11 //   .. region 255 is the smallest box at the peak;
12 //   .. f[i] = exp(-x[i]^2/2) is DECREASING with i (f[0]=1, f[255]=f(R));
13 //   .. w[i] = x[i]/2^53 (region 0 uses the area-based width q = V/f(R));
14 //   .. k[i] = floor( (x[i]/x[i-1]) * 2^53 ) is the fast-accept threshold

38 struct ziggurat_tables {
39   f64 w[256];      // x[i] / 2^53  (region 0: area-based width q/2^53)
40   f64 f[256];      // exp(-x[i]^2 / 2)
41   u64 k[256];      // floor((x[i]/x[i-1]) * 2^53)  fast-accept thresholds
42 
43   static constexpr f64 R = 3.6541528853610088;
44   static constexpr f64 V = 0.00492867323399;
45 };

47 // f(x) = exp(-x^2/2)
48 [[nodiscard]] consteval f64
49 __zig_f(f64 x) noexcept
50 {
51   return math::mkbits::exp_ns::exp_f64(-0.5 * x * x);
52 }

54 [[nodiscard]] consteval ziggurat_tables
55 build_ziggurat() noexcept
56 {
57   ziggurat_tables t{};
58   constexpr f64 M = 9007199254740992.0;      // 2^53
59 
60   const f64 R = ziggurat_tables::R;
61   const f64 V = ziggurat_tables::V;
62   const f64 fR = __zig_f(R);
63   const f64 q = V / fR;      // width of the base rectangle incl. tail mass
64 
65   // base strip + tail
66   t.k[0] = u64((R / q) * M);
67   t.w[0] = q / M;
68   t.f[0] = 1.0;
69 
70   // the topmost (smallest, peak) box uses x = R as its scale seed
71   t.w[255] = R / M;
72   t.f[255] = fR;
73   t.k[1] = 0;
74 
75   // walking the stack downward in index
76   f64 dn = R;      // x[255]
77   f64 tn = R;      // previous x (for the k ratio)
78   bool closed = true;
79   for ( int i = 254; i >= 1; --i ) {
80     const f64 arg = V / dn + __zig_f(dn);
81     if ( !(arg > 0.0 && arg < 1.0) ) closed = false;
82     dn = math::fsqrt(-2.0 * math::mkbits::log_ns::log_f64(arg));
83     t.k[i + 1] = u64((dn / tn) * M);
84     tn = dn;
85     t.f[i] = __zig_f(dn);
86     t.w[i] = dn / M;
87   }
88 
89   const f64 smallest = dn;      // x[1]
90   const bool sane = closed && smallest > 0.20 && smallest < 0.23 && t.f[1] > 0.97 && t.f[1] < 0.98;
91   if ( !sane ) {
92     // static_assert(false, "division by zero in build_ziggurat");
93     // int *boom = nullptr;
94     //(void)*boom;      // actually hard compile err if evaluated, above doesn't work
95     // neither did that
96     __builtin_unreachable();      // THIS SHOULD
97   }
98   return t;
99 }

105 template<ieee754_floating F = f64, rng_concept Rng>
106 [[nodiscard]] inline F
107 normal_ziggurat(Rng &g, F mu = F(0), F sigma = F(1)) noexcept
108 {
109   const f64 *__restrict__ wt = mkbits::ziggurat.w;
110   const f64 *__restrict__ ft = mkbits::ziggurat.f;
111   const u64 *__restrict__ kt = mkbits::ziggurat.k;
112   const f64 R = mkbits::ziggurat_tables::R;
113 
114   u64 u = rng::__impl::next64(g);
115   i64 hz = i64(u >> 11);
116   const i64 sign = -i64(u & 1ULL);
117   hz = (hz ^ sign) - sign;
118   u64 iz = (u >> 1) & 0xFFULL;
119   u64 ahz = u64(hz < 0 ? -hz : hz);
120 
121   if ( ahz < kt[iz] ) [[likely]]
122     return F(mu + sigma * F(f64(hz) * wt[iz]));
123 
124   for ( ;; ) {
125     f64 x = f64(hz) * wt[iz];
126     if ( iz == 0 ) {
127       f64 xt;
128       f64 yt;
129       do {
130         const u64 r1 = ((rng::__impl::next64(g) >> 12) << 1) | 1ULL;
131         const u64 r2 = ((rng::__impl::next64(g) >> 12) << 1) | 1ULL;
132         const f64 u1 = r1 * 0x1.0p-53;
133         const f64 u2 = r2 * 0x1.0p-53;
134         xt = -math::mkbits::log_ns::log_f64(u1) / R;
135         yt = -math::mkbits::log_ns::log_f64(u2);
136       } while ( yt + yt < xt * xt );
137       f64 v = (hz > 0) ? (R + xt) : -(R + xt);
138       return F(mu + sigma * F(v));
139     }
140     const f64 u1 = (rng::__impl::next64(g) >> 11) * 0x1.0p-53;
141     f64 fy = ft[iz] + u1 * (ft[iz - 1] - ft[iz]);
142     f64 ax = x < 0 ? -x : x;
143     f64 phi_x = math::mkbits::exp_ns::exp_f64(-ax * ax * 0.5);
144     if ( fy < phi_x ) return F(mu + sigma * F(x));
145 
146     u = rng::__impl::next64(g);
147     hz = i64(u >> 11);
148     const i64 retry_sign = -i64(u & 1ULL);
149     hz = (hz ^ retry_sign) - retry_sign;
150     iz = (u >> 1) & 0xFFULL;
151     ahz = u64(hz < 0 ? -hz : hz);
152     if ( ahz < kt[iz] ) return F(mu + sigma * F(f64(hz) * wt[iz]));
153   }
154 }
