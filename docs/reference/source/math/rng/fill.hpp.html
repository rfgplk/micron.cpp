25 template<ieee754_floating F, rng_concept Rng>
26 [[gnu::flatten]] inline void
27 fill_uniform(F *__restrict__ out, usize N, Rng &g) noexcept
28 {
29   while ( N >= 4 ) {
30     out[0] = dist::uniform_real<F>(g);
31     out[1] = dist::uniform_real<F>(g);
32     out[2] = dist::uniform_real<F>(g);
33     out[3] = dist::uniform_real<F>(g);
34     out += 4;
35     N -= 4;
36   }
37   while ( N-- != 0 ) *out++ = dist::uniform_real<F>(g);
38 }

40 template<ieee754_floating F, rng_concept Rng>
41 [[gnu::flatten]] inline void
42 fill_uniform(F *__restrict__ out, usize N, Rng &g, F lo, F hi) noexcept
43 {
44   const F r = hi - lo;
45   while ( N >= 4 ) {
46     out[0] = lo + r * dist::uniform_real<F>(g);
47     out[1] = lo + r * dist::uniform_real<F>(g);
48     out[2] = lo + r * dist::uniform_real<F>(g);
49     out[3] = lo + r * dist::uniform_real<F>(g);
50     out += 4;
51     N -= 4;
52   }
53   while ( N-- != 0 ) *out++ = lo + r * dist::uniform_real<F>(g);
54 }

56 template<ieee754_floating F, rng_concept Rng>
57 [[gnu::flatten]] inline void
58 fill_normal(F *__restrict__ out, usize N, Rng &g, F mu = F(0), F sigma = F(1)) noexcept
59 {
60   while ( N >= 4 ) {
61     out[0] = dist::normal_ziggurat<F>(g, mu, sigma);
62     out[1] = dist::normal_ziggurat<F>(g, mu, sigma);
63     out[2] = dist::normal_ziggurat<F>(g, mu, sigma);
64     out[3] = dist::normal_ziggurat<F>(g, mu, sigma);
65     out += 4;
66     N -= 4;
67   }
68   while ( N-- != 0 ) *out++ = dist::normal_ziggurat<F>(g, mu, sigma);
69 }

71 template<rng_concept Rng>
72 [[gnu::flatten]] inline void
73 fill_bytes(u8 *__restrict__ out, usize N, Rng &g) noexcept
74 {
75   while ( N >= 8 ) {
76     const u64 r = rng::__impl::next64(g);
77     __builtin_memcpy(out, &r, 8);
78     out += 8;
79     N -= 8;
80   }
81   if ( N != 0 ) {
82     const u64 r = rng::__impl::next64(g);
83     __builtin_memcpy(out, &r, N);
84   }
85 }

87 template<typename T, rng_concept Rng>
88   requires(micron::is_integral_v<T>)
89 [[gnu::flatten]] inline void
90 fill_uniform_int(T *__restrict__ out, usize N, Rng &g, T lo, T hi) noexcept
91 {
92   T *end = out + N;
93   while ( out != end ) *out++ = dist::uniform_int<T>(g, lo, hi);
