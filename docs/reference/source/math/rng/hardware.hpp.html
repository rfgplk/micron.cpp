38 [[nodiscard]] inline xoshiro256ss
39 seed_from_hw() noexcept
40 {
41   u64 a = 0, b = 0, c = 0, d = 0;
42   const bool got = rdrand64(a) && rdrand64(b) && rdrand64(c) && rdrand64(d) && ((a | b | c | d) != 0);
43   if ( !got ) {
44     micron::timespec_t __ts{};
45     (void)micron::clock_gettime(micron::clock_monotonic, __ts);
46     u64 mix = (static_cast<u64>(__ts.tv_sec) * 1'000'000'000ull) ^ static_cast<u64>(__ts.tv_nsec);
47     micron::timespec_t __rt{};
48     (void)micron::clock_gettime(micron::clock_realtime, __rt);
49     mix ^= (static_cast<u64>(__rt.tv_sec) << 20) ^ static_cast<u64>(__rt.tv_nsec);
50     mix ^= static_cast<u64>(reinterpret_cast<uintptr_t>(&__ts));
51     if constexpr ( __asm_op::rdtsc64_available ) mix ^= __asm_op::rdtsc64();
52     splitmix64 sm{ mix ^ 0xa5a5a5a5a5a5a5a5ULL };
53     a = sm.next();
54     b = sm.next();
55     c = sm.next();
56     d = sm.next();
57   }
58   return xoshiro256ss{ a, b, c, d };
59 }
