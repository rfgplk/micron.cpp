37   [[nodiscard, gnu::flatten]] inline constexpr f64
38   log_f64(f64 x) noexcept
39   {
40     using namespace coeff::log_f64_data;
41 
42     if ( ieee::is_nan(x) ) [[unlikely]]
43       return x;
44     if ( x < 0 ) [[unlikely]]
45       return ieee::qnan_v<f64>();
46     if ( x == 0 ) [[unlikely]]
47       return ieee::inf_v<f64>(1);
48     if ( ieee::is_inf(x) ) [[unlikely]]
49       return x;
50 
51     int k;
52     f64 m = manip::frexp<f64>(x, &k);
53     if ( m < 0x1.6a09e667f3bcdp-1 ) {
54       m = m + m;
55       k -= 1;
56     }
57 
58     f64 f = m - 1.0;
59     f64 s = f / (2.0 + f);
60     f64 z = s * s;
61     f64 w = z * z;
62 
63     // odd
64     f64 t1 = hw::fmadd_sd(Lg6, w, Lg4);
65     t1 = hw::fmadd_sd(t1, w, Lg2);
66     t1 = t1 * w;
67     // even
68     f64 t2 = hw::fmadd_sd(Lg7, w, Lg5);
69     t2 = hw::fmadd_sd(t2, w, Lg3);
70     t2 = hw::fmadd_sd(t2, w, Lg1);
71     t2 = t2 * z;
72     f64 R = t1 + t2;
73 
74     f64 hfsq = 0.5 * f * f;
75     f64 log_m = s * (hfsq + R) + (f - hfsq);
76 
77     // log(x) = k * ln2 + log_m
78     f64 fk = f64(k);
79     return hw::fmadd_sd(fk, ln2_hi, hw::fmadd_sd(fk, ln2_lo, log_m));
80   }

82   [[nodiscard, gnu::flatten]] inline constexpr f32
83   log_f32(f32 x) noexcept
84   {
85     using namespace coeff::log_f32_data;
86 
87     if ( ieee::is_nan(x) ) [[unlikely]]
88       return x;
89     if ( x < 0 ) [[unlikely]]
90       return ieee::qnan_v<f32>();
91     if ( x == 0 ) [[unlikely]]
92       return ieee::inf_v<f32>(1);
93     if ( ieee::is_inf(x) ) [[unlikely]]
94       return x;
95 
96     int k;
97     f32 m = manip::frexp<f32>(x, &k);
98     if ( m < 0x1.6a09e6p-1f ) {
99       m = m + m;
100       k -= 1;
101     }
102 
103     f32 f = m - 1.0f;
104     f32 s = f / (2.0f + f);
105     f32 z = s * s;
106     f32 w = z * z;
107     f32 t1 = hw::fmadd_ss(Lg4, w, Lg2);
108     t1 = t1 * w;
109     f32 t2 = hw::fmadd_ss(Lg3, w, Lg1);
110     t2 = t2 * z;
111     f32 R = t1 + t2;
112     f32 hfsq = 0.5f * f * f;
113     f32 log_m = s * (hfsq + R) + (f - hfsq);
114 
115     f32 fk = f32(k);
116     return hw::fmadd_ss(fk, ln2_hi, hw::fmadd_ss(fk, ln2_lo, log_m));
117   }

119   [[nodiscard, gnu::flatten]] inline constexpr f64
120   log1p_f64(f64 x) noexcept
121   {
122     if ( ieee::is_nan(x) ) [[unlikely]]
123       return x;
124     if ( x == -1.0 ) [[unlikely]]
125       return ieee::inf_v<f64>(1);
126     if ( x < -1.0 ) [[unlikely]]
127       return ieee::qnan_v<f64>();
128     if ( ieee::is_inf(x) ) [[unlikely]]
129       return x;
130     if ( manip::fabs(x) < 0x1.0p-54 ) [[unlikely]]
131       return x;
132     f64 u = 1.0 + x;
133     f64 c = (u - 1.0) - x;
134     return log_f64(u) - c / u;
135   }

279   [[nodiscard, gnu::flatten]] inline constexpr dd64
280   log_dd_kernel_f64(f64 x) noexcept
281   {
282     using namespace coeff::log_f64_data;
283 
284     int k;
285     f64 m = manip::frexp<f64>(x, &k);
286     if ( m < 0x1.6a09e667f3bcdp-1 ) {
287       m = m + m;
288       k -= 1;
289     }
290     // sterbenz f and q
291     const f64 f = m - 1.0;
292     const f64 q = 2.0 + f;
293 
294     const dd64 s = dd::div(dd64{ f, 0.0 }, dd64{ q, 0.0 });
295 
296     // f**2/2 exactly
297     const dd64 ff = dd::two_prod(f, f);
298     const dd64 hfsq{ ff.hi * 0.5, ff.lo * 0.5 };
299 
300     // s**2 in dd64 for the polynomial argument
301     const dd64 z_dd = dd::mul(s, s);
302     const f64 z = z_dd.hi;
303     const f64 w = z * z;
304 
305     // Remez polynomial in z, w
306     f64 t1 = hw::fmadd_sd(Lg6, w, Lg4);
307     t1 = hw::fmadd_sd(t1, w, Lg2);
308     t1 = t1 * w;
309     f64 t2 = hw::fmadd_sd(Lg7, w, Lg5);
310     t2 = hw::fmadd_sd(t2, w, Lg3);
311     t2 = hw::fmadd_sd(t2, w, Lg1);
312     t2 = t2 * z;
313     const f64 R = t1 + t2;
314 
315     // log(1+f) = (f - f**2/2) + s x (f**2/2 + R)
316     const dd64 hfsq_plus_R = dd::add(hfsq, R);
317     const dd64 corr = dd::mul(s, hfsq_plus_R);
318     const dd64 f_minus_hfsq = dd::sub(dd64{ f, 0.0 }, hfsq);
319     const dd64 log_m_dd = dd::add(f_minus_hfsq, corr);
320 
321     // k x ln2
322     const dd64 ln2_dd{ ln2_hi, ln2_lo };
323     const dd64 k_ln2 = dd::mul(ln2_dd, dd64{ f64(k), 0.0 });
324 
325     return dd::add(k_ln2, log_m_dd);
326   }

328   [[nodiscard, gnu::flatten]] inline constexpr dd64
329   log_dd_f64_with_special(f64 x, bool *handled) noexcept
330   {
331 
332     if ( ieee::is_nan(x) ) {
333       *handled = true;
334       return dd64{ x, 0.0 };
335     }
336     if ( x < 0 ) {
337       *handled = true;
338       return dd64{ ieee::qnan_v<f64>(), 0.0 };
339     }
340     if ( x == 0 ) {
341       *handled = true;
342       return dd64{ ieee::inf_v<f64>(1), 0.0 };
343     }
344     if ( ieee::is_inf(x) ) {
345       *handled = true;
346       return dd64{ x, 0.0 };
347     }
348     *handled = false;
349     return log_dd_kernel_f64(x);
350   }
