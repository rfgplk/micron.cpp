197   // pure shift+add kernels (rotation mode)
198   [[gnu::flatten, gnu::always_inline]] inline constexpr void
199   sincos_kernel_q62(i64 z, i64 *c_out, i64 *s_out) noexcept
200   {
201     i64 x = K_Q62;
202     i64 y = 0;
203     for ( int i = 0; i < N_F64; ++i ) {
204       const i64 nm = z >> 63;      // 0 if z >= 0, -1 if z < 0
205       const i64 ys = y >> i;
206       const i64 xs = x >> i;
207       const i64 ai = ATAN_Q62.v[i];
208       const i64 x_d = (ys ^ nm) - nm;
209       const i64 y_d = (xs ^ nm) - nm;
210       const i64 z_d = (ai ^ nm) - nm;
211       x -= x_d;
212       y += y_d;
213       z -= z_d;
214     }
215     *c_out = x;
216     *s_out = y;
217   }

241   // vectoring mode kernel (atan2)
242   [[gnu::flatten, gnu::always_inline]] inline constexpr i64
243   atan2_kernel_q62(i64 x, i64 y) noexcept
244   {
245     i64 z = 0;
246     for ( int i = 0; i < N_F64; ++i ) {
247       const i64 ym = y >> 63;
248       const i64 pm = ~ym;      // 0 if y<0, -1 if y>=0
249       const i64 ys = y >> i;
250       const i64 xs = x >> i;
251       const i64 ai = ATAN_Q62.v[i];
252       const i64 x_d = (ys ^ pm) - pm;
253       const i64 y_d = (xs ^ pm) - pm;
254       const i64 z_d = (ai ^ pm) - pm;
255       // sigma=+1 path: x -= ys, y += xs, z -= ai (pm=0)
256       // sigma=-1 path: x += ys, y -= xs, z += ai (pm=-1)
257       x -= x_d;
258       y += y_d;
259       z -= z_d;
260     }
261     return z;
262   }

308   [[gnu::flatten]] inline constexpr void
309   sincos_f64(f64 x, f64 *s_out, f64 *c_out) noexcept
310   {
311     if ( ieee::is_nan(x) || ieee::is_inf(x) ) [[unlikely]] {
312       *s_out = ieee::qnan_v<f64>();
313       *c_out = ieee::qnan_v<f64>();
314       return;
315     }
316     if ( manip::fabs(x) < 0x1.0p-26 ) [[unlikely]] {
317       *s_out = x;
318       *c_out = 1.0;
319       return;
320     }
321     f64 r;
322     const int q = __rr::reduce_pio2(x, &r);
323     // r is in (-π/2, π/2). Convert to Q2.62.
324     const i64 z = i64(r * F64_TO_Q62);
325     i64 ci, si;
326     sincos_kernel_q62(z, &ci, &si);
327     const f64 c = f64(ci) * Q62_TO_F64;
328     const f64 s = f64(si) * Q62_TO_F64;
329     switch ( q ) {
330     case 0:
331       *s_out = s;
332       *c_out = c;
333       return;
334     case 1:
335       *s_out = c;
336       *c_out = -s;
337       return;
338     case 2:
339       *s_out = -s;
340       *c_out = -c;
341       return;
342     default:
343       *s_out = -c;
344       *c_out = s;
345       return;
346     }
347   }

375   [[nodiscard, gnu::flatten]] inline constexpr f64
376   atan_f64(f64 x) noexcept
377   {
378     if ( ieee::is_nan(x) ) [[unlikely]]
379       return x;
380     if ( ieee::is_inf(x) ) [[unlikely]]
381       return manip::copysign<f64>(0x1.921fb54442d18p+0, x);      // ±π/2
382     if ( ieee::is_zero(x) || manip::fabs(x) < 0x1.0p-27 ) [[unlikely]]
383       return x;
384     const bool inv = manip::fabs(x) > 1.0;
385     const f64 t = inv ? (1.0 / x) : x;
386     const i64 x0 = i64(F64_TO_Q61);
387     const i64 y0 = i64(t * F64_TO_Q61);
388     const i64 z = atan2_kernel_q62(x0, y0);
389     const f64 a = f64(z) * Q62_TO_F64;
390     if ( !inv ) return a;
391     constexpr f64 pio2 = 0x1.921fb54442d18p+0;
392     return manip::signbit(x) ? f64(-pio2 - a) : f64(pio2 - a);
393   }

395   [[nodiscard, gnu::flatten]] inline constexpr f64
396   atan2_f64(f64 y, f64 x) noexcept
397   {
398     if ( ieee::is_nan(x) || ieee::is_nan(y) ) [[unlikely]]
399       return ieee::qnan_v<f64>();
400     constexpr f64 pi = 0x1.921fb54442d18p+1;
401     constexpr f64 pi_half = 0x1.921fb54442d18p+0;
402     const bool y_neg = manip::signbit(y);
403     const bool x_neg = manip::signbit(x);
404     if ( y == 0 ) [[unlikely]] {
405       if ( !x_neg ) return manip::copysign<f64>(0.0, y);
406       return y_neg ? -pi : pi;
407     }
408     if ( x == 0 ) [[unlikely]]
409       return y_neg ? -pi_half : pi_half;
410     if ( ieee::is_inf(x) || ieee::is_inf(y) ) [[unlikely]] {
411       constexpr f64 pi_quarter = 0x1.921fb54442d18p-1;
412       constexpr f64 three_pi_q = 0x1.2d97c7f3321d2p+1;
413       if ( ieee::is_inf(x) && ieee::is_inf(y) ) {
414         if ( !x_neg ) return y_neg ? -pi_quarter : pi_quarter;
415         return y_neg ? -three_pi_q : three_pi_q;
416       }
417       if ( ieee::is_inf(y) ) return y_neg ? -pi_half : pi_half;
418       if ( !x_neg ) return manip::copysign<f64>(0.0, y);
419       return y_neg ? -pi : pi;
420     }
421     const f64 a = atan_f64(y / x);
422     if ( !x_neg ) [[likely]]
423       return a;
424     return y_neg ? f64(a - pi) : f64(a + pi);
425   }

427   [[nodiscard, gnu::flatten]] inline constexpr f64
428   asin_f64(f64 x) noexcept
429   {
430     if ( ieee::is_nan(x) ) [[unlikely]]
431       return x;
432     const f64 ax = manip::fabs(x);
433     if ( ax > 1.0 ) [[unlikely]]
434       return ieee::qnan_v<f64>();
435     if ( ax == 1.0 ) return manip::copysign<f64>(0x1.921fb54442d18p+0, x);
436     // asin(x) = atan(x / sqrt(1 - x^2))
437     const f64 d = hw::sqrt_sd(1.0 - x * x);
438     return atan_f64(x / d);
439   }

442   acos_f64(f64 x) noexcept
443   {

450     const f64 d = hw::sqrt_sd((1.0 - x) * (1.0 + x));
451     return atan2_f64(d, x);

522   [[nodiscard, gnu::flatten]] inline constexpr f32
523   atan_f32(f32 x) noexcept
524   {
525     if ( ieee::is_nan(x) ) [[unlikely]]
526       return x;
527     if ( ieee::is_inf(x) ) [[unlikely]]
528       return manip::copysign<f32>(0x1.921fb6p+0f, x);
529     const bool inv = manip::fabs(x) > 1.0f;
530     const f32 t = inv ? (1.0f / x) : x;

532     const i32 y0 = i32(t * F32_TO_Q29);
533     const i32 z = atan2_kernel_q30(x0, y0);
534     const f32 a = f32(z) * Q30_TO_F32;
535     if ( !inv ) return a;
536     constexpr f32 pio2 = 0x1.921fb6p+0f;
537     return manip::signbit(x) ? f32(-pio2 - a) : f32(pio2 - a);
538   }
