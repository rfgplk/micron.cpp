13 template<u32 Factor, u32 Rows, u32 Limbs> struct __decimal_powers {
14   u16 offsets[Rows + 1]{};
15   u32 limbs[Limbs]{};
16 
17   consteval __decimal_powers()
18   {
19     u32 value[86] = { 1 };
20     u32 used = 1, position = 0;
21     for ( u32 row = 0; row < Rows; ++row ) {
22       offsets[row] = static_cast<u16>(position);
23       for ( u32 i = 0; i < used; ++i ) limbs[position++] = value[i];
24       if ( row + 1 == Rows ) break;
25       u64 carry = 0;
26       for ( u32 i = 0; i < used; ++i ) {
27         const u64 product = static_cast<u64>(value[i]) * Factor + carry;
28         value[i] = static_cast<u32>(product % 1000000000);
29         carry = product / 1000000000;
30       }
31       while ( carry ) {
32         value[used++] = static_cast<u32>(carry % 1000000000);
33         carry /= 1000000000;
34       }
35     }
36     offsets[Rows] = static_cast<u16>(position);
37   }
38 };
39 
40 // Packed base-10^9 powers: 16 KiB, generated by integer constant evaluation.
41 inline constexpr __decimal_powers<1220703125, 83, 3486> __decimal_fives;
42 inline constexpr __decimal_powers<536870912, 34, 565> __decimal_twos;
43 
44 };      // namespace micron::math::__impl
