92   [[nodiscard, gnu::flatten]] inline constexpr f64
93   erf_f64(f64 x) noexcept
94   {
95     using namespace __erf_data;
96     if ( ieee::is_nan(x) ) return x;
97     if ( ieee::is_inf(x) ) return manip::copysign<f64>(1.0, x);
98     f64 ax = manip::fabs(x);
99     if ( ax < 0x1.0p-28 ) return x + efx * x;      // tiny: erf(x) ≈ x*(1 + 2/sqrt(π))
100     if ( ax < 0.84375 ) {
101       f64 z = x * x;
102       f64 r = pp[0] + z * (pp[1] + z * (pp[2] + z * (pp[3] + z * pp[4])));
103       f64 s = qq[0] + z * (qq[1] + z * (qq[2] + z * (qq[3] + z * (qq[4] + z * qq[5]))));
104       return x + x * (r / s);
105     }
106     if ( ax < 1.25 ) {
107       f64 s = ax - 1.0;
108       f64 P = pa[0] + s * (pa[1] + s * (pa[2] + s * (pa[3] + s * (pa[4] + s * (pa[5] + s * pa[6])))));
109       f64 Q = qa[0] + s * (qa[1] + s * (qa[2] + s * (qa[3] + s * (qa[4] + s * (qa[5] + s * qa[6])))));
110       f64 r = erx + P / Q;
111       return manip::signbit(x) ? -r : r;
112     }
113     if ( ax >= 6.0 ) return manip::copysign<f64>(1.0, x);
114     f64 s = 1.0 / (ax * ax);
115     f64 R, S;
116     if ( ax < 2.857142857142857 ) {
117       R = ra[0] + s * (ra[1] + s * (ra[2] + s * (ra[3] + s * (ra[4] + s * (ra[5] + s * (ra[6] + s * ra[7]))))));
118       S = sa[0] + s * (sa[1] + s * (sa[2] + s * (sa[3] + s * (sa[4] + s * (sa[5] + s * (sa[6] + s * (sa[7] + s * sa[8])))))));
119     } else {
120       R = rb[0] + s * (rb[1] + s * (rb[2] + s * (rb[3] + s * (rb[4] + s * (rb[5] + s * rb[6])))));
121       S = sb[0] + s * (sb[1] + s * (sb[2] + s * (sb[3] + s * (sb[4] + s * (sb[5] + s * (sb[6] + s * sb[7]))))));
122     }
123     f64 z_hi = ieee::from_bits<f64>(ieee::to_bits(ax) & 0xffffffff00000000ULL);
124     f64 r = exp_ns::exp_f64(-z_hi * z_hi - 0.5625) * exp_ns::exp_f64((z_hi - ax) * (z_hi + ax) + R / S) / ax;
125     f64 res = 1.0 - r;
126     return manip::signbit(x) ? -res : res;
127   }

129   [[nodiscard, gnu::flatten]] inline constexpr f64
130   erfc_f64(f64 x) noexcept
131   {
132     using namespace __erf_data;
133     if ( ieee::is_nan(x) ) return x;
134     if ( ieee::is_inf(x) ) return manip::signbit(x) ? 2.0 : 0.0;
135     f64 ax = manip::fabs(x);
136     if ( ax < 0.84375 ) {
137       if ( ax < 0x1.0p-56 ) return 1.0 - x;      // tiny
138       f64 z = x * x;
139       f64 r = pp[0] + z * (pp[1] + z * (pp[2] + z * (pp[3] + z * pp[4])));
140       f64 s = qq[0] + z * (qq[1] + z * (qq[2] + z * (qq[3] + z * (qq[4] + z * qq[5]))));
141       f64 y = r / s;
142       if ( ax < 0.25 ) return 1.0 - (x + x * y);
143       f64 r2 = x * y;
144       r2 += (x - 0.5);
145       return 0.5 - r2;
146     }
147     if ( ax < 1.25 ) {
148       f64 s = ax - 1.0;
149       f64 P = pa[0] + s * (pa[1] + s * (pa[2] + s * (pa[3] + s * (pa[4] + s * (pa[5] + s * pa[6])))));
150       f64 Q = qa[0] + s * (qa[1] + s * (qa[2] + s * (qa[3] + s * (qa[4] + s * (qa[5] + s * qa[6])))));
151       if ( manip::signbit(x) ) return 1.0 + (erx + P / Q);
152       return 1.0 - erx - P / Q;
153     }
154     if ( ax >= 28.0 ) return manip::signbit(x) ? 2.0 : 0.0;
155     f64 s = 1.0 / (ax * ax);
156     f64 R, S;
157     if ( ax < 2.857142857142857 ) {
158       R = ra[0] + s * (ra[1] + s * (ra[2] + s * (ra[3] + s * (ra[4] + s * (ra[5] + s * (ra[6] + s * ra[7]))))));
159       S = sa[0] + s * (sa[1] + s * (sa[2] + s * (sa[3] + s * (sa[4] + s * (sa[5] + s * (sa[6] + s * (sa[7] + s * sa[8])))))));
160     } else {
161       R = rb[0] + s * (rb[1] + s * (rb[2] + s * (rb[3] + s * (rb[4] + s * (rb[5] + s * rb[6])))));
162       S = sb[0] + s * (sb[1] + s * (sb[2] + s * (sb[3] + s * (sb[4] + s * (sb[5] + s * (sb[6] + s * sb[7]))))));
163     }
164     f64 z_hi = ieee::from_bits<f64>(ieee::to_bits(ax) & 0xffffffff00000000ULL);
165     f64 r = exp_ns::exp_f64(-z_hi * z_hi - 0.5625) * exp_ns::exp_f64((z_hi - ax) * (z_hi + ax) + R / S) / ax;
166     return manip::signbit(x) ? f64(2.0 - r) : r;
167   }

169   [[nodiscard, gnu::flatten]] inline constexpr f64
170   tgamma_f64(f64 x) noexcept
171   {
172     using namespace coeff::lanczos_data;
173     if ( ieee::is_nan(x) ) return x;
174     if ( x == 0.0 ) return manip::copysign<f64>(ieee::inf_v<f64>(0), x);
175     if ( ieee::is_inf(x) && !manip::signbit(x) ) return x;
176 
177     if ( x < 0.5 ) {
178       if ( x <= 0.0 && round_ns::floor<f64>(x) == x ) return ieee::qnan_v<f64>();
179       f64 sin_pix = trig_ns::sin_f64(0x1.921fb54442d18p+1 * x);
180       if ( sin_pix == 0.0 ) return ieee::qnan_v<f64>();
181       return 0x1.921fb54442d18p+1 / (sin_pix * tgamma_f64(1.0 - x));
182     }
183     f64 z = x - 1.0;
184     f64 a = p[0];
185     for ( int i = 1; i < 9; ++i ) a += p[i] / (z + f64(i));
186     f64 t = z + g + 0.5;
187     f64 sqrt_2pi = 0x1.40d931ff62705p+1;
188     const f64 hp = pow_ns::pow<f64>(t, (z + 0.5) * 0.5);
189     return hw::fp_barrier(f64(sqrt_2pi * hp * exp_ns::exp_f64(-t))) * hp * a;
190   }

192   [[nodiscard, gnu::flatten]] inline constexpr f64
193   lgamma_f64(f64 x) noexcept
194   {
195     using namespace coeff::lanczos_data;
196     if ( ieee::is_nan(x) ) return x;
197     if ( ieee::is_inf(x) ) return ieee::inf_v<f64>(0);
198     if ( x == 0.0 ) return ieee::inf_v<f64>(0);
199 
200     f64 ax = manip::fabs(x);
201     if ( x < 0.5 ) {
202       if ( x <= 0.0 ) {
203         f64 fl = round_ns::floor<f64>(x);
204         if ( fl == x ) return ieee::inf_v<f64>(0);      // negative integer
205       }
206       f64 sin_pix = trig_ns::sin_f64(0x1.921fb54442d18p+1 * x);
207       f64 abs_sinpix = manip::fabs(sin_pix);
208       return f64(1.1447298858494002) - log_ns::log_f64(abs_sinpix) - lgamma_f64(1.0 - x);
209     }
210     f64 z = x - 1.0;
211     f64 a = p[0];
212     for ( int i = 1; i < 9; ++i ) a += p[i] / (z + f64(i));
213     f64 t = z + g + 0.5;
214     (void)ax;
215     return 0.91893853320467267 + (z + 0.5) * log_ns::log_f64(t) - t + log_ns::log_f64(a);
216   }

283   // P = sum (-1)^k a_2k / z^2k, Q = sum (-1)^k a_{2k+1} / z^{2k+1}, with the Hankel
284   // a_k(nu) = prod_{j=1..k} (4nu^2 - (2j-1)^2) / (k! 8^k); every one of these is exact in f64
285   [[gnu::flatten]] inline constexpr f64
286   hankel_PQ_j0(f64 x, f64 *P, f64 *Q) noexcept
287   {
288     const f64 ax = manip::fabs(x);
289     const f64 r = 1.0 / ax;
290     const f64 u = r * r;
291     *P = 1.0 + u * (-0.0703125 + u * (0.112152099609375 + u * (-0.57250142097473145)));
292     *Q = r * (-0.125 + u * (0.0732421875 + u * (-0.22710800170898438)));
293     return ax;
294   }

309   [[nodiscard, gnu::flatten]] inline constexpr f64
310   j0_f64(f64 x) noexcept
311   {
312     if ( ieee::is_nan(x) ) [[unlikely]]
313       return x;
314     if ( ieee::is_inf(x) ) [[unlikely]]
315       return 0.0;
316     const f64 ax = manip::fabs(x);
317     if ( ax < 8.0 ) return __bessel::j0_small(x);
318     f64 P, Q;
319     __bessel::hankel_PQ_j0(x, &P, &Q);
320     const f64 beta = ax - __bessel::pi_quarter;
321     f64 sb, cb;
322     trig_ns::sincos_f64(beta, &sb, &cb);
323     const f64 amp = sqrt_ns::sqrt<f64>(__bessel::two_over_pi / ax);
324     return amp * (P * cb - Q * sb);
325   }

346   [[nodiscard, gnu::flatten]] inline constexpr f64
347   y0_f64(f64 x) noexcept
348   {
349     if ( ieee::is_nan(x) ) [[unlikely]]
350       return x;
351     if ( x == 0.0 ) [[unlikely]]
352       return ieee::inf_v<f64>(1);
353     if ( x < 0.0 ) [[unlikely]]
354       return ieee::qnan_v<f64>();
355     if ( ieee::is_inf(x) ) [[unlikely]]
356       return 0.0;
357     if ( x < 8.0 ) {
358       constexpr f64 gamma_e = 0.5772156649015329;
359       const f64 z = 0.25 * x * x;
360       f64 sum = 0.0;
361       f64 z_pow = 1.0;
362       f64 fact_sq = 1.0;
363       f64 hk = 0.0;
364       for ( int k = 1; k < 25; ++k ) {
365         z_pow *= z;
366         fact_sq *= f64(k) * f64(k);
367         hk += 1.0 / f64(k);
368         const f64 sign = (k & 1) ? 1.0 : -1.0;
369         sum += sign * hk * z_pow / fact_sq;
370       }
371       const f64 j0v = __bessel::j0_small(x);
372       return (__bessel::two_over_pi) * ((log_ns::log_f64(0.5 * x) + gamma_e) * j0v + sum);
373     }
374     f64 P, Q;
375     __bessel::hankel_PQ_j0(x, &P, &Q);
376     const f64 beta = x - __bessel::pi_quarter;
377     f64 sb, cb;
378     trig_ns::sincos_f64(beta, &sb, &cb);
379     const f64 amp = sqrt_ns::sqrt<f64>(__bessel::two_over_pi / x);
380     return amp * (P * sb + Q * cb);
381   }

383   [[nodiscard, gnu::flatten]] inline constexpr f64
384   y1_f64(f64 x) noexcept
385   {
386     if ( ieee::is_nan(x) ) [[unlikely]]
387       return x;
388     if ( x == 0.0 ) [[unlikely]]
389       return ieee::inf_v<f64>(1);
390     if ( x < 0.0 ) [[unlikely]]
391       return ieee::qnan_v<f64>();
392     if ( ieee::is_inf(x) ) [[unlikely]]
393       return 0.0;
394     if ( x < 8.0 ) {
395       constexpr f64 gamma_e = 0.5772156649015329;
396       const f64 z = 0.25 * x * x;
397       f64 sum = 0.0;
398       f64 z_pow_neg = 1.0;
399       f64 fact_prod = 1.0;
400       f64 hk = 0.0;
401       f64 hk1 = 1.0;
402       for ( int k = 0; k < 25; ++k ) {
403         sum += (hk + hk1) * z_pow_neg / fact_prod;
404         z_pow_neg *= -z;
405         fact_prod *= f64(k + 1) * f64(k + 2);
406         hk = hk1;
407         hk1 += 1.0 / f64(k + 2);
408       }
409       const f64 j1v = __bessel::j1_small(x);
410       return (__bessel::two_over_pi) * (-1.0 / x + (log_ns::log_f64(0.5 * x) + gamma_e) * j1v - 0.25 * x * sum);
411     }
412     f64 P, Q;
413     __bessel::hankel_PQ_j1(x, &P, &Q);
414     const f64 beta = x - 3.0 * __bessel::pi_quarter;
415     f64 sb, cb;
416     trig_ns::sincos_f64(beta, &sb, &cb);
417     const f64 amp = sqrt_ns::sqrt<f64>(__bessel::two_over_pi / x);
418     return amp * (P * sb + Q * cb);
419   }
