24 template<ieee754_floating F, usize Dg> struct poly_coeffs {
25   static constexpr usize size = Dg + 1;
26   static constexpr usize degree = Dg;
27   using value_type = F;
28   F data[size];
29 
30   [[nodiscard, gnu::always_inline]] constexpr const F &
31   operator[](usize i) const noexcept
32   {
33     return data[i];
34   }
35 
36   [[nodiscard, gnu::always_inline]] constexpr F &
37   operator[](usize i) noexcept
38   {
39     return data[i];
40   }
41 
42   [[nodiscard, gnu::always_inline]] constexpr const F *
43   begin() const noexcept
44   {
45     return data;
46   }
47 
48   [[nodiscard, gnu::always_inline]] constexpr const F *
49   end() const noexcept
50   {
51     return data + size;
52   }
53 };

66 template<ieee754_floating F>
67 [[nodiscard, gnu::always_inline]] inline constexpr F
68 fma(F a, F b, F c) noexcept
69 {
70   if consteval {
71 #if defined(__micron_arch_wasm_any)
72     return micron::wasm::math::fma(a, b, c);
73 #elif defined(__micron_compiler_clang)
74     return F(a * b + c);
75 #endif
76   }
77 #if defined(__micron_arch_arm32)
78   // VFPv3 uses its native non-fused VMLA; VFPv4 keeps fused VFMA semantics
79   return hw::fmadd<F>(a, b, c);
80 #else
81   // __builtins are ok, compiler splices in place
82   if constexpr ( sizeof(F) == sizeof(float) )
83     return F(__builtin_fmaf(float(a), float(b), float(c)));
84   else if constexpr ( sizeof(F) == sizeof(double) )
85     return F(__builtin_fma(double(a), double(b), double(c)));
86   else
87     return F(__builtin_fmal(static_cast<long double>(a), static_cast<long double>(b), static_cast<long double>(c)));
88 #endif
89 }

91 template<ieee754_floating F, usize N>
92 [[nodiscard, gnu::always_inline]] inline constexpr F
93 horner(const poly_coeffs<F, N> &c, F x) noexcept
94 {
95   const F *__restrict__ const p = c.data;
96   F r = p[N];
97   for ( usize i = N; i-- > 0; ) r = fma<F>(r, x, p[i]);
98   return r;
99 }

101 template<ieee754_floating F>
102 [[nodiscard, gnu::always_inline]] inline constexpr F
103 horner_p(const F *__restrict__ c, usize n, F x) noexcept
104 {
105   F r = c[n];
106   while ( n-- != 0 ) r = fma<F>(r, x, c[n]);
107   return r;
108 }

110 // 2-chain split Horner for even degree polynomials in s = x**2
111 // p(s) = (c0 + c2*s + c4*s**2 + ...) + s*(c1 + c3*s + c5*s**2 + ..)
112 template<ieee754_floating F>
113 [[nodiscard, gnu::always_inline]] inline constexpr F
114 horner_split2(const F *__restrict__ even, usize ne, const F *__restrict__ odd, usize no, F s) noexcept
115 {
116   // p0/p1 will run in parallel
117   F pe = even[ne];
118   F po = odd[no];
119   for ( usize i = ne; i-- > 0; ) pe = fma<F>(pe, s, even[i]);
120   for ( usize i = no; i-- > 0; ) po = fma<F>(po, s, odd[i]);
121   return fma<F>(po, s, pe);
122 }
