39   [[nodiscard, gnu::flatten]] inline constexpr f64
40   exp_f64(f64 x) noexcept
41   {
42     using namespace coeff::exp_f64_data;
43 
44     if ( ieee::is_nan(x) ) [[unlikely]]
45       return x;
46     if ( x > 0x1.62e42fefa39efp+9 ) [[unlikely]]      // 709.7827128933840: largest x with finite exp(x)
47       return ieee::inf_v<f64>(0);
48     if ( x < -745.13 ) [[unlikely]]
49       return 0.0;
50     if ( x == 0.0 ) [[unlikely]]
51       return 1.0;
52 
53     // N = round(x * 32 / ln2)
54     f64 fN = round_ns::rint<f64>(x * inv_ln2_32);
55     i64 N = i64(fN);
56     f64 r = hw::fmadd_sd(-fN, ln2_32_hi, x);
57     r = hw::fmadd_sd(-fN, ln2_32_lo, r);
58 
59     // 1 + r + r**2*(c0 + r*(c1 + r*(c2 + r*(c3 + r*c4))))
60     f64 r2 = r * r;
61     f64 p = hw::fmadd_sd(rem[4], r, rem[3]);
62     p = hw::fmadd_sd(p, r, rem[2]);
63     p = hw::fmadd_sd(p, r, rem[1]);
64     p = hw::fmadd_sd(p, r, rem[0]);
65     f64 exp_r = hw::fmadd_sd(p, r2, r) + 1.0;
66 
67     // 2^(N>>5) * 2^((N&31)/32) * exp_r
68     i64 j = N & 31;
69     i32 k = i32(N >> 5);
70     f64 tw = twoN[j];
71     f64 r_full = tw * exp_r;
72     return manip::ldexp<f64>(r_full, int(k));
73   }

75   [[nodiscard, gnu::flatten]] inline constexpr f32
76   exp_f32(f32 x) noexcept
77   {
78     using namespace coeff::exp_f32_data;
79 
80     if ( ieee::is_nan(x) ) [[unlikely]]
81       return x;
82     if ( x > 88.72283935546875f ) [[unlikely]]      // first f32 above this overflows; kernel yields inf for it
83       return ieee::inf_v<f32>(0);
84     if ( x < -103.972f ) [[unlikely]]
85       return 0.0f;
86     if ( x == 0.0f ) [[unlikely]]
87       return 1.0f;
88 
89     f32 fN = round_ns::rint<f32>(x * inv_ln2_16);
90     i32 N = i32(fN);
91     f32 r = (x - fN * ln2_16_hi) - fN * ln2_16_lo;
92 
93     f32 r2 = r * r;
94     f32 p = hw::fmadd_ss(rem[1], r, rem[0]);
95     f32 exp_r = hw::fmadd_ss(p, r2, r) + 1.0f;
96 
97     i32 j = N & 15;
98     i32 k = N >> 4;
99     return manip::ldexp<f32>(twoN[j] * exp_r, int(k));
100   }

102   [[nodiscard, gnu::flatten]] inline constexpr f64
103   expm1_f64(f64 x) noexcept
104   {
105     using namespace coeff::exp_f64_data;
106     if ( ieee::is_nan(x) ) [[unlikely]]
107       return x;
108     if ( x > 0x1.62e42fefa39efp+9 ) [[unlikely]]      // 709.7827128933840: largest x with finite exp(x)
109       return ieee::inf_v<f64>(0);
110     if ( x < -38.0 ) [[unlikely]]
111       return -1.0;
112     f64 ax = manip::fabs(x);
113     if ( ax < 0x1.0p-7 ) {
114       f64 x2 = x * x;
115       f64 p = hw::fmadd_sd(rem[4], x, rem[3]);
116       p = hw::fmadd_sd(p, x, rem[2]);
117       p = hw::fmadd_sd(p, x, rem[1]);
118       p = hw::fmadd_sd(p, x, rem[0]);
119       return hw::fmadd_sd(p, x2, x);
120     }
121     if ( ax < 0.5 ) {
122       f64 fN = round_ns::rint<f64>(x * inv_ln2_32);
123       i64 N = i64(fN);
124       f64 r = (x - fN * ln2_32_hi) - fN * ln2_32_lo;
125       f64 r2 = r * r;
126       f64 p = hw::fmadd_sd(rem[4], r, rem[3]);
127       p = hw::fmadd_sd(p, r, rem[2]);
128       p = hw::fmadd_sd(p, r, rem[1]);
129       p = hw::fmadd_sd(p, r, rem[0]);
130       f64 exp_r_minus_1 = hw::fmadd_sd(p, r2, r);
131       i64 j = N & 31;
132       i32 k = i32(N >> 5);
133       f64 t = twoN[j];
134       f64 twoN_m1 = t - 1.0;
135       f64 part = twoN_m1 + t * exp_r_minus_1;
136       if ( k == 0 ) return part;
137       return manip::ldexp<f64>(part + 1.0, int(k)) - 1.0;
138     }
139     return exp_f64(x) - 1.0;
140   }

142   [[nodiscard, gnu::flatten]] inline constexpr f32
143   expm1_f32(f32 x) noexcept
144   {
145     using namespace coeff::exp_f32_data;
146     if ( ieee::is_nan(x) ) [[unlikely]]
147       return x;
148     if ( x > 88.72283935546875f ) [[unlikely]]      // first f32 above this overflows
149       return ieee::inf_v<f32>(0);
150     if ( x < -20.0f ) [[unlikely]]
151       return -1.0f;
152     f32 ax = manip::fabs(x);
153     if ( ax < 0x1.0p-3f ) {
154       // expm1(x) = x + x^2*(1/2 + x*(1/6 + x*(1/24))); the 1/24 term keeps this <=1 ULP near 0.1
155       f32 x2 = x * x;
156       f32 p = hw::fmadd_ss(0x1.555556p-5f, x, rem[1]);
157       p = hw::fmadd_ss(p, x, rem[0]);
158       return hw::fmadd_ss(p, x2, x);
159     }
160     return exp_f32(x) - 1.0f;
161   }

163   [[nodiscard, gnu::always_inline]] inline constexpr f64
164   exp2_f64(f64 x) noexcept
165   {
166     using namespace coeff::exp_f64_data;
167     if ( ieee::is_nan(x) ) [[unlikely]]
168       return x;
169     if ( x >= 1024.0 ) [[unlikely]]
170       return ieee::inf_v<f64>(0);
171     if ( x <= -1075.0 ) [[unlikely]]
172       return 0.0;
173 
174     // M = round(x * 32)
175     const f64 fM = round_ns::rint<f64>(x * 32.0);
176     const i64 M = i64(fM);
177     const i32 j = i32(M & 31);
178     const i32 k = i32(M >> 5);
179     // rho = x - M/32
180     const f64 rho = x - fM * 0x1.0p-5;
181     // r = rgo * ln2
182     constexpr f64 ln2_hi = 0x1.62e42fefa39efp-1;
183     constexpr f64 ln2_lo = 0x1.abc9e3b39803fp-56;
184     const f64 r = hw::fmadd_sd(rho, ln2_hi, rho * ln2_lo);
185     // exp(r) = 1 + r + r**2 x P(r)
186     f64 P = hw::fmadd_sd(r, rem[4], rem[3]);
187     P = hw::fmadd_sd(r, P, rem[2]);
188     P = hw::fmadd_sd(r, P, rem[1]);
189     P = hw::fmadd_sd(r, P, rem[0]);
190     const f64 r2 = r * r;
191     const f64 expr = hw::fmadd_sd(r2, P, 1.0 + r);
192     // 2^x = ldexp(twoN[j] * exp(r), k)
193     return manip::ldexp<f64>(twoN[j] * expr, k);
194   }

196   [[nodiscard, gnu::always_inline]] inline constexpr f32
197   exp2_f32(f32 x) noexcept
198   {
199     if ( ieee::is_nan(x) ) [[unlikely]]
200       return x;
201     if ( x >= 128.0f ) [[unlikely]]
202       return ieee::inf_v<f32>(0);
203     if ( x <= -150.0f ) [[unlikely]]
204       return 0.0f;
205     constexpr f32 ln2_hi = 0x1.62e430p-1f;
206     constexpr f32 ln2_lo = -0x1.b9e3afp-26f;
207     const f32 fN = round_ns::rint<f32>(x);
208     const f32 r = x - fN;
209     const f32 y = hw::fmadd_ss(r, ln2_hi, r * ln2_lo);
210     return manip::ldexp<f32>(exp_f32(y), int(i32(fN)));
211   }

214   [[nodiscard, gnu::always_inline]] inline constexpr f64
215   exp10_f64(f64 x) noexcept
216   {
217     if ( ieee::is_nan(x) ) return x;
218     if ( x > 308.25 ) return ieee::inf_v<f64>(0);
219     if ( x < -323.3 ) return 0.0;
220     // ln10 dd64 split
221     constexpr f64 ln10_hi = 0x1.26bb1bbb55516p+1;
222     constexpr f64 ln10_lo = -0x1.f48ad494ea3e9p-53;
223     f64 y = hw::fmadd_sd(x, ln10_hi, x * ln10_lo);
224     return exp_f64(y);
225   }

294   [[nodiscard, gnu::flatten]] inline constexpr dd64
295   exp_dd_kernel_f64(f64 x) noexcept
296   {
297     using namespace coeff::exp_f64_data;
298 
299     const f64 fN = round_ns::rint<f64>(x * inv_ln2_32);
300     const i64 N = i64(fN);
301     const f64 r_hi = hw::fmadd_sd(-fN, ln2_32_hi, x);
302     const f64 r_corr = -fN * ln2_32_lo;
303     const dd64 r_dd = dd::add(dd64{ r_hi, 0.0 }, r_corr);
304 
305     dd64 p = dd64{ rem[4], 0.0 };
306     p = dd::add(dd::mul(p, r_dd), rem[3]);
307     p = dd::add(dd::mul(p, r_dd), rem[2]);
308     p = dd::add(dd::mul(p, r_dd), rem[1]);
309     p = dd::add(dd::mul(p, r_dd), rem[0]);
310 
311     const dd64 r_sq = dd::mul(r_dd, r_dd);
312     const dd64 one_plus_r = dd::add(dd64{ 1.0, 0.0 }, r_dd);
313     const dd64 r2P = dd::mul(r_sq, p);
314     const dd64 exp_r = dd::add(one_plus_r, r2P);
315 
316     const i64 j = N & 31;
317     const i32 k = i32(N >> 5);
318     const dd64 t = dd::mul(dd64{ twoN[j], 0.0 }, exp_r);
319 
320     const f64 hi = manip::ldexp<f64>(t.hi, int(k));
321     const f64 lo = manip::ldexp<f64>(t.lo, int(k));
322     return dd64{ hi, lo };
323   }

325   [[nodiscard, gnu::flatten]] inline constexpr dd64
326   exp_dd_f64_with_special(f64 x, bool *handled) noexcept
327   {
328     if ( ieee::is_nan(x) ) {
329       *handled = true;
330       return dd64{ x, 0.0 };
331     }
332     if ( x > 0x1.62e42fefa39efp+9 ) {      // 709.7827128933840: largest x with finite exp(x)
333       *handled = true;
334       return dd64{ ieee::inf_v<f64>(0), 0.0 };
335     }
336     if ( x < -745.13 ) {
337       *handled = true;
338       return dd64{ 0.0, 0.0 };
339     }
340     if ( x == 0.0 ) {
341       *handled = true;
342       return dd64{ 1.0, 0.0 };
343     }
344     *handled = false;
345     return exp_dd_kernel_f64(x);
346   }
