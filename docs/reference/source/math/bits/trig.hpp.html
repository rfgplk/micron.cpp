57   // cody waite reduction: r = x - N * pi/2
58   [[nodiscard, gnu::always_inline]] inline constexpr int
59   cody_waite(f64 x, f64 *r) noexcept
60   {
61     // N = round(x * 2/pi)
62     f64 fN = round_ns::rint<f64>(x * inv_pio2);
63     i64 N = i64(fN);
64     f64 t = x - fN * pio2_hi;
65     t = t - fN * pio2_mid;
66     t = t - fN * pio2_lo;
67     *r = t;
68     return int(N & 3);
69   }

71   [[nodiscard, gnu::always_inline]] inline constexpr int
72   iter_cody_waite(f64 x, f64 *r) noexcept
73   {
74     int N_total = 0;
75     f64 t = x;
76     for ( int pass = 0; pass < 2 && manip::fabs(t) >= 0x1.0p20; ++pass ) {
77       f64 fN = round_ns::rint<f64>(t * inv_pio2);
78       if ( fN > 0x1.0p20 )
79         fN = 0x1.0p20;
80       else if ( fN < -0x1.0p20 )
81         fN = -0x1.0p20;
82       i64 N = i64(fN);
83       t = t - fN * pio2_hi;
84       t = t - fN * pio2_mid;
85       t = t - fN * pio2_lo;
86       N_total += int(N);
87     }
88     f64 fN = round_ns::rint<f64>(t * inv_pio2);
89     i64 N = i64(fN);
90     t = t - fN * pio2_hi;
91     t = t - fN * pio2_mid;
92     t = t - fN * pio2_lo;
93     N_total += int(N);
94     *r = t;
95     return N_total & 3;
96   }

113   inline constexpr f64 pio2_w0 = 0x1.921fb54400000p+0;      // pi/2 to 33 bits

117   [[nodiscard, gnu::flatten]] inline constexpr int
118   payne_hanek(f64 x, f64 *r) noexcept
119   {
120     using TR = ieee::traits<f64>;
121     using u128 = uint128_t;
122 
123     const u64 b = ieee::to_bits(x);
124     const bool neg = (b & TR::sign_mask) != 0;
125     const u64 m = (b & TR::mant_mask) | TR::implicit_one;
126     const int e = int((b & TR::exp_mask) >> TR::mant_bits) - TR::exp_bias;
127 
128     // NOTE: acc bit J represents weight 2^(J-120) of x*2/pi; bits 120-121 hold q (mod 4); bits 0..119 hold the fractional part
129     const int idx_start = (e >= 83) ? ((e - 83 + 23) / 24) : 0;
130     const int shift_max = e + 44 - 24 * idx_start;
131 
132     u128 acc = u128{ 0 };
133     for ( int k = 0; k < 9; ++k ) {
134       const int idx = idx_start + k;
135       if ( idx >= 66 ) break;
136       const u128 prod = u128{ m } * u128{ u64(ipio2[idx]) };      // up to 77-bit
137       const int shift = shift_max - 24 * k;
138       if ( shift >= 0 && shift < 128 ) {
139         acc = acc + (prod << shift);
140       } else if ( shift < 0 && shift > -77 ) {
141         acc = acc + (prod >> (-shift));
142       } else if ( shift <= -77 ) {
143         break;      // entire prod below precision floor; further chunks also below
144       }
145       // shift >= 128 cannot happen: shift_max <= 127 by construction of idx_start
146     }
147 
148     // bits 120 through 127 of acc hold the integer part
149     const u64 acc_hi = u64(acc >> 64);
150     const u32 int_part = u32(acc_hi >> 56);
151     int q = int(int_part & 3u);
152 
153     u128 frac = acc - (u128{ u64(int_part) } << 120);
154     const u64 frac_hi = u64(frac >> 64);
155     const u64 frac_mid = u64(frac);
156 
157     f64 f_hi = f64(frac_hi) * 0x1.0p-56;
158     f64 f_lo = f64(frac_mid) * 0x1.0p-120;
159     f64 f = f_hi + f_lo;
160 
161     if ( f >= 0.5 ) {
162       f -= 1.0;
163       q = (q + 1) & 3;
164     }
165 
166     f64 t = f * pio2_w0;
167     t = hw::fmadd_sd(f, pio2_w1, t);
168     t = hw::fmadd_sd(f, pio2_w2, t);
169 
170     if ( neg ) {
171       q = (4 - q) & 3;
172       t = -t;
173     }
174     *r = t;
175     return q;
176   }

180   [[nodiscard, gnu::flatten]] inline constexpr int
181   cody_waite_dd(f64 x, dd64 *r) noexcept
182   {
183     const f64 fN = round_ns::rint<f64>(x * inv_pio2);
184     const i64 N = i64(fN);
185     // NOTE: cody-waite needs x - N*C1 EXACT
186     const dd64 p_hi = dd::two_prod(fN, pio2_hi);
187     const dd64 p_mid = dd::two_prod(fN, pio2_mid);
188     const dd64 p_lo = dd::two_prod(fN, pio2_lo);
189     dd64 t = dd::sub(dd64{ x, 0.0 }, p_hi);
190     t = dd::sub(t, p_mid);
191     t = dd::sub(t, p_lo);
192     *r = t;
193     return int(N & 3);
194   }

196   [[nodiscard, gnu::always_inline]] inline constexpr int
197   reduce_pio2(f64 x, f64 *r) noexcept
198   {
199     const f64 ax = manip::fabs(x);
200     if ( ax < 0x1.0p20 ) return cody_waite(x, r);
201     if ( ax < 0x1.0p33 ) {
202       dd64 rd;
203       const int q = cody_waite_dd(x, &rd);
204       *r = rd.to_double();
205       return q;
206     }
207     return __paynehanek::payne_hanek(x, r);
208   }

225   ksin_f64(f64 r) noexcept
226   {
227     using namespace coeff::sin_f64_data;
228     f64 z = r * r;
229     f64 r3 = z * r;
230     f64 p = hw::fmadd_sd(S6, z, S5);

232     p = hw::fmadd_sd(p, z, S3);

234     p = hw::fmadd_sd(p, z, S1);
235     return r + r3 * p;

238   [[nodiscard, gnu::always_inline]] inline constexpr f64
239   kcos_f64(f64 r) noexcept
240   {
241     using namespace coeff::sin_f64_data;
242     f64 z = r * r;
243     f64 hz = 0.5 * z;
244     f64 p = hw::fmadd_sd(C6, z, C5);
245     p = hw::fmadd_sd(p, z, C4);
246     p = hw::fmadd_sd(p, z, C3);
247     p = hw::fmadd_sd(p, z, C2);
248     p = hw::fmadd_sd(p, z, C1);
249     return (1.0 - hz) + z * z * p;

268   kcos_dd_f64(dd64 r) noexcept
269   {
270     using namespace coeff::sin_f64_data;

285   sin_f64(f64 x) noexcept
286   {
287     if ( ieee::is_nan(x) ) [[unlikely]]
288       return x;
289     if ( ieee::is_inf(x) ) [[unlikely]]

291     if ( manip::fabs(x) < 0x1.0p-26 ) [[unlikely]]
292       return x;      // return x if small enough

295     switch ( q ) {
296     case 0:

299       return kcos_f64(r);
300     case 2:
301       return -ksin_f64(r);
302     default:
303       return -kcos_f64(r);

331   sincos_f64(f64 x, f64 *s, f64 *c) noexcept
332   {
333     if ( ieee::is_nan(x) || ieee::is_inf(x) ) [[unlikely]] {
334       *s = ieee::qnan_v<f64>();
335       *c = ieee::qnan_v<f64>();

337     }

341       return;
342     }
343     f64 r;

345     f64 ks = ksin_f64(r);

348     case 0:
349       *s = ks;

353       *s = kc;

355       return;
356     case 2:
357       *s = -ks;
358       *c = -kc;
359       return;
360     default:
361       *s = -kc;
362       *c = ks;
363       return;
364     }

381   [[nodiscard, gnu::flatten]] inline constexpr f64
382   atan_f64(f64 x) noexcept
383   {
384     using namespace coeff::atan_f64_data;
385     if ( ieee::is_nan(x) ) return x;
386     f64 ax = manip::fabs(x);
387     if ( ax >= 0x1.0p+61 ) return manip::copysign<f64>(0x1.921fb54442d18p+0, x);
388     if ( ax < 0x1.0p-29 ) return x;      // tiny
389 
390     int idx;
391     f64 t;
392     if ( ax < 0.4375 ) {
393       idx = 0;
394       t = ax;
395     } else if ( ax < 0.6875 ) {
396       idx = 1;
397       t = (2.0 * ax - 1.0) / (2.0 + ax);
398     } else if ( ax < 1.1875 ) {
399       idx = 2;
400       t = (ax - 1.0) / (ax + 1.0);
401     } else if ( ax < 2.4375 ) {
402       idx = 3;
403       t = (ax - 1.5) / (1.0 + 1.5 * ax);
404     } else {
405       idx = 4;
406       t = -1.0 / ax;
407     }
408 
409     f64 z = t * t;
410     f64 w = z * z;
411     f64 s1 = hw::fmadd_sd(aT[10], w, aT[8]);
412     s1 = hw::fmadd_sd(s1, w, aT[6]);
413     s1 = hw::fmadd_sd(s1, w, aT[4]);
414     s1 = hw::fmadd_sd(s1, w, aT[2]);
415     s1 = hw::fmadd_sd(s1, w, aT[0]);
416     f64 s2 = hw::fmadd_sd(aT[9], w, aT[7]);
417     s2 = hw::fmadd_sd(s2, w, aT[5]);
418     s2 = hw::fmadd_sd(s2, w, aT[3]);
419     s2 = hw::fmadd_sd(s2, w, aT[1]);
420     f64 P = s1 + z * s2;
421     f64 atan_t = t - t * z * P;
422     f64 r = (idx == 0) ? atan_t : f64(atan_lo[idx] + atan_t);
423     return manip::signbit(x) ? -r : r;
424   }

426   [[nodiscard, gnu::flatten]] inline constexpr f64
427   atan2_f64(f64 y, f64 x) noexcept
428   {
429     if ( ieee::is_nan(x) || ieee::is_nan(y) ) [[unlikely]]
430       return ieee::qnan_v<f64>();
431 
432     constexpr f64 pi = 0x1.921fb54442d18p+1;
433     constexpr f64 pi_half = 0x1.921fb54442d18p+0;
434     constexpr f64 pi_quarter = 0x1.921fb54442d18p-1;
435     constexpr f64 three_pi_q = 0x1.2d97c7f3321d2p+1;
436 
437     const bool y_neg = manip::signbit(y);
438     const bool x_neg = manip::signbit(x);
439 
440     if ( y == 0 ) [[unlikely]] {
441       if ( !x_neg ) return manip::copysign<f64>(0.0, y);
442       return y_neg ? -pi : pi;
443     }
444     if ( x == 0 ) [[unlikely]]
445       return y_neg ? -pi_half : pi_half;
446     if ( ieee::is_inf(x) ) [[unlikely]] {
447       if ( ieee::is_inf(y) ) {
448         if ( !x_neg ) return y_neg ? -pi_quarter : pi_quarter;
449         return y_neg ? -three_pi_q : three_pi_q;
450       }
451       if ( !x_neg ) return manip::copysign<f64>(0.0, y);
452       return y_neg ? -pi : pi;
453     }
454     if ( ieee::is_inf(y) ) [[unlikely]]
455       return y_neg ? -pi_half : pi_half;
456 
457     const f64 z = atan_f64(y / x);
458     if ( !x_neg ) [[likely]]
459       return z;
460     return y_neg ? f64(z - pi) : f64(z + pi);
461   }

463   [[nodiscard, gnu::flatten]] inline constexpr f64
464   asin_f64(f64 x) noexcept
465   {
466     if ( ieee::is_nan(x) ) [[unlikely]]
467       return x;
468     f64 ax = manip::fabs(x);
469     if ( ax > 1.0 ) [[unlikely]]
470       return ieee::qnan_v<f64>();
471     if ( ax < 0x1.0p-27 ) [[unlikely]]
472       return x;
473     constexpr f64 pS0 = 1.66666666666666657415e-01;
474     constexpr f64 pS1 = -3.25565818622400915405e-01;
475     constexpr f64 pS2 = 2.01212532134862925881e-01;
476     constexpr f64 pS3 = -4.00555345006794114027e-02;
477     constexpr f64 pS4 = 7.91534994289814532176e-04;
478     constexpr f64 pS5 = 3.47933107596021167570e-05;
479     constexpr f64 qS1 = -2.40339491173441421878e+00;
480     constexpr f64 qS2 = 2.02094576023350569471e+00;
481     constexpr f64 qS3 = -6.88283971605453293030e-01;
482     constexpr f64 qS4 = 7.70381505559019352791e-02;
483     constexpr f64 pio2_hi = 1.57079632679489655800e+00;
484     constexpr f64 pio2_lo = 6.12323399573676603587e-17;
485 
486     if ( ax < 0.5 ) {
487       f64 t = x * x;
488       f64 p = t * (pS0 + t * (pS1 + t * (pS2 + t * (pS3 + t * (pS4 + t * pS5)))));
489       f64 q = 1.0 + t * (qS1 + t * (qS2 + t * (qS3 + t * qS4)));
490       return x + x * (p / q);
491     }
492     f64 w = 1.0 - ax;
493     f64 t = w * 0.5;
494     f64 p = t * (pS0 + t * (pS1 + t * (pS2 + t * (pS3 + t * (pS4 + t * pS5)))));
495     f64 q = 1.0 + t * (qS1 + t * (qS2 + t * (qS3 + t * qS4)));
496     f64 s = sqrt_ns::sqrt<f64>(t);
497 
498     f64 result;
499     if ( ax >= 0.975 ) {
500       f64 wq = p / q;
501       result = pio2_hi - (2.0 * (s + s * wq) - pio2_lo);
502     } else {
503       using T = ieee::traits<f64>;
504       using U = T::uint_type;
505       U sb = ieee::to_bits(s) & 0xffffffff00000000ULL;
506       f64 ws = ieee::from_bits<f64>(sb);
507       f64 c = (t - ws * ws) / (s + ws);
508       f64 r = p / q;
509       f64 pp = 2.0 * s * r - (pio2_lo - 2.0 * c);
510       f64 qq = pio2_hi - 2.0 * ws;
511       result = qq - pp;
512     }
513     return manip::signbit(x) ? -result : result;
514   }

517   acos_f64(f64 x) noexcept
518   {
519     if ( ieee::is_nan(x) ) return x;
520     f64 ax = manip::fabs(x);
521     if ( ax > 1.0 ) return ieee::qnan_v<f64>();
522     constexpr f64 pio2_hi = 0x1.921fb54442d18p+0;
523     constexpr f64 pio2_lo = 0x1.1a62633145c07p-54;
524     if ( ax < 0.5 ) {
525       if ( ax < 0x1.0p-57 ) return pio2_hi + pio2_lo;
526       return pio2_hi - (asin_f64(x) - pio2_lo);
527     }
528     if ( manip::signbit(x) ) {
529       constexpr f64 pi_hi = 0x1.921fb54442d18p+1;
530       constexpr f64 pi_lo = 0x1.1a62633145c07p-53;
531       f64 w = (1.0 + x) * 0.5;
532       f64 s = sqrt_ns::sqrt<f64>(w);
533       f64 r = asin_f64(s);
534       return pi_hi - (2.0 * r - pi_lo);
535     }
536     f64 w = (1.0 - x) * 0.5;
537     f64 s = sqrt_ns::sqrt<f64>(w);
538     return 2.0 * asin_f64(s);
539   }

541   [[nodiscard, gnu::always_inline]] inline constexpr f32
542   ksin_f32(f32 r) noexcept
543   {
544     using namespace coeff::sin_f32_data;
545     const f32 z = r * r;
546     const f32 r3 = z * r;
547     f32 p = hw::fmadd_ss(S4, z, S3);
548     p = hw::fmadd_ss(p, z, S2);
549     p = hw::fmadd_ss(p, z, S1);
550     return r + r3 * p;
551   }

553   [[nodiscard, gnu::always_inline]] inline constexpr f32
554   kcos_f32(f32 r) noexcept
555   {
556     using namespace coeff::sin_f32_data;
557     const f32 z = r * r;
558     const f32 hz = 0.5f * z;
559     f32 p = hw::fmadd_ss(C4, z, C3);
560     p = hw::fmadd_ss(p, z, C2);
561     p = hw::fmadd_ss(p, z, C1);
562     return (1.0f - hz) + z * z * p;
563   }

565   [[nodiscard, gnu::flatten]] inline constexpr f32
566   sin_f32(f32 x) noexcept
567   {
568     if ( ieee::is_nan(x) ) [[unlikely]]
569       return x;
570     if ( ieee::is_inf(x) ) [[unlikely]]
571       return ieee::qnan_v<f32>();
572     if ( manip::fabs(x) < 0x1.0p-12f ) [[unlikely]]
573       return x;
574     f64 rd;
575     const int q = __rr::reduce_pio2(f64(x), &rd);
576     const f32 r = f32(rd);
577     switch ( q ) {
578     case 0:
579       return ksin_f32(r);
580     case 1:
581       return kcos_f32(r);
582     case 2:
583       return -ksin_f32(r);
584     default:
585       return -kcos_f32(r);
586     }
587   }

589   [[nodiscard, gnu::flatten]] inline constexpr f32
590   cos_f32(f32 x) noexcept
591   {
592     if ( ieee::is_nan(x) ) [[unlikely]]
593       return x;
594     if ( ieee::is_inf(x) ) [[unlikely]]
595       return ieee::qnan_v<f32>();
596     if ( manip::fabs(x) < 0x1.0p-12f ) [[unlikely]]
597       return 1.0f;
598     f64 rd;
599     const int q = __rr::reduce_pio2(f64(x), &rd);
600     const f32 r = f32(rd);
601     switch ( q ) {
602     case 0:
603       return kcos_f32(r);
604     case 1:
605       return -ksin_f32(r);
606     case 2:
607       return -kcos_f32(r);
608     default:
609       return ksin_f32(r);
610     }
611   }

613   [[gnu::flatten]] inline constexpr void
614   sincos_f32(f32 x, f32 *s, f32 *c) noexcept
615   {
616     if ( ieee::is_nan(x) || ieee::is_inf(x) ) [[unlikely]] {
617       *s = ieee::qnan_v<f32>();
618       *c = ieee::qnan_v<f32>();
619       return;
620     }
621     if ( manip::fabs(x) < 0x1.0p-12f ) [[unlikely]] {
622       *s = x;
623       *c = 1.0f;
624       return;
625     }
626     f64 rd;
627     const int q = __rr::reduce_pio2(f64(x), &rd);
628     const f32 r = f32(rd);
629     const f32 ks = ksin_f32(r);
630     const f32 kc = kcos_f32(r);
631     switch ( q ) {
632     case 0:
633       *s = ks;
634       *c = kc;
635       return;
636     case 1:
637       *s = kc;
638       *c = -ks;
639       return;
640     case 2:
641       *s = -ks;
642       *c = -kc;
643       return;
644     default:
645       *s = -kc;
646       *c = ks;
647       return;
648     }
649   }

651   [[nodiscard, gnu::flatten]] inline constexpr f32
652   tan_f32(f32 x) noexcept
653   {
654     if ( ieee::is_nan(x) || ieee::is_inf(x) ) [[unlikely]]
655       return ieee::qnan_v<f32>();
656     if ( manip::fabs(x) < 0x1.0p-12f ) [[unlikely]]
657       return x;
658     f64 rd;
659     const int q = __rr::reduce_pio2(f64(x), &rd);
660     const f32 r = f32(rd);
661     const f32 ks = ksin_f32(r);
662     const f32 kc = kcos_f32(r);
663     return (q & 1) ? -kc / ks : ks / kc;
664   }
