38   template<ieee754_floating F>
39   [[nodiscard, gnu::always_inline]] inline constexpr F
40   sqrt(F x) noexcept
41   {
42 #if defined(__micron_arch_wasm_any)
43     if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::sqrt(x);
44 #endif
45     if ( ieee::is_nan(x) ) return x;
46     if ( x < F(0) ) return ieee::qnan_v<F>();
47     if ( x == F(0) ) return x;
48     return hw::sqrt<F>(x);
49   }

51   template<ieee754_floating F, policy::policy_tag P = policy::default_tag<F>>
52   [[nodiscard, gnu::always_inline]] inline constexpr F
53   rsqrt(F x, P = {}) noexcept
54   {
55 #if defined(__micron_arch_wasm_any)
56     policy::__validate_extended<F, P>();
57 #endif
58 #if defined(__micron_arch_wasm_any)
59     if constexpr ( sizeof(F) == 16 ) return F(1) / micron::wasm::math::sqrt(x);
60 #endif
61     if ( ieee::is_nan(x) ) return x;
62     if ( x < F(0) ) return ieee::qnan_v<F>();
63     if ( x == F(0) ) return ieee::inf_v<F>(0);
64     if constexpr ( micron::is_same_v<P, policy::fast_tag> && sizeof(F) == 4 ) {
65       f32 r = hw::rsqrt_approx_ss(f32(x));
66       f32 h = f32(x) * 0.5f;
67       r = r * (1.5f - h * r * r);
68       r = r * (1.5f - h * r * r);
69       return F(r);
70     }
71     return F(1) / hw::sqrt<F>(x);
72   }

77   [[nodiscard, gnu::always_inline]] inline constexpr f64
78   seed_f64(f64 x) noexcept
79   {
80     u64 b = ieee::to_bits(x);
81     u64 sign = b & ieee::traits<f64>::sign_mask;
82     u64 abs_b = b & ~ieee::traits<f64>::sign_mask;
83     abs_b = abs_b / 3 + 0x2A9F76253119D328ULL;
84     return ieee::from_bits<f64>(sign | abs_b);
85   }
86 
87   [[nodiscard, gnu::always_inline]] inline constexpr f32
88   seed_f32(f32 x) noexcept
89   {
90     u32 b = ieee::to_bits(x);
91     u32 sign = b & ieee::traits<f32>::sign_mask;
92     u32 abs_b = b & ~ieee::traits<f32>::sign_mask;
93     abs_b = abs_b / 3 + 0x2A511CC0u;
94     return ieee::from_bits<f32>(sign | abs_b);
95   }

99   template<ieee754_floating F>
100   [[nodiscard]] inline constexpr F
101   cbrt(F x) noexcept
102   {
103 #if defined(__micron_arch_wasm_any)
104     if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::cbrt(x);
105 #endif
106     if ( ieee::is_nan(x) || ieee::is_inf(x) || x == F(0) ) return x;
107     // NOTE: the iteration forms a**3, so the argument is first pulled into [1,8)
108     const int ex = manip::ilogb<F>(x);
109     int k = ex / 3;
110     if ( ex % 3 < 0 ) --k;
111     const F m = manip::ldexp<F>(manip::fabs(x), -3 * k);
112     F a;
113     if constexpr ( sizeof(F) == 4 )
114       a = F(__cbrt_impl::seed_f32(f32(m)));
115     else
116       a = F(__cbrt_impl::seed_f64(f64(m)));
117     // halley's iteration: a * (a**3 + 2m) / (2*(a**3) + m)
118     for ( int i = 0; i < (sizeof(F) == 4 ? 2 : 3); ++i ) {
119       F a3 = a * a * a;
120       a = a * (a3 + F(2) * m) / (F(2) * a3 + m);
121     }
122     return manip::copysign<F>(manip::ldexp<F>(a, k), x);
123   }

125   template<ieee754_floating F>
126   [[nodiscard]] inline constexpr F
127   hypot(F a, F b) noexcept
128   {
129 #if defined(__micron_arch_wasm_any)
130     if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::hypot(a, b);
131 #endif
132     if ( ieee::is_inf(a) || ieee::is_inf(b) ) return ieee::inf_v<F>(0);
133     if ( ieee::is_nan(a) || ieee::is_nan(b) ) return ieee::qnan_v<F>();
134     a = manip::fabs(a);
135     b = manip::fabs(b);
136     if ( a == F(0) ) return b;
137     if ( b == F(0) ) return a;
138 
139     F m = (a > b) ? a : b;
140     int e = manip::ilogb<F>(m);
141     if constexpr ( sizeof(F) == 8 ) {
142       F sa = manip::ldexp<F>(a, -e);
143       F sb = manip::ldexp<F>(b, -e);
144       auto a2 = dd::two_prod(f64(sa), f64(sa));
145       auto b2 = dd::two_prod(f64(sb), f64(sb));
146       auto s = dd::add(a2, b2);
147       F root = F(hw::sqrt_sd(f64(s.hi)));
148       auto r2 = dd::two_prod(f64(root), f64(root));
149       f64 resid_hi = (s.hi - r2.hi) - r2.lo + s.lo;
150       F corr = F(resid_hi / (f64(2) * f64(root)));
151       root = F(root + corr);
152       return manip::ldexp<F>(root, e);
153     } else {
154       F sa = manip::ldexp<F>(a, -e);
155       F sb = manip::ldexp<F>(b, -e);
156       F s = sa * sa + sb * sb;
157       return manip::ldexp<F>(F(hw::sqrt<F>(s)), e);
158     }
159   }
