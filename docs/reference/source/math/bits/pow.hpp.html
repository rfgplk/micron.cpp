37   template<ieee754_floating F>
38   [[nodiscard, gnu::always_inline]] inline constexpr int
39   classify_integer(F y) noexcept
40   {
41     if ( ieee::is_nan(y) || ieee::is_inf(y) ) return 0;
42     using T = ieee::traits<F>;
43     using U = typename T::uint_type;
44     U bits = ieee::to_bits(manip::fabs(y));
45     int e = int((bits & T::exp_mask) >> T::mant_bits) - T::exp_bias;
46     if ( e < 0 ) return 0;
47     if ( e > int(T::mant_bits) ) return 1;
48     U frac_mask = (U(1) << (T::mant_bits - e)) - 1;
49     if ( (bits & T::mant_mask & frac_mask) != 0 ) return 0;
50     if ( e == 0 ) return 2;
51     return ((bits >> (T::mant_bits - e)) & 1) ? 2 : 1;
52   }

54   template<ieee754_floating F>
55   [[nodiscard, gnu::flatten]] inline constexpr F
56   pow_int(F x, long long n) noexcept
57   {
58     if ( n == 0 ) return F(1);
59     bool neg = n < 0;
60     unsigned long long un = neg ? static_cast<unsigned long long>(-n) : static_cast<unsigned long long>(n);
61     F base = x;
62     F r = F(1);
63     while ( un > 0 ) {
64       if ( un & 1 ) r = r * base;
65       base = base * base;
66       un >>= 1;
67     }
68     return neg ? F(F(1) / r) : r;
69   }

73   // NOTE: full IEEE 754-2019
74   template<ieee754_floating F>
75   [[nodiscard, gnu::flatten]] inline constexpr F
76   pow(F x, F y) noexcept
77   {
78 #if defined(__micron_arch_wasm_any)
79     if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::pow(x, y);
80 #endif
81     if ( y == F(0) ) return F(1);
82     if ( x == F(1) ) return F(1);
83     if ( ieee::is_nan(x) || ieee::is_nan(y) ) return ieee::qnan_v<F>();
84 
85     int yint = __impl::classify_integer(y);
86     bool x_neg = manip::signbit(x);
87     F ax = manip::fabs(x);
88 
89     if ( ieee::is_inf(y) ) {
90       if ( ax == F(1) ) return F(1);
91       bool y_pos = !manip::signbit(y);
92       bool x_lt_1 = (ax < F(1));
93       return (y_pos == x_lt_1) ? F(0) : ieee::inf_v<F>(0);
94     }
95 
96     if ( x == F(0) ) {
97       bool y_neg = manip::signbit(y);
98       if ( yint == 2 ) {
99         if ( y_neg ) return ieee::inf_v<F>(x_neg ? 1 : 0);
100         return x;
101       }
102       if ( y_neg ) return ieee::inf_v<F>(0);
103       return F(0);
104     }
105 
106     if ( ieee::is_inf(x) ) {
107       bool y_neg = manip::signbit(y);
108       if ( x_neg ) {
109         if ( yint == 2 ) {
110           return y_neg ? F(-0.0) : ieee::inf_v<F>(1);
111         }
112         return y_neg ? F(0) : ieee::inf_v<F>(0);
113       }
114       return y_neg ? F(0) : ieee::inf_v<F>(0);
115     }
116 
117     if ( x_neg && yint == 0 ) return ieee::qnan_v<F>();
118 
119     if ( yint != 0 ) {
120       constexpr F int_path_max = F(2147483648.0);
121       if ( manip::fabs(y) <= int_path_max ) {
122         long long n = static_cast<long long>(y);
123         F r = __impl::pow_int<F>(ax, n);
124         return (x_neg && yint == 2) ? F(-r) : r;
125       }
126     }
127 
128     F r;
129     if constexpr ( sizeof(F) == 8 ) {
130       f64 lx = log_ns::log_f64(f64(ax));
131       auto p = dd::two_prod(f64(y), lx);
132       f64 ehi = exp_ns::exp_f64(f64(p.hi));
133       r = (ieee::is_inf(ehi) || ehi == f64(0)) ? F(ehi) : F(hw::fmadd_sd(ehi, f64(p.lo), ehi));
134     } else {
135       f64 lx = f64(log_ns::log_f32(f32(ax)));
136       auto p = dd::two_prod(f64(y), lx);
137       f64 ehi = exp_ns::exp_f64(f64(p.hi));
138       r = (ieee::is_inf(ehi) || ehi == f64(0)) ? F(ehi) : F(hw::fmadd_sd(ehi, f64(p.lo), ehi));
139     }
140     return (x_neg && yint == 2) ? F(-r) : r;
141   }
