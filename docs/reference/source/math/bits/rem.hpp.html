32 template<bool Quo, ieee754_floating F>
33 [[nodiscard]] inline constexpr F
34 __fmod_core(F x, F y, u32 &quo) noexcept
35 {
36   using T = ieee::traits<F>;
37   using U = typename T::uint_type;
38 
39   if constexpr ( Quo ) quo = 0;
40   if ( ieee::is_nan(x) || ieee::is_nan(y) || ieee::is_inf(x) || y == F(0) ) return ieee::qnan_v<F>();
41   if ( ieee::is_inf(y) ) return x;
42   if ( x == F(0) ) return x;
43 
44   F ax = manip::fabs(x);
45   F ay = manip::fabs(y);
46   if ( ax < ay ) return x;
47   if ( ax == ay ) {
48     if constexpr ( Quo ) quo = 1;
49     return manip::copysign<F>(F(0), x);
50   }
51 
52   int ex = manip::ilogb<F>(ax);
53   int ey = manip::ilogb<F>(ay);
54 
55   U bx = ieee::to_bits(ax);
56   U by = ieee::to_bits(ay);
57   U mx = bx & T::mant_mask;
58   U my = by & T::mant_mask;
59   if ( ((bx & T::exp_mask) >> T::mant_bits) == 0 ) {
60     while ( (mx & T::implicit_one) == 0 ) mx <<= 1;
61   } else {
62     mx |= T::implicit_one;
63   }
64   if ( ((by & T::exp_mask) >> T::mant_bits) == 0 ) {
65     while ( (my & T::implicit_one) == 0 ) my <<= 1;
66   } else {
67     my |= T::implicit_one;
68   }
69 
70   int diff = ex - ey;
71   for ( int i = 0; i < diff; ++i ) {
72     if constexpr ( Quo ) quo <<= 1;
73     if ( mx >= my ) {
74       mx -= my;
75       if constexpr ( Quo ) quo |= 1u;
76     }
77     mx <<= 1;
78   }
79   if constexpr ( Quo ) quo <<= 1;
80   if ( mx >= my ) {
81     mx -= my;
82     if constexpr ( Quo ) quo |= 1u;
83   }
84 
85   if ( mx == 0 ) return manip::copysign<F>(F(0), x);
86 
87   int shift = 0;
88   while ( (mx & T::implicit_one) == 0 ) {
89     mx <<= 1;
90     ++shift;
91   }
92   int new_exp = ey - shift;
93   if ( new_exp < -T::exp_bias + 1 ) {
94     int subnorm_shift = (-T::exp_bias + 1) - new_exp;
95     mx >>= subnorm_shift;
96     U bits = mx & T::mant_mask;
97     F r = ieee::from_bits<F>(bits);
98     return manip::copysign<F>(r, x);
99   }
100   U packed = (U(new_exp + T::exp_bias) << T::mant_bits) | (mx & T::mant_mask);
101   F r = ieee::from_bits<F>(packed);
102   return manip::copysign<F>(r, x);
103 }

105 // x - n*y with n = x/y rounded to nearest, ties to even
106 template<ieee754_floating F>
107 [[nodiscard]] inline constexpr F
108 __remainder_quo(F x, F y, u32 &quo) noexcept
109 {
110   F r = __fmod_core<true, F>(x, y, quo);
111   const F ay = manip::fabs(y);
112   const F ar = manip::fabs(r);
113   const F ar2 = ar + ar;
114   if ( ar2 > ay || (ar2 == ay && (quo & 1u) != 0u) ) {
115     r = (r > F(0)) ? F(r - ay) : F(r + ay);
116     ++quo;
117   }
118   return r;
119 }

123 fmod(F x, F y) noexcept
124 {
125 #if defined(__micron_arch_wasm_any)
126   if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::fmod(x, y);

132 template<ieee754_floating F>
133 [[nodiscard]] inline constexpr F
134 remainder(F x, F y) noexcept
135 {
136 #if defined(__micron_arch_wasm_any)
137   if constexpr ( sizeof(F) == 16 ) return micron::wasm::math::remainder(x, y);
138 #endif
139   if ( ieee::is_nan(x) || ieee::is_nan(y) || ieee::is_inf(x) || y == F(0) ) return ieee::qnan_v<F>();
140   if ( ieee::is_inf(y) ) return x;
141 
142   u32 quo = 0;
143   return __remainder_quo<F>(x, y, quo);
144 }

146 template<ieee754_floating F>
147 [[nodiscard]] inline constexpr F
148 remquo(F x, F y, int *q) noexcept
149 {
150   if ( ieee::is_nan(x) || ieee::is_nan(y) || ieee::is_inf(x) || y == F(0) ) {
151     *q = 0;
152     return ieee::qnan_v<F>();
153   }
154   if ( ieee::is_inf(y) ) {
155     *q = 0;
156     return x;
157   }
158 
159   u32 quo = 0;
160   const F r = __remainder_quo<F>(x, y, quo);
161   const int n = int(quo & 7u);
162   *q = (manip::signbit(x) != manip::signbit(y)) ? -n : n;
163   return r;
164 }
