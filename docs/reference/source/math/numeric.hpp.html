20 // c[0] constant term, c[N-1] highest order
21 template<ieee754_floating F>
22 [[nodiscard, gnu::flatten]] inline constexpr F
23 polyval(const F *coeffs, usize n, F x) noexcept
24 {
25   if ( n == 0 ) return F(0);
26   F r = coeffs[n - 1];
27   for ( usize i = n - 1; i-- > 0; ) r = math::fma<F>(r, x, coeffs[i]);
28   return r;
29 }

61 template<ieee754_floating F>
62 [[gnu::flatten]] inline constexpr void
63 gradient(const F *__restrict__ y, usize n, F *__restrict__ g) noexcept
64 {
65   if ( n == 0 ) return;
66   if ( n == 1 ) {
67     g[0] = F(0);
68     return;
69   }
70   g[0] = y[1] - y[0];
71   for ( usize i = 1; i + 1 < n; ++i ) g[i] = F(0.5) * (y[i + 1] - y[i - 1]);
72   g[n - 1] = y[n - 1] - y[n - 2];
73 }

75 template<ieee754_floating F>
76 [[gnu::flatten]] inline void
77 gradient(const F *__restrict__ x, const F *__restrict__ y, usize n, F *__restrict__ g) noexcept
78 {
79   if ( n == 0 ) return;
80   if ( n == 1 ) {
81     g[0] = F(0);
82     return;
83   }
84   g[0] = (y[1] - y[0]) / (x[1] - x[0]);
85   for ( usize i = 1; i + 1 < n; ++i ) g[i] = (y[i + 1] - y[i - 1]) / (x[i + 1] - x[i - 1]);
86   g[n - 1] = (y[n - 1] - y[n - 2]) / (x[n - 1] - x[n - 2]);
87 }

97 template<ieee754_floating F>
98 [[nodiscard, gnu::flatten]] inline constexpr F
99 interp(F xq, const F *x, const F *y, usize n) noexcept
100 {
101   if ( n == 0 ) return F(0);
102   if ( n == 1 || xq <= x[0] ) return y[0];
103   if ( xq >= x[n - 1] ) return y[n - 1];
104   usize lo = 0, hi = n - 1;
105   while ( hi - lo > 1 ) {
106     const usize mid = lo + (hi - lo) / 2;
107     if ( x[mid] <= xq )
108       lo = mid;
109     else
110       hi = mid;
111   }
112   const F t = (xq - x[lo]) / (x[lo + 1] - x[lo]);
113   return math::fma<F>(t, y[lo + 1] - y[lo], y[lo]);
114 }

124 // non FFT based
125 template<typename T>
126 [[gnu::flatten]] inline constexpr void
127 convolve(const T *__restrict__ a, usize n, const T *__restrict__ b, usize k, T *__restrict__ out) noexcept
128 {
129   for ( usize i = 0; i + 0 < n + k - 1; ++i ) {
130     T s{ 0 };
131     const usize j_lo = (i + 1 > k) ? (i + 1 - k) : 0;
132     const usize j_hi = (i + 1 < n) ? (i + 1) : n;
133     for ( usize j = j_lo; j < j_hi; ++j ) s = s + a[j] * b[i - j];
134     out[i] = s;
135   }
136 }

138 template<typename T>
139 [[gnu::flatten]] inline constexpr void
140 correlate(const T *__restrict__ a, usize n, const T *__restrict__ b, usize k, T *__restrict__ out) noexcept
141 {
142   for ( usize i = 0; i + 0 < n + k - 1; ++i ) {
143     T s{ 0 };
144     const usize j_lo = (i + 1 > k) ? (i + 1 - k) : 0;
145     const usize j_hi = (i + 1 < n) ? (i + 1) : n;
146     for ( usize j = j_lo; j < j_hi; ++j ) s = s + a[j] * b[k - 1 - (i - j)];
147     out[i] = s;
148   }
149 }

165 template<typename T>
166   requires(micron::is_integral_v<T>)
167 [[nodiscard, gnu::always_inline]] inline constexpr T
168 lcm(T a, T b) noexcept
169 {
170   if ( a == T(0) || b == T(0) ) return T(0);
171   const T g = math::gcd<T>(a, b);
172   T av = (a < T(0)) ? T(-a) : a;
173   T bv = (b < T(0)) ? T(-b) : b;
174   return T((av / g) * bv);
175 }

184 template<typename T>
185   requires(micron::is_integral_v<T>)
186 [[nodiscard, gnu::always_inline]] inline constexpr divmod_result<T>
187 divmod(T a, T b) noexcept
188 {
189   T q = a / b;
190   T r = a - q * b;
191   if constexpr ( micron::is_signed_v<T> ) {
192     if ( (r != T(0)) && ((r < T(0)) != (b < T(0))) ) {
193       q = q - T(1);
194       r = r + b;
195     }
196   }
197   return { q, r };
198 }

223 template<typename T = u64>
224   requires(micron::is_integral_v<T>)
225 [[nodiscard, gnu::flatten]] inline constexpr T
226 comb(int n, int k) noexcept
227 {
228   if ( k < 0 || k > n ) return T(0);
229   if ( k > n - k ) k = n - k;
230   T r{ 1 };
231   for ( int i = 0; i < k; ++i ) {
232     T p{};
233     if ( !__builtin_mul_overflow(r, T(n - i), &p) ) {
234       r = T(p / T(i + 1));
235       continue;
236     }
237     // C(n,i) * (n - i) can exceed T when C(n,i+1) does not; cancel the divisor before multiplying
238     const T g = math::gcd<T>(r, T(i + 1));
239     r = T(T(r / g) * T(T(n - i) / T(T(i + 1) / g)));
240   }
241   return r;
242 }

267 template<ieee754_floating F>
268 [[nodiscard, gnu::flatten]] inline constexpr usize
269 digitize(F x, const F *edges, usize n_edges) noexcept
270 {
271   if ( n_edges == 0 ) return 0;
272   if ( x < edges[0] ) return 0;
273   if ( x >= edges[n_edges - 1] ) return n_edges;
274   usize lo = 0, hi = n_edges - 1;
275   while ( hi - lo > 1 ) {
276     const usize mid = lo + (hi - lo) / 2;
277     if ( edges[mid] <= x )
278       lo = mid;
279     else
280       hi = mid;
281   }
282   return lo + 1;
283 }

285 template<ieee754_floating F>
286 [[gnu::flatten]] inline void
287 histogram(const F *__restrict__ x, usize n_x, F lo, F hi, u64 *__restrict__ out_counts, usize n_bins) noexcept
288 {
289   for ( usize i = 0; i < n_bins; ++i ) out_counts[i] = 0;
290   if ( n_bins == 0 || hi <= lo ) return;
291   const F inv_w = F(n_bins) / (hi - lo);
292   for ( usize i = 0; i < n_x; ++i ) {
293     const F v = x[i];
294     if ( v < lo || v >= hi ) continue;
295     usize b = usize((v - lo) * inv_w);
296     if ( b >= n_bins ) b = n_bins - 1;
297     ++out_counts[b];
298   }
299 }
