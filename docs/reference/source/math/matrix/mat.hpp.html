21 template<arith_scalar T, usize R, usize C>
22   requires(R >= 1 && C >= 1 && R * C * sizeof(T) <= 1024)
23 struct mat: public micron::int_matrix_base_avx<T, C, R> {
24   using base_type = micron::int_matrix_base_avx<T, C, R>;
25   using value_type = T;
26   static constexpr usize rows = R;
27   static constexpr usize cols = C;
28 
29   using base_type::base_type;
30 
31   [[nodiscard, gnu::always_inline]] constexpr T &
32   at(usize r, usize c) noexcept
33   {
34     return this->data[r * C + c];
35   }
36 
37   [[nodiscard, gnu::always_inline]] constexpr const T &
38   at(usize r, usize c) const noexcept
39   {
40     return this->data[r * C + c];
41   }
42 
43   [[nodiscard, gnu::always_inline]] constexpr T &
44   operator[](usize i) noexcept
45   {
46     return this->data[i];
47   }
48 
49   [[nodiscard, gnu::always_inline]] constexpr const T &
50   operator[](usize i) const noexcept
51   {
52     return this->data[i];
53   }
54 
55   using base_type::operator[];
56 
57   [[nodiscard]] static constexpr mat
58   identity() noexcept
59   {
60     mat m{};
61     for ( usize i = 0; i < R * C; ++i ) m.data[i] = T(0);
62     constexpr usize K = (R < C) ? R : C;
63     for ( usize i = 0; i < K; ++i ) m.data[i * C + i] = T(1);
64     return m;
65   }
66 
67   [[nodiscard]] static constexpr mat
68   zero() noexcept
69   {
70     mat m{};
71     for ( usize i = 0; i < R * C; ++i ) m.data[i] = T(0);
72     return m;
73   }
74 };

196 // real matrix (mat * mat) product
197 template<arith_scalar T, usize R, usize K, usize C>
198 [[nodiscard]] inline constexpr mat<T, R, C>
199 matmul(const mat<T, R, K> &a, const mat<T, K, C> &b) noexcept
200 {
201   mat<T, R, C> r{};
202   for ( usize i = 0; i < R; ++i )
203     for ( usize j = 0; j < C; ++j ) {
204       T s = T(0);
205       for ( usize k = 0; k < K; ++k ) s += a.data[i * K + k] * b.data[k * C + j];
206       r.data[i * C + j] = s;
207     }
208   return r;
209 }

211 template<arith_scalar T, usize R, usize C>
212   requires(R >= 2 && R <= 16 && C >= 2 && C <= 16)
213 [[nodiscard]] inline constexpr vec<T, R>
214 matvec(const mat<T, R, C> &m, const vec<T, C> &v) noexcept
215 {
216   vec<T, R> r{};
217   for ( usize i = 0; i < R; ++i ) {
218     T s = T(0);
219     for ( usize j = 0; j < C; ++j ) s += m.data[i * C + j] * v.data[j];
220     r.data[i] = s;
221   }
222   return r;
223 }
