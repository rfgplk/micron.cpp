47 // mkbits::rem::fmod ORs in the implicit one before its subnormal-normalisation loop
48 template<ieee754_floating F>
49 [[nodiscard]] inline F
50 fmod(F x, F y) noexcept
51 {
52   if ( !ieee::is_subnormal(y) ) return mkbits::rem::fmod<F>(x, y);
53   constexpr int k = ieee::traits<F>::mant_bits + 1;
54   const F ys = mkbits::manip::scalbn<F>(y, k);
55   const F r = mkbits::rem::fmod<F>(x, ys);
56   return mkbits::manip::scalbn<F>(mkbits::rem::fmod<F>(mkbits::manip::scalbn<F>(r, k), ys), -k);
57 }

59 template<ieee754_floating F>
60 [[nodiscard]] inline F
61 hypot(F x, F y) noexcept
62 {
63   F ax = mkbits::manip::fabs(x);
64   F ay = mkbits::manip::fabs(y);
65   if ( ieee::is_inf(ax) || ieee::is_inf(ay) ) return ieee::inf_v<F>();
66   if ( ieee::is_nan(ax) || ieee::is_nan(ay) ) return ieee::qnan_v<F>();
67   if ( ax < ay ) {
68     const F t = ax;
69     ax = ay;
70     ay = t;
71   }
72   if ( ay == F(0) ) return ax;
73 
74   const int e = mkbits::manip::ilogb<F>(ax);
75   const F sx = mkbits::manip::scalbn<F>(ax, -e);
76   const F sy = mkbits::manip::scalbn<F>(ay, -e);
77   const F q = F(sx * sx + sy * sy);
78   F r;
79   if constexpr ( sizeof(F) == 4 )
80     r = hw::sqrt_ss(q);
81   else
82     r = hw::sqrt_sd(q);
83   return mkbits::manip::scalbn<F>(r, e);
84 }

86 #if !defined(__MC_SHIM_HW_FMA)
87 // s + e == a + b exactly
88 template<ieee754_floating F>
89 [[gnu::always_inline]] inline void
90 two_sum(F a, F b, F &s, F &e) noexcept
91 {
92   s = hw::fp_barrier(F(a + b));
93   const F bp = hw::fp_barrier(F(s - a));
94   const F lo = hw::fp_barrier(F(a - hw::fp_barrier(F(s - bp))));
95   e = hw::fp_barrier(F(lo + hw::fp_barrier(F(b - bp))));
96 }

99 [[nodiscard]] inline f64
100 fma(f64 a, f64 b, f64 c) noexcept
101 {
102 #if defined(__MC_SHIM_HW_FMA)
103   return hw::fmadd_sd(a, b, c);
104 #else
105   const f64 p = hw::fp_barrier(f64(a * b));
106   const f64 pa = mkbits::manip::fabs(p);
107   // WARNING: dekker's split overflows above 2^996
108   if ( !(mkbits::manip::fabs(a) <= 0x1p996) || !(mkbits::manip::fabs(b) <= 0x1p996) || !(pa >= 0x1p-960) || !(pa <= 0x1p1000)
109        || !(mkbits::manip::fabs(c) <= 0x1p1000) )
110     return f64(p + c);
111 
112   constexpr f64 split = 0x1.0p27 + 1.0;
113   const f64 at = hw::fp_barrier(f64(split * a));
114   const f64 ah = hw::fp_barrier(f64(at - hw::fp_barrier(f64(at - a))));
115   const f64 al = hw::fp_barrier(f64(a - ah));
116   const f64 bt = hw::fp_barrier(f64(split * b));
117   const f64 bh = hw::fp_barrier(f64(bt - hw::fp_barrier(f64(bt - b))));
118   const f64 bl = hw::fp_barrier(f64(b - bh));
119 
120   f64 e = hw::fp_barrier(f64(hw::fp_barrier(f64(ah * bh)) - p));
121   e = hw::fp_barrier(f64(e + hw::fp_barrier(f64(ah * bl))));
122   e = hw::fp_barrier(f64(e + hw::fp_barrier(f64(al * bh))));
123   e = hw::fp_barrier(f64(e + hw::fp_barrier(f64(al * bl))));
124 
125   f64 sh, sl, vh, vl, zh, zl;
126   two_sum<f64>(p, c, sh, sl);        // sh + sl == p + c
127   two_sum<f64>(sl, e, vh, vl);       // vh + vl == sl + e
128   two_sum<f64>(sh, vh, zh, zl);      // zh + zl + vl == a * b + c, exactly
129   return f64(zh + hw::fp_barrier(f64(zl + vl)));
130 #endif
131 }

133 [[nodiscard]] inline f32
134 fma(f32 a, f32 b, f32 c) noexcept
135 {
136 #if defined(__MC_SHIM_HW_FMA)
137   return hw::fmadd_ss(a, b, c);
138 #else
139   // 24 + 24 significand bits, and f32's exponent range cannot over- or underflow f64, so the
140   // product is exact and the sum below rounds once -- into f64
141   const f64 p = hw::fp_barrier(f64(f64(a) * f64(b)));
142   if ( !ieee::is_finite(p) || !ieee::is_finite(f64(c)) ) return f32(p + f64(c));
143 
144   f64 sh, sl;
145   two_sum<f64>(p, f64(c), sh, sl);
146   if ( sl != 0.0 && mkbits::manip::fabs(sh) >= 0x1p-126 ) {
147     // sh sits on an f32 halfway point, where the cast below would round on a tie the exact
148     // residual sl has already decided; step sh off it in sl's direction
149     u64 u = ieee::to_bits(sh);
150     if ( (u & 0x1fffffffULL) == 0x10000000ULL ) u += ((sl < 0.0) == (sh < 0.0)) ? u64(1) : ~u64(0);
151     sh = ieee::from_bits<f64>(u);
152   }
153   return f32(sh);
154 #endif
155 }
