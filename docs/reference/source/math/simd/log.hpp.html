179   [[gnu::flatten]] inline simd::f128
180   log(simd::f128 x) noexcept
181   {
182     using namespace mkbits::coeff::log_f32_data;
183     const uint32x4_t subn = simd::neon::lt(x, simd::neon::splat_f32(0x1.0p-126f));
184     const float32x4_t xn = simd::neon::select(subn, simd::neon::mul(x, simd::neon::splat_f32(0x1.0p25f)), x);
185     const int32x4_t xi = simd::neon::reinterpret_s32_from_f32(xn);
186 
187     int32x4_t k
188         = simd::neon::sub(simd::neon::and_(simd::neon::shr_arith_i32<23>(xi), simd::neon::splat_i32(0xFF)), simd::neon::splat_i32(127));
189     k = simd::neon::sub(k, simd::neon::and_(simd::neon::reinterpret_s32_from_u32(subn), simd::neon::splat_i32(25)));
190 
191     const int32x4_t mant_mask = simd::neon::splat_i32(0x007FFFFF);
192     const int32x4_t bias = simd::neon::splat_i32(0x3F800000);
193     float32x4_t m = simd::neon::reinterpret_f32_from_s32(simd::neon::or_(simd::neon::and_(xi, mant_mask), bias));
194 
195     // m is rebuilt with the implicit bit forced on, so it lands in [1,2) and the split point the
196     // Lg1..Lg4 fit assumes is sqrt(2), not the sqrt(1/2) frexp() hands the scalar twin
197     const float32x4_t sqrt_two = simd::neon::splat_f32(0x1.6a09e6p+0f);
198     const uint32x4_t ge2 = simd::neon::ge(m, sqrt_two);
199     m = simd::neon::select(ge2, simd::neon::mul(m, simd::neon::splat_f32(0.5f)), m);
200     k = simd::neon::add(k, simd::neon::and_(simd::neon::reinterpret_s32_from_u32(ge2), simd::neon::splat_i32(1)));
201     const float32x4_t f = simd::neon::sub(m, simd::neon::splat_f32(1.0f));
202 #if defined(__micron_arch_arm64)
203     const float32x4_t s = simd::neon::div(f, simd::neon::add(simd::neon::splat_f32(2.0f), f));
204 #else
205 
206     const float32x4_t denom = simd::neon::add(simd::neon::splat_f32(2.0f), f);
207     float32x4_t recip = simd::neon::rcp_est(denom);
208     recip = simd::neon::mul(recip, simd::neon::rcp_step(denom, recip));
209     recip = simd::neon::mul(recip, simd::neon::rcp_step(denom, recip));
210     const float32x4_t s = simd::neon::mul(f, recip);
211 #endif
212     const float32x4_t z = simd::neon::mul(s, s);
213     const float32x4_t w = simd::neon::mul(z, z);
214 #if defined(__micron_arm_fma) || defined(__micron_arch_arm64)
215     float32x4_t t1 = simd::neon::fma_f32(simd::neon::splat_f32(Lg2), simd::neon::splat_f32(Lg4), w);
216     t1 = simd::neon::mul(t1, w);
217     float32x4_t t2 = simd::neon::fma_f32(simd::neon::splat_f32(Lg1), simd::neon::splat_f32(Lg3), w);
218     t2 = simd::neon::mul(t2, z);
219 #else
220     float32x4_t t1 = simd::neon::mul(simd::neon::add(simd::neon::splat_f32(Lg2), simd::neon::mul(simd::neon::splat_f32(Lg4), w)), w);
221     float32x4_t t2 = simd::neon::mul(simd::neon::add(simd::neon::splat_f32(Lg1), simd::neon::mul(simd::neon::splat_f32(Lg3), w)), z);
222 #endif
223     const float32x4_t R = simd::neon::add(t1, t2);
224     const float32x4_t hfsq = simd::neon::mul(simd::neon::splat_f32(0.5f), simd::neon::mul(f, f));
225     const float32x4_t log_m = simd::neon::add(simd::neon::mul(s, simd::neon::add(hfsq, R)), simd::neon::sub(f, hfsq));
226     const float32x4_t fk = simd::neon::convert_i32_to_f32(k);
227 #if defined(__micron_arm_fma) || defined(__micron_arch_arm64)
228     float32x4_t y = simd::neon::fma_f32(simd::neon::fma_f32(log_m, fk, simd::neon::splat_f32(ln2_lo)), fk, simd::neon::splat_f32(ln2_hi));
229 #else
230     float32x4_t y = simd::neon::add(
231         simd::neon::add(simd::neon::mul(fk, simd::neon::splat_f32(ln2_hi)), simd::neon::mul(fk, simd::neon::splat_f32(ln2_lo))), log_m);
232 #endif
233     const float32x4_t zero = simd::neon::splat_f32(0.0f);
234     y = simd::neon::select(simd::neon::eq(x, simd::neon::splat_f32(ieee::inf_v<f32>(0))), x, y);
235     // ge() is false for NaN, so the negative and NaN lanes fall through the same select
236     const float32x4_t qn = simd::neon::reinterpret_f32_from_u32(simd::neon::or_(
237         simd::neon::reinterpret_u32_from_f32(x), simd::neon::reinterpret_u32_from_f32(simd::neon::splat_f32(ieee::qnan_v<f32>()))));
238     y = simd::neon::select(simd::neon::ge(x, zero), y, qn);
239     return simd::neon::select(simd::neon::eq(x, zero), simd::neon::splat_f32(ieee::inf_v<f32>(1)), y);
240   }

291     const float64x2_t zero = simd::neon::splat_f64(0.0);
292     const uint64x2_t nneg = simd::neon::ge(x, zero);
293     // nothing exceeds +inf, so ge() against it IS the equality test; the alias layer has no eq() for f64 lanes
294     y = simd::neon::select(simd::neon::ge(x, simd::neon::splat_f64(ieee::inf_v<f64>(0))), x, y);
295     const float64x2_t qn = simd::neon::reinterpret_f64_from_u64(simd::neon::or_(
296         simd::neon::reinterpret_u64_from_f64(x), simd::neon::reinterpret_u64_from_f64(simd::neon::splat_f64(ieee::qnan_v<f64>()))));
297     y = simd::neon::select(nneg, y, qn);
298     return simd::neon::select(simd::neon::and_(nneg, simd::neon::le(x, zero)), simd::neon::splat_f64(ieee::inf_v<f64>(1)), y);
299   }
