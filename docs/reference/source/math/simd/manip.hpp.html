215   [[gnu::always_inline]] inline simd::f128
216   copysign(simd::f128 mag, simd::f128 sgn) noexcept
217   {
218     const uint32x4_t sm = simd::neon::splat_u32(0x80000000u);
219     const uint32x4_t mb = simd::neon::bic(simd::neon::reinterpret_u32_from_f32(mag), sm);
220     const uint32x4_t sb = simd::neon::and_(simd::neon::reinterpret_u32_from_f32(sgn), sm);
221     return simd::neon::reinterpret_f32_from_u32(simd::neon::or_(mb, sb));
222   }
