8 // per iteration:
9 //   ys = y * 2^-i          (FP multiply by 2^-i = exponent decrement = shift)
10 //   xs = x * 2^-i
11 //   sb = sign_mask(z) & -0.0
12 //   x -= ys ^ sb           (subtract if z>=0, add if z<0)
13 //   y += xs ^ sb           (add if z>=0, subtract if z<0)
14 //   z -= atan_i ^ sb       (subtract if z>=0, add if z<0)

57   [[gnu::flatten]] inline simd::d256
58   reduce_pio2_scalar_d256(simd::d256 x, simd::i256 *q_out) noexcept
59   {
60     alignas(32) f64 xs[4];
61     alignas(32) f64 rs[4];
62     alignas(32) i64 qs[4];
63     simd::avx::storeu_f64(reinterpret_cast<double *>(xs), x);
64     for ( usize i = 0; i < 4; ++i ) {
65       if ( !ieee::is_finite(xs[i]) ) [[unlikely]] {
66         rs[i] = 0.0;
67         qs[i] = 0;
68         continue;
69       }
70       // cody_waite's leading x - fN * pio2_hi is exact only while fN fits in 20 bits, since pio2_hi
71       // carries the other 33; above that the full payne_hanek reduction is the one that holds
72       qs[i] = (mkbits::manip::fabs(xs[i]) < 0x1.0p20) ? mkbits::trig_ns::__rr::cody_waite(xs[i], &rs[i])
73                                                       : mkbits::trig_ns::__rr::__paynehanek::payne_hanek(xs[i], &rs[i]);
74     }
75     *q_out = simd::avx::loadu_i256(reinterpret_cast<const __m256i_u *>(qs));
76     return simd::avx::loadu_f64(reinterpret_cast<const double *>(rs));
77   }

79   [[gnu::always_inline]] inline simd::d256
80   reduce_pio2_d256(simd::d256 x, simd::i256 *q_out) noexcept
81   {
82     const simd::d256 ax = simd::avx::andnot_f64(simd::avx::splat_f64(-0.0), x);
83     if ( simd::avx::movemask_f64(simd::avx::cmp_f64<_CMP_LT_OQ>(ax, simd::avx::splat_f64(0x1.0p33))) != 0xF ) [[unlikely]]
84       return reduce_pio2_scalar_d256(x, q_out);
85 
86     const simd::d256 fN
87         = simd::avx::round_f64<_MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC>(simd::avx::mul_f64(x, simd::avx::splat_f64(inv_pio2)));
88     simd::d256 t = simd::fma::fnma_f64(fN, simd::avx::splat_f64(pio2_hi), x);
89     t = simd::fma::fnma_f64(fN, simd::avx::splat_f64(pio2_mid), t);
90     t = simd::fma::fnma_f64(fN, simd::avx::splat_f64(pio2_lo), t);
91     *q_out = simd::avx2::and_i256(simd::avx::cast_f64_to_i256(simd::avx::add_f64(fN, simd::avx::splat_f64(0x1.8p52))),
92                                   simd::avx::splat_i64(3));
93     return t;
94   }

96   [[gnu::always_inline]] inline simd::d256
97   guard_d256(simd::d256 x, simd::d256 y, simd::d256 small) noexcept
98   {
99     const simd::d256 ax = simd::avx::andnot_f64(simd::avx::splat_f64(-0.0), x);
100     y = simd::avx::blendv_f64(y, small, simd::avx::cmp_f64<_CMP_LT_OQ>(ax, simd::avx::splat_f64(0x1.0p-26)));
101     return simd::avx::blendv_f64(y, simd::avx::splat_f64(ieee::qnan_v<f64>()),
102                                  simd::avx::cmp_f64<_CMP_NLT_UQ>(ax, simd::avx::splat_f64(ieee::inf_v<f64>(0))));
103   }

105   [[gnu::flatten]] inline void
106   sincos_kernel_d256(simd::d256 z, simd::d256 *c_out, simd::d256 *s_out) noexcept
107   {
108     simd::d256 x = simd::avx::splat_f64(K_F64);
109     simd::d256 y = simd::avx::zero_f64();
110     const simd::d256 neg_zero = simd::avx::splat_f64(-0.0);
111     for ( int i = 0; i < N_FP_F64; ++i ) {
112       const simd::d256 sm = simd::avx::cmp_f64<_CMP_LT_OQ>(z, simd::avx::zero_f64());
113       const simd::d256 sb = simd::avx::and_f64(sm, neg_zero);
114       const simd::d256 vshift = simd::avx::splat_f64(SHIFT_POW_F64.v[i]);
115       const simd::d256 ys = simd::avx::mul_f64(y, vshift);
116       const simd::d256 xs = simd::avx::mul_f64(x, vshift);
117       const simd::d256 ai = simd::avx::splat_f64(ATAN_F64.v[i]);
118       const simd::d256 ys_s = simd::avx::xor_f64(ys, sb);
119       const simd::d256 xs_s = simd::avx::xor_f64(xs, sb);
120       const simd::d256 ai_s = simd::avx::xor_f64(ai, sb);
121       x = simd::avx::sub_f64(x, ys_s);
122       y = simd::avx::add_f64(y, xs_s);
123       z = simd::avx::sub_f64(z, ai_s);
124     }
125     *c_out = x;
126     *s_out = y;
127   }

131   [[gnu::flatten]] inline simd::d256
132   sin_cordic(simd::d256 x) noexcept
133   {
134     simd::i256 q;
135     const simd::d256 r = __cordic_simd::reduce_pio2_d256(x, &q);
136     simd::d256 c, s;
137     __cordic_simd::sincos_kernel_d256(r, &c, &s);
138     //   q=0: s, q=1: c, q=2: -s, q=3: -c
139     const simd::i256 q_and1 = simd::avx2::and_i256(q, simd::avx::splat_i64(1));
140     const simd::d256 m_odd = simd::avx::cast_i256_to_f64(simd::avx2::eq_i64(q_and1, simd::avx::splat_i64(1)));
141     const simd::d256 m_neg
142         = simd::avx::cast_i256_to_f64(simd::avx2::eq_i64(simd::avx2::and_i256(q, simd::avx::splat_i64(2)), simd::avx::splat_i64(2)));
143     simd::d256 result = simd::avx::blendv_f64(s, c, m_odd);
144     result = simd::avx::blendv_f64(result, fneg(result), m_neg);
145     return __cordic_simd::guard_d256(x, result, x);
146   }

148   [[gnu::flatten]] inline simd::d256
149   cos_cordic(simd::d256 x) noexcept
150   {
151     simd::i256 q;
152     const simd::d256 r = __cordic_simd::reduce_pio2_d256(x, &q);
153     simd::d256 c, s;
154     __cordic_simd::sincos_kernel_d256(r, &c, &s);
155     // cos(x) = sin(x + pi/2) effectively shifts the quadrant by +1
156     const simd::i256 qs = simd::avx2::add_i64(q, simd::avx::splat_i64(1));
157     const simd::i256 q_and1 = simd::avx2::and_i256(qs, simd::avx::splat_i64(1));
158     const simd::d256 m_odd = simd::avx::cast_i256_to_f64(simd::avx2::eq_i64(q_and1, simd::avx::splat_i64(1)));
159     const simd::d256 m_neg
160         = simd::avx::cast_i256_to_f64(simd::avx2::eq_i64(simd::avx2::and_i256(qs, simd::avx::splat_i64(2)), simd::avx::splat_i64(2)));
161     simd::d256 result = simd::avx::blendv_f64(s, c, m_odd);
162     result = simd::avx::blendv_f64(result, fneg(result), m_neg);
163     return __cordic_simd::guard_d256(x, result, simd::avx::splat_f64(1.0));
164   }

173   [[gnu::flatten]] inline simd::d256
174   tan_cordic(simd::d256 x) noexcept
175   {
176     simd::i256 q;
177     const simd::d256 r = __cordic_simd::reduce_pio2_d256(x, &q);
178     simd::d256 c, s;
179     __cordic_simd::sincos_kernel_d256(r, &c, &s);
180     const simd::d256 ratio = simd::avx::div_f64(s, c);
181     const simd::d256 cot = simd::avx::div_f64(fneg(c), s);
182     const simd::i256 q_and1 = simd::avx2::and_i256(q, simd::avx::splat_i64(1));
183     const simd::d256 m_odd = simd::avx::cast_i256_to_f64(simd::avx2::eq_i64(q_and1, simd::avx::splat_i64(1)));
184     return __cordic_simd::guard_d256(x, simd::avx::blendv_f64(ratio, cot, m_odd), x);
185   }

299   [[gnu::always_inline]] inline f32
300   __reduce_lane_f64(f32 xf, int *q) noexcept
301   {
302     const f64 x = f64(xf);
303     if ( !ieee::is_finite(x) ) [[unlikely]] {
304       *q = 0;
305       return 0.0f;
306     }
307     f64 t;
308     // cody_waite's leading x - fN * pio2_hi is exact only while fN fits in 20 bits, since pio2_hi
309     // carries the other 33; above that the full payne_hanek reduction is the one that holds
310     *q = (mkbits::manip::fabs(x) < 0x1.0p20) ? mkbits::trig_ns::__rr::cody_waite(x, &t)
311                                              : mkbits::trig_ns::__rr::__paynehanek::payne_hanek(x, &t);
312     return f32(t);
313   }

315   [[gnu::always_inline]] inline simd::f128
316   guard_f128(simd::f128 x, simd::f128 y, simd::f128 small) noexcept
317   {
318     const float32x4_t ax = simd::neon::abs(x);
319     y = simd::neon::select(simd::neon::lt(ax, simd::neon::splat_f32(0x1.0p-12f)), small, y);
320     return simd::neon::select(simd::neon::lt(ax, simd::neon::splat_f32(ieee::inf_v<f32>(0))), y,
321                               simd::neon::splat_f32(ieee::qnan_v<f32>()));
322   }

324   [[gnu::always_inline]] inline simd::f128
325   reduce_pio2_f128(simd::f128 x, int32x4_t *q_out) noexcept
326   {
327     int q0, q1, q2, q3;
328     const f32 r0 = __reduce_lane_f64(simd::neon::get_lane_f32<0>(x), &q0);
329     const f32 r1 = __reduce_lane_f64(simd::neon::get_lane_f32<1>(x), &q1);
330     const f32 r2 = __reduce_lane_f64(simd::neon::get_lane_f32<2>(x), &q2);
331     const f32 r3 = __reduce_lane_f64(simd::neon::get_lane_f32<3>(x), &q3);
332     const f32 r_arr[4] = { r0, r1, r2, r3 };
333     const int32_t q_arr[4] = { q0, q1, q2, q3 };
334     *q_out = simd::neon::load_i32(q_arr);
335     return simd::neon::load_f32(r_arr);
336   }

338   [[gnu::flatten]] inline void
339   sincos_kernel_f128(simd::f128 z, simd::f128 *c_out, simd::f128 *s_out) noexcept
340   {
341     float32x4_t x = simd::neon::splat_f32(K_F32);
342     float32x4_t y = simd::neon::splat_f32(0.0f);
343     const uint32x4_t sign_bit_mask = simd::neon::splat_u32(0x80000000u);
344     for ( int i = 0; i < N_FP_F32; ++i ) {
345       const uint32x4_t sm = simd::neon::lt(z, simd::neon::splat_f32(0.0f));      // all-1 where z<0
346       const uint32x4_t sb = simd::neon::and_(sm, sign_bit_mask);
347       const float32x4_t vshift = simd::neon::splat_f32(SHIFT_POW_F32.v[i]);
348       const float32x4_t ys = simd::neon::mul(y, vshift);
349       const float32x4_t xs = simd::neon::mul(x, vshift);
350       const float32x4_t ai = simd::neon::splat_f32(ATAN_F32.v[i]);
351       const float32x4_t ys_s = simd::neon::reinterpret_f32_from_u32(simd::neon::xor_(simd::neon::reinterpret_u32_from_f32(ys), sb));
352       const float32x4_t xs_s = simd::neon::reinterpret_f32_from_u32(simd::neon::xor_(simd::neon::reinterpret_u32_from_f32(xs), sb));
353       const float32x4_t ai_s = simd::neon::reinterpret_f32_from_u32(simd::neon::xor_(simd::neon::reinterpret_u32_from_f32(ai), sb));
354       x = simd::neon::sub(x, ys_s);
355       y = simd::neon::add(y, xs_s);
356       z = simd::neon::sub(z, ai_s);
357     }
358     *c_out = x;
359     *s_out = y;
360   }

364   [[gnu::flatten]] inline simd::f128
365   sin_cordic(simd::f128 x) noexcept
366   {
367     int32x4_t q;
368     const float32x4_t r = __cordic_simd_neon::reduce_pio2_f128(x, &q);
369     float32x4_t c, s;
370     __cordic_simd_neon::sincos_kernel_f128(r, &c, &s);
371     const uint32x4_t odd = simd::neon::eq(simd::neon::and_(q, simd::neon::splat_i32(1)), simd::neon::splat_i32(1));
372     const uint32x4_t neg = simd::neon::eq(simd::neon::and_(q, simd::neon::splat_i32(2)), simd::neon::splat_i32(2));
373     float32x4_t result = simd::neon::select(odd, c, s);
374     result = simd::neon::select(neg, simd::neon::neg(result), result);
375     return __cordic_simd_neon::guard_f128(x, result, x);
376   }

378   [[gnu::flatten]] inline simd::f128
379   cos_cordic(simd::f128 x) noexcept
380   {
381     int32x4_t q;
382     const float32x4_t r = __cordic_simd_neon::reduce_pio2_f128(x, &q);
383     float32x4_t c, s;
384     __cordic_simd_neon::sincos_kernel_f128(r, &c, &s);
385     const int32x4_t qs = simd::neon::add(q, simd::neon::splat_i32(1));
386     const uint32x4_t odd = simd::neon::eq(simd::neon::and_(qs, simd::neon::splat_i32(1)), simd::neon::splat_i32(1));
387     const uint32x4_t neg = simd::neon::eq(simd::neon::and_(qs, simd::neon::splat_i32(2)), simd::neon::splat_i32(2));
388     float32x4_t result = simd::neon::select(odd, c, s);
389     result = simd::neon::select(neg, simd::neon::neg(result), result);
390     return __cordic_simd_neon::guard_f128(x, result, simd::neon::splat_f32(1.0f));
391   }

400   [[gnu::flatten]] inline simd::f128
401   tan_cordic(simd::f128 x) noexcept
402   {
403     int32x4_t q;
404     const float32x4_t r = __cordic_simd_neon::reduce_pio2_f128(x, &q);
405     float32x4_t c, s;
406     __cordic_simd_neon::sincos_kernel_f128(r, &c, &s);
407 #if defined(__micron_arch_arm64)
408     const float32x4_t ratio = simd::neon::div(s, c);
409     const float32x4_t cot = simd::neon::div(simd::neon::neg(c), s);
410 #else
411     float32x4_t rc = simd::neon::rcp_est(c);
412     rc = simd::neon::mul(rc, simd::neon::rcp_step(c, rc));
413     rc = simd::neon::mul(rc, simd::neon::rcp_step(c, rc));
414     const float32x4_t ratio = simd::neon::mul(s, rc);
415     float32x4_t rs = simd::neon::rcp_est(s);
416     rs = simd::neon::mul(rs, simd::neon::rcp_step(s, rs));
417     rs = simd::neon::mul(rs, simd::neon::rcp_step(s, rs));
418     const float32x4_t cot = simd::neon::mul(simd::neon::neg(c), rs);
419 #endif
420     const uint32x4_t odd = simd::neon::eq(simd::neon::and_(q, simd::neon::splat_i32(1)), simd::neon::splat_i32(1));
421     return __cordic_simd_neon::guard_f128(x, simd::neon::select(odd, cot, ratio), x);
422   }

438   [[gnu::flatten]] inline simd::d128
439   reduce_scalar(simd::d128 x, int64x2_t *q_out) noexcept
440   {
441     f64 xs[2] = { simd::neon::get_lane_f64<0>(x), simd::neon::get_lane_f64<1>(x) };
442     f64 rs[2];
443     i64 qs[2];
444     for ( usize i = 0; i < 2; ++i ) {
445       if ( !ieee::is_finite(xs[i]) ) [[unlikely]] {
446         rs[i] = 0.0;
447         qs[i] = 0;
448         continue;
449       }
450       qs[i] = (mkbits::manip::fabs(xs[i]) < 0x1.0p20) ? mkbits::trig_ns::__rr::cody_waite(xs[i], &rs[i])
451                                                       : mkbits::trig_ns::__rr::__paynehanek::payne_hanek(xs[i], &rs[i]);
452     }
453     *q_out = simd::neon::load_i64(qs);
454     return simd::neon::load_f64(rs);
455   }
456 
457   [[gnu::always_inline]] inline simd::d128
458   reduce(simd::d128 x, int64x2_t *q_out) noexcept
459   {
460     const int64x2_t big = simd::neon::reinterpret_s64_from_u64(simd::neon::ge(simd::neon::abs(x), simd::neon::splat_f64(0x1.0p33)));
461     if ( (simd::neon::get_lane_i64<0>(big) | simd::neon::get_lane_i64<1>(big)) != 0 ) [[unlikely]]
462       return reduce_scalar(x, q_out);
463 
464     const float64x2_t fN = simd::neon::rint(simd::neon::mul(x, simd::neon::splat_f64(inv_pio2)));
465     float64x2_t t = simd::neon::fms_f64(x, fN, simd::neon::splat_f64(pio2_hi));
466     t = simd::neon::fms_f64(t, fN, simd::neon::splat_f64(pio2_mid));
467     t = simd::neon::fms_f64(t, fN, simd::neon::splat_f64(pio2_lo));
468     *q_out = simd::neon::convert_f64_to_i64(fN);
469     return t;
470   }

472   [[gnu::always_inline]] inline simd::d128
473   guard(simd::d128 x, simd::d128 y, simd::d128 small) noexcept
474   {
475     const float64x2_t ax = simd::neon::abs(x);
476     y = simd::neon::select(simd::neon::lt(ax, simd::neon::splat_f64(0x1.0p-26)), small, y);
477     return simd::neon::select(simd::neon::lt(ax, simd::neon::splat_f64(ieee::inf_v<f64>(0))), y,
478                               simd::neon::splat_f64(ieee::qnan_v<f64>()));
479   }

481   [[gnu::flatten]] inline void
482   sincos_kernel_d128(simd::d128 z, simd::d128 *c_out, simd::d128 *s_out) noexcept
483   {
484     float64x2_t x = simd::neon::splat_f64(K_F64);
485     float64x2_t y = simd::neon::splat_f64(0.0);
486     const uint64x2_t sign_bit_mask = simd::neon::splat_u64(0x8000000000000000ULL);
487     for ( int i = 0; i < N_FP_F64; ++i ) {
488       const uint64x2_t sm = simd::neon::lt(z, simd::neon::splat_f64(0.0));
489       const uint64x2_t sb = simd::neon::and_(sm, sign_bit_mask);
490       const float64x2_t vshift = simd::neon::splat_f64(SHIFT_POW_F64.v[i]);
491       const float64x2_t ys = simd::neon::mul(y, vshift);
492       const float64x2_t xs = simd::neon::mul(x, vshift);
493       const float64x2_t ai = simd::neon::splat_f64(ATAN_F64.v[i]);
494       const float64x2_t ys_s = simd::neon::reinterpret_f64_from_u64(simd::neon::xor_(simd::neon::reinterpret_u64_from_f64(ys), sb));
495       const float64x2_t xs_s = simd::neon::reinterpret_f64_from_u64(simd::neon::xor_(simd::neon::reinterpret_u64_from_f64(xs), sb));
496       const float64x2_t ai_s = simd::neon::reinterpret_f64_from_u64(simd::neon::xor_(simd::neon::reinterpret_u64_from_f64(ai), sb));
497       x = simd::neon::sub(x, ys_s);
498       y = simd::neon::add(y, xs_s);
499       z = simd::neon::sub(z, ai_s);
500     }
501     *c_out = x;
502     *s_out = y;
503   }

507   [[gnu::flatten]] inline simd::d128
508   sin_cordic(simd::d128 x) noexcept
509   {
510     int64x2_t q;
511     const float64x2_t r = __cordic_simd_neon_d::reduce(x, &q);
512     float64x2_t c, s;
513     __cordic_simd_neon_d::sincos_kernel_d128(r, &c, &s);
514     const uint64x2_t odd = simd::neon::eq(simd::neon::and_(q, simd::neon::splat_i64(1)), simd::neon::splat_i64(1));
515     const uint64x2_t neg = simd::neon::eq(simd::neon::and_(q, simd::neon::splat_i64(2)), simd::neon::splat_i64(2));
516     float64x2_t result = simd::neon::select(odd, c, s);
517     result = simd::neon::select(neg, simd::neon::neg(result), result);
518     return __cordic_simd_neon_d::guard(x, result, x);
519   }

521   [[gnu::flatten]] inline simd::d128
522   cos_cordic(simd::d128 x) noexcept
523   {
524     int64x2_t q;
525     const float64x2_t r = __cordic_simd_neon_d::reduce(x, &q);
526     float64x2_t c, s;
527     __cordic_simd_neon_d::sincos_kernel_d128(r, &c, &s);
528     const int64x2_t qs = simd::neon::add(q, simd::neon::splat_i64(1));
529     const uint64x2_t odd = simd::neon::eq(simd::neon::and_(qs, simd::neon::splat_i64(1)), simd::neon::splat_i64(1));
530     const uint64x2_t neg = simd::neon::eq(simd::neon::and_(qs, simd::neon::splat_i64(2)), simd::neon::splat_i64(2));
531     float64x2_t result = simd::neon::select(odd, c, s);
532     result = simd::neon::select(neg, simd::neon::neg(result), result);
533     return __cordic_simd_neon_d::guard(x, result, simd::neon::splat_f64(1.0));
534   }
