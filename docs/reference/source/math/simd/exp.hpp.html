34   [[gnu::flatten]] inline simd::d256
35   exp(simd::d256 x) noexcept
36   {
37     using namespace mkbits::coeff::exp_f64_data;
38 
39     const simd::d256 x0 = x;
40     const simd::d256 hi = simd::avx::splat_f64(0x1.62e42fefa39efp+9);      // 709.7827128933840, largest finite exp
41     const simd::d256 lo = simd::avx::splat_f64(-745.13);
42     x = simd::avx::min_f64(simd::avx::max_f64(x, lo), hi);
43 
44     const simd::d256 fN
45         = simd::avx::round_f64<_MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC>(simd::avx::mul_f64(x, simd::avx::splat_f64(inv_ln2_32)));
46 
47     simd::d256 r = simd::fma::fnma_f64(fN, simd::avx::splat_f64(ln2_32_hi), x);
48     r = simd::fma::fnma_f64(fN, simd::avx::splat_f64(ln2_32_lo), r);
49 
50     simd::d256 p = simd::fma::fma_f64(simd::avx::splat_f64(rem[4]), r, simd::avx::splat_f64(rem[3]));
51     p = simd::fma::fma_f64(p, r, simd::avx::splat_f64(rem[2]));
52     p = simd::fma::fma_f64(p, r, simd::avx::splat_f64(rem[1]));
53     p = simd::fma::fma_f64(p, r, simd::avx::splat_f64(rem[0]));
54     const simd::d256 r2 = simd::avx::mul_f64(r, r);
55     const simd::d256 exp_r = simd::fma::fma_f64(p, r2, simd::avx::add_f64(r, simd::avx::splat_f64(1.0)));
56 
57     const __m128i N32 = simd::avx::convert_f64_to_i32(fN);
58     const __m128i j32 = simd::sse::and_i128(N32, simd::sse::splat_i32(31));
59 
60     const simd::d256 tw = simd::avx2::gather_f64<8>(reinterpret_cast<const double *>(twoN), j32);
61 
62     // two-step 2^k
63     const __m128i k32 = simd::sse::shr_arith_i32(N32, 5);
64     const __m128i kh32 = simd::sse::shr_arith_i32(k32, 1);
65     const __m128i kl32 = simd::sse::sub_i32(k32, kh32);
66     const simd::i256 bh = simd::avx2::add_i64(simd::avx2::widen_i32_to_i64(kh32), simd::avx::splat_i64(1023));
67     const simd::i256 bl = simd::avx2::add_i64(simd::avx2::widen_i32_to_i64(kl32), simd::avx::splat_i64(1023));
68     const simd::d256 sh = simd::avx::cast_i256_to_f64(simd::avx2::shl_i64(bh, 52));
69     const simd::d256 sl = simd::avx::cast_i256_to_f64(simd::avx2::shl_i64(bl, 52));
70     simd::d256 y = simd::avx::mul_f64(simd::avx::mul_f64(simd::avx::mul_f64(tw, exp_r), sh), sl);
71 
72     y = simd::avx::blendv_f64(y, simd::avx::splat_f64(ieee::inf_v<f64>(0)), simd::avx::cmp_f64<_CMP_GT_OQ>(x0, hi));
73     y = simd::avx::blendv_f64(y, simd::avx::zero_f64(), simd::avx::cmp_f64<_CMP_LT_OQ>(x0, lo));
74     return simd::avx::blendv_f64(y, x0, simd::avx::cmp_f64<_CMP_UNORD_Q>(x0, x0));
75   }

81     const simd::f256 x0 = x;
82     const simd::f256 hi = simd::avx::splat_f32(88.72283935546875f);      // largest f32 with finite exp
83     const simd::f256 lo = simd::avx::splat_f32(-103.972f);
84     x = simd::avx::min_f32(simd::avx::max_f32(x, lo), hi);
85     const simd::f256 fN
86         = simd::avx::round_f32<_MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC>(simd::avx::mul_f32(x, simd::avx::splat_f32(inv_ln2_16)));
87     simd::f256 r = simd::fma::fnma_f32(fN, simd::avx::splat_f32(ln2_16_hi), x);
88     r = simd::fma::fnma_f32(fN, simd::avx::splat_f32(ln2_16_lo), r);
89     simd::f256 p = simd::fma::fma_f32(simd::avx::splat_f32(rem[1]), r, simd::avx::splat_f32(rem[0]));
90     const simd::f256 r2 = simd::avx::mul_f32(r, r);
91     const simd::f256 exp_r = simd::fma::fma_f32(p, r2, simd::avx::add_f32(r, simd::avx::splat_f32(1.0f)));
92     const simd::i256 N = simd::avx::convert_f32_to_i32(fN);
93     const simd::i256 j = simd::avx2::and_i256(N, simd::avx::splat_i32(15));
94     const simd::f256 tw = simd::avx2::gather_f32<4>(reinterpret_cast<const float *>(twoN), j);
95     // two-step 2^k
96     const simd::i256 k = simd::avx2::shr_arith_i32(N, 4);
97     const simd::i256 kh = simd::avx2::shr_arith_i32(k, 1);
98     const simd::i256 kl = simd::avx2::sub_i32(k, kh);
99     const simd::f256 sh = simd::avx::cast_i256_to_f32(simd::avx2::shl_i32(simd::avx2::add_i32(kh, simd::avx::splat_i32(127)), 23));
100     const simd::f256 sl = simd::avx::cast_i256_to_f32(simd::avx2::shl_i32(simd::avx2::add_i32(kl, simd::avx::splat_i32(127)), 23));
101     simd::f256 y = simd::avx::mul_f32(simd::avx::mul_f32(simd::avx::mul_f32(tw, exp_r), sh), sl);
102 
103     y = simd::avx::blendv_f32(y, simd::avx::splat_f32(ieee::inf_v<f32>(0)), simd::avx::cmp_f32<_CMP_GT_OQ>(x0, hi));
104     y = simd::avx::blendv_f32(y, simd::avx::zero_f32(), simd::avx::cmp_f32<_CMP_LT_OQ>(x0, lo));
105     return simd::avx::blendv_f32(y, x0, simd::avx::cmp_f32<_CMP_UNORD_Q>(x0, x0));
106   }

154   [[gnu::flatten]] inline simd::f128
155   exp(simd::f128 x) noexcept
156   {
157     using namespace mkbits::coeff::exp_f32_data;
158     const float32x4_t x0 = x;
159     const float32x4_t cap_hi = simd::neon::splat_f32(88.72283935546875f);      // largest f32 with finite exp
160     const float32x4_t cap_lo = simd::neon::splat_f32(-103.972f);
161     x = simd::neon::min(simd::neon::max(x, cap_lo), cap_hi);
162 
163 #if defined(__micron_arch_arm64) || defined(__micron_arm_directed_rounding)
164     const float32x4_t fN = simd::neon::rint(simd::neon::mul(x, simd::neon::splat_f32(inv_ln2_16)));
165 #else
166     const uint32x4_t __sign_mask = simd::neon::splat_u32(0x80000000u);
167     const uint32x4_t __xs = simd::neon::and_(simd::neon::reinterpret_u32_from_f32(x), __sign_mask);
168     const float32x4_t __half
169         = simd::neon::reinterpret_f32_from_u32(simd::neon::or_(__xs, simd::neon::reinterpret_u32_from_f32(simd::neon::splat_f32(0.5f))));
170 #if defined(__micron_arm_fma)
171     const float32x4_t __scaled = simd::neon::fma_f32(__half, x, simd::neon::splat_f32(inv_ln2_16));
172 #else
173     const float32x4_t __scaled = simd::neon::add(simd::neon::mul(x, simd::neon::splat_f32(inv_ln2_16)), __half);
174 #endif
175     const int32x4_t Ni_pre = simd::neon::convert_f32_to_i32(__scaled);
176     const float32x4_t fN = simd::neon::convert_i32_to_f32(Ni_pre);
177 #endif
178 
179 #if defined(__micron_arm_fma) || defined(__micron_arch_arm64)
180     float32x4_t r = simd::neon::fms_f32(x, fN, simd::neon::splat_f32(ln2_16_hi));
181     r = simd::neon::fms_f32(r, fN, simd::neon::splat_f32(ln2_16_lo));
182 
183     float32x4_t p = simd::neon::fma_f32(simd::neon::splat_f32(rem[0]), simd::neon::splat_f32(rem[1]), r);
184 #else
185     float32x4_t r = simd::neon::sub(x, simd::neon::mul(fN, simd::neon::splat_f32(ln2_16_hi)));
186     r = simd::neon::sub(r, simd::neon::mul(fN, simd::neon::splat_f32(ln2_16_lo)));
187     float32x4_t p = simd::neon::add(simd::neon::splat_f32(rem[0]), simd::neon::mul(simd::neon::splat_f32(rem[1]), r));
188 #endif
189     const float32x4_t r2 = simd::neon::mul(r, r);
190     const float32x4_t one = simd::neon::splat_f32(1.0f);
191 #if defined(__micron_arm_fma) || defined(__micron_arch_arm64)
192     const float32x4_t exp_r = simd::neon::fma_f32(simd::neon::add(r, one), p, r2);
193 #else
194     const float32x4_t exp_r = simd::neon::add(simd::neon::add(r, one), simd::neon::mul(p, r2));
195 #endif
196 
197     const int32x4_t Ni = simd::neon::convert_f32_to_i32(fN);
198     const int32x4_t j = simd::neon::and_(Ni, simd::neon::splat_i32(15));
199     const f32 tw_arr[4] = {
200       twoN[simd::neon::get_lane_i32<0>(j)],
201       twoN[simd::neon::get_lane_i32<1>(j)],
202       twoN[simd::neon::get_lane_i32<2>(j)],
203       twoN[simd::neon::get_lane_i32<3>(j)],
204     };
205     const float32x4_t tw = simd::neon::load_f32(tw_arr);
206 
207     // two-step 2^k
208     const int32x4_t k = simd::neon::shr_arith_i32<4>(Ni);
209     const int32x4_t kh = simd::neon::shr_arith_i32<1>(k);
210     const int32x4_t kl = simd::neon::sub(k, kh);
211     const float32x4_t sh = simd::neon::reinterpret_f32_from_s32(simd::neon::shl_i32<23>(simd::neon::add(kh, simd::neon::splat_i32(127))));
212     const float32x4_t sl = simd::neon::reinterpret_f32_from_s32(simd::neon::shl_i32<23>(simd::neon::add(kl, simd::neon::splat_i32(127))));
213     float32x4_t y = simd::neon::mul(simd::neon::mul(simd::neon::mul(tw, exp_r), sh), sl);
214 
215     y = simd::neon::select(simd::neon::gt(x0, cap_hi), simd::neon::splat_f32(ieee::inf_v<f32>(0)), y);
216     y = simd::neon::select(simd::neon::lt(x0, cap_lo), simd::neon::splat_f32(0.0f), y);
217     // ge(x,x) is false only for NaN, which is what the alias layer offers in place of an unordered compare
218     return simd::neon::select(simd::neon::ge(x0, x0), y, x0);
219   }

223   [[gnu::flatten]] inline simd::d128
224   exp(simd::d128 x) noexcept
225   {
226     using namespace mkbits::coeff::exp_f64_data;
227     const float64x2_t x0 = x;
228     const float64x2_t cap_hi = simd::neon::splat_f64(0x1.62e42fefa39efp+9);      // 709.7827128933840
229     const float64x2_t cap_lo = simd::neon::splat_f64(-745.13);
230     x = simd::neon::min(simd::neon::max(x, cap_lo), cap_hi);
231     const float64x2_t fN = simd::neon::rint(simd::neon::mul(x, simd::neon::splat_f64(inv_ln2_32)));
232     float64x2_t r = simd::neon::fms_f64(x, fN, simd::neon::splat_f64(ln2_32_hi));
233     r = simd::neon::fms_f64(r, fN, simd::neon::splat_f64(ln2_32_lo));
234     float64x2_t p = simd::neon::fma_f64(simd::neon::splat_f64(rem[3]), simd::neon::splat_f64(rem[4]), r);
235     p = simd::neon::fma_f64(simd::neon::splat_f64(rem[2]), p, r);
236     p = simd::neon::fma_f64(simd::neon::splat_f64(rem[1]), p, r);
237     p = simd::neon::fma_f64(simd::neon::splat_f64(rem[0]), p, r);
238     const float64x2_t r2 = simd::neon::mul(r, r);
239     const float64x2_t one = simd::neon::splat_f64(1.0);
240     const float64x2_t exp_r = simd::neon::fma_f64(simd::neon::add(r, one), p, r2);
241     const int64x2_t Ni = simd::neon::convert_f64_to_i64(fN);
242     const int64x2_t j = simd::neon::and_(Ni, simd::neon::splat_i64(31));
243     const f64 tw_arr[2] = {
244       twoN[simd::neon::get_lane_i64<0>(j)],
245       twoN[simd::neon::get_lane_i64<1>(j)],
246     };
247     const float64x2_t tw = simd::neon::load_f64(tw_arr);
248     // two-step 2^k
249     const int64x2_t k = simd::neon::shr_arith_i64<5>(Ni);
250     const int64x2_t kh = simd::neon::shr_arith_i64<1>(k);
251     const int64x2_t kl = simd::neon::sub(k, kh);
252     const float64x2_t sh = simd::neon::reinterpret_f64_from_s64(simd::neon::shl_i64<52>(simd::neon::add(kh, simd::neon::splat_i64(1023))));
253     const float64x2_t sl = simd::neon::reinterpret_f64_from_s64(simd::neon::shl_i64<52>(simd::neon::add(kl, simd::neon::splat_i64(1023))));
254     float64x2_t y = simd::neon::mul(simd::neon::mul(simd::neon::mul(tw, exp_r), sh), sl);
255 
256     y = simd::neon::select(simd::neon::gt(x0, cap_hi), simd::neon::splat_f64(ieee::inf_v<f64>(0)), y);
257     y = simd::neon::select(simd::neon::lt(x0, cap_lo), simd::neon::splat_f64(0.0), y);
258     return simd::neon::select(simd::neon::ge(x0, x0), y, x0);
259   }

326   exp(simd::f512 x) noexcept
327   {
328     using namespace mkbits::coeff::exp_f32_data;
329     const simd::f512 hi = simd::avx512::splat_f32(88.722f);
330     const simd::f512 lo = simd::avx512::splat_f32(-103.972f);
331     x = simd::avx512::min_f32(simd::avx512::max_f32(x, lo), hi);
332     const simd::f512 fN = simd::avx512::roundscale_f32<_MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC>(
333         simd::avx512::mul_f32(x, simd::avx512::splat_f32(inv_ln2_16)));
334     simd::f512 r = simd::fma::fnma_f32_v512(fN, simd::avx512::splat_f32(ln2_16_hi), x);
335     r = simd::fma::fnma_f32_v512(fN, simd::avx512::splat_f32(ln2_16_lo), r);
336     simd::f512 p = simd::fma::fma_f32_v512(simd::avx512::splat_f32(rem[1]), r, simd::avx512::splat_f32(rem[0]));
337     const simd::f512 r2 = simd::avx512::mul_f32(r, r);
338     const simd::f512 exp_r = simd::fma::fma_f32_v512(p, r2, simd::avx512::add_f32(r, simd::avx512::splat_f32(1.0f)));
339     const simd::i512 N = simd::avx512::convert_f32_to_i32(fN);
340     const simd::i512 j = simd::avx512::and_i512(N, simd::avx512::splat_i32(15));
341     const simd::f512 tw = simd::avx512::gather_f32<4>(j, reinterpret_cast<const float *>(twoN));
342     // two-step 2^k
343     const simd::i512 k = simd::avx512::shr_arith_i32(N, 4);
344     const simd::i512 kh = simd::avx512::shr_arith_i32(k, 1);
345     const simd::i512 kl = simd::avx512::sub_i32(k, kh);
346     const simd::f512 sh
347         = simd::avx512::cast_i512_to_f32(simd::avx512::shl_i32(simd::avx512::add_i32(kh, simd::avx512::splat_i32(127)), 23));
348     const simd::f512 sl
349         = simd::avx512::cast_i512_to_f32(simd::avx512::shl_i32(simd::avx512::add_i32(kl, simd::avx512::splat_i32(127)), 23));
350     return simd::avx512::mul_f32(simd::avx512::mul_f32(simd::avx512::mul_f32(tw, exp_r), sh), sl);
351   }
