43   inline constexpr f64 pio2 = 0x1.921fb54442d18p+0;

45   // region boundaries, and the numerator/denominator rows they select
46   inline constexpr f64 bnd0 = 0.4375;
47   inline constexpr f64 bnd1 = 0.6875;
48   inline constexpr f64 bnd2 = 1.1875;
49   inline constexpr f64 bnd3 = 2.4375;
50 
51   inline constexpr f64 c_na[5] = { 1.0, 2.0, 1.0, 1.0, 0.0 };
52   inline constexpr f64 c_nb[5] = { 0.0, -1.0, -1.0, -1.5, -1.0 };
53   inline constexpr f64 c_da[5] = { 0.0, 1.0, 1.0, 1.5, 1.0 };
54   inline constexpr f64 c_db[5] = { 1.0, 2.0, 1.0, 1.0, 0.0 };

56   // atan(ax) for ax finite in [2^-29, 2^61)
57   [[nodiscard, gnu::always_inline]] inline constexpr f64
58   atan_core(f64 ax) noexcept
59   {
60     using namespace mkbits::coeff::atan_f64_data;
61     const u32 idx
62         = static_cast<u32>(ax >= bnd0) + static_cast<u32>(ax >= bnd1) + static_cast<u32>(ax >= bnd2) + static_cast<u32>(ax >= bnd3);
63 
64     const f64 t = hw::fmadd_sd(c_na[idx], ax, c_nb[idx]) / hw::fmadd_sd(c_da[idx], ax, c_db[idx]);
65     const f64 z = t * t;
66     const f64 w = z * z;
67 
68     f64 s1 = hw::fmadd_sd(aT[10], w, aT[8]);
69     s1 = hw::fmadd_sd(s1, w, aT[6]);
70     s1 = hw::fmadd_sd(s1, w, aT[4]);
71     s1 = hw::fmadd_sd(s1, w, aT[2]);
72     s1 = hw::fmadd_sd(s1, w, aT[0]);
73 
74     f64 s2 = hw::fmadd_sd(aT[9], w, aT[7]);
75     s2 = hw::fmadd_sd(s2, w, aT[5]);
76     s2 = hw::fmadd_sd(s2, w, aT[3]);
77     s2 = hw::fmadd_sd(s2, w, aT[1]);
78 
79     const f64 P = hw::fmadd_sd(z, s2, s1);
80     const f64 tz = t * z;
81     const f64 at = hw::fmadd_sd(-tz, P, t);      // t - t*z*P
82     // atan_lo[0] is +0.0 and `at` is strictly positive here, so the idx == 0 case needs no branch
83     return f64(atan_lo[idx] + at);
84   }

88   [[nodiscard, gnu::flatten]] inline constexpr f64
89   atan_bl(f64 x) noexcept
90   {
91     if ( ieee::is_nan(x) ) [[unlikely]]
92       return x;
93     const f64 ax = mkbits::manip::fabs(x);
94     if ( ax >= 0x1.0p+61 ) [[unlikely]]
95       return mkbits::manip::copysign<f64>(__atrig::pio2, x);
96     if ( ax < 0x1.0p-29 ) [[unlikely]]
97       return x;
98     return mkbits::manip::copysign<f64>(__atrig::atan_core(ax), x);
99   }

101   [[nodiscard, gnu::flatten]] inline constexpr f64
102   atan2_pos(f64 y, f64 x) noexcept
103   {
104     if ( y == 0.0 ) [[unlikely]]
105       return 0.0;
106     if ( x == 0.0 ) [[unlikely]]
107       return __atrig::pio2;
108     const f64 q = y / x;
109     if ( q >= 0x1.0p+61 ) [[unlikely]]
110       return __atrig::pio2;
111     if ( q < 0x1.0p-29 ) [[unlikely]]
112       return q;
113     return __atrig::atan_core(q);
114   }

116   // full-quadrant atan2
117   [[nodiscard, gnu::flatten]] inline constexpr f64
118   atan2_bl(f64 y, f64 x) noexcept
119   {
120     if ( ieee::is_nan(x) || ieee::is_nan(y) ) [[unlikely]]
121       return ieee::qnan_v<f64>();
122 
123     constexpr f64 pi = 0x1.921fb54442d18p+1;
124     constexpr f64 pi_quarter = 0x1.921fb54442d18p-1;
125     constexpr f64 three_pi_q = 0x1.2d97c7f3321d2p+1;
126 
127     const bool y_neg = mkbits::manip::signbit(y);
128     const bool x_neg = mkbits::manip::signbit(x);
129 
130     if ( y == 0 ) [[unlikely]] {
131       if ( !x_neg ) return mkbits::manip::copysign<f64>(0.0, y);
132       return y_neg ? -pi : pi;
133     }
134     if ( x == 0 ) [[unlikely]]
135       return y_neg ? -__atrig::pio2 : __atrig::pio2;
136     if ( ieee::is_inf(x) ) [[unlikely]] {
137       if ( ieee::is_inf(y) ) {
138         if ( !x_neg ) return y_neg ? -pi_quarter : pi_quarter;
139         return y_neg ? -three_pi_q : three_pi_q;
140       }
141       if ( !x_neg ) return mkbits::manip::copysign<f64>(0.0, y);
142       return y_neg ? -pi : pi;
143     }
144     if ( ieee::is_inf(y) ) [[unlikely]]
145       return y_neg ? -__atrig::pio2 : __atrig::pio2;
146 
147     const f64 z = atan_bl(y / x);
148     const f64 adj = x_neg ? mkbits::manip::copysign<f64>(pi, y) : f64(0.0);
149     return x_neg ? f64(z + adj) : z;
150   }

173   [[gnu::always_inline, gnu::target("avx2,fma")]] inline simd::d256
174   atan_core_p(simd::d256 ax) noexcept
175   {
176     namespace avx = micron::simd::avx;
177     namespace vfma = micron::simd::fma;
178     using namespace mkbits::coeff::atan_f64_data;
179 
180     const simd::d256 m0 = avx::cmp_f64<_CMP_GE_OQ>(ax, avx::splat_f64(bnd0));
181     const simd::d256 m1 = avx::cmp_f64<_CMP_GE_OQ>(ax, avx::splat_f64(bnd1));
182     const simd::d256 m2 = avx::cmp_f64<_CMP_GE_OQ>(ax, avx::splat_f64(bnd2));
183     const simd::d256 m3 = avx::cmp_f64<_CMP_GE_OQ>(ax, avx::splat_f64(bnd3));
184 
185     simd::d256 na = avx::splat_f64(c_na[0]);
186     na = avx::blendv_f64(na, avx::splat_f64(c_na[1]), m0);
187     na = avx::blendv_f64(na, avx::splat_f64(c_na[2]), m1);
188     na = avx::blendv_f64(na, avx::splat_f64(c_na[3]), m2);
189     na = avx::blendv_f64(na, avx::splat_f64(c_na[4]), m3);
190 
191     simd::d256 nb = avx::splat_f64(c_nb[0]);
192     nb = avx::blendv_f64(nb, avx::splat_f64(c_nb[1]), m0);
193     nb = avx::blendv_f64(nb, avx::splat_f64(c_nb[2]), m1);
194     nb = avx::blendv_f64(nb, avx::splat_f64(c_nb[3]), m2);
195     nb = avx::blendv_f64(nb, avx::splat_f64(c_nb[4]), m3);
196 
197     simd::d256 da = avx::splat_f64(c_da[0]);
198     da = avx::blendv_f64(da, avx::splat_f64(c_da[1]), m0);
199     da = avx::blendv_f64(da, avx::splat_f64(c_da[2]), m1);
200     da = avx::blendv_f64(da, avx::splat_f64(c_da[3]), m2);
201     da = avx::blendv_f64(da, avx::splat_f64(c_da[4]), m3);
202 
203     simd::d256 db = avx::splat_f64(c_db[0]);
204     db = avx::blendv_f64(db, avx::splat_f64(c_db[1]), m0);
205     db = avx::blendv_f64(db, avx::splat_f64(c_db[2]), m1);
206     db = avx::blendv_f64(db, avx::splat_f64(c_db[3]), m2);
207     db = avx::blendv_f64(db, avx::splat_f64(c_db[4]), m3);
208 
209     const simd::d256 t = avx::div_f64(vfma::fma_f64(na, ax, nb), vfma::fma_f64(da, ax, db));
210     const simd::d256 z = avx::mul_f64(t, t);
211     const simd::d256 w = avx::mul_f64(z, z);
212 
213     simd::d256 s1 = vfma::fma_f64(avx::splat_f64(aT[10]), w, avx::splat_f64(aT[8]));
214     s1 = vfma::fma_f64(s1, w, avx::splat_f64(aT[6]));
215     s1 = vfma::fma_f64(s1, w, avx::splat_f64(aT[4]));
216     s1 = vfma::fma_f64(s1, w, avx::splat_f64(aT[2]));
217     s1 = vfma::fma_f64(s1, w, avx::splat_f64(aT[0]));
218 
219     simd::d256 s2 = vfma::fma_f64(avx::splat_f64(aT[9]), w, avx::splat_f64(aT[7]));
220     s2 = vfma::fma_f64(s2, w, avx::splat_f64(aT[5]));
221     s2 = vfma::fma_f64(s2, w, avx::splat_f64(aT[3]));
222     s2 = vfma::fma_f64(s2, w, avx::splat_f64(aT[1]));
223 
224     const simd::d256 P = vfma::fma_f64(z, s2, s1);
225     const simd::d256 tz = avx::mul_f64(t, z);
226     // t - t*z*P, as fma(-(t*z), P, t) so it matches the scalar core's fused form exactly
227     const simd::d256 at = vfma::fma_f64(avx::xor_f64(tz, avx::splat_f64(sign_mask_f64)), P, t);
228 
229     simd::d256 lo = avx::splat_f64(atan_lo[0]);
230     lo = avx::blendv_f64(lo, avx::splat_f64(atan_lo[1]), m0);
231     lo = avx::blendv_f64(lo, avx::splat_f64(atan_lo[2]), m1);
232     lo = avx::blendv_f64(lo, avx::splat_f64(atan_lo[3]), m2);
233     lo = avx::blendv_f64(lo, avx::splat_f64(atan_lo[4]), m3);
234     return avx::add_f64(lo, at);
235   }

239   [[gnu::flatten]] inline void
240   atan2_bl_x4(const f64 *y, const f64 *x, f64 *r) noexcept
241   {
242     namespace avx = micron::simd::avx;
243     // f64 is _Float64 on amd64 gcc
244     const simd::d256 vy = avx::loadu_f64(reinterpret_cast<const double *>(y));
245     const simd::d256 vx = avx::loadu_f64(reinterpret_cast<const double *>(x));
246     const simd::d256 sgn = avx::splat_f64(__atrig::sign_mask_f64);
247 
248     const simd::d256 q = avx::div_f64(vy, vx);
249     const simd::d256 aq = avx::andnot_f64(sgn, q);      // |q|
250 
251     const simd::d256 ok
252         = avx::and_f64(avx::cmp_f64<_CMP_GE_OQ>(aq, avx::splat_f64(0x1.0p-29)), avx::cmp_f64<_CMP_LT_OQ>(aq, avx::splat_f64(0x1.0p+61)));
253     if ( avx::movemask_f64(ok) != 0xF ) [[unlikely]] {
254       __atan2_bl_x4_scalar(y, x, r);
255       return;
256     }
257 
258     const simd::d256 z = avx::or_f64(__atrig::atan_core_p(aq), avx::and_f64(q, sgn));      // copysign(core, q)
259     const simd::d256 xneg = avx::cmp_f64<_CMP_LT_OQ>(vx, avx::splat_f64(0.0));
260     const simd::d256 adj = avx::and_f64(avx::or_f64(avx::splat_f64(0x1.921fb54442d18p+1), avx::and_f64(vy, sgn)), xneg);
261     avx::storeu_f64(reinterpret_cast<double *>(r), avx::add_f64(z, adj));
262   }
