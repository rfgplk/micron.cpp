35 // Taylor coefficients of (1 - cos theta)/theta**2 as a polynomial in theta**2
36 template<ieee754_floating F>
37 [[nodiscard, gnu::always_inline]] inline constexpr F
38 lj_a_poly(F t2) noexcept
39 {
40   F r = F(1) / F(3628800);
41   r = r * t2 - F(1) / F(40320);
42   r = r * t2 + F(1) / F(720);
43   r = r * t2 - F(1) / F(24);
44   r = r * t2 + F(0.5);
45   return r;
46 }
47 
48 // Taylor coefficients of (theta - sin theta)/theta**3 as a polynomial in theta**2
49 template<ieee754_floating F>
50 [[nodiscard, gnu::always_inline]] inline constexpr F
51 lj_b_poly(F t2) noexcept
52 {
53   F r = F(1) / F(39916800);
54   r = r * t2 - F(1) / F(362880);
55   r = r * t2 + F(1) / F(5040);
56   r = r * t2 - F(1) / F(120);
57   r = r * t2 + F(1) / F(6);
58   return r;
59 }

61 // Taylor of 1/theta**2
62 template<ieee754_floating F>
63 [[nodiscard, gnu::always_inline]] inline constexpr F
64 lj_inv_poly(F t2) noexcept
65 {
66   F r = F(1) / F(1209600);
67   r = r * t2 + F(1) / F(30240);
68   r = r * t2 + F(1) / F(720);
69   r = r * t2 + F(1) / F(12);
70   return r;
71 }

73 template<ieee754_floating F>
74 [[nodiscard, gnu::always_inline]] inline constexpr vec<F, 3>
75 cross3(const vec<F, 3> &a, const vec<F, 3> &b) noexcept
76 {
77   return vec<F, 3>{ a.data[1] * b.data[2] - a.data[2] * b.data[1], a.data[2] * b.data[0] - a.data[0] * b.data[2],
78                     a.data[0] * b.data[1] - a.data[1] * b.data[0] };
79 }

81 // Branchless left-Jacobian, must faster
82 template<ieee754_floating F>
83 [[nodiscard, gnu::always_inline]] inline constexpr vec<F, 3>
84 left_jacobian_apply_with_ab(const vec<F, 3> &omega, const vec<F, 3> &v, F a, F b) noexcept
85 {
86   const vec<F, 3> wxv = cross3<F>(omega, v);
87   const vec<F, 3> wxwxv = cross3<F>(omega, wxv);
88   return vec<F, 3>{ v.data[0] + a * wxv.data[0] + b * wxwxv.data[0], v.data[1] + a * wxv.data[1] + b * wxwxv.data[1],
89                     v.data[2] + a * wxv.data[2] + b * wxwxv.data[2] };
90 }

92 // Left-Jacobian-inverse application
93 template<ieee754_floating F>
94 [[nodiscard, gnu::always_inline]] inline constexpr vec<F, 3>
95 left_jacobian_inv_apply_with_c(const vec<F, 3> &omega, const vec<F, 3> &t, F c) noexcept
96 {
97   const vec<F, 3> wxt = cross3<F>(omega, t);
98   const vec<F, 3> wxwxt = cross3<F>(omega, wxt);
99   return vec<F, 3>{ t.data[0] - F(0.5) * wxt.data[0] + c * wxwxt.data[0], t.data[1] - F(0.5) * wxt.data[1] + c * wxwxt.data[1],
100                     t.data[2] - F(0.5) * wxt.data[2] + c * wxwxt.data[2] };
101 }

103 template<ieee754_floating F>
104 [[nodiscard, gnu::always_inline]] inline constexpr F
105 small_angle_threshold_sq() noexcept
106 {
107   return F(1) / F(100);
108 }
109 
110 template<ieee754_floating F>
111 [[nodiscard, gnu::flatten]] inline constexpr vec<F, 3>
112 left_jacobian_apply(const vec<F, 3> &omega, const vec<F, 3> &v) noexcept
113 {
114   const F wx = omega.data[0], wy = omega.data[1], wz = omega.data[2];
115   const F theta_sq = wx * wx + wy * wy + wz * wz;
116   F a, b;
117   if ( theta_sq < small_angle_threshold_sq<F>() ) {
118     a = lj_a_poly<F>(theta_sq);
119     b = lj_b_poly<F>(theta_sq);
120   } else {
121     const F theta = math::fsqrt(theta_sq);
122     const F half = theta * F(0.5);
123     F s_half, c_half;
124     math::sincos<F>(half, s_half, c_half);
125     const F sin_theta = F(2) * s_half * c_half;
126     const F cos_theta = c_half * c_half - s_half * s_half;
127     a = (F(1) - cos_theta) / theta_sq;
128     b = (theta - sin_theta) / (theta * theta_sq);
129   }
130   return left_jacobian_apply_with_ab<F>(omega, v, a, b);
131 }
132 
133 template<ieee754_floating F>
134 [[nodiscard, gnu::flatten]] inline constexpr vec<F, 3>
135 left_jacobian_inv_apply(const vec<F, 3> &omega, const vec<F, 3> &t) noexcept
136 {
137   const F wx = omega.data[0], wy = omega.data[1], wz = omega.data[2];
138   const F theta_sq = wx * wx + wy * wy + wz * wz;
139   F coeff;
140   if ( theta_sq < small_angle_threshold_sq<F>() ) {
141     coeff = lj_inv_poly<F>(theta_sq);
142   } else {
143     const F theta = math::fsqrt(theta_sq);
144     const F half = theta * F(0.5);
145     F s_half, c_half;
146     math::sincos<F>(half, s_half, c_half);
147     coeff = (F(1) / theta_sq) - c_half / (F(2) * theta * s_half);
148   }
149   return left_jacobian_inv_apply_with_c<F>(omega, t, coeff);
150 }

212   [[nodiscard, gnu::flatten]] static constexpr SE3
213   exp_map(const vec<F, 6> &xi) noexcept
214   {
215     const vec<F, 3> v{ xi.data[0], xi.data[1], xi.data[2] };
216     const F wx = xi.data[3], wy = xi.data[4], wz = xi.data[5];
217     const vec<F, 3> w{ wx, wy, wz };
218     const F theta_sq = wx * wx + wy * wy + wz * wz;
219 
220     // optimized
221     F scalar;
222     F c_half;
223     F a, b;
224 
225     if ( theta_sq < __se3_impl::small_angle_threshold_sq<F>() ) {
226       scalar = __so3_impl::sinc_half_over_theta<F>(theta_sq);
227       c_half = __so3_impl::cos_half<F>(theta_sq);
228       a = __se3_impl::lj_a_poly<F>(theta_sq);
229       b = __se3_impl::lj_b_poly<F>(theta_sq);
230     } else {
231       const F theta = math::fsqrt(theta_sq);
232       const F half = theta * F(0.5);
233       F s_half;
234       math::sincos<F>(half, s_half, c_half);
235       scalar = s_half / theta;
236       const F sin_theta = F(2) * s_half * c_half;
237       const F cos_theta = c_half * c_half - s_half * s_half;
238       a = (F(1) - cos_theta) / theta_sq;
239       b = (theta - sin_theta) / (theta * theta_sq);
240     }
241 
242     const quat<F> q{ wx * scalar, wy * scalar, wz * scalar, c_half };
243     const vec<F, 3> t_out = __se3_impl::left_jacobian_apply_with_ab<F>(w, v, a, b);
244     return SE3{ SO3<F>{ q }, t_out };
245   }

247   [[nodiscard, gnu::flatten]] static constexpr vec<F, 6>
248   log_map(const SE3 &g) noexcept
249   {
250     F qx = g.R.q.data[0], qy = g.R.q.data[1], qz = g.R.q.data[2], qw = g.R.q.data[3];
251     if ( qw < F(0) ) {
252       qx = -qx;
253       qy = -qy;
254       qz = -qz;
255       qw = -qw;
256     }
257     const F xyz_sq = qx * qx + qy * qy + qz * qz;
258 
259     F wx, wy, wz;
260     F theta_sq;
261     if ( xyz_sq < __so3_impl::small_angle_threshold_sq<F>() ) {
262       const F inv_qw = F(1) / qw;
263       const F y2 = xyz_sq * (inv_qw * inv_qw);
264       F p = F(-1) / F(11);
265       p = p * y2 + F(1) / F(9);
266       p = p * y2 - F(1) / F(7);
267       p = p * y2 + F(1) / F(5);
268       p = p * y2 - F(1) / F(3);
269       p = p * y2 + F(1);
270       const F k = F(2) * p * inv_qw;
271       wx = qx * k;
272       wy = qy * k;
273       wz = qz * k;
274       theta_sq = wx * wx + wy * wy + wz * wz;
275     } else {
276       const F xyz_norm = math::fsqrt(xyz_sq);
277       const F angle = F(2) * math::atan2<F>(xyz_norm, qw);
278       const F k = angle / xyz_norm;
279       wx = qx * k;
280       wy = qy * k;
281       wz = qz * k;
282       theta_sq = angle * angle;
283     }
284     const vec<F, 3> w{ wx, wy, wz };
285 
286     F coeff;
287     if ( theta_sq < __se3_impl::small_angle_threshold_sq<F>() ) {
288       coeff = __se3_impl::lj_inv_poly<F>(theta_sq);
289     } else {
290       const F xyz_norm = math::fsqrt(xyz_sq);
291       const F theta = math::fsqrt(theta_sq);
292       coeff = (F(1) / theta_sq) - qw / (F(2) * theta * xyz_norm);
293     }
294     const vec<F, 3> v_out = __se3_impl::left_jacobian_inv_apply_with_c<F>(w, g.t, coeff);
295 
296     return vec<F, 6>{ v_out.data[0], v_out.data[1], v_out.data[2], wx, wy, wz };
297   }

301   {
302     const auto R3 = SO3<F>::to_matrix(g.R);
303     mat<F, 4, 4> M = mat<F, 4, 4>::zero();

337   [[nodiscard, gnu::flatten]] static constexpr mat<F, 6, 6>
338   adjoint(const SE3 &g) noexcept
339   {
340     const auto R3 = SO3<F>::to_matrix(g.R);
341     const F tx = g.t.data[0], ty = g.t.data[1], tz = g.t.data[2];
342     const F s00 = F(0), s01 = -tz, s02 = ty;
343     const F s10 = tz, s11 = F(0), s12 = -tx;
344     const F s20 = -ty, s21 = tx, s22 = F(0);
345     F tr[9];
346     tr[0] = s00 * R3.data[0] + s01 * R3.data[3] + s02 * R3.data[6];
347     tr[1] = s00 * R3.data[1] + s01 * R3.data[4] + s02 * R3.data[7];
348     tr[2] = s00 * R3.data[2] + s01 * R3.data[5] + s02 * R3.data[8];
349     tr[3] = s10 * R3.data[0] + s11 * R3.data[3] + s12 * R3.data[6];
350     tr[4] = s10 * R3.data[1] + s11 * R3.data[4] + s12 * R3.data[7];
351     tr[5] = s10 * R3.data[2] + s11 * R3.data[5] + s12 * R3.data[8];
352     tr[6] = s20 * R3.data[0] + s21 * R3.data[3] + s22 * R3.data[6];
353     tr[7] = s20 * R3.data[1] + s21 * R3.data[4] + s22 * R3.data[7];
354     tr[8] = s20 * R3.data[2] + s21 * R3.data[5] + s22 * R3.data[8];
355     mat<F, 6, 6> Ad = mat<F, 6, 6>::zero();
356     for ( usize r = 0; r < 3; ++r )
357       for ( usize c = 0; c < 3; ++c ) {
358         Ad.data[r * 6 + c] = R3.data[r * 3 + c];
359         Ad.data[r * 6 + (c + 3)] = tr[r * 3 + c];
360         Ad.data[(r + 3) * 6 + (c + 3)] = R3.data[r * 3 + c];
361       }
362     return Ad;
363   }

365   [[nodiscard, gnu::flatten]] static constexpr mat<F, 6, 6>
366   coadjoint(const SE3 &g) noexcept
367   {
368     const auto R3 = SO3<F>::to_matrix(g.R);
369     const F tx = g.t.data[0], ty = g.t.data[1], tz = g.t.data[2];
370     F tr[9];
371     tr[0] = -tz * R3.data[3] + ty * R3.data[6];
372     tr[1] = -tz * R3.data[4] + ty * R3.data[7];
373     tr[2] = -tz * R3.data[5] + ty * R3.data[8];
374     tr[3] = tz * R3.data[0] - tx * R3.data[6];
375     tr[4] = tz * R3.data[1] - tx * R3.data[7];
376     tr[5] = tz * R3.data[2] - tx * R3.data[8];
377     tr[6] = -ty * R3.data[0] + tx * R3.data[3];
378     tr[7] = -ty * R3.data[1] + tx * R3.data[4];
379     tr[8] = -ty * R3.data[2] + tx * R3.data[5];
380     mat<F, 6, 6> Ad = mat<F, 6, 6>::zero();
381     for ( usize r = 0; r < 3; ++r )
382       for ( usize c = 0; c < 3; ++c ) {
383         Ad.data[r * 6 + c] = R3.data[r * 3 + c];
384         Ad.data[(r + 3) * 6 + c] = tr[r * 3 + c];
385         Ad.data[(r + 3) * 6 + (c + 3)] = R3.data[r * 3 + c];
386       }
387     return Ad;
388   }

390   [[nodiscard, gnu::flatten]] static constexpr vec<F, 6>
391   adjoint_apply(const SE3 &g, const vec<F, 6> &xi) noexcept
392   {
393     const vec<F, 3> v{ xi.data[0], xi.data[1], xi.data[2] };
394     const vec<F, 3> w{ xi.data[3], xi.data[4], xi.data[5] };
395     const auto Rv = SO3<F>::rotate(g.R, v);
396     const auto Rw = SO3<F>::rotate(g.R, w);
397     const auto top = Rv + linalg::ops::cross(g.t, Rw);
398     return vec<F, 6>{ top.data[0], top.data[1], top.data[2], Rw.data[0], Rw.data[1], Rw.data[2] };

402   coadjoint_apply(const SE3 &g, const vec<F, 6> &wrench) noexcept
