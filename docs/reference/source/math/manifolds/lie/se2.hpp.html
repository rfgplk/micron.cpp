34 // Taylor coefficients of sin(w)/w as a polynomial in w**2
35 template<ieee754_floating F>
36 [[nodiscard, gnu::always_inline]] inline constexpr F
37 v_a_poly(F w2) noexcept
38 {
39   F r = F(1) / F(362880);
40   r = r * w2 - F(1) / F(5040);
41   r = r * w2 + F(1) / F(120);
42   r = r * w2 - F(1) / F(6);
43   r = r * w2 + F(1);
44   return r;
45 }
46 
47 // Taylor coefficients of (1 - cos w)/w**2 as a polynomial in w**2
48 template<ieee754_floating F>
49 [[nodiscard, gnu::always_inline]] inline constexpr F
50 v_b_poly(F w2) noexcept
51 {
52   F r = F(1) / F(3628800);
53   r = r * w2 - F(1) / F(40320);
54   r = r * w2 + F(1) / F(720);
55   r = r * w2 - F(1) / F(24);
56   r = r * w2 + F(0.5);
57   return r;
58 }

62 template<ieee754_floating F> struct SE2 {
63   SO2<F> R;
64   vec<F, 2> t;
65 
66   [[nodiscard, gnu::always_inline]] static constexpr SE2
67   identity() noexcept
68   {
69     return SE2{ SO2<F>::identity(), vec<F, 2>{ F(0), F(0) } };
70   }
71 
72   [[nodiscard, gnu::always_inline]] static constexpr SE2
73   from_rt(const SO2<F> &rotation, const vec<F, 2> &translation) noexcept
74   {
75     return SE2{ rotation, translation };
76   }
77 
78   [[nodiscard, gnu::always_inline]] static constexpr SE2
79   from_matrix(const mat<F, 3, 3> &T) noexcept
80   {
81     const mat<F, 2, 2> R2{ T.data[0], T.data[1], T.data[3], T.data[4] };
82     return SE2{ SO2<F>::from_matrix(R2), vec<F, 2>{ T.data[2], T.data[5] } };
83   }
84 
85   [[nodiscard, gnu::flatten]] static constexpr SE2
86   compose(const SE2 &a, const SE2 &b) noexcept
87   {
88     const auto Rb = SO2<F>::rotate(a.R, b.t);
89     return SE2{ SO2<F>::compose(a.R, b.R), a.t + Rb };
90   }
91 
92   [[nodiscard, gnu::flatten]] static constexpr SE2
93   inverse(const SE2 &g) noexcept
94   {
95     const auto Rinv = SO2<F>::inverse(g.R);
96     const auto tinv = SO2<F>::rotate(Rinv, vec<F, 2>{ -g.t.data[0], -g.t.data[1] });
97     return SE2{ Rinv, tinv };
98   }
99 
100   [[nodiscard, gnu::flatten]] static constexpr SE2
101   between(const SE2 &a, const SE2 &b) noexcept
102   {
103     const auto Rinv = SO2<F>::inverse(a.R);
104     return SE2{ SO2<F>::compose(Rinv, b.R), SO2<F>::rotate(Rinv, b.t - a.t) };
105   }
106 
107   [[nodiscard, gnu::flatten]] static constexpr SE2
108   exp_map(const vec<F, 3> &xi) noexcept
109   {
110     const F vx = xi.data[0], vy = xi.data[1], omega = xi.data[2];
111     const F w2 = omega * omega;
112     F a, b;      // V = [[a, -b], [b, a]]
113     if ( w2 < F(0.01) ) {
114       a = __se2_impl::v_a_poly<F>(w2);
115       b = omega * __se2_impl::v_b_poly<F>(w2);
116     } else {
117       F s, c;
118       math::sincos<F>(omega, s, c);
119       a = s / omega;
120       b = (F(1) - c) / omega;
121     }
122     return SE2{ SO2<F>{ omega }, vec<F, 2>{ a * vx - b * vy, b * vx + a * vy } };
123   }
124 
125   [[nodiscard, gnu::flatten]] static constexpr vec<F, 3>
126   log_map(const SE2 &g) noexcept
127   {
128     const F two_pi = F(2) * math::constant_pi<F>;
129     const F ratio = g.R.theta / two_pi;
130     const F k = static_cast<F>(static_cast<i64>(ratio + (ratio >= F(0) ? F(0.5) : F(-0.5))));
131     const F omega = g.R.theta - two_pi * k;
132     const F w2 = omega * omega;
133     F a, b;      // V = [[a, -b], [b, a]]
134     // threshold matched to SO3/SE3 (|omega| < 0.1)
135     if ( w2 < F(0.01) ) {
136       a = __se2_impl::v_a_poly<F>(w2);
137       b = omega * __se2_impl::v_b_poly<F>(w2);
138     } else {
139       F s, c;
140       math::sincos<F>(omega, s, c);
141       a = s / omega;
142       b = (F(1) - c) / omega;
143     }
144     const F det = a * a + b * b;
145     const F inv = F(1) / det;
146     const F vx = (a * g.t.data[0] + b * g.t.data[1]) * inv;
147     const F vy = (-b * g.t.data[0] + a * g.t.data[1]) * inv;
148     return vec<F, 3>{ vx, vy, omega };
149   }
150 
151   [[nodiscard, gnu::flatten]] static constexpr mat<F, 3, 3>
152   to_matrix(const SE2 &g) noexcept
153   {
154     const auto R2 = SO2<F>::to_matrix(g.R);
155     mat<F, 3, 3> M{};
156     M.data[0] = R2.data[0];
157     M.data[1] = R2.data[1];
158     M.data[2] = g.t.data[0];
159     M.data[3] = R2.data[2];
160     M.data[4] = R2.data[3];
161     M.data[5] = g.t.data[1];
162     M.data[6] = F(0);
163     M.data[7] = F(0);
164     M.data[8] = F(1);
165     return M;
166   }
167 
168   [[nodiscard, gnu::always_inline]] static constexpr vec<F, 2>
169   act(const SE2 &g, const vec<F, 2> &point) noexcept
170   {
171     return SO2<F>::rotate(g.R, point) + g.t;
172   }
173 
174   [[nodiscard, gnu::always_inline]] static constexpr vec<F, 2>
175   act_vector(const SE2 &g, const vec<F, 2> &direction) noexcept
176   {
177     return SO2<F>::rotate(g.R, direction);
178   }
179 
180   [[nodiscard, gnu::always_inline]] static constexpr vec<F, 2>
181   inverse_act(const SE2 &g, const vec<F, 2> &point) noexcept
182   {
183     return SO2<F>::inverse_rotate(g.R, point - g.t);
184   }
185 
186   [[nodiscard, gnu::flatten]] static constexpr mat<F, 3, 3>
187   adjoint(const SE2 &g) noexcept
188   {
189     const auto R2 = SO2<F>::to_matrix(g.R);
190     return mat<F, 3, 3>{ R2.data[0], R2.data[1], g.t.data[1], R2.data[2], R2.data[3], -g.t.data[0], F(0), F(0), F(1) };
191   }
192 
193   [[nodiscard, gnu::flatten]] static constexpr vec<F, 3>
194   adjoint_apply(const SE2 &g, const vec<F, 3> &xi) noexcept
195   {
196     const auto v = SO2<F>::rotate(g.R, vec<F, 2>{ xi.data[0], xi.data[1] });
197     return vec<F, 3>{ v.data[0] + g.t.data[1] * xi.data[2], v.data[1] - g.t.data[0] * xi.data[2], xi.data[2] };
198   }
199 
200   [[nodiscard, gnu::flatten]] static constexpr vec<F, 3>
201   coadjoint_apply(const SE2 &g, const vec<F, 3> &wrench) noexcept
202   {
203     const auto force = SO2<F>::rotate(g.R, vec<F, 2>{ wrench.data[0], wrench.data[1] });
204     return vec<F, 3>{ force.data[0], force.data[1], wrench.data[2] + g.t.data[0] * force.data[1] - g.t.data[1] * force.data[0] };
205   }
206 
207   [[nodiscard, gnu::flatten]] static constexpr SE2
208   interpolate(const SE2 &a, const SE2 &b, F t) noexcept
209   {
210     return compose(a, exp_map(log_map(between(a, b)) * t));
211   }
