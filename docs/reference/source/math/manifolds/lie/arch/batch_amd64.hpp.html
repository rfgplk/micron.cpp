27 [[gnu::always_inline]] inline void
28 __transpose4_f32(simd::f128 a, simd::f128 b, simd::f128 c, simd::f128 d, simd::f128 &x, simd::f128 &y, simd::f128 &z) noexcept
29 {
30   const simd::f128 ab0 = simd::sse::unpack_lo_f32(a, b);
31   const simd::f128 ab1 = simd::sse::unpack_hi_f32(a, b);
32   const simd::f128 cd0 = simd::sse::unpack_lo_f32(c, d);
33   const simd::f128 cd1 = simd::sse::unpack_hi_f32(c, d);
34   x = simd::sse::shuffle_f32<0x44>(ab0, cd0);
35   y = simd::sse::shuffle_f32<0xee>(ab0, cd0);
36   z = simd::sse::shuffle_f32<0x44>(ab1, cd1);
37 }

53 [[gnu::always_inline]] inline void
54 __rotate_f32(simd::f128 qx, simd::f128 qy, simd::f128 qz, simd::f128 qw, simd::f128 &x, simd::f128 &y, simd::f128 &z) noexcept
55 {
56   const simd::f128 tx = __vsimd::__fms(qy, z, __vsimd::__mul(qz, y));
57   const simd::f128 ty = __vsimd::__fms(qz, x, __vsimd::__mul(qx, z));
58   const simd::f128 tz = __vsimd::__fms(qx, y, __vsimd::__mul(qy, x));
59   const simd::f128 ux = __vsimd::__fms(qy, tz, __vsimd::__mul(qz, ty));
60   const simd::f128 uy = __vsimd::__fms(qz, tx, __vsimd::__mul(qx, tz));
61   const simd::f128 uz = __vsimd::__fms(qx, ty, __vsimd::__mul(qy, tx));
62   const simd::f128 two = __vsimd::__splat(2.0f);
63   x = __vsimd::__fma(two, __vsimd::__fma(qw, tx, ux), x);
64   y = __vsimd::__fma(two, __vsimd::__fma(qw, ty, uy), y);
65   z = __vsimd::__fma(two, __vsimd::__fma(qw, tz, uz), z);
66 }

87 template<bool Translate>
88 [[gnu::always_inline]] inline void
89 __apply_f32(const quat<f32> &q, const vec<f32, 3> &translation, const vec<f32, 3> *in, vec<f32, 3> *out, usize n) noexcept
90 {
91   const simd::f128 qx = __vsimd::__splat(q.data[0]);
92   const simd::f128 qy = __vsimd::__splat(q.data[1]);
93   const simd::f128 qz = __vsimd::__splat(q.data[2]);
94   const simd::f128 qw = __vsimd::__splat(q.data[3]);
95   usize i = 0;
96   for ( ; i + 4 <= n; i += 4 ) {
97     const float *ip = reinterpret_cast<const float *>(in + i);
98     simd::f128 x, y, z;
99     __transpose4_f32(simd::sse::load_f32(ip), simd::sse::load_f32(ip + 4), simd::sse::load_f32(ip + 8), simd::sse::load_f32(ip + 12), x, y,
100                      z);
101 
102     __rotate_f32(qx, qy, qz, qw, x, y, z);
103     if constexpr ( Translate ) {
104       x = __vsimd::__add(x, __vsimd::__splat(translation.data[0]));
105       y = __vsimd::__add(y, __vsimd::__splat(translation.data[1]));
106       z = __vsimd::__add(z, __vsimd::__splat(translation.data[2]));
107     }
108 
109     simd::f128 a, b, c, d;
110     __transpose4_f32_inv(x, y, z, a, b, c, d);
111     float *op = reinterpret_cast<float *>(out + i);
112     simd::sse::store_f32(op, a);
113     simd::sse::store_f32(op + 4, b);
114     simd::sse::store_f32(op + 8, c);
115     simd::sse::store_f32(op + 12, d);
116   }
117   for ( ; i < n; ++i ) {
118     out[i] = rotate(q, in[i]);
119     if constexpr ( Translate ) out[i] += translation;
120   }
121 }

177   }
