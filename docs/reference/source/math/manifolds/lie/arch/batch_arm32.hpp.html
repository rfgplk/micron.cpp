27 [[gnu::always_inline]] inline void
28 __transpose4_f32(simd::f128 a, simd::f128 b, simd::f128 c, simd::f128 d, simd::f128 &x, simd::f128 &y, simd::f128 &z) noexcept
29 {
30   const simd::f128 ab0 = simd::neon::zip_lo_f32(a, b);
31   const simd::f128 ab1 = simd::neon::zip_hi_f32(a, b);
32   const simd::f128 cd0 = simd::neon::zip_lo_f32(c, d);
33   const simd::f128 cd1 = simd::neon::zip_hi_f32(c, d);
34   x = simd::neon::concat_lo_f32(ab0, cd0);
35   y = simd::neon::concat_hi_f32(ab0, cd0);
36   z = simd::neon::concat_lo_f32(ab1, cd1);
37 }

39 [[gnu::always_inline]] inline void
40 __transpose4_f32_inv(simd::f128 x, simd::f128 y, simd::f128 z, simd::f128 &a, simd::f128 &b, simd::f128 &c, simd::f128 &d) noexcept
41 {
42   const simd::f128 zero = simd::neon::splat_f32(0.0f);
43   const simd::f128 xy0 = simd::neon::zip_lo_f32(x, y);
44   const simd::f128 xy1 = simd::neon::zip_hi_f32(x, y);
45   const simd::f128 z00 = simd::neon::zip_lo_f32(z, zero);
46   const simd::f128 z01 = simd::neon::zip_hi_f32(z, zero);
47   a = simd::neon::concat_lo_f32(xy0, z00);
48   b = simd::neon::concat_hi_f32(xy0, z00);
49   c = simd::neon::concat_lo_f32(xy1, z01);
50   d = simd::neon::concat_hi_f32(xy1, z01);
51 }

53 [[gnu::always_inline]] inline void
54 __rotate_f32(simd::f128 qx, simd::f128 qy, simd::f128 qz, simd::f128 qw, simd::f128 &x, simd::f128 &y, simd::f128 &z) noexcept
55 {
56   const simd::f128 tx = simd::neon::fms_f32(simd::neon::mul(qy, z), qz, y);
57   const simd::f128 ty = simd::neon::fms_f32(simd::neon::mul(qz, x), qx, z);
58   const simd::f128 tz = simd::neon::fms_f32(simd::neon::mul(qx, y), qy, x);
59   const simd::f128 ux = simd::neon::fms_f32(simd::neon::mul(qy, tz), qz, ty);
60   const simd::f128 uy = simd::neon::fms_f32(simd::neon::mul(qz, tx), qx, tz);
61   const simd::f128 uz = simd::neon::fms_f32(simd::neon::mul(qx, ty), qy, tx);
62   const simd::f128 two = simd::neon::splat_f32(2.0f);
63   x = __vsimd::__fma(two, __vsimd::__fma(qw, tx, ux), x);
64   y = __vsimd::__fma(two, __vsimd::__fma(qw, ty, uy), y);
65   z = __vsimd::__fma(two, __vsimd::__fma(qw, tz, uz), z);
66 }

68 template<bool Translate>
69 [[gnu::always_inline]] inline void
70 __apply_f32(const quat<f32> &q, const vec<f32, 3> &translation, const vec<f32, 3> *in, vec<f32, 3> *out, usize n) noexcept
71 {
72   const simd::f128 qx = simd::neon::splat_f32(q.data[0]);
73   const simd::f128 qy = simd::neon::splat_f32(q.data[1]);
74   const simd::f128 qz = simd::neon::splat_f32(q.data[2]);
75   const simd::f128 qw = simd::neon::splat_f32(q.data[3]);
76   usize i = 0;
77   for ( ; i + 4 <= n; i += 4 ) {
78     const float *ip = reinterpret_cast<const float *>(in + i);
79     simd::f128 x, y, z;
80     __transpose4_f32(simd::neon::load_f32(ip), simd::neon::load_f32(ip + 4), simd::neon::load_f32(ip + 8), simd::neon::load_f32(ip + 12), x,
81                      y, z);
82 
83     __rotate_f32(qx, qy, qz, qw, x, y, z);
84     if constexpr ( Translate ) {
85       x = simd::neon::add(x, simd::neon::splat_f32(translation.data[0]));
86       y = simd::neon::add(y, simd::neon::splat_f32(translation.data[1]));
87       z = simd::neon::add(z, simd::neon::splat_f32(translation.data[2]));
88     }
89 
90     simd::f128 a, b, c, d;
91     __transpose4_f32_inv(x, y, z, a, b, c, d);
92     float *op = reinterpret_cast<float *>(out + i);
93     simd::neon::store_f32(op, a);
94     simd::neon::store_f32(op + 4, b);
95     simd::neon::store_f32(op + 8, c);
96     simd::neon::store_f32(op + 12, d);
97   }
98   for ( ; i < n; ++i ) {
99     out[i] = rotate(q, in[i]);
100     if constexpr ( Translate ) out[i] += translation;
101   }
102 }

104 template<bool Translate>
105 [[gnu::always_inline]] inline void
106 __apply_soa_f32(const quat<f32> &q, const vec<f32, 3> &translation, const f32 *in_x, const f32 *in_y, const f32 *in_z, f32 *out_x,
107                 f32 *out_y, f32 *out_z, usize n) noexcept
108 {
109   const simd::f128 qx = simd::neon::splat_f32(q.data[0]);
110   const simd::f128 qy = simd::neon::splat_f32(q.data[1]);
111   const simd::f128 qz = simd::neon::splat_f32(q.data[2]);
112   const simd::f128 qw = simd::neon::splat_f32(q.data[3]);
113   usize i = 0;
114   for ( ; i + 4 <= n; i += 4 ) {
115     simd::f128 x = simd::neon::load_f32(reinterpret_cast<const float *>(in_x + i));
116     simd::f128 y = simd::neon::load_f32(reinterpret_cast<const float *>(in_y + i));
117     simd::f128 z = simd::neon::load_f32(reinterpret_cast<const float *>(in_z + i));
118     __rotate_f32(qx, qy, qz, qw, x, y, z);
119     if constexpr ( Translate ) {
120       x = simd::neon::add(x, simd::neon::splat_f32(translation.data[0]));
121       y = simd::neon::add(y, simd::neon::splat_f32(translation.data[1]));
122       z = simd::neon::add(z, simd::neon::splat_f32(translation.data[2]));
123     }
124     simd::neon::store_f32(reinterpret_cast<float *>(out_x + i), x);
125     simd::neon::store_f32(reinterpret_cast<float *>(out_y + i), y);
126     simd::neon::store_f32(reinterpret_cast<float *>(out_z + i), z);
127   }
128   for ( ; i < n; ++i ) {
129     const auto r = rotate(q, vec<f32, 3>{ in_x[i], in_y[i], in_z[i] });
130     out_x[i] = r.data[0];
131     out_y[i] = r.data[1];
132     out_z[i] = r.data[2];
133     if constexpr ( Translate ) {
134       out_x[i] += translation.data[0];
135       out_y[i] += translation.data[1];
136       out_z[i] += translation.data[2];
137     }
138   }
139 }
