25 [[nodiscard, gnu::always_inline]] inline vec<f32, 3>
26 rotate(const quat<f32> &q, const vec<f32, 3> &v) noexcept
27 {
28   const simd::f128 qv = __vsimd::__zero3(__vsimd::__load(reinterpret_cast<const float *>(q.data)));
29   const simd::f128 vv = __vsimd::__setr(v.data[0], v.data[1], v.data[2], 0.0f);
30   const simd::f128 t = __vsimd::__cross3(qv, vv);
31   const simd::f128 u = __vsimd::__cross3(qv, t);
32   const simd::f128 r = __vsimd::__fma(__vsimd::__splat(2.0f), __vsimd::__fma(__vsimd::__splat(q.data[3]), t, u), vv);
33   vec<f32, 3> out{};
34   __vsimd::__store(reinterpret_cast<float *>(out.data), r);
35   return out;
36 }

40 [[nodiscard, gnu::always_inline]] inline vec<f64, 3>
41 rotate(const quat<f64> &q, const vec<f64, 3> &v) noexcept
42 {
43   const simd::d128 qx = simd::neon::splat_f64(q.data[0]);
44   const simd::d128 qy = simd::neon::splat_f64(q.data[1]);
45   const simd::d128 qz = simd::neon::splat_f64(q.data[2]);
46   const simd::d128 qw = simd::neon::splat_f64(q.data[3]);
47   simd::d128 x = simd::neon::splat_f64(v.data[0]);
48   simd::d128 y = simd::neon::splat_f64(v.data[1]);
49   simd::d128 z = simd::neon::splat_f64(v.data[2]);
50   const simd::d128 tx = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qz, y)), qy, z);
51   const simd::d128 ty = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qx, z)), qz, x);
52   const simd::d128 tz = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qy, x)), qx, y);
53   const simd::d128 ux = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qz, ty)), qy, tz);
54   const simd::d128 uy = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qx, tz)), qz, tx);
55   const simd::d128 uz = simd::neon::fma_f64(simd::neon::neg(simd::neon::mul(qy, tx)), qx, ty);
56   const simd::d128 two = simd::neon::splat_f64(2.0);
57   x = simd::neon::fma_f64(x, two, simd::neon::fma_f64(ux, qw, tx));
58   y = simd::neon::fma_f64(y, two, simd::neon::fma_f64(uy, qw, ty));
59   z = simd::neon::fma_f64(z, two, simd::neon::fma_f64(uz, qw, tz));
60   return vec<f64, 3>{ simd::neon::get_lane_f64<0>(x), simd::neon::get_lane_f64<0>(y), simd::neon::get_lane_f64<0>(z) };
61 }
