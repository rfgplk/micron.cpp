39 template<ieee754_floating F, usize N>
40 [[nodiscard, gnu::noinline]] inline linalg::decomp::eigen_sym_result<F, N>
41 eigensystem(const mat<F, N, N> &P) noexcept
42 {
43   return linalg::decomp::eigen_sym<F, N>(P);
44 }

46 template<ieee754_floating F, usize N>
47 [[nodiscard, gnu::noinline]] inline mat<F, N, N>
48 reconstruct(const linalg::decomp::eigen_sym_result<F, N> &er, const vec<F, N> &values) noexcept
49 {
50   mat<F, N, N> M{};
51   for ( usize r = 0; r < N; ++r ) {
52     for ( usize c = 0; c < N; ++c ) {
53       F acc = F(0);
54       for ( usize k = 0; k < N; ++k ) acc += er.vectors.data[r * N + k] * values.data[k] * er.vectors.data[c * N + k];
55       M.data[r * N + c] = acc;
56     }
57   }
58   return M;
59 }

61 template<ieee754_floating F, usize N, typename Apply>
62 [[nodiscard]] inline mat<F, N, N>
63 sym_func(const mat<F, N, N> &P, Apply apply) noexcept
64 {
65   auto er = eigensystem<F, N>(P);
66   vec<F, N> fl{};
67   for ( usize i = 0; i < N; ++i ) fl.data[i] = apply(er.values.data[i]);
68   return reconstruct<F, N>(er, fl);
69 }

71 template<ieee754_floating F, usize N, typename ApplyFirst, typename ApplySecond>
72 [[nodiscard]] inline sym_pair_result<F, N>
73 sym_func_pair(const mat<F, N, N> &P, ApplyFirst apply_first, ApplySecond apply_second) noexcept
74 {
75   auto er = eigensystem<F, N>(P);
76   vec<F, N> fa{}, fb{};
77   for ( usize i = 0; i < N; ++i ) {
78     fa.data[i] = apply_first(er.values.data[i]);
79     fb.data[i] = apply_second(er.values.data[i]);
80   }
81   return sym_pair_result<F, N>{ reconstruct<F, N>(er, fa), reconstruct<F, N>(er, fb) };
82 }

84 template<ieee754_floating F, usize N>
85 [[nodiscard]] inline mat<F, N, N>
86 symmetrize(const mat<F, N, N> &A) noexcept
87 {
88   mat<F, N, N> S{};
89   for ( usize r = 0; r < N; ++r ) {
90     S.data[r * N + r] = F(0.5) * (A.data[r * N + r] + A.data[r * N + r]);
91     for ( usize c = r + 1; c < N; ++c ) {
92       const F v = F(0.5) * (A.data[r * N + c] + A.data[c * N + r]);
93       S.data[r * N + c] = v;
94       S.data[c * N + r] = v;
95     }
96   }
97   return S;
98 }

100 template<ieee754_floating F, usize N>
101 [[nodiscard, gnu::flatten]] inline F
102 frobenius_norm(const mat<F, N, N> &A) noexcept
103 {
104   F s = F(0);
105   for ( usize i = 0; i < N * N; ++i ) s = math::fma<F>(A.data[i], A.data[i], s);
106   return math::fsqrt(s);
107 }

109 template<ieee754_floating F, usize N>

120 // only the affine-invariant and log-Euclidean arms exist; euclidean_metric and bures_metric
121 // satisfy metric_tag but would silently take the affine-invariant branch
122 template<typename T>
123 concept spd_metric_tag = micron::is_same_v<T, affine_invariant_metric> || micron::is_same_v<T, log_euclidean_metric>;
124 
125 template<ieee754_floating F, usize N>
126   requires(N >= 2)
127 struct spd {
128   using value_type = F;
129   static constexpr usize dim_n = N;
130 
131   [[nodiscard]] static mat<F, N, N>
132   sqrt_pd(const mat<F, N, N> &P) noexcept
133   {
134     return __spd_impl::sym_func<F, N>(P, [](F x) noexcept { return math::fsqrt(x); });
135   }
136 
137   [[nodiscard]] static mat<F, N, N>
138   inv_sqrt_pd(const mat<F, N, N> &P) noexcept
139   {
140     const F eps = math::default_eps<F>();
141     return __spd_impl::sym_func<F, N>(P, [eps](F x) noexcept { return F(1) / math::fsqrt(x > eps ? x : eps); });
142   }
143 
144   struct sqrt_pair {
145     mat<F, N, N> root;
146     mat<F, N, N> inverse_root;
147   };
148 
149   [[nodiscard]] static sqrt_pair
150   sqrt_pair_pd(const mat<F, N, N> &P) noexcept
151   {
152     const F eps = math::default_eps<F>();
153     auto pair = __spd_impl::sym_func_pair<F, N>(
154         P, [](F x) noexcept { return math::fsqrt(x); }, [eps](F x) noexcept { return F(1) / math::fsqrt(x > eps ? x : eps); });
155     return sqrt_pair{ pair.first, pair.second };
156   }
157 
158   [[nodiscard]] static mat<F, N, N>
159   inverse_pd(const mat<F, N, N> &P) noexcept
160   {
161     const F eps = math::default_eps<F>();
162     return __spd_impl::sym_func<F, N>(P, [eps](F x) noexcept { return F(1) / (x > eps ? x : eps); });
163   }
164 
165   [[nodiscard]] static mat<F, N, N>
166   project_to_manifold(const mat<F, N, N> &A) noexcept
167   {
168     auto S = __spd_impl::symmetrize<F, N>(A);
169     const F eps = math::default_eps<F>();
170     return __spd_impl::sym_func<F, N>(S, [eps](F x) noexcept { return x > eps ? x : eps; });
171   }
172 
173   [[nodiscard, gnu::always_inline]] static constexpr mat<F, N, N>
174   project_to_tangent(const mat<F, N, N> &, const mat<F, N, N> &V) noexcept
175   {
176     return __spd_impl::symmetrize<F, N>(V);
177   }
178 
179   template<spd_metric_tag Metric = affine_invariant_metric>
180   [[nodiscard]] static mat<F, N, N>
181   exp_map(const mat<F, N, N> &P, const mat<F, N, N> &V) noexcept
182   {
183     if constexpr ( micron::is_same_v<Metric, log_euclidean_metric> ) {
184       auto LP = linalg::matfunc::logm<F, N>(P).X;
185       auto S = __spd_impl::symmetrize<F, N>(V);
186       mat<F, N, N> sum{};
187       for ( usize i = 0; i < N * N; ++i ) sum.data[i] = LP.data[i] + S.data[i];
188       return linalg::matfunc::expm<F, N>(sum).X;
189     } else {
190       auto roots = sqrt_pair_pd(P);
191       auto Vs = __spd_impl::symmetrize<F, N>(V);
192       auto inner = linalg::ops::gemm(linalg::ops::gemm(roots.inverse_root, Vs), roots.inverse_root);
193       auto E = linalg::matfunc::expm<F, N>(inner).X;
194       return linalg::ops::gemm(linalg::ops::gemm(roots.root, E), roots.root);
195     }
196   }
197 
198   template<spd_metric_tag Metric = affine_invariant_metric>
199   [[nodiscard]] static mat<F, N, N>
200   log_map(const mat<F, N, N> &P, const mat<F, N, N> &Q) noexcept
201   {
202     if constexpr ( micron::is_same_v<Metric, log_euclidean_metric> ) {
203       auto LP = linalg::matfunc::logm<F, N>(P).X;
204       auto LQ = linalg::matfunc::logm<F, N>(Q).X;
205       mat<F, N, N> r{};
206       for ( usize i = 0; i < N * N; ++i ) r.data[i] = LQ.data[i] - LP.data[i];
207       return r;
208     } else {
209       auto roots = sqrt_pair_pd(P);
210       auto inner = linalg::ops::gemm(linalg::ops::gemm(roots.inverse_root, Q), roots.inverse_root);
211       auto L = linalg::matfunc::logm<F, N>(inner).X;
212       return linalg::ops::gemm(linalg::ops::gemm(roots.root, L), roots.root);
213     }
214   }

216   template<spd_metric_tag Metric = affine_invariant_metric>
217   [[nodiscard]] static F
218   distance(const mat<F, N, N> &P, const mat<F, N, N> &Q) noexcept
219   {
220     if constexpr ( micron::is_same_v<Metric, log_euclidean_metric> ) {
221       auto LP = linalg::matfunc::logm<F, N>(P).X;
222       auto LQ = linalg::matfunc::logm<F, N>(Q).X;
223       mat<F, N, N> diff{};
224       for ( usize i = 0; i < N * N; ++i ) diff.data[i] = LP.data[i] - LQ.data[i];
225       return __spd_impl::frobenius_norm<F, N>(diff);
226     } else {
227       auto P_neg_half = inv_sqrt_pd(P);
228       auto inner = linalg::ops::gemm(linalg::ops::gemm(P_neg_half, Q), P_neg_half);
229       auto L = linalg::matfunc::logm<F, N>(inner).X;
230       return __spd_impl::frobenius_norm<F, N>(L);
231     }
232   }

234   template<spd_metric_tag Metric = affine_invariant_metric>
235   [[nodiscard]] static F
236   squared_distance(const mat<F, N, N> &P, const mat<F, N, N> &Q) noexcept
237   {
238     if constexpr ( micron::is_same_v<Metric, log_euclidean_metric> ) {
239       auto LP = linalg::matfunc::logm<F, N>(P).X;
240       auto LQ = linalg::matfunc::logm<F, N>(Q).X;
241       mat<F, N, N> diff{};
242       for ( usize i = 0; i < N * N; ++i ) diff.data[i] = LP.data[i] - LQ.data[i];
243       return __spd_impl::frobenius_squared<F, N>(diff);
244     } else {
245       auto P_neg_half = inv_sqrt_pd(P);
246       auto inner = linalg::ops::gemm(linalg::ops::gemm(P_neg_half, Q), P_neg_half);
247       auto L = linalg::matfunc::logm<F, N>(inner).X;
248       return __spd_impl::frobenius_squared<F, N>(L);
249     }
250   }

252   [[nodiscard]] static mat<F, N, N>
253   retract(const mat<F, N, N> &P, const mat<F, N, N> &V) noexcept
254   {
255     auto Vs = __spd_impl::symmetrize<F, N>(V);
256     auto Pinv = inverse_pd(P);
257     auto VP_invV = linalg::ops::gemm(linalg::ops::gemm(Vs, Pinv), Vs);
258     mat<F, N, N> R{};
259     for ( usize i = 0; i < N * N; ++i ) R.data[i] = P.data[i] + Vs.data[i] + F(0.5) * VP_invV.data[i];
260     return project_to_manifold(R);
261   }

263   [[nodiscard]] static mat<F, N, N>
264   inverse_retract(const mat<F, N, N> &P, const mat<F, N, N> &Q) noexcept
265   {
266     return log_map<affine_invariant_metric>(P, Q);
267   }

269   template<spd_metric_tag Metric = affine_invariant_metric>
270   [[nodiscard]] static F
271   inner(const mat<F, N, N> &P, const mat<F, N, N> &U, const mat<F, N, N> &V) noexcept
272   {
273     if constexpr ( micron::is_same_v<Metric, log_euclidean_metric> ) {
274       F s = F(0);
275       for ( usize i = 0; i < N * N; ++i ) s = math::fma<F>(U.data[i], V.data[i], s);
276       return s;
277     } else {
278       auto Pinv = inverse_pd(P);
279       auto A = linalg::ops::gemm(Pinv, U);
280       auto B = linalg::ops::gemm(Pinv, V);
281       F s = F(0);
282       for ( usize i = 0; i < N; ++i )
283         for ( usize j = 0; j < N; ++j ) s = math::fma<F>(A.data[i * N + j], B.data[j * N + i], s);
284       return s;
285     }
286   }

288   template<spd_metric_tag Metric = affine_invariant_metric>
289   [[nodiscard]] static F
290   norm(const mat<F, N, N> &P, const mat<F, N, N> &V) noexcept
291   {
292     return math::fsqrt(inner<Metric>(P, V, V));
293   }
294 
295   [[nodiscard]] static mat<F, N, N>
296   parallel_transport(const mat<F, N, N> &P, const mat<F, N, N> &Q, const mat<F, N, N> &V) noexcept
297   {
298     auto Pinv = inverse_pd(P);
299     auto QP = linalg::ops::gemm(Q, Pinv);
300     auto E = linalg::matfunc::sqrtm<F, N>(QP).X;
301     auto Et = linalg::ops::transpose(E);
302     return linalg::ops::gemm(linalg::ops::gemm(E, V), Et);
303   }
304 
305   [[nodiscard, gnu::always_inline]] static mat<F, N, N>
306   vector_transport(const mat<F, N, N> &P, const mat<F, N, N> &Q, const mat<F, N, N> &V) noexcept
307   {
308     return parallel_transport(P, Q, V);
309   }
