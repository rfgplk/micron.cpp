32 template<ieee754_floating F, usize N, usize P>
33 [[nodiscard, gnu::flatten]] inline mat<F, P, P>
34 xt_y(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
35 {
36   mat<F, P, P> M{};
37   for ( usize i = 0; i < P; ++i )
38     for ( usize j = 0; j < P; ++j ) {
39       F acc = F(0);
40       for ( usize k = 0; k < N; ++k ) acc = math::fma<F>(X.data[k * P + i], Y.data[k * P + j], acc);
41       M.data[i * P + j] = acc;
42     }
43   return M;
44 }

48 // NOTE: P >= 2 is required
49 template<ieee754_floating F, usize N, usize P>
50   requires(N >= P && P >= 2)
51 struct grassmann {
52   using value_type = F;
53   static constexpr usize rows = N;
54   static constexpr usize cols = P;
55 
56   [[nodiscard, gnu::flatten]] static mat<F, N, P>
57   project_to_tangent(const mat<F, N, P> &X, const mat<F, N, P> &V) noexcept
58   {
59     auto A = __grassmann_impl::xt_y<F, N, P>(X, V);
60     auto XA = linalg::ops::gemm(X, A);
61     mat<F, N, P> R{};
62     for ( usize i = 0; i < N * P; ++i ) R.data[i] = V.data[i] - XA.data[i];
63     return R;
64   }
65 
66   [[nodiscard]] static mat<F, N, P>
67   exp_map(const mat<F, N, P> &X, const mat<F, N, P> &V) noexcept
68   {
69     auto sv = linalg::decomp::svd<F, N, P>(V);
70     F cs[P], sn[P];
71     for ( usize i = 0; i < P; ++i ) {
72       math::sincos<F>(sv.S.data[i], sn[i], cs[i]);
73     }
74     auto XVv = linalg::ops::gemm(X, sv.V);
75     mat<F, N, P> XVcos{};
76     mat<F, N, P> Usin{};
77     for ( usize i = 0; i < N; ++i ) {
78       for ( usize k = 0; k < P; ++k ) {
79         XVcos.data[i * P + k] = XVv.data[i * P + k] * cs[k];
80         Usin.data[i * P + k] = sv.U.data[i * N + k] * sn[k];
81       }
82     }
83     mat<F, N, P> R{};
84     for ( usize r = 0; r < N; ++r ) {
85       for ( usize c = 0; c < P; ++c ) {
86         F acc = F(0);
87         for ( usize k = 0; k < P; ++k ) {
88           const F a = XVcos.data[r * P + k] + Usin.data[r * P + k];
89           acc = math::fma<F>(a, sv.V.data[c * P + k], acc);
90         }
91         R.data[r * P + c] = acc;
92       }
93     }
94     return R;
95   }

97   // Riemannian logarithm (Edelman-Arias-Smith)
98   [[nodiscard]] static mat<F, N, P>
99   log_map(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
100   {
101     const auto Minv = linalg::decomp::inv<F, P>(__grassmann_impl::xt_y<F, N, P>(X, Y));
102     if ( Minv.singular ) return inverse_retract(X, Y);
103     const auto Z = linalg::ops::gemm(Y, Minv.X);
104     mat<F, N, P> T{};
105     for ( usize i = 0; i < N * P; ++i ) T.data[i] = Z.data[i] - X.data[i];
106     const auto sv = linalg::decomp::svd<F, N, P>(T);
107     F th[P];
108     for ( usize i = 0; i < P; ++i ) th[i] = math::atan<F>(sv.S.data[i]);
109     mat<F, N, P> R{};
110     for ( usize r = 0; r < N; ++r ) {
111       for ( usize c = 0; c < P; ++c ) {
112         F acc = F(0);
113         for ( usize k = 0; k < P; ++k ) acc = math::fma<F>(sv.U.data[r * N + k] * th[k], sv.V.data[c * P + k], acc);
114         R.data[r * P + c] = acc;
115       }
116     }
117     return R;
118   }

120   [[nodiscard, gnu::always_inline]] static mat<F, N, P>
121   retract(const mat<F, N, P> &X, const mat<F, N, P> &V) noexcept
122   {
123     return exp_map(X, V);
124   }

126   [[nodiscard, gnu::always_inline]] static mat<F, N, P>
127   inverse_retract(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
128   {
129     mat<F, N, P> diff{};
130     for ( usize i = 0; i < N * P; ++i ) diff.data[i] = Y.data[i] - X.data[i];
131     return project_to_tangent(X, diff);
132   }
133 
134   [[nodiscard]] static vec<F, P>
135   principal_angles(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
136   {
137     auto M = __grassmann_impl::xt_y<F, N, P>(X, Y);
138     auto sv = linalg::decomp::svd<F, P, P>(M);
139     vec<F, P> angles{};
140     for ( usize i = 0; i < P; ++i ) {
141       F s = sv.S.data[i];
142       if ( s > F(1) ) s = F(1);
143       if ( s < F(-1) ) s = F(-1);
144       angles.data[i] = math::acos<F>(s);
145     }
146     return angles;
147   }

149   [[nodiscard, gnu::flatten]] static F
150   distance(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
151   {
152     return math::fsqrt(squared_distance(X, Y));
153   }
154 
155   [[nodiscard, gnu::flatten]] static F
156   squared_distance(const mat<F, N, P> &X, const mat<F, N, P> &Y) noexcept
157   {
158     auto a = principal_angles(X, Y);
159     F s = F(0);
160     for ( usize i = 0; i < P; ++i ) s = math::fma<F>(a.data[i], a.data[i], s);
161     return s;
162   }

164   [[nodiscard, gnu::flatten]] static F
165   inner(const mat<F, N, P> &, const mat<F, N, P> &U, const mat<F, N, P> &V) noexcept
166   {
167     F s = F(0);
168     for ( usize i = 0; i < N * P; ++i ) s = math::fma<F>(U.data[i], V.data[i], s);
169     return s;

172   [[nodiscard, gnu::always_inline]] static F
173   norm(const mat<F, N, P> &p, const mat<F, N, P> &v) noexcept
174   {
175     return math::fsqrt(inner(p, v, v));

178   [[nodiscard]] static mat<F, N, P>
179   project_to_manifold(const mat<F, N, P> &A) noexcept
180   {
181     auto s = linalg::decomp::svd<F, N, P>(A);
182     mat<F, N, P> Q{};
183     for ( usize r = 0; r < N; ++r ) {
184       for ( usize c = 0; c < P; ++c ) {
185         F acc = F(0);
186         for ( usize k = 0; k < P; ++k ) acc = math::fma<F>(s.U.data[r * N + k], s.V.data[c * P + k], acc);
187         Q.data[r * P + c] = acc;
188       }
189     }
190     return Q;
191   }

194   parallel_transport(const mat<F, N, P> &, const mat<F, N, P> &Y, const mat<F, N, P> &V) noexcept
195   {
196     return project_to_tangent(Y, V);
