44 #define __micron_math_builtin_fn(name)                                                                                                     \
45   template<ieee754_floating F> [[nodiscard, gnu::always_inline]] inline constexpr F bi_##name(F x) noexcept                                \
46   {                                                                                                                                        \
47     if constexpr ( sizeof(F) == sizeof(float) )                                                                                            \
48       return F(__builtin_##name##f(float(x)));                                                                                             \
49     else if constexpr ( sizeof(F) == sizeof(double) )                                                                                      \
50       return F(__builtin_##name(double(x)));                                                                                               \
51     else                                                                                                                                   \
52       return F(__builtin_##name##l(static_cast<long double>(x)));                                                                          \
53   }
54 
55 #define __micron_math_builtin_fn2(name)                                                                                                    \
56   template<ieee754_floating F> [[nodiscard, gnu::always_inline]] inline constexpr F bi_##name(F a, F b) noexcept                           \
57   {                                                                                                                                        \
58     if constexpr ( sizeof(F) == sizeof(float) )                                                                                            \
59       return F(__builtin_##name##f(float(a), float(b)));                                                                                   \
60     else if constexpr ( sizeof(F) == sizeof(double) )                                                                                      \
61       return F(__builtin_##name(double(a), double(b)));                                                                                    \
62     else                                                                                                                                   \
63       return F(__builtin_##name##l(static_cast<long double>(a), static_cast<long double>(b)));                                             \
64   }
