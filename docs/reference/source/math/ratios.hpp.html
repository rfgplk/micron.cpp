13 template<uintmax_t N, uintmax_t D = 1>
14   requires(D != 0)
15 class ratio
16 {
17   static constexpr uintmax_t
18   gcd(uintmax_t a, uintmax_t b)
19   {
20     return b == 0 ? a : gcd(b, a % b);
21   }
22 
23 public:
24   static constexpr uintmax_t num = N / gcd(N, D);
25   static constexpr uintmax_t denom = D / gcd(N, D);
26 };
