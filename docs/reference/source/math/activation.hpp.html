22 template<ieee754_floating F>
23 [[nodiscard, gnu::always_inline]] inline constexpr F
24 relu(F x) noexcept
25 {
26   return x > F(0) ? x : F(0);

32 {

37 [[nodiscard, gnu::always_inline]] inline constexpr F
38 elu(F x, F alpha = F(1)) noexcept
39 {
40   return x > F(0) ? x : alpha * mk::exp_ns::expm1<F>(x);

43 template<ieee754_floating F>
44 [[nodiscard, gnu::always_inline]] inline constexpr F
45 gelu(F x) noexcept
46 {
47   constexpr F inv_sqrt2 = F(0.7071067811865475);
48   return F(0.5) * x * (F(1) + mk::special::erf<F>(x * inv_sqrt2));
49 }

51 template<ieee754_floating F>
52 [[nodiscard, gnu::always_inline]] inline constexpr F
53 gelu_approx(F x) noexcept
54 {
55   constexpr F sqrt_2_over_pi = F(0.7978845608028654);
56   const F x3 = x * x * x;
57   const F u = sqrt_2_over_pi * (x + F(0.044715) * x3);
58   return F(0.5) * x * (F(1) + mk::hyp::tanh<F>(u));
59 }

90 [[nodiscard, gnu::always_inline]] inline constexpr F
91 softsign(F x) noexcept
92 {
93   return x / (F(1) + mk::manip::fabs<F>(x));

96 template<ieee754_floating F>
97 [[nodiscard, gnu::always_inline]] inline constexpr F
98 hardswish(F x) noexcept
99 {
100   if ( x <= F(-3) ) return F(0);
101   if ( x >= F(3) ) return x;
102   return x * (x + F(3)) * F(1.0 / 6.0);
103 }
