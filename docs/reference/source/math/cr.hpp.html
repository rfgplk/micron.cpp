28   template<ieee754_floating F>
29   [[nodiscard, gnu::always_inline]] inline constexpr F
30   ulp_of(F x) noexcept
31   {
32     using T = ieee::traits<F>;
33     using U = typename T::uint_type;
34     U b = ieee::to_bits(x) & ~T::sign_mask;
35     if ( b == 0 ) [[unlikely]] {
36       return ieee::from_bits<F>(U(1));
37     }
38     if ( b >= T::exp_mask ) [[unlikely]] {
39       return ieee::inf_v<F>(0);
40     }
41     U raw_e = (b & T::exp_mask) >> T::mant_bits;
42     if ( raw_e == 0 ) [[unlikely]] {
43       return ieee::from_bits<F>(U(1));
44     }
45     if ( raw_e <= U(T::mant_bits) ) {
46       return ieee::from_bits<F>(U(1));
47     }
48     U ulp_raw = (raw_e - U(T::mant_bits)) << T::mant_bits;
49     return ieee::from_bits<F>(ulp_raw);
50   }

57   [[nodiscard, gnu::always_inline]] inline constexpr bool
58   round_test_f64(dd64 y, f64 abs_err) noexcept
59   {
60     const f64 u = ulp_of<f64>(y.hi);
61     const f64 alo = mkbits::manip::fabs(y.lo);
62     // alo + abs_err < 0.5*u  ⇔  rounding boundary is uncrossable
63     return alo + abs_err < 0.5 * u;
64   }
65 
66   [[nodiscard, gnu::always_inline]] inline constexpr bool
67   round_test_f32(f64 y, f64 abs_err) noexcept
68   {
69     const f32 r32 = f32(y);
70     if ( ieee::is_nan(r32) || ieee::is_inf(r32) ) [[unlikely]]
71       return true;
72     const f64 u = f64(ulp_of<f32>(r32));
73     const f64 dist = mkbits::manip::fabs(y - f64(r32));
74     return dist + abs_err < 0.5 * u;
75   }

77   [[nodiscard, gnu::always_inline]] inline constexpr f64
78   round_dd_to_f64(dd64 y) noexcept
79   {
80     return y.hi + y.lo;
81   }

83   [[nodiscard, gnu::always_inline]] inline constexpr f32
84   round_dd_to_f32(dd64 y) noexcept
85   {
86     const dd64 t = dd::fast_two_sum(y.hi, y.lo);
87     const f64 s = t.hi;
88     const f64 resid = t.lo;
89     if ( resid == 0.0 ) {
90       return f32(s);
91     }
92     // round to odd
93     if ( ieee::to_bits(s) & u64(1) ) {
94       return f32(s);
95     }
96     return f32(resid > 0.0 ? ieee::next_up(s) : ieee::next_down(s));
97   }

99   template<typename Stage1, typename Stage2>
100   [[nodiscard, gnu::flatten]] inline constexpr f64
101   drive_f64(f64 x, Stage1 &&s1, Stage2 &&s2) noexcept
102   {
103     const ziv_result r1 = s1(x);
104     if ( round_test_f64(r1.y, r1.abs_err) ) [[likely]] {
105       return round_dd_to_f64(r1.y);
106     }
107     const ziv_result r2 = s2(x);
108     if ( round_test_f64(r2.y, r2.abs_err) ) [[likely]] {
109       return round_dd_to_f64(r2.y);
110     }
111     return round_dd_to_f64(r2.y);
112   }

148   [[nodiscard, gnu::always_inline]] inline constexpr f64
149   abs_err_bound(dd64 y) noexcept
150   {
151     const f64 ah = mkbits::manip::fabs(y.hi);
152     // 2^-72 · max(|hi|, 1).  -72 is the worst-case dd64 polynomial bound
153     // we trust the kernels to deliver; well below half-ulp_f64 for any
154     // |y| ≥ 2^-1022.
155     return (ah > 1.0 ? ah : 1.0) * 0x1.0p-72;
156   }

162   [[nodiscard, gnu::flatten]] inline constexpr f64
163   log_f64(f64 x) noexcept
164   {
165     bool handled = false;
166     const dd64 y = mkbits::log_ns::log_dd_f64_with_special(x, &handled);
167     if ( handled ) return round_dd_to_f64(y);
168     const f64 err = ziv_impl::abs_err_bound(y);
169     if ( round_test_f64(y, err) ) [[likely]]
170       return round_dd_to_f64(y);
171     return round_dd_to_f64(y);
172   }
