47 {
48   constexpr bool tr = micron::same_as<Op, op::trans> or micron::same_as<Op, op::conj_trans>;
49   bits::gemv_kernel<T>(tr, A.rows, A.cols, alpha, A.data, ssize_t(A.ld), 1, x.data, x.inc, beta, y.data, y.inc);
