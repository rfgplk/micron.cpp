72 [[nodiscard, gnu::flatten]] inline f64
73 dot_packed_f64(const __packed_f64 *__restrict__ x, const __packed_f64 *__restrict__ y, usize n) noexcept
74 {
75   __m256d s0 = simd::avx::zero_f64();
76   __m256d s1 = simd::avx::zero_f64();
77   __m256d s2 = simd::avx::zero_f64();
78   __m256d s3 = simd::avx::zero_f64();
79   usize i = 0;
80   for ( ; i + 16 <= n; i += 16 ) {
81     s0 = simd::fma::fma_f64(simd::avx::loadu_f64(x + i + 0), simd::avx::loadu_f64(y + i + 0), s0);
82     s1 = simd::fma::fma_f64(simd::avx::loadu_f64(x + i + 4), simd::avx::loadu_f64(y + i + 4), s1);
83     s2 = simd::fma::fma_f64(simd::avx::loadu_f64(x + i + 8), simd::avx::loadu_f64(y + i + 8), s2);
84     s3 = simd::fma::fma_f64(simd::avx::loadu_f64(x + i + 12), simd::avx::loadu_f64(y + i + 12), s3);
85   }
86   // 4-wide tail
87   for ( ; i + 4 <= n; i += 4 ) s0 = simd::fma::fma_f64(simd::avx::loadu_f64(x + i), simd::avx::loadu_f64(y + i), s0);
88   // horizontal reduction across the 4 chains
89   __m256d sum = simd::avx::add_f64(simd::avx::add_f64(s0, s1), simd::avx::add_f64(s2, s3));
90   __m128d hi = simd::avx::extract_f128_f64<1>(sum);
91   __m128d lo = simd::avx::cast_f64_to_lo128(sum);
92   __m128d half = simd::sse::add_f64(hi, lo);
93   __m128d hh = simd::sse::unpack_hi_f64(half, half);
94   f64 r = simd::sse::extract_low_f64(simd::sse::add_scalar_f64(half, hh));
95   // 1-wide scalar epilogue
96   for ( ; i < n; ++i ) r = math::fma<f64>(x[i], y[i], r);
97   return r;
98 }

127 axpy_packed_f64(double alpha, const __packed_f64 *__restrict__ x, __packed_f64 *__restrict__ y, usize n) noexcept
128 {
129   const __m256d va = simd::avx::splat_f64(alpha);
130   usize i = 0;
131   for ( ; i + 16 <= n; i += 16 ) {
132     __m256d y0 = simd::avx::loadu_f64(y + i + 0);
133     __m256d y1 = simd::avx::loadu_f64(y + i + 4);
134     __m256d y2 = simd::avx::loadu_f64(y + i + 8);
135     __m256d y3 = simd::avx::loadu_f64(y + i + 12);
136     y0 = simd::fma::fma_f64(va, simd::avx::loadu_f64(x + i + 0), y0);
137     y1 = simd::fma::fma_f64(va, simd::avx::loadu_f64(x + i + 4), y1);
138     y2 = simd::fma::fma_f64(va, simd::avx::loadu_f64(x + i + 8), y2);
139     y3 = simd::fma::fma_f64(va, simd::avx::loadu_f64(x + i + 12), y3);
140     simd::avx::storeu_f64(y + i + 0, y0);
141     simd::avx::storeu_f64(y + i + 4, y1);
142     simd::avx::storeu_f64(y + i + 8, y2);
143     simd::avx::storeu_f64(y + i + 12, y3);
144   }
145   for ( ; i + 4 <= n; i += 4 ) {
146     __m256d yk = simd::fma::fma_f64(va, simd::avx::loadu_f64(x + i), simd::avx::loadu_f64(y + i));
147     simd::avx::storeu_f64(y + i, yk);
148   }
149   for ( ; i < n; ++i ) y[i] = math::fma<f64>(alpha, x[i], y[i]);
150 }

174 [[nodiscard, gnu::flatten]] inline f64
175 asum_packed_f64(const __packed_f64 *__restrict__ x, usize n) noexcept
176 {
177   // bit-mask 0x7FFFFFFFFFFFFFFF clears the sign bit (gives |x|).
178   const __m256d sign_mask = simd::avx::cast_i256_to_f64(simd::avx::splat_i64(0x7FFFFFFFFFFFFFFFLL));
179   __m256d s0 = simd::avx::zero_f64();
180   __m256d s1 = simd::avx::zero_f64();
181   __m256d s2 = simd::avx::zero_f64();
182   __m256d s3 = simd::avx::zero_f64();
183   usize i = 0;
184   for ( ; i + 16 <= n; i += 16 ) {
185     s0 = simd::avx::add_f64(s0, simd::avx::and_f64(simd::avx::loadu_f64(x + i + 0), sign_mask));
186     s1 = simd::avx::add_f64(s1, simd::avx::and_f64(simd::avx::loadu_f64(x + i + 4), sign_mask));
187     s2 = simd::avx::add_f64(s2, simd::avx::and_f64(simd::avx::loadu_f64(x + i + 8), sign_mask));
188     s3 = simd::avx::add_f64(s3, simd::avx::and_f64(simd::avx::loadu_f64(x + i + 12), sign_mask));
189   }
190   for ( ; i + 4 <= n; i += 4 ) s0 = simd::avx::add_f64(s0, simd::avx::and_f64(simd::avx::loadu_f64(x + i), sign_mask));
191   __m256d sum = simd::avx::add_f64(simd::avx::add_f64(s0, s1), simd::avx::add_f64(s2, s3));
192   __m128d hi = simd::avx::extract_f128_f64<1>(sum);
193   __m128d lo = simd::avx::cast_f64_to_lo128(sum);
194   __m128d half = simd::sse::add_f64(hi, lo);
195   __m128d hh = simd::sse::unpack_hi_f64(half, half);
196   f64 r = simd::sse::extract_low_f64(simd::sse::add_scalar_f64(half, hh));
197   for ( ; i < n; ++i ) r += mk::manip::fabs<f64>(x[i]);
198   return r;

344 [[nodiscard, gnu::flatten]] inline usize
345 iamax_packed_f64(const __packed_f64 *__restrict__ x, usize n) noexcept
346 {
347   if ( n == 0 ) return 0;
348   if ( n < 8 ) {
349     double mx = mk::manip::fabs<f64>(x[0]);
350     usize idx = 0;
351     for ( usize k = 1; k < n; ++k ) {
352       const double a = mk::manip::fabs<f64>(x[k]);
353       if ( a > mx ) {
354         mx = a;
355         idx = k;
356       }
357     }
358     return idx;
359   }
360   // 2-chain unroll
361   const __m256d abs_mask = simd::avx::cast_i256_to_f64(simd::avx::splat_i64(0x7FFFFFFFFFFFFFFFLL));
362   __m256d vmax0 = simd::avx::and_f64(simd::avx::loadu_f64(x + 0), abs_mask);
363   __m256d vmax1 = simd::avx::and_f64(simd::avx::loadu_f64(x + 4), abs_mask);
364   __m256i vidx0 = simd::avx::setr_i64(0, 1, 2, 3);
365   __m256i vidx1 = simd::avx::setr_i64(4, 5, 6, 7);
366   __m256i vcand0 = simd::avx::setr_i64(8, 9, 10, 11);
367   __m256i vcand1 = simd::avx::setr_i64(12, 13, 14, 15);
368   const __m256i step8 = simd::avx::splat_i64(8);
369   const __m256i step4 = simd::avx::splat_i64(4);
370   usize i = 8;

1377 template<ieee754_floating F>
1378 [[nodiscard, gnu::flatten]] inline constexpr F
1379 dot_kahan(const F *__restrict__ x, const F *__restrict__ x_end, const F *__restrict__ y) noexcept
1380 {
1381   const usize n = usize(x_end - x);
1382   // two accumulators
1383   F s0 = F(0), c0 = F(0);
1384   F s1 = F(0), c1 = F(0);
1385   usize i = 0;
1386   for ( ; i + 2 <= n; i += 2 ) {
1387     const F p0 = x[i + 0] * y[i + 0];
1388     const F yy0 = p0 - c0;
1389     const F t0 = s0 + yy0;
1390     c0 = (t0 - s0) - yy0;
1391     s0 = t0;
1392     const F p1 = x[i + 1] * y[i + 1];
1393     const F yy1 = p1 - c1;
1394     const F t1 = s1 + yy1;
1395     c1 = (t1 - s1) - yy1;
1396     s1 = t1;
1397   }
1398   F s = s0, c = c0;
1399   {
1400     const F yy = (s1 - c1) - c;
1401     const F t = s + yy;
1402     c = (t - s) - yy;
1403     s = t;
1404   }
1405   for ( ; i < n; ++i ) {
1406     const F p = x[i] * y[i];
1407     const F yy = p - c;
1408     const F t = s + yy;
1409     c = (t - s) - yy;
1410     s = t;
1411   }
1412   return s;
1413 }

1508 template<ieee754_floating F>
1509 [[gnu::flatten]] inline constexpr void
1510 rotg(F &a, F &b, F &c, F &s) noexcept
1511 {
1512   const F aa = mk::manip::fabs<F>(a);
1513   const F ab = mk::manip::fabs<F>(b);
1514   F sigma;
1515   if ( aa > ab )
1516     sigma = (a < F(0)) ? F(-1) : F(1);
1517   else
1518     sigma = (b < F(0)) ? F(-1) : F(1);
1519   if ( aa == F(0) && ab == F(0) ) {
1520     c = F(1);
1521     s = F(0);
1522     a = F(0);
1523     b = F(0);
1524     return;
1525   }
1526   const F scale = aa + ab;
1527   const F na = a / scale;
1528   const F nb = b / scale;
1529   const F r = sigma * scale * mk::pow_ns::sqrt<F>(math::fma<F>(na, na, nb * nb));
1530   c = a / r;
1531   s = b / r;
1532   F z;
1533   if ( aa > ab )
1534     z = s;
1535   else if ( c != F(0) )
1536     z = F(1) / c;
1537   else
1538     z = F(1);
1539   a = r;
1540   b = z;
1541 }

1563 template<ieee754_floating F>
1564 [[gnu::flatten]] inline constexpr void
1565 rotmg(F &d1, F &d2, F &x1, F y1, F (&param)[5]) noexcept
1566 {
1567   constexpr F GAM = F(4096);
1568   constexpr F GAMSQ = GAM * GAM;
1569   constexpr F RGAMSQ = F(1) / GAMSQ;
1570 
1571   F flag, h11 = F(0), h21 = F(0), h12 = F(0), h22 = F(0);
1572 
1573   if ( d1 < F(0) ) {
1574     flag = F(-1);
1575     d1 = F(0);
1576     d2 = F(0);
1577     x1 = F(0);
1578   } else {
1579     const F p2 = d2 * y1;
1580     if ( p2 == F(0) ) {
1581       flag = F(-2);
1582       param[0] = flag;
1583       return;
1584     }
1585     const F p1 = d1 * x1;
1586     const F q2 = p2 * y1;
1587     const F q1 = p1 * x1;
1588     if ( mk::manip::fabs<F>(q1) > mk::manip::fabs<F>(q2) ) {
1589       h21 = -y1 / x1;
1590       h12 = p2 / p1;
1591       const F u = F(1) - h12 * h21;
1592       if ( u > F(0) ) {
1593         flag = F(0);
1594         d1 = d1 / u;
1595         d2 = d2 / u;
1596         x1 = x1 * u;
1597       } else {
1598         flag = F(-1);
1599         h11 = h21 = h12 = h22 = F(0);
1600         d1 = d2 = F(0);
1601         x1 = F(0);
1602       }
1603     } else {
1604       if ( q2 < F(0) ) {
1605         flag = F(-1);
1606         h11 = h21 = h12 = h22 = F(0);
1607         d1 = d2 = F(0);
1608         x1 = F(0);
1609       } else {
1610         flag = F(1);
1611         h11 = p1 / p2;
1612         h22 = x1 / y1;
1613         const F u = F(1) + h11 * h22;
1614         const F tmp = d2 / u;
1615         d2 = d1 / u;
1616         d1 = tmp;
1617         x1 = y1 * u;
1618       }
1619     }
1620 
1621     if ( d1 != F(0) ) {
1622       while ( (d1 <= RGAMSQ) || (d1 >= GAMSQ) ) {
1623         if ( flag == F(0) ) {
1624           h11 = F(1);
1625           h22 = F(1);
1626           flag = F(-1);
1627         } else {
1628           h21 = F(-1);
1629           h12 = F(1);
1630           flag = F(-1);
1631         }
1632         if ( d1 <= RGAMSQ ) {
1633           d1 = d1 * GAMSQ;
1634           x1 = x1 / GAM;
1635           h11 = h11 / GAM;
1636           h12 = h12 / GAM;
1637         } else {
1638           d1 = d1 / GAMSQ;
1639           x1 = x1 * GAM;
1640           h11 = h11 * GAM;
1641           h12 = h12 * GAM;
1642         }
1643       }
1644     }
1645     if ( d2 != F(0) ) {
1646       while ( (mk::manip::fabs<F>(d2) <= RGAMSQ) || (mk::manip::fabs<F>(d2) >= GAMSQ) ) {
1647         if ( flag == F(0) ) {
1648           h11 = F(1);
1649           h22 = F(1);
1650           flag = F(-1);
1651         } else {
1652           h21 = F(-1);
1653           h12 = F(1);
1654           flag = F(-1);
1655         }
1656         if ( mk::manip::fabs<F>(d2) <= RGAMSQ ) {
1657           d2 = d2 * GAMSQ;
1658           h21 = h21 / GAM;
1659           h22 = h22 / GAM;
1660         } else {
1661           d2 = d2 / GAMSQ;
1662           h21 = h21 * GAM;
1663           h22 = h22 * GAM;
1664         }
1665       }
1666     }
1667   }
1668 
1669   if ( flag < F(0) ) {
1670     param[1] = h11;
1671     param[2] = h21;
1672     param[3] = h12;
1673     param[4] = h22;
1674   } else if ( flag == F(0) ) {
1675     param[2] = h21;
1676     param[3] = h12;
1677   } else {
1678     param[1] = h11;
1679     param[4] = h22;
1680   }
1681   param[0] = flag;
1682 }
