33 //   R = [ 1-2(y²+z²)   2(xy-wz)    2(xz+wy)  ]
34 //       [ 2(xy+wz)    1-2(x²+z²)   2(yz-wx)  ]
35 //       [ 2(xz-wy)    2(yz+wx)    1-2(x²+y²) ]
36 template<ieee754_floating F, typename V>
37 [[gnu::flatten]] inline constexpr void
38 from_quat(const quat<F> &q, V out) noexcept
39 {
40   const F x = q.data[0], y = q.data[1], z = q.data[2], w = q.data[3];
41   const F xx = x * x, yy = y * y, zz = z * z;
42   const F xy = x * y, xz = x * z, yz = y * z;
43   const F wx = w * x, wy = w * y, wz = w * z;
44   const auto wr = [&](usize r, usize c, F v) constexpr noexcept {
45     if constexpr ( row_view_like<V> )
46       bits::mat_at(out.data, r, c, ssize_t(out.ld), 1) = v;
47     else
48       bits::mat_at(out.data, r, c, 1, ssize_t(out.ld)) = v;
49   };
50   wr(0, 0, F(1) - F(2) * (yy + zz));
51   wr(0, 1, F(2) * (xy - wz));
52   wr(0, 2, F(2) * (xz + wy));
53   wr(1, 0, F(2) * (xy + wz));
54   wr(1, 1, F(1) - F(2) * (xx + zz));
55   wr(1, 2, F(2) * (yz - wx));
56   wr(2, 0, F(2) * (xz - wy));
57   wr(2, 1, F(2) * (yz + wx));
58   wr(2, 2, F(1) - F(2) * (xx + yy));
59 }

72 //   [v]_× = [  0  -z   y ]
73 //           [  z   0  -x ]
74 //           [ -y   x   0 ]
75 template<ieee754_floating F, typename V>
76 [[gnu::flatten]] inline constexpr void
77 skew(F vx, F vy, F vz, V out) noexcept
78 {
79   const auto wr = [&](usize r, usize c, F v) constexpr noexcept {
80     if constexpr ( row_view_like<V> )
81       bits::mat_at(out.data, r, c, ssize_t(out.ld), 1) = v;
82     else
83       bits::mat_at(out.data, r, c, 1, ssize_t(out.ld)) = v;
84   };
85   wr(0, 0, F(0));
86   wr(0, 1, -vz);
87   wr(0, 2, vy);
88   wr(1, 0, vz);
89   wr(1, 1, F(0));
90   wr(1, 2, -vx);
91   wr(2, 0, -vy);
92   wr(2, 1, vx);
93   wr(2, 2, F(0));
94 }

113 template<blas_scalar T, typename V>
114   requires(mat_view_like<V>)
115 [[gnu::flatten]] inline constexpr void
116 outer(T alpha, const vec_view<T> &x, const vec_view<T> &y, V A) noexcept
117 {
118   const ssize_t rs = row_view_like<V> ? ssize_t(A.ld) : ssize_t(1);
119   const ssize_t cs = row_view_like<V> ? ssize_t(1) : ssize_t(A.ld);
120   for ( usize i = 0; i < A.rows; ++i )
121     for ( usize j = 0; j < A.cols; ++j ) bits::mat_at(A.data, i, j, rs, cs) = T(0);
122   level2::ger<T>(alpha, x, y, A);
123 }

125 template<ieee754_floating F, typename V>
126   requires(mat_view_like<V>)
127 [[gnu::flatten]] inline constexpr void
128 givens(usize i, usize j, F c, F s, V out) noexcept
129 {
130   const ssize_t rs = row_view_like<V> ? ssize_t(out.ld) : ssize_t(1);
131   const ssize_t cs = row_view_like<V> ? ssize_t(1) : ssize_t(out.ld);
132   for ( usize r = 0; r < out.rows; ++r )
133     for ( usize cc = 0; cc < out.cols; ++cc ) bits::mat_at(out.data, r, cc, rs, cs) = (r == cc) ? F(1) : F(0);
134   bits::mat_at(out.data, i, i, rs, cs) = c;
135   bits::mat_at(out.data, j, j, rs, cs) = c;
136   bits::mat_at(out.data, i, j, rs, cs) = s;
137   bits::mat_at(out.data, j, i, rs, cs) = -s;
138 }

140 template<blas_scalar T, typename V>
141   requires(mat_view_like<V>)
142 [[gnu::flatten]] inline constexpr void
143 householder(const vec_view<T> &v, V H_out, T beta) noexcept
144 {
145   const ssize_t rs = row_view_like<V> ? ssize_t(H_out.ld) : ssize_t(1);
146   const ssize_t cs = row_view_like<V> ? ssize_t(1) : ssize_t(H_out.ld);
147   for ( usize r = 0; r < H_out.rows; ++r ) {
148     const T mb_vr = -beta * v.data[ssize_t(r) * v.inc];
149     for ( usize c = 0; c < H_out.cols; ++c ) {
150       const T vc = v.data[ssize_t(c) * v.inc];
151       const T diag = (r == c) ? T(1) : T(0);
152       bits::mat_at(H_out.data, r, c, rs, cs) = bits::fma_acc<T>(mb_vr, vc, diag);
153     }
154   }
155 }

157 template<blas_real T, typename V>
158   requires(mat_view_like<V>)
159 [[gnu::flatten]] inline constexpr void
160 householder(const vec_view<T> &v, V H_out) noexcept
161 {
162   T vtv = T(0);
163   for ( usize i = 0; i < v.n; ++i ) {
164     const T vi = v.data[ssize_t(i) * v.inc];
165     vtv = math::fma<T>(vi, vi, vtv);
166   }
167   const T beta = (vtv > T(0)) ? T(2) / vtv : T(0);
168   householder<T>(v, H_out, beta);
169 }

171 template<ieee754_floating F, typename V>
172   requires(mat_view_like<V>)
173 [[gnu::flatten]] inline constexpr void
174 from_axis_angle(F kx, F ky, F kz, F theta, V out) noexcept
175 {
176   const F c = mk::trig::cos<F>(theta);
177   const F s = mk::trig::sin<F>(theta);
178   const F omc = F(1) - c;
179   const auto wr = [&](usize r, usize cc, F v) constexpr noexcept {
180     if constexpr ( row_view_like<V> )
181       bits::mat_at(out.data, r, cc, ssize_t(out.ld), 1) = v;
182     else
183       bits::mat_at(out.data, r, cc, 1, ssize_t(out.ld)) = v;
184   };
185   wr(0, 0, c + kx * kx * omc);
186   wr(0, 1, kx * ky * omc - kz * s);
187   wr(0, 2, kx * kz * omc + ky * s);
188   wr(1, 0, ky * kx * omc + kz * s);
189   wr(1, 1, c + ky * ky * omc);
190   wr(1, 2, ky * kz * omc - kx * s);
191   wr(2, 0, kz * kx * omc - ky * s);
192   wr(2, 1, kz * ky * omc + kx * s);
193   wr(2, 2, c + kz * kz * omc);
194 }
