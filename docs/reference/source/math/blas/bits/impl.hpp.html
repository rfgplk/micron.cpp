68 [[gnu::flatten, gnu::hot]] inline void
69 gemv_n_panel_avx2_f64(usize m, usize n, double alpha, const __packed_f64 *A, ssize_t rs_A, const __packed_f64 *x, double beta,
70                       __packed_f64 *y) noexcept
71 {
72   if ( beta == 0.0 ) {
73     const __m256d z = simd::avx::zero_f64();
74     usize i = 0;
75     for ( ; i + 4 <= m; i += 4 ) simd::avx::storeu_f64(y + i, z);
76     for ( ; i < m; ++i ) y[i] = 0.0;
77   } else if ( beta != 1.0 ) {
78     const __m256d vbeta = simd::avx::splat_f64(beta);
79     usize i = 0;
80     for ( ; i + 4 <= m; i += 4 ) simd::avx::storeu_f64(y + i, simd::avx::mul_f64(vbeta, simd::avx::loadu_f64(y + i)));
81     for ( ; i < m; ++i ) y[i] *= beta;
82   }
83 
84   const __m256d valpha = simd::avx::splat_f64(alpha);
85   usize i = 0;
86   // 4-row dot-product fan-out: one x-load shared across 4 row-loads + 4 FMAs
87   for ( ; i + 4 <= m; i += 4 ) {
88     __m256d acc0 = simd::avx::zero_f64();
89     __m256d acc1 = simd::avx::zero_f64();
90     __m256d acc2 = simd::avx::zero_f64();
91     __m256d acc3 = simd::avx::zero_f64();
92     const __packed_f64 *r0 = A + ssize_t(i + 0) * rs_A;
93     const __packed_f64 *r1 = A + ssize_t(i + 1) * rs_A;
94     const __packed_f64 *r2 = A + ssize_t(i + 2) * rs_A;
95     const __packed_f64 *r3 = A + ssize_t(i + 3) * rs_A;
96     usize j = 0;
97     for ( ; j + 4 <= n; j += 4 ) {
98       const __m256d xv = simd::avx::loadu_f64(x + j);
99       acc0 = simd::fma::fma_f64(simd::avx::loadu_f64(r0 + j), xv, acc0);
100       acc1 = simd::fma::fma_f64(simd::avx::loadu_f64(r1 + j), xv, acc1);
101       acc2 = simd::fma::fma_f64(simd::avx::loadu_f64(r2 + j), xv, acc2);
102       acc3 = simd::fma::fma_f64(simd::avx::loadu_f64(r3 + j), xv, acc3);
103     }
104     const __m256d h01 = simd::avx::hadd_f64(acc0, acc1);
105     const __m256d h23 = simd::avx::hadd_f64(acc2, acc3);
106     const __m256d lo = simd::avx::permute2f128_f64<0x20>(h01, h23);
107     const __m256d hi = simd::avx::permute2f128_f64<0x31>(h01, h23);
108     __m256d sums = simd::avx::add_f64(lo, hi);
109 
110     // j-tail (n%4)
111     if ( j < n ) {
112       alignas(32) double s[4];
113       simd::avx::store_f64(s, sums);
114       for ( usize jj = j; jj < n; ++jj ) {
115         const double xv = x[jj];
116         s[0] = math::fma<double>(r0[jj], xv, s[0]);
117         s[1] = math::fma<double>(r1[jj], xv, s[1]);
118         s[2] = math::fma<double>(r2[jj], xv, s[2]);
119         s[3] = math::fma<double>(r3[jj], xv, s[3]);
120       }
121       sums = simd::avx::load_f64(s);
122     }
123 
124     const __m256d yv = simd::avx::loadu_f64(y + i);
125     simd::avx::storeu_f64(y + i, simd::fma::fma_f64(valpha, sums, yv));
126   }
127   // i-tail
128   for ( ; i < m; ++i ) {
129     __m256d acc = simd::avx::zero_f64();
130     const __packed_f64 *row = A + ssize_t(i) * rs_A;
131     usize j = 0;
132     for ( ; j + 4 <= n; j += 4 ) {
133       acc = simd::fma::fma_f64(simd::avx::loadu_f64(row + j), simd::avx::loadu_f64(x + j), acc);
134     }
135     const __m128d lo = simd::avx::cast_f64_to_lo128(acc);
136     const __m128d hi = simd::avx::extract_f128_f64<1>(acc);
137     const __m128d s2 = simd::sse::add_f64(lo, hi);
138     double tail = simd::sse::extract_low_f64(simd::sse::hadd_f64(s2, s2));
139     for ( ; j < n; ++j ) tail = math::fma<double>(row[j], x[j], tail);
140     y[i] = math::fma<double>(alpha, tail, y[i]);
141   }
142 }

144 [[gnu::flatten, gnu::hot]] inline void
145 gemv_n_panel_avx2_f32(usize m, usize n, float alpha, const __packed_f32 *A, ssize_t rs_A, const __packed_f32 *x, float beta,
146                       __packed_f32 *y) noexcept
147 {
148   if ( beta == 0.0f ) {
149     const __m256 z = simd::avx::zero_f32();
150     usize i = 0;
151     for ( ; i + 8 <= m; i += 8 ) simd::avx::storeu_f32(y + i, z);
152     for ( ; i < m; ++i ) y[i] = 0.0f;
153   } else if ( beta != 1.0f ) {
154     const __m256 vbeta = simd::avx::splat_f32(beta);
155     usize i = 0;
156     for ( ; i + 8 <= m; i += 8 ) simd::avx::storeu_f32(y + i, simd::avx::mul_f32(vbeta, simd::avx::loadu_f32(y + i)));
157     for ( ; i < m; ++i ) y[i] *= beta;
158   }
159 
160   usize i = 0;
161   for ( ; i + 4 <= m; i += 4 ) {
162     __m256 acc0 = simd::avx::zero_f32();
163     __m256 acc1 = simd::avx::zero_f32();
164     __m256 acc2 = simd::avx::zero_f32();
165     __m256 acc3 = simd::avx::zero_f32();
166     const __packed_f32 *r0 = A + ssize_t(i + 0) * rs_A;
167     const __packed_f32 *r1 = A + ssize_t(i + 1) * rs_A;
168     const __packed_f32 *r2 = A + ssize_t(i + 2) * rs_A;
169     const __packed_f32 *r3 = A + ssize_t(i + 3) * rs_A;
170     usize j = 0;
171     for ( ; j + 8 <= n; j += 8 ) {
172       const __m256 xv = simd::avx::loadu_f32(x + j);
173       acc0 = simd::fma::fma_f32(simd::avx::loadu_f32(r0 + j), xv, acc0);
174       acc1 = simd::fma::fma_f32(simd::avx::loadu_f32(r1 + j), xv, acc1);
175       acc2 = simd::fma::fma_f32(simd::avx::loadu_f32(r2 + j), xv, acc2);
176       acc3 = simd::fma::fma_f32(simd::avx::loadu_f32(r3 + j), xv, acc3);
177     }
178     auto reduce8 = [](__m256 v) -> float {
179       __m128 lo = simd::avx::cast_f32_to_lo128(v);
180       __m128 hi = simd::avx::extract_f128_f32<1>(v);
181       __m128 s4 = simd::sse::add_f32(lo, hi);
182       __m128 s2 = simd::sse::hadd_f32(s4, s4);
183       __m128 s1 = simd::sse::hadd_f32(s2, s2);
184       return simd::sse::extract_low_f32(s1);
185     };
186     float s0 = reduce8(acc0), s1 = reduce8(acc1);
187     float s2 = reduce8(acc2), s3 = reduce8(acc3);
188     for ( ; j < n; ++j ) {
189       const float xv = x[j];
190       s0 = math::fma<float>(r0[j], xv, s0);
191       s1 = math::fma<float>(r1[j], xv, s1);
192       s2 = math::fma<float>(r2[j], xv, s2);
193       s3 = math::fma<float>(r3[j], xv, s3);
194     }
195     y[i + 0] = math::fma<float>(alpha, s0, y[i + 0]);
196     y[i + 1] = math::fma<float>(alpha, s1, y[i + 1]);
197     y[i + 2] = math::fma<float>(alpha, s2, y[i + 2]);
198     y[i + 3] = math::fma<float>(alpha, s3, y[i + 3]);
199   }
200   for ( ; i < m; ++i ) {
201     float acc = 0.0f;
202     for ( usize j = 0; j < n; ++j ) acc = math::fma<float>(A[ssize_t(i) * rs_A + ssize_t(j)], x[j], acc);
203     y[i] = math::fma<float>(alpha, acc, y[i]);
204   }
205 }

210 [[gnu::flatten]] inline void
211 gemv_t_panel_avx2_f64(usize m, usize n, double alpha, const __packed_f64 *A, ssize_t rs_A, const __packed_f64 *x, double beta,
212                       __packed_f64 *y) noexcept
213 {
214   constexpr usize NB = 16;       // j-strip width (4 ymm of 4 doubles each)
215   constexpr usize MC = 128;      // m-block height; NB*MC doubles = 16 KiB pack, fits L1
216   alignas(32) double Ap[NB * MC];
217 
218   if ( beta == 0.0 ) {
219     const __m256d z = simd::avx::zero_f64();
220     usize j = 0;
221     for ( ; j + 4 <= n; j += 4 ) simd::avx::storeu_f64(y + j, z);
222     for ( ; j < n; ++j ) y[j] = 0.0;
223   } else if ( beta != 1.0 ) {
224     const __m256d vbeta = simd::avx::splat_f64(beta);
225     usize j = 0;
226     for ( ; j + 4 <= n; j += 4 ) simd::avx::storeu_f64(y + j, simd::avx::mul_f64(vbeta, simd::avx::loadu_f64(y + j)));
227     for ( ; j < n; ++j ) y[j] *= beta;
228   }
229 
230   usize j = 0;
231   for ( ; j + NB <= n; j += NB ) {
232     __m256d y0 = simd::avx::loadu_f64(y + j + 0);
233     __m256d y1 = simd::avx::loadu_f64(y + j + 4);
234     __m256d y2 = simd::avx::loadu_f64(y + j + 8);
235     __m256d y3 = simd::avx::loadu_f64(y + j + 12);
236 
237     for ( usize ic = 0; ic < m; ic += MC ) {
238       const usize mc = (m - ic < MC) ? (m - ic) : MC;
239 
240       for ( usize i = 0; i < mc; ++i ) {
241         const __packed_f64 *src = A + ssize_t(ic + i) * rs_A + ssize_t(j);
242         // Prefetch the next row 8 ahead
243         if ( i + 8 < mc ) {
244           __builtin_prefetch(A + ssize_t(ic + i + 8) * rs_A + ssize_t(j), 0, 1);
245         }
246         simd::avx::store_f64(Ap + i * NB + 0, simd::avx::loadu_f64(src + 0));
247         simd::avx::store_f64(Ap + i * NB + 4, simd::avx::loadu_f64(src + 4));
248         simd::avx::store_f64(Ap + i * NB + 8, simd::avx::loadu_f64(src + 8));
249         simd::avx::store_f64(Ap + i * NB + 12, simd::avx::loadu_f64(src + 12));
250       }
251 
252       for ( usize i = 0; i < mc; ++i ) {
253         const __m256d ax = simd::avx::splat_f64(alpha * x[ic + i]);
254         const __packed_f64 *ap = Ap + i * NB;
255         y0 = simd::fma::fma_f64(ax, simd::avx::load_f64(ap + 0), y0);
256         y1 = simd::fma::fma_f64(ax, simd::avx::load_f64(ap + 4), y1);
257         y2 = simd::fma::fma_f64(ax, simd::avx::load_f64(ap + 8), y2);
258         y3 = simd::fma::fma_f64(ax, simd::avx::load_f64(ap + 12), y3);
259       }
260     }
261 
262     simd::avx::storeu_f64(y + j + 0, y0);
263     simd::avx::storeu_f64(y + j + 4, y1);
264     simd::avx::storeu_f64(y + j + 8, y2);
265     simd::avx::storeu_f64(y + j + 12, y3);
266   }
267 
268   // tail
269   for ( ; j < n; ++j ) {
270     double acc = 0.0;
271     for ( usize i = 0; i < m; ++i ) acc = math::fma<double>(A[ssize_t(i) * rs_A + ssize_t(j)], x[i], acc);
272     y[j] = math::fma<double>(alpha, acc, y[j]);
273   }
274 }

466 template<typename T>
467 [[gnu::flatten]] inline constexpr void
468 gemv_kernel(bool tr, usize m, usize n, T alpha, const T *A, ssize_t rs_A, ssize_t cs_A, const T *x, ssize_t incx, T beta, T *y,
469             ssize_t incy) noexcept
470 {
471   // row-major A panel kernels: tr-case uses axpy fan-out; no-trans uses
472   // dot-product fan-out across 4 consecutive rows.
473   if !consteval {
474     if ( cs_A == 1 && incx == 1 && incy == 1 ) {
475       if constexpr ( ieee754_floating<T> ) {
476 #if defined(__micron_x86_avx2) && defined(__micron_x86_fma)
477         if constexpr ( sizeof(T) == 8 ) {
478           if ( tr ) {
479             gemv_t_panel_avx2_f64(m, n, double(alpha), reinterpret_cast<const __packed_f64 *>(A), rs_A,
480                                   reinterpret_cast<const __packed_f64 *>(x), double(beta), reinterpret_cast<__packed_f64 *>(y));
481           } else {
482             gemv_n_panel_avx2_f64(m, n, double(alpha), reinterpret_cast<const __packed_f64 *>(A), rs_A,
483                                   reinterpret_cast<const __packed_f64 *>(x), double(beta), reinterpret_cast<__packed_f64 *>(y));
484           }
485           return;
486         } else if constexpr ( sizeof(T) == 4 ) {
487           if ( tr ) {
488             gemv_t_panel_avx2_f32(m, n, float(alpha), reinterpret_cast<const __packed_f32 *>(A), rs_A,
489                                   reinterpret_cast<const __packed_f32 *>(x), float(beta), reinterpret_cast<__packed_f32 *>(y));
490           } else {
491             gemv_n_panel_avx2_f32(m, n, float(alpha), reinterpret_cast<const __packed_f32 *>(A), rs_A,
492                                   reinterpret_cast<const __packed_f32 *>(x), float(beta), reinterpret_cast<__packed_f32 *>(y));
493           }
494           return;
495         }
496 #elif defined(__micron_arch_arm64) && defined(__micron_arm_neon)
497         if ( tr ) {
498           if constexpr ( sizeof(T) == 8 ) {
499             gemv_t_panel_neon_f64(m, n, double(alpha), reinterpret_cast<const __packed_f64 *>(A), rs_A,
500                                   reinterpret_cast<const __packed_f64 *>(x), double(beta), reinterpret_cast<__packed_f64 *>(y));
501             return;
502           } else if constexpr ( sizeof(T) == 4 ) {
503             gemv_t_panel_neon_f32(m, n, float(alpha), reinterpret_cast<const __packed_f32 *>(A), rs_A,
504                                   reinterpret_cast<const __packed_f32 *>(x), float(beta), reinterpret_cast<__packed_f32 *>(y));
505             return;
506           }
507         }
508 #elif defined(__micron_arch_arm32) && defined(__micron_arm_neon)
509         if ( tr ) {
510           if constexpr ( sizeof(T) == 4 ) {
511             gemv_t_panel_neon_f32(m, n, float(alpha), reinterpret_cast<const __packed_f32 *>(A), rs_A,
512                                   reinterpret_cast<const __packed_f32 *>(x), float(beta), reinterpret_cast<__packed_f32 *>(y));
513             return;
514           }
515         }
516 #endif
517       }
518     }
519   }
520 
521   const usize m_out = tr ? n : m;
522   const usize n_in = tr ? m : n;
523   const ssize_t step_inner = tr ? rs_A : cs_A;
524   const ssize_t step_outer = tr ? cs_A : rs_A;
525   const bool beta_zero = (beta == T(0));
526   const bool beta_one = (beta == T(1));
527   for ( usize i = 0; i < m_out; ++i ) {
528     const T *row = A + ssize_t(i) * step_outer;
529     // four accumulators
530     T s0{}, s1{}, s2{}, s3{};
531     usize j = 0;
532     if ( incx == 1 && step_inner == 1 ) {
533       for ( ; j + 4 <= n_in; j += 4 ) {
534         s0 = fma_acc<T>(row[j + 0], x[j + 0], s0);
535         s1 = fma_acc<T>(row[j + 1], x[j + 1], s1);
536         s2 = fma_acc<T>(row[j + 2], x[j + 2], s2);
537         s3 = fma_acc<T>(row[j + 3], x[j + 3], s3);
538       }
539     } else {
540       for ( ; j + 4 <= n_in; j += 4 ) {
541         s0 = fma_acc<T>(row[ssize_t(j + 0) * step_inner], x[ssize_t(j + 0) * incx], s0);
542         s1 = fma_acc<T>(row[ssize_t(j + 1) * step_inner], x[ssize_t(j + 1) * incx], s1);
543         s2 = fma_acc<T>(row[ssize_t(j + 2) * step_inner], x[ssize_t(j + 2) * incx], s2);
544         s3 = fma_acc<T>(row[ssize_t(j + 3) * step_inner], x[ssize_t(j + 3) * incx], s3);
545       }
546     }
547     T tail{};
548     for ( ; j < n_in; ++j ) tail = fma_acc<T>(row[ssize_t(j) * step_inner], x[ssize_t(j) * incx], tail);
549     const T acc = ((s0 + s1) + (s2 + s3)) + tail;
550     T &yi = y[ssize_t(i) * incy];
551     if ( beta_zero ) {
552       yi = alpha * acc;
553     } else if ( beta_one ) {
554       yi = fma_acc<T>(alpha, acc, yi);
555     } else {
556       yi = fma_acc<T>(alpha, acc, beta * yi);
557     }
558   }
559 }

739 [[gnu::flatten]] inline void

763         Bp[p * 8 + 3] = base[3 * b_cs];

765         Bp[p * 8 + 5] = base[5 * b_cs];
766         Bp[p * 8 + 6] = base[6 * b_cs];

787         c01 = simd::fma::fma_f64(a0, b1, c01);
788         c10 = simd::fma::fma_f64(a1, b0, c10);

795         c31 = simd::fma::fma_f64(a3, b1, c31);

799       __packed_f64 *r2 = C + ssize_t(i + 2) * rs_C + ssize_t(j);
800       __packed_f64 *r3 = C + ssize_t(i + 3) * rs_C + ssize_t(j);
801       if ( beta_zero ) {
802         simd::avx::storeu_f64(r0 + 0, simd::avx::mul_f64(valpha, c00));
803         simd::avx::storeu_f64(r0 + 4, simd::avx::mul_f64(valpha, c01));
804         simd::avx::storeu_f64(r1 + 0, simd::avx::mul_f64(valpha, c10));
805         simd::avx::storeu_f64(r1 + 4, simd::avx::mul_f64(valpha, c11));
806         simd::avx::storeu_f64(r2 + 0, simd::avx::mul_f64(valpha, c20));
807         simd::avx::storeu_f64(r2 + 4, simd::avx::mul_f64(valpha, c21));
808         simd::avx::storeu_f64(r3 + 0, simd::avx::mul_f64(valpha, c30));
809         simd::avx::storeu_f64(r3 + 4, simd::avx::mul_f64(valpha, c31));
810       } else if ( beta_one ) {
811         simd::avx::storeu_f64(r0 + 0, simd::fma::fma_f64(valpha, c00, simd::avx::loadu_f64(r0 + 0)));

930 template<typename T>
931 [[gnu::flatten]] inline constexpr void
932 gemm_kernel(bool trA, bool trB, usize m, usize n, usize k, T alpha, const T *A, ssize_t rs_A, ssize_t cs_A, const T *B, ssize_t rs_B,
933             ssize_t cs_B, T beta, T *C, ssize_t rs_C, ssize_t cs_C) noexcept
934 {
935   const ssize_t a_rs = trA ? cs_A : rs_A;
936   const ssize_t a_cs = trA ? rs_A : cs_A;
937   const ssize_t b_rs = trB ? cs_B : rs_B;
938   const ssize_t b_cs = trB ? rs_B : cs_B;
939 
940   const bool beta_zero = (beta == T(0));
941   const bool beta_one = (beta == T(1));
942 
943   // inline 4x8 kernel with no panel packing
944 #if defined(__micron_x86_avx2) && defined(__micron_x86_fma)
945   if !consteval {
946     if constexpr ( sizeof(T) == 8 && ieee754_floating<T> ) {
947       const u64 nflops = u64(m) * u64(n) * u64(k);
948       const bool size_ok = (a_cs == 1) ? (nflops < (256ull * 256ull * 256ull)) : (nflops < (64ull * 64ull * 64ull));
949       if ( size_ok && cs_C == 1 && (m % 4u) == 0 && (n % 8u) == 0 && k > 0 && k <= 1024 ) {
950         gemm_4x8_avx2_f64(m, n, k, double(alpha), reinterpret_cast<const __packed_f64 *>(A), a_rs, a_cs,
951                           reinterpret_cast<const __packed_f64 *>(B), b_rs, b_cs, double(beta), reinterpret_cast<__packed_f64 *>(C), rs_C);
952         return;
953       }
954     }
955   }
956 #endif
957 
958   // packed/blocked path for everything the inline kernel can't take or for problems large enough that pack overhead amortizes
959   if !consteval {
960     if ( micron::math::matrix::pack::gemm_should_block_for_layout<T>(m, n, k, b_cs, cs_C) ) {
961       micron::math::matrix::pack::gemm_blocked<T>(m, n, k, alpha, A, a_rs, a_cs, B, b_rs, b_cs, beta, C, rs_C, cs_C);
962       return;
963     }
964   }
965 
966   // scalar fallback path (very small problems and consteval)
967   for ( usize i = 0; i < m; ++i ) {
968     T *c_row = C + ssize_t(i) * rs_C;
969     const T *a_row = A + ssize_t(i) * a_rs;
970 
971     if ( beta_zero ) {
972       if ( cs_C == 1 ) {
973         for ( usize j = 0; j < n; ++j ) c_row[j] = T(0);
974       } else {
975         for ( usize j = 0; j < n; ++j ) c_row[ssize_t(j) * cs_C] = T(0);
976       }
977     } else if ( !beta_one ) {
978       if ( cs_C == 1 ) {
979         for ( usize j = 0; j < n; ++j ) c_row[j] *= beta;
980       } else {
981         for ( usize j = 0; j < n; ++j ) c_row[ssize_t(j) * cs_C] *= beta;
982       }
983     }
984 
985     for ( usize p = 0; p < k; ++p ) {
986       const T a = alpha * a_row[ssize_t(p) * a_cs];
987       const T *b_row = B + ssize_t(p) * b_rs;
988       usize j = 0;
989       if ( b_cs == 1 && cs_C == 1 ) {
990         for ( ; j + 4 <= n; j += 4 ) {
991           c_row[j + 0] = fma_acc<T>(a, b_row[j + 0], c_row[j + 0]);
992           c_row[j + 1] = fma_acc<T>(a, b_row[j + 1], c_row[j + 1]);
993           c_row[j + 2] = fma_acc<T>(a, b_row[j + 2], c_row[j + 2]);
994           c_row[j + 3] = fma_acc<T>(a, b_row[j + 3], c_row[j + 3]);
995         }
996         for ( ; j < n; ++j ) c_row[j] = fma_acc<T>(a, b_row[j], c_row[j]);
997       } else {
998         for ( ; j + 4 <= n; j += 4 ) {
999           T &c0 = c_row[ssize_t(j + 0) * cs_C];
1000           T &c1 = c_row[ssize_t(j + 1) * cs_C];
1001           T &c2 = c_row[ssize_t(j + 2) * cs_C];
1002           T &c3 = c_row[ssize_t(j + 3) * cs_C];
1003           c0 = fma_acc<T>(a, b_row[ssize_t(j + 0) * b_cs], c0);
1004           c1 = fma_acc<T>(a, b_row[ssize_t(j + 1) * b_cs], c1);
1005           c2 = fma_acc<T>(a, b_row[ssize_t(j + 2) * b_cs], c2);
1006           c3 = fma_acc<T>(a, b_row[ssize_t(j + 3) * b_cs], c3);
1007         }
1008         for ( ; j < n; ++j ) {
1009           T &cj = c_row[ssize_t(j) * cs_C];
1010           cj = fma_acc<T>(a, b_row[ssize_t(j) * b_cs], cj);
1011         }
1012       }
1013     }
1014   }

1096 [[gnu::flatten]] inline constexpr void
1097 trsm_left_kernel(bool upper, bool tr, bool unit_diag, usize m, usize n, T alpha, const T *A, ssize_t rs_A, ssize_t cs_A, T *B, ssize_t rs_B,
1098                  ssize_t cs_B) noexcept
1099 {
1100   // pre-scale B by alpha
1101   if ( alpha != T(1) ) {
1102     for ( usize i = 0; i < m; ++i )
1103       for ( usize j = 0; j < n; ++j ) mat_at(B, i, j, rs_B, cs_B) *= alpha;
1104   }
1105   // solve column by column
1106   for ( usize j = 0; j < n; ++j ) {
1107     if ( !tr ) {
1108       if ( upper ) {
1109         for ( usize ii = m; ii-- > 0; ) {
1110           T s = mat_at(B, ii, j, rs_B, cs_B);
1111           for ( usize p = ii + 1; p < m; ++p ) s -= mat_at(A, ii, p, rs_A, cs_A) * mat_at(B, p, j, rs_B, cs_B);
1112           if ( !unit_diag ) s /= mat_at(A, ii, ii, rs_A, cs_A);
1113           mat_at(B, ii, j, rs_B, cs_B) = s;
1114         }
1115       } else {
1116         for ( usize i = 0; i < m; ++i ) {
1117           T s = mat_at(B, i, j, rs_B, cs_B);
1118           for ( usize p = 0; p < i; ++p ) s -= mat_at(A, i, p, rs_A, cs_A) * mat_at(B, p, j, rs_B, cs_B);
1119           if ( !unit_diag ) s /= mat_at(A, i, i, rs_A, cs_A);
1120           mat_at(B, i, j, rs_B, cs_B) = s;
1121         }
