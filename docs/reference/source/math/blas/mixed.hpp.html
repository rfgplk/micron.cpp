32 {
33   if constexpr ( accumulator<T> )

41 template<accumulator F> struct __fraction {
42   F mantissa;
43   i32 exponent;

46 // Normalize via bits so FTZ/DAZ cannot turn a valid subnormal scale into zero.
47 template<accumulator F>
48 constexpr __fraction<F>
49 __normalize(typename ieee::traits<F>::uint_type raw) noexcept
50 {
51   using traits = ieee::traits<F>;
52   using U = typename traits::uint_type;
53   const U exponent = (raw & traits::exp_mask) >> traits::mant_bits;
54   U fraction = raw & traits::mant_mask;
55   i32 e = i32(exponent ? exponent : 1) - traits::exp_bias;
56   if ( !exponent )
57     while ( !(fraction & traits::implicit_one) ) {
58       fraction <<= 1;
59       --e;
60     }
61   return { __builtin_bit_cast(F, (U(traits::exp_bias) << traits::mant_bits) | (fraction & traits::mant_mask)), e };
62 }

65 // Each operand is converted to Acc before multiplication. Output stays in Acc;
66 // narrowing is a separate explicit conversion. beta==0 never reads C. No packing,
67 // runtime ISA selection or implicit heap allocation; compiler vectorization is allowed.
68 template<accumulator Acc, floating_input A, floating_input B>
69 [[nodiscard]] constexpr status
70 gemm_row(bool ta, bool tb, usize m, usize n, usize k, Acc alpha, const A *a, usize lda, const B *b, usize ldb, Acc beta, Acc *c,
71          usize ldc) noexcept
72 {
73   if ( !m || !n ) return status::ok;
74   if ( !c || !__impl::__layout(m, n, ldc)
75        || (k && alpha != Acc(0)
76            && (!a || !b || !__impl::__layout(ta ? k : m, ta ? m : k, lda) || !__impl::__layout(tb ? n : k, tb ? k : n, ldb))) )
77     return status::bad_dimension;
78   for ( usize i = 0; i < m; ++i )
79     for ( usize j = 0; j < n; ++j ) {
80       Acc sum{};
81       if ( alpha != Acc(0) )
82         for ( usize z = 0; z < k; ++z )
83           sum += __impl::__read<Acc>(a[ta ? z * lda + i : i * lda + z]) * __impl::__read<Acc>(b[tb ? j * ldb + z : z * ldb + j]);
84       c[i * ldc + j] = alpha * sum + (beta == Acc(0) ? Acc(0) : beta * c[i * ldc + j]);
85     }
86   return status::ok;
87 }

94 // Exact sum((A-zero_a)*(B-zero_b)) in i32/i64, with a conservative worst-case bound
95 // checked before writing. Both zero points are signed int8 values. Scaling and bias
96 // follow in floating point: output=Acc_sum*scale_a*scale_b[column]+bias[column].
97 template<integer_accumulator Acc>
98 [[nodiscard]] constexpr status
99 gemm_i8_row(bool ta, bool tb, usize m, usize n, usize k, const i8 *a, usize lda, i32 zero_a, const i8 *b, usize ldb, i32 zero_b, Acc *c,
100             usize ldc) noexcept
101 {
102   if ( zero_a < -128 || zero_a > 127 || zero_b < -128 || zero_b > 127 ) return status::bad_zero_point;
103   if ( u64(k) > max_i8_terms<Acc> ) return status::accumulator_overflow;
104   if ( !m || !n ) return status::ok;
105   if ( !c || !__impl::__layout(m, n, ldc)
106        || (k && (!a || !b || !__impl::__layout(ta ? k : m, ta ? m : k, lda) || !__impl::__layout(tb ? n : k, tb ? k : n, ldb))) )
107     return status::bad_dimension;
108   for ( usize i = 0; i < m; ++i )
109     for ( usize j = 0; j < n; ++j ) {
110       Acc sum{};
111       for ( usize z = 0; z < k; ++z )
112         sum += Acc(i32(a[ta ? z * lda + i : i * lda + z]) - zero_a) * Acc(i32(b[tb ? j * ldb + z : z * ldb + j]) - zero_b);
113       c[i * ldc + j] = sum;
114     }
115   return status::ok;
116 }

118 // Positive finite scale and signed-int8 zero point are caller preconditions.
119 // RNE is applied before adding the zero point; saturation follows. NaN maps to
120 // zero_point and infinities saturate. Model interfaces may reject nonfinite input.
121 template<accumulator F>
122 [[nodiscard]] constexpr i8
123 quantize_i8(const F &value, F scale, i32 zero_point = 0) noexcept
124 {
125   using traits = ieee::traits<F>;
126   const auto raw = __builtin_bit_cast(typename traits::uint_type, value);
127   if ( (raw & traits::exp_mask) == traits::exp_mask )
128     return raw & traits::mant_mask ? i8(zero_point) : (raw & traits::sign_mask ? i8(-128) : i8(127));
129   if ( (raw & ~traits::sign_mask) == 0 ) return i8(zero_point);
130   const auto x = __impl::__normalize<F>(raw), s = __impl::__normalize<F>(__builtin_bit_cast(typename traits::uint_type, scale));
131   const i32 exponent = x.exponent - s.exponent;
132   if ( exponent >= 9 ) return raw & traits::sign_mask ? i8(-128) : i8(127);
133   if ( exponent < -1 ) return i8(zero_point);
134   using U = typename traits::uint_type;
135   const F power = __builtin_bit_cast(F, U(exponent + traits::exp_bias) << traits::mant_bits);
136   const F magnitude = (x.mantissa / s.mantissa) * power;
137   const F scaled = raw & traits::sign_mask ? -magnitude : magnitude;
138   if ( scaled >= F(127 - zero_point) ) return 127;
139   if ( scaled <= F(-128 - zero_point) ) return -128;
140   i32 integral = i32(scaled);
141   const F remainder = scaled - F(integral), absolute = remainder < F(0) ? -remainder : remainder;
142   if ( absolute > F(.5) || (absolute == F(.5) && (integral % 2)) ) integral += scaled < F(0) ? -1 : 1;
143   return i8(integral + zero_point);
144 }
