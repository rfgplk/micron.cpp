32 template<op::op_tag Op = op::none, blas_scalar T, typename VA, typename VB>
33   requires(mat_view_like<VA> and mat_view_like<VB>)
34 [[gnu::flatten]] inline constexpr void
35 omatcopy(T alpha, const VA &A, VB B) noexcept
36 {
37   constexpr bool tr = micron::same_as<Op, op::trans> or micron::same_as<Op, op::conj_trans>;
38   const usize m = tr ? A.cols : A.rows;
39   const usize n = tr ? A.rows : A.cols;
40   const ssize_t rs_A = level3::__impl_level3::rs_of(A);
41   const ssize_t cs_A = level3::__impl_level3::cs_of(A);
42   const ssize_t rs_B = level3::__impl_level3::rs_of(B);
43   const ssize_t cs_B = level3::__impl_level3::cs_of(B);
44   for ( usize i = 0; i < m; ++i )
45     for ( usize j = 0; j < n; ++j )
46       bits::mat_at(B.data, i, j, rs_B, cs_B)
47           = alpha * (tr ? bits::mat_at(A.data, j, i, rs_A, cs_A) : bits::mat_at(A.data, i, j, rs_A, cs_A));
48 }

50 template<op::op_tag Op = op::none, blas_scalar T, typename VA>
51   requires(mat_view_like<VA>)
52 [[gnu::flatten]] inline constexpr void
53 imatcopy(T alpha, VA A) noexcept
54 {
55   const ssize_t rs = level3::__impl_level3::rs_of(A);
56   const ssize_t cs = level3::__impl_level3::cs_of(A);
57   if constexpr ( micron::same_as<Op, op::none> ) {
58     for ( usize i = 0; i < A.rows; ++i )
59       for ( usize j = 0; j < A.cols; ++j ) bits::mat_at(A.data, i, j, rs, cs) *= alpha;
60   } else {
61     for ( usize i = 0; i < A.rows; ++i ) {
62       bits::mat_at(A.data, i, i, rs, cs) *= alpha;
63       for ( usize j = i + 1; j < A.cols; ++j ) {
64         T &aij = bits::mat_at(A.data, i, j, rs, cs);
65         T &aji = bits::mat_at(A.data, j, i, rs, cs);
66         const T t = alpha * aji;
67         aji = alpha * aij;
68         aij = t;
69       }
70     }
71   }
72 }

74 template<uplo::uplo_tag U, op::op_tag OpA = op::none, op::op_tag OpB = op::none, blas_scalar T, typename VA, typename VB, typename VC>
75   requires(mat_view_like<VA> and mat_view_like<VB> and mat_view_like<VC>)
76 [[gnu::flatten]] inline constexpr void
77 gemmt(T alpha, const VA &A, const VB &B, T beta, VC C) noexcept
78 {
79   constexpr bool trA = micron::same_as<OpA, op::trans> or micron::same_as<OpA, op::conj_trans>;
80   constexpr bool trB = micron::same_as<OpB, op::trans> or micron::same_as<OpB, op::conj_trans>;
81   constexpr bool upper = micron::same_as<U, uplo::upper>;
82   const usize n = trA ? A.cols : A.rows;      // C is n × n
83   const usize k = trA ? A.rows : A.cols;
84   const ssize_t rs_A = level3::__impl_level3::rs_of(A);
85   const ssize_t cs_A = level3::__impl_level3::cs_of(A);
86   const ssize_t rs_B = level3::__impl_level3::rs_of(B);
87   const ssize_t cs_B = level3::__impl_level3::cs_of(B);
88   const ssize_t rs_C = level3::__impl_level3::rs_of(C);
89   const ssize_t cs_C = level3::__impl_level3::cs_of(C);
90   for ( usize i = 0; i < n; ++i ) {
91     const usize j_lo = upper ? i : 0;
92     const usize j_hi = upper ? n : (i + 1);
93     for ( usize j = j_lo; j < j_hi; ++j ) {
94       T acc{};
95       for ( usize p = 0; p < k; ++p ) {
96         const T a = trA ? bits::mat_at(A.data, p, i, rs_A, cs_A) : bits::mat_at(A.data, i, p, rs_A, cs_A);
97         const T b = trB ? bits::mat_at(B.data, j, p, rs_B, cs_B) : bits::mat_at(B.data, p, j, rs_B, cs_B);
98         acc = bits::fma_acc<T>(a, b, acc);
99       }
100       T &cij = bits::mat_at(C.data, i, j, rs_C, cs_C);
101       if ( beta == T(0) )
102         cij = alpha * acc;
103       else if ( beta == T(1) )
104         cij = bits::fma_acc<T>(alpha, acc, cij);
105       else
106         cij = bits::fma_acc<T>(alpha, acc, beta * cij);
107     }
108   }
109 }
