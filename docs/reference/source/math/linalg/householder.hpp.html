29 // H = I - beta * v * v**T
30 template<ieee754_floating F, usize N> struct householder_t {
31   vec<F, N> v;
32   F beta;
33   usize k0;
34 };
35 
36 template<ieee754_floating F, usize N>
37 [[nodiscard]] inline householder_t<F, N>
38 householder_reflector(const vec<F, N> &x, usize k0) noexcept
39 {
40   householder_t<F, N> h{};
41   h.k0 = k0;
42   for ( usize i = 0; i < N; ++i ) h.v.data[i] = F(0);
43   if ( k0 >= N ) {
44     h.beta = F(0);
45     return h;
46   }
47   F xmax = F(0);
48   for ( usize i = k0; i < N; ++i ) {
49     F a = math::fabs(x.data[i]);
50     if ( a > xmax ) xmax = a;
51   }
52   if ( xmax == F(0) ) {
53     h.beta = F(0);
54     return h;
55   }
56   F sum = F(0);
57   F inv_xmax = F(1) / xmax;
58   for ( usize i = k0; i < N; ++i ) {
59     F s = x.data[i] * inv_xmax;
60     sum = math::fma<F>(s, s, sum);
61   }
62   F norm_x = xmax * math::fsqrt(sum);
63   // pick alpha such that v[k0] = x[k0]
64   F xk = x.data[k0];
65   F sign_xk = (xk >= F(0)) ? F(1) : F(-1);
66   F alpha = -sign_xk * norm_x;
67   // v[k0] = x[k0] - alpha
68   h.v.data[k0] = xk - alpha;
69   for ( usize i = k0 + 1; i < N; ++i ) h.v.data[i] = x.data[i];
70   F vv = h.v.data[k0] * h.v.data[k0];
71   for ( usize i = k0 + 1; i < N; ++i ) vv = math::fma<F>(h.v.data[i], h.v.data[i], vv);
72   h.beta = (vv > F(0)) ? F(2) / vv : F(0);
73   return h;
74 }

76 template<ieee754_floating F, usize R, usize C>
77 inline void
78 apply_householder_left(mat<F, R, C> &A, const vec<F, R> &v, F beta, usize k0) noexcept
79 {
80   if ( beta == F(0) ) return;
81   F w[C];
82   for ( usize j = 0; j < C; ++j ) w[j] = F(0);
83   for ( usize i = k0; i < R; ++i ) {
84     F vi = v.data[i];
85     if ( vi == F(0) ) continue;
86     F *__restrict__ row = A.data + i * C;
87     for ( usize j = 0; j < C; ++j ) w[j] = math::fma<F>(vi, row[j], w[j]);
88   }
89   for ( usize i = k0; i < R; ++i ) {
90     F vi = v.data[i];
91     if ( vi == F(0) ) continue;
92     F bvi = beta * vi;
93     F *__restrict__ row = A.data + i * C;
94     for ( usize j = 0; j < C; ++j ) row[j] -= bvi * w[j];
95   }
96 }

98 template<ieee754_floating F, usize R, usize C>
99 inline void
100 apply_householder_right(mat<F, R, C> &A, const vec<F, C> &v, F beta, usize k0) noexcept
101 {
102   if ( beta == F(0) ) return;
103   F w[R];
104   for ( usize i = 0; i < R; ++i ) {
105     F acc = F(0);
106     F *__restrict__ row = A.data + i * C;
107     for ( usize j = k0; j < C; ++j ) acc = math::fma<F>(row[j], v.data[j], acc);
108     w[i] = acc;
109   }
110   for ( usize i = 0; i < R; ++i ) {
111     F bwi = beta * w[i];
112     F *__restrict__ row = A.data + i * C;
113     for ( usize j = k0; j < C; ++j ) row[j] -= bwi * v.data[j];
114   }
115 }

122 template<ieee754_floating F>
123 [[nodiscard]] inline givens_t<F>
124 givens_rotation(F a, F b) noexcept
125 {
126   givens_t<F> g{};
127   if ( b == F(0) ) {
128     g.c = (a >= F(0)) ? F(1) : F(-1);
129     g.s = F(0);
130     return g;
131   }
132   if ( a == F(0) ) {
133     g.c = F(0);
134     g.s = (b >= F(0)) ? F(1) : F(-1);
135     return g;
136   }
137   if ( math::fabs(b) > math::fabs(a) ) {
138     F t = a / b;
139     F u = math::fsqrt(F(1) + t * t);
140     if ( b < F(0) ) u = -u;
141     g.s = F(1) / u;
142     g.c = g.s * t;
143   } else {
144     F t = b / a;
145     F u = math::fsqrt(F(1) + t * t);
146     if ( a < F(0) ) u = -u;
147     g.c = F(1) / u;
148     g.s = g.c * t;
149   }
150   return g;
151 }

153 // A = Q * H * Q**T
154 template<ieee754_floating F, usize N> struct hess_result {
155   mat<F, N, N> H;
156   mat<F, N, N> Q;
157 };
158 
159 template<ieee754_floating F, usize N>
160 [[nodiscard]] inline hess_result<F, N>
161 hessenberg(const mat<F, N, N> &A) noexcept
162 {
163   hess_result<F, N> r{ A, mat<F, N, N>::identity() };
164   if constexpr ( N <= 2 ) return r;
165 
166   for ( usize k = 0; k + 2 < N; ++k ) {
167     vec<F, N> x{};
168     for ( usize i = 0; i < N; ++i ) x.data[i] = F(0);
169     for ( usize i = k + 1; i < N; ++i ) x.data[i] = r.H.data[i * N + k];
170     auto h = householder_reflector<F, N>(x, k + 1);
171     if ( h.beta == F(0) ) continue;
172     apply_householder_left<F, N, N>(r.H, h.v, h.beta, k + 1);
173     apply_householder_right<F, N, N>(r.H, h.v, h.beta, k + 1);
174     apply_householder_right<F, N, N>(r.Q, h.v, h.beta, k + 1);
175     for ( usize i = k + 2; i < N; ++i ) r.H.data[i * N + k] = F(0);
176   }
177   return r;
178 }

186 template<ieee754_floating F, usize N>
187 [[nodiscard]] inline tridiag_result<F, N>
188 tridiagonalize_sym(const mat<F, N, N> &A) noexcept
189 {
190   auto h = hessenberg<F, N>(A);
191   tridiag_result<F, N> r{};
192   for ( usize i = 0; i < N; ++i ) r.diag.data[i] = h.H.data[i * N + i];
193   for ( usize i = 0; i + 1 < N; ++i ) r.subdiag.data[i] = h.H.data[(i + 1) * N + i];
194   if constexpr ( N >= 1 ) r.subdiag.data[N - 1] = F(0);
195   r.Q = h.Q;
196   return r;
197 }

199 // A = U * B * V**T
200 template<ieee754_floating F, usize R_, usize C_> struct bidiag_result {
201   mat<F, R_, C_> B;
202   mat<F, R_, R_> U;
203   mat<F, C_, C_> V;
204 };
205 
206 template<ieee754_floating F, usize R_, usize C_>
207 [[nodiscard]] inline bidiag_result<F, R_, C_>
208 bidiagonalize(const mat<F, R_, C_> &A) noexcept
209 {
210   bidiag_result<F, R_, C_> r{ A, mat<F, R_, R_>::identity(), mat<F, C_, C_>::identity() };
211   constexpr usize K = (R_ < C_) ? R_ : C_;
212   for ( usize k = 0; k < K; ++k ) {
213     {
214       vec<F, R_> x{};
215       for ( usize i = 0; i < R_; ++i ) x.data[i] = F(0);
216       for ( usize i = k; i < R_; ++i ) x.data[i] = r.B.data[i * C_ + k];
217       auto h = householder_reflector<F, R_>(x, k);
218       if ( h.beta != F(0) ) {
219         apply_householder_left<F, R_, C_>(r.B, h.v, h.beta, k);
220         apply_householder_right<F, R_, R_>(r.U, h.v, h.beta, k);
221         for ( usize i = k + 1; i < R_; ++i ) r.B.data[i * C_ + k] = F(0);
222       }
223     }
224     if ( k + 2 <= C_ ) {
225       vec<F, C_> y{};
226       for ( usize j = 0; j < C_; ++j ) y.data[j] = F(0);
227       for ( usize j = k + 1; j < C_; ++j ) y.data[j] = r.B.data[k * C_ + j];
228       auto h = householder_reflector<F, C_>(y, k + 1);
229       if ( h.beta != F(0) ) {
230         apply_householder_right<F, R_, C_>(r.B, h.v, h.beta, k + 1);
231         apply_householder_right<F, C_, C_>(r.V, h.v, h.beta, k + 1);
232         for ( usize j = k + 2; j < C_; ++j ) r.B.data[k * C_ + j] = F(0);
233       }
234     }
235   }
236   return r;
237 }

242 template<ieee754_floating F>
243 inline void
244 apply_householder_left_dyn(F *__restrict__ A, usize R, usize C, usize ld_A, const F *v, F beta, usize k0) noexcept
245 {
246   if ( beta == F(0) ) return;
247   micron::vector<F, micron::allocator_serial<>, false> w(C, F(0));
248   for ( usize i = k0; i < R; ++i ) {
249     F vi = v[i];
250     if ( vi == F(0) ) continue;
251     const F *row = A + i * ld_A;
252     for ( usize j = 0; j < C; ++j ) w.data()[j] = math::fma<F>(vi, row[j], w.data()[j]);
253   }
254   for ( usize i = k0; i < R; ++i ) {
255     F vi = v[i];
256     if ( vi == F(0) ) continue;
257     F bvi = beta * vi;
258     F *row = A + i * ld_A;
259     for ( usize j = 0; j < C; ++j ) row[j] -= bvi * w.data()[j];
260   }
261 }

263 template<ieee754_floating F>
264 inline void
265 apply_householder_right_dyn(F *__restrict__ A, usize R, usize C, usize ld_A, const F *v, F beta, usize k0) noexcept
266 {
267   if ( beta == F(0) ) return;
268   micron::vector<F, micron::allocator_serial<>, false> w(R, F(0));
269   for ( usize i = 0; i < R; ++i ) {
270     F acc = F(0);
271     const F *row = A + i * ld_A;
272     for ( usize j = k0; j < C; ++j ) acc = math::fma<F>(row[j], v[j], acc);
273     w.data()[i] = acc;
274   }
275   for ( usize i = 0; i < R; ++i ) {
276     F bwi = beta * w.data()[i];
277     F *row = A + i * ld_A;
278     for ( usize j = k0; j < C; ++j ) row[j] -= bwi * v[j];
279   }
280 }

282 template<ieee754_floating F>
283 inline F
284 householder_reflector_dyn(const F *x, usize R, usize k0, F *v_out) noexcept
285 {
286   for ( usize i = 0; i < R; ++i ) v_out[i] = F(0);
287   if ( k0 >= R ) return F(0);
288   F xmax = F(0);
289   for ( usize i = k0; i < R; ++i ) {
290     F a = math::fabs(x[i]);
291     if ( a > xmax ) xmax = a;
292   }
293   if ( xmax == F(0) ) return F(0);
294   F sum = F(0);
295   F inv_xmax = F(1) / xmax;
296   for ( usize i = k0; i < R; ++i ) {
297     F s = x[i] * inv_xmax;
298     sum = math::fma<F>(s, s, sum);
299   }
300   F norm_x = xmax * math::fsqrt(sum);
301   F xk = x[k0];
302   F sign_xk = (xk >= F(0)) ? F(1) : F(-1);
303   F alpha = -sign_xk * norm_x;
304   v_out[k0] = xk - alpha;
305   for ( usize i = k0 + 1; i < R; ++i ) v_out[i] = x[i];
306   F vv = v_out[k0] * v_out[k0];
307   for ( usize i = k0 + 1; i < R; ++i ) vv = math::fma<F>(v_out[i], v_out[i], vv);
308   return (vv > F(0)) ? F(2) / vv : F(0);
309 }
