29 template<ieee754_floating F, usize N>
30 inline void
31 chol_solve(const mat<F, N, N> &L, vec<F, N> &x_inout) noexcept
32 {
33   for ( usize i = 0; i < N; ++i ) {
34     F s = x_inout.data[i];
35     for ( usize j = 0; j < i; ++j ) s = math::fma<F>(-L.data[i * N + j], x_inout.data[j], s);
36     x_inout.data[i] = s / L.data[i * N + i];
37   }
38   for ( usize ii = N; ii-- > 0; ) {
39     F s = x_inout.data[ii];
40     for ( usize j = ii + 1; j < N; ++j ) s = math::fma<F>(-L.data[j * N + ii], x_inout.data[j], s);
41     x_inout.data[ii] = s / L.data[ii * N + ii];
42   }
43 }

45 template<ieee754_floating F>
46 inline void
47 chol_solve(const dynmat<F> &L, dynvec<F> &x_inout) noexcept
48 {
49   const usize N = L.rows;
50   const usize ld = L.ld;
51   const F *__restrict__ Ld = L.data();
52   for ( usize i = 0; i < N; ++i ) {
53     F s = x_inout[i];
54     for ( usize j = 0; j < i; ++j ) s = math::fma<F>(-Ld[i * ld + j], x_inout[j], s);
55     x_inout[i] = s / Ld[i * ld + i];
56   }
57   for ( usize ii = N; ii-- > 0; ) {
58     F s = x_inout[ii];
59     for ( usize j = ii + 1; j < N; ++j ) s = math::fma<F>(-Ld[j * ld + ii], x_inout[j], s);
60     x_inout[ii] = s / Ld[ii * ld + ii];
61   }
62 }

69 template<ieee754_floating F, usize N>
70 [[nodiscard]] inline inv_sympd_result<F, N>
71 inv_sympd(const mat<F, N, N> &A) noexcept
72 {
73   inv_sympd_result<F, N> r{ mat<F, N, N>::zero(), true };
74   auto c = decomp::chol(A);
75   if ( !c.spd ) {
76     r.spd = false;
77     return r;
78   }
79   for ( usize j = 0; j < N; ++j ) {
80     vec<F, N> e{};
81     for ( usize i = 0; i < N; ++i ) e.data[i] = (i == j) ? F(1) : F(0);
82     chol_solve<F, N>(c.L, e);
83     for ( usize i = 0; i < N; ++i ) r.X.data[i * N + j] = e.data[i];
84   }
85   return r;
86 }

98 // SPD-only Cholesky
99 template<ieee754_floating F>
100 [[nodiscard]] inline chol_result_dyn<F>
101 chol(const dynmat<F> &A) noexcept
102 {
103   const usize N = A.rows;
104   chol_result_dyn<F> r{ dynmat<F>::zero(N, N), true };
105   F *__restrict__ L = r.L.data();
106   const usize ld = r.L.ld;
107   const F *__restrict__ Ad = A.data();
108   const usize ldA = A.ld;
109   for ( usize i = 0; i < N; ++i ) {
110     for ( usize j = 0; j <= i; ++j ) {
111       F s = Ad[i * ldA + j];
112       for ( usize k = 0; k < j; ++k ) s = math::fma<F>(-L[i * ld + k], L[j * ld + k], s);
113       if ( i == j ) {
114         if ( s <= F(0) ) {
115           r.spd = false;
116           return r;
117         }
118         L[i * ld + j] = math::fsqrt(s);
119       } else {
120         L[i * ld + j] = s / L[j * ld + j];
121       }
122     }
123   }
124   return r;
125 }

127 template<ieee754_floating F>
128 [[nodiscard]] inline inv_sympd_result_dyn<F>
129 inv_sympd(const dynmat<F> &A) noexcept
130 {
131   const usize N = A.rows;
132   inv_sympd_result_dyn<F> r{ dynmat<F>::zero(N, N), true };
133   auto c = chol<F>(A);
134   if ( !c.spd ) {
135     r.spd = false;
136     return r;
137   }
138   for ( usize j = 0; j < N; ++j ) {
139     dynvec<F> e(N, F(0));
140     e[j] = F(1);
141     chol_solve<F>(c.L, e);
142     for ( usize i = 0; i < N; ++i ) r.X.at(i, j) = e[i];
143   }
144   return r;
145 }

147 template<ieee754_floating F, usize N>
148 [[nodiscard]] inline decomp::log_det_result<F>
149 log_det_sympd(const mat<F, N, N> &A) noexcept
150 {
151   auto c = decomp::chol(A);
152   if ( !c.spd ) return { F(0), 0 };
153   F s = F(0);
154   for ( usize i = 0; i < N; ++i ) s += math::log(c.L.data[i * N + i]);
155   return { F(2) * s, 1 };
156 }

158 template<ieee754_floating F>
159 [[nodiscard]] inline decomp::log_det_result<F>
160 log_det_sympd(const dynmat<F> &A) noexcept
161 {
162   auto c = chol<F>(A);
163   if ( !c.spd ) return { F(0), 0 };
164   const usize N = A.rows;
165   const usize ld = c.L.ld;
166   F s = F(0);
167   for ( usize i = 0; i < N; ++i ) s += math::log(c.L.data()[i * ld + i]);
168   return { F(2) * s, 1 };

178 template<ieee754_floating F>
179 [[nodiscard]] inline chol_pivot_result_dyn<F>
180 chol_pivot(const dynmat<F> &A, F tol = F(0)) noexcept
181 {
182   const usize N = A.rows;
183   chol_pivot_result_dyn<F> r{ dynmat<F>::zero(N, N), micron::vector<usize, micron::allocator_serial<>, false>(N), 0, F(0) };
184   for ( usize i = 0; i < N; ++i ) r.perm.data()[i] = i;
185 
186   micron::vector<F, micron::allocator_serial<>, false> diag(N);
187   for ( usize i = 0; i < N; ++i ) diag.data()[i] = A.at(i, i);
188 
189   dynmat<F> M(N, N);
190   for ( usize i = 0; i < N; ++i )
191     for ( usize j = 0; j < N; ++j ) M.at(i, j) = A.at(i, j);
192 
193   F max_diag = F(0);
194   for ( usize i = 0; i < N; ++i )
195     if ( diag.data()[i] > max_diag ) max_diag = diag.data()[i];
196   const F effective_tol = (tol > F(0)) ? tol : max_diag * default_eps<F>() * F(N);
197 
198   for ( usize k = 0; k < N; ++k ) {
199     F best = diag.data()[r.perm.data()[k]];
200     usize piv = k;
201     for ( usize i = k + 1; i < N; ++i ) {
202       F v = diag.data()[r.perm.data()[i]];
203       if ( v > best ) {
204         best = v;
205         piv = i;
206       }
207     }
208     if ( k == 0 ) r.max_pivot = best;
209     if ( best <= effective_tol ) break;
210 
211     if ( piv != k ) {
212       usize t = r.perm.data()[k];
213       r.perm.data()[k] = r.perm.data()[piv];
214       r.perm.data()[piv] = t;
215     }
216     const usize pk = r.perm.data()[k];
217 
218     F lkk = math::fsqrt(diag.data()[pk]);
219     F *__restrict__ L_data = r.L.data();
220     const usize ld_L = r.L.ld;
221     L_data[pk * ld_L + k] = lkk;
222     const F inv_lkk = F(1) / lkk;
223     const F *__restrict__ L_pk_row = L_data + pk * ld_L;
224     F *__restrict__ diag_p = diag.data();
225     const usize *__restrict__ perm_p = r.perm.data();
226     for ( usize ii = k + 1; ii < N; ++ii ) {
227       const usize pi = perm_p[ii];
228       F *__restrict__ L_pi_row = L_data + pi * ld_L;
229       F sub = F(0);
230       for ( usize j = 0; j < k; ++j ) sub = math::fma<F>(L_pi_row[j], L_pk_row[j], sub);
231       const F lik = (M.at(pi, pk) - sub) * inv_lkk;
232       L_pi_row[k] = lik;
233       diag_p[pi] -= lik * lik;
234     }
235     r.rank = k + 1;
236   }
237   return r;
238 }
