35 template<ieee754_floating F, usize N>
36 [[nodiscard]] inline eigen_sym_result<F, N>
37 eigen_sym(const mat<F, N, N> &A) noexcept
38 {
39   eigen_sym_result<F, N> r{};
40   if constexpr ( N == 1 ) {
41     r.values.data[0] = A.data[0];
42     r.vectors = mat<F, N, N>::identity();
43     r.converged = true;
44     return r;
45   } else {
46     mat<F, N, N> M = A;
47     mat<F, N, N> V = mat<F, N, N>::identity();
48     constexpr int max_sweeps = 50;
49     bool converged = false;
50 
51     for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
52       usize rotations = 0;
53 
54       for ( usize p = 0; p + 1 < N; ++p ) {
55         for ( usize q = p + 1; q < N; ++q ) {
56           F apq = M.data[p * N + q];
57           F app = M.data[p * N + p];
58           F aqq = M.data[q * N + q];
59           // scale-free skip
60           const F dd = math::fabs(app) + math::fabs(aqq);
61           if ( math::fabs(apq) + dd == dd ) continue;
62           ++rotations;
63           F theta = (aqq - app) / (F(2) * apq);
64           F t;
65           if ( theta >= F(0) )
66             t = F(1) / (theta + math::fsqrt(F(1) + theta * theta));
67           else
68             t = F(1) / (theta - math::fsqrt(F(1) + theta * theta));
69           F c = F(1) / math::fsqrt(F(1) + t * t);
70           F s = t * c;
71 
72           M.data[p * N + p] = app - t * apq;
73           M.data[q * N + q] = aqq + t * apq;
74           M.data[p * N + q] = F(0);
75           M.data[q * N + p] = F(0);
76 
77           for ( usize rr = 0; rr < N; ++rr ) {
78             if ( rr == p || rr == q ) continue;
79             F arp = M.data[rr * N + p];
80             F arq = M.data[rr * N + q];
81             F new_arp = c * arp - s * arq;
82             F new_arq = s * arp + c * arq;
83             M.data[rr * N + p] = new_arp;
84             M.data[p * N + rr] = new_arp;
85             M.data[rr * N + q] = new_arq;
86             M.data[q * N + rr] = new_arq;
87           }
88           for ( usize rr = 0; rr < N; ++rr ) {
89             F vrp = V.data[rr * N + p];
90             F vrq = V.data[rr * N + q];
91             V.data[rr * N + p] = c * vrp - s * vrq;
92             V.data[rr * N + q] = s * vrp + c * vrq;
93           }
94         }
95       }
96 
97       if ( rotations == 0 ) {
98         converged = true;
99         break;
100       }
101     }
102 
103     for ( usize i = 0; i < N; ++i ) r.values.data[i] = M.data[i * N + i];
104     r.vectors = V;
105     r.converged = converged;
106     return r;
107   }
108 }

119 template<ieee754_floating F, usize N>
120 inline void
121 francis_step(mat<F, N, N> &H, mat<F, N, N> &Z, usize lo, usize hi, F shift_s, F shift_t) noexcept
122 {
123 
124   const F a00 = H.data[lo * N + lo];
125   const F a01 = H.data[lo * N + (lo + 1)];
126   const F a10 = H.data[(lo + 1) * N + lo];
127   const F a11 = H.data[(lo + 1) * N + (lo + 1)];
128   const F a21 = H.data[(lo + 2) * N + (lo + 1)];
129 
130   F x = a00 * a00 + a01 * a10 - shift_s * a00 + shift_t;
131   F y = a10 * (a00 + a11 - shift_s);
132   F z = a10 * a21;
133 
134   for ( usize k = lo; k + 1 <= hi; ++k ) {
135     const bool short_ref = (k + 1 == hi);
136     if ( k > lo ) {
137       x = H.data[k * N + (k - 1)];
138       y = H.data[(k + 1) * N + (k - 1)];
139       z = (k + 2 <= hi) ? H.data[(k + 2) * N + (k - 1)] : F(0);
140     }
141     if ( short_ref ) z = F(0);
142 
143     F nrm_sq = x * x + y * y + z * z;
144     if ( nrm_sq == F(0) ) continue;
145     F nrm = math::fsqrt(nrm_sq);
146     F sign_x = (x >= F(0)) ? F(1) : F(-1);
147     F alpha = -sign_x * nrm;
148     F v0 = x - alpha;
149     F v1 = y;
150     F v2 = z;
151     F vv = v0 * v0 + v1 * v1 + v2 * v2;
152     if ( vv == F(0) ) continue;
153     F beta = F(2) / vv;
154 
155     const usize col_lo = (k > lo) ? (k - 1) : lo;
156     for ( usize cc = col_lo; cc < N; ++cc ) {
157       F h0 = H.data[k * N + cc];
158       F h1 = H.data[(k + 1) * N + cc];
159       F h2 = short_ref ? F(0) : H.data[(k + 2) * N + cc];
160       F w = v0 * h0 + v1 * h1 + (short_ref ? F(0) : v2 * h2);
161       F bw = beta * w;
162       H.data[k * N + cc] = h0 - bw * v0;
163       H.data[(k + 1) * N + cc] = h1 - bw * v1;
164       if ( !short_ref ) H.data[(k + 2) * N + cc] = h2 - bw * v2;
165     }
166 
167     usize row_top = short_ref ? (k + 3) : (k + 4);
168     if ( row_top > hi + 1 ) row_top = hi + 1;
169     if ( row_top > N ) row_top = N;
170     for ( usize rr = 0; rr < row_top; ++rr ) {
171       F h0 = H.data[rr * N + k];
172       F h1 = H.data[rr * N + (k + 1)];
173       F h2 = short_ref ? F(0) : H.data[rr * N + (k + 2)];
174       F w = h0 * v0 + h1 * v1 + (short_ref ? F(0) : h2 * v2);
175       F bw = beta * w;
176       H.data[rr * N + k] = h0 - bw * v0;
177       H.data[rr * N + (k + 1)] = h1 - bw * v1;
178       if ( !short_ref ) H.data[rr * N + (k + 2)] = h2 - bw * v2;
179     }
180 
181     for ( usize rr = 0; rr < N; ++rr ) {
182       F z0 = Z.data[rr * N + k];
183       F z1 = Z.data[rr * N + (k + 1)];
184       F z2 = short_ref ? F(0) : Z.data[rr * N + (k + 2)];
185       F w = z0 * v0 + z1 * v1 + (short_ref ? F(0) : z2 * v2);
186       F bw = beta * w;
187       Z.data[rr * N + k] = z0 - bw * v0;
188       Z.data[rr * N + (k + 1)] = z1 - bw * v1;
189       if ( !short_ref ) Z.data[rr * N + (k + 2)] = z2 - bw * v2;
190     }
191 
192     if ( k > lo ) {
193       H.data[(k + 1) * N + (k - 1)] = F(0);
194       if ( !short_ref && (k + 2 <= hi) ) H.data[(k + 2) * N + (k - 1)] = F(0);
195     }
196   }
197 }

201 template<ieee754_floating F, usize N>
202 [[nodiscard]] inline schur_result<F, N>
203 schur(const mat<F, N, N> &A) noexcept
204 {
205   schur_result<F, N> r{};
206   if constexpr ( N == 1 ) {
207     r.T = A;
208     r.Z = mat<F, 1, 1>::identity();
209     r.converged = true;
210     return r;
211   } else {
212     auto h = hessenberg<F, N>(A);
213     r.T = h.H;
214     r.Z = h.Q;
215     r.converged = false;
216 
217     const F eps = default_eps<F>();
218     const int max_iter = int(30 * N) + 30;
219     int iter = 0;
220     int stagnant = 0;
221     usize hi = N - 1;
222 
223     while ( true ) {
224 
225       while ( hi > 0 ) {
226         F sub = math::fabs(r.T.data[hi * N + (hi - 1)]);
227         F dd = math::fabs(r.T.data[(hi - 1) * N + (hi - 1)]) + math::fabs(r.T.data[hi * N + hi]);
228         if ( sub <= eps * dd || sub == F(0) ) {
229           r.T.data[hi * N + (hi - 1)] = F(0);
230           --hi;
231           stagnant = 0;
232           continue;
233         }
234         break;
235       }
236       if ( hi == 0 ) {
237         r.converged = true;
238         break;
239       }
240 
241       usize lo = hi;
242       while ( lo > 0 ) {
243         F sub = math::fabs(r.T.data[lo * N + (lo - 1)]);
244         F dd = math::fabs(r.T.data[(lo - 1) * N + (lo - 1)]) + math::fabs(r.T.data[lo * N + lo]);
245         if ( sub <= eps * dd ) {
246           r.T.data[lo * N + (lo - 1)] = F(0);
247           break;
248         }
249         --lo;
250       }
251 
252       if ( hi - lo == 1 ) {
253 
254         if ( lo == 0 ) {
255           r.converged = true;
256           break;
257         }
258         hi = lo - 1;
259         stagnant = 0;
260         continue;
261       }
262 
263       F shift_s, shift_t;
264       const F p = r.T.data[(hi - 1) * N + (hi - 1)];
265       const F q = r.T.data[hi * N + hi];
266       const F rr_off = r.T.data[(hi - 1) * N + hi];
267       const F ss_off = r.T.data[hi * N + (hi - 1)];
268       if ( stagnant >= 10 && stagnant % 10 == 0 ) {
269 
270         F s_excep = math::fabs(r.T.data[hi * N + (hi - 1)]) + (hi >= 2 ? math::fabs(r.T.data[(hi - 1) * N + (hi - 2)]) : F(0));
271         s_excep *= F(1.5);
272         shift_s = F(2) * s_excep;
273         shift_t = s_excep * s_excep;
274       } else {
275         shift_s = p + q;
276         shift_t = p * q - rr_off * ss_off;
277       }
278       __impl_decomp_schur::francis_step<F, N>(r.T, r.Z, lo, hi, shift_s, shift_t);
279       ++iter;
280       ++stagnant;
281       if ( iter >= max_iter ) {
282         r.converged = false;
283         break;
284       }
285     }
286     return r;
287   }
288 }

298 template<ieee754_floating F, usize N>
299 [[nodiscard]] inline eigen_result<F, N>
300 eigen(const mat<F, N, N> &A) noexcept
301 {
302   eigen_result<F, N> r{};
303   auto sr = schur<F, N>(A);
304   r.T = sr.T;
305   r.Z = sr.Z;
306   r.converged = sr.converged;
307 
308   for ( usize i = 0; i < N; ++i ) {
309     r.values_re.data[i] = F(0);
310     r.values_im.data[i] = F(0);
311   }
312   usize i = 0;
313   while ( i < N ) {
314     bool is_block = (i + 1 < N) && (r.T.data[(i + 1) * N + i] != F(0));
315     if ( !is_block ) {
316       r.values_re.data[i] = r.T.data[i * N + i];
317       r.values_im.data[i] = F(0);
318       ++i;
319     } else {
320       F a = r.T.data[i * N + i];
321       F b = r.T.data[i * N + (i + 1)];
322       F c = r.T.data[(i + 1) * N + i];
323       F d = r.T.data[(i + 1) * N + (i + 1)];
324       F tr = a + d;
325       F det = a * d - b * c;
326       F disc = tr * tr * F(0.25) - det;
327       if ( disc >= F(0) ) {
328         F sq = math::fsqrt(disc);
329         r.values_re.data[i] = tr * F(0.5) + sq;
330         r.values_im.data[i] = F(0);
331         r.values_re.data[i + 1] = tr * F(0.5) - sq;
332         r.values_im.data[i + 1] = F(0);
333       } else {
334         F sq = math::fsqrt(-disc);
335         r.values_re.data[i] = tr * F(0.5);
336         r.values_im.data[i] = sq;
337         r.values_re.data[i + 1] = tr * F(0.5);
338         r.values_im.data[i + 1] = -sq;
339       }
340       i += 2;
341     }
342   }
343   return r;
344 }

360 template<ieee754_floating F, usize R_, usize C_>
361   requires(R_ >= C_)
362 [[nodiscard]] inline svd_result<F, R_, C_>
363 svd(const mat<F, R_, C_> &A) noexcept
364 {
365   svd_result<F, R_, C_> r{};
366   r.U = mat<F, R_, R_>::zero();
367   r.V = mat<F, C_, C_>::identity();
368   r.converged = false;
369 
370   mat<F, R_, C_> W = A;
371   constexpr int max_sweeps = 30;
372 
373   for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
374     usize rotations = 0;
375     for ( usize p = 0; p + 1 < C_; ++p ) {
376       for ( usize q = p + 1; q < C_; ++q ) {
377 
378         F a = F(0), b = F(0), c = F(0);
379         for ( usize i = 0; i < R_; ++i ) {
380           const F wp = W.data[i * C_ + p];
381           const F wq = W.data[i * C_ + q];
382           a = math::fma<F>(wp, wp, a);
383           b = math::fma<F>(wq, wq, b);
384           c = math::fma<F>(wp, wq, c);
385         }
386         // scale-free skip
387         const F dd = math::fsqrt(a) * math::fsqrt(b) * F(R_);
388         if ( math::fabs(c) + dd == dd ) continue;
389         ++rotations;
390 
391         F theta = (b - a) / (F(2) * c);
392         F t;
393         if ( theta >= F(0) )
394           t = F(1) / (theta + math::fsqrt(F(1) + theta * theta));
395         else
396           t = F(1) / (theta - math::fsqrt(F(1) + theta * theta));
397         F cs = F(1) / math::fsqrt(F(1) + t * t);
398         F sn = t * cs;
399 
400         for ( usize i = 0; i < R_; ++i ) {
401           F wp = W.data[i * C_ + p];
402           F wq = W.data[i * C_ + q];
403           W.data[i * C_ + p] = cs * wp - sn * wq;
404           W.data[i * C_ + q] = sn * wp + cs * wq;
405         }
406 
407         for ( usize i = 0; i < C_; ++i ) {
408           F vp = r.V.data[i * C_ + p];
409           F vq = r.V.data[i * C_ + q];
410           r.V.data[i * C_ + p] = cs * vp - sn * vq;
411           r.V.data[i * C_ + q] = sn * vp + cs * vq;
412         }
413       }
414     }
415     if ( rotations == 0 ) {
416       r.converged = true;
417       break;
418     }
419   }
420 
421   for ( usize j = 0; j < C_; ++j ) {
422     F s = F(0);
423     for ( usize i = 0; i < R_; ++i ) s = math::fma<F>(W.data[i * C_ + j], W.data[i * C_ + j], s);
424     s = math::fsqrt(s);
425     r.S.data[j] = s;
426     if ( s > F(0) ) {
427       F inv = F(1) / s;
428       for ( usize i = 0; i < R_; ++i ) r.U.data[i * R_ + j] = W.data[i * C_ + j] * inv;
429     } else {
430       for ( usize i = 0; i < R_; ++i ) r.U.data[i * R_ + j] = F(0);
431     }
432   }
433 
434   for ( usize i = 0; i + 1 < C_; ++i ) {
435     usize maxj = i;
436     for ( usize j = i + 1; j < C_; ++j )
437       if ( r.S.data[j] > r.S.data[maxj] ) maxj = j;
438     if ( maxj != i ) {
439       F tmp = r.S.data[i];
440       r.S.data[i] = r.S.data[maxj];
441       r.S.data[maxj] = tmp;
442       for ( usize rr = 0; rr < R_; ++rr ) {
443         F t = r.U.data[rr * R_ + i];
444         r.U.data[rr * R_ + i] = r.U.data[rr * R_ + maxj];
445         r.U.data[rr * R_ + maxj] = t;
446       }
447       for ( usize rr = 0; rr < C_; ++rr ) {
448         F t = r.V.data[rr * C_ + i];
449         r.V.data[rr * C_ + i] = r.V.data[rr * C_ + maxj];
450         r.V.data[rr * C_ + maxj] = t;
451       }
452     }
453   }
454   return r;
455 }

460 template<ieee754_floating F>
461 [[gnu::always_inline]] inline F
462 pythag(F a, F b) noexcept
463 {
464   F absa = math::fabs(a), absb = math::fabs(b);
465   if ( absa > absb ) {
466     F r = absb / absa;
467     return absa * math::fsqrt(F(1) + r * r);
468   }
469   if ( absb == F(0) ) return F(0);
470   F r = absa / absb;
471   return absb * math::fsqrt(F(1) + r * r);
472 }

474 template<ieee754_floating F>
475 [[gnu::always_inline]] inline F
476 sgn_copy(F a, F b) noexcept
477 {
478   return (b >= F(0)) ? math::fabs(a) : -math::fabs(a);
479 }

481 template<ieee754_floating F>
482 inline bool
483 tqli(F *d, F *e, F *Z, usize N, usize ld_Z, int max_iter = 30) noexcept
484 {
485   if ( N <= 1 ) return true;
486 
487   for ( usize l = 0; l < N; ++l ) {
488     int iter = 0;
489     usize m;
490     do {
491 
492       for ( m = l; m + 1 < N; ++m ) {
493         F dd = math::fabs(d[m]) + math::fabs(d[m + 1]);
494         if ( math::fabs(e[m]) + dd == dd ) break;
495       }
496       if ( m != l ) {
497         if ( ++iter > max_iter ) return false;
498         F g = (d[l + 1] - d[l]) / (F(2) * e[l]);
499         F r = pythag<F>(g, F(1));
500         g = d[m] - d[l] + e[l] / (g + sgn_copy<F>(r, g));
501         F s = F(1), c = F(1), p = F(0);
502         usize i = m;
503         bool inner_break = false;
504         while ( i > l ) {
505           --i;
506           F f = s * e[i];
507           F b = c * e[i];
508           r = pythag<F>(f, g);
509           e[i + 1] = r;
510           if ( r == F(0) ) {
511             d[i + 1] -= p;
512             e[m] = F(0);
513             inner_break = true;
514             break;
515           }
516           s = f / r;
517           c = g / r;
518           g = d[i + 1] - p;
519           r = (d[i] - g) * s + F(2) * c * b;
520           p = s * r;
521           d[i + 1] = g + p;
522           g = c * r - b;
523 
524           for ( usize k = 0; k < N; ++k ) {
525             F zi = Z[k * ld_Z + i];
526             F zip = Z[k * ld_Z + i + 1];
527             Z[k * ld_Z + i + 1] = s * zi + c * zip;
528             Z[k * ld_Z + i] = c * zi - s * zip;
529           }
530         }
531         if ( inner_break ) continue;
532         d[l] -= p;
533         e[l] = g;
534         e[m] = F(0);
535       }
536     } while ( m != l );
537   }
538   return true;
539 }

543 template<ieee754_floating F, usize N>
544 [[nodiscard]] inline eigen_sym_result<F, N>
545 eigen_sym_qr(const mat<F, N, N> &A) noexcept
546 {
547   eigen_sym_result<F, N> r{};
548   if constexpr ( N == 1 ) {
549     r.values.data[0] = A.data[0];
550     r.vectors = mat<F, N, N>::identity();
551     r.converged = true;
552     return r;
553   } else {
554     auto tri = tridiagonalize_sym<F, N>(A);
555     F d[N];
556     F e[N];
557     for ( usize i = 0; i < N; ++i ) d[i] = tri.diag.data[i];
558     for ( usize i = 0; i < N; ++i ) e[i] = tri.subdiag.data[i];
559     r.vectors = tri.Q;
560     r.converged = __impl_decomp_qrtri::tqli<F>(d, e, r.vectors.data, N, N);
561     for ( usize i = 0; i < N; ++i ) r.values.data[i] = d[i];
562 
563     for ( usize i = 0; i + 1 < N; ++i ) {
564       usize mn = i;
565       for ( usize j = i + 1; j < N; ++j )
566         if ( r.values.data[j] < r.values.data[mn] ) mn = j;
567       if ( mn != i ) {
568         F tv = r.values.data[i];
569         r.values.data[i] = r.values.data[mn];
570         r.values.data[mn] = tv;
571         for ( usize k = 0; k < N; ++k ) {
572           F tt = r.vectors.data[k * N + i];
573           r.vectors.data[k * N + i] = r.vectors.data[k * N + mn];
574           r.vectors.data[k * N + mn] = tt;
575         }
576       }
577     }
578     return r;
579   }
580 }

582 template<ieee754_floating F> struct eigen_sym_result_dyn {
583   dynvec<F> values;
584   dynmat<F> vectors;
585   bool converged;
586 };
587 
588 template<ieee754_floating F>
589 [[nodiscard]] inline eigen_sym_result_dyn<F>
590 eigen_sym_qr(const dynmat<F> &A) noexcept
591 {
592   const usize N = A.rows;
593   eigen_sym_result_dyn<F> r{ dynvec<F>(N), dynmat<F>::identity(N), true };
594   if ( N <= 1 ) {
595     if ( N == 1 ) r.values[0] = A.at(0, 0);
596     return r;
597   }
598 
599   dynmat<F> H = A;
600   micron::vector<F, micron::allocator_serial<>, false> v(N);
601   micron::vector<F, micron::allocator_serial<>, false> w(N);
602   for ( usize k = 0; k + 2 < N; ++k ) {
603 
604     for ( usize i = 0; i < N; ++i ) v.data()[i] = F(0);
605     F *col_buf = v.data();
606     F xmax = F(0);
607     for ( usize i = k + 1; i < N; ++i ) {
608       F a = math::fabs(H.at(i, k));
609       if ( a > xmax ) xmax = a;
610     }
611     if ( xmax == F(0) ) continue;
612     F sum = F(0);
613     F inv_xmax = F(1) / xmax;
614     for ( usize i = k + 1; i < N; ++i ) {
615       F s = H.at(i, k) * inv_xmax;
616       sum = math::fma<F>(s, s, sum);
617     }
618     F norm_x = xmax * math::fsqrt(sum);
619     F xk = H.at(k + 1, k);
620     F sign_xk = (xk >= F(0)) ? F(1) : F(-1);
621     F alpha = -sign_xk * norm_x;
622     col_buf[k + 1] = xk - alpha;
623     for ( usize i = k + 2; i < N; ++i ) col_buf[i] = H.at(i, k);
624     F vv = col_buf[k + 1] * col_buf[k + 1];
625     for ( usize i = k + 2; i < N; ++i ) vv = math::fma<F>(col_buf[i], col_buf[i], vv);
626     F beta = (vv > F(0)) ? F(2) / vv : F(0);
627     if ( beta == F(0) ) continue;
628 
629     for ( usize j = 0; j < N; ++j ) w.data()[j] = F(0);
630     for ( usize i = k + 1; i < N; ++i ) {
631       F vi = col_buf[i];
632       if ( vi == F(0) ) continue;
633       for ( usize j = 0; j < N; ++j ) w.data()[j] = math::fma<F>(vi, H.at(i, j), w.data()[j]);
634     }
635     for ( usize i = k + 1; i < N; ++i ) {
636       F vi = col_buf[i];
637       if ( vi == F(0) ) continue;
638       F bvi = beta * vi;
639       for ( usize j = 0; j < N; ++j ) H.at(i, j) -= bvi * w.data()[j];
640     }
641 
642     for ( usize i = 0; i < N; ++i ) {
643       F acc = F(0);
644       for ( usize j = k + 1; j < N; ++j ) acc = math::fma<F>(H.at(i, j), col_buf[j], acc);
645       w.data()[i] = acc;
646     }
647     for ( usize i = 0; i < N; ++i ) {
648       F bwi = beta * w.data()[i];
649       for ( usize j = k + 1; j < N; ++j ) H.at(i, j) -= bwi * col_buf[j];
650     }
651 
652     for ( usize i = 0; i < N; ++i ) {
653       F acc = F(0);
654       for ( usize j = k + 1; j < N; ++j ) acc = math::fma<F>(r.vectors.at(i, j), col_buf[j], acc);
655       w.data()[i] = acc;
656     }
657     for ( usize i = 0; i < N; ++i ) {
658       F bwi = beta * w.data()[i];
659       for ( usize j = k + 1; j < N; ++j ) r.vectors.at(i, j) -= bwi * col_buf[j];
660     }
661 
662     for ( usize i = k + 2; i < N; ++i ) H.at(i, k) = F(0);
663   }
664 
665   micron::vector<F, micron::allocator_serial<>, false> d(N);
666   micron::vector<F, micron::allocator_serial<>, false> e(N);
667   for ( usize i = 0; i < N; ++i ) d.data()[i] = H.at(i, i);
668   for ( usize i = 0; i + 1 < N; ++i ) e.data()[i] = H.at(i + 1, i);
669   e.data()[N - 1] = F(0);
670 
671   r.converged = __impl_decomp_qrtri::tqli<F>(d.data(), e.data(), r.vectors.data(), N, r.vectors.ld);
672   for ( usize i = 0; i < N; ++i ) r.values[i] = d.data()[i];
673 
674   for ( usize i = 0; i + 1 < N; ++i ) {
675     usize mn = i;
676     for ( usize j = i + 1; j < N; ++j )
677       if ( r.values[j] < r.values[mn] ) mn = j;
678     if ( mn != i ) {
679       F tv = r.values[i];
680       r.values[i] = r.values[mn];
681       r.values[mn] = tv;
682       for ( usize k = 0; k < N; ++k ) {
683         F tt = r.vectors.at(k, i);
684         r.vectors.at(k, i) = r.vectors.at(k, mn);
685         r.vectors.at(k, mn) = tt;
686       }
687     }
688   }
689   return r;
690 }
