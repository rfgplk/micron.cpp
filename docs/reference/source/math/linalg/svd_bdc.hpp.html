38 // A = U x B x V^T where B is upper bidiagonal in the top-left K×K block
39 template<ieee754_floating F>
40 [[nodiscard]] inline bidiag_result_dyn<F>
41 bidiagonalize(const dynmat<F> &A) noexcept
42 {
43   const usize R = A.rows;
44   const usize C = A.cols;
45   const usize K = (R < C) ? R : C;
46   bidiag_result_dyn<F> r{ A, dynmat<F>::identity(R), dynmat<F>::identity(C) };
47 
48   micron::vector<F, micron::allocator_serial<>, false> x_row(R, F(0));
49   micron::vector<F, micron::allocator_serial<>, false> v_row(R, F(0));
50   micron::vector<F, micron::allocator_serial<>, false> x_col(C, F(0));
51   micron::vector<F, micron::allocator_serial<>, false> v_col(C, F(0));
52 
53   for ( usize k = 0; k < K; ++k ) {
54     // left Householder on column k below row k
55     for ( usize i = 0; i < R; ++i ) x_row.data()[i] = F(0);
56     for ( usize i = k; i < R; ++i ) x_row.data()[i] = r.B.at(i, k);
57     F beta_l = __impl_decomp::householder_reflector_dyn<F>(x_row.data(), R, k, v_row.data());
58     if ( beta_l != F(0) ) {
59       __impl_decomp::apply_householder_left_dyn<F>(r.B.data(), R, C, r.B.ld, v_row.data(), beta_l, k);
60       __impl_decomp::apply_householder_right_dyn<F>(r.U.data(), R, R, r.U.ld, v_row.data(), beta_l, k);
61       for ( usize i = k + 1; i < R; ++i ) r.B.at(i, k) = F(0);
62     }
63     // right Householder on row k right of column k+1
64     if ( k + 2 <= C ) {
65       for ( usize j = 0; j < C; ++j ) x_col.data()[j] = F(0);
66       for ( usize j = k + 1; j < C; ++j ) x_col.data()[j] = r.B.at(k, j);
67       F beta_r = __impl_decomp::householder_reflector_dyn<F>(x_col.data(), C, k + 1, v_col.data());
68       if ( beta_r != F(0) ) {
69         __impl_decomp::apply_householder_right_dyn<F>(r.B.data(), R, C, r.B.ld, v_col.data(), beta_r, k + 1);
70         __impl_decomp::apply_householder_right_dyn<F>(r.V.data(), C, C, r.V.ld, v_col.data(), beta_r, k + 1);
71         for ( usize j = k + 2; j < C; ++j ) r.B.at(k, j) = F(0);
72       }
73     }
74   }
75   return r;
76 }

81 template<ieee754_floating F>
82 [[gnu::always_inline]] inline F
83 pythag(F a, F b) noexcept
84 {
85   F absa = math::fabs(a), absb = math::fabs(b);
86   if ( absa > absb ) {
87     F r = absb / absa;
88     return absa * math::fsqrt(F(1) + r * r);
89   }
90   if ( absb == F(0) ) return F(0);
91   F r = absa / absb;
92   return absb * math::fsqrt(F(1) + r * r);
93 }

95 template<ieee754_floating F>
96 [[gnu::always_inline]] inline F
97 sgn_copy(F a, F b) noexcept
98 {
99   return (b >= F(0)) ? math::fabs(a) : -math::fabs(a);

102 template<ieee754_floating F>
103 inline bool
104 bidiag_svd_diag(F *__restrict__ d, F *__restrict__ e, usize n, F *__restrict__ U_data, usize U_rows, usize U_ld, F *__restrict__ V_data,
105                 usize V_rows, usize V_ld, int max_iter) noexcept
106 {
107   if ( n == 0 ) return true;
108   if ( n == 1 ) {
109     if ( d[0] < F(0) ) {
110       d[0] = -d[0];
111       for ( usize j = 0; j < V_rows; ++j ) V_data[j * V_ld + 0] = -V_data[j * V_ld + 0];
112     }
113     return true;
114   }
115 
116   // scale for negligibility tests
117   F anorm = math::fabs(d[0]);
118   for ( usize i = 0; i + 1 < n; ++i ) {
119     F v = math::fabs(d[i + 1]) + math::fabs(e[i]);
120     if ( v > anorm ) anorm = v;
121   }
122 
123   // process from bottom up
124   for ( usize k_top = n; k_top-- > 0; ) {
125     const usize k = k_top;
126     int iter = 0;
127     while ( true ) {
128       // find l in [0..k] such that the submatrix B[l..k] is irreducible
129       bool flag = true;
130       usize l = 0;
131       bool found = false;
132       for ( usize ll = k; ll-- > 0; ) {
133         // ll in [0..k-1];  test e[ll]
134         if ( math::fabs(e[ll]) + anorm == anorm ) {
135           flag = false;
136           e[ll] = F(0);
137           l = ll + 1;
138           found = true;
139           break;
140         }
141         if ( math::fabs(d[ll]) + anorm == anorm ) {
142           l = ll + 1;
143           found = true;
144           break;
145         }
146       }
147       if ( !found ) {
148         l = 0;
149         flag = false;
150       }
151 
152       if ( flag ) {
153         // chase the zero in d[l-1] across e[l-1..k-1]; left Givens into U cols (l-1, i)
154         F c = F(0), s = F(1);
155         const usize nm = l - 1;
156         for ( usize i = l; i <= k; ++i ) {
157           F f = s * e[i - 1];
158           e[i - 1] = c * e[i - 1];
159           if ( math::fabs(f) + anorm == anorm ) break;
160           F g = d[i];
161           F h = pythag<F>(f, g);
162           d[i] = h;
163           h = F(1) / h;
164           c = g * h;
165           s = -f * h;
166           for ( usize j = 0; j < U_rows; ++j ) {
167             F y = U_data[j * U_ld + nm];
168             F z = U_data[j * U_ld + i];
169             U_data[j * U_ld + nm] = y * c + z * s;
170             U_data[j * U_ld + i] = z * c - y * s;
171           }
172         }
173       }
174 
175       F zk = d[k];
176       if ( l == k ) {
177         if ( zk < F(0) ) {
178           d[k] = -zk;
179           for ( usize j = 0; j < V_rows; ++j ) V_data[j * V_ld + k] = -V_data[j * V_ld + k];
180         }
181         break;
182       }
183 
184       if ( ++iter > max_iter ) return false;
185 
186       // shift from bottom 2x2 of B^T B over rows l..k
187       F x = d[l];
188       const usize nm = k - 1;
189       F y = d[nm];
190       F g = (nm > 0) ? e[nm - 1] : F(0);
191       F h = e[nm];
192       F f = ((y - zk) * (y + zk) + (g - h) * (g + h)) / (F(2) * h * y);
193       F gg = pythag<F>(f, F(1));
194       f = ((x - zk) * (x + zk) + h * ((y / (f + sgn_copy<F>(gg, f))) - h)) / x;
195 
196       // QR sweep over j in [l..nm]
197       F c = F(1), s = F(1);
198       for ( usize j = l; j <= nm; ++j ) {
199         const usize i = j + 1;
200         F gv = e[j];
201         F yv = d[i];
202         F hv = s * gv;
203         gv = c * gv;
204         F zv = pythag<F>(f, hv);
205         if ( j > 0 ) e[j - 1] = zv;
206         if ( zv != F(0) ) {
207           F inv_z = F(1) / zv;
208           c = f * inv_z;
209           s = hv * inv_z;
210         }
211         f = x * c + gv * s;
212         gv = gv * c - x * s;
213         hv = yv * s;
214         yv = yv * c;
215         // right Givens into V cols (j, i)
216         for ( usize jj = 0; jj < V_rows; ++jj ) {
217           F xv = V_data[jj * V_ld + j];
218           F zw = V_data[jj * V_ld + i];
219           V_data[jj * V_ld + j] = xv * c + zw * s;
220           V_data[jj * V_ld + i] = zw * c - xv * s;
221         }
222         F zv2 = pythag<F>(f, hv);
223         d[j] = zv2;
224         if ( zv2 != F(0) ) {
225           F inv_z = F(1) / zv2;
226           c = f * inv_z;
227           s = hv * inv_z;
228         }
229         f = c * gv + s * yv;
230         x = c * yv - s * gv;
231         // left Givens into U cols (j, i)
232         for ( usize jj = 0; jj < U_rows; ++jj ) {
233           F yu = U_data[jj * U_ld + j];
234           F zu = U_data[jj * U_ld + i];
235           U_data[jj * U_ld + j] = yu * c + zu * s;
236           U_data[jj * U_ld + i] = zu * c - yu * s;
237         }
238       }
239       // cleanup the chased entries
240       if ( l > 0 ) e[l - 1] = F(0);
241       e[k - 1] = f;
242       d[k] = x;
243     }
244   }
245   return true;
246 }

248 // Cuppen's algorithm to T = B^T B
249 template<ieee754_floating F>
250 inline F
251 solve_secular_root(const F *d, const F *z2, usize n, usize idx, F rho, F upper_pad, int max_iter, bool *converged = nullptr) noexcept
252 {
253   F lo = d[idx];
254   F hi = (idx + 1 < n) ? d[idx + 1] : (d[n - 1] + upper_pad);
255   F lam = (lo + hi) * F(0.5);
256   bool ok = false;
257 
258   for ( int iter = 0; iter < max_iter; ++iter ) {
259     F f = F(1);
260     F fp = F(0);
261     bool degenerate = false;
262     for ( usize i = 0; i < n; ++i ) {
263       F denom = d[i] - lam;
264       F scale = math::fabs(d[i]) + math::fabs(lam) + F(1);
265       if ( math::fabs(denom) < default_eps<F>() * scale * F(8) ) {
266         degenerate = true;
267         break;
268       }
269       F t = z2[i] / denom;
270       f += rho * t;
271       fp += rho * t / denom;
272     }
273 
274     if ( degenerate ) {
275       // fall back to bisection step
276       F mid = (lo + hi) * F(0.5);
277       if ( mid == lam ) break;
278       lam = mid;
279       continue;
280     }
281 
282     if ( math::fabs(f) <= default_eps<F>() * F(100) * (math::fabs(lam) + F(1)) ) {
283       ok = true;
284       break;
285     }
286 
287     if ( f > F(0) )
288       hi = lam;
289     else
290       lo = lam;
291 
292     F delta = (fp != F(0)) ? -f / fp : F(0);
293     F new_lam = lam + delta;
294     if ( delta == F(0) || new_lam <= lo || new_lam >= hi ) {
295       new_lam = (lo + hi) * F(0.5);
296     }
297     if ( new_lam == lam ) {
298       ok = true;
299       break;
300     }
301     lam = new_lam;
302   }
303   if ( converged ) *converged = ok;
304   return lam;
305 }

307 // Cuppen's recursive symmetric tridiagonal D&C
308 template<ieee754_floating F>
309 inline bool
310 cuppen_tridiag_dnc(F *d, F *e, usize n, F *Q_data, usize Q_ld, usize threshold, int max_iter) noexcept
311 {
312   if ( n == 0 ) return true;
313   if ( n == 1 ) return true;
314 
315   if ( n <= threshold ) {
316     return decomp::__impl_decomp_qrtri::tqli<F>(d, e, Q_data, n, Q_ld, max_iter);
317   }
318 
319   const usize k = n / 2;
320   const F beta = e[k - 1];
321 
322   const F rho = math::fabs(beta);
323   const F sign_beta = (beta >= F(0)) ? F(1) : F(-1);
324 
325   // rank-1 split T = T_hat + rho * v v^T with v = e_{k-1} + sign_beta * e_k
326   d[k - 1] -= rho;
327   d[k] -= rho;
328 
329   // recurse on sub-blocks; each sub-block of Q is currently identity
330   if ( !cuppen_tridiag_dnc<F>(d, e, k, Q_data, Q_ld, threshold, max_iter) ) return false;
331   if ( !cuppen_tridiag_dnc<F>(d + k, e + k, n - k, Q_data + k * Q_ld + k, Q_ld, threshold, max_iter) ) return false;
332 
333   // u_i = (Q^T v)_i where v = e_{k-1} + sign_beta * e_k
334   micron::vector<F, micron::allocator_serial<>, false> u(n, F(0));
335   for ( usize i = 0; i < k; ++i ) u.data()[i] = Q_data[(k - 1) * Q_ld + i];
336   for ( usize i = k; i < n; ++i ) u.data()[i] = sign_beta * Q_data[k * Q_ld + i];
337 
338   // sort d ascending; carry u alongside
339   micron::vector<usize, micron::allocator_serial<>, false> perm(n);
340   for ( usize i = 0; i < n; ++i ) perm.data()[i] = i;
341   for ( usize i = 0; i + 1 < n; ++i ) {
342     usize best = i;
343     for ( usize j = i + 1; j < n; ++j ) {
344       if ( d[perm.data()[j]] < d[perm.data()[best]] ) best = j;
345     }
346     if ( best != i ) {
347       usize t = perm.data()[i];
348       perm.data()[i] = perm.data()[best];
349       perm.data()[best] = t;
350     }
351   }
352   micron::vector<F, micron::allocator_serial<>, false> d_sorted(n, F(0));
353   micron::vector<F, micron::allocator_serial<>, false> u_sorted(n, F(0));
354   for ( usize i = 0; i < n; ++i ) {
355     d_sorted.data()[i] = d[perm.data()[i]];
356     u_sorted.data()[i] = u.data()[perm.data()[i]];
357   }
358 
359   // ρ for secular root finder
360   micron::vector<F, micron::allocator_serial<>, false> z2(n);
361   F sum_z2 = F(0);
362   for ( usize i = 0; i < n; ++i ) {
363     z2.data()[i] = u_sorted.data()[i] * u_sorted.data()[i];
364     sum_z2 += z2.data()[i];
365   }
366   F upper_pad = rho * sum_z2 + (math::fabs(d_sorted.data()[n - 1]) + F(1)) * F(0.01);
367   if ( upper_pad <= F(0) ) upper_pad = F(1);
368 
369   // find new eigenvalues; honestly track per-root convergence
370   micron::vector<F, micron::allocator_serial<>, false> lam(n, F(0));
371   bool all_converged = true;
372   for ( usize i = 0; i < n; ++i ) {
373     bool root_ok = false;
374     lam.data()[i] = solve_secular_root<F>(d_sorted.data(), z2.data(), n, i, rho, upper_pad, max_iter, &root_ok);
375     if ( !root_ok ) all_converged = false;
376   }
377 
378   // normalize each column
379   micron::vector<F, micron::allocator_serial<>, false> V_merge(n * n, F(0));
380   for ( usize i = 0; i < n; ++i ) {
381     F norm_sq = F(0);
382     for ( usize j = 0; j < n; ++j ) {
383       F denom = d_sorted.data()[j] - lam.data()[i];
384       if ( math::fabs(denom) < default_eps<F>() * F(8) ) denom = (denom >= F(0)) ? default_eps<F>() : -default_eps<F>();
385       F vj = u_sorted.data()[j] / denom;
386       V_merge.data()[j * n + i] = vj;
387       norm_sq = math::fma<F>(vj, vj, norm_sq);
388     }
389     F norm = math::fsqrt(norm_sq);
390     if ( norm > F(0) ) {
391       F inv = F(1) / norm;
392       for ( usize j = 0; j < n; ++j ) V_merge.data()[j * n + i] *= inv;
393     }
394   }
395 
396   micron::vector<F, micron::allocator_serial<>, false> V_unperm(n * n, F(0));
397   for ( usize i = 0; i < n; ++i ) {
398     for ( usize j = 0; j < n; ++j ) {
399       // row in original ordering = perm.data()[j]; column = i
400       V_unperm.data()[perm.data()[j] * n + i] = V_merge.data()[j * n + i];
401     }
402   }
403   for ( usize i = 0; i < n; ++i ) d[i] = lam.data()[i];
404 
405   // Q := Q * V_unperm   (Q has columns = old eigenvectors; we transform to new basis)
406   micron::vector<F, micron::allocator_serial<>, false> tmp(n * n, F(0));
407   for ( usize i = 0; i < n; ++i ) {
408     for ( usize j = 0; j < n; ++j ) {
409       F s = F(0);
410       for ( usize kk = 0; kk < n; ++kk ) s = math::fma<F>(Q_data[i * Q_ld + kk], V_unperm.data()[kk * n + j], s);
411       tmp.data()[i * n + j] = s;
412     }
413   }
414   for ( usize i = 0; i < n; ++i )
415     for ( usize j = 0; j < n; ++j ) Q_data[i * Q_ld + j] = tmp.data()[i * n + j];
416 
417   return all_converged;
418 }

429 template<ieee754_floating F>
430 [[nodiscard]] inline svd_bdc_result_dyn<F>
431 svd_bdc(const dynmat<F> &A, int max_iter = 60) noexcept
432 {
433   const usize R = A.rows;
434   const usize C = A.cols;
435   const usize K = (R < C) ? R : C;
436   svd_bdc_result_dyn<F> r{};
437 
438   if ( K == 0 ) {
439     r.U = dynmat<F>::identity(R);
440     r.S = dynvec<F>(0);
441     r.V = dynmat<F>::identity(C);
442     r.converged = true;
443     return r;
444   }
445 
446   // For R < C, bidiagonalize leaves B(R-1, R) outside the K x K block; SVD(A) = (V, S, U) of SVD(A^T)
447   if ( R < C ) {
448     dynmat<F> At(C, R);
449     for ( usize i = 0; i < R; ++i )
450       for ( usize j = 0; j < C; ++j ) At.at(j, i) = A.at(i, j);
451     auto rt = svd_bdc<F>(At, max_iter);
452     r.U = micron::move(rt.V);
453     r.S = micron::move(rt.S);
454     r.V = micron::move(rt.U);
455     r.converged = rt.converged;
456     return r;
457   }
458 
459   auto bd = bidiagonalize<F>(A);
460 
461   micron::vector<F, micron::allocator_serial<>, false> d(K, F(0));
462   micron::vector<F, micron::allocator_serial<>, false> e(K > 0 ? K - 1 : 0, F(0));
463   for ( usize i = 0; i < K; ++i ) d.data()[i] = bd.B.at(i, i);
464   for ( usize i = 0; i + 1 < K; ++i ) e.data()[i] = bd.B.at(i, i + 1);
465 
466   r.converged = __impl_svd_bdc::bidiag_svd_diag<F>(d.data(), e.data(), K, bd.U.data(), R, bd.U.ld, bd.V.data(), C, bd.V.ld, max_iter);
467 
468   // sort singular values descending; permute cols of U (first K) and V (first K)
469   micron::vector<usize, micron::allocator_serial<>, false> idx(K);
470   for ( usize i = 0; i < K; ++i ) idx.data()[i] = i;
471   // simple selection sort by d (small K typically)
472   for ( usize i = 0; i + 1 < K; ++i ) {
473     usize best = i;
474     F best_v = d.data()[idx.data()[i]];
475     for ( usize j = i + 1; j < K; ++j ) {
476       F v = d.data()[idx.data()[j]];
477       if ( v > best_v ) {
478         best = j;
479         best_v = v;
480       }
481     }
482     if ( best != i ) {
483       usize t = idx.data()[i];
484       idx.data()[i] = idx.data()[best];
485       idx.data()[best] = t;
486     }
487   }
488 
489   r.S = dynvec<F>(K);
490   for ( usize i = 0; i < K; ++i ) r.S[i] = d.data()[idx.data()[i]];
491 
492   // permute columns of U[:, 0:K] and V[:, 0:K] using idx
493   r.U = dynmat<F>(R, R);
494   for ( usize i = 0; i < R; ++i ) {
495     for ( usize j = 0; j < K; ++j ) r.U.at(i, j) = bd.U.at(i, idx.data()[j]);
496     for ( usize j = K; j < R; ++j ) r.U.at(i, j) = bd.U.at(i, j);
497   }
498   r.V = dynmat<F>(C, C);
499   for ( usize i = 0; i < C; ++i ) {
500     for ( usize j = 0; j < K; ++j ) r.V.at(i, j) = bd.V.at(i, idx.data()[j]);
501     for ( usize j = K; j < C; ++j ) r.V.at(i, j) = bd.V.at(i, j);
502   }
503 
504   return r;
505 }

507 // SVD via Cuppen
508 template<ieee754_floating F>
509 [[nodiscard]] inline svd_bdc_result_dyn<F>
510 svd_bdc_dnc(const dynmat<F> &A, usize threshold = 16, int max_iter = 60) noexcept
511 {
512   const usize R = A.rows;
513   const usize C = A.cols;
514   svd_bdc_result_dyn<F> r{};
515 
516   if ( R == 0 || C == 0 ) {
517     r.U = dynmat<F>::identity(R);
518     r.S = dynvec<F>(0);
519     r.V = dynmat<F>::identity(C);
520     r.converged = true;
521     return r;
522   }
523 
524   // For R < C, SVD(A) = (V, S, U) of SVD(A^T)
525   if ( R < C ) {
526     dynmat<F> At(C, R);
527     for ( usize i = 0; i < R; ++i )
528       for ( usize j = 0; j < C; ++j ) At.at(j, i) = A.at(i, j);
529     auto rt = svd_bdc_dnc<F>(At, threshold, max_iter);
530     r.U = micron::move(rt.V);
531     r.S = micron::move(rt.S);
532     r.V = micron::move(rt.U);
533     r.converged = rt.converged;
534     return r;
535   }
536 
537   // R >= C path
538   const usize K = C;
539 
540   if ( threshold < K ) threshold = K;
541 
542   auto bd = bidiagonalize<F>(A);
543 
544   // build tridiagonal T = B^T B (C x C symmetric, in d_t and e_t)
545   micron::vector<F, micron::allocator_serial<>, false> d_t(K, F(0));
546   micron::vector<F, micron::allocator_serial<>, false> e_t(K > 0 ? K - 1 : 0, F(0));
547   micron::vector<F, micron::allocator_serial<>, false> d_b(K, F(0));      // diag of B
548   micron::vector<F, micron::allocator_serial<>, false> e_b(K > 0 ? K - 1 : 0, F(0));
549   for ( usize i = 0; i < K; ++i ) d_b.data()[i] = bd.B.at(i, i);
550   for ( usize i = 0; i + 1 < K; ++i ) e_b.data()[i] = bd.B.at(i, i + 1);
551 
552   d_t.data()[0] = d_b.data()[0] * d_b.data()[0];
553   for ( usize i = 1; i < K; ++i ) d_t.data()[i] = d_b.data()[i] * d_b.data()[i] + e_b.data()[i - 1] * e_b.data()[i - 1];
554   for ( usize i = 0; i + 1 < K; ++i ) e_t.data()[i] = d_b.data()[i] * e_b.data()[i];
555 
556   // Cuppen on T
557   dynmat<F> Q = dynmat<F>::identity(K);
558   bool conv = __impl_svd_bdc::cuppen_tridiag_dnc<F>(d_t.data(), e_t.data(), K, Q.data(), Q.ld, threshold, max_iter);
559   r.converged = conv;
560 
561   // d_t now holds eigenvalues; sigma = sqrt(eigenvalue); clamp non-negative
562   micron::vector<F, micron::allocator_serial<>, false> sigma(K, F(0));
563   for ( usize i = 0; i < K; ++i ) {
564     F lam = d_t.data()[i];
565     if ( lam < F(0) ) lam = F(0);
566     sigma.data()[i] = math::fsqrt(lam);
567   }
568 
569   // right singular vectors of B (in bidiag space) = columns of Q
570   // left singular vectors of B = B * Q * diag(1/sigma)
571   dynmat<F> U_inner = dynmat<F>::zero(R, K);
572   // For thin form: U_inner[:, j] = (B * Q[:, j]) / sigma[j]
573   for ( usize j = 0; j < K; ++j ) {
574     // compute B * Q[:, j] using bidiagonal structure
575     // (B * q)_i = d_b[i] * q_i + e_b[i] * q_{i+1}  for i in [0..K-2]
576     // (B * q)_{K-1} = d_b[K-1] * q_{K-1}
577     // (B * q)_i = 0   for i >= K  (since B's bidiagonal is in top K rows; row indices >= K are zero in B)
578     for ( usize i = 0; i + 1 < K; ++i ) {
579       F bq = d_b.data()[i] * Q.at(i, j) + e_b.data()[i] * Q.at(i + 1, j);
580       U_inner.at(i, j) = bq;
581     }
582     if ( K > 0 ) U_inner.at(K - 1, j) = d_b.data()[K - 1] * Q.at(K - 1, j);
583     for ( usize i = K; i < R; ++i ) U_inner.at(i, j) = F(0);
584     // normalize by sigma if nonzero; else the vector itself is in the null space
585     if ( sigma.data()[j] > F(0) ) {
586       F inv = F(1) / sigma.data()[j];
587       for ( usize i = 0; i < R; ++i ) U_inner.at(i, j) *= inv;
588     } else {
589       // sigma == 0: leave U_inner[:, j] as-is (will be re-orthogonalized below or zeroed)
590       for ( usize i = 0; i < R; ++i ) U_inner.at(i, j) = F(0);
591     }
592   }
593 
594   // sort sigma descending; permute U_inner cols and Q cols accordingly
595   micron::vector<usize, micron::allocator_serial<>, false> idx(K);
596   for ( usize i = 0; i < K; ++i ) idx.data()[i] = i;
597   for ( usize i = 0; i + 1 < K; ++i ) {
598     usize best = i;
599     for ( usize j = i + 1; j < K; ++j ) {
600       if ( sigma.data()[idx.data()[j]] > sigma.data()[idx.data()[best]] ) best = j;
601     }
602     if ( best != i ) {
603       usize t = idx.data()[i];
604       idx.data()[i] = idx.data()[best];
605       idx.data()[best] = t;
606     }
607   }
608 
609   r.S = dynvec<F>(K);
610   for ( usize i = 0; i < K; ++i ) r.S[i] = sigma.data()[idx.data()[i]];
611 
612   r.U = dynmat<F>(R, R);
613   for ( usize i = 0; i < R; ++i ) {
614     for ( usize j = 0; j < K; ++j ) {
615       F s = F(0);
616       for ( usize kk = 0; kk < R; ++kk ) s = math::fma<F>(bd.U.at(i, kk), U_inner.at(kk, idx.data()[j]), s);
617       r.U.at(i, j) = s;
618     }
619     for ( usize j = K; j < R; ++j ) r.U.at(i, j) = bd.U.at(i, j);
620   }
621 
622   r.V = dynmat<F>(C, C);
623   for ( usize i = 0; i < C; ++i ) {
624     for ( usize j = 0; j < K; ++j ) {
625       F s = F(0);
626       for ( usize kk = 0; kk < C; ++kk ) s = math::fma<F>(bd.V.at(i, kk), Q.at(kk, idx.data()[j]), s);
627       r.V.at(i, j) = s;
628     }
629     for ( usize j = K; j < C; ++j ) r.V.at(i, j) = bd.V.at(i, j);
630   }
631 
632   return r;
633 }
