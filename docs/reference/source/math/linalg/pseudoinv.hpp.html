35 template<ieee754_floating F> struct svd_result_dyn {
36   dynmat<F> U;      // rows × rows (full)
37   dynvec<F> S;      // length min(rows, cols)
38   dynmat<F> V;      // cols × cols
39   bool converged;
40 };

42 template<ieee754_floating F>
43 [[nodiscard]] inline svd_result_dyn<F>
44 svd(const dynmat<F> &A) noexcept
45 {
46   const usize R = A.rows;
47   const usize C = A.cols;
48   svd_result_dyn<F> r{ dynmat<F>::zero(R, R), dynvec<F>(C), dynmat<F>::identity(C), false };
49 
50   dynmat<F> W = A;
51   constexpr int max_sweeps = 30;
52   F *__restrict__ Wp = W.data();
53   F *__restrict__ Vp = r.V.data();
54   const usize ld_W = W.ld;
55   const usize ld_V = r.V.ld;
56 
57   for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
58     usize rotations = 0;
59     for ( usize p = 0; p + 1 < C; ++p ) {
60       for ( usize q = p + 1; q < C; ++q ) {
61         F *__restrict__ col_p = Wp + p;
62         F *__restrict__ col_q = Wp + q;
63         F a = F(0), b = F(0), c = F(0);
64         for ( usize i = 0; i < R; ++i ) {
65           const F wp = col_p[i * ld_W];
66           const F wq = col_q[i * ld_W];
67           a = math::fma<F>(wp, wp, a);
68           b = math::fma<F>(wq, wq, b);
69           c = math::fma<F>(wp, wq, c);
70         }
71         // scale-free skip
72         const F dd = math::fsqrt(a) * math::fsqrt(b) * F(R);
73         if ( math::fabs(c) + dd == dd ) continue;
74         ++rotations;
75         const F theta = (b - a) / (F(2) * c);
76         const F t
77             = (theta >= F(0)) ? F(1) / (theta + math::fsqrt(F(1) + theta * theta)) : F(1) / (theta - math::fsqrt(F(1) + theta * theta));
78         const F cs = F(1) / math::fsqrt(F(1) + t * t);
79         const F sn = t * cs;
80         for ( usize i = 0; i < R; ++i ) {
81           F *__restrict__ wp_p = col_p + i * ld_W;
82           F *__restrict__ wp_q = col_q + i * ld_W;
83           const F wp = *wp_p;
84           const F wq = *wp_q;
85           *wp_p = cs * wp - sn * wq;
86           *wp_q = sn * wp + cs * wq;
87         }
88         F *__restrict__ vcol_p = Vp + p;
89         F *__restrict__ vcol_q = Vp + q;
90         for ( usize i = 0; i < C; ++i ) {
91           F *__restrict__ vp_p = vcol_p + i * ld_V;
92           F *__restrict__ vp_q = vcol_q + i * ld_V;
93           const F vp = *vp_p;
94           const F vq = *vp_q;
95           *vp_p = cs * vp - sn * vq;
96           *vp_q = sn * vp + cs * vq;
97         }
98       }
99     }
100     if ( rotations == 0 ) {
101       r.converged = true;
102       break;
103     }
104   }
105 
106   F *__restrict__ Up = r.U.data();
107   const usize ld_U = r.U.ld;
108   for ( usize j = 0; j < C; ++j ) {
109     const F *__restrict__ wcol = Wp + j;
110     F s = F(0);
111     for ( usize i = 0; i < R; ++i ) {
112       const F wij = wcol[i * ld_W];
113       s = math::fma<F>(wij, wij, s);
114     }
115     s = math::fsqrt(s);
116     r.S[j] = s;
117     F *__restrict__ ucol = Up + j;
118     if ( s > F(0) ) {
119       const F inv = F(1) / s;
120       for ( usize i = 0; i < R; ++i ) ucol[i * ld_U] = wcol[i * ld_W] * inv;
121     } else {
122       for ( usize i = 0; i < R; ++i ) ucol[i * ld_U] = F(0);
123     }
124   }
125 
126   for ( usize i = 0; i + 1 < C; ++i ) {
127     usize maxj = i;
128     for ( usize j = i + 1; j < C; ++j )
129       if ( r.S[j] > r.S[maxj] ) maxj = j;
130     if ( maxj != i ) {
131       F tmp = r.S[i];
132       r.S[i] = r.S[maxj];
133       r.S[maxj] = tmp;
134       F *__restrict__ ucol_i = Up + i;
135       F *__restrict__ ucol_j = Up + maxj;
136       for ( usize rr = 0; rr < R; ++rr ) {
137         F *p_i = ucol_i + rr * ld_U;
138         F *p_j = ucol_j + rr * ld_U;
139         F t2 = *p_i;
140         *p_i = *p_j;
141         *p_j = t2;
142       }
143       F *__restrict__ vcol_i = Vp + i;
144       F *__restrict__ vcol_j = Vp + maxj;
145       for ( usize rr = 0; rr < C; ++rr ) {
146         F *p_i = vcol_i + rr * ld_V;
147         F *p_j = vcol_j + rr * ld_V;
148         F t2 = *p_i;
149         *p_i = *p_j;
150         *p_j = t2;
151       }
152     }
153   }
154   return r;
155 }

157 template<ieee754_floating F, usize R_, usize C_>
158   requires(R_ >= C_)
159 [[nodiscard]] inline mat<F, C_, R_>
160 pinv(const mat<F, R_, C_> &A, F tol = F(0)) noexcept
161 {
162   auto s = decomp::svd<F, R_, C_>(A);
163   // determine effective tolerance
164   F smax = F(0);
165   constexpr usize K = (R_ < C_) ? R_ : C_;
166   for ( usize i = 0; i < K; ++i )
167     if ( s.S.data[i] > smax ) smax = s.S.data[i];
168   F use_tol = (tol > F(0)) ? tol : F((R_ > C_) ? R_ : C_) * smax * default_eps<F>();
169 
170   // inverse singular values
171   F sinv[K];
172   for ( usize i = 0; i < K; ++i ) sinv[i] = (s.S.data[i] > use_tol) ? F(1) / s.S.data[i] : F(0);
173 
174   mat<F, C_, R_> P{};
175   for ( usize i = 0; i < C_; ++i ) {
176     for ( usize j = 0; j < R_; ++j ) {
177       F acc = F(0);
178       for ( usize k = 0; k < K; ++k ) acc = math::fma<F>(s.V.data[i * C_ + k] * sinv[k], s.U.data[j * R_ + k], acc);
179       P.data[i * R_ + j] = acc;
180     }
181   }
182   return P;
183 }

185 template<ieee754_floating F>
186 [[nodiscard]] inline dynmat<F>
187 pinv(const dynmat<F> &A, F tol = F(0)) noexcept
188 {
189   if ( A.rows < A.cols ) {
190     dynmat<F> At(A.cols, A.rows);
191     for ( usize i = 0; i < A.rows; ++i )
192       for ( usize j = 0; j < A.cols; ++j ) At.at(j, i) = A.at(i, j);
193     auto Pt = pinv<F>(At, tol);
194     dynmat<F> P(A.cols, A.rows);
195     for ( usize i = 0; i < Pt.rows; ++i )
196       for ( usize j = 0; j < Pt.cols; ++j ) P.at(j, i) = Pt.at(i, j);
197     return P;
198   }
199   auto s = svd<F>(A);
200   const usize R = A.rows;
201   const usize C = A.cols;
202   const usize K = (R < C) ? R : C;
203   F smax = F(0);
204   for ( usize i = 0; i < K; ++i )
205     if ( s.S[i] > smax ) smax = s.S[i];
206   F use_tol = (tol > F(0)) ? tol : F((R > C) ? R : C) * smax * default_eps<F>();
207 
208   micron::vector<F, micron::allocator_serial<>, false> sinv(K);
209   F *__restrict__ sinv_p = sinv.data();
210   for ( usize i = 0; i < K; ++i ) sinv_p[i] = (s.S[i] > use_tol) ? F(1) / s.S[i] : F(0);
211 
212   dynmat<F> P(C, R, F(0));
213   F *__restrict__ Pp = P.data();
214   const usize ld_P = P.ld;
215   const F *__restrict__ Vd = s.V.data();
216   const F *__restrict__ Ud = s.U.data();
217   const usize ld_V = s.V.ld;
218   const usize ld_U = s.U.ld;
219   for ( usize i = 0; i < C; ++i ) {
220     const F *__restrict__ vrow = Vd + i * ld_V;
221     F *__restrict__ prow = Pp + i * ld_P;
222     for ( usize j = 0; j < R; ++j ) {
223       const F *__restrict__ urow = Ud + j * ld_U;
224       F acc = F(0);
225       for ( usize k = 0; k < K; ++k ) acc = math::fma<F>(vrow[k] * sinv_p[k], urow[k], acc);
226       prow[j] = acc;
227     }
228   }
229   return P;
230 }

233   requires(R_ >= C_)
234 [[nodiscard]] inline usize
235 rank(const mat<F, R_, C_> &A, F tol = F(0)) noexcept
236 {
237   auto s = decomp::svd<F, R_, C_>(A);
238   constexpr usize K = (R_ < C_) ? R_ : C_;
239   F smax = F(0);
240   for ( usize i = 0; i < K; ++i )
241     if ( s.S.data[i] > smax ) smax = s.S.data[i];
242   F use_tol = (tol > F(0)) ? tol : F((R_ > C_) ? R_ : C_) * smax * default_eps<F>();
243   usize rk = 0;
244   for ( usize i = 0; i < K; ++i )
245     if ( s.S.data[i] > use_tol ) ++rk;
246   return rk;
247 }

249 template<ieee754_floating F>
250 [[nodiscard]] inline usize
251 rank(const dynmat<F> &A, F tol = F(0)) noexcept
252 {
253   if ( A.rows < A.cols ) {
254     dynmat<F> At(A.cols, A.rows);
255     for ( usize i = 0; i < A.rows; ++i )
256       for ( usize j = 0; j < A.cols; ++j ) At.at(j, i) = A.at(i, j);
257     return rank<F>(At, tol);
258   }
259   auto s = svd<F>(A);
260   const usize K = (A.rows < A.cols) ? A.rows : A.cols;
261   F smax = F(0);
262   for ( usize i = 0; i < K; ++i )
263     if ( s.S[i] > smax ) smax = s.S[i];
264   F use_tol = (tol > F(0)) ? tol : F((A.rows > A.cols) ? A.rows : A.cols) * smax * default_eps<F>();
265   usize rk = 0;
266   for ( usize i = 0; i < K; ++i )
267     if ( s.S[i] > use_tol ) ++rk;
268   return rk;
269 }

271 template<ieee754_floating F>
272 [[nodiscard]] inline dynmat<F>
273 orth(const dynmat<F> &A, F tol = F(0)) noexcept
274 {
275   if ( A.rows < A.cols ) {
276     dynmat<F> At(A.cols, A.rows);
277     for ( usize i = 0; i < A.rows; ++i )
278       for ( usize j = 0; j < A.cols; ++j ) At.at(j, i) = A.at(i, j);
279     auto s = svd<F>(At);
280     F smax = F(0);
281     for ( usize i = 0; i < s.S.size(); ++i )
282       if ( s.S[i] > smax ) smax = s.S[i];
283     F use_tol = (tol > F(0)) ? tol : F((A.rows > A.cols) ? A.rows : A.cols) * smax * default_eps<F>();
284     usize rk = 0;
285     for ( usize i = 0; i < s.S.size(); ++i )
286       if ( s.S[i] > use_tol ) ++rk;
287     dynmat<F> O(A.rows, rk);
288     for ( usize i = 0; i < A.rows; ++i )
289       for ( usize j = 0; j < rk; ++j ) O.at(i, j) = s.V.at(i, j);
290     return O;
291   }
292   auto s = svd<F>(A);
293   const usize K = (A.rows < A.cols) ? A.rows : A.cols;
294   F smax = F(0);
295   for ( usize i = 0; i < K; ++i )
296     if ( s.S[i] > smax ) smax = s.S[i];
297   F use_tol = (tol > F(0)) ? tol : F((A.rows > A.cols) ? A.rows : A.cols) * smax * default_eps<F>();
298   usize rk = 0;
299   for ( usize i = 0; i < K; ++i )
300     if ( s.S[i] > use_tol ) ++rk;
301   dynmat<F> O(A.rows, rk);
302   for ( usize i = 0; i < A.rows; ++i )
303     for ( usize j = 0; j < rk; ++j ) O.at(i, j) = s.U.at(i, j);
304   return O;
305 }

307 template<ieee754_floating F>
308 [[nodiscard]] inline dynmat<F>
309 null(const dynmat<F> &A, F tol = F(0)) noexcept
310 {
311   if ( A.rows < A.cols ) {
312     dynmat<F> Ap(A.cols, A.cols, F(0));
313     for ( usize i = 0; i < A.rows; ++i )
314       for ( usize j = 0; j < A.cols; ++j ) Ap.at(i, j) = A.at(i, j);
315     return null<F>(Ap, tol);
316   }
317   auto s = svd<F>(A);
318   const usize K = (A.rows < A.cols) ? A.rows : A.cols;
319   F smax = F(0);
320   for ( usize i = 0; i < K; ++i )
321     if ( s.S[i] > smax ) smax = s.S[i];
322   F use_tol = (tol > F(0)) ? tol : F((A.rows > A.cols) ? A.rows : A.cols) * smax * default_eps<F>();
323   usize rk = 0;
324   for ( usize i = 0; i < K; ++i )
325     if ( s.S[i] > use_tol ) ++rk;
326   const usize ns = A.cols - rk;
327   dynmat<F> N(A.cols, ns);
328   for ( usize i = 0; i < A.cols; ++i )
329     for ( usize j = 0; j < ns; ++j ) N.at(i, j) = s.V.at(i, rk + j);
330   return N;
331 }
