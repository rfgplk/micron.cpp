32 // Thomas algorithm
33 //
34 //   a : sub-diagonal,   length n-1   (a[i] = A[i+1, i])
35 //   b : main diagonal,  length n
36 //   c : super-diagonal, length n-1   (c[i] = A[i, i+1])
37 //   d : right-hand side,length n
38 //
39 // NOTE: A must be diagonally dominant
40 template<ieee754_floating F>
41 [[gnu::flatten]] inline void
42 tridiag_solve(F *__restrict__ a, F *__restrict__ b, F *__restrict__ c, F *__restrict__ d, usize n) noexcept
43 {
44   if ( n == 0 ) return;
45   // pivot guard
46   constexpr F tiny = default_eps<F>() * F(4);
47   if ( n == 1 ) {
48     d[0] = (math::fabs(b[0]) > tiny) ? d[0] / b[0] : F(0);
49     return;
50   }
51 
52   if ( math::fabs(b[0]) > tiny ) {
53     const F inv0 = F(1) / b[0];
54     c[0] = c[0] * inv0;
55     d[0] = d[0] * inv0;
56   } else {
57     c[0] = F(0);
58     d[0] = F(0);
59   }
60   for ( usize i = 1; i < n; ++i ) {
61     const F m = b[i] - a[i - 1] * c[i - 1];
62     if ( math::fabs(m) > tiny ) {
63       const F inv_m = F(1) / m;
64       if ( i + 1 < n ) c[i] = c[i] * inv_m;
65       d[i] = (d[i] - a[i - 1] * d[i - 1]) * inv_m;
66     } else {
67       if ( i + 1 < n ) c[i] = F(0);
68       d[i] = F(0);
69     }
70   }
71 
72   for ( usize i = n - 1; i-- > 0; ) d[i] = d[i] - c[i] * d[i + 1];
73 }

75 // LDL^T factorisation of a 5-banded SPD A
76 //
77 //   d2  : offset-2 super-diagonal,  length n-2  (d2[i] = A[i, i+2])
78 //   d1  : offset-1 super-diagonal,  length n-1  (d1[i] = A[i, i+1])
79 //   diag: main diagonal,            length n
80 template<ieee754_floating F>
81 [[nodiscard]] [[gnu::flatten]] inline bool
82 pent_spd_factor(F *__restrict__ d2, F *__restrict__ d1, F *__restrict__ diag, usize n) noexcept
83 {
84   if ( n == 0 ) return true;
85   if ( n == 1 ) return diag[0] > F(0);
86   for ( usize i = 0; i < n; ++i ) {
87     F l2_i = F(0);
88     F l1_i = F(0);
89     if ( i >= 2 ) {
90       l2_i = d2[i - 2] / diag[i - 2];
91       d2[i - 2] = l2_i;
92     }
93     if ( i >= 1 ) {
94       F num = d1[i - 1];
95       if ( i >= 2 ) num -= l2_i * diag[i - 2] * d1[i - 2];
96       l1_i = num / diag[i - 1];
97       d1[i - 1] = l1_i;
98     }
99     F D_i = diag[i];
100     if ( i >= 1 ) D_i -= l1_i * l1_i * diag[i - 1];
101     if ( i >= 2 ) D_i -= l2_i * l2_i * diag[i - 2];
102     if ( D_i <= F(0) ) return false;
103     diag[i] = D_i;
104   }
105   return true;
106 }

108 // triangular given an LDL^T factor
109 //
110 //   l2  : L offset-2 (length n-2)
111 //   l1  : L offset-1 (length n-1)
112 //   D   : diagonal of D (length n)
113 //   rhs : right-hand side (length n)
114 template<ieee754_floating F>
115 [[gnu::flatten]] inline void
116 pent_spd_solve_factored(const F *__restrict__ l2, const F *__restrict__ l1, const F *__restrict__ D, F *__restrict__ rhs, usize n) noexcept
117 {
118   if ( n == 0 ) return;
119   if ( n == 1 ) {
120     rhs[0] = rhs[0] / D[0];
121     return;
122   }
123   // L y = b
124   for ( usize i = 1; i < n; ++i ) {
125     F y = rhs[i] - l1[i - 1] * rhs[i - 1];
126     if ( i >= 2 ) y -= l2[i - 2] * rhs[i - 2];
127     rhs[i] = y;
128   }
129   for ( usize i = 0; i < n; ++i ) rhs[i] = rhs[i] / D[i];
130   // L^T x = z
131   for ( usize i = n - 1; i-- > 0; ) {
132     F x = rhs[i] - l1[i] * rhs[i + 1];
133     if ( i + 2 < n ) x -= l2[i] * rhs[i + 2];
134     rhs[i] = x;
135   }
136 }

138 template<ieee754_floating F>
139 [[nodiscard]] [[gnu::flatten]] inline bool
140 pent_spd_solve(F *__restrict__ d2, F *__restrict__ d1, F *__restrict__ diag, F *__restrict__ rhs, usize n) noexcept
141 {
142   if ( !pent_spd_factor<F>(d2, d1, diag, n) ) return false;
143   pent_spd_solve_factored<F>(d2, d1, diag, rhs, n);
144   return true;
145 }

147 // selective inverse of A given its LDL^T factor
148 template<ieee754_floating F>
149 [[gnu::flatten]] inline void
150 pent_spd_takahashi(const F *__restrict__ l2, const F *__restrict__ l1, const F *__restrict__ D, usize n, F *__restrict__ z_diag,
151                    F *__restrict__ z_d1, F *__restrict__ z_d2) noexcept
152 {
153   if ( n == 0 ) return;
154   z_diag[n - 1] = F(1) / D[n - 1];
155   if ( n == 1 ) return;
156 
157   // i = n-2
158   z_d1[n - 2] = -l1[n - 2] * z_diag[n - 1];
159   z_diag[n - 2] = F(1) / D[n - 2] - l1[n - 2] * z_d1[n - 2];
160 
161   // i = n-3 down to 0
162   for ( usize ii = n - 2; ii-- > 0; ) {
163     const usize i = ii;
164     // Z[i, i+2] = -l1[i] * Z[i+1, i+2] - l2[i] * Z[i+2, i+2]
165     z_d2[i] = -l1[i] * z_d1[i + 1] - l2[i] * z_diag[i + 2];
166     // Z[i, i+1] = -l1[i] * Z[i+1, i+1] - l2[i] * Z[i+1, i+2]
167     z_d1[i] = -l1[i] * z_diag[i + 1] - l2[i] * z_d1[i + 1];
168     // Z[i, i] = 1/D[i] - l1[i] * Z[i, i+1] - l2[i] * Z[i, i+2]
169     z_diag[i] = F(1) / D[i] - l1[i] * z_d1[i] - l2[i] * z_d2[i];
170   }
171 }

173 template<ieee754_floating F>
174 [[nodiscard]] inline bool
175 gband_lu_solve(F *__restrict__ AB, usize kl, usize ku, usize n, F *__restrict__ rhs) noexcept
176 {
177   const usize nb = kl + ku + 1;      // number of bands
178   auto at = [&](usize band, usize col) -> F & { return AB[band * n + col]; };
179   auto at_const = [&](usize band, usize col) -> const F & { return AB[band * n + col]; };
180 
181   for ( usize k = 0; k + 1 < n; ++k ) {
182     const F piv = at(ku, k);
183     if ( piv == F(0) ) return false;
184     const F inv_piv = F(1) / piv;
185     const usize r_end = (k + kl < n - 1) ? (k + kl) : (n - 1);
186 
187     for ( usize r = k + 1; r <= r_end; ++r ) {
188       const usize band_rk = (r + ku) - k;
189       const F m = at(band_rk, k) * inv_piv;
190       at(band_rk, k) = m;
191       const usize c_end = (k + ku < n - 1) ? (k + ku) : (n - 1);
192       for ( usize c = k + 1; c <= c_end; ++c ) {
193         if ( c > r + ku ) continue;
194         const usize band_rc = (r + ku) - c;
195         const usize band_kc = (k + ku) - c;
196         if ( band_rc >= nb ) continue;
197         if ( c > k + ku ) continue;
198         at(band_rc, c) -= m * at_const(band_kc, c);
199       }
200     }
201   }
202   if ( at(ku, n - 1) == F(0) ) return false;
203 
204   for ( usize r = 0; r < n; ++r ) {
205     F s = rhs[r];
206     const usize j_lo = (r > kl) ? (r - kl) : 0;
207     for ( usize j = j_lo; j < r; ++j ) {
208       const usize band = (r + ku) - j;
209       s -= at_const(band, j) * rhs[j];
210     }
211     rhs[r] = s;
212   }
213 
214   for ( usize ri = n; ri-- > 0; ) {
215     const usize r = ri;
216     F s = rhs[r];
217     const usize j_hi = (r + ku < n - 1) ? (r + ku) : (n - 1);
218     for ( usize j = r + 1; j <= j_hi; ++j ) {
219       if ( j > r + ku ) continue;
220       const usize band = (r + ku) - j;
221       s -= at_const(band, j) * rhs[j];
222     }
223     rhs[r] = s / at_const(ku, r);
224   }
225   return true;
226 }

233 template<ieee754_floating F>
234 [[nodiscard]] inline bandwidth_t<F>
235 bandwidth(const dynmat<F> &A, F tol = F(0)) noexcept
236 {
237   bandwidth_t<F> b{ 0, 0 };
238   const usize N = (A.rows < A.cols) ? A.rows : A.cols;
239   for ( usize i = 0; i < A.rows; ++i ) {
240     for ( usize j = 0; j < A.cols; ++j ) {
241       F v = A.at(i, j);
242       F a = (v < F(0)) ? -v : v;
243       if ( a <= tol ) continue;
244       if ( i > j ) {
245         const usize d = i - j;
246         if ( d > b.kl ) b.kl = d;
247       } else if ( j > i ) {
248         const usize d = j - i;
249         if ( d > b.ku ) b.ku = d;
250       }
251     }
252   }
253   (void)N;
254   return b;
255 }

257 template<ieee754_floating F, usize R, usize C>
258 [[nodiscard]] inline bandwidth_t<F>
259 bandwidth(const mat<F, R, C> &A, F tol = F(0)) noexcept
260 {
261   bandwidth_t<F> b{ 0, 0 };
262   for ( usize i = 0; i < R; ++i ) {
263     for ( usize j = 0; j < C; ++j ) {
264       F v = A.data[i * C + j];
265       F a = (v < F(0)) ? -v : v;
266       if ( a <= tol ) continue;
267       if ( i > j ) {
268         const usize d = i - j;
269         if ( d > b.kl ) b.kl = d;
270       } else if ( j > i ) {
271         const usize d = j - i;
272         if ( d > b.ku ) b.ku = d;
273       }
274     }
275   }
276   return b;
277 }

285 template<ieee754_floating F>
286 [[nodiscard]] inline chol_banded_result_dyn<F>
287 chol_banded(const dynmat<F> &A, usize kl) noexcept
288 {
289   const usize N = A.rows;
290   chol_banded_result_dyn<F> r{ dynmat<F>::zero(N, N), kl, true };
291   F *__restrict__ L = r.L.data();
292   const usize ld = r.L.ld;
293   for ( usize i = 0; i < N; ++i ) {
294     const usize j_lo = (i > kl) ? (i - kl) : 0;
295     for ( usize j = j_lo; j <= i; ++j ) {
296       F s = A.at(i, j);
297       const usize k_lo = (j > kl) ? (j - kl) : 0;      // L(i, k) and L(j, k) both nonzero only for k >= max(i-kl, j-kl)
298       const usize k_lo_eff = (k_lo > j_lo) ? k_lo : j_lo;
299       for ( usize k = k_lo_eff; k < j; ++k ) s = math::fma<F>(-L[i * ld + k], L[j * ld + k], s);
300       if ( i == j ) {
301         if ( s <= F(0) ) {
302           r.spd = false;
303           return r;
304         }
305         L[i * ld + j] = math::fsqrt(s);
306       } else {
307         L[i * ld + j] = s / L[j * ld + j];
308       }
309     }
310   }
311   return r;
312 }
