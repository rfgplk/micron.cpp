35 template<arith_scalar T, usize N>
36 [[nodiscard, gnu::always_inline]] inline constexpr T
37 trace(const mat<T, N, N> &A) noexcept
38 {
39   T s{ 0 };
40   for ( usize i = 0; i < N; ++i ) s = s + A.data[i * N + i];
41   return s;
42 }

45 [[nodiscard]] inline F
46 det(const mat<F, N, N> &A) noexcept
47 {
48   // route through the pivoted LU

52 template<ieee754_floating F, usize N>
53 [[nodiscard]] inline vec<F, N>
54 solve(const mat<F, N, N> &A, const vec<F, N> &b) noexcept
55 {
56   // partial-pivoting LU
57   auto lu = decomp::lu_pivot<F, N>(A);
58   vec<F, N> x = b;
59   if ( lu.singular ) {
60     for ( usize i = 0; i < N; ++i ) x.data[i] = ieee::qnan_v<F>();
61     return x;
62   }
63   decomp::lu_solve<F, N>(lu, x);
64   return x;
65 }

67 template<ieee754_floating F, usize N>
68 [[nodiscard]] inline mat<F, N, N>
69 inv(const mat<F, N, N> &A) noexcept
70 {
71   // pivoted inverse
72   auto r = decomp::inv<F, N>(A);
73   if ( r.singular ) {
74     mat<F, N, N> X = mat<F, N, N>::zero();
75     for ( usize i = 0; i < N * N; ++i ) X.data[i] = ieee::qnan_v<F>();
76     return X;
77   }
78   return r.X;
79 }

81 template<arith_scalar T, usize R1, usize C1, usize R2, usize C2>
82 [[nodiscard]] inline constexpr mat<T, R1 * R2, C1 * C2>
83 kron(const mat<T, R1, C1> &A, const mat<T, R2, C2> &B) noexcept
84 {
85   mat<T, R1 * R2, C1 * C2> K = mat<T, R1 * R2, C1 * C2>::zero();
86   for ( usize i = 0; i < R1; ++i ) {
87     for ( usize j = 0; j < C1; ++j ) {
88       const T a = A.data[i * C1 + j];
89       for ( usize p = 0; p < R2; ++p ) {
90         for ( usize q = 0; q < C2; ++q ) {
91           K.data[(i * R2 + p) * (C1 * C2) + (j * C2 + q)] = a * B.data[p * C2 + q];
92         }
93       }
94     }
95   }
96   return K;
97 }

101 template<ieee754_floating F, usize R_, usize C_>
102 [[nodiscard, gnu::flatten]] inline constexpr F
103 frobenius_norm(const mat<F, R_, C_> &A) noexcept
104 {
105   F s = F(0);
106   for ( usize i = 0; i < R_ * C_; ++i ) s = math::fma<F>(A.data[i], A.data[i], s);
107   return mk::pow_ns::sqrt<F>(s);
108 }

110 template<ieee754_floating F, usize R_, usize C_>
111 [[nodiscard, gnu::flatten]] inline constexpr F
112 norm_inf_mat(const mat<F, R_, C_> &A) noexcept
113 {
114   // ||A||_∞ = max row sum of |A_ij|
115   F m = F(0);
116   for ( usize i = 0; i < R_; ++i ) {
117     F s = F(0);
118     for ( usize j = 0; j < C_; ++j ) s = s + mk::manip::fabs<F>(A.data[i * C_ + j]);
119     if ( s > m ) m = s;
120   }
121   return m;
122 }

125 [[nodiscard, gnu::flatten]] inline constexpr F
126 norm_l1_mat(const mat<F, R_, C_> &A) noexcept
127 {
128   F m = F(0);
129   for ( usize j = 0; j < C_; ++j ) {
130     F s = F(0);
131     for ( usize i = 0; i < R_; ++i ) s = s + mk::manip::fabs<F>(A.data[i * C_ + j]);
132     if ( s > m ) m = s;
133   }
134   return m;

137 template<ieee754_floating F, usize R_, usize C_>
138 [[nodiscard, gnu::flatten]] inline constexpr F
139 norm_p_mat(const mat<F, R_, C_> &A, F p) noexcept
140 {
141   F s = F(0);
142   for ( usize i = 0; i < R_ * C_; ++i ) {
143     s = s + mk::pow_ns::pow<F>(mk::manip::fabs<F>(A.data[i]), p);
144   }
145   return mk::pow_ns::pow<F>(s, F(1) / p);
146 }

148 template<ieee754_floating F, usize N>
149 [[nodiscard]] inline F
150 condition_number(const mat<F, N, N> &A) noexcept
151 {
152   const F na = frobenius_norm(A);
153   const auto Ai = decomp::inv<F, N>(A);      // pivoted
154   if ( Ai.singular ) return ieee::inf_v<F>(0);
155   return na * frobenius_norm(Ai.X);
156 }

158 template<ieee754_floating F, usize R_, usize C_>
159   requires(R_ >= C_)
160 [[nodiscard]] inline vec<F, C_>
161 lstsq(const mat<F, R_, C_> &A, const vec<F, R_> &b) noexcept
162 {
163   auto qr_d = decomp::qr<F, R_, C_>(A);
164   vec<F, C_> y{};
165   for ( usize j = 0; j < C_; ++j ) {
166     F s = F(0);
167     for ( usize i = 0; i < R_; ++i ) s = math::fma<F>(qr_d.Q.data[i * C_ + j], b.data[i], s);
168     y.data[j] = s;
169   }
170   vec<F, C_> x{};
171   for ( usize i = C_; i-- > 0; ) {
172     F s = y.data[i];
173     for ( usize k = i + 1; k < C_; ++k ) s = s - qr_d.R.data[i * C_ + k] * x.data[k];
174     x.data[i] = s / qr_d.R.data[i * C_ + i];
175   }
176   return x;
177 }

179 template<ieee754_floating F, usize R_, usize C_>
180   requires(R_ >= C_)
181 [[nodiscard]] inline mat<F, C_, R_>
182 pinv(const mat<F, R_, C_> &A) noexcept
183 {
184   auto qr_d = decomp::qr<F, R_, C_>(A);
185   mat<F, C_, C_> Rinv = mat<F, C_, C_>::zero();
186   for ( usize j = 0; j < C_; ++j ) {
187     for ( usize i = C_; i-- > 0; ) {
188       F s = (i == j) ? F(1) : F(0);
189       for ( usize k = i + 1; k < C_; ++k ) s = s - qr_d.R.data[i * C_ + k] * Rinv.data[k * C_ + j];
190       Rinv.data[i * C_ + j] = s / qr_d.R.data[i * C_ + i];
191     }
192   }
193   mat<F, C_, R_> P = mat<F, C_, R_>::zero();
194   for ( usize i = 0; i < C_; ++i ) {
195     for ( usize j = 0; j < R_; ++j ) {
196       F s = F(0);
197       for ( usize k = 0; k < C_; ++k ) s = math::fma<F>(Rinv.data[i * C_ + k], qr_d.Q.data[j * C_ + k], s);
198       P.data[i * R_ + j] = s;
199     }
200   }
201   return P;
202 }
