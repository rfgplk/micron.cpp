42 template<ieee754_floating F, usize N>
43 [[nodiscard]] inline lu_result<F, N>
44 lu(const mat<F, N, N> &A) noexcept
45 {
46   lu_result<F, N> r{ mat<F, N, N>::zero(), mat<F, N, N>::zero(), false };
47   // L diagonal = 1
48   for ( usize i = 0; i < N; ++i ) r.L.data[i * N + i] = F(1);
49 
50   for ( usize i = 0; i < N; ++i ) {
51     // U row i
52     for ( usize j = i; j < N; ++j ) {
53       F s = A.data[i * N + j];
54       for ( usize k = 0; k < i; ++k ) s -= r.L.data[i * N + k] * r.U.data[k * N + j];
55       r.U.data[i * N + j] = s;
56     }
57     F piv = r.U.data[i * N + i];
58     if ( piv == F(0) ) {
59       r.singular = true;
60       return r;
61     }
62     F inv = F(1) / piv;
63     // L column i
64     for ( usize j = i + 1; j < N; ++j ) {
65       F s = A.data[j * N + i];
66       for ( usize k = 0; k < i; ++k ) s -= r.L.data[j * N + k] * r.U.data[k * N + i];
67       r.L.data[j * N + i] = s * inv;
68     }
69   }
70   return r;
71 }

73 // Cholesky decomposition.
74 // SPD-only: a non-positive pivot (s <= 0) sets spd=false and returns EARLY, so on
75 // failure r.L is only partially filled (entries past the failing pivot stay zero from

77 // rejected (use chol_pivot for a rank-revealing pivoted factorisation).

83 template<ieee754_floating F, usize N>
84 [[nodiscard]] inline chol_result<F, N>
85 chol(const mat<F, N, N> &A) noexcept
86 {
87   chol_result<F, N> r{ mat<F, N, N>::zero(), true };
88   for ( usize i = 0; i < N; ++i ) {
89     for ( usize j = 0; j <= i; ++j ) {
90       F s = A.data[i * N + j];
91       for ( usize k = 0; k < j; ++k ) s -= r.L.data[i * N + k] * r.L.data[j * N + k];
92       if ( i == j ) {
93         if ( s <= F(0) ) {
94           r.spd = false;
95           return r;
96         }
97         r.L.data[i * N + j] = math::fsqrt(s);
98       } else {
99         r.L.data[i * N + j] = s / r.L.data[j * N + j];
100       }
101     }
102   }
103   return r;
104 }

112 template<ieee754_floating F, usize R_, usize C_>
113 [[nodiscard]] inline qr_result<F, R_, C_>
114 qr(const mat<F, R_, C_> &A) noexcept
115 {
116   qr_result<F, R_, C_> r{ mat<F, R_, C_>::zero(), mat<F, C_, C_>::zero() };
117   vec<F, R_> q[C_];
118   for ( usize j = 0; j < C_; ++j )
119     for ( usize i = 0; i < R_; ++i ) q[j].data[i] = A.data[i * C_ + j];
120 
121   for ( usize j = 0; j < C_; ++j ) {
122     F nj = math::fsqrt(ops::dot(q[j], q[j]));
123     r.R.data[j * C_ + j] = nj;
124     if ( nj > F(0) ) {
125       F inv = F(1) / nj;
126       for ( usize i = 0; i < R_; ++i ) q[j].data[i] *= inv;
127     }
128     for ( usize k = j + 1; k < C_; ++k ) {
129       F p = ops::dot(q[j], q[k]);
130       r.R.data[j * C_ + k] = p;
131       for ( usize i = 0; i < R_; ++i ) q[k].data[i] -= p * q[j].data[i];
132     }
133   }
134   // pack q[] into Q
135   for ( usize j = 0; j < C_; ++j )
136     for ( usize i = 0; i < R_; ++i ) r.Q.data[i * C_ + j] = q[j].data[i];
137   return r;
138 }

150 template<ieee754_floating F>
151 [[nodiscard]] inline eigen_result2<F>
152 eigen_sym2(const mat<F, 2, 2> &A) noexcept
153 {
154   F a = A.data[0], b = A.data[1], c = A.data[3];
155   F tr = a + c;
156   F det = a * c - b * b;
157   // discriminant can dip slightly negative through cancellation; clamp before sqrt.
158   // (under -Ofast the in-fsqrt NaN guard is folded away, so this explicit clamp is mandatory.)
159   F arg = tr * tr * F(0.25) - det;
160   if ( arg < F(0) ) arg = F(0);
161   F disc = math::fsqrt(arg);
162   F l1 = tr * F(0.5) + disc;
163   F l2 = tr * F(0.5) - disc;
164 
165   eigen_result2<F> r{};
166   r.values.data[0] = l1;
167   r.values.data[1] = l2;
168 
169   // eigenvectors
170   if ( b != F(0) ) {
171     F v1y = b;
172     F v1x = l1 - c;
173     F n1 = math::fsqrt(v1x * v1x + v1y * v1y);
174     r.vectors.data[0] = v1x / n1;
175     r.vectors.data[2] = v1y / n1;
176     F v2y = b;
177     F v2x = l2 - c;
178     F n2 = math::fsqrt(v2x * v2x + v2y * v2y);
179     r.vectors.data[1] = v2x / n2;
180     r.vectors.data[3] = v2y / n2;
181   } else {
182     // already diagonal
183     r.vectors.data[0] = F(1);
184     r.vectors.data[1] = F(0);
185     r.vectors.data[2] = F(0);
186     r.vectors.data[3] = F(1);
187   }
188   return r;
189 }

191 template<ieee754_floating F>
192 [[nodiscard]] inline eigen_result3<F>
193 eigen_sym3(const mat<F, 3, 3> &A) noexcept
194 {
195   mat<F, 3, 3> M = A;
196   mat<F, 3, 3> V = mat<F, 3, 3>::identity();
197   constexpr int max_sweeps = 50;
198 
199   for ( int sweep = 0; sweep < max_sweeps; ++sweep ) {
200     int rotations = 0;
201 
202     static constexpr int pq[3][2] = { { 0, 1 }, { 0, 2 }, { 1, 2 } };
203     for ( int k = 0; k < 3; ++k ) {
204       int p = pq[k][0], q = pq[k][1];
205       F app = M.data[p * 3 + p];
206       F aqq = M.data[q * 3 + q];
207       F apq = M.data[p * 3 + q];
208       // scale-free skip, the tqli idiom: nothing left to annihilate at this precision
209       const F dd = math::fabs(app) + math::fabs(aqq);
210       if ( math::fabs(apq) + dd == dd ) continue;
211       ++rotations;
212       F theta = (aqq - app) / (F(2) * apq);
213       F t;
214       if ( theta >= F(0) )
215         t = F(1) / (theta + math::fsqrt(F(1) + theta * theta));
216       else
217         t = F(1) / (theta - math::fsqrt(F(1) + theta * theta));
218       F c = F(1) / math::fsqrt(F(1) + t * t);
219       F s = t * c;
220 
221       F new_app = app - t * apq;
222       F new_aqq = aqq + t * apq;
223       M.data[p * 3 + p] = new_app;
224       M.data[q * 3 + q] = new_aqq;
225       M.data[p * 3 + q] = F(0);
226       M.data[q * 3 + p] = F(0);
227 
228       // update remaining off-diagonals
229       for ( int r = 0; r < 3; ++r ) {
230         if ( r == p || r == q ) continue;
231         F arp = M.data[r * 3 + p];
232         F arq = M.data[r * 3 + q];
233         F new_arp = c * arp - s * arq;
234         F new_arq = s * arp + c * arq;
235         M.data[r * 3 + p] = new_arp;
236         M.data[p * 3 + r] = new_arp;
237         M.data[r * 3 + q] = new_arq;
238         M.data[q * 3 + r] = new_arq;
239       }
240       // update eigenvector matrix V
241       for ( int r = 0; r < 3; ++r ) {
242         F vrp = V.data[r * 3 + p];
243         F vrq = V.data[r * 3 + q];
244         V.data[r * 3 + p] = c * vrp - s * vrq;
245         V.data[r * 3 + q] = s * vrp + c * vrq;
246       }
247     }
248 
249     if ( rotations == 0 ) break;
250   }
251 
252   eigen_result3<F> res{};
253   res.values.data[0] = M.data[0];
254   res.values.data[1] = M.data[4];
255   res.values.data[2] = M.data[8];
256   res.vectors = V;
257 
258   // sort eigenvalues ascending, carrying eigenvector columns (matches eigen_sym_qr's convention).
259   for ( usize i = 0; i + 1 < 3; ++i ) {
260     usize best = i;
261     for ( usize j = i + 1; j < 3; ++j )
262       if ( res.values.data[j] < res.values.data[best] ) best = j;
263     if ( best != i ) {
264       F tv = res.values.data[i];
265       res.values.data[i] = res.values.data[best];
266       res.values.data[best] = tv;
267       for ( usize rr = 0; rr < 3; ++rr ) {
268         F tc = res.vectors.data[rr * 3 + i];
269         res.vectors.data[rr * 3 + i] = res.vectors.data[rr * 3 + best];
270         res.vectors.data[rr * 3 + best] = tc;
271       }
272     }
273   }
274   return res;
275 }

283 template<ieee754_floating F>
284 [[nodiscard]] inline svd2_result<F>
285 svd2(const mat<F, 2, 2> &A) noexcept
286 {
287   mat<F, 2, 2> AtA{};
288   AtA.data[0] = A.data[0] * A.data[0] + A.data[2] * A.data[2];
289   AtA.data[1] = A.data[0] * A.data[1] + A.data[2] * A.data[3];
290   AtA.data[2] = AtA.data[1];
291   AtA.data[3] = A.data[1] * A.data[1] + A.data[3] * A.data[3];
292   auto e = eigen_sym2(AtA);
293 
294   svd2_result<F> r{};
295   r.V = e.vectors;
296   r.S.data[0] = math::fsqrt(e.values.data[0] < F(0) ? F(0) : e.values.data[0]);
297   r.S.data[1] = math::fsqrt(e.values.data[1] < F(0) ? F(0) : e.values.data[1]);
298 
299   vec<F, 2> v0{ r.V.data[0], r.V.data[2] };      // col 0 of V
300   vec<F, 2> v1{ r.V.data[1], r.V.data[3] };      // col 1 of V
301   vec<F, 2> u0 = ops::gemv(A, v0);
302   vec<F, 2> u1 = ops::gemv(A, v1);
303   if ( r.S.data[0] > F(0) ) {
304     F inv = F(1) / r.S.data[0];
305     u0.data[0] *= inv;
306     u0.data[1] *= inv;
307   }
308   if ( r.S.data[1] > F(0) ) {
309     F inv = F(1) / r.S.data[1];
310     u1.data[0] *= inv;
311     u1.data[1] *= inv;
312   }
313   r.U.data[0] = u0.data[0];
314   r.U.data[2] = u0.data[1];
315   r.U.data[1] = u1.data[0];
316   r.U.data[3] = u1.data[1];
317   return r;
318 }

326 template<ieee754_floating F>
327 [[nodiscard]] inline svd3_result<F>
328 svd3(const mat<F, 3, 3> &A) noexcept
329 {
330   mat<F, 3, 3> AtA{};
331   // AtA[i,j] = sum_k A[k,i] * A[k,j]
332   for ( usize i = 0; i < 3; ++i )
333     for ( usize j = 0; j < 3; ++j ) {
334       F s = F(0);
335       for ( usize k = 0; k < 3; ++k ) s += A.data[k * 3 + i] * A.data[k * 3 + j];
336       AtA.data[i * 3 + j] = s;
337     }
338   auto e = eigen_sym3(AtA);
339 
340   svd3_result<F> r{};
341   r.V = e.vectors;
342   for ( usize i = 0; i < 3; ++i ) {
343     F v = e.values.data[i];
344     r.S.data[i] = math::fsqrt(v < F(0) ? F(0) : v);
345   }
346 
347   // U columns = A * V columns scaled by 1/S
348   for ( usize j = 0; j < 3; ++j ) {
349     vec<F, 3> vj{ r.V.data[j], r.V.data[3 + j], r.V.data[6 + j] };
350     vec<F, 3> uj = ops::gemv(A, vj);
351     if ( r.S.data[j] > F(0) ) {
352       F inv = F(1) / r.S.data[j];
353       uj.data[0] *= inv;
354       uj.data[1] *= inv;
355       uj.data[2] *= inv;
356     }
357     r.U.data[j] = uj.data[0];
358     r.U.data[3 + j] = uj.data[1];
359     r.U.data[6 + j] = uj.data[2];
360   }
361 
362   // sort singular values descending, carrying U-columns, S and V-columns together
363   // (umeyama's reflection-scale and the suite convention assume σ descending; eigen_sym3
364   //  emits them in arbitrary Jacobi order).
365   for ( usize i = 0; i + 1 < 3; ++i ) {
366     usize best = i;
367     for ( usize j = i + 1; j < 3; ++j )
368       if ( r.S.data[j] > r.S.data[best] ) best = j;
369     if ( best != i ) {
370       F ts = r.S.data[i];
371       r.S.data[i] = r.S.data[best];
372       r.S.data[best] = ts;
373       for ( usize rr = 0; rr < 3; ++rr ) {
374         F tu = r.U.data[rr * 3 + i];
375         r.U.data[rr * 3 + i] = r.U.data[rr * 3 + best];
376         r.U.data[rr * 3 + best] = tu;
377         F tv = r.V.data[rr * 3 + i];
378         r.V.data[rr * 3 + i] = r.V.data[rr * 3 + best];
379         r.V.data[rr * 3 + best] = tv;
380       }
381     }
382   }
383   return r;
384 }

393 template<ieee754_floating F, usize N>
394 [[nodiscard]] inline lu_pivot_result<F, N>
395 lu_pivot(const mat<F, N, N> &A) noexcept
396 {
397   lu_pivot_result<F, N> r{ A, {}, 1, false };
398   for ( usize i = 0; i < N; ++i ) r.perm[i] = i;
399 
400   F scale[N];
401   for ( usize i = 0; i < N; ++i ) {
402     F m = F(0);
403     for ( usize j = 0; j < N; ++j ) {
404       const F v = math::fabs(r.LU.data[i * N + j]);
405       if ( v > m ) m = v;
406     }
407     scale[i] = m;
408   }
409 
410   for ( usize k = 0; k < N; ++k ) {
411     F best_a = F(0);
412     F best_s = F(1);
413     usize piv = k;
414     for ( usize i = k; i < N; ++i ) {
415       const F a = math::fabs(r.LU.data[i * N + k]);
416       const F si = scale[i];
417       if ( si == F(0) ) continue;      // zero row: skip
418       if ( a * best_s > best_a * si ) {
419         best_a = a;
420         best_s = si;
421         piv = i;
422       }
423     }
424     if ( best_a == F(0) ) {
425       r.singular = true;
426       return r;
427     }
428 
429     if ( piv != k ) {
430       for ( usize j = 0; j < N; ++j ) {
431         F t = r.LU.data[k * N + j];
432         r.LU.data[k * N + j] = r.LU.data[piv * N + j];
433         r.LU.data[piv * N + j] = t;
434       }
435       usize t = r.perm[k];
436       r.perm[k] = r.perm[piv];
437       r.perm[piv] = t;
438       F ts = scale[k];
439       scale[k] = scale[piv];
440       scale[piv] = ts;
441       r.sign = -r.sign;
442     }
443 
444     const F inv_piv = F(1) / r.LU.data[k * N + k];
445     for ( usize i = k + 1; i < N; ++i ) r.LU.data[i * N + k] *= inv_piv;
446     for ( usize i = k + 1; i < N; ++i ) {
447       const F m = r.LU.data[i * N + k];
448       for ( usize j = k + 1; j < N; ++j ) {
449         r.LU.data[i * N + j] = math::fma<F>(-m, r.LU.data[k * N + j], r.LU.data[i * N + j]);
450       }
451     }
452   }
453   return r;
454 }

464 template<ieee754_floating F>
465 [[nodiscard]] inline lu_pivot_result_dyn<F>
466 lu_pivot(const dynmat<F> &A) noexcept
467 {
468   lu_pivot_result_dyn<F> r{ A, micron::vector<usize, micron::allocator_serial<>, false>(A.rows), 1, false };
469   const usize N = A.rows;
470   for ( usize i = 0; i < N; ++i ) r.perm.data()[i] = i;
471   F *__restrict__ LU = r.LU.data();
472   const usize ld = r.LU.ld;
473 
474   micron::vector<F, micron::allocator_serial<>, false> scale(N);
475   for ( usize i = 0; i < N; ++i ) {
476     F m = F(0);
477     for ( usize j = 0; j < N; ++j ) {
478       const F v = math::fabs(LU[i * ld + j]);
479       if ( v > m ) m = v;
480     }
481     scale.data()[i] = m;
482   }
483 
484   for ( usize k = 0; k < N; ++k ) {
485     F best_a = F(0);
486     F best_s = F(1);
487     usize piv = k;
488     for ( usize i = k; i < N; ++i ) {
489       const F a = math::fabs(LU[i * ld + k]);
490       const F si = scale.data()[i];
491       if ( si == F(0) ) continue;
492       if ( a * best_s > best_a * si ) {
493         best_a = a;
494         best_s = si;
495         piv = i;
496       }
497     }
498     if ( best_a == F(0) ) {
499       r.singular = true;
500       return r;
501     }
502 
503     if ( piv != k ) {
504       for ( usize j = 0; j < N; ++j ) {
505         F t = LU[k * ld + j];
506         LU[k * ld + j] = LU[piv * ld + j];
507         LU[piv * ld + j] = t;
508       }
509       usize t = r.perm.data()[k];
510       r.perm.data()[k] = r.perm.data()[piv];
511       r.perm.data()[piv] = t;
512       F ts = scale.data()[k];
513       scale.data()[k] = scale.data()[piv];
514       scale.data()[piv] = ts;
515       r.sign = -r.sign;
516     }
517 
518     const F inv_piv = F(1) / LU[k * ld + k];
519     for ( usize i = k + 1; i < N; ++i ) LU[i * ld + k] *= inv_piv;
520     for ( usize i = k + 1; i < N; ++i ) {
521       const F m = LU[i * ld + k];
522       for ( usize j = k + 1; j < N; ++j ) {
523         LU[i * ld + j] = math::fma<F>(-m, LU[k * ld + j], LU[i * ld + j]);
524       }
525     }
526   }
527   return r;
528 }

530 template<ieee754_floating F, usize N>
531 inline void
532 lu_solve(const lu_pivot_result<F, N> &lu, vec<F, N> &x_inout) noexcept
533 {
534   vec<F, N> b{};
535   for ( usize i = 0; i < N; ++i ) b.data[i] = x_inout.data[lu.perm[i]];
536   for ( usize i = 0; i < N; ++i ) {
537     F s = b.data[i];
538     for ( usize j = 0; j < i; ++j ) s = math::fma<F>(-lu.LU.data[i * N + j], b.data[j], s);
539     b.data[i] = s;
540   }
541   for ( usize ii = N; ii-- > 0; ) {
542     F s = b.data[ii];
543     for ( usize j = ii + 1; j < N; ++j ) s = math::fma<F>(-lu.LU.data[ii * N + j], b.data[j], s);
544     b.data[ii] = s / lu.LU.data[ii * N + ii];
545   }
546   x_inout = b;

701     apply_householder_right<F, R_, R_>(r.Q, h.v, h.beta, k);
702     for ( usize i = k + 1; i < R_; ++i ) r.R.data[i * C_ + k] = F(0);
703   }
704   return r;
705 }

748 template<ieee754_floating F, usize N>
749 [[nodiscard]] inline lu_full_result<F, N>
750 lu_pivot_full(const mat<F, N, N> &A, F tol = F(0)) noexcept
751 {
752   lu_full_result<F, N> r{ A, {}, {}, 1, false, 0 };
753   for ( usize i = 0; i < N; ++i ) {
754     r.perm_row[i] = i;
755     r.perm_col[i] = i;
756   }
757 
758   for ( usize k = 0; k < N; ++k ) {
759     usize pi = k, pj = k;
760     F best = math::fabs(r.LU.data[k * N + k]);
761     for ( usize i = k; i < N; ++i ) {
762       for ( usize j = k; j < N; ++j ) {
763         F a = math::fabs(r.LU.data[i * N + j]);
764         if ( a > best ) {
765           best = a;
766           pi = i;
767           pj = j;
768         }
769       }
770     }
771     if ( best <= tol ) {
772       r.singular = true;
773       r.rank = k;
774       return r;
775     }
776     if ( pi != k ) {
777       for ( usize j = 0; j < N; ++j ) {
778         F t = r.LU.data[k * N + j];
779         r.LU.data[k * N + j] = r.LU.data[pi * N + j];
780         r.LU.data[pi * N + j] = t;
781       }
782       usize t = r.perm_row[k];
783       r.perm_row[k] = r.perm_row[pi];
784       r.perm_row[pi] = t;
785       r.sign = -r.sign;
786     }
787     if ( pj != k ) {
788       for ( usize i = 0; i < N; ++i ) {
789         F t = r.LU.data[i * N + k];
790         r.LU.data[i * N + k] = r.LU.data[i * N + pj];
791         r.LU.data[i * N + pj] = t;
792       }
793       usize t = r.perm_col[k];
794       r.perm_col[k] = r.perm_col[pj];
795       r.perm_col[pj] = t;
796       r.sign = -r.sign;
797     }
798 
799     const F inv_piv = F(1) / r.LU.data[k * N + k];
800     for ( usize i = k + 1; i < N; ++i ) r.LU.data[i * N + k] *= inv_piv;
801     for ( usize i = k + 1; i < N; ++i ) {
802       const F m = r.LU.data[i * N + k];
803       for ( usize j = k + 1; j < N; ++j ) {
804         r.LU.data[i * N + j] = math::fma<F>(-m, r.LU.data[k * N + j], r.LU.data[i * N + j]);
805       }
806     }
807     r.rank = k + 1;
808   }
809   return r;
810 }

893 {
894   // y = P_r * b
895   vec<F, N> y{};
896   for ( usize i = 0; i < N; ++i ) y.data[i] = b.data[lu.perm_row[i]];
897   for ( usize i = 0; i < N; ++i ) {

905     y.data[ii] = s / lu.LU.data[ii * N + ii];

907   // x = P_c * w  ->  x[perm_col[k]] = w[k]
908   vec<F, N> x{};
909   for ( usize i = 0; i < N; ++i ) x.data[i] = F(0);
910   for ( usize k = 0; k < N; ++k ) x.data[lu.perm_col[k]] = y.data[k];
911   return x;
912 }

940 //
941 // A * P = Q * R, R upper trapezoidal, Q orthogonal stored compactly

943 template<ieee754_floating F, usize R_, usize C_>
944   requires(R_ >= 2)
945 struct qr_colpiv_result {
946   householder_sequence<F, R_, C_> Q;
947   mat<F, R_, C_> R;
948   usize perm[C_];
949   int sign;
950   usize rank;
951 };

953 template<ieee754_floating F, usize R_, usize C_>
954   requires(R_ >= 2)
955 [[nodiscard]] inline qr_colpiv_result<F, R_, C_>
956 qr_householder_colpiv(const mat<F, R_, C_> &A, F tol = F(0)) noexcept
957 {
958   qr_colpiv_result<F, R_, C_> r{};
959   constexpr usize K = (R_ < C_) ? R_ : C_;
960   r.R = A;
961   for ( usize j = 0; j < C_; ++j ) r.perm[j] = j;
962   r.sign = 1;
963   r.rank = 0;
964   for ( usize i = 0; i < R_ * K; ++i ) r.Q.V.data[i] = F(0);
965   for ( usize k = 0; k < K; ++k ) r.Q.betas[k] = F(0);
966 
967   F c[C_];
968   for ( usize j = 0; j < C_; ++j ) {
969     F s = F(0);
970     for ( usize i = 0; i < R_; ++i ) {
971       const F v = r.R.data[i * C_ + j];
972       s = math::fma<F>(v, v, s);
973     }
974     c[j] = s;
975   }
976 
977   for ( usize k = 0; k < K; ++k ) {
978     usize p = k;
979     F cmax = c[k];
980     for ( usize j = k + 1; j < C_; ++j ) {
981       if ( c[j] > cmax ) {
982         cmax = c[j];
983         p = j;
984       }
985     }
986     if ( cmax <= tol ) break;
987 
988     if ( p != k ) {
989       for ( usize i = 0; i < R_; ++i ) {
990         F t = r.R.data[i * C_ + k];
991         r.R.data[i * C_ + k] = r.R.data[i * C_ + p];
992         r.R.data[i * C_ + p] = t;
993       }
994       F ct = c[k];
995       c[k] = c[p];
996       c[p] = ct;
997       usize pt = r.perm[k];
998       r.perm[k] = r.perm[p];
999       r.perm[p] = pt;
1000       r.sign = -r.sign;
1001     }
1002 
1003     vec<F, R_> x{};
1004     for ( usize i = 0; i < R_; ++i ) x.data[i] = F(0);
1005     for ( usize i = k; i < R_; ++i ) x.data[i] = r.R.data[i * C_ + k];
1006     auto h = householder_reflector<F, R_>(x, k);
1007     if ( h.beta == F(0) ) break;
1008     apply_householder_left<F, R_, C_>(r.R, h.v, h.beta, k);
1009     for ( usize i = k + 1; i < R_; ++i ) r.R.data[i * C_ + k] = F(0);
1010     for ( usize i = 0; i < R_; ++i ) r.Q.V.data[i * K + k] = h.v.data[i];
1011     r.Q.betas[k] = h.beta;
1012     r.rank = k + 1;
1013 
1014     for ( usize j = k + 1; j < C_; ++j ) {
1015       const F akj = r.R.data[k * C_ + j];
1016       F nc = c[j] - akj * akj;
1017       if ( nc < F(0) ) nc = F(0);
1018       c[j] = nc;
1019     }
1020   }
1021   return r;
1022 }

1084       r.perm.data()[k] = r.perm.data()[p];

1102       if ( nc < F(0) ) nc = F(0);
1103       c.data()[j] = nc;
1104     }
1105   }

1177 {
1178   qr_fullpiv_result<F, R_, C_> r{};
1179   constexpr usize K = (R_ < C_) ? R_ : C_;
1180   r.R = A;
1181   for ( usize i = 0; i < R_; ++i ) {
1182     r.perm_row[i] = i;
1183     r.row_transp[i] = i;
1184   }
1185   for ( usize j = 0; j < C_; ++j ) r.perm_col[j] = j;
1186   r.sign = 1;
1187   r.rank = 0;
1188   for ( usize i = 0; i < R_ * K; ++i ) r.Q.V.data[i] = F(0);
1189   for ( usize k = 0; k < K; ++k ) r.Q.betas[k] = F(0);
1190 
1191   for ( usize k = 0; k < K; ++k ) {
1192     usize pi = k, pj = k;
1193     F best = math::fabs(r.R.data[k * C_ + k]);
1194     for ( usize i = k; i < R_; ++i ) {
1195       for ( usize j = k; j < C_; ++j ) {
1196         F a = math::fabs(r.R.data[i * C_ + j]);
1197         if ( a > best ) {
1198           best = a;
1199           pi = i;
1200           pj = j;
1201         }
1202       }
1203     }
1204     if ( best <= tol ) break;
1205     r.row_transp[k] = pi;
1206     if ( pi != k ) {
1207       for ( usize j = 0; j < C_; ++j ) {
1208         F t = r.R.data[k * C_ + j];
1209         r.R.data[k * C_ + j] = r.R.data[pi * C_ + j];
1210         r.R.data[pi * C_ + j] = t;
1211       }
1212       usize t = r.perm_row[k];
1213       r.perm_row[k] = r.perm_row[pi];
1214       r.perm_row[pi] = t;
1215       r.sign = -r.sign;
1216     }
1217     if ( pj != k ) {
1218       for ( usize i = 0; i < R_; ++i ) {
1219         F t = r.R.data[i * C_ + k];
1220         r.R.data[i * C_ + k] = r.R.data[i * C_ + pj];
1221         r.R.data[i * C_ + pj] = t;
1222       }
1223       usize t = r.perm_col[k];
1224       r.perm_col[k] = r.perm_col[pj];
1225       r.perm_col[pj] = t;
1226       r.sign = -r.sign;
1227     }
1228 
1229     vec<F, R_> x{};
1230     for ( usize i = 0; i < R_; ++i ) x.data[i] = F(0);
1231     for ( usize i = k; i < R_; ++i ) x.data[i] = r.R.data[i * C_ + k];
1232     auto h = householder_reflector<F, R_>(x, k);
1233     if ( h.beta == F(0) ) break;
1234     apply_householder_left<F, R_, C_>(r.R, h.v, h.beta, k);
1235     for ( usize i = k + 1; i < R_; ++i ) r.R.data[i * C_ + k] = F(0);
1236     for ( usize i = 0; i < R_; ++i ) r.Q.V.data[i * K + k] = h.v.data[i];
1237     r.Q.betas[k] = h.beta;
1238     r.rank = k + 1;
1239   }
1240   return r;
1241 }

1504 [[nodiscard]] inline F
1505 rcond(const mat<F, N, N> &A) noexcept
1506 {

1508   for ( usize j = 0; j < N; ++j ) {
1509     F s = F(0);
1510     for ( usize i = 0; i < N; ++i ) s += math::fabs(A.data[i * N + j]);
1511     if ( s > nA ) nA = s;
1512   }
1513   if ( nA == F(0) ) return F(0);
1514   auto inv_r = inv<F, N>(A);
1515   if ( inv_r.singular ) return F(0);
1516   F nInv = F(0);
1517   for ( usize j = 0; j < N; ++j ) {
1518     F s = F(0);
1519     for ( usize i = 0; i < N; ++i ) s += math::fabs(inv_r.X.data[i * N + j]);
1520     if ( s > nInv ) nInv = s;
1521   }
1522   if ( nInv == F(0) ) return F(0);
1523   return F(1) / (nA * nInv);
1524 }
