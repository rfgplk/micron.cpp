28 template<typename T> struct checked_result {
29   T value;
30   bool overflow;
31 
32   [[nodiscard, gnu::always_inline]] constexpr explicit
33   operator bool() const noexcept
34   {
35     return !overflow;
36   }
37 };
38 
39 static_assert(sizeof(checked_result<u64>) <= 16, "checked_result<u64> must fit two registers");
40 static_assert(sizeof(checked_result<i64>) <= 16, "checked_result<i64> must fit two registers");

76 template<arith_integral T>
77 [[nodiscard, gnu::always_inline]] inline constexpr checked_result<T>
78 neg(T a) noexcept
79 {
80   // overflow iff a == T_min
81   if constexpr ( micron::is_signed_v<T> ) {
82     checked_result<T> r{};
83     r.overflow = (a == numeric_limits<T>::min());
84     r.value = r.overflow ? a : T(-a);
85     return r;
86   } else {
87     return { T(0 - a), a != 0 };
88   }
89 }
90 
91 template<arith_integral T>
92 [[nodiscard, gnu::always_inline]] inline constexpr checked_result<T>
93 div(T a, T b) noexcept
94 {
95   checked_result<T> r{};
96   if constexpr ( micron::is_signed_v<T> ) {
97     if ( b == T(-1) && a == numeric_limits<T>::min() ) {
98       r.overflow = true;
99       r.value = a;
100       return r;
101     }
102   }
103   r.overflow = false;
104   r.value = a / b;
105   return r;
106 }

116 template<arith_integral T>
117 [[nodiscard, gnu::always_inline]] inline constexpr T
118 add(T a, T b) noexcept
119 {
120   auto r = checked::add(a, b);
121   if ( !r.overflow ) return r.value;
122   if constexpr ( micron::is_signed_v<T> )
123     return (a < T{ 0 }) ? numeric_limits<T>::min() : numeric_limits<T>::max();
124   else
125     return numeric_limits<T>::max();
126 }
127 
128 template<arith_integral T>
129 [[nodiscard, gnu::always_inline]] inline constexpr T
130 sub(T a, T b) noexcept
131 {
132   auto r = checked::sub(a, b);
133   if ( !r.overflow ) return r.value;
134   if constexpr ( micron::is_signed_v<T> )
135     return (a < T{ 0 }) ? numeric_limits<T>::min() : numeric_limits<T>::max();
136   else
137     return T{ 0 };
138 }
139 
140 template<arith_integral T>
141 [[nodiscard, gnu::always_inline]] inline constexpr T
142 mul(T a, T b) noexcept
143 {
144   auto r = checked::mul(a, b);
145   if ( !r.overflow ) return r.value;
146   if constexpr ( micron::is_signed_v<T> )
147     return ((a < T{ 0 }) ^ (b < T{ 0 })) ? numeric_limits<T>::min() : numeric_limits<T>::max();
148   else
149     return numeric_limits<T>::max();
150 }
151 
152 template<arith_integral T>
153 [[nodiscard, gnu::always_inline]] inline constexpr T
154 neg(T a) noexcept
155 {
156   auto r = checked::neg(a);
157   if ( !r.overflow ) return r.value;
158   if constexpr ( micron::is_signed_v<T> )
159     return numeric_limits<T>::max();
160   else
161     return T{ 0 };
162 }

283 template<typename T>
284   requires(micron::is_integral_v<T> && micron::is_unsigned_v<T>)
285 [[nodiscard, gnu::always_inline]] inline constexpr micron::tuple<T, T>
286 adc(T a, T b, T carry_in) noexcept
287 {
288   auto s0 = checked::add<T>(a, b);
289   auto s1 = checked::add<T>(s0.value, carry_in);
290   return micron::make_tuple(s1.value, T(T(s0.overflow) | T(s1.overflow)));
291 }
292 
293 template<typename T>
294   requires(micron::is_integral_v<T> && micron::is_unsigned_v<T>)
295 [[nodiscard, gnu::always_inline]] inline constexpr micron::tuple<T, T>
296 sbb(T a, T b, T borrow_in) noexcept
297 {
298   auto s0 = checked::sub<T>(a, b);
299   auto s1 = checked::sub<T>(s0.value, borrow_in);
300   return micron::make_tuple(s1.value, T(T(s0.overflow) | T(s1.overflow)));
301 }
302 
303 [[nodiscard, gnu::always_inline]] inline constexpr micron::tuple<u64, u64>
304 mul64(u64 a, u64 b) noexcept
305 {
306 #if defined(__micron_arch_width_64) && defined(__SIZEOF_INT128__)
307   __uint128_t p = static_cast<__uint128_t>(a) * static_cast<__uint128_t>(b);
308   return micron::make_tuple(u64(p), u64(p >> 64));
309 #else
310   const u64 a_lo = u32(a), a_hi = a >> 32;
311   const u64 b_lo = u32(b), b_hi = b >> 32;
312   const u64 p0 = a_lo * b_lo;
313   const u64 p1 = a_lo * b_hi;
314   const u64 p2 = a_hi * b_lo;
315   const u64 p3 = a_hi * b_hi;
316   const u64 mid = (p0 >> 32) + u32(p1) + u32(p2);
317   u64 lo = (p0 & 0xffffffffULL) | (mid << 32);
318   u64 hi = p3 + (p1 >> 32) + (p2 >> 32) + (mid >> 32);
319   return micron::make_tuple(lo, hi);
320 #endif
321 }
