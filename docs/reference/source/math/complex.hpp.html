81   [[gnu::noinline]] constexpr complex &
82   __divide_scaled(complex z) noexcept
83   {
84     constexpr F epsilon = [] {
85       if constexpr ( micron::is_same_v<F, long double> )
86         return F(__micron_ldbl_epsilon);
87       else if constexpr ( sizeof(F) == 2 )
88         return F(0x1p-10);
89       else if constexpr ( sizeof(F) == 4 )
90         return F(micron::numeric_limits<float>::epsilon());
91       else if constexpr ( sizeof(F) == 8 )
92         return F(micron::numeric_limits<double>::epsilon());
93       else
94         return F(0x1p-112);
95     }();
96     constexpr F smallest = [] {
97       if constexpr ( micron::is_same_v<F, long double> )
98         return F(micron::numeric_limits<long double>::min());
99       else if constexpr ( sizeof(F) == 2 )
100         return F(0x1p-14);
101       else if constexpr ( sizeof(F) == 4 )
102         return F(micron::numeric_limits<float>::min());
103       else if constexpr ( sizeof(F) == 8 )
104         return F(micron::numeric_limits<double>::min());
105       else {
106         F value = F(0x1p-1022);
107         for ( unsigned i = 0; i < 4; ++i ) value *= value;
108         return value * F(0x1p-30);
109       }
110     }();
111     constexpr F largest = [=] {
112       if constexpr ( micron::is_same_v<F, long double> )
113         return F(micron::numeric_limits<long double>::max());
114       else if constexpr ( sizeof(F) == 2 )
115         return F(65504);
116       else if constexpr ( sizeof(F) == 4 )
117         return F(micron::numeric_limits<float>::max());
118       else if constexpr ( sizeof(F) == 8 )
119         return F(micron::numeric_limits<double>::max());
120       else
121         return (F(2) / smallest) * (F(2) - epsilon);
122     }();
123     constexpr F small = F(2) * (smallest / epsilon);
124     constexpr F boost = sizeof(F) == 2 ? F(1) / epsilon : F(2) / (epsilon * epsilon);
125     F ar = z.real < 0 ? -z.real : z.real, ai = z.imag < 0 ? -z.imag : z.imag;
126     F numerator = (real < 0 ? -real : real), other = (imag < 0 ? -imag : imag);
127     if ( other > numerator ) numerator = other;
128     F denominator = ar > ai ? ar : ai, a = real, b = imag, c = z.real, d = z.imag, scale = 1;
129     // Scale Smith's formula before its sums overflow or products underflow.
130     if ( numerator > largest / F(2) ) {
131       a *= F(0.5);
132       b *= F(0.5);
133       scale *= F(2);
134     } else if ( numerator && numerator < small ) {
135       a *= boost;
136       b *= boost;
137       scale /= boost;
138     }
139     if ( denominator > largest / F(2) ) {
140       c *= F(0.5);
141       d *= F(0.5);
142       scale *= F(0.5);
143     } else if ( denominator && denominator < small ) {
144       c *= boost;
145       d *= boost;
146       scale *= boost;
147     }
148     auto quotient_sum = [=](F x, F y, F r, F divisor) {
149       F product = y * r;
150       if ( y && r && (product < 0 ? -product : product) < smallest ) return x / divisor + (y / divisor) * r;
151       return (x + product) / divisor;
152     };
153     auto product_ratio = [=](F x, F y, F divisor) {
154       F r = y / divisor, magnitude = x < 0 ? -x : x;
155       if ( (r < 0 ? -r : r) >= smallest || magnitude <= F(1) ) return x * r;
156       F absolute_divisor = divisor < 0 ? -divisor : divisor;
157       if ( absolute_divisor >= F(1) || magnitude <= largest * absolute_divisor ) return (x / divisor) * y;
158       return (x * y) / divisor;
159     };
160     F r, divisor, re, im;
161     if ( ar >= ai ) {
162       r = d / c;
163       divisor = c + d * r;
164       if ( d && (r < 0 ? -r : r) < smallest ) {
165         re = a / c + product_ratio(b / c, d, c);
166         im = b / c - product_ratio(a / c, d, c);
167       } else {
168         re = quotient_sum(a, b, r, divisor);
169         im = quotient_sum(b, -a, r, divisor);
170       }
171     } else {
172       r = c / d;
173       divisor = d + c * r;
174       if ( c && (r < 0 ? -r : r) < smallest ) {
175         re = b / d + product_ratio(a / d, c, d);
176         im = -a / d + product_ratio(b / d, c, d);
177       } else {
178         re = quotient_sum(b, a, r, divisor);
179         im = quotient_sum(-a, b, r, divisor);
180       }
181     }
182     real = re * scale;
183     imag = im * scale;
184     return *this;
185   }

187 public:
188   [[gnu::always_inline]] constexpr complex &
189   operator/=(complex z) noexcept
190   {
191     if constexpr ( sizeof(F) == 2 || sizeof(F) == 4 ) {
192       using W = micron::conditional_t<sizeof(F) == 2, float, double>;
193       W a = real, b = imag, c = z.real, d = z.imag;
194       W reciprocal = W(1) / (c * c + d * d);
195       real = F((a * c + b * d) * reciprocal);
196       imag = F((b * c - a * d) * reciprocal);
197       return *this;
198     }
199     bool bounded;
200     if constexpr ( sizeof(F) == 8 ) {
201       using U = u64;
202       constexpr U sign_clear = ~(U(1) << 63), offset = U(129) << 52;
203       constexpr U mask = U(15) << 60, target = U(4) << 60;
204       U a = bits::bit_cast<U>(real) & sign_clear, b = bits::bit_cast<U>(imag) & sign_clear;
205       U c = bits::bit_cast<U>(z.real) & sign_clear, d = bits::bit_cast<U>(z.imag) & sign_clear;
206       auto outside = [=](U value) { return value ? ((value + offset) ^ target) & mask : U(0); };
207       bounded = !(outside(a) | outside(b) | outside(c) | outside(d)) && bool(c | d);
208     } else if constexpr ( micron::is_same_v<F, long double> && __micron_ldbl_mant_dig == 64 ) {
209       struct __value {
210         u64 significand;
211         u16 exponent;
212       };
213 
214       auto representation = [](const F &value) {
215         if consteval {
216           return bits::bit_cast<__value>(value);
217         }
218         // Read only x87's value bytes; its trailing padding is indeterminate.
219         __value result;
220         micron::cbytecpy<sizeof(result.significand)>(&result.significand, reinterpret_cast<const byte *>(&value));
221         micron::cbytecpy<sizeof(result.exponent)>(&result.exponent, reinterpret_cast<const byte *>(&value) + sizeof(result.significand));
222         return result;
223       };
224       auto a = representation(real), b = representation(imag);
225       auto c = representation(z.real), d = representation(z.imag);
226       auto outside = [](__value value) {
227         unsigned exponent = value.exponent & 0x7fffu;
228         return value.significand ? ((exponent + 129u) ^ 0x4000u) & 0xff00u : 0u;
229       };
230       bounded = !(outside(a) | outside(b) | outside(c) | outside(d)) && bool(c.significand | d.significand);
231     } else {
232       constexpr F lower = F(0x1p-128), upper = F(0x1p128);
233       F a = real < 0 ? -real : real, b = imag < 0 ? -imag : imag;
234       F c = z.real < 0 ? -z.real : z.real, d = z.imag < 0 ? -z.imag : z.imag;
235       F numerator = a > b ? a : b, denominator = c > d ? c : d;
236       bounded = numerator <= upper && denominator <= upper && denominator && (!a || a >= lower) && (!b || b >= lower) && (!c || c >= lower)
237                 && (!d || d >= lower);
238     }
239     if ( bounded ) {
240       F reciprocal = F(1) / (z.real * z.real + z.imag * z.imag);
241       F re = (real * z.real + imag * z.imag) * reciprocal;
242       imag = (imag * z.real - real * z.imag) * reciprocal;
243       real = re;
244       return *this;
245     }
246     return __divide_scaled(z);
247   }
