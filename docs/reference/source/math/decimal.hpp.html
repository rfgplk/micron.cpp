38 // At most ceil(767/9) limbs. Multiplication carries fit in u64 on every backend.
39 inline constexpr void
40 __binary64_digits(micron::__impl::__dec::__decimal &decimal, u64 mantissa, i32 exponent) noexcept
41 {
42   if ( !mantissa ) return;
43   if ( exponent < 0 ) {
44     const i32 zeros = static_cast<i32>(__builtin_ctzll(mantissa));
45     const i32 cancel = zeros < -exponent ? zeros : -exponent;
46     mantissa >>= cancel;
47     exponent += cancel;
48   }
49   u32 limbs[86];
50   const u32 power = static_cast<u32>(exponent < 0 ? -exponent : exponent);
51   const u32 row = exponent < 0 ? power / 13 : power / 29;
52   const u16 *offsets = exponent < 0 ? __decimal_fives.offsets : __decimal_twos.offsets;
53   const u32 *source = (exponent < 0 ? __decimal_fives.limbs : __decimal_twos.limbs) + offsets[row];
54   u32 used = offsets[row + 1] - offsets[row];
55   const u32 low = static_cast<u32>(mantissa % 1000000000), high = static_cast<u32>(mantissa / 1000000000);
56   u64 carry = 0;
57   for ( u32 i = 0; i < used; ++i ) {
58     const u64 product = static_cast<u64>(source[i]) * low + (i ? static_cast<u64>(source[i - 1]) * high : 0) + carry;
59     limbs[i] = static_cast<u32>(product % 1000000000);
60     carry = product / 1000000000;
61   }
62   carry += static_cast<u64>(source[used - 1]) * high;
63   while ( carry ) {
64     limbs[used++] = static_cast<u32>(carry % 1000000000);
65     carry /= 1000000000;
66   }
67   const auto multiply = [&](u32 factor) constexpr {
68     u64 carry = 0;
69     for ( u32 i = 0; i < used; ++i ) {
70       const u64 product = static_cast<u64>(limbs[i]) * factor + carry;
71       const u64 quotient = product / 1000000000;
72       limbs[i] = static_cast<u32>(product - quotient * 1000000000);
73       carry = quotient;
74     }
75     while ( carry ) {
76       limbs[used++] = static_cast<u32>(carry % 1000000000);
77       carry /= 1000000000;
78     }
79   };
80   const i32 scale = exponent < 0 ? -exponent : 0;
81   if ( exponent >= 0 ) {
82     if ( power % 29 ) multiply(1u << (power % 29));
83   } else {
84     u32 remaining = power % 13;
85     u32 factor = 1;
86     while ( remaining-- ) factor *= 5;
87     if ( factor != 1 ) multiply(factor);
88   }
89   u8 reversed[9];
90   u32 top = limbs[used - 1], digits = 0;
91   do {
92     reversed[digits++] = static_cast<u8>(top % 10);
93     top /= 10;
94   } while ( top );
95   decimal.ndigits = digits;
96   for ( u32 i = 0; i < digits; ++i ) decimal.d[i] = reversed[digits - 1 - i];
97   for ( u32 i = used - 1; i-- > 0; ) {
98     decimal.d[decimal.ndigits++] = static_cast<u8>(limbs[i] / 100000000);
99     micron::__impl::__digits::write8(decimal.d + decimal.ndigits, limbs[i] % 100000000);
100     decimal.ndigits += 8;
101   }
102   decimal.point = static_cast<i32>(decimal.ndigits) - scale;
103   micron::__impl::__dec::__dec_trim(decimal);
104 }

106 [[nodiscard]] inline constexpr decimal_result
107 format(char *destination, usize capacity, f64 value, u32 significant_digits) noexcept
108 {
109   const u64 bits = ieee::to_bits(value);
110   const u64 exponent = (bits >> 52) & 0x7ff;
111   if ( exponent == 0x7ff ) return { 0, decimal_error::nonfinite };
112   u64 mantissa = bits & 0xfffffffffffffull;
113   i32 binary_exponent = -1074;
114   if ( exponent ) {
115     mantissa |= u64{ 1 } << 52;
116     binary_exponent = static_cast<i32>(exponent) - 1075;
117   }
118   micron::__impl::__dec::__decimal decimal{};
119   __binary64_digits(decimal, mantissa, binary_exponent);
120   if ( decimal.truncated ) return { 0, decimal_error::internal_capacity };
121 
122   if ( significant_digits && significant_digits < decimal.ndigits ) {
123     const bool up = micron::__impl::__dec::__dec_round_up(decimal, significant_digits);
124     decimal.ndigits = significant_digits;
125     if ( up ) {
126       u32 position = significant_digits;
127       while ( position && decimal.d[position - 1] == 9 ) decimal.d[--position] = 0;
128       if ( position )
129         ++decimal.d[position - 1];
130       else {
131         decimal.d[0] = 1;
132         decimal.ndigits = 1;
133         ++decimal.point;
134       }
135     }
136     micron::__impl::__dec::__dec_trim(decimal);
137   }
138 
139   char output[binary64_decimal_capacity];
140   usize size = 0;
141   if ( bits >> 63 ) output[size++] = '-';
142   if ( !mantissa ) {
143     output[size++] = '0';
144   } else if ( decimal.point > 0 && decimal.point <= 16 && decimal.ndigits <= 32 ) {
145     for ( i32 i = 0; i < decimal.point || i < static_cast<i32>(decimal.ndigits); ++i ) {
146       if ( i == decimal.point ) output[size++] = '.';
147       output[size++] = i < static_cast<i32>(decimal.ndigits) ? static_cast<char>('0' + decimal.d[i]) : '0';
148     }
149   } else if ( decimal.point <= 0 && decimal.point >= -6 && static_cast<i32>(decimal.ndigits) - decimal.point <= 32 ) {
150     output[size++] = '0';
151     output[size++] = '.';
152     for ( i32 i = 0; i < -decimal.point; ++i ) output[size++] = '0';
153     micron::__impl::__digits::ascii(output + size, decimal.d, decimal.ndigits);
154     size += decimal.ndigits;
155   } else {
156     output[size++] = static_cast<char>('0' + decimal.d[0]);
157     if ( decimal.ndigits > 1 ) {
158       output[size++] = '.';
159       micron::__impl::__digits::ascii(output + size, decimal.d + 1, decimal.ndigits - 1);
160       size += decimal.ndigits - 1;
161     }
162     output[size++] = 'e';
163     i32 scientific_exponent = decimal.point - 1;
164     if ( scientific_exponent < 0 ) {
165       output[size++] = '-';
166       scientific_exponent = -scientific_exponent;
167     }
168     char digits[4]{};
169     u32 count = 0;
170     do {
171       digits[count++] = static_cast<char>('0' + scientific_exponent % 10);
172       scientific_exponent /= 10;
173     } while ( scientific_exponent );
174     while ( count ) output[size++] = digits[--count];
175   }
176   if ( !destination || size > capacity ) return { 0, decimal_error::short_output };
177   if consteval {
178     for ( usize i = 0; i < size; ++i ) destination[i] = output[i];
179   } else {
180     __builtin_memcpy(destination, output, size);
181   }
182   return { size };
183 }

187 // Exact finite value, preserving the existing spelling and atomic-output contract
188 [[nodiscard]] inline constexpr decimal_result
189 binary64_to_decimal(char *destination, usize capacity, f64 value) noexcept
190 {
191   return __impl::format(destination, capacity, value, 0);
192 }

194 // Round to nearest significant decimal digits, with ties to even
195 [[nodiscard]] inline constexpr decimal_result
196 binary64_to_decimal(char *destination, usize capacity, f64 value, u32 significant_digits) noexcept
197 {
198   if ( significant_digits < 1 || significant_digits > 17 ) return { 0, decimal_error::invalid_precision };
199   return __impl::format(destination, capacity, value, significant_digits);
200 }
