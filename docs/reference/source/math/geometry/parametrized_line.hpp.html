23 // L(t) = origin + t * direction
24 template<ieee754_floating F, usize Dim>
25   requires(Dim >= 2 && Dim <= 16)
26 struct parametrized_line {
27   vec<F, Dim> origin;
28   vec<F, Dim> direction;
29 
30   [[nodiscard, gnu::always_inline]] constexpr vec<F, Dim>
31   point_at(F t) const noexcept
32   {
33     vec<F, Dim> r{};
34     for ( usize i = 0; i < Dim; ++i ) r.data[i] = origin.data[i] + t * direction.data[i];
35     return r;
36   }
37 
38   // perpendicular distance from p to the line
39   [[nodiscard, gnu::always_inline]] constexpr F
40   distance(const vec<F, Dim> &p) const noexcept
41   {
42     F po[Dim];
43     F dot = F(0);
44     F d_sq = F(0);
45     for ( usize i = 0; i < Dim; ++i ) {
46       po[i] = p.data[i] - origin.data[i];
47       dot += po[i] * direction.data[i];
48       d_sq += direction.data[i] * direction.data[i];
49     }
50     const F u = (d_sq > F(0)) ? dot / d_sq : F(0);
51     F sq = F(0);
52     for ( usize i = 0; i < Dim; ++i ) {
53       const F w = po[i] - u * direction.data[i];
54       sq += w * w;
55     }
56     return math::fsqrt(sq);
57   }
58 
59   // t such that L(t) is on the hyperplane, 0 if parallel
60   [[nodiscard, gnu::always_inline]] constexpr F
61   intersection_param(const hyperplane<F, Dim> &h) const noexcept
62   {
63     F denom = F(0);
64     for ( usize i = 0; i < Dim; ++i ) denom += h.normal.data[i] * direction.data[i];
65     if ( denom == F(0) ) return F(0);
66     F num = -h.offset;
67     for ( usize i = 0; i < Dim; ++i ) num -= h.normal.data[i] * origin.data[i];
68     return num / denom;
69   }
70 
71   [[nodiscard, gnu::always_inline]] constexpr vec<F, Dim>
72   intersection_point(const hyperplane<F, Dim> &h) const noexcept
73   {
74     return point_at(intersection_param(h));
75   }
76 };
