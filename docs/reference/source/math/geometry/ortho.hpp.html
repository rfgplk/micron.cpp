27 // unit vector perpendicular to v
28 template<ieee754_floating F, usize N>
29   requires(N >= 2 && N <= 16)
30 [[nodiscard]] inline constexpr vec<F, N>
31 unit_orthogonal(const vec<F, N> &v) noexcept
32 {
33   if constexpr ( N == 3 ) {
34     F ax = math::fabs(v.data[0]);
35     F ay = math::fabs(v.data[1]);
36     F az = math::fabs(v.data[2]);
37     vec<F, 3> axis{};
38     if ( ax <= ay && ax <= az ) {
39       axis.data[0] = F(1);
40     } else if ( ay <= az ) {
41       axis.data[1] = F(1);
42     } else {
43       axis.data[2] = F(1);
44     }
45     auto c = linalg::ops::cross<F>(v, axis);
46     F nrm = math::fsqrt(c.data[0] * c.data[0] + c.data[1] * c.data[1] + c.data[2] * c.data[2]);
47     if ( nrm > F(0) ) {
48       F inv = F(1) / nrm;
49       for ( usize i = 0; i < 3; ++i ) c.data[i] *= inv;
50     }
51     return c;
52   } else {
53     usize idx_max = 0;
54     F amax = math::fabs(v.data[0]);
55     for ( usize i = 1; i < N; ++i ) {
56       F a = math::fabs(v.data[i]);
57       if ( a > amax ) {
58         amax = a;
59         idx_max = i;
60       }
61     }
62     usize idx_a = idx_max;
63     usize idx_b = (idx_max + 1) % N;
64     vec<F, N> u{};
65     for ( usize i = 0; i < N; ++i ) u.data[i] = F(0);
66     u.data[idx_a] = -v.data[idx_b];
67     u.data[idx_b] = v.data[idx_a];
68     F nrm_sq = u.data[idx_a] * u.data[idx_a] + u.data[idx_b] * u.data[idx_b];
69     F nrm = math::fsqrt(nrm_sq);
70     if ( nrm > F(0) ) {
71       F inv = F(1) / nrm;
72       for ( usize i = 0; i < N; ++i ) u.data[i] *= inv;
73     }
74     return u;
75   }
76 }
