27 // svd3 forms u_j = H*v_j / sigma_j and simply skips the scaling when sigma_j == 0
28 template<ieee754_floating F>
29 inline void
30 complete_u(mat<F, 3, 3> &U) noexcept
31 {
32   vec<F, 3> u0{ U.data[0], U.data[3], U.data[6] };
33   vec<F, 3> u1{ U.data[1], U.data[4], U.data[7] };
34 
35   F n0 = math::fsqrt(u0.data[0] * u0.data[0] + u0.data[1] * u0.data[1] + u0.data[2] * u0.data[2]);
36   if ( !(n0 > F(0.5)) ) {
37     u0 = vec<F, 3>{ F(1), F(0), F(0) };
38     n0 = F(1);
39   }
40   const F inv0 = F(1) / n0;
41   for ( usize i = 0; i < 3; ++i ) u0.data[i] *= inv0;
42 
43   F d = u1.data[0] * u0.data[0] + u1.data[1] * u0.data[1] + u1.data[2] * u0.data[2];
44   for ( usize i = 0; i < 3; ++i ) u1.data[i] -= d * u0.data[i];
45   F n1 = math::fsqrt(u1.data[0] * u1.data[0] + u1.data[1] * u1.data[1] + u1.data[2] * u1.data[2]);
46   if ( !(n1 > F(0.5)) ) {
47     const F ax = math::fabs(u0.data[0]);
48     const F ay = math::fabs(u0.data[1]);
49     const F az = math::fabs(u0.data[2]);
50     u1 = vec<F, 3>{ F(0), F(0), F(0) };
51     if ( ax <= ay && ax <= az )
52       u1.data[0] = F(1);
53     else if ( ay <= az )
54       u1.data[1] = F(1);
55     else
56       u1.data[2] = F(1);
57     d = u1.data[0] * u0.data[0] + u1.data[1] * u0.data[1] + u1.data[2] * u0.data[2];
58     for ( usize i = 0; i < 3; ++i ) u1.data[i] -= d * u0.data[i];
59     n1 = math::fsqrt(u1.data[0] * u1.data[0] + u1.data[1] * u1.data[1] + u1.data[2] * u1.data[2]);
60   }
61   const F inv1 = F(1) / n1;
62   for ( usize i = 0; i < 3; ++i ) u1.data[i] *= inv1;
63 
64   const vec<F, 3> u2 = linalg::ops::cross<F>(u0, u1);
65   for ( usize i = 0; i < 3; ++i ) {
66     U.data[i * 3 + 0] = u0.data[i];
67     U.data[i * 3 + 1] = u1.data[i];
68     U.data[i * 3 + 2] = u2.data[i];
69   }
70 }

74 template<ieee754_floating F>
75 [[nodiscard]] inline transform<F, 3, transform_mode::affine>
76 umeyama(const dynmat<F> &src, const dynmat<F> &dst, bool with_scaling = true) noexcept
77 {
78   transform<F, 3, transform_mode::affine> out{};
79   out.M = mat<F, 4, 4>::identity();
80   const usize N = src.cols;
81   if ( N == 0 || src.rows != 3 || dst.rows != 3 || dst.cols != N ) return out;
82 
83   // centroids
84   vec<F, 3> cs{};
85   vec<F, 3> cd{};
86   for ( usize i = 0; i < 3; ++i ) {
87     F s = F(0), d = F(0);
88     for ( usize j = 0; j < N; ++j ) {
89       s += src.at(i, j);
90       d += dst.at(i, j);
91     }
92     cs.data[i] = s / F(N);
93     cd.data[i] = d / F(N);
94   }
95 
96   // cross-covariance H = (1/N) * sum_j (dst_j - cd) (src_j - cs)^T
97   mat<F, 3, 3> H = mat<F, 3, 3>::zero();
98   F var_src = F(0);
99   for ( usize j = 0; j < N; ++j ) {
100     F sx[3];
101     F dx[3];
102     for ( usize i = 0; i < 3; ++i ) {
103       sx[i] = src.at(i, j) - cs.data[i];
104       dx[i] = dst.at(i, j) - cd.data[i];
105     }
106     for ( usize i = 0; i < 3; ++i )
107       for ( usize k = 0; k < 3; ++k ) H.data[i * 3 + k] += dx[i] * sx[k];
108     for ( usize i = 0; i < 3; ++i ) var_src += sx[i] * sx[i];
109   }
110   for ( usize i = 0; i < 9; ++i ) H.data[i] /= F(N);
111   var_src /= F(N);
112 
113   // SVD of H
114   auto sv = linalg::decomp::svd3<F>(H);
115   // build sign matrix S to ensure det(R) = +1
116   mat<F, 3, 3> Vt{};
117   for ( usize i = 0; i < 3; ++i )
118     for ( usize j = 0; j < 3; ++j ) Vt.data[i * 3 + j] = sv.V.data[j * 3 + i];      // V^T
119 
120   // det(U) * det(V) sign
121   F detU = linalg::ops::det3<F>(sv.U);
122   if ( !(math::fabs(detU) > F(0.5)) ) {
123     __impl_umeyama::complete_u<F>(sv.U);
124     detU = linalg::ops::det3<F>(sv.U);
125   }
126   F detV = linalg::ops::det3<F>(sv.V);
127   F sgn = (detU * detV >= F(0)) ? F(1) : F(-1);
128 
129   // R = U * diag(1, 1, sgn) * V^T
130   mat<F, 3, 3> R = mat<F, 3, 3>::zero();
131   for ( usize i = 0; i < 3; ++i )
132     for ( usize j = 0; j < 3; ++j ) {
133       F s = F(0);
134       for ( usize k = 0; k < 3; ++k ) {
135         F sign_k = (k == 2) ? sgn : F(1);
136         s += sv.U.data[i * 3 + k] * sign_k * Vt.data[k * 3 + j];
137       }
138       R.data[i * 3 + j] = s;
139     }
140 
141   // scale
142   F scale = F(1);
143   if ( with_scaling ) {
144     F trace_Ds = sv.S.data[0] + sv.S.data[1] + sgn * sv.S.data[2];
145     if ( var_src > F(0) ) scale = trace_Ds / var_src;
146   }
147 
148   // translation: cd - scale * R * cs
149   vec<F, 3> t{};
150   for ( usize i = 0; i < 3; ++i ) {
151     F s = cd.data[i];
152     for ( usize j = 0; j < 3; ++j ) s -= scale * R.data[i * 3 + j] * cs.data[j];
153     t.data[i] = s;
154   }
155 
156   // assemble into homogeneous 4x4
157   out.M = mat<F, 4, 4>::zero();
158   for ( usize i = 0; i < 3; ++i )
159     for ( usize j = 0; j < 3; ++j ) out.M.data[i * 4 + j] = scale * R.data[i * 3 + j];
160   for ( usize i = 0; i < 3; ++i ) out.M.data[i * 4 + 3] = t.data[i];
161   out.M.data[3 * 4 + 3] = F(1);
162   return out;
163 }
