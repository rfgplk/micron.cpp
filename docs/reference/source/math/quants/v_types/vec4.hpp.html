254   constexpr vector_4<T>
255   normalized() const
256   {
257     if !consteval {
258 #if defined(__micron_gfx_simd)
259       if constexpr ( micron::is_same_v<T, f32> ) {
260         vector_4<T> out{};
261         const simd::f128 vv = math::__vsimd::__load(reinterpret_cast<const float *>(&x));
262         const simd::f128 n2 = math::__vsimd::__dot4_splat(vv, vv);
263         const simd::f128 gmask = math::__vsimd::__cmp_le(n2, math::__vsimd::__splat(float(math::default_eps<T>())));
264         const simd::f128 r = math::__vsimd::__mul(vv, math::__vsimd::__inv_sqrt_exact(n2));
265         math::__vsimd::__store(reinterpret_cast<float *>(&out.x), math::__vsimd::__andnot(gmask, r));
266         return out;
267       }
268 #endif
269     }
270     T n2 = squared_norm();
271     if ( n2 <= math::default_eps<T>() ) return { T{ 0 }, T{ 0 }, T{ 0 }, T{ 0 } };
272     T inv = math::__vsimd::__inv_sqrt_exact_s(n2);
273     return { x * inv, y * inv, z * inv, w * inv };
274   }

363   constexpr T
364   angle(const vector_4<T> &v) const
365   {
366     T denom = math::fsqrt(squared_norm()) * math::fsqrt(v.squared_norm());
367     if ( denom <= math::fsqrt(math::default_eps<T>()) ) return T{ 0 };
368     T c = dot(v) / denom;
369     c = math::fclamp(c, T{ -1 }, T{ 1 });
370     return math::acos(c);
371   }

378     return dot(v) / denom;

409   constexpr vector_4<T>
410   refract(const vector_4<T> &normal, T eta) const
411   {
412     T nd = dot(normal);
413     T k = T{ 1 } - eta * eta * (T{ 1 } - nd * nd);
414     if ( k < T{ 0 } ) return {};
415     return *this * eta - normal * (eta * nd + math::fsqrt(k));
416   }
