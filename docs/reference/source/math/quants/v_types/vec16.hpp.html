466   constexpr vector_16<T>
467   refract(const vector_16<T> &normal, T eta) const
468   {
469     T nd = dot(normal);
470     T kk = T{ 1 } - eta * eta * (T{ 1 } - nd * nd);
471     if ( kk < T{ 0 } ) return {};
472     return *this * eta - normal * (eta * nd + math::fsqrt(kk));
