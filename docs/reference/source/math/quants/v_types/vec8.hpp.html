306   constexpr vector_8<T>
307   refract(const vector_8<T> &normal, T eta) const
308   {
309     T nd = dot(normal);
310     T k = T{ 1 } - eta * eta * (T{ 1 } - nd * nd);
311     if ( k < T{ 0 } ) return {};
312     return *this * eta - normal * (eta * nd + math::fsqrt(k));
313   }
