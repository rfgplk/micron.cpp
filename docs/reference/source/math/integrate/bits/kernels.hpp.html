32 template<ieee754_floating F>
33 [[nodiscard, gnu::always_inline]] inline F
34 sum_fast(const F *values, usize count) noexcept
35 {
36   F a0 = F(0), a1 = F(0), a2 = F(0), a3 = F(0);
37   usize i = 0;
38   for ( ; i + 4 <= count; i += 4 ) {
39     a0 += values[i];
40     a1 += values[i + 1];
41     a2 += values[i + 2];
42     a3 += values[i + 3];
43   }
44   for ( ; i < count; ++i ) a0 += values[i];
45   return (a0 + a1) + (a2 + a3);
46 }

48 template<ieee754_floating F>
49 [[nodiscard, gnu::always_inline]] inline F
50 weighted_sum_fast(const F *weights, const F *values, usize count) noexcept
51 {
52   F a0 = F(0), a1 = F(0), a2 = F(0), a3 = F(0);
53   usize i = 0;
54   for ( ; i + 4 <= count; i += 4 ) {
55     a0 += weights[i] * values[i];
56     a1 += weights[i + 1] * values[i + 1];
57     a2 += weights[i + 2] * values[i + 2];
58     a3 += weights[i + 3] * values[i + 3];
59   }
60   for ( ; i < count; ++i ) a0 += weights[i] * values[i];
61   return (a0 + a1) + (a2 + a3);
62 }

64 template<ieee754_floating F>
65 inline void
66 diff_uniform(const F *values, F *out, usize count, F dx) noexcept
67 {
68   if ( count == 0 ) return;
69   if ( count == 1 ) {
70     out[0] = F(0);
71     return;
72   }
73   out[0] = (values[1] - values[0]) / dx;
74   const F scale = F(0.5) / dx;
75   for ( usize i = 1; i + 1 < count; ++i ) out[i] = (values[i + 1] - values[i - 1]) * scale;
76   out[count - 1] = (values[count - 1] - values[count - 2]) / dx;
77 }

79 template<ieee754_floating F>
80 inline void
81 cum_trapezoid_uniform(const F *values, F *out, usize count, F dx) noexcept
82 {
83   if ( count == 0 ) return;
84   out[0] = F(0);
85   const F scale = F(0.5) * dx;
86   for ( usize i = 1; i < count; ++i ) out[i] = out[i - 1] + scale * (values[i - 1] + values[i]);
87 }

89 template<ieee754_floating F>
90 inline void
91 affine_transform(const F *values, F *out, usize count, F scale, F offset) noexcept
92 {
93   usize i = 0;
94   for ( ; i + 4 <= count; i += 4 ) {
95     out[i] = values[i] * scale + offset;
96     out[i + 1] = values[i + 1] * scale + offset;
97     out[i + 2] = values[i + 2] * scale + offset;
98     out[i + 3] = values[i + 3] * scale + offset;
99   }
100   for ( ; i < count; ++i ) out[i] = values[i] * scale + offset;
101 }
