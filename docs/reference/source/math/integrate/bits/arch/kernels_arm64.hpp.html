31 [[nodiscard, gnu::always_inline]] inline f32
32 sum_fast(const f32 *values, usize count) noexcept
33 {
34   simd::f128 a0 = simd::neon::splat_f32(0.0f), a1 = a0, a2 = a0, a3 = a0;
35   usize i = 0;
36   for ( ; i + 16 <= count; i += 16 ) {
37     a0 = simd::neon::add(a0, simd::neon::load_f32(reinterpret_cast<const float *>(values + i)));
38     a1 = simd::neon::add(a1, simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 4)));
39     a2 = simd::neon::add(a2, simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 8)));
40     a3 = simd::neon::add(a3, simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 12)));
41   }
42   f32 result = __horizontal(simd::neon::add(simd::neon::add(a0, a1), simd::neon::add(a2, a3)));
43   for ( ; i < count; ++i ) result += values[i];
44   return result;
45 }

63 [[nodiscard, gnu::always_inline]] inline f32
64 weighted_sum_fast(const f32 *weights, const f32 *values, usize count) noexcept
65 {
66   simd::f128 a0 = simd::neon::splat_f32(0.0f), a1 = a0, a2 = a0, a3 = a0;
67   usize i = 0;
68   for ( ; i + 16 <= count; i += 16 ) {
69     a0 = simd::neon::fma_f32(a0, simd::neon::load_f32(reinterpret_cast<const float *>(weights + i)),
70                              simd::neon::load_f32(reinterpret_cast<const float *>(values + i)));
71     a1 = simd::neon::fma_f32(a1, simd::neon::load_f32(reinterpret_cast<const float *>(weights + i + 4)),
72                              simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 4)));
73     a2 = simd::neon::fma_f32(a2, simd::neon::load_f32(reinterpret_cast<const float *>(weights + i + 8)),
74                              simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 8)));
75     a3 = simd::neon::fma_f32(a3, simd::neon::load_f32(reinterpret_cast<const float *>(weights + i + 12)),
76                              simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 12)));
77   }
78   f32 result = __horizontal(simd::neon::add(simd::neon::add(a0, a1), simd::neon::add(a2, a3)));
79   for ( ; i < count; ++i ) result += weights[i] * values[i];

103 inline void
104 diff_uniform(const f32 *values, f32 *out, usize count, f32 dx) noexcept
105 {
106   if ( count < 2 ) {
107     if ( count == 1 ) out[0] = f32(0);
108     return;
109   }
110   out[0] = (values[1] - values[0]) / dx;
111   const simd::f128 scale = simd::neon::splat_f32(float(f32(0.5) / dx));
112   usize i = 1;
113   for ( ; i + 4 <= count - 1; i += 4 ) {
114     const simd::f128 lo = simd::neon::load_f32(reinterpret_cast<const float *>(values + i - 1));
115     const simd::f128 hi = simd::neon::load_f32(reinterpret_cast<const float *>(values + i + 1));
116     simd::neon::store_f32(reinterpret_cast<float *>(out + i), simd::neon::mul(simd::neon::sub(hi, lo), scale));
117   }
118   for ( ; i + 1 < count; ++i ) out[i] = (values[i + 1] - values[i - 1]) * f32(0.5) / dx;
119   out[count - 1] = (values[count - 1] - values[count - 2]) / dx;
120 }

141 inline void
142 cum_trapezoid_uniform(const f32 *values, f32 *out, usize count, f32 dx) noexcept
143 {
144   if ( count == 0 ) return;
145   out[0] = f32(0);
146   const simd::f128 zero = simd::neon::splat_f32(0.0f);
147   const simd::f128 scale = simd::neon::splat_f32(float(f32(0.5) * dx));
148   f32 carry = f32(0);
149   usize i = 1;
150   for ( ; i + 4 <= count; i += 4 ) {
151     simd::f128 value = simd::neon::mul(simd::neon::add(simd::neon::load_f32(reinterpret_cast<const float *>(values + i - 1)),
152                                                        simd::neon::load_f32(reinterpret_cast<const float *>(values + i))),
153                                        scale);
154     value = simd::neon::add(value, simd::neon::ext_f32<3>(zero, value));
155     value = simd::neon::add(value, simd::neon::ext_f32<2>(zero, value));
156     value = simd::neon::add(value, simd::neon::splat_f32(float(carry)));
157     simd::neon::store_f32(reinterpret_cast<float *>(out + i), value);
158     carry = f32(simd::neon::get_lane_f32<3>(value));
159   }
160   for ( ; i < count; ++i ) {
161     carry += f32(0.5) * dx * (values[i - 1] + values[i]);
162     out[i] = carry;
163   }
164 }

166 inline void
167 cum_trapezoid_uniform(const f64 *values, f64 *out, usize count, f64 dx) noexcept
168 {
169   if ( count == 0 ) return;
170   out[0] = f64(0);
171   const simd::d128 zero = simd::neon::splat_f64(0.0);
172   const simd::d128 scale = simd::neon::splat_f64(double(f64(0.5) * dx));
173   f64 carry = f64(0);
174   usize i = 1;
175   for ( ; i + 2 <= count; i += 2 ) {
176     simd::d128 value = simd::neon::mul(simd::neon::add(simd::neon::load_f64(reinterpret_cast<const double *>(values + i - 1)),
177                                                        simd::neon::load_f64(reinterpret_cast<const double *>(values + i))),
178                                        scale);
179     value = simd::neon::add(value, vextq_f64(zero, value, 1));
180     value = simd::neon::add(value, simd::neon::splat_f64(double(carry)));

182     carry = f64(simd::neon::get_lane_f64<1>(value));
183   }
184   for ( ; i < count; ++i ) {
185     carry += f64(0.5) * dx * (values[i - 1] + values[i]);
186     out[i] = carry;

190 inline void
191 affine_transform(const f32 *values, f32 *out, usize count, f32 scale, f32 offset) noexcept
192 {
193   const simd::f128 s = simd::neon::splat_f32(float(scale));
194   const simd::f128 o = simd::neon::splat_f32(float(offset));
195   usize i = 0;
196   for ( ; i + 4 <= count; i += 4 ) {
197     const simd::f128 value = simd::neon::load_f32(reinterpret_cast<const float *>(values + i));
198     simd::neon::store_f32(reinterpret_cast<float *>(out + i), simd::neon::fma_f32(o, value, s));
199   }
200   for ( ; i < count; ++i ) out[i] = values[i] * scale + offset;
201 }
