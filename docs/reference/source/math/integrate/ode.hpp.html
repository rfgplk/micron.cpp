62 template<ieee754_floating F, usize N> struct dense_segment {
63   F t0{ 0 };
64   F t1{ 0 };
65   vec<F, N> y0{};
66   vec<F, N> y1{};
67   vec<F, N> f0{};
68   vec<F, N> f1{};
69 
70   [[nodiscard]] inline vec<F, N>
71   evaluate(F t) const noexcept
72   {
73     vec<F, N> output{};
74     const F h = t1 - t0;
75     if ( h == F(0) ) return y1;
76     F theta = (t - t0) / h;
77     if ( theta < F(0) ) theta = F(0);
78     if ( theta > F(1) ) theta = F(1);
79     const F theta2 = theta * theta;
80     const F theta3 = theta2 * theta;
81     const F h00 = F(2) * theta3 - F(3) * theta2 + F(1);
82     const F h10 = theta3 - F(2) * theta2 + theta;
83     const F h01 = -F(2) * theta3 + F(3) * theta2;
84     const F h11 = theta3 - theta2;
85     for ( usize i = 0; i < N; ++i ) output.data[i] = h00 * y0.data[i] + h * h10 * f0.data[i] + h01 * y1.data[i] + h * h11 * f1.data[i];
86     return output;
87   }
88 };

198 template<usize N, ieee754_floating F>
199 [[nodiscard]] inline bool
200 valid_output(const output_buffer<F, N> *output, F t0, F t_bound) noexcept
201 {
202   if ( output == nullptr ) return true;
203   if ( output->written > output->count ) return false;
204   if ( output->count == 0 ) return true;
205   if ( output->times == nullptr || output->states == nullptr ) return false;
206   const F direction = t_bound >= t0 ? F(1) : F(-1);
207   for ( usize i = output->written; i < output->count; ++i ) {
208     if ( !ieee::is_finite<F>(output->times[i]) || direction * (output->times[i] - t0) < F(0)
209          || direction * (output->times[i] - t_bound) > F(0) )
210       return false;
211     if ( i != output->written && direction * (output->times[i] - output->times[i - 1]) < F(0) ) return false;
212   }
213   return true;
214 }

277 template<ieee754_floating F, typename Evaluate>
278 [[nodiscard]] inline F
279 locate_bracket(F start, F end, F before, F after, Evaluate evaluate) noexcept
280 {
281   F a = start;
282   F b = end;
283   F fa = before;
284   F fb = after;
285   if ( fb == F(0) ) return b;
286   if ( fa == F(0) ) return a;
287   if ( b < a ) {
288     const F time = a;
289     a = b;
290     b = time;
291     const F value = fa;
292     fa = fb;
293     fb = value;
294   }
295   if ( mk::manip::fabs<F>(fa) < mk::manip::fabs<F>(fb) ) {
296     const F time = a;
297     a = b;
298     b = time;
299     const F value = fa;
300     fa = fb;
301     fb = value;
302   }
303   F c = a;
304   F fc = fa;
305   F d = c;
306   bool bisected = true;
307   for ( usize iteration = 0; iteration < 64; ++iteration ) {
308     F trial{};
309     if ( fa != fc && fb != fc ) {
310       trial = a * fb * fc / ((fa - fb) * (fa - fc)) + b * fa * fc / ((fb - fa) * (fb - fc)) + c * fa * fb / ((fc - fa) * (fc - fb));
311     } else {
312       trial = b - fb * (b - a) / (fb - fa);
313     }
314     const F guarded = (F(3) * a + b) * F(0.25);
315     const F guard_lo = guarded < b ? guarded : b;
316     const F guard_hi = guarded < b ? b : guarded;
317     const F tolerance = F(16) * machine_epsilon<F>() * (mk::manip::fabs<F>(a) + mk::manip::fabs<F>(b) + F(1));
318     const bool outside = trial <= guard_lo || trial >= guard_hi;
319     const bool slow_bisection = bisected && mk::manip::fabs<F>(trial - b) >= F(0.5) * mk::manip::fabs<F>(b - c);
320     const bool slow_interpolation = !bisected && mk::manip::fabs<F>(trial - b) >= F(0.5) * mk::manip::fabs<F>(c - d);
321     const bool stale_bisection = bisected && mk::manip::fabs<F>(b - c) <= tolerance;
322     const bool stale_interpolation = !bisected && mk::manip::fabs<F>(c - d) <= tolerance;
323     if ( outside || slow_bisection || slow_interpolation || stale_bisection || stale_interpolation || !ieee::is_finite<F>(trial) ) {
324       trial = F(0.5) * (a + b);
325       bisected = true;
326     } else {
327       bisected = false;
328     }
329 
330     const F value = evaluate(trial);
331     if ( value == F(0) ) return trial;
332     if ( !ieee::is_finite<F>(value) ) {
333       trial = F(0.5) * (a + b);
334       const F midpoint_value = evaluate(trial);
335       if ( !ieee::is_finite<F>(midpoint_value) ) return b;
336       d = c;
337       c = b;
338       fc = fb;
339       if ( (fa < F(0) && midpoint_value > F(0)) || (fa > F(0) && midpoint_value < F(0)) ) {
340         b = trial;
341         fb = midpoint_value;
342       } else {
343         a = trial;
344         fa = midpoint_value;
345       }
346     } else {
347       d = c;
348       c = b;
349       fc = fb;
350       if ( (fa < F(0) && value > F(0)) || (fa > F(0) && value < F(0)) ) {
351         b = trial;
352         fb = value;
353       } else {
354         a = trial;
355         fa = value;
356       }
357     }
358     if ( mk::manip::fabs<F>(fa) < mk::manip::fabs<F>(fb) ) {
359       const F time = a;
360       a = b;
361       b = time;
362       const F fval = fa;
363       fa = fb;
364       fb = fval;
365     }
366     if ( mk::manip::fabs<F>(b - a) <= tolerance ) break;
367   }
368   return b;
369 }

384 write_outputs(const dense_segment<F, N> &dense, F direction, output_buffer<F, N> *output) noexcept
385 {
386   if ( output == nullptr || output->times == nullptr || output->states == nullptr ) return;
387   while ( output->written < output->count ) {
388     const F time = output->times[output->written];
389     if ( direction * (time - dense.t0) < F(0) ) {
390       ++output->written;
391       continue;
392     }
393     if ( direction * (time - dense.t1) > F(0) ) break;
394     output->states[output->written] = dense.evaluate(time);
395     ++output->written;
396   }

409 template<usize N, ieee754_floating F>
410 [[nodiscard]] inline F
411 error_norm(const vec<F, N> &error, const vec<F, N> &before, const vec<F, N> &after, const options<F> &settings) noexcept
412 {
413   F sum = F(0);
414   for ( usize i = 0; i < N; ++i ) {
415     const F magnitude = mk::manip::fabs<F>(before.data[i]) > mk::manip::fabs<F>(after.data[i]) ? mk::manip::fabs<F>(before.data[i])
416                                                                                                : mk::manip::fabs<F>(after.data[i]);
417     const F scale = settings.abs_tol + settings.rel_tol * magnitude;
418     const F ratio = error.data[i] / scale;
419     sum += ratio * ratio;
420   }
421   return mk::pow_ns::sqrt<F>(sum / F(N));
422 }

435 template<method Method, ieee754_floating F>
436 [[nodiscard]] inline F
437 step_factor(F error, F previous_error, const options<F> &settings, bool rejected) noexcept
438 {
439   if ( error == F(0) ) return settings.max_factor;
440   const F exponent = -F(1) / F(estimator_order<Method> + 1);
441   F factor = settings.safety * mk::pow_ns::pow<F>(error, exponent);
442   if ( previous_error > F(0) ) factor *= mk::pow_ns::pow<F>(previous_error, settings.pi_beta);
443   factor = clamp(factor, settings.min_factor, settings.max_factor);
444   if ( rejected && factor > F(1) ) factor = F(1);
445   return factor;
446 }

531 template<usize N, ieee754_floating F, typename Rhs>
532 [[nodiscard]] inline bool
533 rk23_step(Rhs &function, F t, const vec<F, N> &y, F h, rk23_workspace<F, N> &workspace, usize &evaluations) noexcept
534 {
535   if ( !workspace.has_fsal ) {
536     rhs<N, F>(function, t, y, workspace.stages[0]);
537     ++evaluations;
538   }
539   for ( usize i = 0; i < N; ++i ) workspace.temporary.data[i] = y.data[i] + h * F(0.5) * workspace.stages[0].data[i];
540   rhs<N, F>(function, t + h * F(0.5), workspace.temporary, workspace.stages[1]);
541   for ( usize i = 0; i < N; ++i ) workspace.temporary.data[i] = y.data[i] + h * F(0.75) * workspace.stages[1].data[i];
542   rhs<N, F>(function, t + h * F(0.75), workspace.temporary, workspace.stages[2]);
543   for ( usize i = 0; i < N; ++i )
544     workspace.next.data[i] = y.data[i]
545                              + h
546                                    * (F(2.0L / 9.0L) * workspace.stages[0].data[i] + F(1.0L / 3.0L) * workspace.stages[1].data[i]
547                                       + F(4.0L / 9.0L) * workspace.stages[2].data[i]);
548   rhs<N, F>(function, t + h, workspace.next, workspace.stages[3]);
549   evaluations += 3;
550   for ( usize i = 0; i < N; ++i )
551     workspace.error.data[i] = h
552                               * (F(-5.0L / 72.0L) * workspace.stages[0].data[i] + F(1.0L / 12.0L) * workspace.stages[1].data[i]
553                                  + F(1.0L / 9.0L) * workspace.stages[2].data[i] - F(1.0L / 8.0L) * workspace.stages[3].data[i]);
554   return finite(workspace.next) && finite(workspace.error) && finite(workspace.stages[3]);
555 }

557 template<usize N, ieee754_floating F, typename Rhs>
558 [[nodiscard]] inline bool
559 rk45_step(Rhs &function, F t, const vec<F, N> &y, F h, rk45_workspace<F, N> &workspace, usize &evaluations) noexcept
560 {
561   if ( !workspace.has_fsal ) {
562     rhs<N, F>(function, t, y, workspace.stages[0]);
563     ++evaluations;
564   }
565   constexpr F c[6] = { F(1.0L / 5.0L), F(3.0L / 10.0L), F(4.0L / 5.0L), F(8.0L / 9.0L), F(1), F(1) };
566   constexpr F a[5][5] = {
567     { F(1.0L / 5.0L) },
568     { F(3.0L / 40.0L), F(9.0L / 40.0L) },
569     { F(44.0L / 45.0L), F(-56.0L / 15.0L), F(32.0L / 9.0L) },
570     { F(19372.0L / 6561.0L), F(-25360.0L / 2187.0L), F(64448.0L / 6561.0L), F(-212.0L / 729.0L) },
571     { F(9017.0L / 3168.0L), F(-355.0L / 33.0L), F(46732.0L / 5247.0L), F(49.0L / 176.0L), F(-5103.0L / 18656.0L) },
572   };
573   for ( usize stage = 1; stage <= 5; ++stage ) {
574     for ( usize i = 0; i < N; ++i ) {
575       F combination = F(0);
576       for ( usize j = 0; j < stage; ++j ) combination += a[stage - 1][j] * workspace.stages[j].data[i];
577       workspace.temporary.data[i] = y.data[i] + h * combination;
578     }
579     rhs<N, F>(function, t + c[stage - 1] * h, workspace.temporary, workspace.stages[stage]);
580   }
581   constexpr F b[6] = { F(35.0L / 384.0L), F(0), F(500.0L / 1113.0L), F(125.0L / 192.0L), F(-2187.0L / 6784.0L), F(11.0L / 84.0L) };
582   for ( usize i = 0; i < N; ++i ) {
583     F combination = F(0);
584     for ( usize stage = 0; stage < 6; ++stage ) combination += b[stage] * workspace.stages[stage].data[i];
585     workspace.next.data[i] = y.data[i] + h * combination;
586   }
587   rhs<N, F>(function, t + h, workspace.next, workspace.stages[6]);
588   evaluations += 6;
589   constexpr F e[7] = { F(71.0L / 57600.0L), F(0), F(-71.0L / 16695.0L), F(71.0L / 1920.0L), F(-17253.0L / 339200.0L), F(22.0L / 525.0L),
590                        F(-1.0L / 40.0L) };
591   for ( usize i = 0; i < N; ++i ) {
592     F combination = F(0);
593     for ( usize stage = 0; stage < 7; ++stage ) combination += e[stage] * workspace.stages[stage].data[i];
594     workspace.error.data[i] = h * combination;
595   }
596   return finite(workspace.next) && finite(workspace.error) && finite(workspace.stages[6]);
597 }

599 template<usize N, ieee754_floating F, typename Rhs>
600 [[nodiscard]] inline bool
601 dop853_step(Rhs &function, F t, const vec<F, N> &y, F h, dop853_workspace<F, N> &workspace, usize &evaluations) noexcept
602 {
603   using table = dop853_coefficients<F>;
604   if ( !workspace.has_fsal ) {
605     rhs<N, F>(function, t, y, workspace.stages[0]);
606     ++evaluations;
607   }
608   for ( usize stage = 1; stage < 12; ++stage ) {
609     for ( usize i = 0; i < N; ++i ) {
610       F combination = F(0);
611       for ( usize j = 0; j < stage; ++j ) combination += table::a[stage][j] * workspace.stages[j].data[i];
612       workspace.temporary.data[i] = y.data[i] + h * combination;
613     }
614     rhs<N, F>(function, t + table::c[stage] * h, workspace.temporary, workspace.stages[stage]);
615   }
616   for ( usize i = 0; i < N; ++i ) {
617     F combination = F(0);
618     for ( usize stage = 0; stage < 12; ++stage ) combination += table::b[stage] * workspace.stages[stage].data[i];
619     workspace.next.data[i] = y.data[i] + h * combination;
620   }
621   rhs<N, F>(function, t + h, workspace.next, workspace.stages[12]);
622   evaluations += 12;
623   for ( usize i = 0; i < N; ++i ) {
624     F err5 = F(0), err3 = F(0);
625     for ( usize stage = 0; stage < 13; ++stage ) {
626       err5 += table::e5[stage] * workspace.stages[stage].data[i];
627       err3 += table::e3[stage] * workspace.stages[stage].data[i];
628     }
629     const F denominator = mk::pow_ns::sqrt<F>(err5 * err5 + F(0.01) * err3 * err3);
630     workspace.error.data[i] = denominator == F(0) ? F(0) : h * err5 * mk::manip::fabs<F>(err5) / denominator;
631   }
632   return finite(workspace.next) && finite(workspace.error) && finite(workspace.stages[12]);
633 }

672 template<method Method, usize N, ieee754_floating F, typename Rhs, typename Workspace, typename Observer = null_observer,
673          typename Event = no_event>
674   requires(Method == method::rk23 || Method == method::rk45 || Method == method::dop853)
675 [[nodiscard]] inline result<F, N>
676 solve_ivp(Rhs function, micron::__type_identity_t<F> t0, const vec<F, N> &initial, micron::__type_identity_t<F> t_bound,
677           const options<F> &settings, Workspace &workspace, Observer observer = {}, Event event = {}, output_buffer<F, N> *output = nullptr,
678           dense_buffer<F, N> *dense_storage = nullptr) noexcept
679 {
680   result<F, N> result_value{};
681   result_value.t = t0;
682   result_value.y = initial;
683   workspace.has_fsal = false;
684   if ( !__impl::valid_options(settings) || !__impl::finite(initial) || !ieee::is_finite<F>(t0) || !ieee::is_finite<F>(t_bound)
685        || !__impl::valid_output(output, t0, t_bound) || !__impl::valid_dense(dense_storage) ) {
686     result_value.termination = status::invalid_input;
687     return result_value;
688   }
689   if ( t0 == t_bound ) {
690     __impl::observe<N, F>(observer, t0, initial);
691     if ( output != nullptr )
692       while ( output->written < output->count && output->times[output->written] == t0 ) output->states[output->written++] = initial;
693     result_value.t_eval_written = output == nullptr ? 0 : output->written;
694     return result_value;
695   }
696 
697   const F direction = t_bound > t0 ? F(1) : F(-1);
698   const F span = mk::manip::fabs<F>(t_bound - t0);
699   F step_abs = settings.initial_step == F(0) ? span / F(100) : mk::manip::fabs<F>(settings.initial_step);
700   if ( step_abs == F(0) ) step_abs = span;
701   const F max_step = settings.max_step > F(0) ? settings.max_step : span;
702   if ( step_abs > max_step ) step_abs = max_step;
703   F previous_error = F(1);
704   usize consecutive_rejections = 0;
705   bool rejected_before_accept = false;
706 
707   __impl::observe<N, F>(observer, t0, initial);
708   if ( output != nullptr && output->times != nullptr && output->states != nullptr ) {
709     while ( output->written < output->count && output->times[output->written] == t0 ) {
710       output->states[output->written] = initial;
711       ++output->written;
712     }
713   }
714 
715   F event_before = F(0);
716   if constexpr ( !micron::is_same_v<Event, no_event> ) {
717     event_before = __impl::event_value<N, F>(event, t0, initial);
718     if ( !ieee::is_finite<F>(event_before) ) {
719       result_value.termination = status::non_finite;
720       return result_value;
721     }
722     if ( event_before == F(0) ) {
723       result_value.event_index = event.index;
724       result_value.event_time = t0;
725       if ( event.terminal ) {
726         result_value.termination = status::event;
727         result_value.t_eval_written = output == nullptr ? 0 : output->written;
728         return result_value;
729       }
730     }
731   }
732 
733   while ( direction * (result_value.t - t_bound) < F(0) ) {
734     if ( result_value.attempted_steps >= settings.max_steps ) {
735       result_value.termination = status::max_steps;
736       break;
737     }
738     F min_step = settings.min_step;
739     if ( min_step <= F(0) ) min_step = F(16) * machine_epsilon<F>() * (mk::manip::fabs<F>(result_value.t) + F(1));
740     if ( step_abs < min_step ) {
741       result_value.termination = status::step_underflow;
742       break;
743     }
744     F h = direction * step_abs;
745     if ( direction * (result_value.t + h - t_bound) > F(0) ) h = t_bound - result_value.t;
746     if ( result_value.t + h == result_value.t ) {
747       result_value.termination = status::step_underflow;
748       break;
749     }
750 
751     ++result_value.attempted_steps;
752     if ( !__impl::adaptive_step<Method, N, F>(function, result_value.t, result_value.y, h, workspace, result_value.n_evals) ) {
753       result_value.termination = status::non_finite;
754       break;
755     }
756     const F norm = __impl::error_norm<N, F>(workspace.error, result_value.y, workspace.next, settings);
757     if ( !ieee::is_finite<F>(norm) ) {
758       result_value.termination = status::non_finite;
759       break;
760     }
761     if ( norm > F(1) ) {
762       ++result_value.rejected_steps;
763       ++consecutive_rejections;
764       rejected_before_accept = true;
765       if ( consecutive_rejections > settings.rejection_limit ) {
766         result_value.termination = status::rejection_limit;
767         break;
768       }
769       step_abs *= __impl::step_factor<Method>(norm, previous_error, settings, true);
770       workspace.has_fsal = true;
771       continue;
772     }
773 
774     consecutive_rejections = 0;
775     workspace.last.t0 = result_value.t;
776     workspace.last.t1 = result_value.t + h;
777     workspace.last.y0 = result_value.y;
778     workspace.last.y1 = workspace.next;
779     workspace.last.f0 = workspace.stages[0];
780     constexpr usize endpoint = Method == method::rk23 ? 3 : (Method == method::rk45 ? 6 : 12);
781     workspace.last.f1 = workspace.stages[endpoint];
782     result_value.t += h;
783     result_value.y = workspace.next;
784     result_value.last_step = h;
785     ++result_value.accepted_steps;
786     __impl::accept_fsal<Method>(workspace);
787 
788     bool terminal_event = false;
789     if constexpr ( !micron::is_same_v<Event, no_event> ) {
790       const F event_after = __impl::event_value<N, F>(event, result_value.t, result_value.y);
791       if ( !ieee::is_finite<F>(event_after) ) {
792         result_value.termination = status::non_finite;
793         break;
794       }
795       if ( __impl::crossed<F>(event_before, event_after, event.direction) ) {
796         const F root = __impl::locate_event<N, F>(event, workspace.last, event_before, event_after);
797         result_value.event_index = event.index;
798         result_value.event_time = root;
799         if ( event.terminal ) {
800           const F event_step = root - workspace.last.t0;
801           result_value.t = root;
802           result_value.y = workspace.last.evaluate(root);
803           workspace.last.t1 = root;
804           workspace.last.y1 = result_value.y;
805           __impl::rhs<N, F>(function, root, result_value.y, workspace.last.f1);
806           ++result_value.n_evals;
807           result_value.last_step = event_step;
808           result_value.termination = status::event;
809           terminal_event = true;
810         }
811       }
812       event_before = event_after;
813     }
814 
815     __impl::write_outputs<N, F>(workspace.last, direction, output);
816     if ( !__impl::store_dense<N, F>(workspace.last, dense_storage) ) {
817       result_value.termination = status::storage_full;
818       break;
819     }
820     if ( terminal_event ) {
821       __impl::observe<N, F>(observer, result_value.t, result_value.y);
822       break;
823     }
824 
825     __impl::observe<N, F>(observer, result_value.t, result_value.y);
826     F factor = __impl::step_factor<Method>(norm, previous_error, settings, rejected_before_accept);
827     rejected_before_accept = false;
828     previous_error = norm > F(1e-4L) ? norm : F(1e-4L);
829     step_abs *= factor;
830     if ( step_abs > max_step ) step_abs = max_step;
831   }
832 
833   result_value.t_eval_written = output == nullptr ? 0 : output->written;
834   result_value.dense_segments_written = dense_storage == nullptr ? 0 : dense_storage->written;
835   return result_value;
836 }

883 template<method Method, usize N, ieee754_floating F, typename Rhs, typename Observer = null_observer, typename Event = no_event>
884   requires(Method == method::euler || Method == method::midpoint || Method == method::rk4)
885 [[nodiscard]] inline result<F, N>
886 integrate_fixed(Rhs function, micron::__type_identity_t<F> t0, const vec<F, N> &initial, micron::__type_identity_t<F> t_bound,
887                 micron::__type_identity_t<F> step, fixed_workspace<F, N> &workspace, Observer observer = {}, Event event = {},
888                 output_buffer<F, N> *output = nullptr, dense_buffer<F, N> *dense_storage = nullptr, usize max_steps = 100000) noexcept
889 {
890   result<F, N> result_value{};
891   result_value.t = t0;
892   result_value.y = initial;
893   if ( step == F(0) || max_steps == 0 || !__impl::finite(initial) || !ieee::is_finite<F>(step) || !ieee::is_finite<F>(t0)
894        || !ieee::is_finite<F>(t_bound) || !__impl::valid_output(output, t0, t_bound) || !__impl::valid_dense(dense_storage) ) {
895     result_value.termination = status::invalid_input;
896     return result_value;
897   }
898   if ( t0 == t_bound ) {
899     __impl::observe<N, F>(observer, t0, initial);
900     if ( output != nullptr && output->written < output->count && output->times[output->written] == t0 ) {
901       output->states[output->written++] = initial;
902       result_value.t_eval_written = output->written;
903     }
904     return result_value;
905   }
906   const F direction = t_bound > t0 ? F(1) : F(-1);
907   const F step_abs = mk::manip::fabs<F>(step);
908   __impl::observe<N, F>(observer, t0, initial);
909   if ( output != nullptr ) {
910     while ( output->written < output->count && output->times[output->written] == t0 ) output->states[output->written++] = initial;
911   }
912   F event_before = F(0);
913   if constexpr ( !micron::is_same_v<Event, no_event> ) {
914     event_before = __impl::event_value<N, F>(event, t0, initial);
915     if ( !ieee::is_finite<F>(event_before) ) {
916       result_value.termination = status::non_finite;
917       return result_value;
918     }
919     if ( event_before == F(0) ) {
920       result_value.event_index = event.index;
921       result_value.event_time = t0;
922       if ( event.terminal ) {
923         result_value.termination = status::event;
924         result_value.t_eval_written = output == nullptr ? 0 : output->written;
925         return result_value;
926       }
927     }
928   }
929   while ( direction * (result_value.t - t_bound) < F(0) ) {
930     if ( result_value.attempted_steps >= max_steps ) {
931       result_value.termination = status::max_steps;
932       break;
933     }
934     F h = direction * step_abs;
935     if ( direction * (result_value.t + h - t_bound) > F(0) ) h = t_bound - result_value.t;
936     vec<F, N> &next = workspace.next;
937     __impl::fixed_step<Method, N, F>(function, result_value.t, result_value.y, h, workspace, next, result_value.n_evals);
938     ++result_value.attempted_steps;
939     if ( !__impl::finite(next) ) {
940       result_value.termination = status::non_finite;
941       break;
942     }
943     workspace.last.t0 = result_value.t;
944     workspace.last.t1 = result_value.t + h;
945     workspace.last.y0 = result_value.y;
946     workspace.last.y1 = next;
947     workspace.last.f0 = workspace.k1;
948     if constexpr ( Method == method::euler )
949       workspace.last.f1 = workspace.k2;
950     else if constexpr ( Method == method::midpoint )
951       workspace.last.f1 = workspace.k3;
952     else
953       workspace.last.f1 = workspace.temporary;
954     result_value.t += h;
955     result_value.y = next;
956     result_value.last_step = h;
957     ++result_value.accepted_steps;
958     bool terminal_event = false;
959     if constexpr ( !micron::is_same_v<Event, no_event> ) {
960       const F event_after = __impl::event_value<N, F>(event, result_value.t, result_value.y);
961       if ( !ieee::is_finite<F>(event_after) ) {
962         result_value.termination = status::non_finite;
963         break;
964       }
965       if ( __impl::crossed<F>(event_before, event_after, event.direction) ) {
966         const F root = __impl::locate_event<N, F>(event, workspace.last, event_before, event_after);
967         result_value.event_index = event.index;
968         result_value.event_time = root;
969         if ( event.terminal ) {
970           const F event_step = root - workspace.last.t0;
971           result_value.t = root;
972           result_value.y = workspace.last.evaluate(root);
973           workspace.last.t1 = root;
974           workspace.last.y1 = result_value.y;
975           __impl::rhs<N, F>(function, root, result_value.y, workspace.last.f1);
976           ++result_value.n_evals;
977           result_value.last_step = event_step;
978           result_value.termination = status::event;
979           terminal_event = true;
980         }
981       }
982       event_before = event_after;
983     }
984     __impl::write_outputs(workspace.last, direction, output);
985     if ( !__impl::store_dense(workspace.last, dense_storage) ) {
986       result_value.termination = status::storage_full;
987       break;
988     }
989     if ( terminal_event ) {
990       __impl::observe<N, F>(observer, result_value.t, result_value.y);
991       break;
992     }
993     __impl::observe<N, F>(observer, result_value.t, result_value.y);
994   }
995   result_value.t_eval_written = output == nullptr ? 0 : output->written;
996   result_value.dense_segments_written = dense_storage == nullptr ? 0 : dense_storage->written;
997   return result_value;
998 }

1081 template<ieee754_floating F> struct runtime_workspace {
1082   F *stages{ nullptr };                // stage_capacity * length
1083   F *temporary{ nullptr };             // length
1084   F *next{ nullptr };                  // length
1085   F *error{ nullptr };                 // length
1086   F *start{ nullptr };                 // length, retained for last dense step
1087   F *start_derivative{ nullptr };      // length
1088   F *end_derivative{ nullptr };        // length
1089   usize length{ 0 };
1090   usize stage_capacity{ 0 };
1091   bool has_fsal{ false };
1092   F last_t0{ 0 };
1093   F last_t1{ 0 };
1094 
1095   [[nodiscard]] inline bool
1096   valid(usize n, usize required_stages) const noexcept
1097   {
1098     return n != 0 && length >= n && stage_capacity >= required_stages && stages != nullptr && temporary != nullptr && next != nullptr
1099            && error != nullptr && start != nullptr && start_derivative != nullptr && end_derivative != nullptr;
1100   }
1101 
1102   inline void
1103   dense_evaluate(F t, F *out, usize n) const noexcept
1104   {
1105     const F h = last_t1 - last_t0;
1106     if ( h == F(0) ) {
1107       for ( usize i = 0; i < n; ++i ) out[i] = next[i];
1108       return;
1109     }
1110     F theta = (t - last_t0) / h;
1111     if ( theta < F(0) ) theta = F(0);
1112     if ( theta > F(1) ) theta = F(1);
1113     const F theta2 = theta * theta;
1114     const F theta3 = theta2 * theta;
1115     const F h00 = F(2) * theta3 - F(3) * theta2 + F(1);
1116     const F h10 = theta3 - F(2) * theta2 + theta;
1117     const F h01 = -F(2) * theta3 + F(3) * theta2;
1118     const F h11 = theta3 - theta2;
1119     for ( usize i = 0; i < n; ++i ) out[i] = h00 * start[i] + h * h10 * start_derivative[i] + h01 * next[i] + h * h11 * end_derivative[i];
1120   }
1121 };

1131 template<ieee754_floating F> struct runtime_dense_buffer {
1132   F *times{ nullptr };             // interleaved t0, t1 pairs
1133   F *coefficients{ nullptr };      // y0, y1, f0, f1 blocks per segment
1134   usize capacity{ 0 };
1135   usize stride{ 0 };
1136   usize written{ 0 };
1137 
1138   [[nodiscard]] inline bool
1139   valid(usize n) const noexcept
1140   {
1141     return written <= capacity && (stride == 0 || stride >= n) && (capacity == 0 || (times != nullptr && coefficients != nullptr));
1142   }
1143 
1144   inline void
1145   evaluate(usize segment, F t, F *out, usize n) const noexcept
1146   {
1147     if ( out == nullptr || segment >= written ) return;
1148     const usize width = stride == 0 ? n : stride;
1149     const F t0 = times[2 * segment];
1150     const F t1 = times[2 * segment + 1];
1151     const F *base = coefficients + segment * 4 * width;
1152     const F *y0 = base;
1153     const F *y1 = base + width;
1154     const F *f0 = base + 2 * width;
1155     const F *f1 = base + 3 * width;
1156     const F h = t1 - t0;
1157     if ( h == F(0) ) {
1158       for ( usize i = 0; i < n; ++i ) out[i] = y1[i];
1159       return;
1160     }
1161     F theta = (t - t0) / h;
1162     if ( theta < F(0) ) theta = F(0);
1163     if ( theta > F(1) ) theta = F(1);
1164     const F theta2 = theta * theta;
1165     const F theta3 = theta2 * theta;
1166     const F h00 = F(2) * theta3 - F(3) * theta2 + F(1);
1167     const F h10 = theta3 - F(2) * theta2 + theta;
1168     const F h01 = -F(2) * theta3 + F(3) * theta2;
1169     const F h11 = theta3 - theta2;
1170     for ( usize i = 0; i < n; ++i ) out[i] = h00 * y0[i] + h * h10 * f0[i] + h01 * y1[i] + h * h11 * f1[i];
1171   }
1172 };

1210 template<ieee754_floating F>
1211 [[nodiscard]] inline F
1212 runtime_error_norm(const F *error, const F *before, const F *after, usize n, const options<F> &settings) noexcept
1213 {
1214   F sum = F(0);
1215   for ( usize i = 0; i < n; ++i ) {

1217     const F after_abs = mk::manip::fabs<F>(after[i]);
1218     const F scale = settings.abs_tol + settings.rel_tol * (before_abs > after_abs ? before_abs : after_abs);

1221   }

1227 template<method Method, ieee754_floating F, typename Rhs>
1228 [[nodiscard]] inline bool
1229 runtime_adaptive_step(Rhs &function, F t, const F *y, usize n, F h, runtime_workspace<F> &workspace, usize &evaluations) noexcept
1230 {
1231   auto stage = [&](usize s) noexcept -> F * { return workspace.stages + s * n; };
1232   if ( !workspace.has_fsal ) {
1233     runtime_rhs(function, t, y, stage(0), n);
1234     ++evaluations;
1235   }
1236   if constexpr ( Method == method::rk23 ) {
1237     for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = y[i] + h * F(0.5) * stage(0)[i];
1238     runtime_rhs(function, t + h * F(0.5), workspace.temporary, stage(1), n);
1239     for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = y[i] + h * F(0.75) * stage(1)[i];
1240     runtime_rhs(function, t + h * F(0.75), workspace.temporary, stage(2), n);
1241     for ( usize i = 0; i < n; ++i )
1242       workspace.next[i] = y[i] + h * (F(2.0L / 9.0L) * stage(0)[i] + F(1.0L / 3.0L) * stage(1)[i] + F(4.0L / 9.0L) * stage(2)[i]);
1243     runtime_rhs(function, t + h, workspace.next, stage(3), n);
1244     evaluations += 3;
1245     for ( usize i = 0; i < n; ++i )
1246       workspace.error[i] = h
1247                            * (F(-5.0L / 72.0L) * stage(0)[i] + F(1.0L / 12.0L) * stage(1)[i] + F(1.0L / 9.0L) * stage(2)[i]
1248                               - F(1.0L / 8.0L) * stage(3)[i]);
1249   } else if constexpr ( Method == method::rk45 ) {
1250     constexpr F c[5] = { F(1.0L / 5.0L), F(3.0L / 10.0L), F(4.0L / 5.0L), F(8.0L / 9.0L), F(1) };
1251     constexpr F a[5][5] = {
1252       { F(1.0L / 5.0L) },
1253       { F(3.0L / 40.0L), F(9.0L / 40.0L) },
1254       { F(44.0L / 45.0L), F(-56.0L / 15.0L), F(32.0L / 9.0L) },
1255       { F(19372.0L / 6561.0L), F(-25360.0L / 2187.0L), F(64448.0L / 6561.0L), F(-212.0L / 729.0L) },
1256       { F(9017.0L / 3168.0L), F(-355.0L / 33.0L), F(46732.0L / 5247.0L), F(49.0L / 176.0L), F(-5103.0L / 18656.0L) },
1257     };
1258     for ( usize s = 1; s <= 5; ++s ) {
1259       for ( usize i = 0; i < n; ++i ) {
1260         F combination = F(0);
1261         for ( usize j = 0; j < s; ++j ) combination += a[s - 1][j] * stage(j)[i];
1262         workspace.temporary[i] = y[i] + h * combination;
1263       }
1264       runtime_rhs(function, t + c[s - 1] * h, workspace.temporary, stage(s), n);
1265     }
1266     constexpr F b[6] = { F(35.0L / 384.0L), F(0), F(500.0L / 1113.0L), F(125.0L / 192.0L), F(-2187.0L / 6784.0L), F(11.0L / 84.0L) };
1267     for ( usize i = 0; i < n; ++i ) {
1268       F combination = F(0);
1269       for ( usize s = 0; s < 6; ++s ) combination += b[s] * stage(s)[i];
1270       workspace.next[i] = y[i] + h * combination;
1271     }
1272     runtime_rhs(function, t + h, workspace.next, stage(6), n);
1273     evaluations += 6;
1274     constexpr F e[7] = { F(71.0L / 57600.0L), F(0), F(-71.0L / 16695.0L), F(71.0L / 1920.0L), F(-17253.0L / 339200.0L), F(22.0L / 525.0L),
1275                          F(-1.0L / 40.0L) };
1276     for ( usize i = 0; i < n; ++i ) {
1277       F combination = F(0);
1278       for ( usize s = 0; s < 7; ++s ) combination += e[s] * stage(s)[i];
1279       workspace.error[i] = h * combination;
1280     }
1281   } else {
1282     using table = dop853_coefficients<F>;
1283     for ( usize s = 1; s < 12; ++s ) {
1284       for ( usize i = 0; i < n; ++i ) {
1285         F combination = F(0);
1286         for ( usize j = 0; j < s; ++j ) combination += table::a[s][j] * stage(j)[i];
1287         workspace.temporary[i] = y[i] + h * combination;
1288       }
1289       runtime_rhs(function, t + table::c[s] * h, workspace.temporary, stage(s), n);
1290     }
1291     for ( usize i = 0; i < n; ++i ) {
1292       F combination = F(0);
1293       for ( usize s = 0; s < 12; ++s ) combination += table::b[s] * stage(s)[i];
1294       workspace.next[i] = y[i] + h * combination;
1295     }
1296     runtime_rhs(function, t + h, workspace.next, stage(12), n);
1297     evaluations += 12;
1298     for ( usize i = 0; i < n; ++i ) {
1299       F err5 = F(0), err3 = F(0);
1300       for ( usize s = 0; s < 13; ++s ) {
1301         err5 += table::e5[s] * stage(s)[i];
1302         err3 += table::e3[s] * stage(s)[i];
1303       }
1304       const F denominator = mk::pow_ns::sqrt<F>(err5 * err5 + F(0.01) * err3 * err3);
1305       workspace.error[i] = denominator == F(0) ? F(0) : h * err5 * mk::manip::fabs<F>(err5) / denominator;
1306     }
1307   }
1308   constexpr usize endpoint = Method == method::rk23 ? 3 : (Method == method::rk45 ? 6 : 12);
1309   return runtime_finite(workspace.next, n) && runtime_finite(workspace.error, n) && runtime_finite(stage(endpoint), n);
1310 }

1312 template<ieee754_floating F>
1313 inline void
1314 runtime_write_outputs(runtime_workspace<F> &workspace, F direction, usize n, runtime_output_buffer<F> *output) noexcept
1315 {
1316   if ( output == nullptr || output->times == nullptr || output->states == nullptr ) return;
1317   const usize stride = output->stride == 0 ? n : output->stride;
1318   while ( output->written < output->count ) {
1319     const F time = output->times[output->written];
1320     if ( direction * (time - workspace.last_t0) < F(0) ) {
1321       ++output->written;
1322       continue;
1323     }
1324     if ( direction * (time - workspace.last_t1) > F(0) ) break;
1325     workspace.dense_evaluate(time, output->states + output->written * stride, n);
1326     ++output->written;
1327   }
1328 }

1351 template<ieee754_floating F>
1352 [[nodiscard]] inline bool
1353 runtime_store_dense(const runtime_workspace<F> &workspace, usize n, runtime_dense_buffer<F> *storage) noexcept
1354 {
1355   if ( storage == nullptr ) return true;
1356   if ( storage->written >= storage->capacity ) return false;
1357   const usize segment = storage->written++;
1358   const usize width = storage->stride == 0 ? n : storage->stride;
1359   storage->times[2 * segment] = workspace.last_t0;
1360   storage->times[2 * segment + 1] = workspace.last_t1;
1361   F *base = storage->coefficients + segment * 4 * width;
1362   F *y0 = base;
1363   F *y1 = base + width;
1364   F *f0 = base + 2 * width;
1365   F *f1 = base + 3 * width;
1366   for ( usize i = 0; i < n; ++i ) {
1367     y0[i] = workspace.start[i];
1368     y1[i] = workspace.next[i];
1369     f0[i] = workspace.start_derivative[i];
1370     f1[i] = workspace.end_derivative[i];
1371   }
1372   return true;

1378 {
1379   if ( output == nullptr ) return true;
1380   if ( output->written > output->count || (output->stride != 0 && output->stride < n) ) return false;
1381   if ( output->count == 0 ) return true;
1382   if ( output->times == nullptr || output->states == nullptr ) return false;

1388     if ( i != output->written && direction * (output->times[i] - output->times[i - 1]) < F(0) ) return false;
1389   }
1390   return true;

1406                     const options<F> &settings, runtime_workspace<F> &workspace, Observer observer, Event event,
1407                     runtime_output_buffer<F> *output, runtime_dense_buffer<F> *dense_storage) noexcept
1408 {
1409   runtime_result<F> result_value{};
1410   result_value.t = t0;
1411   constexpr usize required_stages = __impl::runtime_stages<Method>;
1412   if ( state == nullptr || !workspace.valid(n, required_stages) || !__impl::valid_options(settings) || !__impl::runtime_finite(state, n)
1413        || !ieee::is_finite<F>(t0) || !ieee::is_finite<F>(t_bound) || !__impl::runtime_valid_output(output, n, t0, t_bound)
1414        || !__impl::runtime_valid_dense(dense_storage, n) ) {
1415     result_value.termination = status::invalid_input;
1416     return result_value;
1417   }
1418   workspace.has_fsal = false;
1419   const usize output_stride = output == nullptr || output->stride == 0 ? n : output->stride;
1420   if ( t0 == t_bound ) {
1421     if constexpr ( requires { observer(t0, state, n); } ) observer(t0, state, n);
1422     if ( output != nullptr ) {
1423       while ( output->written < output->count && output->times[output->written] == t0 ) {
1424         for ( usize i = 0; i < n; ++i ) output->states[output->written * output_stride + i] = state[i];
1425         ++output->written;
1426       }
1427       result_value.t_eval_written = output->written;
1428     }
1429     if constexpr ( !micron::is_same_v<Event, no_event> ) {
1430       const F initial_event = __impl::runtime_event_value<F>(event, t0, state, n);
1431       if ( !ieee::is_finite<F>(initial_event) )
1432         result_value.termination = status::non_finite;
1433       else if ( initial_event == F(0) ) {
1434         result_value.event_index = event.index;
1435         result_value.event_time = t0;
1436         if ( event.terminal ) result_value.termination = status::event;
1437       }
1438     }
1439     return result_value;
1440   }
1441   const F direction = t_bound > t0 ? F(1) : F(-1);
1442   const F span = mk::manip::fabs<F>(t_bound - t0);
1443   F step_abs = settings.initial_step == F(0) ? span / F(100) : mk::manip::fabs<F>(settings.initial_step);
1444   const F max_step = settings.max_step > F(0) ? settings.max_step : span;
1445   if ( step_abs > max_step ) step_abs = max_step;
1446   F previous_error = F(1);
1447   usize consecutive_rejections = 0;
1448   bool rejected_before_accept = false;
1449   if constexpr ( requires { observer(t0, state, n); } ) observer(t0, state, n);
1450   if ( output != nullptr ) {
1451     while ( output->written < output->count && output->times[output->written] == t0 ) {
1452       for ( usize i = 0; i < n; ++i ) output->states[output->written * output_stride + i] = state[i];
1453       ++output->written;
1454     }
1455   }
1456   F event_before = F(0);
1457   if constexpr ( !micron::is_same_v<Event, no_event> ) {
1458     event_before = __impl::runtime_event_value<F>(event, t0, state, n);
1459     if ( !ieee::is_finite<F>(event_before) ) {
1460       result_value.termination = status::non_finite;
1461       return result_value;
1462     }
1463     if ( event_before == F(0) ) {
1464       result_value.event_index = event.index;
1465       result_value.event_time = t0;
1466       if ( event.terminal ) {
1467         result_value.termination = status::event;
1468         result_value.t_eval_written = output == nullptr ? 0 : output->written;
1469         return result_value;
1470       }
1471     }
1472   }
1473   while ( direction * (result_value.t - t_bound) < F(0) ) {
1474     if ( result_value.attempted_steps >= settings.max_steps ) {
1475       result_value.termination = status::max_steps;
1476       break;
1477     }
1478     F min_step = settings.min_step;
1479     if ( min_step <= F(0) ) min_step = F(16) * machine_epsilon<F>() * (mk::manip::fabs<F>(result_value.t) + F(1));
1480     if ( step_abs < min_step ) {
1481       result_value.termination = status::step_underflow;
1482       break;
1483     }
1484     F h = direction * step_abs;
1485     if ( direction * (result_value.t + h - t_bound) > F(0) ) h = t_bound - result_value.t;
1486     ++result_value.attempted_steps;
1487     if ( !__impl::runtime_adaptive_step<Method>(function, result_value.t, state, n, h, workspace, result_value.n_evals) ) {
1488       result_value.termination = status::non_finite;
1489       break;
1490     }
1491     const F norm = __impl::runtime_error_norm(workspace.error, state, workspace.next, n, settings);
1492     if ( !ieee::is_finite<F>(norm) ) {
1493       result_value.termination = status::non_finite;
1494       break;
1495     }
1496     if ( norm > F(1) ) {
1497       ++result_value.rejected_steps;
1498       if ( ++consecutive_rejections > settings.rejection_limit ) {
1499         result_value.termination = status::rejection_limit;
1500         break;
1501       }
1502       rejected_before_accept = true;
1503       step_abs *= __impl::step_factor<Method>(norm, previous_error, settings, true);
1504       workspace.has_fsal = true;
1505       continue;
1506     }
1507     consecutive_rejections = 0;
1508     workspace.last_t0 = result_value.t;
1509     workspace.last_t1 = result_value.t + h;
1510     constexpr usize endpoint = Method == method::rk23 ? 3 : (Method == method::rk45 ? 6 : 12);
1511     F *first_stage = workspace.stages;
1512     F *last_stage = workspace.stages + endpoint * n;
1513     for ( usize i = 0; i < n; ++i ) {
1514       workspace.start[i] = state[i];
1515       workspace.start_derivative[i] = first_stage[i];
1516       workspace.end_derivative[i] = last_stage[i];
1517       state[i] = workspace.next[i];
1518       first_stage[i] = last_stage[i];
1519     }
1520     workspace.has_fsal = true;
1521     result_value.t += h;
1522     result_value.last_step = h;
1523     ++result_value.accepted_steps;
1524 
1525     bool terminal_event = false;
1526     if constexpr ( !micron::is_same_v<Event, no_event> ) {
1527       const F event_after = __impl::runtime_event_value<F>(event, result_value.t, state, n);
1528       if ( !ieee::is_finite<F>(event_after) ) {
1529         result_value.termination = status::non_finite;
1530         break;
1531       }
1532       if ( __impl::crossed<F>(event_before, event_after, event.direction) ) {
1533         const F root = __impl::runtime_locate_event<F>(event, workspace, n, event_before, event_after);
1534         result_value.event_index = event.index;
1535         result_value.event_time = root;
1536         if ( event.terminal ) {
1537           workspace.dense_evaluate(root, workspace.temporary, n);
1538           workspace.last_t1 = root;
1539           for ( usize i = 0; i < n; ++i ) state[i] = workspace.next[i] = workspace.temporary[i];
1540           __impl::runtime_rhs(function, root, state, workspace.end_derivative, n);
1541           ++result_value.n_evals;
1542           result_value.t = root;
1543           result_value.last_step = root - workspace.last_t0;
1544           result_value.termination = status::event;
1545           terminal_event = true;
1546         }
1547       }
1548       event_before = event_after;
1549     }
1550 
1551     __impl::runtime_write_outputs(workspace, direction, n, output);
1552     if ( !__impl::runtime_store_dense(workspace, n, dense_storage) ) {
1553       result_value.termination = status::storage_full;
1554       break;
1555     }
1556     if constexpr ( requires { observer(result_value.t, state, n); } ) observer(result_value.t, state, n);
1557     if ( terminal_event ) break;
1558     const F factor = __impl::step_factor<Method>(norm, previous_error, settings, rejected_before_accept);
1559     rejected_before_accept = false;
1560     previous_error = norm > F(1e-4L) ? norm : F(1e-4L);
1561     step_abs *= factor;
1562     if ( step_abs > max_step ) step_abs = max_step;
1563   }
1564   result_value.t_eval_written = output == nullptr ? 0 : output->written;
1565   result_value.dense_segments_written = dense_storage == nullptr ? 0 : dense_storage->written;
1566   return result_value;
1567 }

1595 template<method Method, ieee754_floating F, typename Rhs>
1596   requires(Method == method::euler || Method == method::midpoint || Method == method::rk4)
1597 [[nodiscard]] inline runtime_result<F>
1598 integrate_fixed(Rhs function, micron::__type_identity_t<F> t0, F *state, usize n, micron::__type_identity_t<F> t_bound,
1599                 micron::__type_identity_t<F> step, runtime_workspace<F> &workspace, usize max_steps = 100000) noexcept
1600 {
1601   runtime_result<F> result_value{};
1602   result_value.t = t0;
1603   constexpr usize required = Method == method::euler ? 2 : (Method == method::midpoint ? 3 : 5);
1604   if ( state == nullptr || step == F(0) || max_steps == 0 || !workspace.valid(n, required) || !ieee::is_finite<F>(step)
1605        || !ieee::is_finite<F>(t0) || !ieee::is_finite<F>(t_bound) || !__impl::runtime_finite(state, n) ) {
1606     result_value.termination = status::invalid_input;
1607     return result_value;
1608   }
1609   if ( t0 == t_bound ) return result_value;
1610   const F direction = t_bound > t0 ? F(1) : F(-1);
1611   const F step_abs = mk::manip::fabs<F>(step);
1612   auto stage = [&](usize s) noexcept -> F * { return workspace.stages + s * n; };
1613   while ( direction * (result_value.t - t_bound) < F(0) ) {
1614     if ( result_value.attempted_steps >= max_steps ) {
1615       result_value.termination = status::max_steps;
1616       break;
1617     }
1618     F h = direction * step_abs;
1619     if ( direction * (result_value.t + h - t_bound) > F(0) ) h = t_bound - result_value.t;
1620     if ( result_value.t + h == result_value.t ) {
1621       result_value.termination = status::step_underflow;
1622       break;
1623     }
1624     __impl::runtime_rhs(function, result_value.t, state, stage(0), n);
1625     if constexpr ( Method == method::euler ) {
1626       for ( usize i = 0; i < n; ++i ) workspace.next[i] = state[i] + h * stage(0)[i];
1627       __impl::runtime_rhs(function, result_value.t + h, workspace.next, stage(1), n);
1628       result_value.n_evals += 2;
1629     } else if constexpr ( Method == method::midpoint ) {
1630       for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = state[i] + F(0.5) * h * stage(0)[i];
1631       __impl::runtime_rhs(function, result_value.t + F(0.5) * h, workspace.temporary, stage(1), n);
1632       for ( usize i = 0; i < n; ++i ) workspace.next[i] = state[i] + h * stage(1)[i];
1633       __impl::runtime_rhs(function, result_value.t + h, workspace.next, stage(2), n);
1634       result_value.n_evals += 3;
1635     } else {
1636       for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = state[i] + F(0.5) * h * stage(0)[i];
1637       __impl::runtime_rhs(function, result_value.t + F(0.5) * h, workspace.temporary, stage(1), n);
1638       for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = state[i] + F(0.5) * h * stage(1)[i];
1639       __impl::runtime_rhs(function, result_value.t + F(0.5) * h, workspace.temporary, stage(2), n);
1640       for ( usize i = 0; i < n; ++i ) workspace.temporary[i] = state[i] + h * stage(2)[i];
1641       __impl::runtime_rhs(function, result_value.t + h, workspace.temporary, stage(3), n);
1642       for ( usize i = 0; i < n; ++i )
1643         workspace.next[i] = state[i] + h * (stage(0)[i] + F(2) * stage(1)[i] + F(2) * stage(2)[i] + stage(3)[i]) / F(6);
1644       __impl::runtime_rhs(function, result_value.t + h, workspace.next, stage(4), n);
1645       result_value.n_evals += 5;
1646     }
1647     const usize endpoint = Method == method::euler ? 1 : (Method == method::midpoint ? 2 : 4);
1648     if ( !__impl::runtime_finite(workspace.next, n) || !__impl::runtime_finite(stage(endpoint), n) ) {
1649       result_value.termination = status::non_finite;
1650       break;
1651     }
1652     for ( usize i = 0; i < n; ++i ) {
1653       workspace.start[i] = state[i];
1654       workspace.start_derivative[i] = stage(0)[i];
1655       workspace.end_derivative[i] = stage(endpoint)[i];
1656       state[i] = workspace.next[i];
1657     }
1658     result_value.t += h;
1659     result_value.last_step = h;
1660     ++result_value.accepted_steps;
1661     ++result_value.attempted_steps;
1662   }
1663   return result_value;
1664 }
