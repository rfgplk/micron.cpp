32 template<ieee754_floating F>
33 inline void
34 merge(quad_result<F> &into, const quad_result<F> &part) noexcept
35 {
36   into.value += part.value;
37   into.abs_err += part.abs_err;
38   into.n_evals += part.n_evals;
39   into.resabs += part.resabs;
40   into.resasc += part.resasc;
41   into.n_intervals += part.n_intervals;
42   into.n_splits += part.n_splits;
43   into.n_roundoff += part.n_roundoff;
44   if ( into.status == quad_status::ok && part.status != quad_status::ok ) into.status = part.status;
45 }

47 template<ieee754_floating F, typename Fn>
48 [[nodiscard]] inline quad_result<F>
49 tanh_sinh_finite(Fn f, F a, F b, const tanh_sinh_options<F> &options) noexcept
50 {
51   using work_type = micron::conditional_t<(sizeof(F) < sizeof(f64)), f64, F>;
52   quad_result<F> result{};
53   if ( a == b ) return result;
54   if ( b < a ) {
55     quad_result<F> r = tanh_sinh_finite<F>(f, b, a, options);
56     r.value = -r.value;
57     return r;
58   }
59   if ( options.max_levels == 0 || options.max_levels >= sizeof(usize) * 8 || options.max_terms == 0 || options.abs_tol < F(0)
60        || options.rel_tol < F(0) ) {
61     result.status = quad_status::invalid_input;
62     return result;
63   }
64 
65   const work_type midpoint = work_type(0.5) * (work_type(a) + work_type(b));
66   const work_type halfwidth = work_type(0.5) * (work_type(b) - work_type(a));
67   const work_type pi2 = work_type(0.5) * constant_pi<work_type>;
68   work_type previous = work_type(0);
69   bool have_previous = false;
70 
71   for ( usize level = 0; level < options.max_levels; ++level ) {
72     const work_type h = work_type(1) / work_type(usize(1) << level);
73     const usize refinement = level < sizeof(usize) * 8 - 1 ? (usize(1) << level) : usize(-1) / options.max_terms;
74     const usize term_limit = options.max_terms <= usize(-1) / refinement ? options.max_terms * refinement : usize(-1);
75     work_type sum = work_type(0);
76     usize quiet_tail = 0;
77     for ( usize k = 0; k < term_limit; ++k ) {
78       const work_type t = work_type(k) * h;
79       const work_type sh = mk::hyp::sinh<work_type>(t);
80       const work_type ch = mk::hyp::cosh<work_type>(t);
81       const work_type u = pi2 * sh;
82       const work_type z = mk::hyp::tanh<work_type>(u);
83       const work_type cu = mk::hyp::cosh<work_type>(u);
84       if ( !ieee::is_finite<work_type>(cu) || z >= work_type(1) ) break;
85       const work_type weight = pi2 * ch / (cu * cu);
86       work_type term{};
87       if ( k == 0 ) {
88         const F value = f(F(midpoint));
89         ++result.n_evals;
90         if ( !ieee::is_finite<F>(value) ) {
91           result.status = quad_status::non_finite;
92           return result;
93         }
94         term = static_cast<work_type>(weight) * static_cast<work_type>(value);
95       } else {
96         const work_type endpoint_distance = halfwidth * (work_type(1) - z);
97         F xp = F(work_type(b) - endpoint_distance);
98         F xm = F(work_type(a) + endpoint_distance);
99         if ( xp >= b ) xp = mk::manip::nextafter<F>(b, a);
100         if ( xm <= a ) xm = mk::manip::nextafter<F>(a, b);
101         const F fp = f(xp);
102         const F fm = f(xm);
103         result.n_evals += 2;
104         if ( !ieee::is_finite<F>(fp) || !ieee::is_finite<F>(fm) ) {
105           result.status = quad_status::non_finite;
106           return result;
107         }
108         term = static_cast<work_type>(weight) * static_cast<work_type>(fp + fm);
109       }
110       sum += term;
111       if ( k > 4 && mk::manip::fabs<work_type>(h * halfwidth * term) <= work_type(options.abs_tol) * work_type(0.001) )
112         ++quiet_tail;
113       else
114         quiet_tail = 0;
115       if ( quiet_tail == 3 ) break;
116       if ( k + 1 == term_limit ) result.status = quad_status::max_evaluations;
117     }
118 
119     const work_type current = halfwidth * h * sum;
120     if ( have_previous ) {
121       const work_type delta = mk::manip::fabs<work_type>(current - previous);
122       const F error = F(work_type(16) * delta);
123       result.value = F(current);
124       result.abs_err = error;
125       result.resabs = mk::manip::fabs<F>(result.value);
126       result.n_intervals = level + 1;
127       const F rel = options.rel_tol * mk::manip::fabs<F>(result.value);
128       const F tolerance = options.abs_tol > rel ? options.abs_tol : rel;
129       if ( error <= tolerance ) {
130         result.status = quad_status::ok;
131         return result;
132       }
133     }
134     previous = current;
135     have_previous = true;
136   }
137 
138   result.value = F(previous);
139   result.resabs = mk::manip::fabs<F>(result.value);
140   result.n_intervals = options.max_levels;
141   if ( result.status == quad_status::ok ) result.status = quad_status::max_depth;
142   return result;
143 }

147 template<ieee754_floating F, callable_real<F> Fn>
148 [[nodiscard]] inline quad_result<F>
149 tanh_sinh(Fn f, F a, F b, const tanh_sinh_options<F> &options = {}) noexcept
150 {
151   const int ai = ieee::inf_sign<F>(a);
152   const int bi = ieee::inf_sign<F>(b);
153   if ( ai == 0 && bi == 0 ) return __impl_weighted::tanh_sinh_finite<F>(f, a, b, options);
154   if ( (ai == 1 && bi == 0) || (ai == 0 && bi == -1) || (ai == 1 && bi == -1) ) {
155     quad_result<F> r = tanh_sinh<F>(f, b, a, options);
156     r.value = -r.value;
157     return r;
158   }
159   if ( ai == 0 && bi == 1 ) {
160     auto g = [&](F t) noexcept -> F {
161       const F q = F(1) - t;
162       return f(a + t / q) / (q * q);
163     };
164     return __impl_weighted::tanh_sinh_finite<F>(g, F(0), F(1), options);
165   }
166   if ( ai == -1 && bi == 0 ) {
167     auto g = [&](F t) noexcept -> F {
168       const F q = F(1) - t;
169       return f(b - t / q) / (q * q);
170     };
171     return __impl_weighted::tanh_sinh_finite<F>(g, F(0), F(1), options);
172   }
173   if ( ai == -1 && bi == 1 ) {
174     auto g = [&](F t) noexcept -> F {
175       const F angle = constant_pi<F> * (t - F(0.5));
176       const F c = mk::trig::cos<F>(angle);
177       return f(mk::trig::tan<F>(angle)) * constant_pi<F> / (c * c);
178     };
179     return __impl_weighted::tanh_sinh_finite<F>(g, F(0), F(1), options);
180   }
181   quad_result<F> result{};
182   result.status = quad_status::invalid_input;
183   return result;
184 }

235 template<ieee754_floating F, callable_real<F> Fn, usize Capacity>
236 [[nodiscard]] inline quad_result<F>
237 quad_cauchy(Fn f, F a, F b, F pole, const quad_options<F> &options, quad_workspace<F, Capacity> &workspace) noexcept
238 {
239   if ( b < a ) {
240     quad_result<F> r = quad_cauchy<F>(f, b, a, pole, options, workspace);
241     r.value = -r.value;
242     return r;
243   }
244   if ( pole <= a || pole >= b ) {
245     auto weighted = [&](F x) noexcept -> F { return f(x) / (x - pole); };
246     return quad<F>(weighted, a, b, options, workspace);
247   }
248 
249   const F dl = pole - a;
250   const F dr = b - pole;
251   const F symmetric = dl < dr ? dl : dr;
252   auto cancelled = [&](F t) noexcept -> F { return (f(pole + t) - f(pole - t)) / t; };
253   quad_result<F> result = quad<F>(cancelled, F(0), symmetric, options, workspace);
254   if ( dl > symmetric ) {
255     auto left = [&](F x) noexcept -> F { return f(x) / (x - pole); };
256     quad_result<F> part = quad<F>(left, a, pole - symmetric, options, workspace);
257     __impl_weighted::merge(result, part);
258   }
259   if ( dr > symmetric ) {
260     auto right = [&](F x) noexcept -> F { return f(x) / (x - pole); };
261     quad_result<F> part = quad<F>(right, pole + symmetric, b, options, workspace);
262     __impl_weighted::merge(result, part);
263   }
264   return result;
265 }

267 template<ieee754_floating F, callable_real<F> Fn>
268 [[nodiscard]] inline quad_result<F>
269 quad_algebraic(Fn f, F a, F b, F alpha, F beta, const tanh_sinh_options<F> &options = {}) noexcept
270 {
271   if ( alpha <= F(-1) || beta <= F(-1) ) {
272     quad_result<F> result{};
273     result.status = quad_status::invalid_input;
274     return result;
275   }
276   auto weighted = [&](F x) noexcept -> F { return f(x) * mk::pow_ns::pow<F>(x - a, alpha) * mk::pow_ns::pow<F>(b - x, beta); };
277   return tanh_sinh<F>(weighted, a, b, options);
278 }

280 template<ieee754_floating F, callable_real<F> Fn>
281 [[nodiscard]] inline quad_result<F>
282 quad_algebraic_log(Fn f, F a, F b, F alpha, F beta, bool log_left, bool log_right, const tanh_sinh_options<F> &options = {}) noexcept
283 {
284   if ( alpha <= F(-1) || beta <= F(-1) ) {
285     quad_result<F> result{};
286     result.status = quad_status::invalid_input;
287     return result;
288   }
289   auto weighted = [&](F x) noexcept -> F {
290     F w = mk::pow_ns::pow<F>(x - a, alpha) * mk::pow_ns::pow<F>(b - x, beta);
291     if ( log_left ) w *= mk::log_ns::log<F>(x - a);
292     if ( log_right ) w *= mk::log_ns::log<F>(b - x);
293     return f(x) * w;
294   };
295   return tanh_sinh<F>(weighted, a, b, options);
296 }
