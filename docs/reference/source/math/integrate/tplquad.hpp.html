26 template<ieee754_floating F, typename Fn>
27 [[nodiscard]] inline quad_result<F>
28 tplquad(Fn f, F ax, F bx, F ay, F by, F az, F bz, F atol, F rtol, usize max_depth = 30) noexcept
29 {
30   quad_result<F> total{};
31   const F third_atol = atol / F(3);
32   const F third_rtol = rtol / F(3);
33   auto outer_fn = [&](F x) noexcept -> F {
34     auto inner_fn = [&](F y, F z) noexcept -> F { return f(x, y, z); };
35     quad_result<F> inner = dblquad<F>(inner_fn, ay, by, az, bz, third_atol, third_rtol, max_depth);
36     total.n_evals += inner.n_evals;
37     if ( inner.status != quad_status::ok ) total.status = inner.status;
38     return inner.value;
39   };
40   quad_result<F> outer = quad<F>(outer_fn, ax, bx, third_atol, third_rtol, max_depth);
41   total.value = outer.value;
42   total.abs_err = outer.abs_err;
43   total.n_evals += outer.n_evals;
44   if ( outer.status != quad_status::ok ) total.status = outer.status;
45   return total;
46 }
