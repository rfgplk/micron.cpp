47 template<ieee754_floating F>
48 inline void
49 cum_trapezoid(const F *xs, const F *ys, F *out, usize n) noexcept
50 {
51   if ( n == 0 ) return;
52   F y_left = ys[0];
53   out[0] = F(0);
54   for ( usize i = 1; i < n; ++i ) {
55     const F y_right = ys[i];
56     const F dx = xs[i] - xs[i - 1];
57     out[i] = math::fma<F>(F(0.5) * dx, y_right + y_left, out[i - 1]);
58     y_left = y_right;
59   }
60 }

62 template<ieee754_floating F>
63 inline void
64 cum_trapezoid(const F *ys, F *out, usize n, F dx) noexcept
65 {
66   if ( ys != out ) {
67     __integrate_arch::cum_trapezoid_uniform(ys, out, n, dx);
68     return;
69   }
70   if ( n == 0 ) return;
71   F y_left = ys[0];
72   out[0] = F(0);
73   for ( usize i = 1; i < n; ++i ) {
74     const F y_right = ys[i];
75     out[i] = math::fma<F>(F(0.5) * dx, y_right + y_left, out[i - 1]);
76     y_left = y_right;
77   }
78 }

101 template<ieee754_floating F>
102 [[nodiscard, gnu::always_inline]] inline F
103 irregular_simpson_panel(F x0, F x1, F x2, F y0, F y1, F y2) noexcept
104 {
105   const F h0 = x1 - x0;
106   const F h1 = x2 - x1;
107   if ( h0 == F(0) || h1 == F(0) ) return F(0);
108   const F h = h0 + h1;
109   return (h / F(6)) * ((F(2) - h1 / h0) * y0 + (h * h / (h0 * h1)) * y1 + (F(2) - h0 / h1) * y2);
110 }

114 template<ieee754_floating F, accumulation_policy Policy = accumulation_policy::fast>
115 [[nodiscard]] inline F
116 simpson(const F *xs, const F *ys, usize n) noexcept
117 {
118   if ( n < 2 ) return F(0);
119   __impl_accumulate::sum<F, Policy> total{};
120   usize i = 0;
121   for ( ; i + 2 < n; i += 2 )
122     total.add(__impl_samples::irregular_simpson_panel<F>(xs[i], xs[i + 1], xs[i + 2], ys[i], ys[i + 1], ys[i + 2]));
123   if ( i + 1 < n ) total.add(F(0.5) * (xs[i + 1] - xs[i]) * (ys[i] + ys[i + 1]));
124   return total.get();
125 }

136 template<ieee754_floating F>
137 inline void
138 cum_simpson(const F *xs, const F *ys, F *out, usize n) noexcept
139 {
140   if ( n == 0 ) return;
141   out[0] = F(0);
142   if ( n == 1 ) return;
143 
144   F y0 = ys[0];
145   F y1 = ys[1];
146   out[1] = F(0.5) * (xs[1] - xs[0]) * (y0 + y1);
147   usize i = 2;
148   for ( ; i < n; i += 2 ) {
149     const F y2 = ys[i];
150     out[i] = out[i - 2] + __impl_samples::irregular_simpson_panel<F>(xs[i - 2], xs[i - 1], xs[i], y0, y1, y2);
151     if ( i + 1 < n ) {
152       const F y3 = ys[i + 1];
153       out[i + 1] = out[i] + F(0.5) * (xs[i + 1] - xs[i]) * (y2 + y3);
154       y0 = y2;
155       y1 = y3;
156     }
157   }
158 }

160 template<ieee754_floating F>
161 inline void
162 cum_simpson(const F *ys, F *out, usize n, F dx) noexcept
163 {
164   if ( n == 0 ) return;
165   out[0] = F(0);
166   if ( n == 1 ) return;
167 
168   F y0 = ys[0];
169   F y1 = ys[1];
170   out[1] = F(0.5) * dx * (y0 + y1);
171   usize i = 2;
172   for ( ; i < n; i += 2 ) {
173     const F y2 = ys[i];
174     out[i] = out[i - 2] + (dx / F(3)) * (y0 + F(4) * y1 + y2);
175     if ( i + 1 < n ) {
176       const F y3 = ys[i + 1];
177       out[i + 1] = out[i] + F(0.5) * dx * (y2 + y3);
178       y0 = y2;
179       y1 = y3;
180     }
181   }
182 }

203 template<ieee754_floating F>
204 [[nodiscard]] inline F
205 sampled_romberg(const F *y, usize n, F dx) noexcept
206 {
207   if ( n < 2 ) return F(0);
208   const usize panels = n - 1;
209   if ( (panels & (panels - 1)) != 0 ) return F(0);
210 
211   constexpr usize cap = sizeof(usize) * 8;
212   F row[cap]{};
213   usize levels = 1;
214   for ( usize p = panels; p > 1; p >>= 1 ) ++levels;
215 
216   for ( usize level = 0; level < levels; ++level ) {
217     const usize stride = panels >> level;
218     const usize count = usize(1) << level;
219     F t = F(0.5) * (y[0] + y[panels]);
220     for ( usize j = 1; j < count; ++j ) t += y[j * stride];
221     row[level] = dx * F(stride) * t;
222     F factor = F(4);
223     for ( usize j = level; j > 0; --j ) {
224       row[j - 1] = row[j] + (row[j] - row[j - 1]) / (factor - F(1));
225       factor *= F(4);
226     }
227   }
228   return row[0];
229 }
