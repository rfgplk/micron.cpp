55 template<ieee754_floating F>
56 [[nodiscard, gnu::always_inline]] inline F
57 halton(usize i, u32 base) noexcept
58 {
59   F factor = F(1);
60   F value = F(0);
61   while ( i > 0 ) {
62     factor /= F(base);
63     value += factor * F(i % base);
64     i /= base;
65   }
66   return value;
67 }

69 [[nodiscard, gnu::always_inline]] inline constexpr u64
70 mix64(u64 value) noexcept
71 {
72   value += 0x9e3779b97f4a7c15ULL;
73   value = (value ^ (value >> 30)) * 0xbf58476d1ce4e5b9ULL;
74   value = (value ^ (value >> 27)) * 0x94d049bb133111ebULL;
75   return value ^ (value >> 31);
76 }

88 [[nodiscard]] inline constexpr u32
89 gcd(u32 a, u32 b) noexcept
90 {
91   while ( b != 0 ) {
92     const u32 r = a % b;
93     a = b;
94     b = r;
95   }
96   return a;
97 }

99 template<ieee754_floating F>
100 [[nodiscard]] inline F
101 scrambled_halton_coordinate(usize index, u32 base, u64 scramble) noexcept
102 {
103   const u32 multiplier = u32(scramble % (base - 1)) + 1;
104   const u32 shift = u32((scramble >> 32) % base);
105   F factor = F(1);
106   F value = F(0);
107   while ( index > 0 ) {
108     factor /= F(base);
109     const u32 digit = u32(index % base);
110     value += factor * F((multiplier * digit + shift) % base);
111     index /= base;
112   }
113   return value + F(shift) * factor / F(base - 1);
114 }

129 template<usize D>
130 inline void
131 sobol_directions(u64 (&directions)[D][64]) noexcept
132 {
133   for ( usize bit = 0; bit < 64; ++bit ) directions[0][bit] = u64(1) << (63 - bit);
134   for ( usize dimension = 1; dimension < D; ++dimension ) {
135     const sobol_parameter &parameter = sobol_parameters[dimension - 1];
136     const usize degree = parameter.degree;
137     for ( usize bit = 0; bit < degree; ++bit ) directions[dimension][bit] = u64(parameter.initial[bit]) << (63 - bit);
138     for ( usize bit = degree; bit < 64; ++bit ) {
139       u64 value = directions[dimension][bit - degree] ^ (directions[dimension][bit - degree] >> degree);
140       for ( usize k = 1; k < degree; ++k ) {
141         if ( (parameter.polynomial >> (degree - 1 - k)) & 1U ) value ^= directions[dimension][bit - k];
142       }
143       directions[dimension][bit] = value;
144     }
145   }
146 }

149 [[nodiscard]] inline F
150 volume(const F (&lo)[D], const F (&hi)[D]) noexcept
151 {
152   F result = F(1);
153   for ( usize d = 0; d < D; ++d ) result *= hi[d] - lo[d];
154   return result;

158 [[nodiscard]] inline bool
159 valid_bounds(const F (&lo)[D], const F (&hi)[D]) noexcept
160 {
161   for ( usize d = 0; d < D; ++d )

166 template<ieee754_floating F>
167 inline void
168 welford_add(F x, usize n, F &mean, F &m2) noexcept
169 {
170   const F delta = x - mean;
171   mean += delta / F(n);
172   m2 += delta * (x - mean);
173 }

175 template<ieee754_floating F>
176 inline void
177 finish(monte_carlo_result<F> &result, F mean, F m2, usize count, F volume) noexcept
178 {
179   result.estimate = volume * mean;
180   if ( count > 1 ) result.variance = volume * volume * m2 / F(count - 1);
181   result.standard_error = count > 0 ? mk::pow_ns::sqrt<F>(result.variance / F(count)) : F(0);
182 }

186 template<usize D, ieee754_floating F, typename Fn, rng::rng_concept Rng>
187   requires(D >= 1 && D <= 16)
188 [[nodiscard]] inline monte_carlo_result<F>
189 monte_carlo_detailed(Fn f, const F (&lo)[D], const F (&hi)[D], usize n_samples, Rng &generator) noexcept
190 {
191   monte_carlo_result<F> result{};
192   if ( n_samples == 0 ) {
193     result.status = monte_carlo_status::no_samples;
194     return result;
195   }
196   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
197     result.status = monte_carlo_status::invalid_input;
198     return result;
199   }
200   F mean = F(0), m2 = F(0), point[D]{};
201   for ( usize sample = 0; sample < n_samples; ++sample ) {
202     for ( usize d = 0; d < D; ++d ) {
203       const F unit = rng::dist::uniform_real<F>(generator);
204       point[d] = math::fma<F>(unit, hi[d] - lo[d], lo[d]);
205     }
206     const F value = f(point);
207     ++result.evaluations;
208     if ( !ieee::is_finite<F>(value) ) {
209       result.status = monte_carlo_status::non_finite;
210       return result;
211     }
212     __impl_monte_carlo::welford_add(value, sample + 1, mean, m2);
213   }
214   __impl_monte_carlo::finish(result, mean, m2, n_samples, __impl_monte_carlo::volume(lo, hi));
215   return result;
216 }

232 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, rng::rng_concept Rng, usize BatchSize>
233   requires(D >= 1 && D <= 16)
234 [[nodiscard]] inline monte_carlo_result<F>
235 monte_carlo_batch_detailed(Fn f, const F (&lo)[D], const F (&hi)[D], usize n_samples, Rng &generator,
236                            monte_carlo_batch_workspace<D, F, BatchSize> &workspace) noexcept
237 {
238   monte_carlo_result<F> result{};
239   if ( n_samples == 0 ) {
240     result.status = monte_carlo_status::no_samples;
241     return result;
242   }
243   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
244     result.status = monte_carlo_status::invalid_input;
245     return result;
246   }
247   for ( usize d = 0; d < D; ++d ) workspace.coordinate_views[d] = workspace.coordinates[d];
248   F mean = F(0), m2 = F(0);
249   usize completed = 0;
250   while ( completed < n_samples ) {
251     const usize remaining = n_samples - completed;
252     const usize count = remaining < BatchSize ? remaining : BatchSize;
253     for ( usize sample = 0; sample < count; ++sample )
254       for ( usize d = 0; d < D; ++d ) workspace.coordinates[d][sample] = rng::dist::uniform_real<F>(generator);
255     for ( usize d = 0; d < D; ++d )
256       __integrate_arch::affine_transform(workspace.coordinates[d], workspace.coordinates[d], count, hi[d] - lo[d], lo[d]);
257     f(workspace.coordinate_views, workspace.values, count);
258     for ( usize sample = 0; sample < count; ++sample ) {
259       const F value = workspace.values[sample];
260       ++result.evaluations;
261       if ( !ieee::is_finite<F>(value) ) {
262         result.status = monte_carlo_status::non_finite;
263         return result;
264       }
265       __impl_monte_carlo::welford_add(value, completed + sample + 1, mean, m2);
266     }
267     completed += count;
268   }
269   __impl_monte_carlo::finish(result, mean, m2, n_samples, __impl_monte_carlo::volume(lo, hi));
270   return result;
271 }

281 template<usize D, ieee754_floating F, typename Fn, rng::rng_concept Rng>
282   requires(D >= 1 && D <= 16)
283 [[nodiscard]] inline monte_carlo_result<F>
284 antithetic_monte_carlo(Fn f, const F (&lo)[D], const F (&hi)[D], usize n_samples, Rng &generator) noexcept
285 {
286   monte_carlo_result<F> result{};
287   if ( n_samples == 0 ) {
288     result.status = monte_carlo_status::no_samples;
289     return result;
290   }
291   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
292     result.status = monte_carlo_status::invalid_input;
293     return result;
294   }
295   F mean = F(0), m2 = F(0), point[D]{}, antithetic[D]{};
296   usize observations = 0;
297   while ( result.evaluations < n_samples ) {
298     for ( usize d = 0; d < D; ++d ) {
299       const F unit = rng::dist::uniform_real<F>(generator);
300       point[d] = math::fma<F>(unit, hi[d] - lo[d], lo[d]);
301       antithetic[d] = math::fma<F>(F(1) - unit, hi[d] - lo[d], lo[d]);
302     }
303     const F first = f(point);
304     ++result.evaluations;
305     F observation = first;
306     if ( result.evaluations < n_samples ) {
307       const F second = f(antithetic);
308       ++result.evaluations;
309       if ( !ieee::is_finite<F>(second) ) {
310         result.status = monte_carlo_status::non_finite;
311         return result;
312       }
313       observation = F(0.5) * (first + second);
314     }
315     if ( !ieee::is_finite<F>(observation) ) {
316       result.status = monte_carlo_status::non_finite;
317       return result;
318     }
319     __impl_monte_carlo::welford_add(observation, ++observations, mean, m2);
320   }
321   __impl_monte_carlo::finish(result, mean, m2, observations, __impl_monte_carlo::volume(lo, hi));
322   return result;
323 }

325 template<usize D, ieee754_floating F, typename Fn, rng::rng_concept Rng>
326   requires(D >= 1 && D <= 16)
327 [[nodiscard]] inline monte_carlo_result<F>
328 stratified_monte_carlo(Fn f, const F (&lo)[D], const F (&hi)[D], usize n_samples, Rng &generator) noexcept
329 {
330   monte_carlo_result<F> result{};
331   if ( n_samples == 0 ) {
332     result.status = monte_carlo_status::no_samples;
333     return result;
334   }
335   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
336     result.status = monte_carlo_status::invalid_input;
337     return result;
338   }
339   u32 multiplier[D]{}, shift[D]{};
340   for ( usize d = 0; d < D; ++d ) {
341     u32 candidate = u32(generator.next() % n_samples);
342     if ( candidate == 0 ) candidate = 1;
343     while ( __impl_monte_carlo::gcd(candidate, u32(n_samples)) != 1 ) {
344       ++candidate;
345       if ( candidate >= n_samples ) candidate = 1;
346     }
347     multiplier[d] = candidate;
348     shift[d] = u32(generator.next() % n_samples);
349   }
350   F mean = F(0), m2 = F(0), point[D]{};
351   for ( usize sample = 0; sample < n_samples; ++sample ) {
352     for ( usize d = 0; d < D; ++d ) {
353       const usize stratum = (usize(multiplier[d]) * sample + shift[d]) % n_samples;
354       const F unit = (F(stratum) + rng::dist::uniform_real<F>(generator)) / F(n_samples);
355       point[d] = math::fma<F>(unit, hi[d] - lo[d], lo[d]);
356     }
357     const F value = f(point);
358     ++result.evaluations;
359     if ( !ieee::is_finite<F>(value) ) {
360       result.status = monte_carlo_status::non_finite;
361       return result;
362     }
363     __impl_monte_carlo::welford_add(value, sample + 1, mean, m2);
364   }
365   __impl_monte_carlo::finish(result, mean, m2, n_samples, __impl_monte_carlo::volume(lo, hi));
366   return result;
367 }

369 template<usize D, ieee754_floating F, typename Fn>
370   requires(D >= 1 && D <= 16)
371 [[nodiscard]] inline F
372 quasi_monte_carlo(Fn f, const F (&lo)[D], const F (&hi)[D], usize n_samples) noexcept
373 {
374   if ( n_samples == 0 || !__impl_monte_carlo::valid_bounds(lo, hi) ) return F(0);
375   F sum = F(0), point[D]{};
376   for ( usize sample = 1; sample <= n_samples; ++sample ) {
377     for ( usize d = 0; d < D; ++d ) {
378       const F unit = __impl_monte_carlo::halton<F>(sample, __impl_monte_carlo::halton_primes[d]);
379       point[d] = math::fma<F>(unit, hi[d] - lo[d], lo[d]);
380     }
381     sum += f(point);
382   }
383   return __impl_monte_carlo::volume(lo, hi) * sum / F(n_samples);
384 }

388 [[nodiscard]] inline monte_carlo_result<F>
389 scrambled_halton(Fn f, const F (&lo)[D], const F (&hi)[D], usize samples_per_replica, usize replicas = 8,
390                  u64 seed = 0x8b8b8b8b243f6a88ULL) noexcept

399     return result;
400   }
401   const F volume = __impl_monte_carlo::volume(lo, hi);
402   F replica_mean = F(0), replica_m2 = F(0), point[D]{};

406       for ( usize d = 0; d < D; ++d ) {

418     }

420     __impl_monte_carlo::welford_add(estimate, replica + 1, replica_mean, replica_m2);
421   }
422   result.estimate = replica_mean;
423   if ( replicas > 1 ) result.variance = replica_m2 / F(replicas - 1);
424   result.standard_error = mk::pow_ns::sqrt<F>(result.variance / F(replicas));

482 template<usize D, ieee754_floating F, typename Fn>
483   requires(D >= 1 && D <= 16)
484 [[nodiscard]] inline monte_carlo_result<F>
485 scrambled_sobol(Fn f, const F (&lo)[D], const F (&hi)[D], usize samples_per_replica, usize replicas = 8,
486                 u64 seed = 0xd1b54a32d192ed03ULL) noexcept
487 {
488   monte_carlo_result<F> result{};
489   if ( samples_per_replica == 0 || replicas == 0 ) {
490     result.status = monte_carlo_status::no_samples;
491     return result;
492   }
493   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
494     result.status = monte_carlo_status::invalid_input;
495     return result;
496   }
497   u64 directions[D][64]{};
498   __impl_monte_carlo::sobol_directions<D>(directions);
499   const F volume = __impl_monte_carlo::volume(lo, hi);
500   F replica_mean = F(0), replica_m2 = F(0), point[D]{};
501   for ( usize replica = 0; replica < replicas; ++replica ) {
502     u64 state[D]{};
503     F sum = F(0);
504     for ( usize sample = 0; sample < samples_per_replica; ++sample ) {
505       if ( sample != 0 ) {
506         const usize bit = usize(__builtin_ctzll(u64(sample)));
507         for ( usize d = 0; d < D; ++d ) state[d] ^= directions[d][bit];
508       }
509       for ( usize d = 0; d < D; ++d ) {
510         const u64 shift = __impl_monte_carlo::mix64(seed ^ (u64(replica + 1) << 32) ^ u64(d + 1));
511         const F unit = __impl_monte_carlo::unit_from_u64<F>(state[d] ^ shift);
512         point[d] = math::fma<F>(unit, hi[d] - lo[d], lo[d]);
513       }
514       const F value = f(point);
515       ++result.evaluations;
516       if ( !ieee::is_finite<F>(value) ) {
517         result.status = monte_carlo_status::non_finite;
518         return result;
519       }
520       sum += value;
521     }
522     const F estimate = volume * sum / F(samples_per_replica);
523     __impl_monte_carlo::welford_add(estimate, replica + 1, replica_mean, replica_m2);
524   }
525   result.estimate = replica_mean;
526   if ( replicas > 1 ) result.variance = replica_m2 / F(replicas - 1);
527   result.standard_error = mk::pow_ns::sqrt<F>(result.variance / F(replicas));
528   return result;
529 }
530 
531 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, usize BatchSize>
532   requires(D >= 1 && D <= 16)
533 [[nodiscard]] inline monte_carlo_result<F>
534 scrambled_sobol_batch(Fn f, const F (&lo)[D], const F (&hi)[D], usize samples_per_replica, usize replicas, u64 seed,
535                       monte_carlo_batch_workspace<D, F, BatchSize> &workspace) noexcept
536 {
537   monte_carlo_result<F> result{};
538   if ( samples_per_replica == 0 || replicas == 0 ) {
539     result.status = monte_carlo_status::no_samples;
540     return result;
541   }
542   if ( !__impl_monte_carlo::valid_bounds(lo, hi) ) {
543     result.status = monte_carlo_status::invalid_input;
544     return result;
545   }
546   for ( usize d = 0; d < D; ++d ) workspace.coordinate_views[d] = workspace.coordinates[d];
547   u64 directions[D][64]{};
548   __impl_monte_carlo::sobol_directions<D>(directions);
549   const F volume = __impl_monte_carlo::volume(lo, hi);
550   F replica_mean = F(0), replica_m2 = F(0);
551   for ( usize replica = 0; replica < replicas; ++replica ) {
552     u64 state[D]{};
553     F sum = F(0);
554     usize completed = 0;
555     while ( completed < samples_per_replica ) {
556       const usize remaining = samples_per_replica - completed;
557       const usize count = remaining < BatchSize ? remaining : BatchSize;
558       for ( usize sample = 0; sample < count; ++sample ) {
559         const usize index = completed + sample;
560         if ( index != 0 ) {
561           const usize bit = usize(__builtin_ctzll(u64(index)));
562           for ( usize d = 0; d < D; ++d ) state[d] ^= directions[d][bit];
563         }
564         for ( usize d = 0; d < D; ++d ) {
565           const u64 shift = __impl_monte_carlo::mix64(seed ^ (u64(replica + 1) << 32) ^ u64(d + 1));
566           workspace.coordinates[d][sample] = __impl_monte_carlo::unit_from_u64<F>(state[d] ^ shift);
567         }
568       }
569       for ( usize d = 0; d < D; ++d )
570         __integrate_arch::affine_transform(workspace.coordinates[d], workspace.coordinates[d], count, hi[d] - lo[d], lo[d]);
571       f(workspace.coordinate_views, workspace.values, count);
572       for ( usize sample = 0; sample < count; ++sample ) {
573         if ( !ieee::is_finite<F>(workspace.values[sample]) ) {
574           result.status = monte_carlo_status::non_finite;
575           return result;
576         }
577         sum += workspace.values[sample];
578       }
579       result.evaluations += count;
580       completed += count;
581     }
582     const F estimate = volume * sum / F(samples_per_replica);
583     __impl_monte_carlo::welford_add(estimate, replica + 1, replica_mean, replica_m2);
584   }
585   result.estimate = replica_mean;
586   if ( replicas > 1 ) result.variance = replica_m2 / F(replicas - 1);
587   result.standard_error = mk::pow_ns::sqrt<F>(result.variance / F(replicas));
588   return result;
589 }
