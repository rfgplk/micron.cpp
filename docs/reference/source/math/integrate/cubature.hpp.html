105 template<usize D, ieee754_floating F, typename Fn>
106 [[nodiscard]] inline bool
107 evaluate(Fn &f, cubature_region<D, F> &region, usize &n_evals) noexcept
108 {
109   constexpr F lambda2 = F(0.358568582800318091990645153907937L);
110   constexpr F lambda4 = F(0.948683298050513799599668063329816L);
111   constexpr F lambda5 = F(0.688247201611685297721628734293623L);
112   constexpr F weight2 = F(980.0L / 6561.0L);
113   constexpr F weight4 = F(200.0L / 19683.0L);
114   constexpr F weight_e2 = F(245.0L / 486.0L);
115   constexpr F weight_e4 = F(25.0L / 729.0L);
116   constexpr F ratio = F((9.0L / 70.0L) / (9.0L / 10.0L));
117 
118   F point[D]{};
119   for ( usize d = 0; d < D; ++d ) point[d] = region.center[d];
120   const F center_value = f(point);
121   ++n_evals;
122   if ( !ieee::is_finite<F>(center_value) ) return false;
123 
124   F sum2 = F(0), sum3 = F(0), sum4 = F(0), sum5 = F(0);
125   F difference[D]{};
126   for ( usize d = 0; d < D; ++d ) {
127     const F delta2 = lambda2 * region.halfwidth[d];
128     point[d] = region.center[d] - delta2;
129     const F m2 = f(point);
130     point[d] = region.center[d] + delta2;
131     const F p2 = f(point);
132     const F delta4 = lambda4 * region.halfwidth[d];
133     point[d] = region.center[d] - delta4;
134     const F m4 = f(point);
135     point[d] = region.center[d] + delta4;
136     const F p4 = f(point);
137     point[d] = region.center[d];
138     n_evals += 4;
139     if ( !ieee::is_finite<F>(m2) || !ieee::is_finite<F>(p2) || !ieee::is_finite<F>(m4) || !ieee::is_finite<F>(p4) ) return false;
140     sum2 += m2 + p2;
141     sum3 += m4 + p4;
142     difference[d] = mk::manip::fabs<F>(m2 + p2 - F(2) * center_value - ratio * (m4 + p4 - F(2) * center_value));
143   }
144 
145   for ( usize i = 0; i + 1 < D; ++i ) {
146     for ( usize j = i + 1; j < D; ++j ) {
147       const F di = lambda4 * region.halfwidth[i];
148       const F dj = lambda4 * region.halfwidth[j];
149       for ( usize signs = 0; signs < 4; ++signs ) {
150         point[i] = region.center[i] + ((signs & 1) ? di : -di);
151         point[j] = region.center[j] + ((signs & 2) ? dj : -dj);
152         const F value = f(point);
153         ++n_evals;
154         if ( !ieee::is_finite<F>(value) ) return false;
155         sum4 += value;
156       }
157       point[i] = region.center[i];
158       point[j] = region.center[j];
159     }
160   }
161 
162   for ( usize mask = 0; mask < (usize(1) << D); ++mask ) {
163     for ( usize d = 0; d < D; ++d ) {
164       const F delta = lambda5 * region.halfwidth[d];
165       point[d] = region.center[d] + ((mask & (usize(1) << d)) ? delta : -delta);
166     }
167     const F value = f(point);
168     ++n_evals;
169     if ( !ieee::is_finite<F>(value) ) return false;
170     sum5 += value;
171   }
172 
173   F volume = F(1);
174   for ( usize d = 0; d < D; ++d ) volume *= F(2) * region.halfwidth[d];
175   const F dd = F(D);
176   const F weight1 = (F(12824) - F(9120) * dd + F(400) * dd * dd) / F(19683);
177   const F weight3 = (F(1820) - F(400) * dd) / F(19683);
178   const F weight5 = F(6859) / F(19683) / F(usize(1) << D);
179   const F weight_e1 = (F(729) - F(950) * dd + F(50) * dd * dd) / F(729);
180   const F weight_e3 = (F(265) - F(100) * dd) / F(1458);
181   const F seventh = volume * (weight1 * center_value + weight2 * sum2 + weight3 * sum3 + weight4 * sum4 + weight5 * sum5);
182   const F fifth = volume * (weight_e1 * center_value + weight_e2 * sum2 + weight_e3 * sum3 + weight_e4 * sum4);
183   region.value = seventh;
184   region.error = mk::manip::fabs<F>(seventh - fifth);
185   const F floor = F(50) * machine_epsilon<F>() * mk::manip::fabs<F>(seventh);
186   if ( region.error < floor ) region.error = floor;
187 
188   region.split_dimension = 0;
189   for ( usize d = 1; d < D; ++d ) {
190     if ( difference[d] > difference[region.split_dimension]
191          || (difference[d] == difference[region.split_dimension] && region.halfwidth[d] > region.halfwidth[region.split_dimension]) )
192       region.split_dimension = d;
193   }
194   return true;
195 }

197 template<usize D, ieee754_floating F, typename Fn, usize Capacity, usize BatchSize>
198 [[nodiscard]] inline bool
199 evaluate_batch(Fn &f, cubature_region<D, F> &region, usize &n_evals,
200                cubature_batch_workspace<D, F, Capacity, BatchSize> &workspace) noexcept
201 {
202   constexpr F lambda2 = F(0.358568582800318091990645153907937L);
203   constexpr F lambda4 = F(0.948683298050513799599668063329816L);
204   constexpr F lambda5 = F(0.688247201611685297721628734293623L);
205   constexpr F weight2 = F(980.0L / 6561.0L);
206   constexpr F weight4 = F(200.0L / 19683.0L);
207   constexpr F weight_e2 = F(245.0L / 486.0L);
208   constexpr F weight_e4 = F(25.0L / 729.0L);
209   constexpr F ratio = F((9.0L / 70.0L) / (9.0L / 10.0L));
210   constexpr usize axis_count = 1 + 4 * D;
211 
212   for ( usize d = 0; d < D; ++d ) workspace.coordinate_views[d] = workspace.coordinates[d];
213   F point[D]{};
214   for ( usize d = 0; d < D; ++d ) point[d] = region.center[d];
215   usize queued = 0;
216   auto emit = [&](const F(&p)[D]) noexcept {
217     for ( usize d = 0; d < D; ++d ) workspace.coordinates[d][queued] = p[d];
218     ++queued;
219   };
220   auto evaluate_queued = [&](F *capture, usize &captured, F &sum) noexcept -> bool {
221     if ( queued == 0 ) return true;
222     f(workspace.coordinate_views, workspace.values, queued);
223     n_evals += queued;
224     for ( usize i = 0; i < queued; ++i ) {
225       if ( !ieee::is_finite<F>(workspace.values[i]) ) return false;
226       if ( capture != nullptr )
227         capture[captured++] = workspace.values[i];
228       else
229         sum += workspace.values[i];
230     }
231     queued = 0;
232     return true;
233   };
234 
235   F axis_values[axis_count]{};
236   usize captured = 0;
237   F ignored = F(0);
238   emit(point);
239   if ( queued == BatchSize && !evaluate_queued(axis_values, captured, ignored) ) return false;
240   for ( usize d = 0; d < D; ++d ) {
241     const F delta2 = lambda2 * region.halfwidth[d];
242     point[d] = region.center[d] - delta2;
243     emit(point);
244     if ( queued == BatchSize && !evaluate_queued(axis_values, captured, ignored) ) return false;
245     point[d] = region.center[d] + delta2;
246     emit(point);
247     if ( queued == BatchSize && !evaluate_queued(axis_values, captured, ignored) ) return false;
248     const F delta4 = lambda4 * region.halfwidth[d];
249     point[d] = region.center[d] - delta4;
250     emit(point);
251     if ( queued == BatchSize && !evaluate_queued(axis_values, captured, ignored) ) return false;
252     point[d] = region.center[d] + delta4;
253     emit(point);
254     if ( queued == BatchSize && !evaluate_queued(axis_values, captured, ignored) ) return false;
255     point[d] = region.center[d];
256   }
257   if ( !evaluate_queued(axis_values, captured, ignored) ) return false;
258 
259   const F center_value = axis_values[0];
260   F sum2 = F(0), sum3 = F(0), sum4 = F(0), sum5 = F(0);
261   F difference[D]{};
262   for ( usize d = 0; d < D; ++d ) {
263     const usize base = 1 + 4 * d;
264     const F m2 = axis_values[base], p2 = axis_values[base + 1];
265     const F m4 = axis_values[base + 2], p4 = axis_values[base + 3];
266     sum2 += m2 + p2;
267     sum3 += m4 + p4;
268     difference[d] = mk::manip::fabs<F>(m2 + p2 - F(2) * center_value - ratio * (m4 + p4 - F(2) * center_value));
269   }
270 
271   for ( usize i = 0; i + 1 < D; ++i ) {
272     for ( usize j = i + 1; j < D; ++j ) {
273       const F di = lambda4 * region.halfwidth[i];
274       const F dj = lambda4 * region.halfwidth[j];
275       for ( usize signs = 0; signs < 4; ++signs ) {
276         point[i] = region.center[i] + ((signs & 1) ? di : -di);
277         point[j] = region.center[j] + ((signs & 2) ? dj : -dj);
278         emit(point);
279         if ( queued == BatchSize && !evaluate_queued(nullptr, captured, sum4) ) return false;
280       }
281       point[i] = region.center[i];
282       point[j] = region.center[j];
283     }
284   }
285   if ( !evaluate_queued(nullptr, captured, sum4) ) return false;
286 
287   for ( usize mask = 0; mask < (usize(1) << D); ++mask ) {
288     for ( usize d = 0; d < D; ++d ) {
289       const F delta = lambda5 * region.halfwidth[d];
290       point[d] = region.center[d] + ((mask & (usize(1) << d)) ? delta : -delta);
291     }
292     emit(point);
293     if ( queued == BatchSize && !evaluate_queued(nullptr, captured, sum5) ) return false;
294   }
295   if ( !evaluate_queued(nullptr, captured, sum5) ) return false;
296 
297   F volume = F(1);
298   for ( usize d = 0; d < D; ++d ) volume *= F(2) * region.halfwidth[d];
299   const F dd = F(D);
300   const F weight1 = (F(12824) - F(9120) * dd + F(400) * dd * dd) / F(19683);
301   const F weight3 = (F(1820) - F(400) * dd) / F(19683);
302   const F weight5 = F(6859) / F(19683) / F(usize(1) << D);
303   const F weight_e1 = (F(729) - F(950) * dd + F(50) * dd * dd) / F(729);
304   const F weight_e3 = (F(265) - F(100) * dd) / F(1458);
305   const F seventh = volume * (weight1 * center_value + weight2 * sum2 + weight3 * sum3 + weight4 * sum4 + weight5 * sum5);
306   const F fifth = volume * (weight_e1 * center_value + weight_e2 * sum2 + weight_e3 * sum3 + weight_e4 * sum4);
307   region.value = seventh;
308   region.error = mk::manip::fabs<F>(seventh - fifth);
309   const F floor = F(50) * machine_epsilon<F>() * mk::manip::fabs<F>(seventh);
310   if ( region.error < floor ) region.error = floor;
311   region.split_dimension = 0;
312   for ( usize d = 1; d < D; ++d ) {
313     if ( difference[d] > difference[region.split_dimension]
314          || (difference[d] == difference[region.split_dimension] && region.halfwidth[d] > region.halfwidth[region.split_dimension]) )
315       region.split_dimension = d;
316   }
317   return true;
318 }

320 template<usize D, ieee754_floating F, usize Capacity>
321 inline void
322 push(cubature_workspace<D, F, Capacity> &workspace, const cubature_region<D, F> &region) noexcept
323 {
324   usize i = workspace.size++;
325   while ( i != 0 ) {
326     const usize parent = (i - 1) >> 1;
327     if ( workspace.heap[parent].error >= region.error ) break;
328     workspace.heap[i] = workspace.heap[parent];
329     i = parent;
330   }
331   workspace.heap[i] = region;
332 }

334 template<usize D, ieee754_floating F, usize Capacity>
335 [[nodiscard]] inline cubature_region<D, F>
336 pop(cubature_workspace<D, F, Capacity> &workspace) noexcept
337 {
338   const cubature_region<D, F> result = workspace.heap[0];
339   const cubature_region<D, F> tail = workspace.heap[--workspace.size];
340   if ( workspace.size == 0 ) return result;
341   usize i = 0;
342   while ( true ) {
343     const usize left = i * 2 + 1;
344     if ( left >= workspace.size ) break;
345     const usize right = left + 1;
346     const usize child = right < workspace.size && workspace.heap[right].error > workspace.heap[left].error ? right : left;
347     if ( workspace.heap[child].error <= tail.error ) break;
348     workspace.heap[i] = workspace.heap[child];
349     i = child;
350   }
351   workspace.heap[i] = tail;
352   return result;
353 }

392 template<usize D, ieee754_floating F, callable_real_d<F, D> Fn, usize Capacity>
393   requires(D >= 2 && D <= 16)
394 [[nodiscard]] inline cubature_result<F>
395 cubature(Fn f, const F (&lower)[D], const F (&upper)[D], const cubature_options<F> &options,
396          cubature_workspace<D, F, Capacity> &workspace) noexcept
397 {
398   cubature_result<F> result{};
399   workspace.reset();
400   if ( options.abs_tol < F(0) || options.rel_tol < F(0) ) {
401     result.status = quad_status::invalid_input;
402     return result;
403   }
404   const usize requested_limit = options.max_regions == 0 ? Capacity : options.max_regions;
405   const usize region_limit = requested_limit < Capacity ? requested_limit : Capacity;
406   constexpr usize eval_cost = __impl_cubature::point_count<D>;
407   if ( options.max_evals != 0 && options.max_evals < eval_cost ) {
408     result.status = quad_status::max_evaluations;
409     return result;
410   }
411 
412   cubature_region<D, F> initial{};
413   F orientation = F(1);
414   for ( usize d = 0; d < D; ++d ) {
415     F lo = lower[d];
416     F hi = upper[d];
417     if ( !ieee::is_finite<F>(lo) || !ieee::is_finite<F>(hi) ) {
418       result.status = quad_status::invalid_input;
419       return result;
420     }
421     if ( hi < lo ) {
422       const F swap = lo;
423       lo = hi;
424       hi = swap;
425       orientation = -orientation;
426     }
427     if ( lo == hi ) return result;
428     initial.center[d] = F(0.5) * (lo + hi);
429     initial.halfwidth[d] = F(0.5) * (hi - lo);
430   }
431   if ( !__impl_cubature::evaluate<D, F>(f, initial, result.n_evals) ) {
432     result.status = quad_status::non_finite;
433     return result;
434   }
435   __impl_cubature::push(workspace, initial);
436   __impl_accumulate::runtime_sum<F> value_sum{ options.accumulation };
437   __impl_accumulate::runtime_sum<F> error_sum{ options.accumulation };
438   value_sum.add(initial.value);
439   error_sum.add(initial.error);
440   result.n_regions = 1;
441 
442   while ( true ) {
443     const F value = value_sum.get();
444     F error = error_sum.get();
445     if ( error < F(0) ) error = F(0);
446     const F tolerance
447         = options.abs_tol > options.rel_tol * mk::manip::fabs<F>(value) ? options.abs_tol : options.rel_tol * mk::manip::fabs<F>(value);
448     if ( error <= tolerance ) break;
449     if ( workspace.size >= region_limit ) {
450       result.status = quad_status::max_intervals;
451       break;
452     }
453     if ( options.max_evals != 0 && result.n_evals + 2 * eval_cost > options.max_evals ) {
454       result.status = quad_status::max_evaluations;
455       break;
456     }
457     const cubature_region<D, F> parent = __impl_cubature::pop(workspace);
458     cubature_region<D, F> left = parent;
459     cubature_region<D, F> right = parent;
460     const usize split = parent.split_dimension;
461     left.halfwidth[split] *= F(0.5);
462     right.halfwidth[split] = left.halfwidth[split];
463     left.center[split] -= left.halfwidth[split];
464     right.center[split] += right.halfwidth[split];
465     if ( !__impl_cubature::evaluate<D, F>(f, left, result.n_evals) || !__impl_cubature::evaluate<D, F>(f, right, result.n_evals) ) {
466       __impl_cubature::push(workspace, parent);
467       result.status = quad_status::non_finite;
468       break;
469     }
470     value_sum.add(-parent.value);
471     value_sum.add(left.value);
472     value_sum.add(right.value);
473     error_sum.add(-parent.error);
474     error_sum.add(left.error);
475     error_sum.add(right.error);
476     __impl_cubature::push(workspace, left);
477     __impl_cubature::push(workspace, right);
478     ++result.n_splits;
479     result.n_regions = workspace.size;
480   }
481   result.value = orientation * value_sum.get();
482   result.abs_err = error_sum.get();
483   if ( result.abs_err < F(0) ) result.abs_err = F(0);
484   return result;
485 }

499 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, usize Capacity>
500   requires(D >= 2 && D <= 16)
501 [[nodiscard]] inline cubature_result<F>
502 cubature_batch(Fn f, const F (&lower)[D], const F (&upper)[D], const cubature_options<F> &options,
503                cubature_workspace<D, F, Capacity> &workspace) noexcept
504 {
505   auto scalar = [&](const F(&point)[D]) noexcept -> F {
506     const F *coordinates[D];
507     for ( usize d = 0; d < D; ++d ) coordinates[d] = point + d;
508     F value{};
509     f(coordinates, &value, 1);
510     return value;
511   };
512   return cubature<D, F>(scalar, lower, upper, options, workspace);

515 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, usize Capacity, usize BatchSize>
516   requires(D >= 2 && D <= 16)
517 [[nodiscard]] inline cubature_result<F>
518 cubature_batch(Fn f, const F (&lower)[D], const F (&upper)[D], const cubature_options<F> &options,
519                cubature_batch_workspace<D, F, Capacity, BatchSize> &workspace) noexcept
520 {
521   cubature_result<F> result{};
522   workspace.reset();
523   if ( options.abs_tol < F(0) || options.rel_tol < F(0) ) {
524     result.status = quad_status::invalid_input;
525     return result;
526   }
527   const usize requested_limit = options.max_regions == 0 ? Capacity : options.max_regions;
528   const usize region_limit = requested_limit < Capacity ? requested_limit : Capacity;
529   constexpr usize eval_cost = __impl_cubature::point_count<D>;
530   if ( options.max_evals != 0 && options.max_evals < eval_cost ) {
531     result.status = quad_status::max_evaluations;
532     return result;
533   }
534 
535   cubature_region<D, F> initial{};
536   F orientation = F(1);
537   for ( usize d = 0; d < D; ++d ) {
538     F lo = lower[d];
539     F hi = upper[d];
540     if ( !ieee::is_finite<F>(lo) || !ieee::is_finite<F>(hi) ) {
541       result.status = quad_status::invalid_input;
542       return result;
543     }
544     if ( hi < lo ) {
545       const F swap = lo;
546       lo = hi;
547       hi = swap;
548       orientation = -orientation;
549     }
550     if ( lo == hi ) return result;
551     initial.center[d] = F(0.5) * (lo + hi);
552     initial.halfwidth[d] = F(0.5) * (hi - lo);
553   }
554   if ( !__impl_cubature::evaluate_batch<D, F>(f, initial, result.n_evals, workspace) ) {
555     result.status = quad_status::non_finite;
556     return result;
557   }
558   __impl_cubature::push(workspace, initial);
559   __impl_accumulate::runtime_sum<F> value_sum{ options.accumulation };
560   __impl_accumulate::runtime_sum<F> error_sum{ options.accumulation };
561   value_sum.add(initial.value);
562   error_sum.add(initial.error);
563   result.n_regions = 1;
564 
565   while ( true ) {
566     const F value = value_sum.get();
567     F error = error_sum.get();
568     if ( error < F(0) ) error = F(0);
569     const F tolerance
570         = options.abs_tol > options.rel_tol * mk::manip::fabs<F>(value) ? options.abs_tol : options.rel_tol * mk::manip::fabs<F>(value);
571     if ( error <= tolerance ) break;
572     if ( workspace.size >= region_limit ) {
573       result.status = quad_status::max_intervals;
574       break;
575     }
576     if ( options.max_evals != 0 && result.n_evals + 2 * eval_cost > options.max_evals ) {
577       result.status = quad_status::max_evaluations;
578       break;
579     }
580     const cubature_region<D, F> parent = __impl_cubature::pop(workspace);
581     cubature_region<D, F> left = parent;
582     cubature_region<D, F> right = parent;
583     const usize split = parent.split_dimension;
584     left.halfwidth[split] *= F(0.5);
585     right.halfwidth[split] = left.halfwidth[split];
586     left.center[split] -= left.halfwidth[split];
587     right.center[split] += right.halfwidth[split];
588     if ( !__impl_cubature::evaluate_batch<D, F>(f, left, result.n_evals, workspace)
589          || !__impl_cubature::evaluate_batch<D, F>(f, right, result.n_evals, workspace) ) {
590       __impl_cubature::push(workspace, parent);
591       result.status = quad_status::non_finite;
592       break;
593     }
594     value_sum.add(-parent.value);
595     value_sum.add(left.value);
596     value_sum.add(right.value);
597     error_sum.add(-parent.error);
598     error_sum.add(left.error);
599     error_sum.add(right.error);
600     __impl_cubature::push(workspace, left);
601     __impl_cubature::push(workspace, right);
602     ++result.n_splits;
603     result.n_regions = workspace.size;
604   }
605   result.value = orientation * value_sum.get();
606   result.abs_err = error_sum.get();
607   if ( result.abs_err < F(0) ) result.abs_err = F(0);
608   return result;
609 }
