25 template<usize D, usize Fixed, ieee754_floating F, typename Fn>
26 inline quad_result<F>
27 nquad_impl(Fn &f, F (&x)[D], const F (&lo)[D], const F (&hi)[D], F atol, F rtol, usize max_depth, quad_result<F> &acc) noexcept
28 {
29   if constexpr ( Fixed == D ) {
30     quad_result<F> r{};
31     r.value = f(x);
32     return r;
33   } else {
34     auto outer_fn = [&](F xi) noexcept -> F {
35       x[Fixed] = xi;
36       if constexpr ( Fixed + 1 == D ) {
37         return f(x);
38       } else {
39         quad_result<F> inner = nquad_impl<D, Fixed + 1, F>(f, x, lo, hi, atol, rtol, max_depth, acc);
40         acc.n_evals += inner.n_evals;
41         if ( inner.status != quad_status::ok ) acc.status = inner.status;
42         return inner.value;
43       }
44     };
45     return quad<F>(outer_fn, lo[Fixed], hi[Fixed], atol, rtol, max_depth);
46   }
47 }

51 template<usize D, ieee754_floating F, typename Fn>
52   requires(D >= 1)
53 [[nodiscard]] inline quad_result<F>
54 nquad(Fn f, const F (&lo)[D], const F (&hi)[D], F atol, F rtol, usize max_depth = 30) noexcept
55 {
56   quad_result<F> total{};
57   const F per_atol = atol / F(D);
58   const F per_rtol = rtol / F(D);
59 
60   if constexpr ( D == 1 ) {
61     auto fn1 = [&](F xi) noexcept -> F {
62       F xs[1] = { xi };
63       return f(xs);
64     };
65     return quad<F>(fn1, lo[0], hi[0], atol, rtol, max_depth);
66   } else {
67     F x[D]{};
68     quad_result<F> q = __impl_nquad::nquad_impl<D, 0, F>(f, x, lo, hi, per_atol, per_rtol, max_depth, total);
69     total.value = q.value;
70     total.abs_err = q.abs_err;
71     total.n_evals += q.n_evals;
72     if ( q.status != quad_status::ok ) total.status = q.status;
73     return total;
74   }
75 }
