26 template<ieee754_floating F, typename Fn>
27 [[nodiscard]] inline quad_result<F>
28 dblquad(Fn f, F ax, F bx, F ay, F by, F atol, F rtol, usize max_depth = 30) noexcept
29 {
30   quad_result<F> outer{};
31   const F half_atol = F(0.5) * atol;
32   const F half_rtol = F(0.5) * rtol;
33   auto outer_fn = [&](F x) noexcept -> F {
34     auto inner_fn = [&](F y) noexcept -> F { return f(x, y); };
35     quad_result<F> inner = quad<F>(inner_fn, ay, by, half_atol, half_rtol, max_depth);
36     outer.n_evals += inner.n_evals;
37     if ( inner.status != quad_status::ok ) outer.status = inner.status;
38     return inner.value;
39   };
40   quad_result<F> outer_q = quad<F>(outer_fn, ax, bx, half_atol, half_rtol, max_depth);
41   outer.value = outer_q.value;
42   outer.abs_err = outer_q.abs_err;
43   outer.n_evals += outer_q.n_evals;
44   if ( outer_q.status != quad_status::ok ) outer.status = outer_q.status;
45   return outer;
46 }

48 template<ieee754_floating F, typename Fn, typename YL, typename YU>
49 [[nodiscard]] inline quad_result<F>
50 dblquad(Fn f, F ax, F bx, YL yl, YU yu, F atol, F rtol, usize max_depth = 30) noexcept
51 {
52   quad_result<F> outer{};
53   const F half_atol = F(0.5) * atol;
54   const F half_rtol = F(0.5) * rtol;
55   auto outer_fn = [&](F x) noexcept -> F {
56     const F a = yl(x);
57     const F b = yu(x);
58     auto inner_fn = [&](F y) noexcept -> F { return f(x, y); };
59     quad_result<F> inner = quad<F>(inner_fn, a, b, half_atol, half_rtol, max_depth);
60     outer.n_evals += inner.n_evals;
61     if ( inner.status != quad_status::ok ) outer.status = inner.status;
62     return inner.value;
63   };
64   quad_result<F> outer_q = quad<F>(outer_fn, ax, bx, half_atol, half_rtol, max_depth);
65   outer.value = outer_q.value;
66   outer.abs_err = outer_q.abs_err;
67   outer.n_evals += outer_q.n_evals;
68   if ( outer_q.status != quad_status::ok ) outer.status = outer_q.status;
69   return outer;
70 }
