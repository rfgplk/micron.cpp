73 template<ieee754_floating F, usize MaxPoints, usize MaxOrder>
74 [[nodiscard]] inline bool
75 fornberg_weights(const F *nodes, usize count, F x0, usize derivative, F *weights,
76                  fornberg_workspace<F, MaxPoints, MaxOrder> &workspace) noexcept
77 {
78   if ( count == 0 || count > MaxPoints || derivative > MaxOrder || derivative >= count ) return false;
79   for ( usize i = 0; i < count; ++i )
80     for ( usize order = 0; order <= derivative; ++order ) workspace.coefficients[i][order] = F(0);
81   workspace.coefficients[0][0] = F(1);
82   F c1 = F(1);
83   F c4 = nodes[0] - x0;
84   for ( usize i = 1; i < count; ++i ) {
85     const usize highest = i < derivative ? i : derivative;
86     F c2 = F(1);
87     const F c5 = c4;
88     c4 = nodes[i] - x0;
89     for ( usize j = 0; j < i; ++j ) {
90       const F c3 = nodes[i] - nodes[j];
91       if ( c3 == F(0) ) return false;
92       c2 *= c3;
93       if ( j + 1 == i ) {
94         for ( usize order = highest; order > 0; --order )
95           workspace.coefficients[i][order]
96               = c1 * (F(order) * workspace.coefficients[i - 1][order - 1] - c5 * workspace.coefficients[i - 1][order]) / c2;
97         workspace.coefficients[i][0] = -c1 * c5 * workspace.coefficients[i - 1][0] / c2;
98       }
99       for ( usize order = highest; order > 0; --order )
100         workspace.coefficients[j][order] = (c4 * workspace.coefficients[j][order] - F(order) * workspace.coefficients[j][order - 1]) / c3;
101       workspace.coefficients[j][0] = c4 * workspace.coefficients[j][0] / c3;
102     }
103     c1 = c2;
104   }
105   for ( usize i = 0; i < count; ++i ) weights[i] = workspace.coefficients[i][derivative];
106   return true;
107 }

109 template<usize Derivative, usize Points, ieee754_floating F, callable_real<F> Fn>
110   requires(Derivative >= 1 && Derivative <= 12 && Points > Derivative && Points <= 25 && (Points % 2 == 1))
111 [[nodiscard]] inline F
112 central_nth(Fn f, F x, F h) noexcept
113 {
114   F nodes[Points]{};
115   F weights[Points]{};
116   constexpr usize radius = Points / 2;
117   for ( usize i = 0; i < Points; ++i ) nodes[i] = F(i) - F(radius);
118   fornberg_workspace<F, Points, Derivative> workspace{};
119   (void)fornberg_weights(nodes, Points, F(0), Derivative, weights, workspace);
120   F sum = F(0);
121   for ( usize i = 0; i < Points; ++i ) sum += weights[i] * f(math::fma<F>(nodes[i], h, x));
122   F scale = F(1);
123   for ( usize order = 0; order < Derivative; ++order ) scale *= h;
124   return sum / scale;
125 }

161 template<ieee754_floating F>
162 [[nodiscard, gnu::always_inline]] inline constexpr F
163 h_central(F x = F(0)) noexcept
164 {
165   const F ax = mk::manip::fabs<F>(x);
166   const F scale = ax > F(1) ? ax : F(1);
167   if constexpr ( sizeof(F) == sizeof(float) )
168     return F(1e-2L) * scale;
169   else if constexpr ( sizeof(F) == sizeof(double) )
170     return F(1e-5L) * scale;
171   else
172     return F(1e-6L) * scale;
173 }

204 [[nodiscard, gnu::always_inline]] inline constexpr F

210 // f'(x) ≈ (-f(x+2h) + 8 f(x+h) - 8 f(x-h) + f(x-2h)) / (12 h)
211 template<ieee754_floating F, callable_real<F> Fn>
212 [[nodiscard, gnu::always_inline]] inline constexpr F
213 central4(Fn f, F x, F h) noexcept
214 {
215   const F h2 = F(2) * h;
216   const F p2 = f(x + h2);
217   const F p1 = f(x + h);
218   const F m1 = f(x - h);
219   const F m2 = f(x - h2);
220   return (-p2 + F(8) * p1 - F(8) * m1 + m2) / (F(12) * h);
221 }

223 template<ieee754_floating F, callable_real<F> Fn>
224 [[nodiscard]] inline F
225 adaptive(Fn f, F x, F h0 = F(0), F tol = F(0), usize max_levels = 12) noexcept
226 {
227   if ( h0 == F(0) ) h0 = h_central<F>(x);
228   if ( tol == F(0) ) tol = math::default_eps<F>() * F(100);
229   auto step = [&](F h) noexcept -> F { return central<F>(f, x, h); };
230   return richardson::extrapolate<F>(step, h0, F(2), 2, max_levels, tol);
231 }

233 template<ieee754_floating F, callable_real<F> Fn>
234 [[nodiscard]] inline derivative_result<F>
235 adaptive_diagnostic(Fn f, F x, F h0 = F(0), F tol = F(0), usize max_levels = 12) noexcept
236 {
237   derivative_result<F> result{};
238   if ( h0 == F(0) ) h0 = h_central<F>(x);
239   if ( tol == F(0) ) tol = math::default_eps<F>() * F(100);
240   if ( h0 <= F(0) || tol < F(0) || max_levels == 0 ) {
241     result.status = derivative_status::invalid_input;
242     return result;
243   }
244   F h = h0;
245   F previous = central4<F>(f, x, h);
246   result.n_evals = 4;
247   for ( usize level = 1; level < max_levels; ++level ) {
248     h *= F(0.5);
249     const F current = central4<F>(f, x, h);
250     result.n_evals += 4;
251     result.value = current;
252     result.abs_err = mk::manip::fabs<F>(current - previous) / F(15);
253     result.step = h;
254     result.levels = level + 1;
255     if ( !ieee::is_finite<F>(current) ) {
256       result.status = derivative_status::non_finite;
257       return result;
258     }
259     if ( result.abs_err <= tol ) return result;
260     previous = current;
261   }
262   result.status = derivative_status::max_levels;
263   return result;
264 }

282 template<usize Derivative, usize Points, ieee754_floating F, callable_real_batch<F> Fn>
283   requires(Derivative >= 1 && Derivative <= 12 && Points > Derivative && Points <= 25 && (Points % 2 == 1))
284 [[nodiscard]] inline F
285 central_nth_batch(Fn f, F x, F h) noexcept
286 {
287   F offsets[Points]{};
288   F nodes[Points]{};
289   F values[Points]{};
290   F weights[Points]{};
291   constexpr usize radius = Points / 2;
292   for ( usize i = 0; i < Points; ++i ) {
293     offsets[i] = F(i) - F(radius);
294     nodes[i] = math::fma<F>(offsets[i], h, x);
295   }
296   fornberg_workspace<F, Points, Derivative> workspace{};
297   (void)fornberg_weights(offsets, Points, F(0), Derivative, weights, workspace);
298   f(nodes, values, Points);
299   const F sum = __integrate_arch::weighted_sum_fast(weights, values, Points);
300   F scale = F(1);
301   for ( usize order = 0; order < Derivative; ++order ) scale *= h;
302   return sum / scale;
303 }

305 template<usize D, ieee754_floating F, callable_real_d<F, D> Fn>
306   requires(D >= 2 and D <= 16)
307 [[nodiscard]] inline vec<F, D>
308 gradient(Fn f, const F (&x)[D], F h) noexcept
309 {
310   vec<F, D> g{};
311   F xp[D];
312   for ( usize d = 0; d < D; ++d ) xp[d] = x[d];
313   for ( usize d = 0; d < D; ++d ) {
314     xp[d] = x[d] + h;
315     const F fph = f(xp);
316     xp[d] = x[d] - h;
317     const F fmh = f(xp);
318     xp[d] = x[d];
319     g.data[d] = (fph - fmh) / (F(2) * h);
320   }
321   return g;
322 }

324 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, usize Capacity>
325   requires(D >= 1 && D <= 16 && Capacity >= 2 * D)
326 [[nodiscard]] inline vec<F, D>
327 gradient_batch(Fn f, const F (&x)[D], F h, derivative_batch_workspace<D, F, Capacity> &workspace) noexcept
328 {
329   vec<F, D> result{};
330   for ( usize d = 0; d < D; ++d ) workspace.coordinate_views[d] = workspace.coordinates[d];
331   for ( usize sample = 0; sample < 2 * D; ++sample )
332     for ( usize d = 0; d < D; ++d ) workspace.coordinates[d][sample] = x[d];
333   for ( usize d = 0; d < D; ++d ) {
334     workspace.coordinates[d][2 * d] += h;
335     workspace.coordinates[d][2 * d + 1] -= h;
336   }
337   f(workspace.coordinate_views, workspace.values, 2 * D);
338   const F scale = F(0.5) / h;
339   for ( usize d = 0; d < D; ++d ) result.data[d] = (workspace.values[2 * d] - workspace.values[2 * d + 1]) * scale;
340   return result;
341 }

344   requires(D >= 1 and D <= 16 and R >= 1 and R <= 16)
345 [[nodiscard]] inline mat<F, R, D>
346 jacobian(Fn f, const F (&x)[D], F h) noexcept
347 {
348   mat<F, R, D> J{};
349   F xp[D];
350   for ( usize d = 0; d < D; ++d ) xp[d] = x[d];
351   for ( usize d = 0; d < D; ++d ) {
352     xp[d] = x[d] + h;
353     auto fp = f(xp);
354     xp[d] = x[d] - h;
355     auto fm = f(xp);
356     xp[d] = x[d];

358   }
359   return J;

383 // (f(x+he_i+he_j) - f(x+he_i-he_j) - f(x-he_i+he_j) + f(x-he_i-he_j)) / (4 h**2)
384 template<usize D, ieee754_floating F, callable_real_d<F, D> Fn>
385   requires(D >= 1 and D <= 16)
386 [[nodiscard]] inline mat<F, D, D>
387 hessian(Fn f, const F (&x)[D], F h) noexcept
388 {
389   mat<F, D, D> H{};
390   F xp[D];
391   for ( usize d = 0; d < D; ++d ) xp[d] = x[d];
392   const F f0 = f(x);
393   const F h2 = h * h;
394 
395   // diagonal first
396   for ( usize i = 0; i < D; ++i ) {
397     xp[i] = x[i] + h;
398     const F fp = f(xp);
399     xp[i] = x[i] - h;
400     const F fm = f(xp);
401     xp[i] = x[i];
402     H.data[i * D + i] = (fp - F(2) * f0 + fm) / h2;
403   }
404   // off-diagonal
405   for ( usize i = 0; i < D; ++i ) {
406     for ( usize j = i + 1; j < D; ++j ) {
407       xp[i] = x[i] + h;
408       xp[j] = x[j] + h;
409       const F pp = f(xp);
410       xp[j] = x[j] - h;
411       const F pm = f(xp);
412       xp[i] = x[i] - h;
413       xp[j] = x[j] + h;
414       const F mp = f(xp);
415       xp[j] = x[j] - h;
416       const F mm = f(xp);
417       xp[i] = x[i];
418       xp[j] = x[j];
419       const F val = (pp - pm - mp + mm) / (F(4) * h2);
420       H.data[i * D + j] = val;
421       H.data[j * D + i] = val;
422     }
423   }
424   return H;
425 }

427 template<usize D, ieee754_floating F, callable_real_batch_d<F> Fn, usize Capacity>
428   requires(D >= 1 && D <= 16 && Capacity >= 1 + 2 * D * D)
429 [[nodiscard]] inline mat<F, D, D>
430 hessian_batch(Fn f, const F (&x)[D], F h, derivative_batch_workspace<D, F, Capacity> &workspace) noexcept
431 {
432   mat<F, D, D> result{};
433   constexpr usize count = 1 + 2 * D * D;
434   for ( usize d = 0; d < D; ++d ) workspace.coordinate_views[d] = workspace.coordinates[d];
435   for ( usize sample = 0; sample < count; ++sample )
436     for ( usize d = 0; d < D; ++d ) workspace.coordinates[d][sample] = x[d];
437 
438   usize sample = 1;
439   for ( usize d = 0; d < D; ++d ) {
440     workspace.coordinates[d][sample++] += h;
441     workspace.coordinates[d][sample++] -= h;
442   }
443   for ( usize i = 0; i < D; ++i ) {
444     for ( usize j = i + 1; j < D; ++j ) {
445       workspace.coordinates[i][sample] += h;
446       workspace.coordinates[j][sample++] += h;
447       workspace.coordinates[i][sample] += h;
448       workspace.coordinates[j][sample++] -= h;
449       workspace.coordinates[i][sample] -= h;
450       workspace.coordinates[j][sample++] += h;
451       workspace.coordinates[i][sample] -= h;
452       workspace.coordinates[j][sample++] -= h;
453     }
454   }
455   f(workspace.coordinate_views, workspace.values, count);
456   const F h2 = h * h;
457   sample = 1;
458   for ( usize d = 0; d < D; ++d ) {
459     result.data[d * D + d] = (workspace.values[sample] - F(2) * workspace.values[0] + workspace.values[sample + 1]) / h2;
460     sample += 2;
461   }
462   for ( usize i = 0; i < D; ++i ) {
463     for ( usize j = i + 1; j < D; ++j ) {
464       const F value
465           = (workspace.values[sample] - workspace.values[sample + 1] - workspace.values[sample + 2] + workspace.values[sample + 3])
466             / (F(4) * h2);
467       result.data[i * D + j] = value;
468       result.data[j * D + i] = value;
469       sample += 4;
470     }
471   }
472   return result;
473 }

475 template<ieee754_floating F>
476 inline void
477 diff(const F *y, F *dy, usize n, F dx) noexcept
478 {
479   if ( y != dy ) {
480     __integrate_arch::diff_uniform(y, dy, n, dx);
481     return;
482   }
483   if ( n == 0 ) return;
484   if ( n == 1 ) {
485     dy[0] = F(0);
486     return;
487   }
488   F previous = y[0];
489   F current = y[1];
490   dy[0] = (current - previous) / dx;
491   for ( usize i = 1; i + 1 < n; ++i ) {
492     const F next = y[i + 1];
493     dy[i] = (next - previous) / (F(2) * dx);
494     previous = current;
495     current = next;
496   }
497   dy[n - 1] = (current - previous) / dx;
498 }

500 // non-uniform spacing form
501 template<ieee754_floating F>
502 inline void
503 diff(const F *xs, const F *ys, F *dy, usize n) noexcept
504 {
505   if ( n == 0 ) return;
506   if ( n == 1 ) {
507     dy[0] = F(0);
508     return;
509   }
510   F y_previous = ys[0];
511   F y_current = ys[1];
512   dy[0] = (y_current - y_previous) / (xs[1] - xs[0]);
513   for ( usize i = 1; i + 1 < n; ++i ) {
514     const F y_next = ys[i + 1];
515     const F h_l = xs[i] - xs[i - 1];
516     const F h_r = xs[i + 1] - xs[i];
517     const F h_l2 = h_l * h_l;
518     const F h_r2 = h_r * h_r;
519     dy[i] = (-h_r2 * y_previous + (h_r2 - h_l2) * y_current + h_l2 * y_next) / (h_l * h_r * (h_l + h_r));
520     y_previous = y_current;
521     y_current = y_next;
522   }
523   dy[n - 1] = (y_current - y_previous) / (xs[n - 1] - xs[n - 2]);
524 }

527 inline void
528 gradient_irregular(const F *xs, const F *ys, F *gradient_out, usize n) noexcept
529 {
530   diff<F>(xs, ys, gradient_out, n);
531 }

533 template<ieee754_floating F>
534 inline void
535 diff2(const F *xs, const F *ys, F *d2y, usize n) noexcept
536 {
537   if ( n == 0 ) return;
538   if ( n < 3 ) {
539     for ( usize i = 0; i < n; ++i ) d2y[i] = F(0);
540     return;
541   }
542   {
543     const F h0 = xs[1] - xs[0];
544     const F h1 = xs[2] - xs[1];
545     d2y[0] = F(2) / (h0 + h1) * ((ys[2] - ys[1]) / h1 - (ys[1] - ys[0]) / h0);
546   }
547   for ( usize i = 1; i + 1 < n; ++i ) {
548     const F h_l = xs[i] - xs[i - 1];
549     const F h_r = xs[i + 1] - xs[i];
550     d2y[i] = F(2) / (h_l + h_r) * ((ys[i + 1] - ys[i]) / h_r - (ys[i] - ys[i - 1]) / h_l);
551   }
552   {
553     const F h_lm1 = xs[n - 2] - xs[n - 3];
554     const F h_l = xs[n - 1] - xs[n - 2];
555     d2y[n - 1] = F(2) / (h_lm1 + h_l) * ((ys[n - 1] - ys[n - 2]) / h_l - (ys[n - 2] - ys[n - 3]) / h_lm1);
556   }
557 }
