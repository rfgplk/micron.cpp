124 template<ieee754_floating F>
125 [[gnu::always_inline]] inline void
126 add(F x, accumulation_policy policy, F &sum, F &correction) noexcept
127 {
128   if ( policy == accumulation_policy::fast ) {
129     sum += x;
130     return;
131   }
132   const F next = sum + x;
133   if ( mk::manip::fabs<F>(sum) >= mk::manip::fabs<F>(x) )
134     correction += (sum - next) + x;
135   else
136     correction += (x - next) + sum;
137   sum = next;
138 }

141 [[nodiscard, gnu::always_inline]] inline F
142 total(F sum, F correction, accumulation_policy policy) noexcept
143 {
144   return policy == accumulation_policy::accurate ? sum + correction : sum;

147 template<ieee754_floating F, typename Fn, typename Table>
148 [[gnu::always_inline]] inline bool
149 gk_rule(Fn &f, F a, F b, quad_interval<F> &out, usize &n_evals) noexcept
150 {
151   const F half_w = F(0.5) * (b - a);
152   const F abs_half_w = mk::manip::fabs<F>(half_w);
153   const F mid = F(0.5) * (a + b);
154   const F fmid = f(mid);
155   ++n_evals;
156   if ( !ieee::is_finite<F>(fmid) ) return false;
157 
158   F fv1[Table::half]{};
159   F fv2[Table::half]{};
160   fv1[0] = fmid;
161   fv2[0] = fmid;
162   F resk = Table::wk[0] * fmid;
163   F resg = Table::wg[0] * fmid;
164   F resabs = Table::wk[0] * mk::manip::fabs<F>(fmid);
165 
166   for ( usize k = 1; k < Table::half; ++k ) {
167     const F absc = half_w * Table::nodes[k];
168     const F fm = f(mid - absc);
169     const F fp = f(mid + absc);
170     n_evals += 2;
171     if ( !ieee::is_finite<F>(fm) || !ieee::is_finite<F>(fp) ) return false;
172     fv1[k] = fm;
173     fv2[k] = fp;
174     const F pair = fm + fp;
175     resk = math::fma<F>(Table::wk[k], pair, resk);
176     if ( Table::wg[k] != F(0) ) resg = math::fma<F>(Table::wg[k], pair, resg);
177     resabs = math::fma<F>(Table::wk[k], mk::manip::fabs<F>(fm) + mk::manip::fabs<F>(fp), resabs);
178   }
179 
180   const F mean = F(0.5) * resk;
181   F resasc = Table::wk[0] * mk::manip::fabs<F>(fmid - mean);
182   for ( usize k = 1; k < Table::half; ++k )
183     resasc += Table::wk[k] * (mk::manip::fabs<F>(fv1[k] - mean) + mk::manip::fabs<F>(fv2[k] - mean));
184 
185   out.a = a;
186   out.b = b;
187   out.value = half_w * resk;
188   out.resabs = abs_half_w * resabs;
189   out.resasc = abs_half_w * resasc;
190   F error = mk::manip::fabs<F>(half_w * (resk - resg));
191   if ( out.resasc != F(0) && error != F(0) ) {
192     F ratio = F(200) * error / out.resasc;
193     if ( ratio < F(1) ) error = out.resasc * ratio * mk::pow_ns::sqrt<F>(ratio);
194   }
195   const F floor = F(50) * machine_epsilon<F>() * out.resabs;
196   out.error = error > floor ? error : floor;
197   return ieee::is_finite<F>(out.value) && ieee::is_finite<F>(out.error);
198 }

200 template<ieee754_floating F, typename Fn, typename Table>
201 [[gnu::always_inline]] inline bool
202 gk_rule_batch(Fn &f, F a, F b, quad_interval<F> &out, usize &n_evals) noexcept
203 {
204   constexpr usize count = Table::half * 2 - 1;
205   const F half_w = F(0.5) * (b - a);
206   const F abs_half_w = mk::manip::fabs<F>(half_w);
207   const F mid = F(0.5) * (a + b);
208   F points[count]{};
209   F values[count]{};
210   points[0] = mid;
211   for ( usize k = 1; k < Table::half; ++k ) {
212     const F absc = half_w * Table::nodes[k];
213     points[2 * k - 1] = mid - absc;
214     points[2 * k] = mid + absc;
215   }
216   f(points, values, count);
217   n_evals += count;
218   for ( usize i = 0; i < count; ++i )
219     if ( !ieee::is_finite<F>(values[i]) ) return false;
220 
221   const F fmid = values[0];
222   F resk = Table::wk[0] * fmid;
223   F resg = Table::wg[0] * fmid;
224   F resabs = Table::wk[0] * mk::manip::fabs<F>(fmid);
225   for ( usize k = 1; k < Table::half; ++k ) {
226     const F fm = values[2 * k - 1];
227     const F fp = values[2 * k];
228     const F pair = fm + fp;
229     resk = math::fma<F>(Table::wk[k], pair, resk);
230     if ( Table::wg[k] != F(0) ) resg = math::fma<F>(Table::wg[k], pair, resg);
231     resabs = math::fma<F>(Table::wk[k], mk::manip::fabs<F>(fm) + mk::manip::fabs<F>(fp), resabs);
232   }
233 
234   const F mean = F(0.5) * resk;
235   F resasc = Table::wk[0] * mk::manip::fabs<F>(fmid - mean);
236   for ( usize k = 1; k < Table::half; ++k )
237     resasc += Table::wk[k] * (mk::manip::fabs<F>(values[2 * k - 1] - mean) + mk::manip::fabs<F>(values[2 * k] - mean));
238 
239   out.a = a;
240   out.b = b;
241   out.value = half_w * resk;
242   out.resabs = abs_half_w * resabs;
243   out.resasc = abs_half_w * resasc;
244   F error = mk::manip::fabs<F>(half_w * (resk - resg));
245   if ( out.resasc != F(0) && error != F(0) ) {
246     const F ratio = F(200) * error / out.resasc;
247     if ( ratio < F(1) ) error = out.resasc * ratio * mk::pow_ns::sqrt<F>(ratio);
248   }
249   const F floor = F(50) * machine_epsilon<F>() * out.resabs;
250   out.error = error > floor ? error : floor;
251   return ieee::is_finite<F>(out.value) && ieee::is_finite<F>(out.error);
252 }

256 evaluate(Fn &f, F a, F b, gauss_kronrod_rule rule, quad_interval<F> &out, usize &n_evals) noexcept

270 template<ieee754_floating F, usize Capacity>
271 [[gnu::always_inline]] inline void
272 heap_push(quad_workspace<F, Capacity> &ws, const quad_interval<F> &v) noexcept
273 {
274   usize i = ws.size++;
275   while ( i != 0 ) {
276     const usize parent = (i - 1) >> 1;
277     if ( ws.heap[parent].error >= v.error ) break;
278     ws.heap[i] = ws.heap[parent];
279     i = parent;
280   }
281   ws.heap[i] = v;
282 }

284 template<ieee754_floating F, usize Capacity>
285 [[nodiscard, gnu::always_inline]] inline quad_interval<F>
286 heap_pop(quad_workspace<F, Capacity> &ws) noexcept
287 {
288   const quad_interval<F> result = ws.heap[0];
289   const quad_interval<F> tail = ws.heap[--ws.size];
290   if ( ws.size == 0 ) return result;
291   usize i = 0;
292   while ( true ) {
293     const usize left = i * 2 + 1;
294     if ( left >= ws.size ) break;
295     const usize right = left + 1;
296     const usize child = right < ws.size && ws.heap[right].error > ws.heap[left].error ? right : left;
297     if ( ws.heap[child].error <= tail.error ) break;
298     ws.heap[i] = ws.heap[child];
299     i = child;
300   }
301   ws.heap[i] = tail;
302   return result;
303 }

308 {

313 template<ieee754_floating F, typename Fn, usize Capacity, bool Batch = false>
314 [[nodiscard]] inline quad_result<F>
315 finite(Fn f, F a, F b, const quad_options<F> &options, quad_workspace<F, Capacity> &ws) noexcept
316 {
317   quad_result<F> result{};
318   ws.reset();
319   if ( a == b ) return result;
320   if ( options.abs_tol < F(0) || options.rel_tol < F(0) || options.max_depth == 0
321        || (options.n_breakpoints != 0 && options.breakpoints == nullptr) ) {
322     result.status = quad_status::invalid_input;
323     return result;
324   }
325   for ( usize i = 0; i < options.n_breakpoints; ++i ) {
326     if ( !ieee::is_finite<F>(options.breakpoints[i]) ) {
327       result.status = quad_status::invalid_input;
328       return result;
329     }
330   }
331 
332   const usize requested_limit = options.max_intervals == 0 ? Capacity : options.max_intervals;
333   const usize interval_limit = requested_limit < Capacity ? requested_limit : Capacity;
334   const usize eval_cost = options.rule == gauss_kronrod_rule::gk15 ? 15 : 21;
335   F value = F(0), value_c = F(0), error = F(0), error_c = F(0), resabs = F(0), resabs_c = F(0), resasc = F(0), resasc_c = F(0);
336 
337   F left = a;
338   usize initialized = 0;
339   while ( left < b ) {
340     F right = b;
341     for ( usize i = 0; i < options.n_breakpoints; ++i ) {
342       const F p = options.breakpoints[i];
343       if ( p > left && p < right && p < b ) right = p;
344     }
345     if ( initialized >= interval_limit ) {
346       result.status = quad_status::max_intervals;
347       break;
348     }
349     if ( options.max_evals != 0 && result.n_evals + eval_cost > options.max_evals ) {
350       result.status = quad_status::max_evaluations;
351       break;
352     }
353     quad_interval<F> initial{};
354     bool evaluated{};
355     if constexpr ( Batch )
356       evaluated = evaluate_batch<F>(f, left, right, options.rule, initial, result.n_evals);
357     else
358       evaluated = evaluate<F>(f, left, right, options.rule, initial, result.n_evals);
359     if ( !evaluated ) {
360       result.status = quad_status::non_finite;
361       ws.reset();
362       return result;
363     }
364     heap_push(ws, initial);
365     add(initial.value, options.accumulation, value, value_c);
366     add(initial.error, options.accumulation, error, error_c);
367     add(initial.resabs, options.accumulation, resabs, resabs_c);
368     add(initial.resasc, options.accumulation, resasc, resasc_c);
369     ++initialized;
370     left = right;
371   }
372 
373   if ( ws.size == 0 ) return result;
374   result.n_intervals = ws.size;
375   const F initial_error = total(error, error_c, options.accumulation);
376 
377   while ( result.status == quad_status::ok ) {
378     const F current_value = total(value, value_c, options.accumulation);
379     const F current_error = total(error, error_c, options.accumulation);
380     if ( current_error <= tolerance(options, current_value) ) break;
381     if ( ws.size >= interval_limit ) {
382       result.status = quad_status::max_intervals;
383       break;
384     }
385     if ( options.max_evals != 0 && result.n_evals + 2 * eval_cost > options.max_evals ) {
386       result.status = quad_status::max_evaluations;
387       break;
388     }
389 
390     const quad_interval<F> parent = heap_pop(ws);
391     if ( parent.depth >= options.max_depth ) {
392       heap_push(ws, parent);
393       result.status = quad_status::max_depth;
394       break;
395     }
396     const F mid = F(0.5) * (parent.a + parent.b);
397     if ( mid == parent.a || mid == parent.b ) {
398       heap_push(ws, parent);
399       result.status = quad_status::roundoff;
400       break;
401     }
402 
403     quad_interval<F> lo{}, hi{};
404     bool lo_ok{}, hi_ok{};
405     if constexpr ( Batch ) {
406       lo_ok = evaluate_batch<F>(f, parent.a, mid, options.rule, lo, result.n_evals);
407       hi_ok = evaluate_batch<F>(f, mid, parent.b, options.rule, hi, result.n_evals);
408     } else {
409       lo_ok = evaluate<F>(f, parent.a, mid, options.rule, lo, result.n_evals);
410       hi_ok = evaluate<F>(f, mid, parent.b, options.rule, hi, result.n_evals);
411     }
412     if ( !lo_ok || !hi_ok ) {
413       heap_push(ws, parent);
414       result.status = quad_status::non_finite;
415       break;
416     }
417     lo.depth = parent.depth + 1;
418     hi.depth = parent.depth + 1;
419 
420     const F child_value = lo.value + hi.value;
421     const F child_error = lo.error + hi.error;
422     const F scale = mk::manip::fabs<F>(parent.value) + F(1);
423     if ( mk::manip::fabs<F>(parent.value - child_value) <= F(100) * machine_epsilon<F>() * scale && child_error >= F(0.99) * parent.error )
424       ++result.n_roundoff;
425 
426     add(-parent.value, options.accumulation, value, value_c);
427     add(child_value, options.accumulation, value, value_c);
428     add(-parent.error, options.accumulation, error, error_c);
429     add(child_error, options.accumulation, error, error_c);
430     add(-parent.resabs, options.accumulation, resabs, resabs_c);
431     add(lo.resabs + hi.resabs, options.accumulation, resabs, resabs_c);
432     add(-parent.resasc, options.accumulation, resasc, resasc_c);
433     add(lo.resasc + hi.resasc, options.accumulation, resasc, resasc_c);
434     heap_push(ws, lo);
435     heap_push(ws, hi);
436     ++result.n_splits;
437     result.n_intervals = ws.size;
438 
439     if ( result.n_roundoff >= 20 ) result.status = quad_status::roundoff;
440     const F now_error = total(error, error_c, options.accumulation);
441     if ( result.n_splits > 50 && initial_error > F(0) && now_error > initial_error * F(1000000) ) result.status = quad_status::divergent;
442   }
443 
444   result.value = total(value, value_c, options.accumulation);
445   result.abs_err = total(error, error_c, options.accumulation);
446   result.resabs = total(resabs, resabs_c, options.accumulation);
447   result.resasc = total(resasc, resasc_c, options.accumulation);
448   if ( result.abs_err < F(0) ) result.abs_err = F(0);
449   if ( !ieee::is_finite<F>(result.value) || !ieee::is_finite<F>(result.abs_err) ) result.status = quad_status::non_finite;
450   return result;
451 }

453 template<ieee754_floating F, typename Fn, usize Capacity>
454 [[nodiscard]] inline quad_result<F>
455 dispatch_bounds(Fn f, F a, F b, const quad_options<F> &options, quad_workspace<F, Capacity> &ws) noexcept
456 {
457   if ( a != a || b != b ) {
458     quad_result<F> result{};
459     result.status = quad_status::invalid_input;
460     return result;
461   }
462   const int ai = ieee::inf_sign<F>(a);
463   const int bi = ieee::inf_sign<F>(b);
464   if ( ai == 0 && bi == 0 ) {
465     if ( b < a ) {
466       quad_result<F> r = finite<F>(f, b, a, options, ws);
467       r.value = -r.value;
468       return r;
469     }
470     return finite<F>(f, a, b, options, ws);
471   }
472 
473   if ( (ai == 1 && bi == 0) || (ai == 0 && bi == -1) || (ai == 1 && bi == -1) ) {
474     quad_result<F> r = dispatch_bounds<F>(f, b, a, options, ws);
475     r.value = -r.value;
476     return r;
477   }
478 
479   quad_options<F> transformed = options;
480   transformed.breakpoints = nullptr;
481   transformed.n_breakpoints = 0;
482 
483   if ( ai == 0 && bi == 1 ) {
484     auto g = [&](F t) noexcept -> F {
485       const F q = F(1) - t;
486       const F x = a + t / q;
487       return f(x) / (q * q);
488     };
489     return finite<F>(g, F(0), F(1), transformed, ws);
490   }
491   if ( ai == -1 && bi == 0 ) {
492     auto g = [&](F t) noexcept -> F {
493       const F q = F(1) - t;
494       const F x = b - t / q;
495       return f(x) / (q * q);
496     };
497     return finite<F>(g, F(0), F(1), transformed, ws);
498   }
499   if ( ai == -1 && bi == 1 ) {
500     auto g = [&](F t) noexcept -> F {
501       const F angle = constant_pi<F> * (t - F(0.5));
502       const F c = mk::trig::cos<F>(angle);
503       return f(mk::trig::tan<F>(angle)) * constant_pi<F> / (c * c);
504     };
505     return finite<F>(g, F(0), F(1), transformed, ws);
506   }
507 
508   quad_result<F> result{};
509   result.status = quad_status::invalid_input;
510   return result;
511 }

587 template<usize M, ieee754_floating F>
588 [[nodiscard, gnu::always_inline]] inline F
589 norm(const vec<F, M> &value, quad_error_norm mode) noexcept
590 {
591   F result = F(0);
592   if ( mode == quad_error_norm::maximum ) {
593     for ( usize component = 0; component < M; ++component ) {
594       const F magnitude = mk::manip::fabs<F>(value.data[component]);
595       if ( magnitude > result ) result = magnitude;
596     }
597     return result;
598   }
599   for ( usize component = 0; component < M; ++component ) result += value.data[component] * value.data[component];
600   return mk::pow_ns::sqrt<F>(result);
601 }

603 template<usize M, ieee754_floating F, typename Fn, typename Table>
604 [[nodiscard]] inline bool
605 evaluate_rule(Fn &f, F a, F b, quad_vec_interval<M, F> &out, usize &n_evals, quad_error_norm mode) noexcept
606 {
607   const F half_w = F(0.5) * (b - a);
608   const F abs_half_w = mk::manip::fabs<F>(half_w);
609   const F mid = F(0.5) * (a + b);
610   vec<F, M> negative[Table::half]{};
611   vec<F, M> positive[Table::half]{};
612   negative[0] = f(mid);
613   positive[0] = negative[0];
614   ++n_evals;
615   F resk[M]{}, resg[M]{}, resabs[M]{};
616   for ( usize component = 0; component < M; ++component ) {
617     const F center = negative[0].data[component];
618     if ( !ieee::is_finite<F>(center) ) return false;
619     resk[component] = Table::wk[0] * center;
620     resg[component] = Table::wg[0] * center;
621     resabs[component] = Table::wk[0] * mk::manip::fabs<F>(center);
622   }
623   for ( usize k = 1; k < Table::half; ++k ) {
624     const F absc = half_w * Table::nodes[k];
625     negative[k] = f(mid - absc);
626     positive[k] = f(mid + absc);
627     n_evals += 2;
628     for ( usize component = 0; component < M; ++component ) {
629       const F fm = negative[k].data[component];
630       const F fp = positive[k].data[component];
631       if ( !ieee::is_finite<F>(fm) || !ieee::is_finite<F>(fp) ) return false;
632       const F pair = fm + fp;
633       resk[component] = math::fma<F>(Table::wk[k], pair, resk[component]);
634       if ( Table::wg[k] != F(0) ) resg[component] = math::fma<F>(Table::wg[k], pair, resg[component]);
635       resabs[component] = math::fma<F>(Table::wk[k], mk::manip::fabs<F>(fm) + mk::manip::fabs<F>(fp), resabs[component]);
636     }
637   }
638 
639   out.a = a;
640   out.b = b;
641   for ( usize component = 0; component < M; ++component ) {
642     const F mean = F(0.5) * resk[component];
643     F resasc = Table::wk[0] * mk::manip::fabs<F>(negative[0].data[component] - mean);
644     for ( usize k = 1; k < Table::half; ++k )
645       resasc += Table::wk[k]
646                 * (mk::manip::fabs<F>(negative[k].data[component] - mean) + mk::manip::fabs<F>(positive[k].data[component] - mean));
647     const F scaled_abs = abs_half_w * resabs[component];
648     const F scaled_asc = abs_half_w * resasc;
649     F error = mk::manip::fabs<F>(half_w * (resk[component] - resg[component]));
650     if ( scaled_asc != F(0) && error != F(0) ) {
651       const F ratio = F(200) * error / scaled_asc;
652       if ( ratio < F(1) ) error = scaled_asc * ratio * mk::pow_ns::sqrt<F>(ratio);
653     }
654     const F floor = F(50) * machine_epsilon<F>() * scaled_abs;
655     out.value.data[component] = half_w * resk[component];
656     out.error.data[component] = error > floor ? error : floor;
657   }
658   out.priority = norm(out.error, mode);
659   return ieee::is_finite<F>(out.priority);
660 }

662 template<usize M, ieee754_floating F, typename Fn>
663 [[nodiscard]] inline bool
664 evaluate(Fn &f, F a, F b, gauss_kronrod_rule rule, quad_vec_interval<M, F> &out, usize &n_evals, quad_error_norm mode) noexcept
665 {
666   if ( rule == gauss_kronrod_rule::gk15 ) return evaluate_rule<M, F, Fn, coeff::gk::gk_15_7<F>>(f, a, b, out, n_evals, mode);
667   return evaluate_rule<M, F, Fn, coeff::gk::gk_21_10<F>>(f, a, b, out, n_evals, mode);
668 }

670 template<usize M, ieee754_floating F, usize Capacity>
671 inline void
672 push(quad_vec_workspace<M, F, Capacity> &workspace, const quad_vec_interval<M, F> &interval) noexcept
673 {
674   usize i = workspace.size++;
675   while ( i != 0 ) {
676     const usize parent = (i - 1) >> 1;
677     if ( workspace.heap[parent].priority >= interval.priority ) break;
678     workspace.heap[i] = workspace.heap[parent];
679     i = parent;
680   }
681   workspace.heap[i] = interval;
682 }

684 template<usize M, ieee754_floating F, usize Capacity>
685 [[nodiscard]] inline quad_vec_interval<M, F>
686 pop(quad_vec_workspace<M, F, Capacity> &workspace) noexcept
687 {
688   const quad_vec_interval<M, F> result = workspace.heap[0];
689   const quad_vec_interval<M, F> tail = workspace.heap[--workspace.size];
690   if ( workspace.size == 0 ) return result;
691   usize i = 0;
692   while ( true ) {
693     const usize left = i * 2 + 1;
694     if ( left >= workspace.size ) break;
695     const usize right = left + 1;
696     const usize child = right < workspace.size && workspace.heap[right].priority > workspace.heap[left].priority ? right : left;
697     if ( workspace.heap[child].priority <= tail.priority ) break;
698     workspace.heap[i] = workspace.heap[child];
699     i = child;
700   }
701   workspace.heap[i] = tail;
702   return result;
703 }

705 template<usize M, ieee754_floating F, typename Fn, usize Capacity>
706 [[nodiscard]] inline quad_vec_result<M, F>
707 finite(Fn f, F a, F b, const quad_options<F> &options, quad_vec_workspace<M, F, Capacity> &workspace, quad_error_norm mode) noexcept
708 {
709   quad_vec_result<M, F> result{};
710   workspace.reset();
711   if ( a == b ) return result;
712   if ( options.abs_tol < F(0) || options.rel_tol < F(0) || options.max_depth == 0
713        || (options.n_breakpoints != 0 && options.breakpoints == nullptr) ) {
714     result.status = quad_status::invalid_input;
715     return result;
716   }
717   for ( usize i = 0; i < options.n_breakpoints; ++i ) {
718     if ( !ieee::is_finite<F>(options.breakpoints[i]) ) {
719       result.status = quad_status::invalid_input;
720       return result;
721     }
722   }
723   const usize requested_limit = options.max_intervals == 0 ? Capacity : options.max_intervals;
724   const usize interval_limit = requested_limit < Capacity ? requested_limit : Capacity;
725   const usize eval_cost = options.rule == gauss_kronrod_rule::gk15 ? 15 : 21;
726   __impl_accumulate::runtime_sum<F> values[M];
727   __impl_accumulate::runtime_sum<F> errors[M];
728   for ( usize component = 0; component < M; ++component ) {
729     values[component].policy = options.accumulation;
730     errors[component].policy = options.accumulation;
731   }
732 
733   F left = a;
734   while ( left < b ) {
735     F right = b;
736     for ( usize i = 0; i < options.n_breakpoints; ++i ) {
737       const F point = options.breakpoints[i];
738       if ( point > left && point < right && point < b ) right = point;
739     }
740     if ( workspace.size >= interval_limit ) {
741       result.status = quad_status::max_intervals;
742       break;
743     }
744     if ( options.max_evals != 0 && result.n_evals + eval_cost > options.max_evals ) {
745       result.status = quad_status::max_evaluations;
746       break;
747     }
748     quad_vec_interval<M, F> interval{};
749     if ( !evaluate<M, F>(f, left, right, options.rule, interval, result.n_evals, mode) ) {
750       result.status = quad_status::non_finite;
751       workspace.reset();
752       return result;
753     }
754     push(workspace, interval);
755     for ( usize component = 0; component < M; ++component ) {
756       values[component].add(interval.value.data[component]);
757       errors[component].add(interval.error.data[component]);
758     }
759     left = right;
760   }
761   if ( workspace.size == 0 ) return result;
762 
763   while ( result.status == quad_status::ok ) {
764     for ( usize component = 0; component < M; ++component ) {
765       result.value.data[component] = values[component].get();
766       result.abs_err.data[component] = errors[component].get();
767     }
768     result.error_norm = norm(result.abs_err, mode);
769     const F value_norm = norm(result.value, mode);
770     const F relative = options.rel_tol * value_norm;
771     const F tolerance = options.abs_tol > relative ? options.abs_tol : relative;
772     if ( result.error_norm <= tolerance ) break;
773     if ( workspace.size >= interval_limit ) {
774       result.status = quad_status::max_intervals;
775       break;
776     }
777     if ( options.max_evals != 0 && result.n_evals + 2 * eval_cost > options.max_evals ) {
778       result.status = quad_status::max_evaluations;
779       break;
780     }
781     const quad_vec_interval<M, F> parent = pop(workspace);
782     if ( parent.depth >= options.max_depth ) {
783       push(workspace, parent);
784       result.status = quad_status::max_depth;
785       break;
786     }
787     const F mid = F(0.5) * (parent.a + parent.b);
788     if ( mid == parent.a || mid == parent.b ) {
789       push(workspace, parent);
790       result.status = quad_status::roundoff;
791       break;
792     }
793     quad_vec_interval<M, F> lo{}, hi{};
794     if ( !evaluate<M, F>(f, parent.a, mid, options.rule, lo, result.n_evals, mode)
795          || !evaluate<M, F>(f, mid, parent.b, options.rule, hi, result.n_evals, mode) ) {
796       push(workspace, parent);
797       result.status = quad_status::non_finite;
798       break;
799     }
800     lo.depth = parent.depth + 1;
801     hi.depth = parent.depth + 1;
802     for ( usize component = 0; component < M; ++component ) {
803       values[component].add(-parent.value.data[component]);
804       values[component].add(lo.value.data[component] + hi.value.data[component]);
805       errors[component].add(-parent.error.data[component]);
806       errors[component].add(lo.error.data[component] + hi.error.data[component]);
807     }
808     push(workspace, lo);
809     push(workspace, hi);
810   }
811   for ( usize component = 0; component < M; ++component ) {
812     result.value.data[component] = values[component].get();
813     result.abs_err.data[component] = errors[component].get();
814     if ( result.abs_err.data[component] < F(0) ) result.abs_err.data[component] = F(0);
815   }
816   result.error_norm = norm(result.abs_err, mode);
817   result.n_intervals = workspace.size;
818   return result;
819 }

823 template<usize M, ieee754_floating F, typename Fn, usize Capacity>
824   requires(M >= 2 && M <= 16)
825 [[nodiscard]] inline quad_vec_result<M, F>
826 quad_vec(Fn f, F a, F b, const quad_options<F> &options, quad_vec_workspace<M, F, Capacity> &workspace,
827          quad_error_norm norm = quad_error_norm::maximum) noexcept
828 {
829   if ( a != a || b != b ) {
830     quad_vec_result<M, F> result{};
831     result.status = quad_status::invalid_input;
832     return result;
833   }
834   const int ai = ieee::inf_sign<F>(a);
835   const int bi = ieee::inf_sign<F>(b);
836   if ( ai == 0 && bi == 0 ) {
837     if ( b < a ) {
838       quad_vec_result<M, F> result = __impl_quad_vec::finite<M, F>(f, b, a, options, workspace, norm);
839       for ( usize component = 0; component < M; ++component ) result.value.data[component] = -result.value.data[component];
840       return result;
841     }
842     return __impl_quad_vec::finite<M, F>(f, a, b, options, workspace, norm);
843   }
844   if ( (ai == 1 && bi == 0) || (ai == 0 && bi == -1) || (ai == 1 && bi == -1) ) {
845     quad_vec_result<M, F> result = quad_vec<M, F>(f, b, a, options, workspace, norm);
846     for ( usize component = 0; component < M; ++component ) result.value.data[component] = -result.value.data[component];
847     return result;
848   }
849   quad_options<F> transformed = options;
850   transformed.breakpoints = nullptr;
851   transformed.n_breakpoints = 0;
852   if ( ai == 0 && bi == 1 ) {
853     auto g = [&](F t) noexcept {
854       const F q = F(1) - t;
855       auto value = f(a + t / q);
856       for ( usize component = 0; component < M; ++component ) value.data[component] /= q * q;
857       return value;
858     };
859     return __impl_quad_vec::finite<M, F>(g, F(0), F(1), transformed, workspace, norm);
860   }
861   if ( ai == -1 && bi == 0 ) {
862     auto g = [&](F t) noexcept {
863       const F q = F(1) - t;
864       auto value = f(b - t / q);
865       for ( usize component = 0; component < M; ++component ) value.data[component] /= q * q;
866       return value;
867     };
868     return __impl_quad_vec::finite<M, F>(g, F(0), F(1), transformed, workspace, norm);
869   }
870   if ( ai == -1 && bi == 1 ) {
871     auto g = [&](F t) noexcept {
872       const F angle = constant_pi<F> * (t - F(0.5));
873       const F c = mk::trig::cos<F>(angle);
874       auto value = f(mk::trig::tan<F>(angle));
875       const F jacobian = constant_pi<F> / (c * c);
876       for ( usize component = 0; component < M; ++component ) value.data[component] *= jacobian;
877       return value;
878     };
879     return __impl_quad_vec::finite<M, F>(g, F(0), F(1), transformed, workspace, norm);
880   }
881   quad_vec_result<M, F> result{};
882   result.status = quad_status::invalid_input;
883   return result;
884 }

895 template<usize M, ieee754_floating F, typename Fn, usize Capacity>
896   requires(M >= 2 && M <= 16)
897 [[nodiscard]] inline quad_vec_result<M, F>
898 quad_vec(Fn f, F a, F b, const quad_options<F> &options, quad_workspace<F, Capacity> &workspace,
899          quad_error_norm norm = quad_error_norm::maximum) noexcept
900 {
901   quad_vec_result<M, F> result{};
902   F norm_acc = F(0);
903   for ( usize component = 0; component < M; ++component ) {
904     auto scalar = [&](F x) noexcept -> F { return f(x).data[component]; };
905     quad_result<F> r = quad<F>(scalar, a, b, options, workspace);
906     result.value.data[component] = r.value;
907     result.abs_err.data[component] = r.abs_err;
908     result.n_evals += r.n_evals;
909     if ( r.n_intervals > result.n_intervals ) result.n_intervals = r.n_intervals;
910     if ( r.status != quad_status::ok && result.status == quad_status::ok ) result.status = r.status;
911     if ( norm == quad_error_norm::maximum ) {
912       if ( r.abs_err > norm_acc ) norm_acc = r.abs_err;
913     } else {
914       norm_acc += r.abs_err * r.abs_err;
915     }
916   }
917   result.error_norm = norm == quad_error_norm::maximum ? norm_acc : mk::pow_ns::sqrt<F>(norm_acc);
918   return result;
919 }

921 template<ieee754_floating F, callable_real_batch<F> Fn, usize Capacity>
922 [[nodiscard]] inline quad_result<F>
923 quad_batch(Fn f, F a, F b, const quad_options<F> &options, quad_workspace<F, Capacity> &workspace) noexcept
924 {
925   if ( ieee::inf_sign<F>(a) == 0 && ieee::inf_sign<F>(b) == 0 ) {
926     if ( b < a ) {
927       quad_result<F> result = __impl_quad::finite<F, Fn, Capacity, true>(f, b, a, options, workspace);
928       result.value = -result.value;
929       return result;
930     }
931     return __impl_quad::finite<F, Fn, Capacity, true>(f, a, b, options, workspace);
932   }
933   auto scalar = [&](F x) noexcept -> F {
934     F y{};
935     f(&x, &y, 1);
936     return y;
937   };
938   return quad<F>(scalar, a, b, options, workspace);
939 }
