27 template<ieee754_floating F>
28 [[nodiscard]] inline cubic_spline_1d<F>
29 make_pchip(raw_slice<const F> xs, raw_slice<const F> ys, build_info<F> *info = nullptr) noexcept
30 {
31   cubic_spline_1d<F> s{};
32   s.bc = bc_kind::clamped;
33 
34   if ( xs.size() != ys.size() ) {
35     if ( info ) info->status = build_status::size_mismatch;
36     return s;
37   }
38   if ( xs.size() < 2 ) {
39     if ( info ) info->status = build_status::too_few_points;
40     return s;
41   }
42   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
43     if ( info ) info->status = build_status::non_monotonic_x;
44     return s;
45   }
46   const usize n = xs.size();
47 
48   vector<F> m(n, F(0));
49   m.set_size(n);
50   __impl_splines_bits::pchip_slopes<F>(xs.ptr, ys.ptr, m.data(), n);
51 
52   s.xs.reserve(n);
53   s.seg.reserve(n - 1);
54   for ( usize i = 0; i < n; ++i ) s.xs.emplace_back(xs[i]);
55   poly_coeffs<F, 3> zero{};
56   zero.data[0] = zero.data[1] = zero.data[2] = zero.data[3] = F(0);
57   for ( usize i = 0; i + 1 < n; ++i ) s.seg.emplace_back(zero);
58   __impl_splines_bits::build_cubic_segments_from_slopes<F>(xs.ptr, ys.ptr, m.data(), s.seg.data(), n);
59 
60   if ( info ) info->status = build_status::ok;
61   return s;
62 }
