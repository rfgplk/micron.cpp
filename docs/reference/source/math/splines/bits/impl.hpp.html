23 template<ieee754_floating F>
24 [[nodiscard, gnu::always_inline]] inline constexpr bool
25 strictly_increasing(const F *__restrict__ x, usize n) noexcept
26 {
27   if ( n < 2 ) return true;
28   for ( usize i = 1; i < n; ++i )
29     if ( !(x[i] > x[i - 1]) ) return false;
30   return true;
31 }

33 template<ieee754_floating F>
34 [[gnu::always_inline]] inline void
35 binary_advance(F knot, F query, usize candidate, usize &index) noexcept
36 {
37 #if defined(__OPTIMIZE__) && defined(__micron_arch_amd64)
38   if constexpr ( sizeof(F) == 8 ) {
39 #if defined(__micron_x86_avx)
40     asm volatile("vucomisd %[knot], %[query]\n\t"
41 #else
42     asm volatile("ucomisd %[knot], %[query]\n\t"
43 #endif
44                  "cmovae %[candidate], %[index]"
45                  : [index] "+&r"(index)
46                  : [candidate] "r"(candidate), [knot] "x"(knot), [query] "x"(query)
47                  : "cc");
48   } else if constexpr ( sizeof(F) == 4 ) {
49 #if defined(__micron_x86_avx)
50     asm volatile("vucomiss %[knot], %[query]\n\t"
51 #else
52     asm volatile("ucomiss %[knot], %[query]\n\t"
53 #endif
54                  "cmovae %[candidate], %[index]"
55                  : [index] "+&r"(index)
56                  : [candidate] "r"(candidate), [knot] "x"(knot), [query] "x"(query)
57                  : "cc");
58   } else {
59     // long double / f128 have no xmm register class; ucomis* cannot take them
60     if ( knot <= query ) index = candidate;
61   }
62 #else
63   if ( knot <= query ) index = candidate;
64 #endif
65 }

67 template<ieee754_floating F>
68 [[nodiscard, gnu::always_inline]] inline usize
69 locate_segment(const F *__restrict__ xs, usize n, F x, usize &last) noexcept
70 {
71   const usize last_seg = (n >= 2) ? (n - 2) : 0;
72 
73   if ( !(x > xs[0]) ) {
74     last = 0;
75     return 0;
76   }
77   if ( !(x < xs[n - 1]) ) {
78     last = last_seg;
79     return last_seg;
80   }
81 
82   const usize i = last <= last_seg ? last : last_seg;
83   if ( x >= xs[i] && x <= xs[i + 1] ) {
84     last = i;
85     return i;
86   }
87   if ( i + 1 <= last_seg && x >= xs[i + 1] && x <= xs[i + 2] ) {
88     last = i + 1;
89     return i + 1;
90   }
91 
92 #if defined(__OPTIMIZE__) && defined(__micron_arch_amd64)
93   usize lo = 0;
94   usize step = micron::bit_floor(last_seg);
95   if ( step != 0 && last_seg == (step << 1) - 1 ) {
96     for ( ; step != 0; step >>= 1 ) {
97       const usize candidate = lo + step;
98       binary_advance<F>(xs[candidate], x, candidate, lo);
99     }
100   } else {
101     for ( ; step != 0; step >>= 1 ) {
102       const usize unbounded = lo + step;
103       const usize candidate = unbounded < last_seg ? unbounded : last_seg;
104       binary_advance<F>(xs[candidate], x, candidate, lo);
105     }
106   }
107 #else
108   usize lo = 0;
109   usize hi = n - 1;
110   while ( hi - lo > 1 ) {
111     const usize mid = lo + ((hi - lo) >> 1);
112     if ( xs[mid] <= x )
113       lo = mid;
114     else
115       hi = mid;
116   }
117 #endif
118   last = lo;
119   return lo;
120 }

122 template<ieee754_floating F>
123 [[nodiscard, gnu::always_inline]] inline F
124 eval_cubic_local(const poly_coeffs<F, 3> &p, F t) noexcept
125 {
126   F r = math::fma<F>(p.data[3], t, p.data[2]);
127   r = math::fma<F>(r, t, p.data[1]);
128   r = math::fma<F>(r, t, p.data[0]);
129   return r;
130 }

148 template<ieee754_floating F>
149 [[nodiscard, gnu::always_inline]] inline F
150 eval_cubic_antideriv_local(const poly_coeffs<F, 3> &p, F t) noexcept
151 {
152   // via horner
153   F r = math::fma<F>(p.data[3] * F(0.25), t, p.data[2] * (F(1) / F(3)));
154   r = math::fma<F>(r, t, p.data[1] * F(0.5));
155   r = math::fma<F>(r, t, p.data[0]);
156   return r * t;
157 }

159 template<ieee754_floating F>
160 inline void
161 build_cubic_segments(const F *__restrict__ xs, const F *__restrict__ ys, const F *__restrict__ M, poly_coeffs<F, 3> *__restrict__ seg,
162                      usize n) noexcept
163 {
164   const F sixth = F(1) / F(6);
165   for ( usize i = 0; i + 1 < n; ++i ) {
166     const F h = xs[i + 1] - xs[i];
167     const F dy = ys[i + 1] - ys[i];
168     const F Mi = M[i];
169     const F Mi1 = M[i + 1];
170     seg[i].data[0] = ys[i];
171     seg[i].data[1] = dy / h - h * (F(2) * Mi + Mi1) * sixth;
172     seg[i].data[2] = F(0.5) * Mi;
173     seg[i].data[3] = (Mi1 - Mi) / (F(6) * h);
174   }
175 }

177 template<ieee754_floating F>
178 inline void
179 build_cubic_segments_from_slopes(const F *__restrict__ xs, const F *__restrict__ ys, const F *__restrict__ m,
180                                  poly_coeffs<F, 3> *__restrict__ seg, usize n) noexcept
181 {
182   for ( usize i = 0; i + 1 < n; ++i ) {
183     const F h = xs[i + 1] - xs[i];
184     const F inv_h = F(1) / h;
185     const F delta = (ys[i + 1] - ys[i]) * inv_h;
186     const F mi = m[i];
187     const F mi1 = m[i + 1];
188     seg[i].data[0] = ys[i];
189     seg[i].data[1] = mi;
190     seg[i].data[2] = (F(3) * delta - F(2) * mi - mi1) * inv_h;
191     seg[i].data[3] = (mi + mi1 - F(2) * delta) * (inv_h * inv_h);
192   }
193 }

195 template<ieee754_floating F>
196 inline void
197 pchip_slopes(const F *__restrict__ xs, const F *__restrict__ ys, F *__restrict__ m, usize n) noexcept
198 {
199   if ( n == 0 ) return;
200   if ( n == 1 ) {
201     m[0] = F(0);
202     return;
203   }
204   if ( n == 2 ) {
205     const F s = (ys[1] - ys[0]) / (xs[1] - xs[0]);
206     m[0] = s;
207     m[1] = s;
208     return;
209   }
210 
211   for ( usize i = 1; i + 1 < n; ++i ) {
212     const F h0 = xs[i] - xs[i - 1];
213     const F h1 = xs[i + 1] - xs[i];
214     const F s0 = (ys[i] - ys[i - 1]) / h0;
215     const F s1 = (ys[i + 1] - ys[i]) / h1;
216     if ( s0 * s1 <= F(0) ) {
217       m[i] = F(0);
218     } else {
219       const F w0 = F(2) * h1 + h0;
220       const F w1 = h1 + F(2) * h0;
221       m[i] = (w0 + w1) / (w0 / s0 + w1 / s1);
222     }
223   }
224 
225   {
226     const F h0 = xs[1] - xs[0];
227     const F h1 = xs[2] - xs[1];
228     const F s0 = (ys[1] - ys[0]) / h0;
229     const F s1 = (ys[2] - ys[1]) / h1;
230     F t = ((F(2) * h0 + h1) * s0 - h0 * s1) / (h0 + h1);
231     if ( t * s0 <= F(0) )
232       t = F(0);
233     else if ( s0 * s1 <= F(0) && (t < F(0) ? -t : t) > F(3) * (s0 < F(0) ? -s0 : s0) )
234       t = F(3) * s0;
235     m[0] = t;
236   }
237   {
238     const F h0 = xs[n - 2] - xs[n - 3];
239     const F h1 = xs[n - 1] - xs[n - 2];
240     const F s0 = (ys[n - 2] - ys[n - 3]) / h0;
241     const F s1 = (ys[n - 1] - ys[n - 2]) / h1;
242     F t = ((F(2) * h1 + h0) * s1 - h1 * s0) / (h0 + h1);
243     if ( t * s1 <= F(0) )
244       t = F(0);
245     else if ( s0 * s1 <= F(0) && (t < F(0) ? -t : t) > F(3) * (s1 < F(0) ? -s1 : s1) )
246       t = F(3) * s1;
247     m[n - 1] = t;
248   }
249 }

251 template<ieee754_floating F>
252 [[nodiscard, gnu::always_inline]] inline bool
253 is_sorted_nondecreasing(const F *__restrict__ x, usize n, usize probe = 8) noexcept
254 {
255   (void)probe;
256   if ( n == 0 ) return true;
257   if ( !(x[0] == x[0]) ) return false;
258   for ( usize i = 1; i < n; ++i )
259     if ( !(x[i] >= x[i - 1]) ) return false;
260   return true;
261 }
