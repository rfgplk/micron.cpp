30 template<ieee754_floating F>
31 inline void
32 positive_reciprocal_in_place(F *__restrict__ values, usize count) noexcept
33 {
34   for ( usize i = 0; i < count; ++i ) values[i] = values[i] > F(0) ? F(1) / values[i] : F(0);
35 }

37 template<ieee754_floating F>
38 inline void
39 cubic_horner_batch(const poly_coeffs<F, 3> &p, F origin, const F *__restrict__ x, F *__restrict__ out, usize count) noexcept
40 {
41   for ( usize i = 0; i < count; ++i ) {
42     const F u = x[i] - origin;
43     F value = math::fma<F>(p[3], u, p[2]);
44     value = math::fma<F>(value, u, p[1]);
45     out[i] = math::fma<F>(value, u, p[0]);
46   }
47 }
48 
49 template<ieee754_floating F>
50 inline void
51 linear_batch(F origin, F y0, F slope, const F *__restrict__ x, F *__restrict__ out, usize count) noexcept
52 {
53   for ( usize i = 0; i < count; ++i ) out[i] = math::fma<F>(slope, x[i] - origin, y0);
54 }

56 template<ieee754_floating F>
57 inline void
58 cubic_horner_stream_batch(const poly_coeffs<F, 3> &p, F origin, const F *__restrict__ x, F *__restrict__ out, usize count) noexcept
59 {
60   cubic_horner_batch<F>(p, origin, x, out, count);
61 }

66 {

78 template<ieee754_floating F>
79 inline void
80 packed_curve_horner(const F *__restrict__ coeff, F u, F *__restrict__ out, usize dimensions) noexcept
81 {
82   for ( usize d = 0; d < dimensions; ++d ) {
83     F value = math::fma<F>(coeff[3 * dimensions + d], u, coeff[2 * dimensions + d]);
84     value = math::fma<F>(value, u, coeff[dimensions + d]);
85     out[d] = math::fma<F>(value, u, coeff[d]);
86   }
87 }

89 template<ieee754_floating F>
90 inline void
91 curve_horner(const F *__restrict__ a, const F *__restrict__ b, const F *__restrict__ c, const F *__restrict__ d, F u, F *__restrict__ out,
92              usize dimensions) noexcept
93 {
94   for ( usize axis = 0; axis < dimensions; ++axis ) {
95     F value = math::fma<F>(d[axis], u, c[axis]);
96     value = math::fma<F>(value, u, b[axis]);
97     out[axis] = math::fma<F>(value, u, a[axis]);
98   }
99 }
