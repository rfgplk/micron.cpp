27 template<ieee754_floating F>
28 [[nodiscard]] inline vector<F>
29 make_unclamped_uniform_knots(usize n_ctrl, u32 degree, F t_min, F t_max) noexcept
30 {
31   vector<F> out;
32   if ( degree == 0 || n_ctrl < degree + 1 || !(t_max > t_min) ) return out;
33   const usize count = n_ctrl + degree + 1;
34   const F spacing = (t_max - t_min) / F(n_ctrl - degree);
35   out.reserve(count);
36   for ( usize i = 0; i < count; ++i ) out.emplace_back(t_min + (F(i) - F(degree)) * spacing);
37   return out;
38 }

47 template<ieee754_floating F>
48 [[nodiscard]] inline vector<F>
49 greville_abscissae(raw_slice<const F> knots, u32 degree) noexcept
50 {
51   vector<F> out;
52   if ( degree == 0 || knots.size() < usize(2 * degree + 2) ) return out;
53   const usize n_ctrl = knots.size() - degree - 1;
54   out.reserve(n_ctrl);
55   for ( usize i = 0; i < n_ctrl; ++i ) {
56     F sum = F(0);
57     for ( u32 j = 1; j <= degree; ++j ) sum += knots[i + j];
58     out.emplace_back(sum / F(degree));
59   }
60   return out;
61 }

63 template<ieee754_floating F>
64 [[nodiscard]] inline vector<F>
65 make_knots(knot_style style, usize n_ctrl, u32 degree, F t_min, F t_max, raw_slice<const F> samples = {}) noexcept
66 {
67   if ( style == knot_style::clamped_uniform ) return make_uniform_clamped_knots<F>(n_ctrl, degree, t_min, t_max);
68   if ( style == knot_style::unclamped_uniform ) return make_unclamped_uniform_knots<F>(n_ctrl, degree, t_min, t_max);
69   if ( style == knot_style::periodic_uniform ) return make_periodic_uniform_knots<F>(n_ctrl, degree, t_min, t_max);
70   if ( samples.size() != n_ctrl ) return {};
71   return make_averaged_clamped_knots<F>(samples.ptr, samples.size(), degree);

74 template<ieee754_floating F>
75 inline void
76 basis(const bspline<F> &s, F x, F *__restrict__ out, usize out_count) noexcept
77 {
78   const usize n_ctrl = s.ctrl.size();
79   for ( usize i = 0; i < out_count; ++i ) out[i] = F(0);
80   if ( n_ctrl == 0 || out_count < n_ctrl || s.degree == 0 ) return;
81   const u32 p = s.degree;
82   const F *u = s.knots.data();
83   if ( x < u[p] ) x = u[p];
84   if ( x > u[n_ctrl] ) x = u[n_ctrl];
85   usize last = s.last_hit;
86   const usize span = __impl_bspline::bspline_span<F>(u, n_ctrl, p, x, last);
87   F local[bspline_max_degree + 1];
88   __impl_bspline::bspline_basis<F>(u, span, p, x, local);
89   for ( u32 j = 0; j <= p; ++j ) out[span - p + j] = local[j];
90 }

105 template<ieee754_floating F>
106 inline void
107 local_basis_derivatives(const F *u, usize span, u32 degree, F x, u32 order, F out[bspline_max_degree + 1][bspline_max_degree + 1]) noexcept
108 {
109   F ndu[bspline_max_degree + 1][bspline_max_degree + 1]{};
110   F left[bspline_max_degree + 1]{};
111   F right[bspline_max_degree + 1]{};
112   ndu[0][0] = F(1);
113   for ( u32 j = 1; j <= degree; ++j ) {
114     left[j] = x - u[span + 1 - j];
115     right[j] = u[span + j] - x;
116     F saved = F(0);
117     for ( u32 r = 0; r < j; ++r ) {
118       ndu[j][r] = right[r + 1] + left[j - r];
119       const F temp = ndu[j][r] != F(0) ? ndu[r][j - 1] / ndu[j][r] : F(0);
120       ndu[r][j] = saved + right[r + 1] * temp;
121       saved = left[j - r] * temp;
122     }
123     ndu[j][j] = saved;
124   }
125   for ( u32 j = 0; j <= degree; ++j ) out[0][j] = ndu[j][degree];
126 
127   F a[2][bspline_max_degree + 1]{};
128   for ( u32 r = 0; r <= degree; ++r ) {
129     u32 s1 = 0, s2 = 1;
130     a[0][0] = F(1);
131     for ( u32 k = 1; k <= order; ++k ) {
132       F d = F(0);
133       const i32 rk = i32(r) - i32(k);
134       const i32 pk = i32(degree) - i32(k);
135       if ( r >= k ) {
136         const F denom = ndu[pk + 1][rk];
137         a[s2][0] = denom != F(0) ? a[s1][0] / denom : F(0);
138         d = a[s2][0] * ndu[rk][pk];
139       }
140       const i32 j1 = rk >= -1 ? 1 : -rk;
141       const i32 j2 = i32(r) - 1 <= pk ? i32(k) - 1 : i32(degree) - i32(r);
142       for ( i32 j = j1; j <= j2; ++j ) {
143         const F denom = ndu[pk + 1][rk + j];
144         a[s2][j] = denom != F(0) ? (a[s1][j] - a[s1][j - 1]) / denom : F(0);
145         d += a[s2][j] * ndu[rk + j][pk];
146       }
147       if ( i32(r) <= pk ) {
148         const F denom = ndu[pk + 1][r];
149         a[s2][k] = denom != F(0) ? -a[s1][k - 1] / denom : F(0);
150         d += a[s2][k] * ndu[r][pk];
151       }
152       out[k][r] = d;
153       const u32 swap = s1;
154       s1 = s2;
155       s2 = swap;
156     }
157   }
158   F scale = F(degree);
159   for ( u32 k = 1; k <= order; ++k ) {
160     for ( u32 j = 0; j <= degree; ++j ) out[k][j] *= scale;
161     scale *= F(degree - k);
162   }
163 }

165 template<ieee754_floating F>
166 [[nodiscard]] inline bool
167 valid_knots(raw_slice<const F> knots, usize n_ctrl, u32 degree) noexcept
168 {
169   if ( degree == 0 || degree > bspline_max_degree || n_ctrl < degree + 1 ) return false;
170   if ( knots.size() != n_ctrl + degree + 1 ) return false;
171   for ( usize i = 1; i < knots.size(); ++i )
172     if ( knots[i] < knots[i - 1] ) return false;
173   return knots[degree] < knots[n_ctrl];
174 }

176 template<ieee754_floating F>
177 [[nodiscard]] inline bool
178 banded_qr_solve(raw_slice<const F> xs, raw_slice<const F> ys, raw_slice<const F> weights, raw_slice<const F> knots, usize n_ctrl,
179                 u32 degree, F *solution) noexcept
180 {
181   const usize width = usize(degree + 1);
182   vector<F> R(n_ctrl * width, F(0));
183   R.set_size(n_ctrl * width);
184   vector<F> transformed(n_ctrl, F(0));
185   transformed.set_size(n_ctrl);
186   usize last = degree;
187   for ( usize sample = 0; sample < xs.size(); ++sample ) {
188     const usize span = __impl_bspline::bspline_span<F>(knots.ptr, n_ctrl, degree, xs[sample], last);
189     const usize first = span - degree;
190     F row[bspline_max_degree + 1]{};
191     __impl_bspline::bspline_basis<F>(knots.ptr, span, degree, xs[sample], row);
192     const F scale = weights.size() ? F(math::fsqrt(weights[sample])) : F(1);
193     for ( u32 j = 0; j <= degree; ++j ) row[j] *= scale;
194     F rhs = scale * ys[sample];
195     for ( usize column = first; column <= span; ++column ) {
196       const usize local_column = column - first;
197       const F a = R[column * width];
198       const F b = row[local_column];
199       if ( b == F(0) ) continue;
200       const F radius = F(math::fsqrt(a * a + b * b));
201       if ( !(radius > F(0)) ) continue;
202       const F inverse_radius = F(1) / radius;
203       const F cosine = a * inverse_radius;
204       const F sine = b * inverse_radius;
205       const usize last_column = span < column + degree ? span : column + degree;
206       for ( usize j = column; j <= last_column; ++j ) {
207         const usize r_index = column * width + j - column;
208         const usize row_index = j - first;
209         const F old_r = R[r_index];
210         const F old_row = row[row_index];
211         R[r_index] = math::fma<F>(sine, old_row, cosine * old_r);
212         row[row_index] = math::fma<F>(cosine, old_row, -sine * old_r);
213       }
214       const F old_rhs = transformed[column];
215       transformed[column] = math::fma<F>(sine, rhs, cosine * old_rhs);
216       rhs = math::fma<F>(cosine, rhs, -sine * old_rhs);
217     }
218   }
219   const F tiny = sizeof(F) <= 4 ? F(1e-12) : F(1e-28);
220   for ( usize ii = n_ctrl; ii-- > 0; ) {
221     F rhs = transformed[ii];
222     const usize end = ii + degree < n_ctrl - 1 ? ii + degree : n_ctrl - 1;
223     for ( usize j = ii + 1; j <= end; ++j ) rhs -= R[ii * width + j - ii] * solution[j];
224     const F diagonal = R[ii * width];
225     if ( !(math::fabs(diagonal) > tiny) ) return false;
226     solution[ii] = rhs / diagonal;
227   }
228   return true;
229 }

231 template<ieee754_floating F>
232 [[nodiscard]] inline bool
233 banded_normal_solve(raw_slice<const F> xs, raw_slice<const F> ys, raw_slice<const F> weights, raw_slice<const F> knots, usize n_ctrl,
234                     u32 degree, F *solution) noexcept
235 {
236   const usize width = usize(degree + 1);
237   vector<F> lower(n_ctrl * width, F(0));
238   lower.set_size(n_ctrl * width);
239   for ( usize i = 0; i < n_ctrl; ++i ) solution[i] = F(0);
240   usize last = degree;
241   for ( usize sample = 0; sample < xs.size(); ++sample ) {
242     const usize span = __impl_bspline::bspline_span<F>(knots.ptr, n_ctrl, degree, xs[sample], last);
243     const usize first = span - degree;
244     F row[bspline_max_degree + 1]{};
245     __impl_bspline::bspline_basis<F>(knots.ptr, span, degree, xs[sample], row);
246     const F weight = weights.size() ? weights[sample] : F(1);
247     for ( u32 j = 0; j <= degree; ++j ) {
248       const usize column = first + j;
249       solution[column] = math::fma<F>(weight * row[j], ys[sample], solution[column]);
250       for ( u32 k = 0; k <= j; ++k ) {
251         const usize other = first + k;
252         lower[column * width + column - other] = math::fma<F>(weight * row[j], row[k], lower[column * width + column - other]);
253       }
254     }
255   }
256   for ( usize i = 0; i < n_ctrl; ++i ) {
257     const usize begin = i > degree ? i - degree : 0;
258     for ( usize j = begin; j <= i; ++j ) {
259       F value = lower[i * width + i - j];
260       usize k_begin = i > degree ? i - degree : 0;
261       const usize j_begin = j > degree ? j - degree : 0;
262       if ( j_begin > k_begin ) k_begin = j_begin;
263       for ( usize k = k_begin; k < j; ++k ) value -= lower[i * width + i - k] * lower[j * width + j - k];
264       if ( i == j ) {
265         if ( !(value > F(0)) ) return false;
266         lower[i * width] = F(math::fsqrt(value));
267       } else {
268         lower[i * width + i - j] = value / lower[j * width];
269       }
270     }
271   }
272   for ( usize i = 0; i < n_ctrl; ++i ) {
273     F value = solution[i];
274     const usize begin = i > degree ? i - degree : 0;
275     for ( usize j = begin; j < i; ++j ) value -= lower[i * width + i - j] * solution[j];
276     solution[i] = value / lower[i * width];
277   }
278   for ( usize ii = n_ctrl; ii-- > 0; ) {
279     F value = solution[ii];
280     const usize end = ii + degree < n_ctrl - 1 ? ii + degree : n_ctrl - 1;
281     for ( usize j = ii + 1; j <= end; ++j ) value -= lower[j * width + j - ii] * solution[j];
282     solution[ii] = value / lower[ii * width];
283   }
284   return true;
285 }

289 template<ieee754_floating F>
290 [[nodiscard]] inline vector<F>
291 basis_derivatives(const bspline<F> &s, F x, u32 order) noexcept
292 {
293   vector<F> out;
294   const usize n_ctrl = s.ctrl.size();
295   if ( n_ctrl == 0 || s.degree == 0 ) return out;
296   if ( order > s.degree ) order = s.degree;
297   out = vector<F>(usize(order + 1) * n_ctrl, F(0));
298   out.set_size(usize(order + 1) * n_ctrl);
299   const F *u = s.knots.data();
300   if ( x < u[s.degree] ) x = u[s.degree];
301   if ( x > u[n_ctrl] ) x = u[n_ctrl];
302   usize last = s.last_hit;
303   const usize span = __impl_bspline::bspline_span<F>(u, n_ctrl, s.degree, x, last);
304   F local[bspline_max_degree + 1][bspline_max_degree + 1]{};
305   __impl_bspline_helpers::local_basis_derivatives<F>(u, span, s.degree, x, order, local);
306   for ( u32 k = 0; k <= order; ++k )
307     for ( u32 j = 0; j <= s.degree; ++j ) out[usize(k) * n_ctrl + span - s.degree + j] = local[k][j];
308   return out;
309 }

311 template<ieee754_floating F>
312 [[nodiscard]] inline sparse::csr<F, u32>
313 design_matrix(const bspline<F> &s, raw_slice<const F> x) noexcept
314 {
315   const usize n_ctrl = s.ctrl.size();
316   sparse::csr<F, u32> out(x.size(), n_ctrl);
317   if ( n_ctrl == 0 || s.degree == 0 ) return out;
318   const usize per_row = usize(s.degree + 1);
319   out.inner.reserve(x.size() * per_row);
320   out.values.reserve(x.size() * per_row);
321   out.outer[0] = 0;
322   usize last = s.last_hit;
323   for ( usize i = 0; i < x.size(); ++i ) {
324     F value = x[i];
325     if ( value < s.knots[s.degree] ) value = s.knots[s.degree];
326     if ( value > s.knots[n_ctrl] ) value = s.knots[n_ctrl];
327     const usize span = __impl_bspline::bspline_span<F>(s.knots.data(), n_ctrl, s.degree, value, last);
328     F local[bspline_max_degree + 1];
329     __impl_bspline::bspline_basis<F>(s.knots.data(), span, s.degree, value, local);
330     for ( u32 j = 0; j <= s.degree; ++j ) {
331       out.inner.emplace_back(u32(span - s.degree + j));
332       out.values.emplace_back(local[j]);
333     }
334     out.outer[i + 1] = u32(out.values.size());
335   }
336   return out;
337 }

339 template<ieee754_floating F>
340 [[nodiscard]] inline bspline<F>
341 antiderivative_spline(const bspline<F> &s, F constant = F(0)) noexcept
342 {
343   bspline<F> out{};
344   const usize n_ctrl = s.ctrl.size();
345   if ( n_ctrl == 0 || s.degree >= bspline_max_degree ) return out;
346   const u32 p = s.degree;
347   out.degree = p + 1;
348   out.knots.reserve(s.knots.size() + 2);
349   out.knots.emplace_back(s.knots[0]);
350   for ( usize i = 0; i < s.knots.size(); ++i ) out.knots.emplace_back(s.knots[i]);
351   out.knots.emplace_back(s.knots[s.knots.size() - 1]);
352   out.ctrl.reserve(n_ctrl + 1);
353   F value = constant;
354   out.ctrl.emplace_back(value);
355   for ( usize i = 0; i < n_ctrl; ++i ) {
356     value += (s.knots[i + p + 1] - s.knots[i]) * s.ctrl[i] / F(p + 1);
357     out.ctrl.emplace_back(value);
358   }
359   return out;
360 }

370 template<ieee754_floating F>
371 [[nodiscard]] inline bspline<F>
372 insert_knot(const bspline<F> &s, F knot, u32 repetitions = 1, build_info<F> *info = nullptr) noexcept
373 {
374   bspline<F> current = s;
375   if ( s.ctrl.size() == 0 || repetitions == 0 ) {
376     if ( info ) info->status = build_status::ok;
377     return current;
378   }
379   for ( u32 repeat = 0; repeat < repetitions; ++repeat ) {
380     const u32 p = current.degree;
381     const usize n_ctrl = current.ctrl.size();
382     const F lo = current.knots[p];
383     const F hi = current.knots[n_ctrl];
384     if ( knot < lo || knot > hi ) {
385       if ( info ) info->status = build_status::invalid_argument;
386       return {};
387     }
388     usize last = current.last_hit;
389     const usize span = __impl_bspline::bspline_span<F>(current.knots.data(), n_ctrl, p, knot, last);
390     u32 multiplicity = 0;
391     for ( usize i = 0; i < current.knots.size(); ++i )
392       if ( current.knots[i] == knot ) ++multiplicity;
393     const u32 limit = (knot == lo || knot == hi) ? p + 1 : p;
394     if ( multiplicity >= limit ) {
395       if ( info ) info->status = build_status::degenerate;
396       return {};
397     }
398     bspline<F> next{};
399     next.degree = p;
400     next.knots.reserve(current.knots.size() + 1);
401     for ( usize i = 0; i <= span; ++i ) next.knots.emplace_back(current.knots[i]);
402     next.knots.emplace_back(knot);
403     for ( usize i = span + 1; i < current.knots.size(); ++i ) next.knots.emplace_back(current.knots[i]);
404     next.ctrl.reserve(n_ctrl + 1);
405     for ( usize i = 0; i <= span - p; ++i ) next.ctrl.emplace_back(current.ctrl[i]);
406     for ( usize i = span - p + 1; i <= span - multiplicity; ++i ) {
407       const F denom = current.knots[i + p] - current.knots[i];
408       const F alpha = denom > F(0) ? (knot - current.knots[i]) / denom : F(0);
409       next.ctrl.emplace_back(math::fma<F>(alpha, current.ctrl[i] - current.ctrl[i - 1], current.ctrl[i - 1]));
410     }
411     for ( usize i = span - multiplicity; i < n_ctrl; ++i ) next.ctrl.emplace_back(current.ctrl[i]);
412     current = micron::move(next);
413   }
414   if ( info ) info->status = build_status::ok;
415   return current;
416 }

418 template<ieee754_floating F>
419 [[nodiscard]] inline bspline<F>
420 refine_knots(const bspline<F> &s, raw_slice<const F> knots, build_info<F> *info = nullptr) noexcept
421 {
422   bspline<F> out = s;
423   for ( usize i = 0; i < knots.size(); ++i ) {
424     build_info<F> local{};
425     out = insert_knot<F>(out, knots[i], 1, &local);
426     if ( local.status != build_status::ok ) {
427       if ( info ) info->status = local.status;
428       return {};
429     }
430   }
431   if ( info ) info->status = build_status::ok;
432   return out;
433 }

440 template<ieee754_floating F>
441 [[nodiscard]] inline bspline_split<F>
442 split(const bspline<F> &s, F knot, build_info<F> *info = nullptr) noexcept
443 {
444   bspline_split<F> out{};
445   if ( s.ctrl.size() == 0 || !(knot > s.knots[s.degree] && knot < s.knots[s.ctrl.size()]) ) {
446     if ( info ) info->status = build_status::invalid_argument;
447     return out;
448   }
449   u32 multiplicity = 0;
450   for ( usize i = 0; i < s.knots.size(); ++i )
451     if ( s.knots[i] == knot ) ++multiplicity;
452   const u32 insertions = multiplicity < s.degree ? s.degree - multiplicity : 0;
453   auto refined = insert_knot<F>(s, knot, insertions, info);
454   if ( refined.ctrl.size() == 0 ) return out;
455   usize first = 0;
456   while ( first < refined.knots.size() && refined.knots[first] < knot ) ++first;
457   const usize left_ctrl = first;
458   // The p-fold knot is shared by both children.  Its shared control point is
459   // immediately before the first copy of the split knot.
460   const usize right_start = first - 1;
461 
462   out.left.degree = refined.degree;
463   out.left.ctrl.reserve(left_ctrl);
464   for ( usize i = 0; i < left_ctrl; ++i ) out.left.ctrl.emplace_back(refined.ctrl[i]);
465   out.left.knots.reserve(left_ctrl + refined.degree + 1);
466   for ( usize i = 0; i < first; ++i ) out.left.knots.emplace_back(refined.knots[i]);
467   for ( u32 i = 0; i <= refined.degree; ++i ) out.left.knots.emplace_back(knot);
468 
469   out.right.degree = refined.degree;
470   out.right.ctrl.reserve(refined.ctrl.size() - right_start);
471   for ( usize i = right_start; i < refined.ctrl.size(); ++i ) out.right.ctrl.emplace_back(refined.ctrl[i]);
472   out.right.knots.reserve(out.right.ctrl.size() + refined.degree + 1);
473   for ( u32 i = 0; i <= refined.degree; ++i ) out.right.knots.emplace_back(knot);
474   for ( usize i = first + refined.degree; i < refined.knots.size(); ++i ) out.right.knots.emplace_back(refined.knots[i]);
475   if ( info ) info->status = build_status::ok;
476   return out;
477 }

479 template<ieee754_floating F>
480 [[nodiscard]] inline bspline<F>
481 make_lsq_bspline(raw_slice<const F> xs, raw_slice<const F> ys, raw_slice<const F> knots, u32 degree, raw_slice<const F> weights = {},
482                  lsq_method method = lsq_method::householder_qr, build_info<F> *info = nullptr) noexcept
483 {
484   bspline<F> out{};
485   out.degree = degree;
486   if ( xs.size() != ys.size() || (weights.size() != 0 && weights.size() != xs.size()) ) {
487     if ( info ) info->status = build_status::size_mismatch;
488     return out;
489   }
490   if ( knots.size() <= degree + 1 ) {
491     if ( info ) info->status = build_status::too_few_points;
492     return out;
493   }
494   const usize n_ctrl = knots.size() - degree - 1;
495   if ( xs.size() < n_ctrl || !__impl_bspline_helpers::valid_knots<F>(knots, n_ctrl, degree) ) {
496     if ( info ) info->status = build_status::invalid_argument;
497     return out;
498   }
499   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
500     if ( info ) info->status = build_status::non_monotonic_x;
501     return out;
502   }
503   out.knots.reserve(knots.size());
504   for ( usize i = 0; i < knots.size(); ++i ) out.knots.emplace_back(knots[i]);
505   vector<F> solution(n_ctrl, F(0));
506   solution.set_size(n_ctrl);
507   for ( usize i = 0; i < xs.size(); ++i ) {
508     if ( xs[i] < knots[degree] || xs[i] > knots[n_ctrl] || (weights.size() && !(weights[i] > F(0))) ) {
509       if ( info ) info->status = build_status::invalid_argument;
510       return {};
511     }
512   }
513   const bool solved = method == lsq_method::householder_qr
514                           ? __impl_bspline_helpers::banded_qr_solve<F>(xs, ys, weights, knots, n_ctrl, degree, solution.data())
515                           : __impl_bspline_helpers::banded_normal_solve<F>(xs, ys, weights, knots, n_ctrl, degree, solution.data());
516   if ( !solved ) {
517     if ( info ) info->status = build_status::singular_system;
518     return {};
519   }
520   out.ctrl.reserve(n_ctrl);
521   for ( usize i = 0; i < n_ctrl; ++i ) out.ctrl.emplace_back(solution[i]);
522   if ( info ) info->status = build_status::ok;
523   return out;
524 }

526 template<ieee754_floating F>
527 inline void
528 evaluate(const bspline<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize n) noexcept
529 {
530   for ( usize i = 0; i < n; ++i ) out[i] = evaluate<F>(s, xq[i]);
