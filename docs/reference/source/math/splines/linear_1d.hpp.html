26 template<ieee754_floating F> struct linear_1d {
27   vector<F> xs;
28   vector<F> ys;
29   mutable usize last_hit{ 0 };
30   extrap mode{ extrap::linear_continue };
31 };

33 template<ieee754_floating F>
34 [[nodiscard]] inline linear_1d<F>
35 make_linear(raw_slice<const F> xs, raw_slice<const F> ys, build_info<F> *info = nullptr) noexcept
36 {
37   linear_1d<F> s{};
38   if ( xs.size() != ys.size() ) {
39     if ( info ) info->status = build_status::size_mismatch;
40     return s;
41   }
42   if ( xs.size() < 2 ) {
43     if ( info ) info->status = build_status::too_few_points;
44     return s;
45   }
46   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
47     if ( info ) info->status = build_status::non_monotonic_x;
48     return s;
49   }
50   const usize n = xs.size();
51   s.xs.reserve(n);
52   s.ys.reserve(n);
53   for ( usize i = 0; i < n; ++i ) {
54     s.xs.emplace_back(xs[i]);
55     s.ys.emplace_back(ys[i]);
56   }
57   if ( info ) info->status = build_status::ok;
58   return s;
59 }

61 template<ieee754_floating F>
62 [[nodiscard, gnu::flatten]] inline F
63 evaluate(const linear_1d<F> &s, F x) noexcept
64 {
65   const usize n = s.xs.size();
66   const F *__restrict__ xs = s.xs.data();
67   const F *__restrict__ ys = s.ys.data();
68   if ( n == 0 ) return F(0);
69   if ( n == 1 ) return ys[0];
70 
71   if ( x <= xs[0] ) {
72     if ( s.mode == extrap::error_value ) return F(0);
73     if ( s.mode == extrap::clamp_to_endpoints ) return ys[0];
74 
75     const F slope = (ys[1] - ys[0]) / (xs[1] - xs[0]);
76     return math::fma<F>(slope, x - xs[0], ys[0]);
77   }
78   if ( x >= xs[n - 1] ) {
79     if ( s.mode == extrap::error_value ) return F(0);
80     if ( s.mode == extrap::clamp_to_endpoints ) return ys[n - 1];
81     const F slope = (ys[n - 1] - ys[n - 2]) / (xs[n - 1] - xs[n - 2]);
82     return math::fma<F>(slope, x - xs[n - 1], ys[n - 1]);
83   }
84 
85   const usize i = __impl_splines_bits::locate_segment<F>(xs, n, x, s.last_hit);
86   const F t = (x - xs[i]) / (xs[i + 1] - xs[i]);
87   return math::fma<F>(t, ys[i + 1] - ys[i], ys[i]);
88 }

90 template<ieee754_floating F>
91 inline void
92 evaluate(const linear_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize n) noexcept
93 {
94   if ( !__impl_splines_bits::is_sorted_nondecreasing<F>(xq, n) ) {
95     for ( usize i = 0; i < n; ++i ) out[i] = evaluate<F>(s, xq[i]);
96     return;
97   }
98 
99   const usize ns = s.xs.size();
100   const F *__restrict__ xs = s.xs.data();
101   const F *__restrict__ ys = s.ys.data();
102   if ( ns < 2 ) {
103     for ( usize i = 0; i < n; ++i ) out[i] = (ns == 0) ? F(0) : ys[0];
104     return;
105   }
106 
107   const F left_slope = (ys[1] - ys[0]) / (xs[1] - xs[0]);
108   const F right_slope = (ys[ns - 1] - ys[ns - 2]) / (xs[ns - 1] - xs[ns - 2]);
109   usize query = 0;
110   while ( query < n && xq[query] <= xs[0] ) {
111     const F x = xq[query];
112     if ( x <= xs[0] ) {
113       if ( s.mode == extrap::error_value ) {
114         out[query] = F(0);
115       } else if ( s.mode == extrap::clamp_to_endpoints ) {
116         out[query] = ys[0];
117       } else {
118         out[query] = math::fma<F>(left_slope, x - xs[0], ys[0]);
119       }
120     }
121     ++query;
122   }
123 
124   usize idx = 0;
125   if ( query < n && xq[query] < xs[ns - 1] ) idx = __impl_splines_bits::locate_segment<F>(xs, ns, xq[query], idx);
126   while ( query < n && xq[query] < xs[ns - 1] ) {
127     while ( idx + 1 < ns - 1 && xq[query] > xs[idx + 1] ) ++idx;
128     const usize begin = query;
129     while ( query < n && xq[query] < xs[ns - 1] && xq[query] <= xs[idx + 1] ) ++query;
130     const F slope = (ys[idx + 1] - ys[idx]) / (xs[idx + 1] - xs[idx]);
131     __spline_arch::linear_batch(xs[idx], ys[idx], slope, xq + begin, out + begin, query - begin);
132   }
133   while ( query < n ) {
134     const F x = xq[query];
135     if ( x >= xs[ns - 1] ) {
136       if ( s.mode == extrap::error_value ) {
137         out[query] = F(0);
138       } else if ( s.mode == extrap::clamp_to_endpoints ) {
139         out[query] = ys[ns - 1];
140       } else {
141         out[query] = math::fma<F>(right_slope, x - xs[ns - 1], ys[ns - 1]);
142       }
143     }
144     ++query;
145   }
146   if ( n != 0 ) {
147     if ( xq[n - 1] <= xs[0] )
148       s.last_hit = 0;
149     else if ( xq[n - 1] >= xs[ns - 1] )
150       s.last_hit = ns - 2;
151     else
152       s.last_hit = idx;
153   }
154 }
