41 template<ieee754_floating F>
42 [[nodiscard]] inline bool
43 solve_cubic_M(const F *xs, const F *ys, usize n, bc_kind bc, F left_slope, F right_slope, F *M_out) noexcept
44 {
45   if ( n < 2 ) return false;
46 
47   if ( n == 2 ) {
48 
49     M_out[0] = F(0);
50     M_out[1] = F(0);
51     return true;
52   }
53 
54   vector<F> a(n - 1, F(0));      // sub-diagonal
55   vector<F> b(n, F(0));          // main diagonal
56   vector<F> c(n - 1, F(0));      // super-diagonal
57   vector<F> d(n, F(0));          // rhs / solution
58   a.set_size(n - 1);
59   b.set_size(n);
60   c.set_size(n - 1);
61   d.set_size(n);
62 
63   for ( usize i = 1; i + 1 < n; ++i ) {
64     const F h0 = xs[i] - xs[i - 1];
65     const F h1 = xs[i + 1] - xs[i];
66     const F del0 = (ys[i] - ys[i - 1]) / h0;
67     const F del1 = (ys[i + 1] - ys[i]) / h1;
68     a[i - 1] = h0;
69     b[i] = F(2) * (h0 + h1);
70     c[i] = h1;
71     d[i] = F(6) * (del1 - del0);
72   }
73 
74   if ( bc == bc_kind::natural ) {
75 
76     b[0] = F(1);
77     c[0] = F(0);
78     d[0] = F(0);
79     a[n - 2] = F(0);
80     b[n - 1] = F(1);
81     d[n - 1] = F(0);
82   } else if ( bc == bc_kind::clamped ) {
83 
84     const F h0 = xs[1] - xs[0];
85     const F del0 = (ys[1] - ys[0]) / h0;
86     b[0] = F(2) * h0;
87     c[0] = h0;
88     d[0] = F(6) * (del0 - left_slope);
89 
90     const F hL = xs[n - 1] - xs[n - 2];
91     const F delL = (ys[n - 1] - ys[n - 2]) / hL;
92     a[n - 2] = hL;
93     b[n - 1] = F(2) * hL;
94     d[n - 1] = F(6) * (right_slope - delL);
95   } else /* bc_kind::not_a_knot */ {
96     if ( n < 4 ) {
97       // n == 3 (n == 2 already returned above): the not-a-knot spline with a
98       // single interior knot degenerates to the unique parabola through the 3
99       // points, whose second derivative is constant
100       const F h0 = xs[1] - xs[0];
101       const F h1 = xs[2] - xs[1];
102       const F del0 = (ys[1] - ys[0]) / h0;
103       const F del1 = (ys[2] - ys[1]) / h1;
104       const F second_dd = (del1 - del0) / (xs[2] - xs[0]);      // f[x0,x1,x2]
105       const F Mconst = F(2) * second_dd;
106       M_out[0] = Mconst;
107       M_out[1] = Mconst;
108       M_out[2] = Mconst;
109       return true;
110     }
111 
112     const usize m = n - 2;
113     vector<F> ai(m == 0 ? 1 : m, F(0));      // sub
114     vector<F> bi(m == 0 ? 1 : m, F(0));      // diag
115     vector<F> ci(m == 0 ? 1 : m, F(0));      // super
116     vector<F> di(m == 0 ? 1 : m, F(0));      // rhs / sol
117     if ( m > 0 ) {
118       ai.set_size(m);
119       bi.set_size(m);
120       ci.set_size(m);
121       di.set_size(m);
122     }
123 
124     {
125 
126       const F h0 = xs[1] - xs[0];
127       const F h1 = xs[2] - xs[1];
128       const F del0 = (ys[1] - ys[0]) / h0;
129       const F del1 = (ys[2] - ys[1]) / h1;
130       bi[0] = h0 + F(2) * h1;
131       if ( m > 1 ) ci[0] = h1 - h0;
132       di[0] = F(6) * h1 * (del1 - del0) / (h0 + h1);
133     }
134 
135     for ( usize i = 2; i + 1 < n - 1; ++i ) {
136       const usize r = i - 1;
137       const F hp = xs[i] - xs[i - 1];
138       const F hn = xs[i + 1] - xs[i];
139       const F delp = (ys[i] - ys[i - 1]) / hp;
140       const F deln = (ys[i + 1] - ys[i]) / hn;
141       ai[r - 1] = hp;
142       bi[r] = F(2) * (hp + hn);
143       ci[r] = hn;
144       di[r] = F(6) * (deln - delp);
145     }
146 
147     if ( m >= 2 ) {
148 
149       const F hLm1 = xs[n - 2] - xs[n - 3];
150       const F hL = xs[n - 1] - xs[n - 2];
151       const F delLm1 = (ys[n - 2] - ys[n - 3]) / hLm1;
152       const F delL = (ys[n - 1] - ys[n - 2]) / hL;
153       ai[m - 2] = hLm1 - hL;
154       bi[m - 1] = F(2) * hLm1 + hL;
155       di[m - 1] = F(6) * hLm1 * (delL - delLm1) / (hLm1 + hL);
156     }
157 
158     linalg::tridiag_solve<F>(ai.data(), bi.data(), ci.data(), di.data(), m);
159 
160     for ( usize r = 0; r < m; ++r ) M_out[r + 1] = di[r];
161     {
162       const F h0 = xs[1] - xs[0];
163       const F h1 = xs[2] - xs[1];
164 
165       M_out[0] = (h0 + h1) / h1 * M_out[1] - h0 / h1 * M_out[2];
166     }
167     {
168       const F hLm1 = xs[n - 2] - xs[n - 3];
169       const F hL = xs[n - 1] - xs[n - 2];
170 
171       M_out[n - 1] = (hL + hLm1) / hLm1 * M_out[n - 2] - hL / hLm1 * M_out[n - 3];
172     }
173     return true;
174   }
175 
176   linalg::tridiag_solve<F>(a.data(), b.data(), c.data(), d.data(), n);
177   for ( usize i = 0; i < n; ++i ) M_out[i] = d[i];
178   return true;
179 }

184 template<ieee754_floating F> struct cubic_system_factor {
185   vector<F> storage;
186   usize size{ 0 };
187   bc_kind bc{ bc_kind::natural };
188   bool quadratic_case{ false };
189 
190   [[nodiscard, gnu::always_inline]] F *
191   lower() noexcept
192   {
193     return storage.data();
194   }
195 
196   [[nodiscard, gnu::always_inline]] F *
197   inverse_diagonal() noexcept
198   {
199     return storage.data() + size;
200   }
201 
202   [[nodiscard, gnu::always_inline]] F *
203   upper() noexcept
204   {
205     return storage.data() + 2 * size;
206   }
207 };

209 template<ieee754_floating F>
210 [[nodiscard]] inline bool
211 factor_cubic_system(const F *xs, usize n, bc_kind bc, cubic_system_factor<F> &factor) noexcept
212 {
213   if ( n < 2 ) return false;
214   factor.bc = bc;
215   factor.quadratic_case = bc == bc_kind::not_a_knot && n == 3;
216   if ( n == 2 || factor.quadratic_case ) {
217     factor.size = 0;
218     return true;
219   }
220   factor.size = bc == bc_kind::not_a_knot ? n - 2 : n;
221   const usize m = factor.size;
222   factor.storage = vector<F>(3 * m, F(0));
223   factor.storage.set_size(3 * m);
224   F *lower = factor.lower();
225   F *diagonal = factor.inverse_diagonal();
226   F *upper = factor.upper();
227 
228   if ( bc == bc_kind::not_a_knot ) {
229     const F h0 = xs[1] - xs[0];
230     const F h1 = xs[2] - xs[1];
231     diagonal[0] = h0 + F(2) * h1;
232     if ( m > 1 ) upper[0] = h1 - h0;
233     for ( usize i = 2; i + 1 < n - 1; ++i ) {
234       const usize row = i - 1;
235       const F hp = xs[i] - xs[i - 1];
236       const F hn = xs[i + 1] - xs[i];
237       lower[row] = hp;
238       diagonal[row] = F(2) * (hp + hn);
239       upper[row] = hn;
240     }
241     if ( m >= 2 ) {
242       const F h0_last = xs[n - 2] - xs[n - 3];
243       const F h1_last = xs[n - 1] - xs[n - 2];
244       lower[m - 1] = h0_last - h1_last;
245       diagonal[m - 1] = F(2) * h0_last + h1_last;
246     }
247   } else {
248     for ( usize i = 1; i + 1 < n; ++i ) {
249       const F h0 = xs[i] - xs[i - 1];
250       const F h1 = xs[i + 1] - xs[i];
251       lower[i] = h0;
252       diagonal[i] = F(2) * (h0 + h1);
253       upper[i] = h1;
254     }
255     if ( bc == bc_kind::natural ) {
256       diagonal[0] = F(1);
257       diagonal[n - 1] = F(1);
258     } else {
259       const F h0 = xs[1] - xs[0];
260       const F h1 = xs[n - 1] - xs[n - 2];
261       diagonal[0] = F(2) * h0;
262       upper[0] = h0;
263       lower[n - 1] = h1;
264       diagonal[n - 1] = F(2) * h1;
265     }
266   }
267 
268   constexpr F tiny = default_eps<F>() * F(4);
269   if ( !(math::fabs(diagonal[0]) > tiny) ) return false;
270   diagonal[0] = F(1) / diagonal[0];
271   if ( m > 1 ) upper[0] *= diagonal[0];
272   for ( usize i = 1; i < m; ++i ) {
273     const F pivot = diagonal[i] - lower[i] * upper[i - 1];
274     if ( !(math::fabs(pivot) > tiny) ) return false;
275     diagonal[i] = F(1) / pivot;
276     if ( i + 1 < m ) upper[i] *= diagonal[i];
277   }
278   return true;
279 }

281 template<ieee754_floating F>
282 inline void
283 solve_cubic_factored(const cubic_system_factor<F> &factor, const F *xs, const F *ys, usize n, F left_slope, F right_slope,
284                      F *moments) noexcept
285 {
286   if ( n == 2 ) {
287     moments[0] = moments[1] = F(0);
288     return;
289   }
290   if ( factor.quadratic_case ) {
291     const F h0 = xs[1] - xs[0];
292     const F h1 = xs[2] - xs[1];
293     const F delta0 = (ys[1] - ys[0]) / h0;
294     const F delta1 = (ys[2] - ys[1]) / h1;
295     const F value = F(2) * (delta1 - delta0) / (xs[2] - xs[0]);
296     moments[0] = moments[1] = moments[2] = value;
297     return;
298   }
299 
300   const usize m = factor.size;
301   F *rhs = moments;
302   if ( factor.bc == bc_kind::not_a_knot ) {
303     const F h0 = xs[1] - xs[0];
304     const F h1 = xs[2] - xs[1];
305     const F delta0 = (ys[1] - ys[0]) / h0;
306     const F delta1 = (ys[2] - ys[1]) / h1;
307     rhs[0] = F(6) * h1 * (delta1 - delta0) / (h0 + h1);
308     for ( usize i = 2; i + 1 < n - 1; ++i ) {
309       const F hp = xs[i] - xs[i - 1];
310       const F hn = xs[i + 1] - xs[i];
311       rhs[i - 1] = F(6) * ((ys[i + 1] - ys[i]) / hn - (ys[i] - ys[i - 1]) / hp);
312     }
313     if ( m >= 2 ) {
314       const F hp = xs[n - 2] - xs[n - 3];
315       const F hn = xs[n - 1] - xs[n - 2];
316       rhs[m - 1] = F(6) * hp * ((ys[n - 1] - ys[n - 2]) / hn - (ys[n - 2] - ys[n - 3]) / hp) / (hp + hn);
317     }
318   } else {
319     for ( usize i = 1; i + 1 < n; ++i ) {
320       const F hp = xs[i] - xs[i - 1];
321       const F hn = xs[i + 1] - xs[i];
322       rhs[i] = F(6) * ((ys[i + 1] - ys[i]) / hn - (ys[i] - ys[i - 1]) / hp);
323     }
324     if ( factor.bc == bc_kind::natural ) {
325       rhs[0] = rhs[n - 1] = F(0);
326     } else {
327       const F h0 = xs[1] - xs[0];
328       const F hn = xs[n - 1] - xs[n - 2];
329       rhs[0] = F(6) * ((ys[1] - ys[0]) / h0 - left_slope);
330       rhs[n - 1] = F(6) * (right_slope - (ys[n - 1] - ys[n - 2]) / hn);
331     }
332   }
333 
334   const F *lower = factor.storage.data();
335   const F *inverse = factor.storage.data() + m;
336   const F *upper = factor.storage.data() + 2 * m;
337   rhs[0] *= inverse[0];
338   for ( usize i = 1; i < m; ++i ) rhs[i] = (rhs[i] - lower[i] * rhs[i - 1]) * inverse[i];
339   for ( usize i = m - 1; i-- > 0; ) rhs[i] -= upper[i] * rhs[i + 1];
340 
341   if ( factor.bc == bc_kind::not_a_knot ) {
342     for ( usize i = m; i-- > 0; ) moments[i + 1] = moments[i];
343     const F h0 = xs[1] - xs[0];
344     const F h1 = xs[2] - xs[1];
345     moments[0] = (h0 + h1) * moments[1] / h1 - h0 * moments[2] / h1;
346     const F hp = xs[n - 2] - xs[n - 3];
347     const F hn = xs[n - 1] - xs[n - 2];
348     moments[n - 1] = (hn + hp) * moments[n - 2] / hp - hn * moments[n - 3] / hp;
349   }
350 }

354 template<ieee754_floating F>
355 [[nodiscard]] inline cubic_spline_1d<F>
356 make_cubic(raw_slice<const F> xs, raw_slice<const F> ys, bc_kind bc = bc_kind::natural, F left_slope = F(0), F right_slope = F(0),
357            build_info<F> *info = nullptr) noexcept
358 {
359   cubic_spline_1d<F> s{};
360   s.bc = bc;
361   if ( xs.size() != ys.size() ) {
362     if ( info ) info->status = build_status::size_mismatch;
363     return s;
364   }
365   if ( xs.size() < 2 ) {
366     if ( info ) info->status = build_status::too_few_points;
367     return s;
368   }
369   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
370     if ( info ) info->status = build_status::non_monotonic_x;
371     return s;
372   }
373   const usize n = xs.size();
374 
375   // relative-spacing guard
376   {
377     const F span = xs[n - 1] - xs[0];
378     const F min_h = F(1e-12) * span;
379     for ( usize i = 0; i + 1 < n; ++i ) {
380       if ( xs[i + 1] - xs[i] <= min_h ) {
381         if ( info ) info->status = build_status::degenerate;
382         return s;
383       }
384     }
385   }
386 
387   vector<F> M(n, F(0));
388   M.set_size(n);
389   __impl_cubic_1d::cubic_system_factor<F> factor;
390   if ( !__impl_cubic_1d::factor_cubic_system<F>(xs.ptr, n, bc, factor) ) {
391     if ( info ) info->status = build_status::singular_system;
392     return s;
393   }
394   __impl_cubic_1d::solve_cubic_factored<F>(factor, xs.ptr, ys.ptr, n, left_slope, right_slope, M.data());
395 
396   s.xs.reserve(n);
397   s.seg.reserve(n - 1);
398   for ( usize i = 0; i < n; ++i ) s.xs.emplace_back(xs[i]);
399   poly_coeffs<F, 3> zero{};
400   zero.data[0] = zero.data[1] = zero.data[2] = zero.data[3] = F(0);
401   for ( usize i = 0; i + 1 < n; ++i ) s.seg.emplace_back(zero);
402   __impl_splines_bits::build_cubic_segments<F>(xs.ptr, ys.ptr, M.data(), s.seg.data(), n);
403 
404   if ( info ) info->status = build_status::ok;
405   return s;
406 }

408 template<ieee754_floating F>
409 [[nodiscard, gnu::flatten]] inline F
410 evaluate(const cubic_spline_1d<F> &s, F x) noexcept
411 {
412   const usize n = s.xs.size();
413   const F *__restrict__ xs = s.xs.data();
414   const auto *__restrict__ seg = s.seg.data();
415   if ( n == 0 ) return F(0);
416   if ( n == 1 ) return seg ? seg[0].data[0] : F(0);
417 
418   if ( x <= xs[0] ) {
419     if ( s.mode == extrap::error_value ) return F(0);
420     if ( s.mode == extrap::clamp_to_endpoints ) return seg[0].data[0];
421 
422     const F slope = seg[0].data[1];
423     return math::fma<F>(slope, x - xs[0], seg[0].data[0]);
424   }
425   if ( x >= xs[n - 1] ) {
426     if ( s.mode == extrap::error_value ) return F(0);
427     if ( s.mode == extrap::clamp_to_endpoints ) {
428       const auto &p = seg[n - 2];
429       const F t = xs[n - 1] - xs[n - 2];
430       return __impl_splines_bits::eval_cubic_local<F>(p, t);
431     }
432     const auto &p = seg[n - 2];
433     const F t_end = xs[n - 1] - xs[n - 2];
434     const F y_end = __impl_splines_bits::eval_cubic_local<F>(p, t_end);
435     const F slope_end = __impl_splines_bits::eval_cubic_deriv1_local<F>(p, t_end);
436     return math::fma<F>(slope_end, x - xs[n - 1], y_end);
437   }
438 
439   const usize i = __impl_splines_bits::locate_segment<F>(xs, n, x, s.last_hit);
440   return __impl_splines_bits::eval_cubic_local<F>(seg[i], x - xs[i]);
441 }

443 template<ieee754_floating F>
444 inline void
445 evaluate(const cubic_spline_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize n) noexcept
446 {
447   if ( !__impl_splines_bits::is_sorted_nondecreasing<F>(xq, n) ) {
448     for ( usize i = 0; i < n; ++i ) out[i] = evaluate<F>(s, xq[i]);
449     return;
450   }
451 
452   const usize ns = s.xs.size();
453   if ( ns < 2 ) {
454     for ( usize i = 0; i < n; ++i ) out[i] = evaluate<F>(s, xq[i]);
455     return;
456   }
457   const F *__restrict__ xs = s.xs.data();
458   const auto *__restrict__ seg = s.seg.data();
459 
460   usize query = 0;
461   while ( query < n && xq[query] <= xs[0] ) {
462     out[query] = evaluate<F>(s, xq[query]);
463     ++query;
464   }
465 
466   usize idx = 0;
467   if ( query < n && xq[query] < xs[ns - 1] ) idx = __impl_splines_bits::locate_segment<F>(xs, ns, xq[query], idx);
468   while ( query < n && xq[query] < xs[ns - 1] ) {
469     while ( idx + 1 < ns - 1 && xq[query] > xs[idx + 1] ) ++idx;
470     const usize begin = query;
471     while ( query < n && xq[query] < xs[ns - 1] && xq[query] <= xs[idx + 1] ) ++query;
472     __spline_arch::cubic_horner_batch(seg[idx], xs[idx], xq + begin, out + begin, query - begin);
473   }
474   while ( query < n ) {
475     out[query] = evaluate<F>(s, xq[query]);
476     ++query;
477   }
478   if ( n != 0 ) {
479     if ( xq[n - 1] <= xs[0] )
480       s.last_hit = 0;
481     else if ( xq[n - 1] >= xs[ns - 1] )
482       s.last_hit = ns - 2;
483     else
484       s.last_hit = idx;
485   }
486 }

488 template<ieee754_floating F>
489 [[nodiscard]] inline F
490 derivative(const cubic_spline_1d<F> &s, F x, u32 order = 1) noexcept
491 {
492   const usize n = s.xs.size();
493   if ( n < 2 ) return F(0);
494   const F *__restrict__ xs = s.xs.data();
495   const auto *__restrict__ seg = s.seg.data();
496   if ( order == 0 ) return evaluate<F>(s, x);
497   const bool outside = x < xs[0] || x > xs[n - 1];
498   if ( outside && (s.mode != extrap::linear_continue || order != 1) ) return F(0);
499   if ( x <= xs[0] ) {
500     if ( order == 1 ) return seg[0].data[1];
501     if ( order == 2 ) return F(2) * seg[0].data[2];
502     return F(0);
503   }
504   if ( x >= xs[n - 1] ) {
505     const auto &p = seg[n - 2];
506     const F t = xs[n - 1] - xs[n - 2];
507     if ( order == 1 ) return __impl_splines_bits::eval_cubic_deriv1_local<F>(p, t);
508     if ( order == 2 ) return __impl_splines_bits::eval_cubic_deriv2_local<F>(p, t);
509     return F(0);
510   }
511   const usize i = __impl_splines_bits::locate_segment<F>(xs, n, x, s.last_hit);
512   const F t = x - xs[i];
513   if ( order == 1 ) return __impl_splines_bits::eval_cubic_deriv1_local<F>(seg[i], t);
514   if ( order == 2 ) return __impl_splines_bits::eval_cubic_deriv2_local<F>(seg[i], t);
515 
516   if ( order == 3 ) return F(6) * seg[i].data[3];
517   return F(0);
518 }
519 
520 template<ieee754_floating F>
521 [[nodiscard]] inline F
522 integral(const cubic_spline_1d<F> &s, F a, F b) noexcept
523 {
524   if ( a == b ) return F(0);
525   const F sign = (a < b) ? F(1) : F(-1);
526   if ( a > b ) {
527     const F t = a;
528     a = b;
529     b = t;
530   }
531   const usize n = s.xs.size();
532   if ( n < 2 ) return F(0);
533   const F *__restrict__ xs = s.xs.data();
534   const auto *__restrict__ seg = s.seg.data();
535 
536   usize ia = 0, ib = 0;
537   ia = __impl_splines_bits::locate_segment<F>(xs, n, a < xs[0] ? xs[0] : a, s.last_hit);
538   ib = __impl_splines_bits::locate_segment<F>(xs, n, b > xs[n - 1] ? xs[n - 1] : b, s.last_hit);
539 
540   if ( a < xs[0] ) a = xs[0];
541   if ( a > xs[n - 1] ) a = xs[n - 1];
542   if ( b < xs[0] ) b = xs[0];
543   if ( b > xs[n - 1] ) b = xs[n - 1];
544 
545   F sum = F(0);
546   if ( ia == ib ) {
547     const auto &p = seg[ia];
548     const F lo = a - xs[ia];
549     const F hi = b - xs[ia];
550     sum = __impl_splines_bits::eval_cubic_antideriv_local<F>(p, hi) - __impl_splines_bits::eval_cubic_antideriv_local<F>(p, lo);
551   } else {
552 
553     {
554       const auto &p = seg[ia];
555       const F lo = a - xs[ia];
556       const F hi = xs[ia + 1] - xs[ia];
557       sum += __impl_splines_bits::eval_cubic_antideriv_local<F>(p, hi) - __impl_splines_bits::eval_cubic_antideriv_local<F>(p, lo);
558     }
559 
560     for ( usize i = ia + 1; i < ib; ++i ) {
561       const auto &p = seg[i];
562       const F hi = xs[i + 1] - xs[i];
563       sum += __impl_splines_bits::eval_cubic_antideriv_local<F>(p, hi);
564     }
565 
566     {
567       const auto &p = seg[ib];
568       const F hi = b - xs[ib];
569       sum += __impl_splines_bits::eval_cubic_antideriv_local<F>(p, hi);
570     }
571   }
572   return sign * sum;
573 }
