91 template<ieee754_floating F>
92 [[nodiscard]] inline F
93 evaluate(const nearest_1d<F> &s, F x, spline_cursor &cursor) noexcept
94 {
95   const usize n = s.xs.size();
96   if ( n < 2 || x <= s.xs[0] || x >= s.xs[n - 1] ) return evaluate<F>(s, x);
97   const usize segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, x, cursor.segment);
98   return ((x + x) < (s.xs[segment] + s.xs[segment + 1])) ? s.ys[segment] : s.ys[segment + 1];
99 }

101 template<ieee754_floating F>
102 [[nodiscard]] inline F
103 evaluate(const linear_1d<F> &s, F x, spline_cursor &cursor) noexcept
104 {
105   const usize n = s.xs.size();
106   if ( n < 2 || x <= s.xs[0] || x >= s.xs[n - 1] ) return evaluate<F>(s, x);
107   const usize segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, x, cursor.segment);
108   const F fraction = (x - s.xs[segment]) / (s.xs[segment + 1] - s.xs[segment]);
109   return math::fma<F>(fraction, s.ys[segment + 1] - s.ys[segment], s.ys[segment]);
110 }

122 template<ieee754_floating F>
123 [[nodiscard]] inline F
124 evaluate(const bspline<F> &s, F x, spline_cursor &cursor) noexcept
125 {
126   const usize n_ctrl = s.ctrl.size();
127   if ( n_ctrl == 0 || s.degree == 0 ) return F(0);
128   if ( x < s.knots[s.degree] ) x = s.knots[s.degree];
129   if ( x > s.knots[n_ctrl] ) x = s.knots[n_ctrl];
130   const usize span = __impl_bspline::bspline_span<F>(s.knots.data(), n_ctrl, s.degree, x, cursor.segment);
131   switch ( s.degree ) {
132   case 1:
133     return __impl_bspline::deboor_fixed<1, F>(s.knots.data(), s.ctrl.data(), span, x);
134   case 2:
135     return __impl_bspline::deboor_fixed<2, F>(s.knots.data(), s.ctrl.data(), span, x);
136   case 3:
137     return __impl_bspline::deboor_fixed<3, F>(s.knots.data(), s.ctrl.data(), span, x);
138   case 4:
139     return __impl_bspline::deboor_fixed<4, F>(s.knots.data(), s.ctrl.data(), span, x);
140   case 5:
141     return __impl_bspline::deboor_fixed<5, F>(s.knots.data(), s.ctrl.data(), span, x);
142   default:
143     break;
144   }
145   F d[bspline_max_degree + 1];
146   for ( u32 j = 0; j <= s.degree; ++j ) d[j] = s.ctrl[span - s.degree + j];
147   for ( u32 r = 1; r <= s.degree; ++r )
148     for ( u32 j = s.degree; j >= r; --j ) {
149       const F denom = s.knots[span + 1 + j - r] - s.knots[span + j - s.degree];
150       const F alpha = denom > F(0) ? (x - s.knots[span + j - s.degree]) / denom : F(0);
151       d[j] = math::fma<F>(alpha, d[j] - d[j - 1], d[j - 1]);
152     }
153   return d[s.degree];
154 }

156 template<ieee754_floating F, usize D>
157 [[nodiscard]] inline vec<F, D>
158 evaluate(const cubic_curve_nd<F, D> &c, F t, spline_cursor &cursor) noexcept
159 {
160   const usize n = c.ts.size();
161   if ( n < 2 || t <= c.ts[0] || t >= c.ts[n - 1] ) return evaluate<F, D>(c, t);
162   const usize segment = __impl_splines_bits::locate_segment<F>(c.ts.data(), n, t, cursor.segment);
163   const F u = t - c.ts[segment];
164   vec<F, D> out{};
165   __spline_arch::curve_horner(c.seg[segment].a.data, c.seg[segment].b.data, c.seg[segment].c.data, c.seg[segment].d.data, u, out.data, D);
166   return out;
167 }

178 template<ieee754_floating F>
179 inline void
180 evaluate_sorted(const cubic_spline_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize count) noexcept
181 {
182   const usize n = s.xs.size();
183   if ( n < 2 ) {
184     for ( usize i = 0; i < count; ++i ) out[i] = evaluate<F>(s, xq[i]);
185     return;
186   }
187   usize query = 0;
188   while ( query < count && xq[query] <= s.xs[0] ) {
189     out[query] = evaluate<F>(s, xq[query]);
190     ++query;
191   }
192   usize segment = 0;
193   if ( query < count && xq[query] < s.xs[n - 1] ) segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, xq[query], segment);
194   while ( query < count && xq[query] < s.xs[n - 1] ) {
195     while ( segment + 1 < n - 1 && xq[query] > s.xs[segment + 1] ) ++segment;
196     const usize begin = query;
197     while ( query < count && xq[query] < s.xs[n - 1] && xq[query] <= s.xs[segment + 1] ) ++query;
198     __spline_arch::cubic_horner_batch(s.seg[segment], s.xs[segment], xq + begin, out + begin, query - begin);
199   }
200   while ( query < count ) {
201     out[query] = evaluate<F>(s, xq[query]);
202     ++query;
203   }
204   if ( count ) {
205     if ( xq[count - 1] <= s.xs[0] )
206       s.last_hit = 0;
207     else if ( xq[count - 1] >= s.xs[n - 1] )
208       s.last_hit = n - 2;
209     else
210       s.last_hit = segment;
211   }
212 }

214 template<ieee754_floating F>
215 inline void
216 evaluate_sorted(const linear_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize count) noexcept
217 {
218   const usize n = s.xs.size();
219   if ( n < 2 ) {
220     for ( usize i = 0; i < count; ++i ) out[i] = evaluate<F>(s, xq[i]);
221     return;
222   }
223   usize query = 0;
224   while ( query < count && xq[query] <= s.xs[0] ) {
225     out[query] = evaluate<F>(s, xq[query]);
226     ++query;
227   }
228   usize segment = 0;
229   if ( query < count && xq[query] < s.xs[n - 1] ) segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, xq[query], segment);
230   while ( query < count && xq[query] < s.xs[n - 1] ) {
231     while ( segment + 1 < n - 1 && xq[query] > s.xs[segment + 1] ) ++segment;
232     const usize begin = query;
233     while ( query < count && xq[query] < s.xs[n - 1] && xq[query] <= s.xs[segment + 1] ) ++query;
234     const F slope = (s.ys[segment + 1] - s.ys[segment]) / (s.xs[segment + 1] - s.xs[segment]);
235     __spline_arch::linear_batch(s.xs[segment], s.ys[segment], slope, xq + begin, out + begin, query - begin);
236   }
237   while ( query < count ) {
238     out[query] = evaluate<F>(s, xq[query]);
239     ++query;
240   }
241   if ( count ) s.last_hit = xq[count - 1] >= s.xs[n - 1] ? n - 2 : segment;
242 }

244 template<ieee754_floating F>
245 inline void
246 evaluate_sorted(const nearest_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize count) noexcept
247 {
248   const usize n = s.xs.size();
249   if ( n < 2 ) {
250     for ( usize i = 0; i < count; ++i ) out[i] = evaluate<F>(s, xq[i]);
251     return;
252   }
253   usize segment = 0;
254   for ( usize i = 0; i < count; ++i ) {
255     if ( xq[i] <= s.xs[0] || xq[i] >= s.xs[n - 1] ) {
256       out[i] = evaluate<F>(s, xq[i]);
257       if ( xq[i] >= s.xs[n - 1] ) segment = n - 2;
258       continue;
259     }
260     while ( segment + 1 < n - 1 && xq[i] > s.xs[segment + 1] ) ++segment;
261     out[i] = ((xq[i] + xq[i]) < (s.xs[segment] + s.xs[segment + 1])) ? s.ys[segment] : s.ys[segment + 1];
262   }
263   if ( count ) s.last_hit = segment;
264 }

276 template<ieee754_floating F>
277 inline void
278 evaluate_streaming(const cubic_spline_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize count) noexcept
279 {
280   if ( !__impl_splines_bits::is_sorted_nondecreasing<F>(xq, count) || s.xs.size() < 2 ) {
281     for ( usize i = 0; i < count; ++i ) out[i] = evaluate<F>(s, xq[i]);
282     return;
283   }
284   const usize n = s.xs.size();
285   usize query = 0;
286   while ( query < count && xq[query] <= s.xs[0] ) {
287     out[query] = evaluate<F>(s, xq[query]);
288     ++query;
289   }
290   usize segment = 0;
291   if ( query < count && xq[query] < s.xs[n - 1] ) segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, xq[query], segment);
292   while ( query < count && xq[query] < s.xs[n - 1] ) {
293     while ( segment + 1 < n - 1 && xq[query] > s.xs[segment + 1] ) ++segment;
294     const usize begin = query;
295     while ( query < count && xq[query] < s.xs[n - 1] && xq[query] <= s.xs[segment + 1] ) ++query;
296     __spline_arch::cubic_horner_stream_batch(s.seg[segment], s.xs[segment], xq + begin, out + begin, query - begin);
297   }
298   while ( query < count ) {
299     out[query] = evaluate<F>(s, xq[query]);
300     ++query;
301   }
302   if ( count * sizeof(F) >= 64 ) __spline_arch::spline_store_fence();
303 }

305 template<ieee754_floating F>
306 inline void
307 evaluate_streaming(const linear_1d<F> &s, const F *__restrict__ xq, F *__restrict__ out, usize count) noexcept
308 {
309   if ( !__impl_splines_bits::is_sorted_nondecreasing<F>(xq, count) || s.xs.size() < 2 ) {
310     for ( usize i = 0; i < count; ++i ) out[i] = evaluate<F>(s, xq[i]);
311     return;
312   }
313   const usize n = s.xs.size();
314   const F left_slope = (s.ys[1] - s.ys[0]) / (s.xs[1] - s.xs[0]);
315   const F right_slope = (s.ys[n - 1] - s.ys[n - 2]) / (s.xs[n - 1] - s.xs[n - 2]);
316   usize query = 0;
317   while ( query < count && xq[query] <= s.xs[0] ) {
318     const F x = xq[query];
319     out[query] = s.mode == extrap::error_value
320                      ? F(0)
321                      : (s.mode == extrap::clamp_to_endpoints ? s.ys[0] : math::fma<F>(left_slope, x - s.xs[0], s.ys[0]));
322     ++query;
323   }
324   usize segment = 0;
325   if ( query < count && xq[query] < s.xs[n - 1] ) segment = __impl_splines_bits::locate_segment<F>(s.xs.data(), n, xq[query], segment);
326   while ( query < count && xq[query] < s.xs[n - 1] ) {
327     while ( segment + 1 < n - 1 && xq[query] > s.xs[segment + 1] ) ++segment;
328     const usize begin = query;
329     while ( query < count && xq[query] < s.xs[n - 1] && xq[query] <= s.xs[segment + 1] ) ++query;
330     const F slope = (s.ys[segment + 1] - s.ys[segment]) / (s.xs[segment + 1] - s.xs[segment]);
331     __spline_arch::linear_stream_batch(s.xs[segment], s.ys[segment], slope, xq + begin, out + begin, query - begin);
332   }
333   while ( query < count ) {
334     const F x = xq[query];
335     out[query] = s.mode == extrap::error_value
336                      ? F(0)
337                      : (s.mode == extrap::clamp_to_endpoints ? s.ys[n - 1] : math::fma<F>(right_slope, x - s.xs[n - 1], s.ys[n - 1]));
338     ++query;
339   }
340   if ( count * sizeof(F) >= 64 ) __spline_arch::spline_store_fence();
341 }
