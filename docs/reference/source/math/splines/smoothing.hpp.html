33 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
34 // reinsch_workspace
35 //
36 //  h         h_i = xs[i+1] - xs[i]
37 //  r_diag    diagonal of R
38 //  r_d1      super diagonal of R
39 //  m_diag    diagonal of Q W^{-1} Q^T
40 //  m_d1      offset-1 of Q W^{-1} Q^T
41 //  m_d2      offset-2 of Q W^{-1} Q^T
42 //  rhs0      Q y
43 //  c_diag    diagonal of A = R + \ M (per \)
44 //  c_d1      offset-1 of A
45 //  c_d2      offset-2 of A
46 //  gamma     interior second derivatives γ
47 //  z_diag    Z = A^{-1} diagonal
48 //  z_d1      Z offset-1
49 //  z_d2      Z offset-2
50 //  y_hat     smoothed values
51 //  q_t_g     (Q^T y) projected back to n-vector
52

82 template<ieee754_floating F> struct reinsch_workspace {
83   vector<F> storage;
84   workspace_array<F> h, inv_h, inv_w;
85   workspace_array<F> r_diag, r_d1;
86   workspace_array<F> m_diag, m_d1, m_d2;
87   workspace_array<F> rhs0;
88   workspace_array<F> c_diag, c_d1, c_d2;
89   workspace_array<F> gamma;
90   workspace_array<F> z_diag, z_d1, z_d2;
91   workspace_array<F> y_hat;
92   workspace_array<F> q_t_g;
93 };

95 template<ieee754_floating F>
96 inline void
97 reinsch_update_rhs(reinsch_workspace<F> &ws, const F *__restrict__ ys, usize n) noexcept
98 {
99   for ( usize i = 0; i + 2 < n; ++i ) {
100     const F qi = ws.inv_h[i];
101     const F qi1 = ws.inv_h[i + 1];
102     ws.rhs0[i] = qi * ys[i] - (qi + qi1) * ys[i + 1] + qi1 * ys[i + 2];
103   }
104 }

106 template<ieee754_floating F>
107 inline void
108 reinsch_precompute(reinsch_workspace<F> &ws, const F *__restrict__ xs, const F *__restrict__ ys, const F *w, bool uniform_w,
109                    usize n) noexcept
110 {
111   const usize m = n - 2;
112   const usize m1 = m ? m - 1 : 0;
113   const usize m2 = m > 1 ? m - 2 : 0;
114   const usize total = (n - 1) * 2 + n + m * 6 + m1 * 4 + m2 * 3 + n * 2;
115   ws.storage = vector<F>(total, F(0));
116   ws.storage.set_size(total);
117   usize offset = 0;
118   auto bind = [&](workspace_array<F> &array, usize size) {
119     array.ptr = ws.storage.data() + offset;
120     array.length = size;
121     offset += size;
122   };
123   bind(ws.h, n - 1);
124   bind(ws.inv_h, n - 1);
125   bind(ws.inv_w, n);
126   bind(ws.r_diag, m);
127   bind(ws.r_d1, m1);
128   bind(ws.m_diag, m);
129   bind(ws.m_d1, m1);
130   bind(ws.m_d2, m2);
131   bind(ws.rhs0, m);
132   bind(ws.c_diag, m);
133   bind(ws.c_d1, m1);
134   bind(ws.c_d2, m2);
135   bind(ws.gamma, m);
136   bind(ws.z_diag, m);
137   bind(ws.z_d1, m1);
138   bind(ws.z_d2, m2);
139   bind(ws.y_hat, n);
140   bind(ws.q_t_g, n);
141 
142   for ( usize i = 0; i + 1 < n; ++i ) {
143     ws.h[i] = xs[i + 1] - xs[i];
144     ws.inv_h[i] = F(1) / ws.h[i];
145   }
146   for ( usize i = 0; i < n; ++i ) ws.inv_w[i] = uniform_w ? F(1) : F(1) / w[i];
147 
148   const F third = F(1) / F(3);
149   const F sixth = F(1) / F(6);
150 
151   const F *__restrict__ h = ws.h.data();
152   const F *__restrict__ inv_h = ws.inv_h.data();
153   const F *__restrict__ inv_w = ws.inv_w.data();
154   for ( usize i = 0; i + 2 < n; ++i ) {
155     const F hi = h[i];
156     const F hi1 = h[i + 1];
157     const F qi = inv_h[i];
158     const F qi1 = inv_h[i + 1];
159     const F qm = qi + qi1;
160 
161     ws.r_diag[i] = (hi + hi1) * third;
162     ws.m_diag[i] = qi * qi * inv_w[i] + qm * qm * inv_w[i + 1] + qi1 * qi1 * inv_w[i + 2];
163 
164     if ( i + 3 < n ) {
165       ws.r_d1[i] = hi1 * sixth;
166       const F qj = qi1;
167       const F qjm = qi1 + inv_h[i + 2];
168       ws.m_d1[i] = -qm * qj * inv_w[i + 1] - qj * qjm * inv_w[i + 2];
169     }
170     if ( i + 4 < n ) {
171       ws.m_d2[i] = qi1 * inv_h[i + 2] * inv_w[i + 2];
172     }
173   }
174   reinsch_update_rhs<F>(ws, ys, n);
175 }

180 template<ieee754_floating F>
181 [[nodiscard]] inline bool
182 reinsch_factor_lambda(reinsch_workspace<F> &ws, usize n, F lambda) noexcept
183 {
184   const usize m = n - 2;
185   for ( usize i = 0; i < m; ++i ) ws.c_diag[i] = ws.r_diag[i] + lambda * ws.m_diag[i];
186   for ( usize i = 0; i + 1 < m; ++i ) ws.c_d1[i] = ws.r_d1[i] + lambda * ws.m_d1[i];
187   for ( usize i = 0; i + 2 < m; ++i ) ws.c_d2[i] = lambda * ws.m_d2[i];
188   return linalg::pent_spd_factor<F>(ws.c_d2.data(), ws.c_d1.data(), ws.c_diag.data(), m);
189 }

191 template<ieee754_floating F>
192 inline void
193 reinsch_apply_factored(reinsch_workspace<F> &ws, const F *__restrict__ ys, usize n, F lambda, F *out_rss) noexcept
194 {
195   const usize m = n - 2;
196   for ( usize i = 0; i < m; ++i ) ws.gamma[i] = ws.rhs0[i];
197   linalg::pent_spd_solve_factored<F>(ws.c_d2.data(), ws.c_d1.data(), ws.c_diag.data(), ws.gamma.data(), m);
198 
199   for ( usize k = 0; k < n; ++k ) ws.q_t_g[k] = F(0);
200   for ( usize i = 0; i < m; ++i ) {
201     const F value = ws.gamma[i];
202     const F qi = ws.inv_h[i];
203     const F qi1 = ws.inv_h[i + 1];
204     ws.q_t_g[i] = math::fma<F>(value, qi, ws.q_t_g[i]);
205     ws.q_t_g[i + 1] = math::fma<F>(-value, qi + qi1, ws.q_t_g[i + 1]);
206     ws.q_t_g[i + 2] = math::fma<F>(value, qi1, ws.q_t_g[i + 2]);
207   }
208   F rss = F(0);
209   for ( usize k = 0; k < n; ++k ) {
210     const F correction = lambda * ws.inv_w[k] * ws.q_t_g[k];
211     ws.y_hat[k] = ys[k] - correction;
212     rss = math::fma<F>(correction, correction, rss);
213   }
214   if ( out_rss ) *out_rss = rss;
215 }

217 template<ieee754_floating F>
218 [[nodiscard]] inline F
219 reinsch_trace_inverse_m(reinsch_workspace<F> &ws, usize n) noexcept
220 {
221   const usize m = n - 2;
222   linalg::pent_spd_takahashi<F>(ws.c_d2.data(), ws.c_d1.data(), ws.c_diag.data(), m, ws.z_diag.data(), ws.z_d1.data(), ws.z_d2.data());
223   F t0 = F(0), t1 = F(0), t2 = F(0), t3 = F(0);
224   usize i = 0;
225   for ( ; i + 4 <= m; i += 4 ) {
226     t0 = math::fma<F>(ws.z_diag[i], ws.m_diag[i], t0);
227     t1 = math::fma<F>(ws.z_diag[i + 1], ws.m_diag[i + 1], t1);
228     t2 = math::fma<F>(ws.z_diag[i + 2], ws.m_diag[i + 2], t2);
229     t3 = math::fma<F>(ws.z_diag[i + 3], ws.m_diag[i + 3], t3);
230   }
231   for ( ; i < m; ++i ) t0 = math::fma<F>(ws.z_diag[i], ws.m_diag[i], t0);
232   F trace = (t0 + t1) + (t2 + t3);
233   for ( usize j = 0; j + 1 < m; ++j ) trace += F(2) * ws.z_d1[j] * ws.m_d1[j];
234   for ( usize j = 0; j + 2 < m; ++j ) trace += F(2) * ws.z_d2[j] * ws.m_d2[j];
235   return trace;
236 }

238 template<ieee754_floating F>
239 [[nodiscard]] inline bool
240 reinsch_solve_one_lambda(reinsch_workspace<F> &ws, const F *__restrict__ xs, const F *__restrict__ ys, const F *w, bool uniform_w, usize n,
241                          F lambda, bool need_trace, F *out_rss, F *out_tr) noexcept
242 {
243   (void)xs;
244   (void)w;
245   (void)uniform_w;
246   if ( !reinsch_factor_lambda<F>(ws, n, lambda) ) return false;
247   reinsch_apply_factored<F>(ws, ys, n, lambda, out_rss);
248   if ( need_trace && out_tr ) *out_tr = reinsch_trace_inverse_m<F>(ws, n);
249   return true;
250 }

254 template<ieee754_floating F>
255 [[nodiscard, gnu::always_inline]] inline F
256 gcv_score(F rss, F lambda, F tr_inv_m, usize n) noexcept
257 {
258   const F denom_term = lambda * tr_inv_m;
259   if ( !(denom_term > F(0)) ) return F(1e30);
260   return F(n) * rss / (denom_term * denom_term);
261 }

272 template<ieee754_floating F>
273 [[nodiscard]] inline F
274 gcv_search(reinsch_workspace<F> &ws, const F *xs, const F *ys, const F *w, bool uniform_w, usize n, build_info<F> *info) noexcept
275 {
276   const F phi = (math::fsqrt(F(5)) - F(1)) * F(0.5);
277   F a = F(-12);
278   F b = F(12);
279   F c = b - phi * (b - a);
280   F d = a + phi * (b - a);
281 
282   auto eval_at = [&](F log_lambda) -> F {
283     const F lambda = math::mk::exp_ns::exp10<F>(log_lambda);
284     F rss = F(0), tr = F(0);
285     if ( !reinsch_solve_one_lambda<F>(ws, xs, ys, w, uniform_w, n, lambda, true, &rss, &tr) ) return F(1e30);
286     return gcv_score<F>(rss, lambda, tr, n);
287   };
288 
289   F fc = eval_at(c);
290   F fd = eval_at(d);
291   usize iter = 0;
292   for ( iter = 0; iter < 60; ++iter ) {
293     if ( b - a < F(1e-3) ) break;
294     if ( fc < fd ) {
295       b = d;
296       d = c;
297       fd = fc;
298       c = b - phi * (b - a);
299       fc = eval_at(c);
300     } else {
301       a = c;
302       c = d;
303       fc = fd;
304       d = a + phi * (b - a);
305       fd = eval_at(d);
306     }
307   }
308   if ( info ) info->n_iterations = iter + 2;      // +2 for the initial fc, fd
309   const F log_lambda_opt = F(0.5) * (a + b);
310   return math::mk::exp_ns::exp10<F>(log_lambda_opt);
311 }

315 template<ieee754_floating F>
316 [[nodiscard]] inline cubic_spline_1d<F>
317 make_smoothing(raw_slice<const F> xs, raw_slice<const F> ys, raw_slice<const F> w, F lambda, build_info<F> *info = nullptr) noexcept
318 {
319   cubic_spline_1d<F> s{};
320   s.bc = bc_kind::natural;
321 
322   if ( xs.size() != ys.size() ) {
323     if ( info ) info->status = build_status::size_mismatch;
324     return s;
325   }
326   if ( xs.size() < 4 ) {
327     if ( info ) info->status = build_status::too_few_points;
328     return s;
329   }
330   if ( w.size() != 0 && w.size() != xs.size() ) {
331     if ( info ) info->status = build_status::size_mismatch;
332     return s;
333   }
334   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
335     if ( info ) info->status = build_status::non_monotonic_x;
336     return s;
337   }
338   const usize n = xs.size();
339   const bool uniform_w = (w.size() == 0);
340 
341   if ( lambda == F(0) ) {
342     return make_cubic<F>(xs, ys, bc_kind::natural, F(0), F(0), info);
343   }
344 
345   __impl_smoothing::reinsch_workspace<F> ws;
346   __impl_smoothing::reinsch_precompute<F>(ws, xs.ptr, ys.ptr, w.ptr, uniform_w, n);
347 
348   F lam = lambda;
349   if ( lambda < F(0) ) lam = __impl_smoothing::gcv_search<F>(ws, xs.ptr, ys.ptr, w.ptr, uniform_w, n, info);
350 
351   F rss = F(0);
352   if ( !__impl_smoothing::reinsch_solve_one_lambda<F>(ws, xs.ptr, ys.ptr, w.ptr, uniform_w, n, lam, false, &rss, nullptr) ) {
353     if ( info ) info->status = build_status::singular_system;
354     return s;
355   }
356 
357   build_info<F> bi{};
358   auto out = __impl_smoothing::build_smoothing_spline_from_y_hat<F>(xs.ptr, ws.y_hat.data(), n, &bi);
359   if ( info ) {
360     info->status = bi.status;
361     info->residual = (lambda < F(0)) ? lam : rss;
362   }
363   return out;
364 }

366 template<ieee754_floating F>
367 [[nodiscard]] inline cubic_spline_1d<F>
368 make_adaptive_knots(raw_slice<const F> xs, raw_slice<const F> ys, F max_abs_err, usize max_knots, build_info<F> *info = nullptr) noexcept
369 {
370   cubic_spline_1d<F> result{};
371   result.bc = bc_kind::natural;
372 
373   if ( xs.size() != ys.size() ) {
374     if ( info ) info->status = build_status::size_mismatch;
375     return result;
376   }
377   if ( xs.size() < 4 ) {
378     if ( info ) info->status = build_status::too_few_points;
379     return result;
380   }
381   if ( !(max_abs_err > F(0)) || max_knots < 4 ) {
382     if ( info ) info->status = build_status::invalid_argument;
383     return result;
384   }
385   if ( !__impl_splines_bits::strictly_increasing<F>(xs.ptr, xs.size()) ) {
386     if ( info ) info->status = build_status::non_monotonic_x;
387     return result;
388   }
389 
390   const usize n = xs.size();
391   if ( max_knots > n ) max_knots = n;
392 
393   vector<usize> active(0);
394   active.reserve(max_knots);
395   active.emplace_back(0);
396   active.emplace_back(n / 3);
397   active.emplace_back((2 * n) / 3);
398   active.emplace_back(n - 1);
399   vector<usize> uniq(0);
400   uniq.reserve(active.size());
401   for ( usize i = 0; i < active.size(); ++i ) {
402     if ( uniq.size() == 0 || uniq[uniq.size() - 1] != active[i] ) uniq.emplace_back(active[i]);
403   }
404   active = uniq;
405   if ( active.size() < 4 ) {
406     usize k = 1;
407     while ( active.size() < 4 && k + 1 < n ) {
408       bool present = false;
409       for ( usize j = 0; j < active.size(); ++j )
410         if ( active[j] == k ) {
411           present = true;
412           break;
413         }
414       if ( !present ) {
415         usize j = 0;
416         while ( j < active.size() && active[j] < k ) ++j;
417         active.emplace_back(F(0));
418         for ( usize jj = active.size() - 1; jj > j; --jj ) active[jj] = active[jj - 1];
419         active[j] = k;
420       }
421       ++k;
422     }
423   }
424 
425   vector<F> kx(0), ky(0);
426   kx.reserve(max_knots);
427   ky.reserve(max_knots);
428 
429   usize iter = 0;
430   while ( true ) {
431     ++iter;
432     kx.set_size(0);
433     ky.set_size(0);
434     for ( usize i = 0; i < active.size(); ++i ) {
435       kx.emplace_back(xs[active[i]]);
436       ky.emplace_back(ys[active[i]]);
437     }
438     raw_slice<const F> kxs(kx.data(), kx.size());
439     raw_slice<const F> kys(ky.data(), ky.size());
440     result = make_cubic<F>(kxs, kys, bc_kind::natural, F(0), F(0), nullptr);
441 
442     F worst = F(0);
443     usize worst_i = 0;
444     bool found = false;
445     usize active_pos = 0;
446     for ( usize i = 0; i < n; ++i ) {
447       if ( active_pos < active.size() && active[active_pos] == i ) {
448         ++active_pos;
449         continue;
450       }
451       const F y_pred = evaluate<F>(result, xs[i]);
452       const F err = (ys[i] >= y_pred) ? (ys[i] - y_pred) : (y_pred - ys[i]);
453       if ( err > worst ) {
454         worst = err;
455         worst_i = i;
456         found = true;
457       }
458     }
459 
460     if ( !found || worst <= max_abs_err || active.size() >= max_knots ) {
461       if ( info ) {
462         info->status = (worst <= max_abs_err) ? build_status::ok : build_status::max_iter;
463         info->n_iterations = iter;
464         info->residual = worst;
465       }
466       return result;
467     }
468 
469     usize j = 0;
470     while ( j < active.size() && active[j] < worst_i ) ++j;
471     active.emplace_back(0);
472     for ( usize jj = active.size() - 1; jj > j; --jj ) active[jj] = active[jj - 1];
473     active[j] = worst_i;
474   }
475 }
