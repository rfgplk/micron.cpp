113 inline bool
114 fits(const rational &a, rational_limits limits) noexcept
115 {
116   return limits.max_limbs && a.numerator().size() <= limits.max_limbs && a.denominator().size() <= limits.max_limbs
117          && a.numerator().size() <= limits.max_work_limbs && a.denominator().size() <= limits.max_work_limbs;
118 }
119 
120 inline bool
121 product_fits(usize a, usize b, usize maximum) noexcept
122 {
123   return a == 0 || b == 0 || (a <= maximum && b <= maximum - a);
124 }

132 inline rational_result
133 normalize(arbint<> numerator, arbuint<> denominator, rational_limits limits)
134 {
135   if ( denominator.is_zero() ) return fail(rational_error::zero_denominator);
136   if ( !limits.max_limbs || denominator.size() > limits.max_work_limbs || numerator.size() > limits.max_work_limbs )
137     return fail(rational_error::limb_limit);
138   if ( numerator.is_zero() ) return {};
139   const auto common = micron::math::gcd(numerator.magnitude(), denominator);
140   if ( common != 1 ) {
141     numerator /= arbint<>(common, false);
142     denominator /= common;
143   }
144   if ( numerator.size() > limits.max_limbs || denominator.size() > limits.max_limbs ) return fail(rational_error::limb_limit);
145   return { access::make(micron::move(numerator), micron::move(denominator)), rational_error::none };
146 }

150 inline rational_result
151 make_rational(const arbint<> &numerator, const arbint<> &denominator, rational_limits limits = {})
152 {
153   if ( numerator.size() > limits.max_work_limbs || denominator.size() > limits.max_work_limbs )
154     return __impl::fail(rational_error::limb_limit);
155   auto adjusted = numerator;
156   if ( denominator.negative() ) adjusted.negate();
157   return __impl::normalize(micron::move(adjusted), denominator.magnitude(), limits);
158 }

160 inline rational_result
161 make_rational(i64 numerator, i64 denominator = 1, rational_limits limits = {})

163   return make_rational(arbint<>(numerator), arbint<>(denominator), limits);

166 inline rational_result
167 rational_negate(const rational &a, rational_limits limits = {})
168 {
169   if ( !__impl::fits(a, limits) ) return __impl::fail(rational_error::limb_limit);
170   auto numerator = a.numerator();
171   numerator.negate();
172   return { __impl::access::make(micron::move(numerator), a.denominator()), rational_error::none };
173 }

175 inline rational_result
176 rational_add(const rational &a, const rational &b, rational_limits limits = {})
177 {
178   using namespace __impl;
179   if ( !fits(a, limits) || !fits(b, limits) ) return fail(rational_error::limb_limit);
180   if ( a.is_zero() ) return { b };
181   if ( b.is_zero() ) return { a };
182   if ( a.denominator() == b.denominator() && a.numerator() == -b.numerator() ) return {};
183   const auto common = micron::math::gcd(a.denominator(), b.denominator());
184   const auto ad = a.denominator() / common;
185   const auto bd = b.denominator() / common;
186   if ( !product_fits(a.numerator().size(), bd.size(), limits.max_work_limbs)
187        || !product_fits(b.numerator().size(), ad.size(), limits.max_work_limbs)
188        || !product_fits(ad.size(), b.denominator().size(), limits.max_work_limbs) )
189     return fail(rational_error::limb_limit);
190   auto left = a.numerator() * arbint<>(bd, false);
191   const auto right = b.numerator() * arbint<>(ad, false);
192   if ( left.sign() == right.sign() && (left.size() >= limits.max_work_limbs || right.size() >= limits.max_work_limbs) )
193     return fail(rational_error::limb_limit);
194   left += right;
195   return normalize(micron::move(left), ad * b.denominator(), limits);
196 }

198 inline rational_result
199 rational_subtract(const rational &a, const rational &b, rational_limits limits = {})
200 {
201   auto negative = rational_negate(b, limits);
202   if ( !negative ) return negative;
203   return rational_add(a, negative.value, limits);
204 }

206 inline rational_result
207 rational_multiply(const rational &a, const rational &b, rational_limits limits = {})
208 {
209   using namespace __impl;
210   if ( !fits(a, limits) || !fits(b, limits) ) return fail(rational_error::limb_limit);
211   if ( a.is_zero() || b.is_zero() ) return {};
212   const auto ga = micron::math::gcd(a.numerator().magnitude(), b.denominator());
213   const auto gb = micron::math::gcd(b.numerator().magnitude(), a.denominator());
214   const auto an = a.numerator() / arbint<>(ga, false);
215   const auto bn = b.numerator() / arbint<>(gb, false);
216   const auto ad = a.denominator() / gb;
217   const auto bd = b.denominator() / ga;
218   if ( !product_fits(an.size(), bn.size(), limits.max_work_limbs) || !product_fits(ad.size(), bd.size(), limits.max_work_limbs) )
219     return fail(rational_error::limb_limit);
220   return normalize(an * bn, ad * bd, limits);
221 }

223 inline rational_result
224 rational_divide(const rational &a, const rational &b, rational_limits limits = {})
225 {
226   using namespace __impl;
227   if ( b.is_zero() ) return fail(rational_error::zero_denominator);
228   if ( !fits(a, limits) || !fits(b, limits) ) return fail(rational_error::limb_limit);
229   if ( a.is_zero() ) return {};
230   auto inverse = access::make(arbint<>(b.denominator(), b.numerator().negative()), b.numerator().magnitude());
231   return rational_multiply(a, inverse, limits);
232 }

234 inline rational_result
235 rational_power(const rational &a, i64 exponent, rational_limits limits = {})
236 {
237   using namespace __impl;
238   const u64 power = exponent < 0 ? static_cast<u64>(-(exponent + 1)) + 1 : static_cast<u64>(exponent);
239   if ( power > limits.max_power ) return fail(rational_error::exponent_limit);
240   if ( !fits(a, limits) ) return fail(rational_error::limb_limit);
241   // 0^0 is deliberately undefined, rather than silently introducing a value
242   if ( a.is_zero() && exponent <= 0 ) return fail(rational_error::zero_denominator);
243   auto base = exponent < 0 ? rational_divide(make_rational(1).value, a, limits) : rational_result{ a };
244   if ( !base ) return base;
245   auto result = make_rational(1, 1, limits);
246   u64 remaining = power;
247   while ( remaining ) {
248     if ( remaining & 1 ) {
249       result = rational_multiply(result.value, base.value, limits);
250       if ( !result ) return result;
251     }
252     remaining >>= 1;
253     if ( remaining ) {
254       base = rational_multiply(base.value, base.value, limits);
255       if ( !base ) return base;
256     }
257   }
258   return result;
259 }

261 inline rational_order_result
262 rational_compare(const rational &a, const rational &b, rational_limits limits = {})
263 {
264   using namespace __impl;
265   if ( !fits(a, limits) || !fits(b, limits) ) return { 0, rational_error::limb_limit };
266   if ( a.sign() != b.sign() ) return { a.sign() < b.sign() ? -1 : 1 };
267   if ( a.denominator() == b.denominator() ) return { cmp(a.numerator(), b.numerator()) };
268   const auto common = micron::math::gcd(a.denominator(), b.denominator());
269   const auto ad = a.denominator() / common;
270   const auto bd = b.denominator() / common;
271   if ( !product_fits(a.numerator().size(), bd.size(), limits.max_work_limbs)
272        || !product_fits(b.numerator().size(), ad.size(), limits.max_work_limbs) )
273     return { 0, rational_error::limb_limit };
274   const auto left = a.numerator() * arbint<>(bd, false);
275   const auto right = b.numerator() * arbint<>(ad, false);
276   return { cmp(left, right) };
277 }
