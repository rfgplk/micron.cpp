30 template<ieee754_floating T>
31 [[nodiscard]] inline constexpr quaternion<T>
32 from_axis_angle(T ax, T ay, T az, T angle) noexcept
33 {
34   const T aax = math::fabs(ax);
35   const T aay = math::fabs(ay);
36   const T aaz = math::fabs(az);
37   const T m = math::fmax(aax, math::fmax(aay, aaz));
38   if ( m == T(0) ) return identity<T>();
39   const T inv_m = T(1) / m;
40   const T sx = ax * inv_m;
41   const T sy = ay * inv_m;
42   const T sz = az * inv_m;
43   const T inv_n = math::frsqrt(sx * sx + sy * sy + sz * sz) * inv_m;
44   const T half = angle * T(0.5);
45   const T s = math::sin<T>(half);
46   const T c = math::cos<T>(half);
47   const T k = s * inv_n;
48   return quaternion<T>{ ax * k, ay * k, az * k, c };
49 }

63 // NOTE: uses atan2(|v|, w) instead of acos(w) for numerical stability (slower)
64 // returns angle in [0, 2pi] with the recovered axis chosen so that from_axis_angle(axis, angle) reconstructs the input quaternion
65 template<ieee754_floating T>
66 [[nodiscard]] inline constexpr axis_angle_t<T>
67 to_axis_angle(const quaternion<T> &q) noexcept
68 {
69   const T ax = math::fabs(q.x);
70   const T ay = math::fabs(q.y);
71   const T az = math::fabs(q.z);
72   const T aw = math::fabs(q.w);
73   const T m = math::fmax(math::fmax(ax, ay), math::fmax(az, aw));
74   if ( m == T(0) ) {
75     return axis_angle_t<T>{ micron::vector_3<T>{ T(1), T(0), T(0) }, T(0) };
76   }
77   const T inv_m = T(1) / m;
78   const T sx = q.x * inv_m;
79   const T sy = q.y * inv_m;
80   const T sz = q.z * inv_m;
81   const T sw = q.w * inv_m;
82   const T inv_n = math::frsqrt(sx * sx + sy * sy + sz * sz + sw * sw) * inv_m;
83   const T nx = q.x * inv_n;
84   const T ny = q.y * inv_n;
85   const T nz = q.z * inv_n;
86   const T nw = q.w * inv_n;
87   const T v2 = nx * nx + ny * ny + nz * nz;
88   const T v_norm = math::fsqrt(v2);
89   // atan2 is exact for every representable v_norm; only the 1/v_norm below needs a floor
90   if ( v_norm <= __safe_min_n2<T>() ) {
91     return axis_angle_t<T>{ micron::vector_3<T>{ T(1), T(0), T(0) }, T(0) };
92   }
93   const T half_angle = math::atan2<T>(v_norm, nw);
94   const T inv_v = T(1) / v_norm;
95   return axis_angle_t<T>{ micron::vector_3<T>{ nx * inv_v, ny * inv_v, nz * inv_v }, T(2) * half_angle };
96 }

98 // NOTE: faster acos(w)-based decomposition
99 template<ieee754_floating T>
100 [[nodiscard]] inline constexpr axis_angle_t<T>
101 to_axis_angle_fast(const quaternion<T> &q) noexcept
102 {
103   T w_clamped = q.w;
104   if ( w_clamped > T(1) ) w_clamped = T(1);
105   if ( w_clamped < T(-1) ) w_clamped = T(-1);
106   const T half_angle = math::acos<T>(w_clamped);
107   const T s = math::sin<T>(half_angle);
108   if ( math::fabs(s) < math::default_eps<T>() ) {
109     return axis_angle_t<T>{ micron::vector_3<T>{ T(1), T(0), T(0) }, T(0) };
110   }
111   const T inv_s = T(1) / s;
112   return axis_angle_t<T>{ micron::vector_3<T>{ q.x * inv_s, q.y * inv_s, q.z * inv_s }, T(2) * half_angle };
113 }

115 // Shepperd's method
116 template<ieee754_floating T>
117 [[nodiscard]] inline constexpr quaternion<T>
118 from_matrix(const mat<T, 3, 3> &R) noexcept
119 {
120   const T *__restrict__ d = R.data;
121   const T m00 = d[0], m01 = d[1], m02 = d[2];
122   const T m10 = d[3], m11 = d[4], m12 = d[5];
123   const T m20 = d[6], m21 = d[7], m22 = d[8];
124   const T tr = m00 + m11 + m22;
125   const T eps = math::default_eps<T>();
126   T x, y, z, w;
127   if ( tr > T(0) ) {
128     const T r = tr + T(1);
129     const T s = math::fsqrt(r > T(0) ? r : T(0)) * T(2);
130     const T inv_s = (s > eps) ? T(1) / s : T(0);
131     w = T(0.25) * s;
132     x = (m21 - m12) * inv_s;
133     y = (m02 - m20) * inv_s;
134     z = (m10 - m01) * inv_s;
135   } else if ( m00 > m11 && m00 > m22 ) {
136     const T r = T(1) + m00 - m11 - m22;
137     const T s = math::fsqrt(r > T(0) ? r : T(0)) * T(2);
138     const T inv_s = (s > eps) ? T(1) / s : T(0);
139     x = T(0.25) * s;
140     y = (m01 + m10) * inv_s;
141     z = (m02 + m20) * inv_s;
142     w = (m21 - m12) * inv_s;
143   } else if ( m11 > m22 ) {
144     const T r = T(1) + m11 - m00 - m22;
145     const T s = math::fsqrt(r > T(0) ? r : T(0)) * T(2);
146     const T inv_s = (s > eps) ? T(1) / s : T(0);
147     y = T(0.25) * s;
148     x = (m01 + m10) * inv_s;
149     z = (m12 + m21) * inv_s;
150     w = (m02 - m20) * inv_s;
151   } else {
152     const T r = T(1) + m22 - m00 - m11;
153     const T s = math::fsqrt(r > T(0) ? r : T(0)) * T(2);
154     const T inv_s = (s > eps) ? T(1) / s : T(0);
155     z = T(0.25) * s;
156     x = (m02 + m20) * inv_s;
157     y = (m12 + m21) * inv_s;
158     w = (m10 - m01) * inv_s;
159   }
160   const T n2 = x * x + y * y + z * z + w * w;
161   // exact-zero OR non-finite (NaN/Inf input matrix yields NaN n2) -> fall back to identity
162   if ( n2 == T(0) || !ieee::is_finite(n2) ) return identity<T>();
163   const T inv_n = math::frsqrt(n2);
164   return quaternion<T>{ x * inv_n, y * inv_n, z * inv_n, w * inv_n };
165 }

167 template<ieee754_floating T>
168 [[nodiscard, gnu::always_inline]] inline constexpr mat<T, 3, 3>
169 to_matrix(const quaternion<T> &q) noexcept
170 {
171   quat<T> qq{};
172   qq.data[0] = q.x;
173   qq.data[1] = q.y;
174   qq.data[2] = q.z;
175   qq.data[3] = q.w;
176   return blas::identities::from_quat<T>(qq);
177 }

179 // v' = (2w^2/|q|^2 - 1) v + (2 (u . v)/|q|^2) u + (2 w/|q|^2) (u x v)
180 template<ieee754_floating T>
181 [[nodiscard]] inline constexpr micron::vector_3<T>
182 rotate(const quaternion<T> &q, const micron::vector_3<T> &v) noexcept
183 {
184   const T ux = q.x, uy = q.y, uz = q.z, w = q.w;
185   const T n2 = ux * ux + uy * uy + uz * uz + w * w;
186   if ( n2 == T(0) ) return v;
187   const T inv_n2 = T(1) / n2;
188   const T two_inv_n2 = T(2) * inv_n2;
189   const T a = w * w * two_inv_n2 - T(1);
190   const T b = (ux * v.x + uy * v.y + uz * v.z) * two_inv_n2;
191   const T c = w * two_inv_n2;
192   const T cx = uy * v.z - uz * v.y;
193   const T cy = uz * v.x - ux * v.z;
194   const T cz = ux * v.y - uy * v.x;
195   return micron::vector_3<T>{ a * v.x + b * ux + c * cx, a * v.y + b * uy + c * cy, a * v.z + b * uz + c * cz };
196 }

225 template<ieee754_floating T>
226 [[nodiscard]] inline constexpr micron::vector_3<T>
227 __orthonormal_axis(const micron::vector_3<T> &a) noexcept
228 {
229   const T ax = math::fabs(a.x), ay = math::fabs(a.y), az = math::fabs(a.z);
230   const micron::vector_3<T> basis = (ax <= ay && ax <= az) ? micron::vector_3<T>{ T(1), T(0), T(0) }
231                                     : (ay <= az)           ? micron::vector_3<T>{ T(0), T(1), T(0) }
232                                                            : micron::vector_3<T>{ T(0), T(0), T(1) };
233   const micron::vector_3<T> axis{ a.y * basis.z - a.z * basis.y, a.z * basis.x - a.x * basis.z, a.x * basis.y - a.y * basis.x };
234   const T inv = math::frsqrt(axis.x * axis.x + axis.y * axis.y + axis.z * axis.z);
235   return micron::vector_3<T>{ axis.x * inv, axis.y * inv, axis.z * inv };
236 }

238 // shortest-arc rotation taking the unit vector a onto the unit vector b
239 template<ieee754_floating T>
240 [[nodiscard]] inline constexpr quaternion<T>
241 from_unit_vectors(const micron::vector_3<T> &a, const micron::vector_3<T> &b) noexcept
242 {
243   const T d = a.x * b.x + a.y * b.y + a.z * b.z;
244   const T cx = a.y * b.z - a.z * b.y;
245   const T cy = a.z * b.x - a.x * b.z;
246   const T cz = a.x * b.y - a.y * b.x;
247   const T w = T(1) + d;
248   if ( w > math::default_eps<T>() ) return quaternion<T>{ cx, cy, cz, w }.normalized();
249   const micron::vector_3<T> axis = __orthonormal_axis<T>(a);
250   return quaternion<T>{ axis.x, axis.y, axis.z, T(0) };
251 }

253 template<ieee754_floating T>
254 [[nodiscard]] inline constexpr quaternion<T>
255 from_two_vectors(const micron::vector_3<T> &a, const micron::vector_3<T> &b) noexcept
256 {
257   const T ma = math::fmax(math::fabs(a.x), math::fmax(math::fabs(a.y), math::fabs(a.z)));
258   const T mb = math::fmax(math::fabs(b.x), math::fmax(math::fabs(b.y), math::fabs(b.z)));
259   if ( ma == T(0) || mb == T(0) ) return identity<T>();
260   const T ia = T(1) / math::fmax(ma, __safe_min_n2<T>());
261   const T ib = T(1) / math::fmax(mb, __safe_min_n2<T>());
262   const micron::vector_3<T> sa{ a.x * ia, a.y * ia, a.z * ia };
263   const micron::vector_3<T> sb{ b.x * ib, b.y * ib, b.z * ib };
264   const T s = math::fsqrt(sa.x * sa.x + sa.y * sa.y + sa.z * sa.z) * math::fsqrt(sb.x * sb.x + sb.y * sb.y + sb.z * sb.z);
265   const T d = sa.x * sb.x + sa.y * sb.y + sa.z * sb.z;
266   const T cx = sa.y * sb.z - sa.z * sb.y;
267   const T cy = sa.z * sb.x - sa.x * sb.z;
268   const T cz = sa.x * sb.y - sa.y * sb.x;
269   const T w = s + d;
270   if ( w > math::default_eps<T>() * s ) return quaternion<T>{ cx, cy, cz, w }.normalized();
271   const micron::vector_3<T> axis = __orthonormal_axis<T>(sa);
272   return quaternion<T>{ axis.x, axis.y, axis.z, T(0) };
273 }
