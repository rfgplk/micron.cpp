48 // AoS->SoA transpose of 4 vectors of 4 doubles each. After this call:
49 //   xs = {v0[0], v1[0], v2[0], v3[0]}
50 //   ys = {v0[1], v1[1], v2[1], v3[1]}
51 //   zs = {v0[2], v1[2], v2[2], v3[2]}
52 //   ws = {v0[3], v1[3], v2[3], v3[3]}
53 [[gnu::always_inline]] inline void
54 transpose4x4_pd(__m256d v0, __m256d v1, __m256d v2, __m256d v3, __m256d &xs, __m256d &ys, __m256d &zs, __m256d &ws) noexcept
55 {
56   const __m256d t01_lo = simd::avx::unpack_lo_f64(v0, v1);      // {v0[0], v1[0], v0[2], v1[2]}
57   const __m256d t01_hi = simd::avx::unpack_hi_f64(v0, v1);      // {v0[1], v1[1], v0[3], v1[3]}
58   const __m256d t23_lo = simd::avx::unpack_lo_f64(v2, v3);
59   const __m256d t23_hi = simd::avx::unpack_hi_f64(v2, v3);
60   xs = simd::avx::permute2f128_f64<0x20>(t01_lo, t23_lo);
61   ys = simd::avx::permute2f128_f64<0x20>(t01_hi, t23_hi);
62   zs = simd::avx::permute2f128_f64<0x31>(t01_lo, t23_lo);
63   ws = simd::avx::permute2f128_f64<0x31>(t01_hi, t23_hi);
64 }

66 // SoA->AoS transpose
67 [[gnu::always_inline]] inline void
68 transpose4x4_pd_inv(__m256d xs, __m256d ys, __m256d zs, __m256d ws, __m256d &v0, __m256d &v1, __m256d &v2, __m256d &v3) noexcept
69 {
70   const __m256d t01_lo = simd::avx::unpack_lo_f64(xs, ys);
71   const __m256d t01_hi = simd::avx::unpack_hi_f64(xs, ys);
72   const __m256d t23_lo = simd::avx::unpack_lo_f64(zs, ws);
73   const __m256d t23_hi = simd::avx::unpack_hi_f64(zs, ws);
74   v0 = simd::avx::permute2f128_f64<0x20>(t01_lo, t23_lo);
75   v1 = simd::avx::permute2f128_f64<0x20>(t01_hi, t23_hi);
76   v2 = simd::avx::permute2f128_f64<0x31>(t01_lo, t23_lo);
77   v3 = simd::avx::permute2f128_f64<0x31>(t01_hi, t23_hi);
78 }

80 [[nodiscard, gnu::always_inline]] inline __m256d
81 normalizable_all() noexcept
82 {
83   const __m256d z = simd::avx::splat_f64(0.0);
84   return simd::avx::cmp_f64<_CMP_EQ_OQ>(z, z);
85 }

87 [[nodiscard, gnu::always_inline]] inline __m256d
88 keep_normalizable(__m256d acc, __m256d n2, double lo) noexcept
89 {
90   const __m256d ok = simd::avx::and_f64(simd::avx::cmp_f64<_CMP_GT_OQ>(n2, simd::avx::splat_f64(lo)),
91                                         simd::avx::cmp_f64<_CMP_LT_OQ>(n2, simd::avx::splat_f64(ieee::inf_v<double>())));
92   return simd::avx::and_f64(acc, ok);
93 }

108 // AoS->SoA transpose of 4 f32 quaternions packed as 4 float32x4_t
109 [[gnu::always_inline]] inline void
110 transpose4x4_ps(float32x4_t v0, float32x4_t v1, float32x4_t v2, float32x4_t v3, float32x4_t &xs, float32x4_t &ys, float32x4_t &zs,
111                 float32x4_t &ws) noexcept
112 {
113   const float32x4_t t01_lo = simd::neon::zip_lo_f32(v0, v1);      // {v0[0], v1[0], v0[1], v1[1]}
114   const float32x4_t t01_hi = simd::neon::zip_hi_f32(v0, v1);      // {v0[2], v1[2], v0[3], v1[3]}
115   const float32x4_t t23_lo = simd::neon::zip_lo_f32(v2, v3);
116   const float32x4_t t23_hi = simd::neon::zip_hi_f32(v2, v3);
117   xs = simd::neon::concat_lo_f32(t01_lo, t23_lo);
118   ys = simd::neon::concat_hi_f32(t01_lo, t23_lo);
119   zs = simd::neon::concat_lo_f32(t01_hi, t23_hi);
120   ws = simd::neon::concat_hi_f32(t01_hi, t23_hi);
121 }

194 // batched_multiply: out[i] = a[i] * b[i] (Hamilton product) for i in [0, n)
195 template<ieee754_floating T>
196 inline void
197 batched_multiply(const quaternion<T> *a, const quaternion<T> *b, quaternion<T> *out, usize n) noexcept
198 {
199 #if defined(__micron_x86_avx2) && defined(__micron_x86_fma)
200   if constexpr ( sizeof(T) == 8 ) {
201     usize i = 0;
202     for ( ; i + 4 <= n; i += 4 ) {
203       const double *ap = reinterpret_cast<const double *>(a + i);
204       const double *bp = reinterpret_cast<const double *>(b + i);
205       const __m256d a0 = simd::avx::loadu_f64(ap + 0);
206       const __m256d a1 = simd::avx::loadu_f64(ap + 4);
207       const __m256d a2 = simd::avx::loadu_f64(ap + 8);
208       const __m256d a3 = simd::avx::loadu_f64(ap + 12);
209       const __m256d b0 = simd::avx::loadu_f64(bp + 0);
210       const __m256d b1 = simd::avx::loadu_f64(bp + 4);
211       const __m256d b2 = simd::avx::loadu_f64(bp + 8);
212       const __m256d b3 = simd::avx::loadu_f64(bp + 12);
213 
214       __m256d axs, ays, azs, aws, bxs, bys, bzs, bws;
215       __impl::transpose4x4_pd(a0, a1, a2, a3, axs, ays, azs, aws);
216       __impl::transpose4x4_pd(b0, b1, b2, b3, bxs, bys, bzs, bws);
217 
218       // 16 FMAs = 8 cyc on 2 FMA ports; transposes add ~6 cyc; ~14 cyc/4-quats = 3.5 cyc/quat
219       __m256d rxs = simd::avx::mul_f64(aws, bxs);
220       rxs = simd::fma::fma_f64(axs, bws, rxs);
221       rxs = simd::fma::fma_f64(ays, bzs, rxs);
222       rxs = simd::fma::fnma_f64(azs, bys, rxs);
223 
224       __m256d rys = simd::avx::mul_f64(aws, bys);
225       rys = simd::fma::fnma_f64(axs, bzs, rys);
226       rys = simd::fma::fma_f64(ays, bws, rys);
227       rys = simd::fma::fma_f64(azs, bxs, rys);
228 
229       __m256d rzs = simd::avx::mul_f64(aws, bzs);
230       rzs = simd::fma::fma_f64(axs, bys, rzs);
231       rzs = simd::fma::fnma_f64(ays, bxs, rzs);
232       rzs = simd::fma::fma_f64(azs, bws, rzs);
233 
234       __m256d rws = simd::avx::mul_f64(aws, bws);
235       rws = simd::fma::fnma_f64(axs, bxs, rws);
236       rws = simd::fma::fnma_f64(ays, bys, rws);
237       rws = simd::fma::fnma_f64(azs, bzs, rws);
238 
239       __m256d r0, r1, r2, r3;
240       __impl::transpose4x4_pd_inv(rxs, rys, rzs, rws, r0, r1, r2, r3);
241       double *op = reinterpret_cast<double *>(out + i);
242       simd::avx::storeu_f64(op + 0, r0);
243       simd::avx::storeu_f64(op + 4, r1);
244       simd::avx::storeu_f64(op + 8, r2);
245       simd::avx::storeu_f64(op + 12, r3);
246     }
247     for ( ; i < n; ++i ) out[i] = multiply<T>(a[i], b[i]);
248     return;
249   }
250 #endif
251 #if defined(__micron_quat_batched_neon)
252   if constexpr ( sizeof(T) == 4 ) {
253     // f32: 4 quaternions per iteration via SoA on float32x4_t
254     usize i = 0;
255     for ( ; i + 4 <= n; i += 4 ) {
256       const float *ap = reinterpret_cast<const float *>(a + i);
257       const float *bp = reinterpret_cast<const float *>(b + i);
258       const float32x4_t a0 = simd::neon::load_f32(ap + 0);
259       const float32x4_t a1 = simd::neon::load_f32(ap + 4);
260       const float32x4_t a2 = simd::neon::load_f32(ap + 8);
261       const float32x4_t a3 = simd::neon::load_f32(ap + 12);
262       const float32x4_t b0 = simd::neon::load_f32(bp + 0);
263       const float32x4_t b1 = simd::neon::load_f32(bp + 4);
264       const float32x4_t b2 = simd::neon::load_f32(bp + 8);
265       const float32x4_t b3 = simd::neon::load_f32(bp + 12);
266 
267       float32x4_t axs, ays, azs, aws, bxs, bys, bzs, bws;
268       __impl::transpose4x4_ps(a0, a1, a2, a3, axs, ays, azs, aws);
269       __impl::transpose4x4_ps(b0, b1, b2, b3, bxs, bys, bzs, bws);
270 
271       float32x4_t rxs = simd::neon::mul(aws, bxs);
272       rxs = simd::neon::fma_f32(rxs, axs, bws);
273       rxs = simd::neon::fma_f32(rxs, ays, bzs);
274       rxs = simd::neon::fms_f32(rxs, azs, bys);
275 
276       float32x4_t rys = simd::neon::mul(aws, bys);
277       rys = simd::neon::fms_f32(rys, axs, bzs);
278       rys = simd::neon::fma_f32(rys, ays, bws);
279       rys = simd::neon::fma_f32(rys, azs, bxs);
280 
281       float32x4_t rzs = simd::neon::mul(aws, bzs);
282       rzs = simd::neon::fma_f32(rzs, axs, bys);
283       rzs = simd::neon::fms_f32(rzs, ays, bxs);
284       rzs = simd::neon::fma_f32(rzs, azs, bws);
285 
286       float32x4_t rws = simd::neon::mul(aws, bws);
287       rws = simd::neon::fms_f32(rws, axs, bxs);
288       rws = simd::neon::fms_f32(rws, ays, bys);
289       rws = simd::neon::fms_f32(rws, azs, bzs);
290 
291       float32x4_t r0, r1, r2, r3;
292       __impl::transpose4x4_ps_inv(rxs, rys, rzs, rws, r0, r1, r2, r3);
293       float *op = reinterpret_cast<float *>(out + i);
294       simd::neon::store_f32(op + 0, r0);
295       simd::neon::store_f32(op + 4, r1);
296       simd::neon::store_f32(op + 8, r2);
297       simd::neon::store_f32(op + 12, r3);
298     }
299     for ( ; i < n; ++i ) out[i] = multiply<T>(a[i], b[i]);
300     return;
301   }
302 #if defined(__micron_arch_arm64)
303   if constexpr ( sizeof(T) == 8 ) {
304     // f64 on arm64: 2 quaternions per iteration via SoA on float64x2_t.
305     usize i = 0;
306     for ( ; i + 2 <= n; i += 2 ) {
307       const double *ap = reinterpret_cast<const double *>(a + i);
308       const double *bp = reinterpret_cast<const double *>(b + i);
309       const float64x2_t alo0 = simd::neon::load_f64(ap + 0);
310       const float64x2_t ahi0 = simd::neon::load_f64(ap + 2);
311       const float64x2_t alo1 = simd::neon::load_f64(ap + 4);
312       const float64x2_t ahi1 = simd::neon::load_f64(ap + 6);
313       const float64x2_t blo0 = simd::neon::load_f64(bp + 0);
314       const float64x2_t bhi0 = simd::neon::load_f64(bp + 2);
315       const float64x2_t blo1 = simd::neon::load_f64(bp + 4);
316       const float64x2_t bhi1 = simd::neon::load_f64(bp + 6);
317 
318       float64x2_t axs, ays, azs, aws, bxs, bys, bzs, bws;
319       __impl::transpose2x4_pd(alo0, ahi0, alo1, ahi1, axs, ays, azs, aws);
320       __impl::transpose2x4_pd(blo0, bhi0, blo1, bhi1, bxs, bys, bzs, bws);
321 
322       float64x2_t rxs = simd::neon::mul(aws, bxs);
323       rxs = simd::neon::fma_f64(rxs, axs, bws);
324       rxs = simd::neon::fma_f64(rxs, ays, bzs);
325       rxs = simd::neon::fms_f64(rxs, azs, bys);
326 
327       float64x2_t rys = simd::neon::mul(aws, bys);
328       rys = simd::neon::fms_f64(rys, axs, bzs);
329       rys = simd::neon::fma_f64(rys, ays, bws);
330       rys = simd::neon::fma_f64(rys, azs, bxs);
331 
332       float64x2_t rzs = simd::neon::mul(aws, bzs);
333       rzs = simd::neon::fma_f64(rzs, axs, bys);
334       rzs = simd::neon::fms_f64(rzs, ays, bxs);
335       rzs = simd::neon::fma_f64(rzs, azs, bws);
336 
337       float64x2_t rws = simd::neon::mul(aws, bws);
338       rws = simd::neon::fms_f64(rws, axs, bxs);
339       rws = simd::neon::fms_f64(rws, ays, bys);
340       rws = simd::neon::fms_f64(rws, azs, bzs);
341 
342       float64x2_t rlo0, rhi0, rlo1, rhi1;
343       __impl::transpose2x4_pd_inv(rxs, rys, rzs, rws, rlo0, rhi0, rlo1, rhi1);

358 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
359 // batched_normalize: out[i] = a[i] / |a[i]|
360 template<ieee754_floating T>
361 inline void
362 batched_normalize(const quaternion<T> *in, quaternion<T> *out, usize n) noexcept
363 {
364 #if defined(__micron_x86_avx2) && defined(__micron_x86_fma)
365   if constexpr ( sizeof(T) == 8 ) {
366     const __m256d one = simd::avx::splat_f64(1.0);
367     __m256d ok = __impl::normalizable_all();
368     usize i = 0;
369     for ( ; i + 4 <= n; i += 4 ) {
370       const double *ip = reinterpret_cast<const double *>(in + i);
371       const __m256d q0 = simd::avx::loadu_f64(ip + 0);
372       const __m256d q1 = simd::avx::loadu_f64(ip + 4);
373       const __m256d q2 = simd::avx::loadu_f64(ip + 8);
374       const __m256d q3 = simd::avx::loadu_f64(ip + 12);
375 
376       __m256d xs, ys, zs, ws;
377       __impl::transpose4x4_pd(q0, q1, q2, q3, xs, ys, zs, ws);
378 
379       __m256d n2 = simd::avx::mul_f64(xs, xs);
380       n2 = simd::fma::fma_f64(ys, ys, n2);
381       n2 = simd::fma::fma_f64(zs, zs, n2);
382       n2 = simd::fma::fma_f64(ws, ws, n2);
383       ok = __impl::keep_normalizable(ok, n2, __safe_min_n2<T>());
384 
385       const __m256d sq = simd::avx::sqrt_f64(n2);
386       const __m256d inv = simd::avx::div_f64(one, sq);
387 
388       xs = simd::avx::mul_f64(xs, inv);
389       ys = simd::avx::mul_f64(ys, inv);
390       zs = simd::avx::mul_f64(zs, inv);
391       ws = simd::avx::mul_f64(ws, inv);
392 
393       __m256d r0, r1, r2, r3;
394       __impl::transpose4x4_pd_inv(xs, ys, zs, ws, r0, r1, r2, r3);
395       double *op = reinterpret_cast<double *>(out + i);
396       simd::avx::storeu_f64(op + 0, r0);
397       simd::avx::storeu_f64(op + 4, r1);
398       simd::avx::storeu_f64(op + 8, r2);
399       simd::avx::storeu_f64(op + 12, r3);
400     }
401     if ( !__impl::all_normalizable(ok) ) {
402       for ( usize k = 0; k < i; ++k ) out[k] = normalize<T>(in[k]);
403     }
404     for ( ; i < n; ++i ) out[i] = normalize<T>(in[i]);
405     return;
406   }

408 #if defined(__micron_quat_batched_neon)
409   if constexpr ( sizeof(T) == 4 ) {
410     // NOTE: we multiply by rsqrt(n**2) rather than dividing by sqrt(n**2)
411     // NEON's vrsqrteq_f32 + one NR step lands at ~22 bits of mantissa, close enough to fp32 precision
412     float32x4_t ok = simd::neon::splat_f32(1.0f);
413     usize i = 0;
414     for ( ; i + 4 <= n; i += 4 ) {

417       const float32x4_t q1 = simd::neon::load_f32(ip + 4);

423

425       n2 = simd::neon::fma_f32(n2, ys, ys);
426       n2 = simd::neon::fma_f32(n2, zs, zs);
427       n2 = simd::neon::fma_f32(n2, ws, ws);
428       ok = __impl::keep_normalizable(ok, n2, __safe_min_n2<T>());
429 
430       float32x4_t inv = simd::neon::rsqrt_est(n2);
431       inv = simd::neon::mul(inv, simd::neon::rsqrt_step(simd::neon::mul(n2, inv), inv));
432       inv = simd::neon::mul(inv, simd::neon::rsqrt_step(simd::neon::mul(n2, inv), inv));
433 
434       xs = simd::neon::mul(xs, inv);

443       simd::neon::store_f32(op + 4, r1);

449     }
450     for ( ; i < n; ++i ) out[i] = normalize<T>(in[i]);
451     return;
452   }
453 #if defined(__micron_arch_arm64)
454   if constexpr ( sizeof(T) == 8 ) {
455     const float64x2_t one = simd::neon::splat_f64(1.0);
456     float64x2_t ok = simd::neon::splat_f64(1.0);
457     usize i = 0;
458     for ( ; i + 2 <= n; i += 2 ) {
459       const double *ip = reinterpret_cast<const double *>(in + i);
460       const float64x2_t lo0 = simd::neon::load_f64(ip + 0);
461       const float64x2_t hi0 = simd::neon::load_f64(ip + 2);

463       const float64x2_t hi1 = simd::neon::load_f64(ip + 6);

476 
477       xs = simd::neon::mul(xs, inv);
478       ys = simd::neon::mul(ys, inv);

484       double *op = reinterpret_cast<double *>(out + i);
485       simd::neon::store_f64(op + 0, rlo0);
486       simd::neon::store_f64(op + 2, rhi0);

490     if ( !__impl::all_normalizable(ok) ) {
491       for ( usize k = 0; k < i; ++k ) out[k] = normalize<T>(in[k]);

502 // batched_rotate: out[i] = q[i] * vec3(v[i]) * conj(q[i])
503 template<ieee754_floating T>
504 inline void
505 batched_rotate(const quaternion<T> *q, const micron::vector_3<T> *v, micron::vector_3<T> *out, usize n) noexcept

507   for ( usize i = 0; i < n; ++i ) out[i] = rotate<T>(q[i], v[i]);
