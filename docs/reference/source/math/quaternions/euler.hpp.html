31 // Public tag types
32 enum class rotations : u8 {
33   // Tait-Bryan body-axis labels (aerospace)
34   roll = 0,       // X-axis
35   pitch = 1,      // Y-axis
36   yaw = 2,        // Z-axis
37   // Proper-Euler symbol aliases (same physical axes)
38   alpha = roll,
39   beta = pitch,
40   gamma = yaw,
41 };
42 
43 enum class frame : u8 {
44   body = 0,       // intrinsic:  q' = q x dq
45   world = 1,      // extrinsic:  q' = dq x q
46 };
47 
48 enum class euler_order : u8 {
49   // Tait-Bryan (3 distinct axes)
50   XYZ = 0,
51   XZY = 1,
52   YXZ = 2,
53   YZX = 3,
54   ZXY = 4,
55   ZYX = 5,
56   // Proper-Euler (axis repeated)
57   XYX = 6,
58   XZX = 7,
59   YXY = 8,
60   YZY = 9,
61   ZXZ = 10,
62   ZYZ = 11,
63 };

85 // body-frame
86 template<ieee754_floating T>
87 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
88 mul_x_body(const quaternion<T> &q, T s, T c) noexcept
89 {
90   return quaternion<T>{ q.w * s + q.x * c, q.y * c + q.z * s, q.z * c - q.y * s, q.w * c - q.x * s };
91 }

93 // body-frame
94 template<ieee754_floating T>
95 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
96 mul_y_body(const quaternion<T> &q, T s, T c) noexcept
97 {
98   return quaternion<T>{ q.x * c - q.z * s, q.w * s + q.y * c, q.x * s + q.z * c, q.w * c - q.y * s };
99 }

103 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
104 mul_z_body(const quaternion<T> &q, T s, T c) noexcept
105 {
106   return quaternion<T>{ q.x * c + q.y * s, q.y * c - q.x * s, q.w * s + q.z * c, q.w * c - q.z * s };
107 }

110 template<ieee754_floating T>
111 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
112 mul_x_world(const quaternion<T> &q, T s, T c) noexcept
113 {
114   return quaternion<T>{ c * q.x + s * q.w, c * q.y - s * q.z, c * q.z + s * q.y, c * q.w - s * q.x };

117 // world-frame
118 template<ieee754_floating T>
119 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
120 mul_y_world(const quaternion<T> &q, T s, T c) noexcept
121 {
122   return quaternion<T>{ c * q.x + s * q.z, c * q.y + s * q.w, c * q.z - s * q.x, c * q.w - s * q.y };

125 // world-frame
126 template<ieee754_floating T>
127 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
128 mul_z_world(const quaternion<T> &q, T s, T c) noexcept
129 {
130   return quaternion<T>{ c * q.x - s * q.y, c * q.y + s * q.x, c * q.z + s * q.w, c * q.w - s * q.z };

133 // fast unit normalization of an already-near-unit quaternion (used by rotate_safe)
134 template<ieee754_floating T>
135 [[nodiscard, gnu::always_inline]] inline constexpr quaternion<T>
136 unit_renorm(const quaternion<T> &q) noexcept
137 {
138   const T n2 = q.x * q.x + q.y * q.y + q.z * q.z + q.w * q.w;
139   const T inv = math::frsqrt(n2);
140   return quaternion<T>{ q.x * inv, q.y * inv, q.z * inv, q.w * inv };
141 }

192 template<rotations Axis, frame F = frame::body, ieee754_floating T>
193 [[gnu::always_inline]] inline constexpr void
194 rotate(quaternion<T> &q, T angle) noexcept
195 {
196   T s, c;
197   math::sincos<T>(angle * T(0.5), s, c);
198   if constexpr ( F == frame::body ) {
199     if constexpr ( __impl_euler::axis_of_v<Axis> == u8(0) )
200       q = __impl_euler::mul_x_body<T>(q, s, c);
201     else if constexpr ( __impl_euler::axis_of_v<Axis> == u8(1) )
202       q = __impl_euler::mul_y_body<T>(q, s, c);
203     else
204       q = __impl_euler::mul_z_body<T>(q, s, c);
205   } else {
206     if constexpr ( __impl_euler::axis_of_v<Axis> == u8(0) )
207       q = __impl_euler::mul_x_world<T>(q, s, c);
208     else if constexpr ( __impl_euler::axis_of_v<Axis> == u8(1) )
209       q = __impl_euler::mul_y_world<T>(q, s, c);
210     else
211       q = __impl_euler::mul_z_world<T>(q, s, c);
212   }
213 }

241 template<ieee754_floating T>
242 inline constexpr void
243 rotate(quaternion<T> &q, rotations axis, T angle, frame f = frame::body) noexcept
244 {
245   T s, c;
246   math::sincos<T>(angle * T(0.5), s, c);
247   const u8 ax = ((axis == rotations::roll) || (axis == rotations::alpha))   ? u8(0)
248                 : ((axis == rotations::pitch) || (axis == rotations::beta)) ? u8(1)
249                                                                             : u8(2);
250   if ( f == frame::body ) {
251     if ( ax == u8(0) )
252       q = __impl_euler::mul_x_body<T>(q, s, c);
253     else if ( ax == u8(1) )
254       q = __impl_euler::mul_y_body<T>(q, s, c);
255     else
256       q = __impl_euler::mul_z_body<T>(q, s, c);
257   } else {
258     if ( ax == u8(0) )
259       q = __impl_euler::mul_x_world<T>(q, s, c);
260     else if ( ax == u8(1) )
261       q = __impl_euler::mul_y_world<T>(q, s, c);
262     else
263       q = __impl_euler::mul_z_world<T>(q, s, c);
264   }
265 }

268 // single-axis vector rotation
269 template<rotations Axis, ieee754_floating T>
270 [[nodiscard, gnu::always_inline]] inline constexpr micron::vector_3<T>
271 rotate_axis(const micron::vector_3<T> &v, T angle) noexcept
272 {
273   T s, c;
274   math::sincos<T>(angle, s, c);
275   if constexpr ( __impl_euler::axis_of_v<Axis> == u8(0) ) {
276     return micron::vector_3<T>{ v.x, c * v.y - s * v.z, s * v.y + c * v.z };
277   } else if constexpr ( __impl_euler::axis_of_v<Axis> == u8(1) ) {
278     return micron::vector_3<T>{ c * v.x + s * v.z, v.y, -s * v.x + c * v.z };
279   } else {
280     return micron::vector_3<T>{ c * v.x - s * v.y, s * v.x + c * v.y, v.z };
281   }
282 }

284 // %%%%%%%%%%%%%%%%%%%%%%%%%%#%%%%%%%%%%%%
285 // euler to quaternion (all 12 sequences, closed form)
286 template<euler_order Ord, ieee754_floating T>
287 [[nodiscard, gnu::flatten]] inline constexpr quaternion<T>
288 from_euler(T a, T b, T c) noexcept
289 {
290   T s1, c1, s2, c2, s3, c3;
291   math::sincos<T>(a * T(0.5), s1, c1);
292   math::sincos<T>(b * T(0.5), s2, c2);
293   math::sincos<T>(c * T(0.5), s3, c3);
294 
295   // Tait-Bryan
296   if constexpr ( Ord == euler_order::XYZ ) {
297     return quaternion<T>{ s1 * c2 * c3 + c1 * s2 * s3, c1 * s2 * c3 - s1 * c2 * s3, c1 * c2 * s3 + s1 * s2 * c3,
298                           c1 * c2 * c3 - s1 * s2 * s3 };
299   } else if constexpr ( Ord == euler_order::XZY ) {
300     return quaternion<T>{ s1 * c2 * c3 - c1 * s2 * s3, c1 * c2 * s3 - s1 * s2 * c3, c1 * s2 * c3 + s1 * c2 * s3,
301                           c1 * c2 * c3 + s1 * s2 * s3 };
302   } else if constexpr ( Ord == euler_order::YXZ ) {
303     return quaternion<T>{ c1 * s2 * c3 + s1 * c2 * s3, s1 * c2 * c3 - c1 * s2 * s3, c1 * c2 * s3 - s1 * s2 * c3,
304                           c1 * c2 * c3 + s1 * s2 * s3 };
305   } else if constexpr ( Ord == euler_order::YZX ) {
306     return quaternion<T>{ c1 * c2 * s3 + s1 * s2 * c3, s1 * c2 * c3 + c1 * s2 * s3, c1 * s2 * c3 - s1 * c2 * s3,
307                           c1 * c2 * c3 - s1 * s2 * s3 };
308   } else if constexpr ( Ord == euler_order::ZXY ) {
309     return quaternion<T>{ c1 * s2 * c3 - s1 * c2 * s3, c1 * c2 * s3 + s1 * s2 * c3, s1 * c2 * c3 + c1 * s2 * s3,
310                           c1 * c2 * c3 - s1 * s2 * s3 };
311   } else if constexpr ( Ord == euler_order::ZYX ) {
312     return quaternion<T>{ c1 * c2 * s3 - s1 * s2 * c3, c1 * s2 * c3 + s1 * c2 * s3, s1 * c2 * c3 - c1 * s2 * s3,
313                           c1 * c2 * c3 + s1 * s2 * s3 };
314     // true Euler
315   } else if constexpr ( Ord == euler_order::XYX ) {
316     return quaternion<T>{ c1 * c2 * s3 + s1 * c2 * c3, c1 * s2 * c3 + s1 * s2 * s3, s1 * s2 * c3 - c1 * s2 * s3,
317                           c1 * c2 * c3 - s1 * c2 * s3 };
318   } else if constexpr ( Ord == euler_order::XZX ) {
319     return quaternion<T>{ c1 * c2 * s3 + s1 * c2 * c3, c1 * s2 * s3 - s1 * s2 * c3, s1 * s2 * s3 + c1 * s2 * c3,
320                           c1 * c2 * c3 - s1 * c2 * s3 };
321   } else if constexpr ( Ord == euler_order::YXY ) {
322     return quaternion<T>{ c1 * s2 * c3 + s1 * s2 * s3, c1 * c2 * s3 + s1 * c2 * c3, c1 * s2 * s3 - s1 * s2 * c3,
323                           c1 * c2 * c3 - s1 * c2 * s3 };
324   } else if constexpr ( Ord == euler_order::YZY ) {
325     return quaternion<T>{ s1 * s2 * c3 - c1 * s2 * s3, c1 * c2 * s3 + s1 * c2 * c3, s1 * s2 * s3 + c1 * s2 * c3,
326                           c1 * c2 * c3 - s1 * c2 * s3 };
327   } else if constexpr ( Ord == euler_order::ZXZ ) {
328     return quaternion<T>{ c1 * s2 * c3 + s1 * s2 * s3, s1 * s2 * c3 - c1 * s2 * s3, c1 * c2 * s3 + s1 * c2 * c3,
329                           c1 * c2 * c3 - s1 * c2 * s3 };
330   } else {      // ZYZ
331     return quaternion<T>{ c1 * s2 * s3 - s1 * s2 * c3, c1 * s2 * c3 + s1 * s2 * s3, c1 * c2 * s3 + s1 * c2 * c3,
332                           c1 * c2 * c3 - s1 * c2 * s3 };
333   }
334 }

345 
346 template<euler_order Ord, ieee754_floating T>
347 [[nodiscard, gnu::flatten]] inline constexpr mat<T, 3, 3>
348 from_euler_matrix(T a, T b, T c) noexcept
349 {
350   T sa, ca, sb, cb, sg, cg;
351   math::sincos<T>(a, sa, ca);
352   math::sincos<T>(b, sb, cb);
353   math::sincos<T>(c, sg, cg);
354   mat<T, 3, 3> R{};
355   T *__restrict__ d = R.data;
356 
357   if constexpr ( Ord == euler_order::XYZ ) {
358     d[0] = cb * cg;
359     d[1] = -cb * sg;
360     d[2] = sb;
361     d[3] = sa * sb * cg + ca * sg;
362     d[4] = ca * cg - sa * sb * sg;
363     d[5] = -sa * cb;
364     d[6] = sa * sg - ca * sb * cg;
365     d[7] = ca * sb * sg + sa * cg;
366     d[8] = ca * cb;
367   } else if constexpr ( Ord == euler_order::XZY ) {
368     d[0] = cb * cg;
369     d[1] = -sb;
370     d[2] = cb * sg;
371     d[3] = ca * sb * cg + sa * sg;
372     d[4] = ca * cb;
373     d[5] = ca * sb * sg - sa * cg;
374     d[6] = sa * sb * cg - ca * sg;
375     d[7] = sa * cb;
376     d[8] = sa * sb * sg + ca * cg;
377   } else if constexpr ( Ord == euler_order::YXZ ) {
378     d[0] = ca * cg + sa * sb * sg;
379     d[1] = sa * sb * cg - ca * sg;
380     d[2] = sa * cb;
381     d[3] = cb * sg;
382     d[4] = cb * cg;
383     d[5] = -sb;
384     d[6] = ca * sb * sg - sa * cg;
385     d[7] = sa * sg + ca * sb * cg;
386     d[8] = ca * cb;
387   } else if constexpr ( Ord == euler_order::YZX ) {
388     d[0] = ca * cb;
389     d[1] = sa * sg - ca * sb * cg;
390     d[2] = ca * sb * sg + sa * cg;
391     d[3] = sb;
392     d[4] = cb * cg;
393     d[5] = -cb * sg;
394     d[6] = -sa * cb;
395     d[7] = sa * sb * cg + ca * sg;
396     d[8] = ca * cg - sa * sb * sg;
397   } else if constexpr ( Ord == euler_order::ZXY ) {
398     d[0] = ca * cg - sa * sb * sg;
399     d[1] = -sa * cb;
400     d[2] = ca * sg + sa * sb * cg;
401     d[3] = sa * cg + ca * sb * sg;
402     d[4] = ca * cb;
403     d[5] = sa * sg - ca * sb * cg;
404     d[6] = -cb * sg;
405     d[7] = sb;
406     d[8] = cb * cg;
407   } else if constexpr ( Ord == euler_order::ZYX ) {
408     d[0] = ca * cb;
409     d[1] = ca * sb * sg - sa * cg;
410     d[2] = ca * sb * cg + sa * sg;
411     d[3] = sa * cb;
412     d[4] = sa * sb * sg + ca * cg;
413     d[5] = sa * sb * cg - ca * sg;
414     d[6] = -sb;
415     d[7] = cb * sg;
416     d[8] = cb * cg;
417   } else if constexpr ( Ord == euler_order::XYX ) {
418     d[0] = cb;
419     d[1] = sb * sg;
420     d[2] = sb * cg;
421     d[3] = sa * sb;
422     d[4] = ca * cg - sa * cb * sg;
423     d[5] = -ca * sg - sa * cb * cg;
424     d[6] = -ca * sb;
425     d[7] = sa * cg + ca * cb * sg;
426     d[8] = ca * cb * cg - sa * sg;
427   } else if constexpr ( Ord == euler_order::XZX ) {
428     d[0] = cb;
429     d[1] = -sb * cg;
430     d[2] = sb * sg;
431     d[3] = ca * sb;
432     d[4] = ca * cb * cg - sa * sg;
433     d[5] = -ca * cb * sg - sa * cg;
434     d[6] = sa * sb;
435     d[7] = sa * cb * cg + ca * sg;
436     d[8] = ca * cg - sa * cb * sg;
437   } else if constexpr ( Ord == euler_order::YXY ) {
438     d[0] = ca * cg - sa * cb * sg;
439     d[1] = sa * sb;
440     d[2] = ca * sg + sa * cb * cg;
441     d[3] = sb * sg;
442     d[4] = cb;
443     d[5] = -sb * cg;
444     d[6] = -sa * cg - ca * cb * sg;
445     d[7] = ca * sb;
446     d[8] = ca * cb * cg - sa * sg;
447   } else if constexpr ( Ord == euler_order::YZY ) {
448     d[0] = ca * cb * cg - sa * sg;
449     d[1] = -ca * sb;
450     d[2] = ca * cb * sg + sa * cg;
451     d[3] = sb * cg;
452     d[4] = cb;
453     d[5] = sb * sg;
454     d[6] = -sa * cb * cg - ca * sg;
455     d[7] = sa * sb;
456     d[8] = ca * cg - sa * cb * sg;
457   } else if constexpr ( Ord == euler_order::ZXZ ) {
458     d[0] = ca * cg - sa * cb * sg;
459     d[1] = -ca * sg - sa * cb * cg;
460     d[2] = sa * sb;
461     d[3] = sa * cg + ca * cb * sg;
462     d[4] = ca * cb * cg - sa * sg;
463     d[5] = -ca * sb;
464     d[6] = sb * sg;
465     d[7] = sb * cg;
466     d[8] = cb;
467   } else {      // ZYZ
468     d[0] = ca * cb * cg - sa * sg;
469     d[1] = -ca * cb * sg - sa * cg;
470     d[2] = ca * sb;
471     d[3] = sa * cb * cg + ca * sg;
472     d[4] = ca * cg - sa * cb * sg;
473     d[5] = sa * sb;
474     d[6] = -sb * cg;
475     d[7] = sb * sg;
476     d[8] = cb;
477   }
478   return R;
479 }

488 template<euler_order Ord, ieee754_floating T>
489 [[nodiscard, gnu::flatten]] inline constexpr mat<T, 4, 4>
490 from_euler_matrix4(T a, T b, T c) noexcept
491 {
492   const mat<T, 3, 3> R = from_euler_matrix<Ord, T>(a, b, c);
493   mat<T, 4, 4> M{};
494   T *__restrict__ d = M.data;
495   const T *__restrict__ r = R.data;
496   d[0] = r[0];
497   d[1] = r[1];
498   d[2] = r[2];
499   d[3] = T(0);
500   d[4] = r[3];
501   d[5] = r[4];
502   d[6] = r[5];
503   d[7] = T(0);
504   d[8] = r[6];
505   d[9] = r[7];
506   d[10] = r[8];
507   d[11] = T(0);
508   d[12] = T(0);
509   d[13] = T(0);
510   d[14] = T(0);
511   d[15] = T(1);
512   return M;
513 }

528 template<ieee754_floating T>
529 [[nodiscard, gnu::always_inline]] inline constexpr T
530 clamp_unit(T x) noexcept
531 {
532   if ( x > T(1) ) return T(1);
533   if ( x < T(-1) ) return T(-1);
534   return x;
535 }

537 template<ieee754_floating T>
538 [[nodiscard, gnu::always_inline]] inline constexpr T
539 gimbal_floor() noexcept
540 {
541   return T(1) - math::default_eps<T>();

546 template<euler_order Ord, ieee754_floating T>
547 [[nodiscard, gnu::flatten]] inline constexpr micron::vector_3<T>
548 to_euler_matrix(const mat<T, 3, 3> &R) noexcept
549 {
550   const T *__restrict__ d = R.data;
551   const T R00 = d[0], R01 = d[1], R02 = d[2];
552   const T R10 = d[3], R11 = d[4], R12 = d[5];
553   const T R20 = d[6], R21 = d[7], R22 = d[8];
554   const T glim = __impl_euler::gimbal_floor<T>();
555   T a, b, c;
556 
557   // NOTE: gimbal-lock formulas are unified across both beta branches per sequence
558   // each chosen atan2 pair gives sin/cos of the merged angle with the
559   // correct sign in both gimbal poles, so no further sign branching needed
560   if constexpr ( Ord == euler_order::XYZ ) {
561     const T sb = __impl_euler::clamp_unit<T>(R02);
562     b = math::asin<T>(sb);
563     if ( math::fabs(sb) >= glim ) {
564       c = T(0);
565       a = math::atan2<T>(R21, R11);
566     } else {
567       a = math::atan2<T>(-R12, R22);
568       c = math::atan2<T>(-R01, R00);
569     }
570   } else if constexpr ( Ord == euler_order::XZY ) {
571     const T sb = __impl_euler::clamp_unit<T>(-R01);
572     b = math::asin<T>(sb);
573     if ( math::fabs(sb) >= glim ) {
574       c = T(0);
575       a = math::atan2<T>(-R12, R22);
576     } else {
577       a = math::atan2<T>(R21, R11);
578       c = math::atan2<T>(R02, R00);
579     }
580   } else if constexpr ( Ord == euler_order::YXZ ) {
581     const T sb = __impl_euler::clamp_unit<T>(-R12);
582     b = math::asin<T>(sb);
583     if ( math::fabs(sb) >= glim ) {
584       c = T(0);
585       a = math::atan2<T>(-R20, R00);
586     } else {
587       a = math::atan2<T>(R02, R22);
588       c = math::atan2<T>(R10, R11);
589     }
590   } else if constexpr ( Ord == euler_order::YZX ) {
591     const T sb = __impl_euler::clamp_unit<T>(R10);
592     b = math::asin<T>(sb);
593     if ( math::fabs(sb) >= glim ) {
594       c = T(0);
595       a = math::atan2<T>(R02, R22);
596     } else {
597       a = math::atan2<T>(-R20, R00);
598       c = math::atan2<T>(-R12, R11);
599     }
600   } else if constexpr ( Ord == euler_order::ZXY ) {
601     const T sb = __impl_euler::clamp_unit<T>(R21);
602     b = math::asin<T>(sb);
603     if ( math::fabs(sb) >= glim ) {
604       c = T(0);
605       a = math::atan2<T>(R10, R00);
606     } else {
607       a = math::atan2<T>(-R01, R11);
608       c = math::atan2<T>(-R20, R22);
609     }
610   } else if constexpr ( Ord == euler_order::ZYX ) {
611     const T sb = __impl_euler::clamp_unit<T>(-R20);
612     b = math::asin<T>(sb);
613     if ( math::fabs(sb) >= glim ) {
614       c = T(0);
615       a = math::atan2<T>(-R01, R11);
616     } else {
617       a = math::atan2<T>(R10, R00);
618       c = math::atan2<T>(R21, R22);
619     }
620   } else if constexpr ( Ord == euler_order::XYX ) {
621     const T cb = __impl_euler::clamp_unit<T>(R00);
622     b = math::acos<T>(cb);
623     if ( math::fabs(cb) >= glim ) {
624       c = T(0);
625       a = math::atan2<T>(R21, R11);
626     } else {
627       a = math::atan2<T>(R10, -R20);
628       c = math::atan2<T>(R01, R02);
629     }
630   } else if constexpr ( Ord == euler_order::XZX ) {
631     const T cb = __impl_euler::clamp_unit<T>(R00);
632     b = math::acos<T>(cb);
633     if ( math::fabs(cb) >= glim ) {
634       c = T(0);
635       a = math::atan2<T>(-R12, R22);
636     } else {
637       a = math::atan2<T>(R20, R10);
638       c = math::atan2<T>(R02, -R01);
639     }
640   } else if constexpr ( Ord == euler_order::YXY ) {
641     const T cb = __impl_euler::clamp_unit<T>(R11);
642     b = math::acos<T>(cb);
643     if ( math::fabs(cb) >= glim ) {
644       c = T(0);
645       a = math::atan2<T>(-R20, R00);
646     } else {
647       a = math::atan2<T>(R01, R21);
648       c = math::atan2<T>(R10, -R12);
649     }
650   } else if constexpr ( Ord == euler_order::YZY ) {
651     const T cb = __impl_euler::clamp_unit<T>(R11);
652     b = math::acos<T>(cb);
653     if ( math::fabs(cb) >= glim ) {
654       c = T(0);
655       a = math::atan2<T>(R02, R22);
656     } else {
657       a = math::atan2<T>(R21, -R01);
658       c = math::atan2<T>(R12, R10);
659     }
660   } else if constexpr ( Ord == euler_order::ZXZ ) {
661     const T cb = __impl_euler::clamp_unit<T>(R22);
662     b = math::acos<T>(cb);
663     if ( math::fabs(cb) >= glim ) {
664       c = T(0);
665       a = math::atan2<T>(R10, R00);
666     } else {
667       a = math::atan2<T>(R02, -R12);
668       c = math::atan2<T>(R20, R21);
669     }
670   } else {      // ZYZ
671     const T cb = __impl_euler::clamp_unit<T>(R22);
672     b = math::acos<T>(cb);
673     if ( math::fabs(cb) >= glim ) {
674       c = T(0);
675       a = math::atan2<T>(-R01, R11);
676     } else {
677       a = math::atan2<T>(R12, R02);
678       c = math::atan2<T>(R21, -R20);
679     }
680   }
681   return micron::vector_3<T>{ a, b, c };
682 }

684 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
685 // quaternion to euler (all 12 sequences)
686 
687 template<euler_order Ord, ieee754_floating T>
688 [[nodiscard, gnu::flatten]] inline constexpr micron::vector_3<T>
689 to_euler(const quaternion<T> &q) noexcept
690 {
691   const T x = q.x, y = q.y, z = q.z, w = q.w;
692   const T xx = x * x, yy = y * y, zz = z * z;
693   const T xy = x * y, xz = x * z, yz = y * z;
694   const T wx = w * x, wy = w * y, wz = w * z;
695   // matrix elements (lazy-eval per sequence)
696   const T R00 = T(1) - T(2) * (yy + zz);
697   const T R01 = T(2) * (xy - wz);
698   const T R02 = T(2) * (xz + wy);
699   const T R10 = T(2) * (xy + wz);
700   const T R11 = T(1) - T(2) * (xx + zz);
701   const T R12 = T(2) * (yz - wx);
702   const T R20 = T(2) * (xz - wy);
703   const T R21 = T(2) * (yz + wx);
704   const T R22 = T(1) - T(2) * (xx + yy);
705   mat<T, 3, 3> R{};
706   R.data[0] = R00;
707   R.data[1] = R01;
708   R.data[2] = R02;
709   R.data[3] = R10;
710   R.data[4] = R11;
711   R.data[5] = R12;
712   R.data[6] = R20;
713   R.data[7] = R21;
714   R.data[8] = R22;
715   return to_euler_matrix<Ord, T>(R);
716 }
