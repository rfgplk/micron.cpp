64 // q_dot = (1/2) * q (X) (omega, 0)
65 // NOTE: omega is the angular velocity in the BODY frame
66 // For a world-frame angular velocity omega_w, use (1/2) * (omega_w, 0) (X) q
67 template<ieee754_floating T>
68 [[nodiscard]] inline constexpr quaternion<T>
69 derivative(const quaternion<T> &q, const micron::vector_3<T> &omega) noexcept
70 {
71   const quaternion<T> w_q{ omega.x, omega.y, omega.z, T(0) };
72   const auto p = multiply<T>(q, w_q);
73   return quaternion<T>{ p.x * T(0.5), p.y * T(0.5), p.z * T(0.5), p.w * T(0.5) };
74 }

76 template<ieee754_floating T>
77 [[nodiscard]] inline constexpr quaternion<T>
78 integrate(const quaternion<T> &q, const micron::vector_3<T> &omega, T dt) noexcept
79 {
80   const T n2 = omega.x * omega.x + omega.y * omega.y + omega.z * omega.z;
81   const T half_phi2 = n2 * dt * dt * T(0.25);
82 
83   T sx, sy, sz, c;
84   if ( half_phi2 < __impl_kinematics::small_half_phi_sq<T>() ) {
85     const T hp4 = half_phi2 * half_phi2;
86     const T axis_scale = T(0.5) * dt * (T(1) - half_phi2 * (T(1) / T(6)) + hp4 * (T(1) / T(120)));
87     c = T(1) - half_phi2 * T(0.5) + hp4 * (T(1) / T(24));
88     sx = omega.x * axis_scale;
89     sy = omega.y * axis_scale;
90     sz = omega.z * axis_scale;
91   } else {
92     const T n = math::fsqrt(n2);
93     const T half_phi = T(0.5) * n * dt;
94     const T s = math::sin<T>(half_phi);
95     c = math::cos<T>(half_phi);
96     const T axis_scale = s / n;
97     sx = omega.x * axis_scale;
98     sy = omega.y * axis_scale;
99     sz = omega.z * axis_scale;
100   }
101   const quaternion<T> dq{ sx, sy, sz, c };
102   return multiply<T>(q, dq).normalized();
103 }

105 template<ieee754_floating T>
106 [[nodiscard]] inline constexpr quaternion<T>
107 integrate_small_angle(const quaternion<T> &q, const micron::vector_3<T> &omega, T dt) noexcept
108 {
109   const T n2 = omega.x * omega.x + omega.y * omega.y + omega.z * omega.z;
110   const T phi2 = n2 * dt * dt;
111   const T phi4 = phi2 * phi2;
112   const T s = T(0.5) * dt * (T(1) - phi2 * (T(1) / T(24)) + phi4 * (T(1) / T(1920)));
113   const T c = T(1) - phi2 * T(0.125) + phi4 * (T(1) / T(384));
114   const quaternion<T> dq{ omega.x * s, omega.y * s, omega.z * s, c };
115   return multiply<T>(q, dq).normalized();
116 }

118 template<ieee754_floating T>
119 [[nodiscard]] inline constexpr quaternion<T>
120 integrate_pade(const quaternion<T> &q, const micron::vector_3<T> &omega, T dt) noexcept
121 {
122   const T h = T(0.5) * dt;
123   const T vx = omega.x * h, vy = omega.y * h, vz = omega.z * h;
124   const T v2 = vx * vx + vy * vy + vz * vz;
125   const T w0 = T(1) - v2 * (T(1) / T(12));
126   const T a = w0 * w0;
127   const T b = v2 * T(0.25);
128   const T denom = a + b;
129   if ( denom <= __safe_min_n2<T>() ) return quaternion<T>::nan();
130   const T inv_d = T(1) / denom;
131   const T num_w = (a - b) * inv_d;
132   const T num_v = w0 * inv_d;
133   const quaternion<T> dq{ vx * num_v, vy * num_v, vz * num_v, num_w };
134   return multiply<T>(q, dq).normalized();
135 }

137 template<ieee754_floating T>
138 [[nodiscard]] inline constexpr micron::vector_3<T>
139 angular_velocity(const quaternion<T> &q0, const quaternion<T> &q1, T dt) noexcept
140 {
141   if ( !ieee::is_finite<T>(dt) || math::fabs(dt) <= math::default_eps<T>() ) {
142     return __impl_kinematics::nan_vec<T>();
143   }
144   const auto dq = multiply<T>(inverse_unit<T>(q0), q1);
145   const auto aa = to_axis_angle<T>(dq);
146   const T s = aa.angle / dt;
147   return micron::vector_3<T>{ aa.axis.x * s, aa.axis.y * s, aa.axis.z * s };
148 }

150 template<ieee754_floating T>
151 [[nodiscard]] inline constexpr micron::vector_3<T>
152 log_map_pade(const quaternion<T> &q) noexcept
153 {
154   const T sgn = (q.w >= T(0)) ? T(1) : T(-1);
155   const T cx = q.x * sgn, cy = q.y * sgn, cz = q.z * sgn, cw = q.w * sgn;
156   if ( cw < __impl_kinematics::log_pade_cw_floor<T>() ) {
157     const auto aa = to_axis_angle<T>(quaternion<T>{ cx, cy, cz, cw });
158     return micron::vector_3<T>{ aa.axis.x * aa.angle, aa.axis.y * aa.angle, aa.axis.z * aa.angle };
159   }
160   const T s2 = cx * cx + cy * cy + cz * cz;
161   const T cw2 = cw * cw;
162   const T num = T(15) * cw2 + T(4) * s2;
163   const T den = (T(15) * cw2 + T(9) * s2) * cw;
164   const T scale = T(2) * num / den;
165   return micron::vector_3<T>{ cx * scale, cy * scale, cz * scale };
166 }

168 template<ieee754_floating T>
169 [[nodiscard]] inline constexpr micron::vector_3<T>
170 angular_velocity_pade(const quaternion<T> &q0, const quaternion<T> &q1, T dt) noexcept
171 {
172   if ( !ieee::is_finite<T>(dt) || math::fabs(dt) <= math::default_eps<T>() ) {
173     return __impl_kinematics::nan_vec<T>();
174   }
175   const auto v = log_map_pade<T>(multiply<T>(inverse_unit<T>(q0), q1));
176   const T inv_dt = T(1) / dt;
177   return micron::vector_3<T>{ v.x * inv_dt, v.y * inv_dt, v.z * inv_dt };
178 }
