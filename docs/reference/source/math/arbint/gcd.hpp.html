19 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
20 // the gcd ladder
21 //
22 // THREE RUNGS AND FOUR THRESHOLDS, and that mismatch is the one thing to get right here.
23 // threshold::gcd_hgcd is NOT a rung -- it is the half-gcd recursion's own cutoff, the size below
24 // which hgcd iterates instead of recursing, exactly the way `cutoff` is internal to the Karatsuba
25 // recursion and never appears in pick_mul. gcd_dc is where the ladder starts calling hgcd at all.
26 // somebody will otherwise write a fourth case and wonder why it is dead.
27 //
28 // WHAT IS BUILT is gcd_tiers_built, and the ladder saturates there rather than pretending, the same
29 // way mul() clamps to mpn::tiers_built. gcd_tier_cap is the separate question of what the automatic
30 // path may SELECT, which is a measurement result and not a capability.
31 //
32 // there is no solver:: tag for a gcd tier and there should not be: the Solver template parameter
33 // names a MULTIPLICATION tier and means nothing here. pinning a gcd tier is gcd_with<>, which is
34 // what the differential and the crossover bench call, and the class always takes the ladder.

43 // ALL THREE TIERS ARE BUILT. what the automatic path may SELECT is gcd_tier_cap, which is a
44 // measurement result and not a capability -- Lehmer earned its window and sits at 1; the dc tier
45 // ships built, differentially tested and capped out until the crossover bench finds one for it,
46 // the same treatment comba and toom-3 have. gcd_with<gcd_algo::dc> and a --def on the cap are how
47 // it gets exercised meanwhile.
48 inline constexpr gcd_algo gcd_tiers_built = gcd_algo::dc;

50 static_assert(static_cast<u8>(gcd_tier_cap) <= static_cast<u8>(gcd_tiers_built),
51               "arbint: MICRON_ARBINT_GCD_TIER_CAP names a tier above mpn::gcd_tiers_built -- the ladder cannot select "
52               "what is not built");

54 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
55 // the divide-and-conquer tier
56 //
57 // reduce the pair by half with hgcd, repeat, and hand the remainder to Lehmer once the operands are
58 // small enough that the recursion's constant no longer pays. THE MATRIX IS DISCARDED at every
59 // round: gcd wants only hgcd's side effect on the pair, which is a reduction that preserves the
60 // gcd. gcdext is the caller that would want the matrix, and it stays on the Lehmer cofactors.
61 
62 [[nodiscard, gnu::flatten]] inline constexpr usize
63 gcd_dc_itch(usize un, usize vn) noexcept
64 {
65   if ( vn > un ) {
66     const usize t = un;
67     un = vn;
68     vn = t;
69   }
70   if ( vn == 0 ) return 0;
71 
72   // the two working copies, then whichever of the three consumers is largest -- they are sequential
73   usize w = hgcd_mat_itch(un) + hgcd_itch(un);
74   const usize d = subdiv_step_itch(un);
75   if ( d > w ) w = d;
76   // the hand-off may happen at any size the loop exits at, including un itself when the pair is
77   // already below the threshold, so Lehmer is charged at full width rather than at the threshold
78   const usize l = gcd_lehmer_itch(un, un);
79   if ( l > w ) w = l;
80 
81   return 2u * un + w;
82 }

84 // same contract as gcd_binary and gcd_lehmer: un >= vn >= 1, both normalized and nonzero, gp holds
85 // vn limbs, the operands survive the call, and the return is the gcd's length.
86 //
87 // the operands are copied in because hgcd MUTATES its pair in place -- it has to, the recursion
88 // works on a sub-slice of the same buffers -- and because the differential runs two tiers back to
89 // back on the same inputs.
90 [[nodiscard]] inline constexpr usize
91 gcd_dc(limb_t *gp, const limb_t *up, usize un, const limb_t *vp, usize vn, limb_t *scratch) noexcept
92 {
93   limb_t *a = scratch;      // un
94   limb_t *b = a + un;       // un
95   limb_t *const work = b + un;
96 
97   copyi(a, up, un);
98   copyi(b, vp, vn);
99   zero(b + vn, un - vn);
100 
101   usize n = un;
102   for ( ;; ) {
103     usize an = normalize(a, n);
104     const usize bn = normalize(b, n);
105     if ( bn == 0 ) {
106       n = an;
107       break;
108     }
109     if ( an < bn ) {
110       limb_t *const t = a;
111       a = b;
112       b = t;
113       an = bn;
114     }
115     n = an;
116     if ( n <= threshold::gcd_dc ) break;
117 
118     // hgcd reads n limbs of both, so the shorter one is padded rather than trusted to be clean
119     zero(b + normalize(b, n), n - normalize(b, n));
120 
121     hgcd_mat M{};
122     hgcd_mat_init(M, n, work);
123     const usize k = hgcd(a, b, n, M, work + hgcd_mat_itch(n));
124     if ( k != 0 ) {
125       n = k;
126       continue;
127     }
128 
129     // hgcd could not move the pair -- one genuine Euclidean step, which always can
130     __gcd_hook hook{};
131     const usize t = __subdiv_step(a, b, n, hook, work);
132     if ( t == 0 ) break;
133     n = t;
134   }
135 
136   usize an = normalize(a, n);
137   usize bn = normalize(b, n);
138   if ( bn == 0 ) {
139     if ( an == 0 ) return 0;
140     copyi(gp, a, an);
141     return an;
142   }
143   if ( an < bn ) {
144     limb_t *const t = a;
145     a = b;
146     b = t;
147     const usize tn = an;
148     an = bn;
149     bn = tn;
150   }
151   return gcd_lehmer(gp, a, an, b, bn, work);
152 }

157 [[nodiscard, gnu::flatten]] inline constexpr usize
158 gcd_tier_itch_for(gcd_algo k, usize un, usize vn) noexcept
159 {
160   if ( k == gcd_algo::binary ) return gcd_binary_itch(un, vn);
161   if ( k == gcd_algo::lehmer ) return gcd_lehmer_itch(un, vn);
162   return gcd_dc_itch(un, vn);
163 }

165 [[nodiscard, gnu::flatten]] inline constexpr usize
166 gcd_tier_itch(usize un, usize vn) noexcept
167 {
168   // every tier the LADDER could pick, maxed rather than summed -- it picks one at runtime, and a
169   // bounded frame is sized once at instantiation.
170   const usize a = gcd_binary_itch(un, vn);
171   const usize b = gcd_lehmer_itch(un, vn);
172   usize m = a > b ? a : b;
173 
174   // the dc tier is charged only where the ladder could actually reach it. BOTH gates are
175   // compile-time constants at instantiation -- a --def on the cap or the threshold is visible right
176   // here -- and gcd() dispatches on the REDUCED pair, which is never longer than un, so a size that
177   // cannot select dc on the way in cannot select it later either. without this, every bounded
178   // arbuint would carry hgcd's frame, including the widths that can never reach it.
179   if ( static_cast<u8>(gcd_tier_cap) >= static_cast<u8>(gcd_algo::dc) && un >= threshold::gcd_dc ) {
180     const usize c = gcd_dc_itch(un, vn);
181     if ( c > m ) m = c;
182   }
183   return m;
184 }

186 // the prologue's levelling division, plus whichever tier follows it. the two are SEQUENTIAL and
187 // share everything past the reduced pair, so this is a max and not a sum.
188 [[nodiscard, gnu::flatten]] inline constexpr usize
189 gcd_itch(usize un, usize vn) noexcept
190 {
191   if ( un < vn ) {
192     const usize t = un;
193     un = vn;
194     vn = t;
195   }
196   if ( vn == 0 ) return 0;
197   if ( un == vn ) return gcd_tier_itch(vn, vn);
198 
199   const usize pair = vn;      // the remainder the levelling division leaves behind
200   const usize head = (un - vn + 1u) + divrem_itch(un, vn);
201   const usize tier = gcd_tier_itch(vn, vn);
202   return pair + (head > tier ? head : tier);
203 }

205 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
206 // the friendly entry
207 //
208 // gp takes min(un, vn) limbs and the return is the gcd's length, zero only when both operands were
209 // zero. up and vp survive the call.
210 //
211 // the degenerate cases are stripped HERE and never reach a tier, which is what lets every tier
212 // state "normalized and nonzero" as a precondition instead of re-checking. the answers match
213 // micron::math::gcd<T> in math/generic.hpp exactly -- gcd(0,0) == 0, gcd(0,x) == x.
214 
215 [[nodiscard]] inline constexpr usize
216 gcd(limb_t *gp, const limb_t *up, usize un, const limb_t *vp, usize vn, limb_t *scratch) noexcept
217 {
218   un = normalize(up, un);
219   vn = normalize(vp, vn);
220 
221   if ( vn == 0 ) {
222     copyi(gp, up, un);
223     return un;
224   }
225   if ( un == 0 ) {
226     copyi(gp, vp, vn);
227     return vn;
228   }
229   if ( cmp_var(up, un, vp, vn) < 0 ) {
230     const limb_t *tp = up;
231     up = vp;
232     vp = tp;
233     const usize tn = un;
234     un = vn;
235     vn = tn;
236   }
237 
238   const limb_t *ap = up;
239   usize an = un;
240   const limb_t *bp = vp;
241   usize bn = vn;
242   limb_t *work = scratch;
243 
244   if ( un > vn ) {
245     // one division levels the sizes, so the tier below sees a balanced pair. after it the operands
246     // are (v, u mod v), both at most vn limbs, which is what gcd_itch charges the tier for.
247     limb_t *const b = scratch;      // vn
248     limb_t *const q = b + vn;       // un - vn + 1
249     limb_t *const dwork = q + (un - vn + 1u);
250     divrem(q, b, up, un, vp, vn, dwork);
251     bn = normalize(b, vn);
252     if ( bn == 0 ) {
253       copyi(gp, vp, vn);
254       return vn;
255     }
256     ap = vp;
257     an = vn;
258     bp = b;
259     work = scratch + vn;
260   }
261 
262   if ( an < bn ) {
263     const limb_t *tp = ap;
264     ap = bp;
265     bp = tp;
266     const usize tn = an;
267     an = bn;
268     bn = tn;
269   }
270 
271   switch ( clamp_to(pick_gcd(an, bn), gcd_tier_cap) ) {
272   case gcd_algo::binary:
273     return gcd_binary(gp, ap, an, bp, bn, work);
274   case gcd_algo::dc:
275     return gcd_dc(gp, ap, an, bp, bn, work);
276   default:
277     return gcd_lehmer(gp, ap, an, bp, bn, work);
278   }
279 }

281 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
282 // pinned tiers
283 
284 // the pinned entry may name a tier the ladder would never select, so this CANNOT defer to
285 // gcd_itch's cap-and-threshold gate -- it charges the named tier outright. deferring is how a
286 // gcd_with<dc> in the differential or the bench gets a frame sized for Lehmer and overruns it
287 // without failing a single correctness check.
288 template<gcd_algo K>
289 [[nodiscard, gnu::flatten]] inline constexpr usize
290 gcd_itch_with(usize un, usize vn) noexcept
291 {
292   if ( un < vn ) {
293     const usize t = un;
294     un = vn;
295     vn = t;
296   }
297   if ( vn == 0 ) return 0;
298   if ( un == vn ) return gcd_tier_itch_for(K, vn, vn);
299 
300   const usize pair = vn;
301   const usize head = (un - vn + 1u) + divrem_itch(un, vn);
302   const usize tier = gcd_tier_itch_for(K, vn, vn);
303   return pair + (head > tier ? head : tier);
304 }

306 // the prologue is shared with gcd() -- the degenerate cases and the levelling division are not the
307 // tier's business either way -- and only the final dispatch is pinned.
308 template<gcd_algo K>
309 [[nodiscard]] inline constexpr usize
310 gcd_with(limb_t *gp, const limb_t *up, usize un, const limb_t *vp, usize vn, limb_t *scratch) noexcept
311 {
312   static_assert(static_cast<u8>(K) <= static_cast<u8>(gcd_tiers_built),
313                 "arbint: this gcd tier is not implemented yet -- see mpn::gcd_tiers_built");
314 
315   un = normalize(up, un);
316   vn = normalize(vp, vn);
317   if ( vn == 0 ) {
318     copyi(gp, up, un);
319     return un;
320   }
321   if ( un == 0 ) {
322     copyi(gp, vp, vn);
323     return vn;
324   }
325   if ( cmp_var(up, un, vp, vn) < 0 ) {
326     const limb_t *tp = up;
327     up = vp;
328     vp = tp;
329     const usize tn = un;
330     un = vn;
331     vn = tn;
332   }
333 
334   const limb_t *ap = up;
335   usize an = un;
336   const limb_t *bp = vp;
337   usize bn = vn;
338   limb_t *work = scratch;
339 
340   if ( un > vn ) {
341     limb_t *const b = scratch;
342     limb_t *const q = b + vn;
343     limb_t *const dwork = q + (un - vn + 1u);
344     divrem(q, b, up, un, vp, vn, dwork);
345     bn = normalize(b, vn);
346     if ( bn == 0 ) {
347       copyi(gp, vp, vn);
348       return vn;
349     }
350     ap = vp;
351     an = vn;
352     bp = b;
353     work = scratch + vn;
354   }
355 
356   if ( an < bn ) {
357     const limb_t *tp = ap;
358     ap = bp;
359     bp = tp;
360     const usize tn = an;
361     an = bn;
362     bn = tn;
363   }
364 
365   if constexpr ( K == gcd_algo::binary )
366     return gcd_binary(gp, ap, an, bp, bn, work);
367   else if constexpr ( K == gcd_algo::dc )
368     return gcd_dc(gp, ap, an, bp, bn, work);
369   else
370     return gcd_lehmer(gp, ap, an, bp, bn, work);
371 }
