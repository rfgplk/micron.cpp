14 // mpn
15 //
16 // naked pointers and lengths;
17 // no ownership;
18 // no sign;
19 // no normalization
20 //
21 //  .. lengths are in limbs and every routine that takes n requires n >= 1 unless it says otherwise
22 //  .. inputs need not be normalized; outputs are not normalized either (call normalize())
23 //  .. rp may equal ap or bp for the in-place families (add/sub/shift/logic); it may not partially overlap
24 //  .. the multiply outputs take __restrict__ and genuinely require a distinct destination

56 [[nodiscard, gnu::flatten]] inline constexpr int
57 cmp(const limb_t *ap, const limb_t *bp, usize n) noexcept
58 {
59   for ( usize i = n; i-- > 0; )
60     if ( ap[i] != bp[i] ) return ap[i] < bp[i] ? -1 : 1;
61   return 0;
62 }

100 [[nodiscard, gnu::flatten]] inline constexpr limb_t
101 mul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
102 {
103   limb_t cy = 0;
104   for ( usize i = 0; i < n; ++i ) {
105     limb_t lo = 0, hi = 0;
106     muladd_wide(ap[i], b, cy, lo, hi);
107     rp[i] = lo;
108     cy = hi;
109   }
110   return cy;
111 }

113 // rp += ap * b. a*b + cy + rp[i] <= (2^w-1)^2 + 2(2^w-1) == 2^2w - 1
114 [[nodiscard, gnu::flatten]] inline constexpr limb_t
115 addmul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
116 {
117   limb_t cy = 0;
118   for ( usize i = 0; i < n; ++i ) {
119     limb_t lo = 0, hi = 0;
120     muladd2_wide(ap[i], b, cy, rp[i], lo, hi);
121     rp[i] = lo;
122     cy = hi;
123   }
124   return cy;
125 }

127 // rp -= ap * b. hi + bw cannot wrap
128 [[nodiscard, gnu::flatten]] inline constexpr limb_t
129 submul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
130 {
131   limb_t cy = 0;
132   for ( usize i = 0; i < n; ++i ) {
133     limb_t lo = 0, hi = 0;
134     muladd_wide(ap[i], b, cy, lo, hi);
135     limb_t d = 0;
136     const limb_t bw = subb(rp[i], lo, 0, d);
137     rp[i] = d;
138     cy = static_cast<limb_t>(hi + bw);
139   }
140   return cy;
141 }

143 [[nodiscard, gnu::flatten]] inline constexpr limb_t
144 lshift(limb_t *rp, const limb_t *ap, usize n, usize cnt) noexcept
145 {
146   if ( n == 0 ) return 0;
147   const usize tnc = limb_bits - cnt;
148   limb_t high = ap[n - 1];
149   const limb_t ret = static_cast<limb_t>(high >> tnc);
150   for ( usize i = n - 1; i > 0; --i ) {
151     const limb_t low = ap[i - 1];
152     rp[i] = static_cast<limb_t>((high << cnt) | (low >> tnc));
153     high = low;
154   }
155   rp[0] = static_cast<limb_t>(high << cnt);
156   return ret;
157 }
158 
159 [[nodiscard, gnu::flatten]] inline constexpr limb_t
160 rshift(limb_t *rp, const limb_t *ap, usize n, usize cnt) noexcept
161 {
162   if ( n == 0 ) return 0;
163   const usize tnc = limb_bits - cnt;
164   limb_t low = ap[0];
165   const limb_t ret = static_cast<limb_t>(low << tnc);
166   for ( usize i = 0; i + 1 < n; ++i ) {
167     const limb_t high = ap[i + 1];
168     rp[i] = static_cast<limb_t>((low >> cnt) | (high << tnc));
169     low = high;
170   }
171   rp[n - 1] = static_cast<limb_t>(low >> cnt);
172   return ret;
173 }

202 [[nodiscard, gnu::flatten]] inline constexpr limb_t
203 add_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
204 {
205   limb_t cy = b;
206   usize i = 0;
207   for ( ; i < n && cy != 0; ++i ) cy = addc(ap[i], cy, 0, rp[i]);
208   if ( rp != ap )
209     for ( ; i < n; ++i ) rp[i] = ap[i];
210   return cy;
211 }

224 [[nodiscard, gnu::flatten]] inline constexpr limb_t
225 add(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn) noexcept
226 {
227   const limb_t cy = add_n(rp, ap, bp, bn);
228   if ( an == bn ) return cy;
229   return add_1(rp + bn, ap + bn, an - bn, cy);
230 }

232 [[nodiscard, gnu::flatten]] inline constexpr limb_t
233 sub(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn) noexcept
234 {
235   const limb_t bw = sub_n(rp, ap, bp, bn);
236   if ( an == bn ) return bw;
237   return sub_1(rp + bn, ap + bn, an - bn, bw);
238 }

240 [[nodiscard, gnu::flatten]] inline constexpr limb_t
241 neg(limb_t *rp, const limb_t *ap, usize n) noexcept
242 {
243   usize i = 0;
244   for ( ; i < n; ++i ) {
245     if ( ap[i] != 0 ) {
246       rp[i] = static_cast<limb_t>(~ap[i] + 1u);
247       for ( ++i; i < n; ++i ) rp[i] = static_cast<limb_t>(~ap[i]);
248       return 1;
249     }
250     rp[i] = 0;
251   }
252   return 0;
253 }

260 [[nodiscard, gnu::flatten]] inline constexpr limb_t
261 cnd_add_n(limb_t cnd, limb_t *rp, const limb_t *ap, const limb_t *bp, usize n) noexcept

372 [[nodiscard, gnu::always_inline]] inline constexpr usize
373 normalize(const limb_t *p, usize n) noexcept
374 {
375   while ( n > 0 && p[n - 1] == 0 ) --n;
376   return n;
377 }

379 [[nodiscard, gnu::flatten]] inline constexpr usize
380 bitlen(const limb_t *p, usize n) noexcept
381 {
382   n = normalize(p, n);
383   if ( n == 0 ) return 0;
384   return (n - 1u) * limb_bits + limb_bitlen(p[n - 1]);
385 }

395 [[nodiscard, gnu::flatten]] inline constexpr usize
396 scan1(const limb_t *p, usize n) noexcept
397 {
398   for ( usize i = 0; i < n; ++i )
399     if ( p[i] != 0 ) return i * limb_bits + limb_ctz(p[i]);

403 [[nodiscard, gnu::always_inline]] inline constexpr bool
404 testbit(const limb_t *p, usize n, usize bit) noexcept
405 {
406   const usize i = bit / limb_bits;
407   if ( i >= n ) return false;
408   return ((p[i] >> (bit % limb_bits)) & limb_t{ 1 }) != 0;
409 }
