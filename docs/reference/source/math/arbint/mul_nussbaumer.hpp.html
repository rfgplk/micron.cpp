57 [[nodiscard, gnu::flatten]] inline constexpr __nuss_plan
58 __nuss_make_plan(usize an, usize bn) noexcept
59 {
60   if ( an == 0u || bn == 0u ) return { 0u, 0u, 0u, 0u, 0u, false };
61   const usize abits = __nuss_size_mul(an, limb_bits);
62   const usize bbits = __nuss_size_mul(bn, limb_bits);
63   if ( abits == __nuss_size_max || bbits == __nuss_size_max ) return { 0u, 0u, 0u, 0u, 0u, false };
64 
65   for ( usize k = 0u; k <= threshold::nussbaumer_max_log; ++k ) {
66     const usize q = limb_bits - 2u - k;
67     const usize ad = __nuss_ceil_div(abits, q);
68     const usize bd = __nuss_ceil_div(bbits, q);
69     const usize need = __nuss_size_add(ad, bd);
70     const usize n = usize{ 1 } << k;
71     if ( need != __nuss_size_max && need - 1u <= n ) return { q, k, n, ad, bd, true };
72   }
73   return { 0u, 0u, 0u, 0u, 0u, false };
74 }

206 [[nodiscard, gnu::always_inline]] inline constexpr dlimb_t
207 __nuss_coeff_mul_bits(__nuss_coeff a, __nuss_coeff b) noexcept
208 {
209   // The plan reserves every transform-growth bit, so leaf operands are signed one-limb values
210   // whose high limb is only sign extension.  A native signed widening multiply is therefore exact.
211 #if defined(__micron_arch_width_64)
212   const int128_t product = static_cast<int128_t>(static_cast<slimb_t>(a.lo)) * static_cast<int128_t>(static_cast<slimb_t>(b.lo));
213 #else
214   const i64 product = static_cast<i64>(static_cast<slimb_t>(a.lo)) * static_cast<i64>(static_cast<slimb_t>(b.lo));
215 #endif
216   return static_cast<dlimb_t>(product);
217 }

226 [[nodiscard, gnu::always_inline]] inline constexpr __nuss_coeff
227 __nuss_coeff_shr(__nuss_coeff a, usize shift) noexcept
228 {
229   if ( shift == 0u ) return a;
230   return { static_cast<limb_t>((a.lo >> shift) | (a.hi << (limb_bits - shift))), static_cast<limb_t>(a.hi >> shift) };

233 [[nodiscard, gnu::always_inline]] inline constexpr bool
234 __nuss_coeff_divexact_pow2(__nuss_coeff &a, usize shift) noexcept
235 {
236   if ( shift == 0u ) return true;
237   const limb_t mask = static_cast<limb_t>((limb_t{ 1 } << shift) - 1u);
238   const bool exact = (a.lo & mask) == 0u;
239   const limb_t sign = static_cast<limb_t>(a.hi >> (limb_bits - 1u));
240   const limb_t sign_mask = static_cast<limb_t>(0u - sign);
241   a.lo = static_cast<limb_t>((a.lo >> shift) | (a.hi << (limb_bits - shift)));
242   a.hi = static_cast<limb_t>((a.hi >> shift) | (sign_mask << (limb_bits - shift)));
243   return exact;
244 }

287 // rp = ap * Z^e in Z[Z]/(Z^r + 1), 0 <= e < 2r
288 inline constexpr void
289 __nuss_twiddle_copy(__nuss_coeffs rp, __nuss_const_coeffs ap, usize e) noexcept
290 {
291   const usize r = rp.n;
292   e &= 2u * r - 1u;
293   const bool outer_neg = e >= r;
294   if ( outer_neg ) e -= r;
295   if ( e == 0u ) {
296     __nuss_copy_cneg(rp, ap, outer_neg);
297     return;
298   }
299 
300   const usize straight = r - e;
301   __nuss_copy_cneg(__nuss_coeff_subspan(rp, e, straight), __nuss_coeff_subspan(ap, 0u, straight), outer_neg);
302   __nuss_copy_cneg(__nuss_coeff_subspan(rp, 0u, e), __nuss_coeff_subspan(ap, straight, e), !outer_neg);
303 }

305 // p *= Z^e in Z[Z]/(Z^r + 1), 0 <= e < 2r
306 inline constexpr void
307 __nuss_twiddle(__nuss_coeffs p, usize e) noexcept
308 {
309   const usize r = p.n;
310   e &= 2u * r - 1u;
311   const bool outer_neg = e >= r;
312   if ( outer_neg ) e -= r;
313 
314   if ( e == 0u ) {
315     if ( outer_neg )
316       for ( usize i = 0u; i < r; ++i ) __nuss_coeff_store(p, i, __nuss_coeff_neg(__nuss_coeff_load(p, i)));
317     return;
318   }
319 
320   const usize cycles = usize{ 1 } << limb_ctz(static_cast<limb_t>(e));
321   for ( usize start = 0u; start < cycles; ++start ) {
322     usize src = start;
323     __nuss_coeff carry = __nuss_coeff_load(p, src);
324     do {
325       usize dst = src + e;
326       const bool wrap = dst >= r;
327       if ( wrap ) dst -= r;
328       const __nuss_coeff next = __nuss_coeff_load(p, dst);
329       carry = __nuss_coeff_cneg(carry, wrap != outer_neg);
330       __nuss_coeff_store(p, dst, carry);
331       carry = next;
332       src = dst;
333     } while ( src != start );
334   }
335 }

337 inline constexpr void
338 __nuss_butterfly_to(__nuss_coeffs a, __nuss_const_coeffs b, __nuss_coeffs diff) noexcept
339 {

360 inline constexpr void
361 __nuss_forward_node(__nuss_coeffs a, usize base, usize half, usize stride, usize r, usize root_step, __nuss_coeffs temp) noexcept
362 {
363   for ( usize j = 0u; j < half; ++j ) {
364     __nuss_coeffs u = __nuss_coeff_subspan(a, (base + j) * r, r);
365     __nuss_coeffs v = __nuss_coeff_subspan(a, (base + j + half) * r, r);
366     const usize e = j * stride * root_step;
367     if ( e == 0u ) {
368       __nuss_butterfly(u, v);
369     } else {
370       __nuss_butterfly_to(u, __nuss_as_const(v), temp);
371       __nuss_twiddle_copy(v, __nuss_as_const(temp), e);
372     }
373   }
374 }

376 inline constexpr void
377 __nuss_forward_block(__nuss_coeffs a, usize base, usize len, usize stride, usize r, usize root_step, __nuss_coeffs temp) noexcept
378 {
379   usize stage_stride = stride;
380   for ( usize stage_len = len; stage_len > 1u; stage_len >>= 1u ) {
381     const usize half = stage_len >> 1u;
382     for ( usize block = base; block < base + len; block += stage_len )
383       __nuss_forward_node(a, block, half, stage_stride, r, root_step, temp);
384     stage_stride <<= 1u;
385   }
386 }

391 __nuss_forward_rec(__nuss_coeffs a, usize base, usize len, usize stride, usize r, usize root_step, __nuss_coeffs temp) noexcept

411 __nuss_inverse_node(__nuss_coeffs a, usize base, usize half, usize stride, usize r, usize root_step, __nuss_coeffs temp) noexcept
412 {
413   for ( usize j = 0u; j < half; ++j ) {

415     __nuss_coeffs v = __nuss_coeff_subspan(a, (base + j + half) * r, r);
416     const usize e = j * stride * root_step;
417     if ( e == 0u ) {

427 __nuss_inverse_block(__nuss_coeffs a, usize base, usize len, usize stride, usize r, usize root_step, __nuss_coeffs temp) noexcept
428 {
429   usize stage_stride = stride * (len >> 1u);
430   for ( usize stage_len = 2u; stage_len <= len; stage_len <<= 1u ) {
431     const usize half = stage_len >> 1u;
432     for ( usize block = base; block < base + len; block += stage_len )

494 template<> struct __nuss_leaf_accumulator<true> {
495   dlimb_t bits;
496 
497   constexpr void
498   add(__nuss_coeff a, __nuss_coeff b, bool twice = false) noexcept
499   {
500     dlimb_t p = __nuss_coeff_mul_bits(a, b);
501     if ( twice ) p += p;
502     bits += p;
503   }
504 
505   constexpr void
506   sub(__nuss_coeff a, __nuss_coeff b, bool twice = false) noexcept
507   {
508     dlimb_t p = __nuss_coeff_mul_bits(a, b);
509     if ( twice ) p += p;
510     bits -= p;
511   }
512 
513   [[nodiscard]] constexpr __nuss_coeff
514   combine(const __nuss_leaf_accumulator &other) const noexcept
515   {
516     const dlimb_t sum = bits + other.bits;
517     return { lo_half(sum), hi_half(sum) };
518   }
519 };

521 template<> struct __nuss_leaf_accumulator<false> {
522   __nuss_coeff value;
523 
524   constexpr void
525   add(__nuss_coeff a, __nuss_coeff b, bool twice = false) noexcept
526   {
527     __nuss_coeff p = __nuss_coeff_mul(a, b);
528     if ( twice ) p = __nuss_coeff_add(p, p);
529     value = __nuss_coeff_add(value, p);
530   }
531 
532   constexpr void
533   sub(__nuss_coeff a, __nuss_coeff b, bool twice = false) noexcept
534   {
535     __nuss_coeff p = __nuss_coeff_mul(a, b);
536     if ( twice ) p = __nuss_coeff_add(p, p);
537     value = __nuss_coeff_sub(value, p);
538   }
539 
540   [[nodiscard]] constexpr __nuss_coeff
541   combine(const __nuss_leaf_accumulator &other) const noexcept
542   {
543     return __nuss_coeff_add(value, other.value);
544   }

549 __nuss_mul_leaf_coefficient(__nuss_const_coeffs ap, __nuss_const_coeffs bp, usize n, usize d) noexcept
550 {
551   __nuss_leaf_accumulator<Wide> acc0{}, acc1{};
552   usize i = 0u;
553   for ( ; i + 1u <= d; i += 2u ) {
554     acc0.add(__nuss_coeff_load(ap, i), __nuss_coeff_load(bp, d - i));
555     acc1.add(__nuss_coeff_load(ap, i + 1u), __nuss_coeff_load(bp, d - i - 1u));
556   }
557   if ( i <= d ) acc0.add(__nuss_coeff_load(ap, i), __nuss_coeff_load(bp, d - i));
558 
559   i = d + 1u;
560   for ( ; i + 1u < n; i += 2u ) {
561     acc0.sub(__nuss_coeff_load(ap, i), __nuss_coeff_load(bp, d + n - i));
562     acc1.sub(__nuss_coeff_load(ap, i + 1u), __nuss_coeff_load(bp, d + n - i - 1u));
563   }
564   if ( i < n ) acc0.sub(__nuss_coeff_load(ap, i), __nuss_coeff_load(bp, d + n - i));
565   return acc0.combine(acc1);
566 }

604 template<bool Wide>
605 [[nodiscard, gnu::always_inline]] inline constexpr __nuss_coeff
606 __nuss_sqr_leaf_coefficient(__nuss_const_coeffs ap, usize n, usize d) noexcept
607 {
608   __nuss_leaf_accumulator<Wide> acc0{}, acc1{};
609   const usize pos_last = d >> 1u;
610   usize i = 0u;
611   for ( ; i + 1u <= pos_last; i += 2u ) {
612     acc0.add(__nuss_coeff_load(ap, i), __nuss_coeff_load(ap, d - i), i != d - i);
613     acc1.add(__nuss_coeff_load(ap, i + 1u), __nuss_coeff_load(ap, d - i - 1u), i + 1u != d - i - 1u);
614   }
615   if ( i <= pos_last ) acc0.add(__nuss_coeff_load(ap, i), __nuss_coeff_load(ap, d - i), i != d - i);
616 
617   const usize neg_last = (d + n) >> 1u;
618   i = d + 1u;
619   for ( ; i + 1u <= neg_last; i += 2u ) {
620     acc0.sub(__nuss_coeff_load(ap, i), __nuss_coeff_load(ap, d + n - i), i != d + n - i);
621     acc1.sub(__nuss_coeff_load(ap, i + 1u), __nuss_coeff_load(ap, d + n - i - 1u), i + 1u != d + n - i - 1u);
622   }
623   if ( i <= neg_last ) acc0.sub(__nuss_coeff_load(ap, i), __nuss_coeff_load(ap, d + n - i), i != d + n - i);
624   return acc0.combine(acc1);
625 }

666 inline constexpr void
667 __nuss_transpose(__nuss_coeffs rp, __nuss_const_coeffs ap, usize rows, usize cols) noexcept
668 {
669 #if defined(__micron_arbint_kern_coefficients)
670   if ( !__builtin_is_constant_evaluated() ) {
671     constexpr usize lanes = __kern::vec_lanes;
672     const usize full_rows = rows - rows % lanes;
673     const usize full_cols = cols - cols % lanes;
674     for ( usize col = 0u; col < full_cols; col += lanes ) {
675       for ( usize row = 0u; row < full_rows; row += lanes ) {
676         const usize out = col * rows + row;
677         const usize in = row * cols + col;
678         __kern::coeff_transpose_square(rp.lo + out, rp.hi + out, rows, ap.lo + in, ap.hi + in, cols);
679       }
680       for ( usize row = full_rows; row < rows; ++row ) {
681         for ( usize i = 0u; i < lanes; ++i ) {
682           const usize out = (col + i) * rows + row;
683           const usize in = row * cols + col + i;
684           rp.lo[out] = ap.lo[in];
685           rp.hi[out] = ap.hi[in];
686         }
687       }
688     }
689     for ( usize col = full_cols; col < cols; ++col ) {
690       for ( usize row = 0u; row < rows; ++row ) {
691         const usize out = col * rows + row;
692         const usize in = row * cols + col;
693         rp.lo[out] = ap.lo[in];
694         rp.hi[out] = ap.hi[in];
695       }
696     }
697     return;
698   }
699 #endif
700   constexpr usize tile = 8u;
701   for ( usize row0 = 0u; row0 < rows; row0 += tile ) {
702     const usize row_n = rows - row0 < tile ? rows - row0 : tile;
703     for ( usize col0 = 0u; col0 < cols; col0 += tile ) {
704       const usize col_n = cols - col0 < tile ? cols - col0 : tile;
705       for ( usize col = 0u; col < col_n; ++col ) {
706         const usize out = (col0 + col) * rows + row0;
707         for ( usize row = 0u; row < row_n; ++row ) {
708           const usize in = (row0 + row) * cols + col0 + col;
709           rp.lo[out + row] = ap.lo[in];
710           rp.hi[out + row] = ap.hi[in];
711         }
712       }
713     }
714   }
715 }

724 [[nodiscard]] inline constexpr bool
725 __nuss_divexact_coeffs(__nuss_coeffs a, usize shift) noexcept
726 {
727 #if defined(__micron_arbint_kern_coefficients)
728   if ( !__builtin_is_constant_evaluated() ) return __kern::coeff_divexact_pow2(a.lo, a.hi, a.n, shift);
729 #endif
730   bool exact = true;
731   for ( usize i = 0u; i < a.n; ++i ) {
732     __nuss_coeff v = __nuss_coeff_load(a, i);
733     exact = __nuss_coeff_divexact_pow2(v, shift) && exact;
734     __nuss_coeff_store(a, i, v);
735   }
736   return exact;

739 inline constexpr bool
740 __nuss_fold(__nuss_coeffs rp, __nuss_coeffs a, usize s, usize r) noexcept
741 {
742   const usize outer_n = 2u * s;
743   const usize div_shift = __nuss_log2(outer_n);
744   const bool exact = __nuss_divexact_coeffs(a, div_shift);
745 
746   for ( usize i = 0u; i < s; ++i ) {
747     __nuss_coeffs lo = __nuss_coeff_subspan(a, i * r, r);
748     const __nuss_const_coeffs hi = __nuss_coeff_subspan(__nuss_as_const(a), (i + s) * r, r);
749     __nuss_coeff_store(lo, 0u, __nuss_coeff_sub(__nuss_coeff_load(lo, 0u), __nuss_coeff_load(hi, r - 1u)));
750     if ( r > 1u )
751       __nuss_add_coeffs(__nuss_coeff_subspan(lo, 1u, r - 1u), __nuss_coeff_subspan(__nuss_as_const(lo), 1u, r - 1u),
752                         __nuss_coeff_subspan(hi, 0u, r - 1u));
753   }
754   __nuss_transpose(rp, __nuss_coeff_subspan(__nuss_as_const(a), 0u, s * r), s, r);
755   return exact;
756 }

758 inline constexpr bool
759 __nuss_mul_loaded(__nuss_coeffs rp, __nuss_coeffs a, __nuss_coeffs b, usize s, usize r, limb_t *scratch) noexcept
760 {
761   const usize outer_n = 2u * s;
762   __nuss_coeffs temp = __nuss_coeff_span(scratch, r);
763   __nuss_forward(a, outer_n, r, temp);
764   __nuss_forward(b, outer_n, r, temp);
765 
766   bool exact = true;
767   for ( usize i = 0u; i < outer_n; ++i ) {
768     __nuss_coeffs ai = __nuss_coeff_subspan(a, i * r, r);
769     const __nuss_const_coeffs bi = __nuss_coeff_subspan(__nuss_as_const(b), i * r, r);
770     exact = __nuss_mul_negacyclic(ai, __nuss_as_const(ai), bi, scratch) && exact;
771   }
772 
773   __nuss_inverse(a, outer_n, r, temp);
774   return __nuss_fold(rp, a, s, r) && exact;
775 }

778 __nuss_sqr_loaded(__nuss_coeffs rp, __nuss_coeffs a, usize s, usize r, limb_t *scratch) noexcept
779 {

781   __nuss_coeffs temp = __nuss_coeff_span(scratch, r);
782   __nuss_forward(a, outer_n, r, temp);

784   bool exact = true;
785   for ( usize i = 0u; i < outer_n; ++i ) {
786     __nuss_coeffs ai = __nuss_coeff_subspan(a, i * r, r);
787     exact = __nuss_sqr_negacyclic(ai, __nuss_as_const(ai), scratch) && exact;
788   }
789 
790   __nuss_inverse(a, outer_n, r, temp);
791   return __nuss_fold(rp, a, s, r) && exact;
792 }

796 {
797   const usize n = rp.n;
798   if ( n <= __nuss_leaf_n ) {
799     __nuss_mul_leaf(rp, ap, bp, scratch);
800     return true;
801   }
802 
803   const usize k = __nuss_log2(n);
804   const usize r = __nuss_inner_n(k);
805   const usize s = n / r;
806   __nuss_coeffs a = __nuss_coeff_span(scratch, 2u * n);
807   __nuss_coeffs b = __nuss_coeff_span(scratch + 4u * n, 2u * n);
808   limb_t *const rec = scratch + 8u * n;
809   __nuss_reshape(a, ap, s, r);
810   __nuss_reshape(b, bp, s, r);
811   return __nuss_mul_loaded(rp, a, b, s, r, rec);
812 }

847 [[nodiscard, gnu::always_inline]] inline constexpr limb_t
848 __nuss_low_mask(usize bits) noexcept
849 {
850   return bits >= limb_bits ? limb_max : static_cast<limb_t>((limb_t{ 1 } << bits) - 1u);
851 }
852 
853 [[nodiscard, gnu::always_inline]] inline constexpr limb_t
854 __nuss_digit(const limb_t *ap, usize an, usize digit, usize q) noexcept
855 {
856   const usize bit = digit * q;
857   const usize word = bit / limb_bits;
858   const usize shift = bit % limb_bits;
859   if ( word >= an ) return 0u;
860   limb_t v = static_cast<limb_t>(ap[word] >> shift);
861   if ( shift != 0u && q > limb_bits - shift && word + 1u < an ) v = static_cast<limb_t>(v | (ap[word + 1u] << (limb_bits - shift)));
862   return static_cast<limb_t>(v & __nuss_low_mask(q));
863 }

865 struct __nuss_packer {
866   limb_t *rp;
867   usize rn;
868   usize out;
869   usize used;
870   limb_t acc;
871   bool exact;
872 
873   constexpr void
874   push(limb_t digit, usize q) noexcept
875   {
876     const usize room = limb_bits - used;
877     acc = static_cast<limb_t>(acc | static_cast<limb_t>(digit << used));
878     if ( q < room ) {
879       used += q;
880       return;
881     }
882 
883     if ( out < rn )
884       rp[out] = acc;
885     else if ( acc != 0u )
886       exact = false;
887     ++out;
888     if ( q == room ) {
889       used = 0u;
890       acc = 0u;
891     } else {
892       used = q - room;
893       acc = static_cast<limb_t>(digit >> room);
894     }
895   }
896 
897   constexpr void
898   finish() noexcept
899   {
900     if ( used != 0u ) {
901       if ( out < rn )
902         rp[out] = acc;
903       else if ( acc != 0u )
904         exact = false;
905       ++out;
906     }
907     while ( out < rn ) rp[out++] = 0u;
908   }
909 };

911 inline constexpr bool
912 __nuss_reassemble(limb_t *rp, usize rn, __nuss_const_coeffs cp, usize q) noexcept
913 {
914   __nuss_coeff carry{};
915   __nuss_packer pack{ rp, rn, 0u, 0u, 0u, true };
916   for ( usize i = 0u; i < cp.n; ++i ) {
917     const __nuss_coeff v = __nuss_coeff_add(__nuss_coeff_load(cp, i), carry);
918     if ( __nuss_coeff_negative(v) ) return false;
919     pack.push(static_cast<limb_t>(v.lo & __nuss_low_mask(q)), q);
920     carry = __nuss_coeff_shr(v, q);
921   }
922   while ( !__nuss_coeff_is_zero(carry) ) {
923     if ( __nuss_coeff_negative(carry) ) return false;
924     pack.push(static_cast<limb_t>(carry.lo & __nuss_low_mask(q)), q);
925     carry = __nuss_coeff_shr(carry, q);
926   }
927   pack.finish();
928   return pack.exact;
929 }

931 inline constexpr bool
932 __nuss_mul_product(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn, const __nuss_plan &plan, limb_t *scratch) noexcept
933 {
934   __nuss_coeffs cp = __nuss_coeff_span(scratch, plan.n);
935   limb_t *const work = scratch + 2u * plan.n;
936   bool exact = true;
937 
938   if ( plan.n <= __nuss_leaf_n ) {
939     __nuss_coeffs accum = __nuss_coeff_span(work, plan.n);
940     __nuss_zero_coeffs(cp);
941     __nuss_zero_coeffs(accum);
942     for ( usize i = 0u; i < plan.a_digits; ++i ) {
943       const __nuss_coeff a{ __nuss_digit(ap, an, i, plan.digit_bits), 0u };
944       __nuss_coeff_store(cp, i, a);
945       for ( usize j = 0u; j < plan.b_digits; ++j ) {
946         const __nuss_coeff b{ __nuss_digit(bp, bn, j, plan.digit_bits), 0u };
947         const usize ij = i + j;
948         const __nuss_coeff old = __nuss_coeff_load(accum, ij);
949         __nuss_coeff_store(accum, ij, __nuss_coeff_add(old, __nuss_coeff_mul(a, b)));
950       }
951     }
952     __nuss_copy_coeffs(cp, __nuss_as_const(accum));
953   } else {
954     const usize r = __nuss_inner_n(plan.log_n);
955     const usize s = plan.n / r;
956     __nuss_coeffs a = __nuss_coeff_span(work, 2u * plan.n);
957     __nuss_coeffs b = __nuss_coeff_span(work + 4u * plan.n, 2u * plan.n);
958     limb_t *const rec = work + 8u * plan.n;
959     __nuss_zero_coeffs(a);
960     __nuss_zero_coeffs(b);
961     for ( usize d = 0u; d < plan.a_digits; ++d )
962       __nuss_coeff_store(a, (d % s) * r + d / s, { __nuss_digit(ap, an, d, plan.digit_bits), 0u });
963     for ( usize d = 0u; d < plan.b_digits; ++d )
964       __nuss_coeff_store(b, (d % s) * r + d / s, { __nuss_digit(bp, bn, d, plan.digit_bits), 0u });
965     exact = __nuss_mul_loaded(cp, a, b, s, r, rec);
966   }
967   return __nuss_reassemble(rp, an + bn, __nuss_as_const(cp), plan.digit_bits) && exact;
968 }

970 inline constexpr bool
971 __nuss_sqr_product(limb_t *rp, const limb_t *ap, usize an, const __nuss_plan &plan, limb_t *scratch) noexcept
972 {
973   __nuss_coeffs cp = __nuss_coeff_span(scratch, plan.n);
974   limb_t *const work = scratch + 2u * plan.n;
975   bool exact = true;
976 
977   if ( plan.n <= __nuss_leaf_n ) {
978     __nuss_zero_coeffs(cp);
979     for ( usize i = 0u; i < plan.a_digits; ++i ) __nuss_coeff_store(cp, i, { __nuss_digit(ap, an, i, plan.digit_bits), 0u });
980     __nuss_sqr_leaf(cp, __nuss_as_const(cp), work);
981   } else {
982     const usize r = __nuss_inner_n(plan.log_n);
983     const usize s = plan.n / r;
984     __nuss_coeffs a = __nuss_coeff_span(work, 2u * plan.n);
985     limb_t *const rec = work + 4u * plan.n;
986     __nuss_zero_coeffs(a);
987     for ( usize d = 0u; d < plan.a_digits; ++d )
988       __nuss_coeff_store(a, (d % s) * r + d / s, { __nuss_digit(ap, an, d, plan.digit_bits), 0u });
989     exact = __nuss_sqr_loaded(cp, a, s, r, rec);
990   }
991   return __nuss_reassemble(rp, 2u * an, __nuss_as_const(cp), plan.digit_bits) && exact;
992 }

1006       lo = mid;

1011 }

1013 [[nodiscard, gnu::flatten]] inline constexpr usize
1014 __nuss_max_partner(usize fixed, usize wanted) noexcept
1015 {
1016   usize lo = 0u;
1017   usize hi = wanted + 1u;
1018   if ( hi == 0u ) hi = wanted;
1019   while ( lo + 1u < hi ) {
1020     const usize mid = lo + (hi - lo) / 2u;
1021     if ( __nuss_make_plan(mid, fixed).valid )
1022       lo = mid;
1023     else
1024       hi = mid;
1025   }
1026   return lo;
1027 }

1029 struct __nuss_mul_shape {
1030   usize an;
1031   usize bn;
1032   __nuss_plan plan;
1033   bool direct;
1034 };
1035 
1036 [[nodiscard, gnu::flatten]] inline constexpr __nuss_mul_shape
1037 __nuss_pick_mul_shape(usize an, usize bn) noexcept
1038 {
1039   const bool long_axis = an / bn >= 8u;
1040   const __nuss_plan whole = __nuss_make_plan(an, bn);
1041   if ( !long_axis && whole.valid ) return { an, bn, whole, true };
1042 
1043   if ( long_axis ) {
1044     const usize three_bn = __nuss_size_mul(3u, bn);
1045     const usize block = an < three_bn ? an : three_bn;
1046     const __nuss_plan p = __nuss_make_plan(block, bn);
1047     if ( p.valid ) return { block, bn, p, false };
1048   }
1049 
1050   const usize balanced = __nuss_max_balanced_limbs();
1051   const usize bblock = bn < balanced ? bn : balanced;
1052   usize wanted = an;
1053   if ( long_axis ) {
1054     const usize three_b = __nuss_size_mul(3u, bblock);
1055     wanted = an < three_b ? an : three_b;
1056   }
1057   const usize ablock = __nuss_max_partner(bblock, wanted);
1058   return { ablock, bblock, __nuss_make_plan(ablock, bblock), false };
1059 }

1090 inline constexpr void
1091 __nuss_add_partial(limb_t *rp, usize rn, usize off, const limb_t *part, usize pn) noexcept
1092 {
1093   limb_t cy = add_n(rp + off, rp + off, part, pn);
1094   usize i = off + pn;
1095   while ( cy != 0u && i < rn ) {
1096     cy = addc(rp[i], 0u, cy, rp[i]);
1097     ++i;
1098   }

1101 inline constexpr void
1102 mul_nussbaumer(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn,
1103                limb_t *scratch) noexcept
1104 {
1105   const __nuss_mul_shape shape = __nuss_pick_mul_shape(an, bn);
1106   if ( shape.direct ) {
1107     (void)__nuss_mul_product(rp, ap, an, bp, bn, shape.plan, scratch);
1108     return;
1109   }
1110 
1111   const usize rn = an + bn;
1112   zero(rp, rn);
1113   for ( usize bj = 0u; bj < bn; bj += shape.bn ) {
1114     const usize btake = bn - bj < shape.bn ? bn - bj : shape.bn;
1115     for ( usize ai = 0u; ai < an; ai += shape.an ) {
1116       const usize atake = an - ai < shape.an ? an - ai : shape.an;
1117       const __nuss_plan p = __nuss_make_plan(atake, btake);
1118       (void)__nuss_mul_product(scratch, ap + ai, atake, bp + bj, btake, p, scratch);
1119       __nuss_add_partial(rp, rn, ai + bj, scratch, atake + btake);
1120     }
1121   }
1122 }

1124 inline constexpr void
1125 sqr_nussbaumer(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch) noexcept
1126 {
1127   const __nuss_plan whole = __nuss_make_plan(n, n);
1128   if ( whole.valid ) {
1129     (void)__nuss_sqr_product(rp, ap, n, whole, scratch);
1130     return;
1131   }
1132 
1133   const usize block = __nuss_max_balanced_limbs();
1134   const usize rn = 2u * n;
1135   zero(rp, rn);
1136   for ( usize i = 0u; i < n; i += block ) {
1137     const usize itake = n - i < block ? n - i : block;
1138     const __nuss_plan sp = __nuss_make_plan(itake, itake);
1139     (void)__nuss_sqr_product(scratch, ap + i, itake, sp, scratch);
1140     __nuss_add_partial(rp, rn, 2u * i, scratch, 2u * itake);
1141 
1142     for ( usize j = 0u; j < i; j += block ) {
1143       const usize jtake = n - j < block ? n - j : block;
1144       const __nuss_plan mp = __nuss_make_plan(itake, jtake);
1145       (void)__nuss_mul_product(scratch, ap + i, itake, ap + j, jtake, mp, scratch);
1146       __nuss_add_partial(rp, rn, i + j, scratch, itake + jtake);
1147       __nuss_add_partial(rp, rn, i + j, scratch, itake + jtake);
1148     }
1149   }
1150 }
