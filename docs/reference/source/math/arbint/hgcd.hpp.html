16 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
17 // half-gcd
18 //
19 //  ..T(n) = 2*T(n/2) + Theta(M(n))
20 //  ..T(n) = Theta(M(n) log n)
21 //
22 //
23 // (via Moller's formulation [Niels Moller, "On Schoenhage's algorithm and subquadratic integer
24 // gcd computation", Math. Comp. 77, 2008,)
25 // matrix convention
26 //     ( a )   ( u00  u01 ) ( a' )              a' = u11*a - u01*b       b' = u00*b - u10*a
27 //     ( b ) = ( u10  u11 ) ( b' )

36 struct hgcd_mat {

42 [[nodiscard, gnu::always_inline]] inline constexpr usize
43 hgcd_mat_itch(usize n) noexcept
44 {
45   return 4u * (((n + 1u) / 2u) + 2u);
46 }

48 [[gnu::flatten]] inline constexpr void
49 hgcd_mat_init(hgcd_mat &M, usize n, limb_t *space) noexcept
50 {
51   const usize s = ((n + 1u) / 2u) + 2u;
52   zero(space, 4u * s);
53   M.alloc = s;
54   M.n = 1;
55   M.p[0][0] = space;
56   M.p[0][1] = space + s;
57   M.p[1][0] = space + 2u * s;
58   M.p[1][1] = space + 3u * s;
59   M.p[0][0][0] = 1;
60   M.p[1][1][0] = 1;
61 }

69 // M <- M * m1, for a single-limb matrix m1
70 [[gnu::flatten]] inline constexpr void
71 hgcd_mat_mul_1(hgcd_mat &M, const mat1 &m1, limb_t *tp) noexcept
72 {
73   const mat1 t{ m1.u00, m1.u10, m1.u01, m1.u11 };
74 
75   const usize w = M.n + 2u;
76   limb_t *const r0 = tp;
77   limb_t *const r1 = tp + w;
78 
79   const usize n0 = mat1_mul_vector(t, r0, r1, M.p[0][0], M.p[0][1], M.n);
80   copyi(M.p[0][0], r0, w);
81   copyi(M.p[0][1], r1, w);
82 
83   const usize n1 = mat1_mul_vector(t, r0, r1, M.p[1][0], M.p[1][1], M.n);
84   copyi(M.p[1][0], r0, w);
85   copyi(M.p[1][1], r1, w);
86 
87   M.n = n0 > n1 ? n0 : n1;
88 }

90 [[nodiscard, gnu::always_inline]] inline constexpr usize
91 matrix22_mul_itch(usize rn, usize mn) noexcept
92 {
93   const usize w = rn + mn + 1u;
94   const usize a = rn > mn ? rn : mn;
95   const usize b = rn > mn ? mn : rn;
96   return 4u * w + (rn + mn) + __mul_itch_bound(a, b);
97 }

99 [[gnu::always_inline]] inline constexpr void
100 __mul_ord(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn, limb_t *sc) noexcept
101 {
102   if ( an >= bn )
103     mul(rp, ap, an, bp, bn, sc);
104   else
105     mul(rp, bp, bn, ap, an, sc);
106 }

108 // never flatten this (or anything that invokes mul)
109 inline constexpr void
110 matrix22_mul(hgcd_mat &M, const hgcd_mat &M1, limb_t *tp) noexcept
111 {
112   const usize rn = M.n;
113   const usize mn = M1.n;
114   const usize pn = rn + mn;
115   const usize w = pn + 1u;
116 
117   limb_t *const o00 = tp;
118   limb_t *const o01 = o00 + w;
119   limb_t *const o10 = o01 + w;
120   limb_t *const o11 = o10 + w;
121   limb_t *const prod = o11 + w;
122   limb_t *const work = prod + pn;
123 
124   limb_t *const out[2][2] = { { o00, o01 }, { o10, o11 } };
125 
126   for ( usize i = 0; i < 2u; ++i ) {
127     for ( usize j = 0; j < 2u; ++j ) {
128       limb_t *const o = out[i][j];
129       __mul_ord(o, M.p[i][0], rn, M1.p[0][j], mn, work);
130       __mul_ord(prod, M.p[i][1], rn, M1.p[1][j], mn, work);
131       o[pn] = add_n(o, o, prod, pn);
132     }
133   }
134 
135   usize nn = 1;
136   for ( usize i = 0; i < 2u; ++i ) {
137     for ( usize j = 0; j < 2u; ++j ) {
138       const usize k = normalize(out[i][j], w);
139       if ( k > nn ) nn = k;
140     }
141   }
142 
143   for ( usize i = 0; i < 2u; ++i )
144     for ( usize j = 0; j < 2u; ++j ) copyi(M.p[i][j], out[i][j], w);
145 
146   M.n = nn;
147 }

149 [[nodiscard, gnu::always_inline]] inline constexpr usize
150 hgcd_mat_adjust_itch(usize p, usize mn) noexcept
151 {
152   const usize a = p > mn ? p : mn;
153   const usize b = p > mn ? mn : p;
154   return 3u * (p + mn) + __mul_itch_bound(a, b);
155 }

157 [[nodiscard, gnu::always_inline]] inline constexpr usize
158 __adjust_width(usize k, usize mn, usize h) noexcept
159 {
160   usize w = k > mn + 1u ? k : mn + 1u;
161   if ( w > h ) w = h;
162   return w;
163 }

165 [[nodiscard]] inline constexpr usize
166 hgcd_mat_adjust(const hgcd_mat &M, usize n, limb_t *ap, limb_t *bp, usize p, limb_t *tp) noexcept
167 {
168   const usize mn = M.n;
169   const usize w = p + mn;
170 
171   limb_t *const t0 = tp;          // u11 * a_lo
172   limb_t *const t1 = t0 + w;      // u10 * a_lo
173   limb_t *const t2 = t1 + w;      // u01 * b_lo, then u00 * b_lo
174   limb_t *const work = t2 + w;
175 
176   __mul_ord(t0, ap, p, M.p[1][1], mn, work);
177   __mul_ord(t1, ap, p, M.p[1][0], mn, work);
178   __mul_ord(t2, bp, p, M.p[0][1], mn, work);
179 
180   // a <- a_hi*B^p + t0 - t2
181   copyi(ap, t0, p);
182   limb_t ah = add(ap + p, ap + p, n - p, t0 + p, mn);
183   ah = static_cast<limb_t>(ah - sub(ap, ap, n, t2, w));
184 
185   __mul_ord(t2, bp, p, M.p[0][0], mn, work);
186 
187   // b <- b_hi*B^p + t2 - t1
188   copyi(bp, t2, p);
189   limb_t bh = add(bp + p, bp + p, n - p, t2 + p, mn);
190   bh = static_cast<limb_t>(bh - sub(bp, bp, n, t1, w));
191 
192   (void)ah;
193   (void)bh;
194 
195   const usize an = normalize(ap, n);
196   const usize bn = normalize(bp, n);
197   return an > bn ? an : bn;
198 }

200 [[nodiscard, gnu::always_inline]] inline constexpr usize
201 hgcd_mat_update_q_itch(usize mn, usize qn) noexcept
202 {
203   const usize a = mn > qn ? mn : qn;
204   const usize b = mn > qn ? qn : mn;
205   return (mn + qn + 1u) + __mul_itch_bound(a, b);
206 }

208 // requires M.n + qn + 1 <= M.alloc
209 inline constexpr void
210 hgcd_mat_update_q(hgcd_mat &M, const limb_t *qp, usize qn, limb_t *tp) noexcept
211 {
212   const usize mn = M.n;
213   const usize pn = mn + qn;
214   const usize w = pn + 1u;
215 
216   limb_t *const t = tp;
217   limb_t *const work = t + w;
218 
219   usize nn = 1;
220   for ( usize i = 0; i < 2u; ++i ) {
221     __mul_ord(t, M.p[i][0], mn, qp, qn, work);
222     t[pn] = add(t, t, pn, M.p[i][1], mn);
223 
224     copyi(M.p[i][1], M.p[i][0], mn);
225     copyi(M.p[i][0], t, w);
226 
227     const usize k = normalize(t, w);
228     if ( k > nn ) nn = k;
229   }
230   M.n = nn;
231 }

233 [[nodiscard, gnu::always_inline]] inline constexpr bool
234 __mat1_make_even(mat1 &m) noexcept
235 {
236   if ( m.u00 > m.u01 ) return true;
237 
238   const limb_t q = static_cast<limb_t>(m.u01 / m.u00);
239   m.u01 = static_cast<limb_t>(m.u01 - q * m.u00);
240   m.u11 = static_cast<limb_t>(m.u11 - q * m.u10);
241 
242   return !(m.u00 == 1u && m.u01 == 0u && m.u10 == 0u && m.u11 == 1u);
243 }

266 template<class Hook>
267 [[nodiscard]] inline constexpr usize
268 __subdiv_step(limb_t *ap, limb_t *bp, usize n, Hook &hook, limb_t *tp) noexcept
269 {
270   const usize an = normalize(ap, n);
271   const usize bn = normalize(bp, n);
272   if ( bn == 0 || an < bn ) return 0;
273 
274   limb_t *const qp = tp;                       // an - bn + 1
275   limb_t *const rp = qp + (an - bn + 1u);      // bn
276   limb_t *const work = rp + bn;
277 
278   divrem(qp, rp, ap, an, bp, bn, work);
279   const usize qn = normalize(qp, an - bn + 1u);
280   const usize rn = normalize(rp, bn);
281 
282   // nothing has moved yet, so a refusal costs the division and no correctness
283   if ( !hook.quotient(qp, qn) ) return 0;
284 
285   copyi(ap, bp, bn);
286   zero(ap + bn, n - bn);
287   copyi(bp, rp, rn);
288   zero(bp + rn, n - rn);
289   return bn;
290 }

292 [[nodiscard, gnu::always_inline]] inline constexpr usize
293 hgcd_subdiv_pair_itch(usize n) noexcept
294 {
295   const usize d = divrem_cap_itch(n, n);
296   const usize u = hgcd_mat_update_q_itch_max(n);
297   return (n + 1u) + 3u * n + 2u + (d > u ? d : u);
298 }

300 [[nodiscard]] inline constexpr usize
301 hgcd_subdiv_pair(limb_t *ap, limb_t *bp, usize n, hgcd_mat &M, limb_t *tp) noexcept
302 {
303   const usize an = normalize(ap, n);
304   const usize bn = normalize(bp, n);
305   if ( bn == 0 || an < bn ) return 0;
306 
307   limb_t *const q1 = tp;                       // an - bn + 1
308   limb_t *const r1 = q1 + (an - bn + 1u);      // bn
309   limb_t *const q2 = r1 + bn;                  // bn + 1
310   limb_t *const r2 = q2 + bn + 1u;             // bn
311   limb_t *const work = r2 + bn;
312 
313   divrem(q1, r1, ap, an, bp, bn, work);
314   const usize q1n = normalize(q1, an - bn + 1u);
315   const usize r1n = normalize(r1, bn);
316   if ( r1n == 0 ) return 0;
317 
318   divrem(q2, r2, bp, bn, r1, r1n, work);
319   const usize q2n = normalize(q2, bn - r1n + 1u);
320   const usize r2n = normalize(r2, r1n);
321 
322   if ( M.n + q1n + q2n + 2u > M.alloc ) return 0;
323 
324   hgcd_mat_update_q(M, q1, q1n, work);
325   hgcd_mat_update_q(M, q2, q2n, work);
326 
327   copyi(ap, r1, r1n);
328   zero(ap + r1n, n - r1n);
329   copyi(bp, r2, r2n);
330   zero(bp + r2n, n - r2n);
331   return r1n;
332 }

334 [[nodiscard, gnu::always_inline]] inline constexpr usize
335 hgcd_step_itch(usize n) noexcept
336 {
337   const usize win = 2u * n + hgcd_mat_mul_1_itch((n + 1u) / 2u + 2u);
338   const usize div = hgcd_subdiv_pair_itch(n);
339   return win > div ? win : div;
340 }

342 [[nodiscard]] inline constexpr usize
343 hgcd_step(usize n, limb_t *ap, limb_t *bp, usize s, hgcd_mat &M, limb_t *tp) noexcept
344 {
345   const limb_t top = static_cast<limb_t>(ap[n - 1u] | bp[n - 1u]);
346   if ( top != 0 && M.n + 2u <= M.alloc && cmp_var(ap, n, bp, n) > 0 ) {
347     const usize sh = limb_clz(top);
348     limb_t ah = 0, al = 0, bh = 0, bl = 0;
349     __window2(ah, al, ap, n, sh);
350     __window2(bh, bl, bp, n, sh);
351 
352     mat1 m1{};
353     if ( hgcd2_sel(m1, ah, al, bh, bl) && __mat1_make_even(m1) ) {
354       limb_t *const t0 = tp;
355       limb_t *const t1 = tp + n;
356       usize n2 = 0;
357       if ( mat1_mul_inverse_vector(m1, t0, t1, ap, bp, n, n2) ) {
358         if ( n2 != 0 && cmp_var(t0, n2, t1, n2) > 0 ) {
359           const usize an = normalize(t0, n2);
360           if ( an > s ) {
361             copyi(ap, t0, n);
362             copyi(bp, t1, n);
363             hgcd_mat_mul_1(M, m1, tp + 2u * n);
364             return an;
365           }
366         }
367       }
368     }
369   }
370 
371   return hgcd_subdiv_pair(ap, bp, n, M, tp);
372 }

374 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
375 // recursion
376 //
377 // reduce (a; b) of n limbs until the pair fits about half that, accumulating the matrix
378 // Moller's shape; reduce the top half by recursion, take a few steps to trim, reduce again, and finish by iterating
379 //
380 //   pre:   n >= 2, a > b, ap[n-1] | bp[n-1] != 0, M the identity, M.alloc >= (n+1)/2 + 2
381 //   post:  0   -- no progress, and ap, bp and M are untouched
382 //          nn  -- the reduced pair is in (ap; bp) at nn limbs, det M == 1, and
383 //                 M * (a'; b') == (a; b) EXACTLY against the pair that came in
384 
385 [[nodiscard]] inline constexpr usize
386 hgcd_itch(usize n) noexcept
387 {
388   const usize s = n / 2u + 1u;
389   if ( n <= s ) return 0;
390 
391   const usize step = hgcd_step_itch(n);
392   if ( n < threshold::gcd_hgcd ) return step;
393 
394   const usize p = n / 2u;
395   const usize h = n - p;                    // first sub-problem
396   const usize hn = (h + 1u) / 2u + 2u;      // widest matrix it may hand back
397 
398   const usize sub = h > p ? h : p;
399   const usize rec = hgcd_itch(sub);
400 
401   usize m = rec;
402   const usize adj1 = hgcd_mat_adjust_itch(p, hn);
403   if ( adj1 > m ) m = adj1;
404   if ( step > m ) m = step;
405 
406   const usize pn = (p + 1u) / 2u + 2u;
407   usize inner = rec;
408   const usize adj3 = hgcd_mat_adjust_itch(p, pn);
409   if ( adj3 > inner ) inner = adj3;
410   const usize mm = matrix22_mul_itch((n + 1u) / 2u + 2u, pn);
411   if ( mm > inner ) inner = mm;
412 
413   const usize blk = hgcd_mat_itch(p) + inner;
414   return blk > m ? blk : m;
415 }

417 [[nodiscard]] inline constexpr usize
418 hgcd(limb_t *ap, limb_t *bp, usize n, hgcd_mat &M, limb_t *tp) noexcept
419 {
420   const usize s = n / 2u + 1u;
421   if ( n <= s ) return 0;
422 
423   usize nn = n;
424   bool success = false;
425 
426   if ( n >= threshold::gcd_hgcd ) {
427     const usize p = n / 2u;
428     {
429       const usize h = n - p;
430       const usize save = M.alloc;
431 
432       usize want = (h + 1u) / 2u + 2u;
433       if ( h >= 2u && want > h - 1u ) want = h - 1u;
434       M.alloc = want < save ? want : save;
435 
436       const usize k = hgcd(ap + p, bp + p, h, M, tp);
437       M.alloc = save;
438       if ( k > 0 ) {
439         nn = hgcd_mat_adjust(M, p + __adjust_width(k, M.n, h), ap, bp, p, tp);
440         success = true;
441       }
442     }
443 
444     const usize n2 = (3u * n) / 4u + 1u;
445     while ( nn > n2 ) {
446       if ( normalize(bp, nn) == 0 ) return success ? nn : 0;
447       const usize k = hgcd_step(nn, ap, bp, s, M, tp);
448       if ( k == 0 ) return success ? nn : 0;
449       nn = k;
450       success = true;
451     }
452 
453     // reduce again, this time folding the result in with a matrix product
454     if ( nn > s + 2u ) {
455       const usize p2 = 2u * s - nn + 1u;
456       const usize m = nn - p2;
457 
458       // what M can still absorb
459       usize room = M.alloc > M.n + 1u ? M.alloc - M.n - 1u : 0u;
460       if ( m >= 2u && room > m - 1u ) room = m - 1u;
461       if ( m >= 2u && room >= 3u ) {
462         hgcd_mat M1{};
463         hgcd_mat_init(M1, m, tp);
464         if ( M1.alloc > room ) M1.alloc = room;
465         limb_t *const w = tp + hgcd_mat_itch(m);
466 
467         const usize k = hgcd(ap + p2, bp + p2, m, M1, w);
468         if ( k > 0 ) {
469           nn = hgcd_mat_adjust(M1, p2 + __adjust_width(k, M1.n, m), ap, bp, p2, w);
470           matrix22_mul(M, M1, w);
471           success = true;
472         }
473       }
474     }
475   }
476 
477   for ( ;; ) {
478     if ( nn <= s ) break;
479     if ( normalize(bp, nn) == 0 ) break;
480     const usize k = hgcd_step(nn, ap, bp, s, M, tp);
481     if ( k == 0 ) break;
482     nn = k;
483     success = true;
484   }
485 
486   return success ? nn : 0;
487 }
