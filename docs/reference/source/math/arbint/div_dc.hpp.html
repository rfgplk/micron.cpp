32 // 2n limbs by n limbs, n quotient limbs
33 inline constexpr limb_t
34 dc_div_qr_n(limb_t *qp, limb_t *np, const limb_t *dp, usize n, limb_t dinv, limb_t *tp, limb_t *mulsc) noexcept
35 {
36   const usize lo = n >> 1;
37   const usize hi = n - lo;      // hi >= lo, so every mul below already has an >= bn
38 
39   limb_t qh = (hi < threshold::div_dc) ? sbpi1_div_qr(qp + lo, np + 2u * lo, 2u * hi, dp + lo, hi, dinv)
40                                        : dc_div_qr_n(qp + lo, np + 2u * lo, dp + lo, hi, dinv, tp, mulsc);
41 
42   mul(tp, qp + lo, hi, dp, lo, mulsc);
43 
44   limb_t cy = sub_n(np + lo, np + lo, tp, n);
45   if ( qh != 0 ) cy = static_cast<limb_t>(cy + sub_n(np + n, np + n, dp, lo));
46 
47   while ( cy != 0 ) {
48     qh = static_cast<limb_t>(qh - sub_1(qp + lo, qp + lo, hi, 1));
49     cy = static_cast<limb_t>(cy - add_n(np + lo, np + lo, dp, n));
50   }
51 
52   const limb_t ql = (lo < threshold::div_dc) ? sbpi1_div_qr(qp, np + hi, 2u * lo, dp + hi, lo, dinv)
53                                              : dc_div_qr_n(qp, np + hi, dp + hi, lo, dinv, tp, mulsc);
54 
55   mul(tp, dp, hi, qp, lo, mulsc);
56 
57   cy = sub_n(np, np, tp, n);
58   if ( ql != 0 ) cy = static_cast<limb_t>(cy + sub_n(np + lo, np + lo, dp, hi));
59 
60   while ( cy != 0 ) {
61     (void)sub_1(qp, qp, lo, 1);
62     cy = static_cast<limb_t>(cy - add_n(np, np, dp, n));
63   }
64   return qh;
65 }

67 inline constexpr limb_t
68 dc_div_qr(limb_t *qp, limb_t *np, usize nn, const limb_t *dp, usize dn, limb_t dinv, limb_t *scratch) noexcept
69 {
70   limb_t *const tp = scratch;
71   limb_t *const mulsc = tp + dn;
72 
73   const usize qn = nn - dn;
74 
75   const limb_t qh = static_cast<limb_t>(cmp(np + qn, dp, dn) >= 0);
76   if ( qh != 0 ) (void)sub_n(np + qn, np + qn, dp, dn);
77 
78   const usize r = qn % dn;
79   for ( usize j = qn - r; j >= dn; ) {
80     j -= dn;
81     (void)dc_div_qr_n(qp + r + j, np + r + j, dp, dn, dinv, tp, mulsc);
82   }
83   if ( r != 0 ) (void)sbpi1_div_qr(qp, np, r + dn, dp, dn, dinv);
84 
85   return qh;
86 }
