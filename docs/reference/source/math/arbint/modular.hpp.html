22 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
23 // modular layer
24 //
25 //   montgomery<U>   odd modulus, precomputed once; even moduli throw; the fast path for exponentiation
26 //   barrett<U>      any modulus, precomputed once
27 //   powmod/mulmod   one-shot, no context to keep, dispatching on parity
28 //
29 // WARNING: montgomery::mul and montgomery::sqr take and return montgomery form,
30 //          while montgomery::pow takes and returns an ordinary residue
31 //
32 //
33 // NOTE: PINNED SOLVERS DO NOT REACH IN HERE

43 [[nodiscard, gnu::always_inline]] inline constexpr usize
44 mod_max(usize a, usize b) noexcept
45 {
46   return a > b ? a : b;
47 }
48 
49 template<class U> struct mod_sizes {
50   static constexpr bool bounded = U::bounded;
51   static constexpr usize cap = U::cap_limbs;
52 
53   // __pow2_mod at its widest, k == 2*cap: np(k+1) + qp(k-n+2) + rp(n) + divrem_itch(k+1, n)
54   // Barrett's ctor shares it with a Newton reciprocal, whose FFT can need more than division.
55   static constexpr usize setup
56       = bounded ? mod_max((2u * cap + 1u) + (cap + 2u) + cap + mpn::divrem_cap_itch(2u * cap + 1u, cap), 2u * cap + mpn::invert_n_itch(cap))
57                 : 1u;
58 
59   // prod(2cap) + qp(cap+2) + rp(cap) + divrem_itch(2cap, cap)
60   // mulmod, sqrmod, __reduced and barrett::reduce all carve out of this one
61   static constexpr usize wide = bounded ? 2u * cap + (cap + 2u) + cap + mpn::divrem_cap_itch(2u * cap, cap) : 1u;
62 
63   // __mont_op / __bar_op both lay two n-limb operands down first and then hand the rest to the engine
64   static constexpr usize op = bounded ? 2u * cap + mod_max(mpn::mont_op_itch(cap), mpn::barrett_op_itch(cap)) : 1u;
65 
66   // out(n) + the window. powm_itch_max and NOT powm_itch(cap, ...)
67   static constexpr usize powm = bounded ? cap + mpn::powm_itch_max(cap, mpn::threshold::powm_table_limbs) : 1u;
68 };

75 template<arb_unsigned U> class montgomery
76 {
77 public:
78   using value_type = U;
79   using allocator_type = typename U::allocator_type;
80 
81   static constexpr usize width_bits = U::width_bits;
82   static constexpr bool bounded = U::bounded;
83   static constexpr usize cap_limbs = U::cap_limbs;
84 
85 private:
86   using sizes = __arb::mod_sizes<U>;
87   template<usize N> using scratch_for = __arb::scratch<bounded ? N : 1u, allocator_type, bounded>;
88 
89   U md;                  // modulus, odd
90   U r1;                  // R mod m
91   U r2;                  // R^2 mod m
92   mpn::limb_t minv;      // -m^-1 mod B
93   u32 mn;                // Montgomery width in limbs; R == B^mn
94 
95   static_assert(
96       !bounded || sizes::powm * sizeof(mpn::limb_t) <= (usize{ 1 } << 18),
97       "arbint: a bounded powmod at this width would park more than 256 KiB of frame; use the dynamic arbuint<> for a modulus this large");
98 
99   [[nodiscard, gnu::always_inline]] constexpr const mpn::limb_t *
100   mlimbs() const noexcept
101   {
102     return md.limbs();
103   }
104 
105   // B^k mod m, for k == mn (R) and k == 2*mn (R^2)
106   constexpr void
107   __pow2_mod(U &out, usize k)
108   {
109     const usize n = mn;
110     scratch_for<sizes::setup> sc(k + 1u + (k - n + 1u) + n + mpn::divrem_itch(k + 1u, n));
111     mpn::limb_t *const np = sc.get();               // k + 1
112     mpn::limb_t *const qp = np + k + 1u;            // k + 1 - n + 1
113     mpn::limb_t *const rp = qp + (k - n + 2u);      // n
114     mpn::limb_t *const work = rp + n;
115 
116     mpn::zero(np, k);
117     np[k] = 1;
118     mpn::divrem(qp, rp, np, k + 1u, mlimbs(), n, work);
119     out.__assign_limbs(rp, n);
120   }
121 
122 public:
123   // ctors
124   explicit constexpr montgomery(const U &modulus) : md(modulus), r1(), r2(), minv(0), mn(0)
125   {
126     if ( md.is_zero() ) micron::exc<micron::except::domain_error>("micron::math::montgomery modulus is zero");
127     if ( md.even() ) micron::exc<micron::except::domain_error>("micron::math::montgomery requires an odd modulus");
128     mn = static_cast<u32>(md.size());
129     minv = static_cast<mpn::limb_t>(static_cast<mpn::limb_t>(0) - mpn::binv_odd(md.limbs()[0]));
130     __pow2_mod(r1, mn);
131     __pow2_mod(r2, 2u * static_cast<usize>(mn));
132   }
133 
134   constexpr montgomery(const montgomery &) = default;
135   constexpr montgomery(montgomery &&) noexcept = default;
136   constexpr montgomery &operator=(const montgomery &) = default;
137   constexpr montgomery &operator=(montgomery &&) noexcept = default;
138   ~montgomery() = default;
139 
140   [[nodiscard, gnu::always_inline]] constexpr const U &
141   modulus() const noexcept
142   {
143     return md;
144   }
145 
146   [[nodiscard, gnu::always_inline]] constexpr const U &
147   one_form() const noexcept
148   {
149     return r1;
150   }
151 
152   [[nodiscard, gnu::always_inline]] constexpr usize
153   width() const noexcept
154   {
155     return mn;
156   }
157 
158   [[nodiscard, gnu::always_inline]] constexpr mpn::limb_t
159   inv_limb() const noexcept
160   {
161     return minv;
162   }

167   // a*R mod m. a need not be reduced
168   [[nodiscard]] constexpr U
169   to_form(const U &a) const
170   {
171     U t = __reduced(a);
172     return __mont_op(t, r2, false);
173   }
174 
175   // a*R^-1 mod m
176   [[nodiscard]] constexpr U
177   from_form(const U &a) const
178   {
179     const usize n = mn;
180     scratch_for<sizes::op> sc(2u * n + mpn::from_mont_itch(n));
181     mpn::limb_t *const tp = sc.get();
182     mpn::limb_t *const out = tp + 2u * n;
183 
184     const mpn::limb_t *const ap = a.limbs();
185     const usize an = a.size();
186     mpn::copyi(out, ap, an < n ? an : n);
187     if ( an < n ) mpn::zero(out + an, n - an);
188 
189     const mpn::mont_ctx c{ mlimbs(), n, minv };
190     mpn::copyi(tp, out, n);
191     mpn::zero(tp + n, n);
192     mpn::redc(out, tp, c);
193 
194     U r;
195     r.__assign_limbs(out, n);
196     return r;
197   }
198 
199   [[nodiscard]] constexpr U
200   reduce(const U &t) const
201   {
202     return from_form(t);
203   }
204 
205   [[nodiscard]] constexpr U
206   mul(const U &a, const U &b) const
207   {
208     return __mont_op(a, b, false);
209   }
210 
211   [[nodiscard]] constexpr U
212   sqr(const U &a) const
213   {
214     return __mont_op(a, a, true);
215   }
216 
217   [[nodiscard]] constexpr U
218   pow(const U &a, const U &ex) const
219   {
220     return __powm_into(a, ex.limbs(), ex.size());
221   }
222 
223   [[nodiscard]] constexpr U
224   pow(const U &a, u64 ex) const
225   {
226     mpn::limb_t buf[(sizeof(u64) + sizeof(mpn::limb_t) - 1u) / sizeof(mpn::limb_t)] = {};
227     constexpr usize take = sizeof(buf) / sizeof(buf[0]);
228     u64 v = ex;
229     for ( usize i = 0; i < take; ++i ) {
230       buf[i] = static_cast<mpn::limb_t>(v);
231       v = (mpn::limb_bits >= 64u) ? 0u : (v >> (mpn::limb_bits & 63u));
232     }
233     return __powm_into(a, buf, take);
234   }
235 
236 private:
237   [[nodiscard]] constexpr U
238   __reduced(const U &a) const
239   {
240     // a mod m, by the shortest route that is correct for both widths
241     const usize n = mn;
242     const usize an = a.size();
243     if ( an < n || mpn::cmp_var(a.limbs(), an, mlimbs(), n) < 0 ) {
244       U t(a);
245       return t;
246     }
247     scratch_for<sizes::wide> sc((an - n + 1u) + n + mpn::divrem_itch(an, n));
248     mpn::limb_t *const qp = sc.get();
249     mpn::limb_t *const rp = qp + (an - n + 1u);
250     mpn::limb_t *const work = rp + n;
251     mpn::divrem(qp, rp, a.limbs(), an, mlimbs(), n, work);
252     U r;
253     r.__assign_limbs(rp, n);
254     return r;
255   }
256 
257   [[nodiscard]] constexpr U
258   __mont_op(const U &a, const U &b, bool square) const
259   {
260     const usize n = mn;
261     scratch_for<sizes::op> sc(2u * n + mpn::mont_op_itch(n));
262     mpn::limb_t *const av = sc.get();      // n
263     mpn::limb_t *const bv = av + n;        // n
264     mpn::limb_t *const work = bv + n;
265 
266     __spread(av, a);
267     if ( !square ) __spread(bv, b);
268 
269     const mpn::mont_ctx c{ mlimbs(), n, minv };
270     if ( square )
271       mpn::mont_sqr(av, av, c, work);
272     else
273       mpn::mont_mul(av, av, bv, c, work);
274 
275     U r;
276     r.__assign_limbs(av, n);
277     return r;
278   }
279 
280   [[gnu::always_inline]] constexpr void
281   __spread(mpn::limb_t *dst, const U &a) const noexcept
282   {
283     const usize n = mn;
284     const usize an = a.size();
285     const usize take = an < n ? an : n;
286     mpn::copyi(dst, a.limbs(), take);
287     if ( take < n ) mpn::zero(dst + take, n - take);
288   }
289 
290   [[nodiscard]] constexpr U
291   __powm_into(const U &a, const mpn::limb_t *ep, usize en) const
292   {
293     const usize n = mn;
294     const usize an = a.size();
295     const usize ebits = mpn::bitlen(ep, en);
296     scratch_for<sizes::powm> sc(n + mpn::powm_itch(n, an != 0 ? an : 1u, ebits));
297     mpn::limb_t *const out = sc.get();
298     mpn::limb_t *const work = out + n;
299 
300     mpn::powm_with<mpn::modalgo::redc>(out, a.limbs(), an, ep, en, mlimbs(), n, work);
301     U r;
302     r.__assign_limbs(out, n);
303     return r;
304   }
305 };

309 template<arb_unsigned U> class barrett
310 {
311 public:
312   using value_type = U;
313   using wide_type = __arb::wide_t<U>;
314   using allocator_type = typename U::allocator_type;
315 
316   static constexpr usize width_bits = U::width_bits;
317   static constexpr bool bounded = U::bounded;
318   static constexpr usize cap_limbs = U::cap_limbs;
319 
320 private:
321   using sizes = __arb::mod_sizes<U>;
322   template<usize N> using scratch_for = __arb::scratch<bounded ? N : 1u, allocator_type, bounded>;
323 
324   U md;
325   wide_type mw;      // modulus normalized
326   wide_type iv;      // reciprocal
327   u32 mn;
328   u32 sh;
329 
330   static_assert(
331       !bounded || sizes::powm * sizeof(mpn::limb_t) <= (usize{ 1 } << 18),
332       "arbint: a bounded powmod at this width would park more than 256 KiB of frame; use the dynamic arbuint<> for a modulus this large");
333 
334 public:
335   // ctors
336   explicit constexpr barrett(const U &modulus) : md(modulus), mw(), iv(), mn(0), sh(0)
337   {
338     if ( md.is_zero() ) micron::exc<micron::except::domain_error>("micron::math::barrett modulus is zero");
339     mn = static_cast<u32>(md.size());
340     const usize n = mn;
341     sh = static_cast<u32>(mpn::limb_clz(md.limbs()[n - 1u]));
342 
343     scratch_for<sizes::setup> sc(2u * n + mpn::invert_n_itch(n));
344     mpn::limb_t *const mp = sc.get();      // n
345     mpn::limb_t *const ip = mp + n;        // n
346     mpn::limb_t *const work = ip + n;
347 
348     if ( sh != 0 )
349       (void)mpn::lshift(mp, md.limbs(), n, sh);
350     else
351       mpn::copyi(mp, md.limbs(), n);
352     mpn::invert_n(ip, mp, n, work);
353 
354     mw.__assign_limbs(mp, n);
355     iv.__assign_limbs(ip, n);
356   }
357 
358   constexpr barrett(const barrett &) = default;
359   constexpr barrett(barrett &&) noexcept = default;
360   constexpr barrett &operator=(const barrett &) = default;
361   constexpr barrett &operator=(barrett &&) noexcept = default;
362   ~barrett() = default;
363 
364   [[nodiscard, gnu::always_inline]] constexpr const U &
365   modulus() const noexcept
366   {
367     return md;
368   }
369 
370   [[nodiscard, gnu::always_inline]] constexpr const wide_type &
371   inverse() const noexcept
372   {
373     return iv;
374   }
375 
376   [[nodiscard, gnu::always_inline]] constexpr usize
377   width() const noexcept
378   {
379     return mn;
380   }
381 
382   [[nodiscard, gnu::always_inline]] constexpr usize
383   shift() const noexcept

386   }

388   [[nodiscard]] constexpr U
389   reduce(const U &a) const
390   {
391     const usize n = mn;
392     const usize an = a.size();
393     if ( an < n || mpn::cmp_var(a.limbs(), an, md.limbs(), n) < 0 ) return a;
394     scratch_for<sizes::wide> sc((an - n + 1u) + n + mpn::divrem_itch(an, n));
395     mpn::limb_t *const qp = sc.get();
396     mpn::limb_t *const rp = qp + (an - n + 1u);
397     mpn::limb_t *const work = rp + n;
398     mpn::divrem(qp, rp, a.limbs(), an, md.limbs(), n, work);
399     U r;
400     r.__assign_limbs(rp, n);
401     return r;
402   }

447   __bar_op(const U &a, const U &b, bool square) const
448   {
449     const usize n = mn;
450     scratch_for<sizes::op> sc(2u * n + mpn::barrett_op_itch(n));
451     mpn::limb_t *const av = sc.get();
452     mpn::limb_t *const bv = av + n;
453     mpn::limb_t *const work = bv + n;
454

457

459     if ( square )
460       mpn::barrett_sqr(av, av, c, work);

462       mpn::barrett_mul(av, av, bv, c, work);
463 
464     U r;
465     r.__assign_limbs(av, n);
466     return r;
467   }

477     mpn::limb_t *const work = out + n;

480     U r;
481     r.__assign_limbs(out, n);
482     return r;
483   }

489 // a*b mod m
490 template<usize B, arb_solver S, class A>
491 [[nodiscard]] inline constexpr arbuint<B, S, A>
492 mulmod(const arbuint<B, S, A> &a, const arbuint<B, S, A> &b, const arbuint<B, S, A> &m)
493 {
494   using U = arbuint<B, S, A>;
495   using sizes = __arb::mod_sizes<U>;
496   const usize n = m.size();
497   if ( n == 0 ) micron::exc<micron::except::domain_error>("micron::math::mulmod modulus is zero");
498 
499   const usize an = a.size();
500   const usize bn = b.size();
501   if ( an == 0 || bn == 0 ) return U::zero();
502 
503   const usize pn = an + bn;
504   __arb::scratch<U::bounded ? sizes::wide : 1u, A, U::bounded> sc(pn + (pn > n ? pn - n + 1u : 1u) + n
505                                                                   + mpn::divrem_itch(pn > n ? pn : n, n));
506   mpn::limb_t *const prod = sc.get();
507   mpn::limb_t *const qp = prod + pn;
508   mpn::limb_t *const rp = qp + (pn > n ? pn - n + 1u : 1u);
509   mpn::limb_t *const work = rp + n;
510 
511   if ( an >= bn )
512     U::__mul_into(prod, a.limbs(), an, b.limbs(), bn, work);
513   else
514     U::__mul_into(prod, b.limbs(), bn, a.limbs(), an, work);
515 
516   const usize rn = mpn::normalize(prod, pn);
517   U r;
518   if ( rn < n || mpn::cmp_var(prod, rn, m.limbs(), n) < 0 ) {
519     r.__assign_limbs(prod, rn);
520     return r;
521   }
522   mpn::divrem(qp, rp, prod, rn, m.limbs(), n, work);
523   r.__assign_limbs(rp, n);
524   return r;
525 }

527 // a*a mod m via squaring ladder
528 template<usize B, arb_solver S, class A>
529 [[nodiscard]] inline constexpr arbuint<B, S, A>
530 sqrmod(const arbuint<B, S, A> &a, const arbuint<B, S, A> &m)
531 {
532   using U = arbuint<B, S, A>;
533   using sizes = __arb::mod_sizes<U>;
534   const usize n = m.size();
535   if ( n == 0 ) micron::exc<micron::except::domain_error>("micron::math::sqrmod modulus is zero");
536 
537   const usize an = a.size();
538   if ( an == 0 ) return U::zero();
539 
540   const usize pn = 2u * an;
541   __arb::scratch<U::bounded ? sizes::wide : 1u, A, U::bounded> sc(pn + (pn > n ? pn - n + 1u : 1u) + n
542                                                                   + mpn::divrem_itch(pn > n ? pn : n, n));
543   mpn::limb_t *const prod = sc.get();
544   mpn::limb_t *const qp = prod + pn;
545   mpn::limb_t *const rp = qp + (pn > n ? pn - n + 1u : 1u);
546   mpn::limb_t *const work = rp + n;
547 
548   U::__sqr_into(prod, a.limbs(), an, work);
549 
550   const usize rn = mpn::normalize(prod, pn);
551   U r;
552   if ( rn < n || mpn::cmp_var(prod, rn, m.limbs(), n) < 0 ) {
553     r.__assign_limbs(prod, rn);
554     return r;
555   }
556   mpn::divrem(qp, rp, prod, rn, m.limbs(), n, work);
557   r.__assign_limbs(rp, n);
558   return r;
559 }

561 // base^exp mod m via the sliding window
562 template<usize B, arb_solver S, class A>
563 [[nodiscard]] inline constexpr arbuint<B, S, A>
564 powmod(const arbuint<B, S, A> &a, const arbuint<B, S, A> &ex, const arbuint<B, S, A> &m)
565 {
566   using U = arbuint<B, S, A>;
567   using sizes = __arb::mod_sizes<U>;
568   const usize n = m.size();
569   if ( n == 0 ) micron::exc<micron::except::domain_error>("micron::math::powmod modulus is zero");
570 
571   const usize an = a.size();
572   const usize en = ex.size();
573   const usize ebits = mpn::bitlen(ex.limbs(), en);
574 
575   __arb::scratch<U::bounded ? sizes::powm : 1u, A, U::bounded> sc(n + mpn::powm_itch(n, an != 0 ? an : 1u, ebits));
576   mpn::limb_t *const out = sc.get();
577   mpn::limb_t *const work = out + n;
578 
579   mpn::powm(out, a.limbs(), an, ex.limbs(), en, m.limbs(), n, work);
580   U r;
581   r.__assign_limbs(out, n);
582   return r;
583 }

592 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
593 // signed
594 template<usize B, arb_solver S, class A>
595 [[nodiscard]] inline constexpr arbint<B, S, A>
596 powmod(const arbint<B, S, A> &a, const arbint<B, S, A> &ex, const arbint<B, S, A> &m)
597 {
598   using SI = arbint<B, S, A>;
599   using U = arbuint<B, S, A>;
600   if ( ex.sign() < 0 ) micron::exc<micron::except::domain_error>("micron::math::powmod negative exponent");
601 
602   const U mm = abs(m).magnitude();
603   U base = abs(a).magnitude();
604   if ( mm.is_zero() ) micron::exc<micron::except::domain_error>("micron::math::powmod modulus is zero");
605 
606   base %= mm;
607   if ( a.sign() < 0 && !base.is_zero() ) base = mm - base;
608 
609   return SI(powmod(base, abs(ex).magnitude(), mm), false);
610 }
