15 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
16 // Toom-Cook 3
17 // five 1/3-size products instead of nine, O(n^1.465)
18 //
19 //  ..points        0     1    -1     2    inf
20 //  ..A(x)B(x)     v0    v1   vm1    v2   vinf
21 //
22 // step ordereding, writing (c4 c3 c2 c1 c0):
23 //     v2  -= vm1        (16 8 4 2 1) - (1 -1 1 -1 1) = (15 9 3 3 0)
24 //     v2  /= 3                                       = ( 5 3 1 1 0)
25 //     vm1  = (v1-vm1)/2                              = ( 0 1 0 1 0)
26 //     v1  -= v0                                      = ( 1 1 1 1 0)
27 //     v2   = (v2-v1)/2                               = ( 2 1 0 0 0)
28 //     v2  -= 2*vinf                                  = ( 0 1 0 0 0)  == c3
29 //     v1  -= vm1 + vinf                              = ( 0 0 1 0 0)  == c2
30 //     vm1 -= v2                                      = ( 0 0 0 1 0)  == c1
31 //
32 // NOTE: rp distinct from ap and bp, holds an + bn limbs
33 // balanced only; an == bn == n
34

45 inline constexpr limb_t inv3 = static_cast<limb_t>(limb_bits == 64 ? 0xAAAAAAAAAAAAAAABull : 0xAAAAAAABull);
46 inline constexpr limb_t third_max = static_cast<limb_t>(limb_max / 3u);
47 
48 [[gnu::flatten]] inline constexpr void
49 divexact_by3(limb_t *rp, const limb_t *up, usize n) noexcept
50 {
51   limb_t c = 0;
52   for ( usize i = 0; i < n; ++i ) {
53     limb_t l = 0;
54     const limb_t b = subb(up[i], c, 0, l);
55     const limb_t q = static_cast<limb_t>(l * inv3);
56     rp[i] = q;
57     c = static_cast<limb_t>(b + (q > third_max) + (q > static_cast<limb_t>(third_max * 2u)));
58   }
59 }

70 [[nodiscard, gnu::flatten]] inline constexpr usize
71 toom3_itch(usize n, bool force_top = false) noexcept
72 {
73   usize total = 0;
74   if ( force_top && n >= 3u && n < threshold::mul_toom3 ) {
75     const usize k = (n + 2u) / 3u;
76     total += 6u * (k + 1u) + 5u * (2u * k + 2u);
77     n = k + 1u;
78   }
79   while ( n >= threshold::mul_toom3 && n >= 3u ) {
80     const usize k = (n + 2u) / 3u;
81     total += 6u * (k + 1u) + 5u * (2u * k + 2u);
82     n = k + 1u;
83   }
84   return total + karatsuba_itch(n, karatsuba_force);
85 }

91 // evaluation
92 //
93 // a0 + a1 + a2 and a0 + 2a1 + 4a2 are at most 3 and 7 times a limb block; both go in k+1 limbs
94 
95 // s = a0 + a1 + a2, k+1 limbs
96 inline constexpr void
97 toom3_eval1(limb_t *s, const limb_t *a0, const limb_t *a1, const limb_t *a2, usize k, usize t) noexcept
98 {
99   copyi(s, a0, k);
100   s[k] = add_n(s, s, a1, k);
101   s[k] = static_cast<limb_t>(s[k] + add(s, s, k, a2, t));
102 }
103 
104 // s = a0 + 2*a1 + 4*a2 by Horner, k+1 limbs
105 inline constexpr void
106 toom3_eval2(limb_t *s, const limb_t *a0, const limb_t *a1, const limb_t *a2, usize k, usize t) noexcept
107 {
108   copyi(s, a2, t);
109   zero(s + t, k + 1u - t);
110   (void)lshift(s, s, k + 1u, 1);                              // 2*a2, top limb still tiny
111   s[k] = static_cast<limb_t>(s[k] + add_n(s, s, a1, k));      // + a1
112   (void)lshift(s, s, k + 1u, 1);                              // *2
113   s[k] = static_cast<limb_t>(s[k] + add_n(s, s, a0, k));      // + a0
114 }
115 
116 // s = |a0 - a1 + a2| in k+1 limbs
117 [[nodiscard]] inline constexpr bool
118 toom3_evalm1(limb_t *s, limb_t *tmp, const limb_t *a0, const limb_t *a1, const limb_t *a2, usize k, usize t) noexcept
119 {
120   copyi(tmp, a0, k);
121   tmp[k] = add(tmp, tmp, k, a2, t);      // a0 + a2, at most 2*B^k so the top limb is 0 or 1
122 
123   if ( tmp[k] != 0 || cmp(tmp, a1, k) >= 0 ) {
124     copyi(s, tmp, k + 1u);
125     s[k] = static_cast<limb_t>(s[k] - sub_n(s, s, a1, k));
126     return false;
127   }
128   (void)sub_n(s, a1, tmp, k);
129   s[k] = 0;
130   return true;
131 }

133 inline constexpr void
134 toom_interpolate_5pts(limb_t *__restrict__ rp, limb_t *w0, limb_t *w1, limb_t *wm1, limb_t *w2, limb_t *winf, usize k, usize n, usize t,
135                       usize kk, bool sm) noexcept
136 {
137   if ( sm )
138     (void)add_n(w2, w2, wm1, kk);
139   else
140     (void)sub_n(w2, w2, wm1, kk);
141   divexact_by3(w2, w2, kk);
142 
143   if ( sm )
144     (void)add_n(wm1, w1, wm1, kk);
145   else
146     (void)sub_n(wm1, w1, wm1, kk);
147   (void)rshift(wm1, wm1, kk, 1);      // exact: (v1 -/+ vm1) is always even
148 
149   (void)sub_n(w1, w1, w0, kk);
150   (void)sub_n(w2, w2, w1, kk);
151   (void)rshift(w2, w2, kk, 1);      // exact again
152 
153   // w2 -= 2*vinf
154   (void)sub_n(w2, w2, winf, kk);
155   (void)sub_n(w2, w2, winf, kk);
156 
157   (void)sub_n(w1, w1, wm1, kk);
158   (void)sub_n(w1, w1, winf, kk);
159   (void)sub_n(wm1, wm1, w2, kk);
160 
161   const usize rn = 2u * n;
162   zero(rp, rn);
163   copyi(rp, w0, 2u * k);
164   (void)add(rp + k, rp + k, rn - k, wm1, normalize(wm1, kk));
165   (void)add(rp + 2u * k, rp + 2u * k, rn - 2u * k, w1, normalize(w1, kk));
166   (void)add(rp + 3u * k, rp + 3u * k, rn - 3u * k, w2, normalize(w2, kk));
167   (void)add(rp + 4u * k, rp + 4u * k, rn - 4u * k, winf, normalize(winf, 2u * t));
168 }

170 inline constexpr void
171 mul_toom3(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, const limb_t *__restrict__ bp, usize n, limb_t *scratch,
172           usize cutoff) noexcept
173 {
174   const usize k = (n + 2u) / 3u;
175   const usize t = n - 2u * k;
176   const usize kk = 2u * k + 2u;
177 
178   limb_t *const as1 = scratch;
179   limb_t *const as2 = as1 + (k + 1u);
180   limb_t *const asm1 = as2 + (k + 1u);
181   limb_t *const bs1 = asm1 + (k + 1u);
182   limb_t *const bs2 = bs1 + (k + 1u);
183   limb_t *const bsm1 = bs2 + (k + 1u);
184   limb_t *const w0 = bsm1 + (k + 1u);
185   limb_t *const w1 = w0 + kk;
186   limb_t *const wm1 = w1 + kk;
187   limb_t *const w2 = wm1 + kk;
188   limb_t *const winf = w2 + kk;
189   limb_t *const rec = winf + kk;
190 
191   const limb_t *const a0 = ap;
192   const limb_t *const a1 = ap + k;
193   const limb_t *const a2 = ap + 2u * k;
194   const limb_t *const b0 = bp;
195   const limb_t *const b1 = bp + k;
196   const limb_t *const b2 = bp + 2u * k;
197 
198   toom3_eval1(as1, a0, a1, a2, k, t);
199   toom3_eval1(bs1, b0, b1, b2, k, t);
200   toom3_eval2(as2, a0, a1, a2, k, t);
201   toom3_eval2(bs2, b0, b1, b2, k, t);
202   const bool sa = toom3_evalm1(asm1, w0, a0, a1, a2, k, t);      // w0 is free scratch until v0 lands
203   const bool sb = toom3_evalm1(bsm1, w1, b0, b1, b2, k, t);
204   const bool sm = sa != sb;
205 
206   __mul_rec3(w1, as1, k + 1u, bs1, k + 1u, rec, cutoff);
207   __mul_rec3(wm1, asm1, k + 1u, bsm1, k + 1u, rec, cutoff);
208   __mul_rec3(w2, as2, k + 1u, bs2, k + 1u, rec, cutoff);
209   __mul_rec3(w0, a0, k, b0, k, rec, cutoff);
210   zero(w0 + 2u * k, kk - 2u * k);
211   __mul_rec3(winf, a2, t, b2, t, rec, cutoff);
212   zero(winf + 2u * t, kk - 2u * t);
213 
214   toom_interpolate_5pts(rp, w0, w1, wm1, w2, winf, k, n, t, kk, sm);
215 }

217 inline constexpr void
218 __mul_rec3(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn, limb_t *scratch,
219            usize cutoff) noexcept
220 {
221   if ( an == bn && an >= threshold::mul_toom3 && toom3_split_ok(an) )
222     mul_toom3(rp, ap, bp, an, scratch, cutoff);
223   else
224     __mul_rec(rp, ap, an, bp, bn, scratch, cutoff);
225 }

265 inline constexpr void
266 sqr_toom3(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch, usize cutoff) noexcept
267 {
268   const usize k = (n + 2u) / 3u;
269   const usize t = n - 2u * k;
270   const usize kk = 2u * k + 2u;
271 
272   limb_t *const as1 = scratch;
273   limb_t *const as2 = as1 + (k + 1u);
274   limb_t *const asm1 = as2 + (k + 1u);
275   limb_t *const w0 = asm1 + (k + 1u);
276   limb_t *const w1 = w0 + kk;
277   limb_t *const wm1 = w1 + kk;
278   limb_t *const w2 = wm1 + kk;
279   limb_t *const winf = w2 + kk;
280   limb_t *const rec = winf + kk;
281 
282   const limb_t *const a0 = ap;
283   const limb_t *const a1 = ap + k;
284   const limb_t *const a2 = ap + 2u * k;
285 
286   toom3_eval1(as1, a0, a1, a2, k, t);
287   toom3_eval2(as2, a0, a1, a2, k, t);
288   (void)toom3_evalm1(asm1, w0, a0, a1, a2, k, t);
289 
290   __sqr_rec3(w1, as1, k + 1u, rec, cutoff);
291   __sqr_rec3(wm1, asm1, k + 1u, rec, cutoff);
292   __sqr_rec3(w2, as2, k + 1u, rec, cutoff);
293   __sqr_rec3(w0, a0, k, rec, cutoff);
294   zero(w0 + 2u * k, kk - 2u * k);
295   __sqr_rec3(winf, a2, t, rec, cutoff);
296   zero(winf + 2u * t, kk - 2u * t);
297 
298   toom_interpolate_5pts(rp, w0, w1, wm1, w2, winf, k, n, t, kk, false);
299 }
