17 // Lehmer (accelerated Euclid)
18 //
19 // Lehmer simulates a whole run of Euclidean steps on the leading limbs;
20 // each step costs a single limb divide, then applies the accumulated 2x2 matrix to the full operands
21 // run of ten steps costs ten limb divides plus one pass
22

25 namespace math

27 namespace mpn

30 [[nodiscard, gnu::flatten]] inline constexpr usize
31 gcd_lehmer_itch(usize un, usize vn) noexcept
32 {
33   if ( vn > un ) {
34     const usize t = un;
35     un = vn;
36     vn = t;
37   }
38   if ( vn == 0 ) return 0;
39   return 4u * un + (un + 1u) + divrem_cap_itch(un, vn);
40 }
41 
42 // un >= vn >= 1
43 [[nodiscard]] inline constexpr usize
44 gcd_lehmer(limb_t *gp, const limb_t *up, usize un, const limb_t *vp, usize vn, limb_t *scratch) noexcept
45 {
46   limb_t *a = scratch;           // un
47   limb_t *b = a + un;            // un
48   limb_t *t = b + un;            // un
49   limb_t *s = t + un;            // un
50   limb_t *const q = s + un;      // un + 1
51   limb_t *const work = q + un + 1u;
52 
53   usize an = un, bn = vn;
54   copyi(a, up, un);
55   copyi(b, vp, vn);
56 
57   for ( ;; ) {
58     if ( bn == 0 ) break;
59     if ( an <= 2 && bn <= 2 ) {
60       limb_t g1 = 0, g0 = 0;
61       gcd_22_any(g1, g0, an > 1 ? a[1] : 0, a[0], bn > 1 ? b[1] : 0, b[0]);
62       a[0] = g0;
63       a[1] = g1;
64       an = normalize(a, 2);
65       bn = 0;
66       break;
67     }
68     if ( bn == 1 ) {
69       a[0] = gcd_1(a, an, b[0]);
70       an = a[0] != 0 ? 1u : 0u;
71       bn = 0;
72       break;
73     }
74 
75     if ( an == bn ) {
76       const usize sh = limb_clz(a[an - 1u]);
77       limb_t A = 0, A0 = 0, B = 0, B0 = 0;
78       __window2(A, A0, a, an, sh);
79       __window2(B, B0, b, an, sh);
80 
81       mat1 m{};
82       if ( hgcd2_sel(m, A, A0, B, B0) ) {
83         usize n2 = 0;
84         if ( mat1_mul_inverse_vector(m, t, s, a, b, an, n2) ) {
85           limb_t *sw = a;
86           a = t;
87           t = sw;
88           sw = b;
89           b = s;
90           s = sw;
91           an = normalize(a, n2);
92           bn = normalize(b, n2);
93           if ( cmp_var(a, an, b, bn) < 0 ) {
94             sw = a;
95             a = b;
96             b = sw;
97             const usize tn = an;
98             an = bn;
99             bn = tn;
100           }
101           continue;
102         }
103         // the matrix was not valid for the full operands
104       }
105     }
106 
107     // no run could be validated
108     divrem(q, t, a, an, b, bn, work);
109     const usize tn = normalize(t, bn);
110     limb_t *sw = a;
111     a = b;
112     b = t;
113     t = sw;
114     an = bn;
115     bn = tn;
116   }
117 
118   if ( an == 0 ) return 0;
119   copyi(gp, a, an);
120   return an;
121 }
