34 struct barrett_ctx {
35   const limb_t *mp;      // n limbs, normalized
36   const limb_t *ip;      // n limbs; the reciprocal is B^n + ip
37   usize n;
38   usize shift;      // 0 .. limb_bits-1
39 };
40 
41 [[nodiscard]] inline constexpr barrett_ctx
42 barrett_make(limb_t *mwork, limb_t *iwork, const limb_t *mp, usize n, limb_t *scratch) noexcept
43 {
44   const usize s = limb_clz(mp[n - 1]);
45   if ( s != 0 )
46     (void)lshift(mwork, mp, n, s);
47   else
48     copyi(mwork, mp, n);
49   invert_n(iwork, mwork, n, scratch);
50   return barrett_ctx{ mwork, iwork, n, s };
51 }

53 [[nodiscard, gnu::always_inline]] inline constexpr usize
54 barrett_make_itch(usize n) noexcept
55 {
56   return invert_n_itch(n);
57 }

59 [[nodiscard, gnu::flatten]] inline constexpr usize
60 barrett_reduce_itch(usize n) noexcept
61 {
62   usize w = __mul_itch_bound(n + 1u, n);
63   const usize v = __mul_itch_bound(n, n);
64   if ( v > w ) w = v;
65   // xw 2n+1, P 2n+1, T n+2, q3 n, U 2n, rr n+1
66   return (2u * n + 1u) + (2u * n + 1u) + (n + 2u) + n + (2u * n) + (n + 1u) + w;
67 }

69 // rp = {xp, xn} mod m, both unshifted, rp taking n limbs
70 inline constexpr void
71 barrett_reduce(limb_t *rp, const limb_t *xp, usize xn, const barrett_ctx &c, limb_t *scratch) noexcept
72 {
73   const usize n = c.n;
74   limb_t *const xw = scratch;               // 2n + 1
75   limb_t *const pp = xw + 2u * n + 1u;      // 2n + 1
76   limb_t *const tt = pp + 2u * n + 1u;      // n + 2
77   limb_t *const q3 = tt + n + 2u;           // n
78   limb_t *const uu = q3 + n;                // 2n
79   limb_t *const rr = uu + 2u * n;           // n + 1
80   limb_t *const work = rr + n + 1u;
81 
82   if ( xn > 2u * n ) xn = 2u * n;
83   copyi(xw, xp, xn);
84   zero(xw + xn, 2u * n + 1u - xn);
85   if ( c.shift != 0 ) xw[2u * n] = lshift(xw, xw, 2u * n, c.shift);
86 
87   // q1 = xw + n - 1, n + 1 limbs
88   mul(pp, xw + n - 1u, n + 1u, c.ip, n, work);
89 
90   // T = floor(q2 / B^n) = q1 + floor(q1*ip / B^n)
91   copyi(tt, pp + n, n + 1u);
92   tt[n + 1u] = add_n(tt, tt, xw + n - 1u, n + 1u);
93 
94   // q3 = floor(T / B)
95   if ( tt[n + 1u] != 0 ) [[unlikely]]
96     for ( usize i = 0; i < n; ++i ) q3[i] = limb_max;
97   else
98     copyi(q3, tt + 1u, n);
99 
100   mul(uu, q3, n, c.mp, n, work);
101 
102   (void)sub_n(rr, xw, uu, n + 1u);      // wraps by design
103 
104   for ( ;; ) {
105     if ( rr[n] == 0 && cmp(rr, c.mp, n) < 0 ) break;
106     (void)sub(rr, rr, n + 1u, c.mp, n);
107   }
108 
109   if ( c.shift != 0 )
110     (void)rshift(rp, rr, n, c.shift);
111   else
112     copyi(rp, rr, n);
113 }

115 // rp = a*b mod m, operands reduced, n limbs each
116 inline constexpr void
117 barrett_mul(limb_t *rp, const limb_t *ap, const limb_t *bp, const barrett_ctx &c, limb_t *scratch) noexcept
118 {
119   limb_t *const prod = scratch;      // 2n
120   limb_t *const work = prod + 2u * c.n;
121   mul(prod, ap, c.n, bp, c.n, work);
122   barrett_reduce(rp, prod, 2u * c.n, c, work);
123 }
124 
125 inline constexpr void
126 barrett_sqr(limb_t *rp, const limb_t *ap, const barrett_ctx &c, limb_t *scratch) noexcept
127 {
128   limb_t *const prod = scratch;      // 2n
129   limb_t *const work = prod + 2u * c.n;
130   sqr(prod, ap, c.n, work);
131   barrett_reduce(rp, prod, 2u * c.n, c, work);
132 }

134 [[nodiscard, gnu::flatten]] inline constexpr usize
135 barrett_op_itch(usize n) noexcept
136 {
137   usize w = __mul_itch_bound(n, n);
138   const usize s = __sqr_itch_bound(n);
139   if ( s > w ) w = s;
140   const usize r = barrett_reduce_itch(n);
141   if ( r > w ) w = r;
142   return 2u * n + w;
143 }

145 [[nodiscard, gnu::flatten]] inline constexpr usize
146 powm_window_for(usize ebits, usize n, usize budget) noexcept
147 {
148   usize k = powm_window(ebits);
149   if ( k > threshold::powm_window_cap ) k = threshold::powm_window_cap;
150   if ( n == 0 ) n = 1;
151   while ( k > 1u && ((usize{ 1 } << (k - 1u)) * n) > budget ) --k;
152   return k;
153 }

155 // %%%%%%%%%%%%%%%%%%%%%%%%
156 // scratch
157 [[nodiscard, gnu::flatten]] inline constexpr usize
158 __powm_work_itch(usize n, usize bn) noexcept
159 {
160   usize mont = mont_op_itch(n);
161   const usize t = to_mont_itch(bn, n);
162   if ( t > mont ) mont = t;
163 
164   usize bar = barrett_make_itch(n);
165   const usize bop = barrett_op_itch(n);
166   if ( bop > bar ) bar = bop;
167   const usize bred = (bn > n ? (bn - n + 1u) : 1u) + divrem_cap_itch(bn > n ? bn : n, n);
168   if ( bred > bar ) bar = bred;
169   bar += 2u * n;
170 
171   return mont > bar ? mont : bar;
172 }

174 [[nodiscard, gnu::flatten]] inline constexpr usize
175 powm_itch(usize n, usize bn, usize ebits, usize budget = threshold::powm_table_limbs) noexcept
176 {
177   if ( n == 0 ) return 0;
178   if ( bn == 0 ) bn = 1;
179   const usize k = powm_window_for(ebits, n, budget);
180   const usize tbl = (usize{ 1 } << (k - 1u)) * n;
181   return tbl + 2u * n + __powm_work_itch(n, bn);
182 }
183 
184 [[nodiscard, gnu::flatten]] inline constexpr usize
185 powm_itch_max(usize cap, usize budget = threshold::powm_table_limbs) noexcept
186 {
187   if ( cap == 0 ) return 0;
188   const usize tbl = budget > cap ? budget : cap;
189   return tbl + 2u * cap + __powm_work_itch(cap, cap);
190 }

195 [[nodiscard, gnu::always_inline]] inline constexpr usize
196 __powm_bits(const limb_t *ep, usize en, usize lo, usize len) noexcept
197 {
198   usize w = 0;
199   for ( usize j = len; j-- > 0; ) w = (w << 1) | (testbit(ep, en, lo + j) ? 1u : 0u);
200   return w;
201 }

203 template<class Engine>
204 inline constexpr void
205 __powm_window(limb_t *rp, const limb_t *ep, usize en, usize ebits, usize n, usize k, limb_t *tbl, limb_t *acc, limb_t *tmp,
206               Engine &eng) noexcept
207 {
208   const usize entries = usize{ 1 } << (k - 1u);
209 
210   // odd powers: T[j] = base^(2j+1)
211   if ( entries > 1u ) {
212     eng.sqr(tmp, tbl);
213     for ( usize j = 1; j < entries; ++j ) eng.mul(tbl + j * n, tbl + (j - 1u) * n, tmp);
214   }
215 
216   bool first = true;
217   usize i = ebits;
218   while ( i > 0 ) {
219     if ( !testbit(ep, en, i - 1u) ) {
220       if ( !first ) eng.sqr(acc, acc);
221       --i;
222       continue;
223     }
224     usize len = k < i ? k : i;
225     while ( len > 1u && !testbit(ep, en, i - len) ) --len;
226     const usize w = __powm_bits(ep, en, i - len, len);
227 
228     if ( first ) {
229       copyi(acc, tbl + ((w - 1u) >> 1) * n, n);
230       first = false;
231     } else {
232       for ( usize t = 0; t < len; ++t ) eng.sqr(acc, acc);
233       eng.mul(acc, acc, tbl + ((w - 1u) >> 1) * n);
234     }
235     i -= len;
236   }
237   copyi(rp, acc, n);
238 }

240 template<modalgo K>
241 inline constexpr void
242 __powm_engine(limb_t *rp, const limb_t *bp, usize bn, const limb_t *ep, usize en, usize ebits, const limb_t *mp, usize n, usize k,
243               limb_t *tbl, limb_t *acc, limb_t *tmp, limb_t *work) noexcept
244 {
245   if constexpr ( K == modalgo::redc ) {
246     const mont_ctx c = mont_make(mp, n);
247 
248     struct eng_t {
249       const mont_ctx &c;
250       limb_t *w;
251 
252       [[gnu::always_inline]] constexpr void
253       mul(limb_t *r, const limb_t *a, const limb_t *b) const noexcept
254       {
255         mont_mul(r, a, b, c, w);
256       }
257 
258       [[gnu::always_inline]] constexpr void
259       sqr(limb_t *r, const limb_t *a) const noexcept
260       {
261         mont_sqr(r, a, c, w);
262       }
263     } eng{ c, work };
264 
265     to_mont(tbl, bp, bn, c, work);
266     if ( is_zero(tbl, n) ) {
267       // b == 0 mod m, and e > 0 here, so the answer is 0 whatever the window would do
268       zero(rp, n);
269       return;
270     }
271     __powm_window(acc, ep, en, ebits, n, k, tbl, acc, tmp, eng);
272     from_mont(rp, acc, c, work);
273   } else {
274     limb_t *const mw = work;        // n
275     limb_t *const iw = mw + n;      // n
276     limb_t *const bw = iw + n;      // the rest
277     const barrett_ctx c = barrett_make(mw, iw, mp, n, bw);
278 
279     struct eng_t {
280       const barrett_ctx &c;
281       limb_t *w;
282 
283       [[gnu::always_inline]] constexpr void
284       mul(limb_t *r, const limb_t *a, const limb_t *b) const noexcept
285       {
286         barrett_mul(r, a, b, c, w);
287       }
288 
289       [[gnu::always_inline]] constexpr void
290       sqr(limb_t *r, const limb_t *a) const noexcept
291       {
292         barrett_sqr(r, a, c, w);
293       }
294     } eng{ c, bw };
295 
296     if ( bn > n || cmp_var(bp, bn, mp, n) >= 0 ) {
297       const usize qn = bn > n ? bn - n + 1u : 1u;
298       limb_t *const qp = bw;
299       limb_t *const dw = qp + qn;
300       if ( bn < n ) {
301         copyi(tbl, bp, bn);
302         zero(tbl + bn, n - bn);
303       } else {
304         divrem(qp, tbl, bp, bn, mp, n, dw);
305       }
306     } else {
307       copyi(tbl, bp, bn);
308       zero(tbl + bn, n - bn);
309     }
310     if ( is_zero(tbl, n) ) {
311       zero(rp, n);
312       return;
313     }
314     __powm_window(rp, ep, en, ebits, n, k, tbl, acc, tmp, eng);
315   }
316 }

318 inline constexpr void
319 powm(limb_t *rp, const limb_t *bp, usize bn, const limb_t *ep, usize en, const limb_t *mp, usize n, limb_t *scratch,
320      usize budget = threshold::powm_table_limbs) noexcept
321 {
322   n = normalize(mp, n);
323   if ( n == 0 ) return;
324 
325   if ( n == 1 && mp[0] == 1 ) {
326     rp[0] = 0;
327     return;
328   }
329 
330   en = normalize(ep, en);
331   if ( en == 0 ) {
332     zero(rp, n);
333     rp[0] = 1;
334     return;
335   }
336 
337   bn = normalize(bp, bn);
338   if ( bn == 0 ) {
339     zero(rp, n);
340     return;
341   }
342 
343   const usize ebits = bitlen(ep, en);
344   const usize k = powm_window_for(ebits, n, budget);
345   const usize entries = usize{ 1 } << (k - 1u);
346 
347   limb_t *const tbl = scratch;                // entries * n
348   limb_t *const acc = tbl + entries * n;      // n
349   limb_t *const tmp = acc + n;                // n
350   limb_t *const work = tmp + n;
351 
352   const bool odd = (mp[0] & limb_t{ 1 }) != 0;
353   const bool use_mont = (threshold::powm_engine == 0u) ? odd : (threshold::powm_engine == 1u);
354 
355   if ( use_mont )
356     __powm_engine<modalgo::redc>(rp, bp, bn, ep, en, ebits, mp, n, k, tbl, acc, tmp, work);
357   else
358     __powm_engine<modalgo::barrett>(rp, bp, bn, ep, en, ebits, mp, n, k, tbl, acc, tmp, work);
359 }

363 // pinned engines
364 template<modalgo K>
365 inline constexpr void
366 powm_with(limb_t *rp, const limb_t *bp, usize bn, const limb_t *ep, usize en, const limb_t *mp, usize n, limb_t *scratch,
367           usize budget = threshold::powm_table_limbs) noexcept
368 {
369   static_assert(static_cast<u8>(K) <= static_cast<u8>(mod_engines_built),
370                 "arbint: this modular engine is not implemented yet -- see mpn::mod_engines_built");
371   n = normalize(mp, n);
372   if ( n == 0 ) return;
373   if ( n == 1 && mp[0] == 1 ) {
374     rp[0] = 0;
375     return;
376   }
377   en = normalize(ep, en);
378   if ( en == 0 ) {
379     zero(rp, n);
380     rp[0] = 1;
381     return;
382   }
383   bn = normalize(bp, bn);
384   if ( bn == 0 ) {
385     zero(rp, n);
386     return;
387   }
388 
389   const usize ebits = bitlen(ep, en);
390   const usize k = powm_window_for(ebits, n, budget);
391   const usize entries = usize{ 1 } << (k - 1u);
392 
393   limb_t *const tbl = scratch;
394   limb_t *const acc = tbl + entries * n;
395   limb_t *const tmp = acc + n;
396   limb_t *const work = tmp + n;
397 
398   __powm_engine<K>(rp, bp, bn, ep, en, ebits, mp, n, k, tbl, acc, tmp, work);
399 }
