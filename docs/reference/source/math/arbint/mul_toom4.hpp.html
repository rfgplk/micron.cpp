16 // Toom-Cook 4
17 // seven 1/4-size products instead of sixteen, O(n^1.404)
18 //
19 //     points        0     1    -1     2    -2    1/2    inf
20 //     A(x)B(x)     w0    w2    w3    w4    w1     w5     w6

22 // NOTE: rp distinct from ap and bp, holds an + bn limbs; balanced only; an == bn == n

31 [[nodiscard, gnu::always_inline]] inline constexpr limb_t
32 binv_odd(limb_t d) noexcept
33 {
34 
35   limb_t x = d;
36   for ( usize i = 0; i < 6u; ++i ) x = static_cast<limb_t>(x * static_cast<limb_t>(limb_t{ 2 } - static_cast<limb_t>(d * x)));
37   return x;
38 }
39 
40 inline constexpr limb_t inv9 = binv_odd(9u);
41 inline constexpr limb_t inv15 = binv_odd(15u);
42 
43 static_assert(binv_odd(3u) == inv3, "arbint: the general binary inverse must agree with the hand-written 3^-1");
44 static_assert(static_cast<limb_t>(9u * inv9) == 1u, "arbint: 9 * 9^-1 must be 1 mod 2^limb_bits");
45 static_assert(static_cast<limb_t>(15u * inv15) == 1u, "arbint: 15 * 15^-1 must be 1 mod 2^limb_bits");

47 [[gnu::flatten]] inline constexpr void
48 divexact_1_odd(limb_t *rp, const limb_t *up, usize n, limb_t d, limb_t dinv) noexcept
49 {
50   limb_t cy = 0;
51   for ( usize i = 0; i < n; ++i ) {
52     limb_t s = 0;
53     const limb_t bw = subb(up[i], cy, 0, s);
54     const limb_t q = static_cast<limb_t>(s * dinv);
55     rp[i] = q;
56     limb_t lo = 0, hi = 0;
57     mul_wide(q, d, lo, hi);
58     cy = static_cast<limb_t>(hi + bw);
59   }
60 }

74 [[gnu::flatten]] inline constexpr void
75 rsh1add_n(limb_t *rp, const limb_t *up, const limb_t *vp, usize n) noexcept
76 {
77   const limb_t cy = add_n(rp, up, vp, n);
78   (void)rshift(rp, rp, n, 1);
79   rp[n - 1u] = static_cast<limb_t>(rp[n - 1u] | (cy << (limb_bits - 1u)));

82 [[nodiscard, gnu::flatten]] inline constexpr bool
83 toom4_eval_pm1(limb_t *xp1, limb_t *xm1, limb_t *tp, const limb_t *x0, const limb_t *x1, const limb_t *x2, const limb_t *x3, usize k,
84                usize s) noexcept
85 {
86   xp1[k] = add_n(xp1, x0, x2, k);
87   tp[k] = add(tp, x1, k, x3, s);
88   const bool neg = cmp(xp1, tp, k + 1u) < 0;
89   if ( neg )
90     (void)sub_n(xm1, tp, xp1, k + 1u);
91   else
92     (void)sub_n(xm1, xp1, tp, k + 1u);
93   (void)add_n(xp1, xp1, tp, k + 1u);
94   return neg;
95 }

97 [[nodiscard, gnu::flatten]] inline constexpr bool
98 toom4_eval_pm2(limb_t *xp2, limb_t *xm2, limb_t *tp, const limb_t *x0, const limb_t *x1, const limb_t *x2, const limb_t *x3, usize k,
99                usize s) noexcept
100 {
101   limb_t cy = lshift(xp2, x2, k, 2);
102   xp2[k] = static_cast<limb_t>(cy + add_n(xp2, xp2, x0, k));
103 
104   zero(tp, k + 1u);
105   copyi(tp, x3, s);
106   cy = lshift(tp, tp, k, 2);
107   tp[k] = static_cast<limb_t>(cy + add_n(tp, tp, x1, k));
108   cy = lshift(tp, tp, k, 1);
109   tp[k] = static_cast<limb_t>((tp[k] << 1) | cy);
110 
111   const bool neg = cmp(xp2, tp, k + 1u) < 0;
112   if ( neg )
113     (void)sub_n(xm2, tp, xp2, k + 1u);
114   else
115     (void)sub_n(xm2, xp2, tp, k + 1u);
116   (void)add_n(xp2, xp2, tp, k + 1u);
117   return neg;
118 }

120 [[gnu::flatten]] inline constexpr void
121 toom4_eval_half(limb_t *xh, const limb_t *x0, const limb_t *x1, const limb_t *x2, const limb_t *x3, usize k, usize s) noexcept
122 {
123   limb_t c = lshift(xh, x0, k, 1);
124   xh[k] = static_cast<limb_t>(c + add_n(xh, xh, x1, k));
125   c = lshift(xh, xh, k, 1);
126   xh[k] = static_cast<limb_t>((xh[k] << 1) | c);
127   xh[k] = static_cast<limb_t>(xh[k] + add_n(xh, xh, x2, k));
128   c = lshift(xh, xh, k, 1);
129   xh[k] = static_cast<limb_t>((xh[k] << 1) | c);
130   xh[k] = static_cast<limb_t>(xh[k] + add(xh, xh, k, x3, s));
131 }

133 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
134 // seven point interpolation
135 //
136 // Bodrato's sequence
137 //     W5 = W5 + W4          W5 = W5 - W2*65      W1 = W5 - W1
138 //     W1 = (W4 - W1)/2      W2 = W2 - W6 - W0    W5 = (W5 - W3*8)/9
139 //     W4 = W4 - W0          W5 = (W5 + W2*45)/2  W3 = W3 - W5
140 //     W4 = (W4 - W1)/4      W4 = (W4 - W2)/3     W1 = (W1/15 + W5)/2
141 //          - W6*16          W2 = W2 - W4         W5 = W5 - W1
142 //     W3 = (W2 - W3)/2
143 //     W2 = W2 - W3
144 // on exit the seven buffers hold c0..c6 and rp holds their sum at strides of k limbs
145 inline constexpr void
146 toom_interpolate_7pts(limb_t *__restrict__ rp, limb_t *w0, limb_t *w1, limb_t *w2, limb_t *w3, limb_t *w4, limb_t *w5, limb_t *w6,
147                       limb_t *tp, usize k, usize n, usize s, usize m, bool w1_neg, bool w3_neg) noexcept
148 {
149   (void)add_n(w5, w5, w4, m);
150 
151   if ( w1_neg )
152     rsh1add_n(w1, w1, w4, m);
153   else {
154     (void)sub_n(w1, w4, w1, m);
155     (void)rshift(w1, w1, m, 1);
156   }
157 
158   (void)sub_n(w4, w4, w0, m);
159   (void)sub_n(w4, w4, w1, m);
160   (void)rshift(w4, w4, m, 2);
161   (void)lshift(tp, w6, m, 4);
162   (void)sub_n(w4, w4, tp, m);
163 
164   if ( w3_neg )
165     rsh1add_n(w3, w3, w2, m);
166   else {
167     (void)sub_n(w3, w2, w3, m);
168     (void)rshift(w3, w3, m, 1);
169   }
170 
171   (void)sub_n(w2, w2, w3, m);
172 
173   (void)submul_1(w5, w2, m, 65u);
174   (void)sub_n(w2, w2, w6, m);
175   (void)sub_n(w2, w2, w0, m);
176   (void)addmul_1(w5, w2, m, 45u);
177   (void)rshift(w5, w5, m, 1);
178 
179   (void)sub_n(w4, w4, w2, m);
180   divexact_by3(w4, w4, m);
181   (void)sub_n(w2, w2, w4, m);
182 
183   (void)sub_n(w1, w5, w1, m);
184 
185   (void)lshift(tp, w3, m, 3);
186   (void)sub_n(w5, w5, tp, m);
187   divexact_by9(w5, w5, m);
188   (void)sub_n(w3, w3, w5, m);
189 
190   divexact_by15(w1, w1, m);
191   rsh1add_n(w1, w1, w5, m);
192   w1[m - 1u] = static_cast<limb_t>(w1[m - 1u] & (limb_max >> 1));
193   (void)sub_n(w5, w5, w1, m);
194 
195   const usize rn = 2u * n;
196   zero(rp, rn);
197   copyi(rp, w0, 2u * k);
198   (void)add(rp + k, rp + k, rn - k, w1, normalize(w1, m));
199   (void)add(rp + 2u * k, rp + 2u * k, rn - 2u * k, w2, normalize(w2, m));
200   (void)add(rp + 3u * k, rp + 3u * k, rn - 3u * k, w3, normalize(w3, m));
201   (void)add(rp + 4u * k, rp + 4u * k, rn - 4u * k, w4, normalize(w4, m));
202   (void)add(rp + 5u * k, rp + 5u * k, rn - 5u * k, w5, normalize(w5, m));
203   (void)add(rp + 6u * k, rp + 6u * k, rn - 6u * k, w6, normalize(w6, 2u * s));
204 }

206 [[nodiscard, gnu::always_inline]] inline constexpr bool
207 toom4_split_ok(usize n) noexcept
208 {
209   return n >= 4u && n > 3u * ((n + 3u) / 4u);

215 [[nodiscard, gnu::flatten]] inline constexpr usize
216 toom4_itch(usize n, bool force_top = false) noexcept
217 {
218   usize total = 0;
219   if ( force_top && n >= 4u && n < threshold::mul_toom4 ) {
220     const usize k = (n + 3u) / 4u;
221     total += 11u * (k + 1u) + 8u * (2u * k + 2u);
222     n = k + 1u;
223   }
224   while ( n >= threshold::mul_toom4 && n >= 4u ) {
225     const usize k = (n + 3u) / 4u;
226     total += 11u * (k + 1u) + 8u * (2u * k + 2u);
227     n = k + 1u;
228   }
229   return total + toom3_itch(n);
230 }

232 [[nodiscard, gnu::flatten]] inline constexpr usize
233 sqr_toom4_itch(usize n, bool force_top = false) noexcept
234 {
235   usize total = 0;
236   if ( force_top && n >= 4u && n < threshold::sqr_toom4 ) {
237     const usize k = (n + 3u) / 4u;
238     total += 6u * (k + 1u) + 8u * (2u * k + 2u);
239     n = k + 1u;
240   }
241   while ( n >= threshold::sqr_toom4 && n >= 4u ) {
242     const usize k = (n + 3u) / 4u;
243     total += 6u * (k + 1u) + 8u * (2u * k + 2u);
244     n = k + 1u;
245   }
246   return total + sqr_toom3_itch(n);
247 }

275 inline constexpr void
276 mul_toom4(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, const limb_t *__restrict__ bp, usize n, limb_t *scratch,
277           usize cutoff) noexcept
278 {
279   const usize k = (n + 3u) / 4u;
280   const usize s = n - 3u * k;      // 1 <= s <= k
281   const usize m = 2u * k + 2u;
282   const usize e = k + 1u;
283 
284   limb_t *const ap1 = scratch;
285   limb_t *const am1 = ap1 + e;
286   limb_t *const ap2 = am1 + e;
287   limb_t *const am2 = ap2 + e;
288   limb_t *const ah = am2 + e;
289   limb_t *const bp1 = ah + e;
290   limb_t *const bm1 = bp1 + e;
291   limb_t *const bp2 = bm1 + e;
292   limb_t *const bm2 = bp2 + e;
293   limb_t *const bh = bm2 + e;
294   limb_t *const et = bh + e;
295   limb_t *const w0 = et + e;
296   limb_t *const w1 = w0 + m;
297   limb_t *const w2 = w1 + m;
298   limb_t *const w3 = w2 + m;
299   limb_t *const w4 = w3 + m;
300   limb_t *const w5 = w4 + m;
301   limb_t *const w6 = w5 + m;
302   limb_t *const tp = w6 + m;
303   limb_t *const rec = tp + m;
304 
305   const limb_t *const a0 = ap;
306   const limb_t *const a1 = ap + k;
307   const limb_t *const a2 = ap + 2u * k;
308   const limb_t *const a3 = ap + 3u * k;
309   const limb_t *const b0 = bp;
310   const limb_t *const b1 = bp + k;
311   const limb_t *const b2 = bp + 2u * k;
312   const limb_t *const b3 = bp + 3u * k;
313 
314   const bool an1 = toom4_eval_pm1(ap1, am1, et, a0, a1, a2, a3, k, s);
315   const bool bn1 = toom4_eval_pm1(bp1, bm1, et, b0, b1, b2, b3, k, s);
316   const bool an2 = toom4_eval_pm2(ap2, am2, et, a0, a1, a2, a3, k, s);
317   const bool bn2 = toom4_eval_pm2(bp2, bm2, et, b0, b1, b2, b3, k, s);
318   toom4_eval_half(ah, a0, a1, a2, a3, k, s);
319   toom4_eval_half(bh, b0, b1, b2, b3, k, s);
320 
321   __mul_rec4(w2, ap1, e, bp1, e, rec, cutoff);      // f(1)
322   __mul_rec4(w3, am1, e, bm1, e, rec, cutoff);      // |f(-1)|
323   __mul_rec4(w4, ap2, e, bp2, e, rec, cutoff);      // f(2)
324   __mul_rec4(w1, am2, e, bm2, e, rec, cutoff);      // |f(-2)|
325   __mul_rec4(w5, ah, e, bh, e, rec, cutoff);        // 64 * f(1/2)
326   __mul_rec4(w0, a0, k, b0, k, rec, cutoff);        // f(0)
327   zero(w0 + 2u * k, m - 2u * k);
328   __mul_rec4(w6, a3, s, b3, s, rec, cutoff);      // f(inf)
329   zero(w6 + 2u * s, m - 2u * s);
330 
331   toom_interpolate_7pts(rp, w0, w1, w2, w3, w4, w5, w6, tp, k, n, s, m, an2 != bn2, an1 != bn1);
332 }

334 inline constexpr void
335 sqr_toom4(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch, usize cutoff) noexcept
336 {
337   const usize k = (n + 3u) / 4u;
338   const usize s = n - 3u * k;
339   const usize m = 2u * k + 2u;
340   const usize e = k + 1u;
341 
342   limb_t *const ap1 = scratch;
343   limb_t *const am1 = ap1 + e;
344   limb_t *const ap2 = am1 + e;
345   limb_t *const am2 = ap2 + e;
346   limb_t *const ah = am2 + e;
347   limb_t *const et = ah + e;
348   limb_t *const w0 = et + e;
349   limb_t *const w1 = w0 + m;
350   limb_t *const w2 = w1 + m;
351   limb_t *const w3 = w2 + m;
352   limb_t *const w4 = w3 + m;
353   limb_t *const w5 = w4 + m;
354   limb_t *const w6 = w5 + m;
355   limb_t *const tp = w6 + m;
356   limb_t *const rec = tp + m;
357 
358   const limb_t *const a0 = ap;
359   const limb_t *const a1 = ap + k;
360   const limb_t *const a2 = ap + 2u * k;
361   const limb_t *const a3 = ap + 3u * k;
362 
363   (void)toom4_eval_pm1(ap1, am1, et, a0, a1, a2, a3, k, s);
364   (void)toom4_eval_pm2(ap2, am2, et, a0, a1, a2, a3, k, s);
365   toom4_eval_half(ah, a0, a1, a2, a3, k, s);
366 
367   __sqr_rec4(w2, ap1, e, rec, cutoff);
368   __sqr_rec4(w3, am1, e, rec, cutoff);
369   __sqr_rec4(w4, ap2, e, rec, cutoff);
370   __sqr_rec4(w1, am2, e, rec, cutoff);
371   __sqr_rec4(w5, ah, e, rec, cutoff);
372   __sqr_rec4(w0, a0, k, rec, cutoff);
373   zero(w0 + 2u * k, m - 2u * k);
374   __sqr_rec4(w6, a3, s, rec, cutoff);
375   zero(w6 + 2u * s, m - 2u * s);
376 
377   toom_interpolate_7pts(rp, w0, w1, w2, w3, w4, w5, w6, tp, k, n, s, m, false, false);
378 }
