17 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
18 // Barrett division
19 // (precomputed n-limb reciprocal into a multiply)
20 //
21 // ..invert_limb(d)       one limb      floor((B^2 - 1)/d) - B
22 // ..invert_pi1(d1, d0)   two limbs     3-by-2 estimator
23 // ..invert_n(ip, dp, n)  n limbs       floor((B^2n - 1)/D) - B^n
24 //
25 // dc_div_qr_n's recurrence D(n) = 2*D(n/2) + 2*M(n/2) solves to about 4.4*M(n) for n quotient limbs against an n-limb divisor
26 //
27 // TODO: convert estimate via mullo/low-half mul
28 //
29 // WARNING: never gnu::flatten on anything calling mul(), sqr() or divrem(); comptimes _explode_

41 inline constexpr void
42 invert_n_basecase(limb_t *ip, const limb_t *dp, usize n, limb_t *scratch) noexcept
43 {
44   if ( n == 1 ) {
45     ip[0] = invert_limb(dp[0]);
46     return;
47   }
48 
49   limb_t *const np = scratch;      // 2n
50   for ( usize i = 0; i < 2u * n; ++i ) np[i] = limb_max;
51 
52   (void)sbpi1_div_qr(ip, np, 2u * n, dp, n, invert_pi1(dp[n - 1], dp[n - 2]));
53 }
54 
55 [[nodiscard, gnu::always_inline]] inline constexpr usize
56 invert_n_basecase_itch(usize n) noexcept
57 {
58   return n == 1 ? 0u : 2u * n;
59 }

75 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
76 // the exact fixup
77 //
78 // {pp, 2n} and {rp, 2n} are scratch; ip is corrected in place
79 
80 inline constexpr void
81 __invert_n_fixup(limb_t *ip, const limb_t *dp, usize n, limb_t *pp, limb_t *rp, limb_t *work) noexcept
82 {
83   // P = D * (B^n + V) = D*V + D*B^n
84   mul(pp, dp, n, ip, n, work);
85   limb_t cy = add_n(pp + n, pp + n, dp, n);
86 
87   // P at or above B^2n means V overshot
88   while ( cy != 0 ) {
89     __micron_arbint_mu_audit(inv_fixup_down);
90     (void)sub_1(ip, ip, n, 1);
91     cy = static_cast<limb_t>(cy - sub(pp, pp, 2u * n, dp, n));
92   }
93 
94   // R = B^2n - P
95   (void)neg(rp, pp, 2u * n);
96   for ( ;; ) {
97     if ( is_zero(rp + n, n) && cmp(rp, dp, n) <= 0 ) break;
98     __micron_arbint_mu_audit(inv_fixup_up);
99     (void)add_1(ip, ip, n, 1);
100     (void)sub(rp, rp, 2u * n, dp, n);
101   }
102 }

104 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
105 // Newton
106 //
107 // one step costs roughly
108 // T(n) = T(n/2) + O(M(n)) = O(M(n))
109 //
110 // with h = ceil(n/2), l = n - h and D_h the top h limbs of D:
111 //
112 // B^2n / D  ~=  B^l * (B^2h / D_h)
113 
114 inline constexpr void
115 invert_n(limb_t *ip, const limb_t *dp, usize n, limb_t *scratch) noexcept
116 {
117   if ( n < threshold::inv_newton ) {
118     invert_n_basecase(ip, dp, n, scratch);
119     return;
120   }
121 
122   const usize h = (n + 1u) / 2u;
123   const usize l = n - h;
124 
125   invert_n(ip + l, dp + l, h, scratch);
126   zero(ip, l);
127 
128   limb_t *const ep = scratch;               // 2n + 1
129   limb_t *const wp = ep + 2u * n + 1u;      // n + h
130   limb_t *const xp = wp + n + h;            // n + 1
131   limb_t *const pp = xp + n + 1u;           // 3n + 2
132   limb_t *const work = pp + 3u * n + 2u;
133 
134   mul(wp, dp, n, ip + l, h, work);      // W = D * V_h; n >= h, so the order is legal
135 
136   zero(ep, n);
137   (void)neg(ep + n, dp, n);      // C = B^n - D, in the top half
138   const limb_t bw = sub_n(ep + l, ep + l, wp, n + h);
139   const bool eneg = (bw != 0);
140   if ( eneg ) (void)neg(ep, ep, 2u * n);
141   const usize en = normalize(ep, 2u * n);
142 
143   if ( en != 0 ) {
144     copyi(xp, ip, n);
145     xp[n] = 1;      // X_0 = B^n + V, n + 1 limbs
146     const usize xn = n + 1u;
147     if ( xn >= en )
148       mul(pp, xp, xn, ep, en, work);
149     else
150       mul(pp, ep, en, xp, xn, work);
151 
152     const usize pn = xn + en;
153     if ( pn > 2u * n ) {
154       usize dl = pn - 2u * n;
155       dl = normalize(pp + 2u * n, dl);
156       if ( dl > n ) dl = n;
157       if ( dl != 0 ) {
158         if ( eneg )
159           (void)sub(ip, ip, n, pp + 2u * n, dl);
160         else
161           (void)add(ip, ip, n, pp + 2u * n, dl);
162       }
163     }
164   }
165 
166   __invert_n_fixup(ip, dp, n, pp, ep, work);
167 }

169 [[nodiscard, gnu::flatten]] inline constexpr usize
170 invert_n_itch(usize n) noexcept
171 {
172   if ( n < 2u ) return 0;
173   usize best = 0;
174   usize m = n;
175   while ( m >= threshold::inv_newton ) {
176     const usize h = (m + 1u) / 2u;
177     const usize lev = (2u * m + 1u) + (m + h) + (m + 1u) + (3u * m + 2u);
178     usize w = mul_itch(m, h);
179     const usize w2 = __mul_itch_bound(2u * m + 1u, m + 1u);
180     if ( w2 > w ) w = w2;
181     const usize w3 = mul_itch(m, m);
182     if ( w3 > w ) w = w3;
183     if ( lev + w > best ) best = lev + w;
184     m = h;
185   }
186   const usize base = invert_n_basecase_itch(m);
187   return base > best ? base : best;
188 }

193 [[nodiscard, gnu::always_inline]] inline constexpr usize
194 mu_block_size(usize qn, usize dn) noexcept
195 {
196   if ( qn == 0 || dn < 2u ) return 2u;
197   const usize blocks = (qn - 1u) / dn + 1u;      // ceil(qn / dn)
198   usize in = (qn - 1u) / blocks + 1u;            // ceil(qn / blocks), which is <= dn
199   if ( in < 2u ) in = 2u;
200   if ( in > dn ) in = dn;
201   return in;
202 }

204 // %%%%%%%%%%%%%%%%%%%%
205 // one block
206 
207 inline constexpr void
208 mu_div_qr_block(limb_t *qp, limb_t *np, const limb_t *dp, usize dn, const limb_t *ip, usize in, limb_t *tp, limb_t *work) noexcept
209 {
210   const limb_t *const u1 = np + dn;
211 
212   mul(tp, u1, in, ip, in, work);
213   limb_t cy = add_n(qp, tp + in, u1, in);
214   cy = static_cast<limb_t>(cy + add_1(qp, qp, in, 1));
215 
216   // q^ ran past B^in - 1
217   if ( cy != 0 ) [[unlikely]] {
218     __micron_arbint_mu_audit(mu_saturations);
219     for ( usize i = 0; i < in; ++i ) qp[i] = limb_max;
220   }
221 
222   mul(tp, dp, dn, qp, in, work);
223   cy = sub_n(np, np, tp, dn + in);
224 
225   // q^ was too big
226   while ( cy != 0 ) {
227     __micron_arbint_mu_audit(mu_corrections_down);
228     (void)sub_1(qp, qp, in, 1);
229     cy = static_cast<limb_t>(cy - add(np, np, dn + in, dp, dn));
230   }
231 
232   // q^ was too small
233   for ( ;; ) {
234     if ( is_zero(np + dn, in) && cmp(np, dp, dn) < 0 ) break;
235     __micron_arbint_mu_audit(mu_corrections_up);
236     (void)add_1(qp, qp, in, 1);
237     (void)sub(np, np, dn + in, dp, dn);
238   }
239 }

244 [[nodiscard]] inline constexpr limb_t
245 mu_div_qr(limb_t *qp, limb_t *np, usize nn, const limb_t *dp, usize dn, limb_t *scratch) noexcept
246 {
247   const usize qn = nn - dn;
248   const usize in = mu_block_size(qn, dn);
249 
250   limb_t *const ip = scratch;             // in
251   limb_t *const tp = ip + in;             // dn + in
252   limb_t *const work = tp + dn + in;      // max(invert_n_itch(in), mul_itch(dn, in), dc_div_itch(dn))
253 
254   invert_n(ip, dp + dn - in, in, work);
255 
256   const limb_t qh = static_cast<limb_t>(cmp(np + qn, dp, dn) >= 0);
257   if ( qh != 0 ) (void)sub_n(np + qn, np + qn, dp, dn);
258 
259   const usize r = qn % in;
260   for ( usize j = qn - r; j >= in; ) {
261     j -= in;
262     mu_div_qr_block(qp + r + j, np + r + j, dp, dn, ip, in, tp, work);
263   }
264 
265   if ( r != 0 ) {
266     const limb_t dinv = invert_pi1(dp[dn - 1], dp[dn - 2]);
267     if ( div_wants_dc(r + dn, dn) )
268       (void)dc_div_qr(qp, np, r + dn, dp, dn, dinv, work);
269     else
270       (void)sbpi1_div_qr(qp, np, r + dn, dp, dn, dinv);
271   }
272   return qh;
273 }

275 [[nodiscard, gnu::flatten]] inline constexpr usize
276 mu_div_qr_itch(usize nn, usize dn) noexcept
277 {
278   if ( nn <= dn || dn < 2u ) return 0;
279   const usize qn = nn - dn;
280   const usize in = mu_block_size(qn, dn);
281 
282   usize w = invert_n_itch(in);
283   const usize m = mul_itch(dn, in);
284   if ( m > w ) w = m;
285   const usize e = mul_itch(in, in);
286   if ( e > w ) w = e;
287   if ( (qn % in) != 0 && div_wants_dc(qn % in + dn, dn) ) {
288     const usize t = dc_div_itch(dn);
289     if ( t > w ) w = t;
290   }
291   return in + (dn + in) + w;
292 }

294 [[nodiscard, gnu::always_inline]] inline constexpr bool
295 div_wants_mu(usize nn, usize dn) noexcept
296 {
297   return static_cast<u8>(clamp_to(pick_div(nn, dn), div_tier_cap)) >= static_cast<u8>(divalgo::mu) && dn >= 2u && nn > dn;
298 }

305 [[nodiscard, gnu::flatten]] inline constexpr usize
306 divrem_itch(usize nn, usize dn) noexcept
307 {
308   if ( dn <= 1 ) return 0;
309   usize t = nn + 1u + dn;
310   if ( div_wants_mu(nn + 1u, dn) )
311     t += mu_div_qr_itch(nn + 1u, dn);
312   else if ( div_wants_dc(nn + 1u, dn) )
313     t += dc_div_itch(dn);
314   return t;
315 }

317 // Bounds every division with numerator <= nn and divisor <= dn. Reducing the
318 // divisor can change the Barrett block size and enter a larger FFT padding band.
319 [[nodiscard]] inline constexpr usize
320 divrem_cap_itch(usize nn, usize dn) noexcept
321 {
322   if ( dn > nn ) dn = nn;
323   if ( dn <= 1u ) return 0u;
324   usize work = 0u;
325   if constexpr ( MICRON_ARBINT_DIV_TIER_CAP >= 1 )
326     if ( dn >= threshold::div_dc ) work = dc_div_itch(dn);
327   if constexpr ( MICRON_ARBINT_DIV_TIER_CAP >= 2 ) {
328     if ( dn >= threshold::div_mu ) {
329       usize inner = invert_n_itch(dn);
330       inner = __mul_itch_max(inner, mul_solver_cap_itch<solver::automatic>(dn));
331       inner = __mul_itch_max(inner, work);
332       work = __mul_itch_max(work, 3u * dn + inner);
333     }
334   }
335   return nn + 1u + dn + work;
336 }

338 // qp: nn - dn + 1 limbs, rp: dn limbs, scratch: divrem_itch(nn, dn) limbs
339 // requires nn >= dn >= 1 and {dp, dn} normalized in the sense that dp[dn-1] != 0
340 inline constexpr void
341 divrem(limb_t *qp, limb_t *rp, const limb_t *np, usize nn, const limb_t *dp, usize dn, limb_t *scratch) noexcept
342 {
343   if ( dn == 1 ) {
344     rp[0] = divrem_1(qp, np, nn, dp[0]);
345     return;
346   }
347 
348   const usize sh = limb_clz(dp[dn - 1]);
349   limb_t *const dwork = scratch;
350   limb_t *const nwork = scratch + dn;
351 
352   if ( sh != 0 ) {
353     (void)lshift(dwork, dp, dn, sh);
354     nwork[nn] = lshift(nwork, np, nn, sh);
355   } else {
356     copyi(dwork, dp, dn);
357     copyi(nwork, np, nn);
358     nwork[nn] = 0;
359   }
360 
361   if ( div_wants_mu(nn + 1u, dn) ) {
362     (void)mu_div_qr(qp, nwork, nn + 1u, dwork, dn, nwork + nn + 1u);
363   } else {
364     const limb_t dinv = invert_pi1(dwork[dn - 1], dwork[dn - 2]);
365     if ( div_wants_dc(nn + 1u, dn) )
366       (void)dc_div_qr(qp, nwork, nn + 1u, dwork, dn, dinv, nwork + nn + 1u);
367     else
368       (void)sbpi1_div_qr(qp, nwork, nn + 1u, dwork, dn, dinv);
369   }
370 
371   if ( sh != 0 )
372     (void)rshift(rp, nwork, dn, sh);
373   else
374     copyi(rp, nwork, dn);
375 }

378 // pinned tiers

380 template<divalgo K>
381 inline constexpr void
382 divrem_with(limb_t *qp, limb_t *rp, const limb_t *np, usize nn, const limb_t *dp, usize dn, limb_t *scratch) noexcept
383 {
384   static_assert(static_cast<u8>(K) <= static_cast<u8>(div_tiers_built), "arbint: this division tier is not implemented yet");
385   if ( dn == 1 ) {
386     rp[0] = divrem_1(qp, np, nn, dp[0]);
387     return;
388   }
389 
390   const usize sh = limb_clz(dp[dn - 1]);
391   limb_t *const dwork = scratch;
392   limb_t *const nwork = scratch + dn;
393 
394   if ( sh != 0 ) {
395     (void)lshift(dwork, dp, dn, sh);
396     nwork[nn] = lshift(nwork, np, nn, sh);
397   } else {
398     copyi(dwork, dp, dn);
399     copyi(nwork, np, nn);
400     nwork[nn] = 0;
401   }
402 
403   limb_t *const tier = nwork + nn + 1u;
404   bool done = false;
405   if constexpr ( K == divalgo::mu ) {
406     if ( nn + 1u > dn ) {
407       (void)mu_div_qr(qp, nwork, nn + 1u, dwork, dn, tier);
408       done = true;
409     }
410   }
411   if ( !done ) {
412     const limb_t dinv = invert_pi1(dwork[dn - 1], dwork[dn - 2]);
413     if constexpr ( K != divalgo::sbpi1 ) {
414       if ( dn >= 6u && nn + 1u > dn ) {
415         (void)dc_div_qr(qp, nwork, nn + 1u, dwork, dn, dinv, tier);
416         done = true;
417       }
418     }
419     if ( !done ) (void)sbpi1_div_qr(qp, nwork, nn + 1u, dwork, dn, dinv);
420   }
421 
422   if ( sh != 0 )
423     (void)rshift(rp, nwork, dn, sh);
424   else
425     copyi(rp, nwork, dn);
426 }

428 template<divalgo K>
429 [[nodiscard, gnu::flatten]] inline constexpr usize
430 div_itch_with(usize nn, usize dn) noexcept
431 {
432   if ( dn <= 1 ) return 0;
433   usize t = nn + 1u + dn;
434   if constexpr ( K == divalgo::mu ) {
435     if ( nn + 1u > dn ) return t + mu_div_qr_itch(nn + 1u, dn);
436   }
437   if constexpr ( K != divalgo::sbpi1 ) {
438     if ( dn >= 6u && nn + 1u > dn ) return t + dc_div_itch(dn);
439   }
440   return t;
441 }
