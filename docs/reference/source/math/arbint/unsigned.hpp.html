155   [[nodiscard]] explicit constexpr
156   operator I() const noexcept
157   {
158     using U = micron::make_unsigned_t<I>;
159     U acc = 0;
160     const usize take = (sizeof(U) + sizeof(mpn::limb_t) - 1u) / sizeof(mpn::limb_t);
161     const usize n = s.size() < take ? s.size() : take;

203   constexpr arbuint &
204   operator+=(const arbuint &o)
205   {
206     const usize an = s.size();
207     const usize bn = o.s.size();
208     if ( bn == 0 ) return *this;
209     const usize mx = an > bn ? an : bn;
210     const usize mn = an > bn ? bn : an;
211 
212     (void)s.ensure(mx + 1u);
213 
214     mpn::limb_t *r = s.limbs();
215     const mpn::limb_t *b = o.s.limbs();
216     const usize cap = s.capacity();
217 
218     mpn::limb_t cy = mpn::add_n(r, r, b, mn);
219     if ( an > bn )
220       cy = mpn::add_1(r + mn, r + mn, an - mn, cy);
221     else if ( bn > an )
222       cy = mpn::add_1(r + mn, b + mn, bn - mn, cy);
223 
224     usize n = mx;
225     if ( cy != 0 && n < cap ) {
226       r[n] = cy;
227       ++n;
228     }
229     s.finish(n);
230     return *this;
231   }

233   constexpr arbuint &
234   operator-=(const arbuint &o)
235   {
236     const usize an = s.size();
237     const usize bn = o.s.size();
238     if ( bn == 0 ) return *this;
239 
240     const int c = mpn::cmp_var(s.limbs(), an, o.s.limbs(), bn);
241     if ( c == 0 ) {
242       s.finish(0);
243       return *this;
244     }
245     if ( c > 0 ) {
246       (void)mpn::sub(s.limbs(), s.limbs(), an, o.s.limbs(), bn);
247       s.finish(an);
248       return *this;
249     }
250 
251     if constexpr ( !bounded ) {
252       s.finish(0);
253       return *this;
254     } else {
255 
256       (void)s.ensure(cap_limbs);
257       mpn::limb_t *r = s.limbs();
258       mpn::zero(r + an, cap_limbs - an);
259       mpn::limb_t ext[cap_limbs] = {};
260       mpn::copyi(ext, o.s.limbs(), bn);
261       (void)mpn::sub_n(r, r, ext, cap_limbs);
262       s.finish(cap_limbs);
263       return *this;
264     }
265   }

267   constexpr arbuint &
268   operator*=(const arbuint &o)
269   {
270     const usize an = s.size();
271     const usize bn = o.s.size();
272     if ( an == 0 || bn == 0 ) {
273       s.finish(0);
274       return *this;
275     }
276     if ( &o == this ) return __square();
277 
278     const usize rn = an + bn;
279     const usize itch = an >= bn ? mpn::mul_solver_itch<Solver>(an, bn) : mpn::mul_solver_itch<Solver>(bn, an);
280     scratch_for<mul_scratch_limbs> sc(rn + itch);
281     mpn::limb_t *t = sc.get();
282     mpn::limb_t *work = t + rn;
283 
284     if ( an >= bn )
285       __mul_into(t, s.limbs(), an, o.s.limbs(), bn, work);
286     else
287       __mul_into(t, o.s.limbs(), bn, s.limbs(), an, work);
288 
289     (void)s.ensure(rn);
290     const usize cap = s.capacity();
291     mpn::copyi(s.limbs(), t, rn < cap ? rn : cap);
292     s.finish(rn);
293     return *this;

310   constexpr arbuint &
311   operator<<=(usize k)
312   {
313     const usize an = s.size();
314     if ( an == 0 || k == 0 ) return *this;
315     const usize wh = k / mpn::limb_bits;
316     const usize bi = k % mpn::limb_bits;
317 
318     (void)s.ensure(an + wh + 1u);
319     const usize cap = s.capacity();
320     const usize room = (an + wh + 1u) < cap ? (an + wh + 1u) : cap;
321     if ( room <= wh ) {
322       s.finish(0);
323       return *this;
324     }
325     const usize keep = room - wh;
326     const usize src = keep < an ? keep : an;
327 
328     mpn::limb_t *r = s.limbs();
329     usize n = wh + src;
330     if ( bi != 0 ) {
331       const mpn::limb_t top = mpn::lshift(r + wh, r, src, bi);
332       if ( top != 0 && n < room ) {
333         r[n] = top;
334         ++n;
335       }
336     } else {
337       mpn::copyd(r + wh, r, src);
338     }
339     mpn::zero(r, wh);
340     s.finish(n);
341     return *this;
342   }

344   constexpr arbuint &
345   operator>>=(usize k)
346   {
347     const usize an = s.size();
348     if ( an == 0 || k == 0 ) return *this;
349     const usize wh = k / mpn::limb_bits;
350     const usize bi = k % mpn::limb_bits;
351     if ( wh >= an ) {
352       s.finish(0);
353       return *this;
354     }
355     const usize keep = an - wh;
356     mpn::limb_t *r = s.limbs();
357     if ( bi != 0 )
358       (void)mpn::rshift(r, r + wh, keep, bi);
359     else
360       mpn::copyi(r, r + wh, keep);
361     s.finish(keep);
362     return *this;
363   }

440   template<micron::integral I>
441   constexpr void
442   __assign_integral(I v)
443   {
444     using U = micron::make_unsigned_t<I>;
445     U u = static_cast<U>(v);
446     if ( u == 0 ) {
447       s.finish(0);
448       return;
449     }
450     constexpr usize take = (sizeof(U) + sizeof(mpn::limb_t) - 1u) / sizeof(mpn::limb_t);
451     if ( !s.ensure(take) ) {
452 
453       (void)s.ensure(cap_limbs);
454     }
455     const usize cap = s.capacity();
456     const usize lim = take < cap ? take : cap;
457     mpn::limb_t *r = s.limbs();
458     for ( usize i = 0; i < lim; ++i ) {
459       r[i] = static_cast<mpn::limb_t>(u);
460       if constexpr ( sizeof(U) > sizeof(mpn::limb_t) )
461         u = static_cast<U>(u >> mpn::limb_bits);
462       else
463         u = 0;
464     }
465     s.finish(lim);
466   }

477   constexpr arbuint &
478   __mul_add_1(mpn::limb_t mul, mpn::limb_t add)
479   {
480     const usize n = s.size();
481     (void)s.ensure(n + 1u);
482     const usize cap = s.capacity();
483     mpn::limb_t *r = s.limbs();
484     mpn::limb_t hi = (n != 0) ? mpn::mul_1(r, r, n, mul) : mpn::limb_t{ 0 };
485     if ( n != 0 ) hi = static_cast<mpn::limb_t>(hi + mpn::add_1(r, r, n, add));
486     usize k = n;
487     if ( n == 0 ) {
488       if ( add != 0 && cap > 0 ) {
489         r[0] = add;
490         k = 1;
491       }
492     } else if ( hi != 0 && k < cap ) {
493       r[k] = hi;
494       ++k;
495     }
496     s.finish(k);
497     return *this;
498   }

535   constexpr arbuint &
536   __square()
537   {
538     const usize an = s.size();
539     if ( an == 0 ) {
540       s.finish(0);
541       return *this;
542     }
543     const usize rn = 2u * an;
544     scratch_for<mul_scratch_limbs> sc(rn + mpn::sqr_solver_itch<Solver>(an));
545     mpn::limb_t *t = sc.get();
546     __sqr_into(t, s.limbs(), an, t + rn);
547 
548     (void)s.ensure(rn);
549     const usize cap = s.capacity();
550     mpn::copyi(s.limbs(), t, rn < cap ? rn : cap);
551     s.finish(rn);
552     return *this;
553   }

555   constexpr void
556   __divmod_both(const arbuint &o, arbuint &q)
557   {
558     const usize an = s.size();
559     const usize bn = o.s.size();
560     if ( bn == 0 ) micron::exc<micron::except::domain_error>("micron::math::arbuint division by zero");
561     if ( an < bn ) {
562       q.s.finish(0);
563       return;
564     }
565 
566     const usize qn = an - bn + 1u;
567     const usize itch = mpn::divrem_itch(an, bn);
568     scratch_for<div_scratch_limbs> sc(qn + bn + itch);
569     mpn::limb_t *qp = sc.get();
570     mpn::limb_t *rp = qp + qn;
571     mpn::limb_t *work = rp + bn;
572 
573     mpn::divrem(qp, rp, s.limbs(), an, o.s.limbs(), bn, work);
574 
575     (void)q.s.ensure(qn);
576     const usize qcap = q.s.capacity();
577     mpn::copyi(q.s.limbs(), qp, qn < qcap ? qn : qcap);
578     q.s.finish(qn);
579 
580     (void)s.ensure(bn);
581     const usize rcap = s.capacity();
582     mpn::copyi(s.limbs(), rp, bn < rcap ? bn : rcap);
583     s.finish(bn);
584   }

586   constexpr void
587   __divmod_into(const arbuint &o, bool want_quotient)
588   {
589     const usize an = s.size();
590     const usize bn = o.s.size();
591     if ( bn == 0 ) micron::exc<micron::except::domain_error>("micron::math::arbuint division by zero");
592     if ( an < bn ) {
593       if ( want_quotient ) s.finish(0);
594       return;
595     }
596 
597     const usize qn = an - bn + 1u;
598     const usize itch = mpn::divrem_itch(an, bn);
599     scratch_for<div_scratch_limbs> sc(qn + bn + itch);
600     mpn::limb_t *qp = sc.get();
601     mpn::limb_t *rp = qp + qn;
602     mpn::limb_t *work = rp + bn;
603 
604     mpn::divrem(qp, rp, s.limbs(), an, o.s.limbs(), bn, work);
605 
606     const mpn::limb_t *src = want_quotient ? qp : rp;
607     const usize n = want_quotient ? qn : bn;
608     (void)s.ensure(n);
609     const usize cap = s.capacity();
610     mpn::copyi(s.limbs(), src, n < cap ? n : cap);
611     s.finish(n);
612   }

615   flip_bits() noexcept

617   {
618     (void)s.ensure(cap_limbs);

698 template<usize B, arb_solver S, class A, micron::integral I>
699 [[nodiscard, gnu::flatten]] inline constexpr int
700 cmp(const arbuint<B, S, A> &a, I v) noexcept
701 {
702   using U = micron::make_unsigned_t<I>;
703   constexpr usize take = (sizeof(U) + sizeof(mpn::limb_t) - 1u) / sizeof(mpn::limb_t);
704   U u = static_cast<U>(v);
705   mpn::limb_t tmp[take] = {};
706   for ( usize i = 0; i < take; ++i ) {
707     tmp[i] = static_cast<mpn::limb_t>(u);
708     if constexpr ( sizeof(U) > sizeof(mpn::limb_t) )
709       u = static_cast<U>(u >> mpn::limb_bits);
710     else
711       u = 0;
712   }
713   return mpn::cmp_var(a.limbs(), a.size(), tmp, mpn::normalize(tmp, take));
714 }

769 template<usize B, arb_solver S, class A>
770 [[nodiscard]] inline constexpr arbuint<B, S, A>
771 pow(const arbuint<B, S, A> &a, u64 ex)
772 {
773   if ( ex == 0 ) return arbuint<B, S, A>::one();
774   arbuint<B, S, A> base(a);
775   arbuint<B, S, A> acc = arbuint<B, S, A>::one();
776   while ( ex != 0 ) {
777     if ( (ex & 1u) != 0 ) acc *= base;
778     ex >>= 1;
779     if ( ex != 0 ) base *= base;
780   }
781   return acc;
782 }
