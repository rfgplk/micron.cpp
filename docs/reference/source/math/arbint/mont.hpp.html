18 // Montgomery arithmetic
19 // (multiplication mod m with no division in the loop)
20 //
21 // representation is a*R mod m with R = B^n

30 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%
31 // context
32 
33 // preconditions:
34 // ..mp[0] must be ODD
35 // ..mp[n-1] must be NONZERO
36 struct mont_ctx {
37   const limb_t *mp;      // n limbs, borrowed, must outlive the context
38   usize n;
39   limb_t minv;      // -m^-1 mod B
40 };
41 
42 [[nodiscard, gnu::always_inline]] inline constexpr mont_ctx
43 mont_make(const limb_t *mp, usize n) noexcept
44 {
45   return mont_ctx{ mp, n, static_cast<limb_t>(static_cast<limb_t>(0) - binv_odd(mp[0])) };
46 }

48 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
49 // REDC
50 //
51 // rp = T * R^-1 mod m, for {tp, 2n} holding T < m*R
52 
53 [[gnu::flatten]] inline constexpr void
54 redc(limb_t *rp, limb_t *tp, const mont_ctx &c) noexcept
55 {
56   const usize n = c.n;
57   limb_t carry = 0;
58   for ( usize i = 0; i < n; ++i ) {
59     const limb_t u = static_cast<limb_t>(tp[i] * c.minv);
60     const limb_t cy = addmul_1(tp + i, c.mp, n, u);
61     carry = static_cast<limb_t>(carry + add_1(tp + i + n, tp + i + n, n - i, cy));
62   }
63   const limb_t ge = static_cast<limb_t>(carry != 0 || cmp(tp + n, c.mp, n) >= 0);
64   (void)cnd_sub_n(ge, rp, tp + n, c.mp, n);
65 }

68 redc_itch(usize) noexcept

73 inline constexpr void
74 mont_mul(limb_t *rp, const limb_t *ap, const limb_t *bp, const mont_ctx &c, limb_t *scratch) noexcept
75 {
76   limb_t *const prod = scratch;              // 2n
77   limb_t *const work = prod + 2u * c.n;      // mul_itch(n, n)
78   mul(prod, ap, c.n, bp, c.n, work);
79   redc(rp, prod, c);
80 }

82 inline constexpr void
83 mont_sqr(limb_t *rp, const limb_t *ap, const mont_ctx &c, limb_t *scratch) noexcept
84 {
85   limb_t *const prod = scratch;              // 2n
86   limb_t *const work = prod + 2u * c.n;      // sqr_itch(n)
87   sqr(prod, ap, c.n, work);
88   redc(rp, prod, c);
89 }

91 // rp = a * R mod m
92 inline constexpr void
93 to_mont(limb_t *rp, const limb_t *ap, usize an, const mont_ctx &c, limb_t *scratch) noexcept
94 {
95   const usize n = c.n;
96   an = normalize(ap, an);
97   if ( an == 0 ) {
98     zero(rp, n);
99     return;
100   }
101   const usize nn = an + n;
102   limb_t *const np = scratch;             // an + n
103   limb_t *const qp = np + nn;             // an + 1
104   limb_t *const work = qp + an + 1u;      // divrem_itch(an + n, n)
105 
106   zero(np, n);
107   copyi(np + n, ap, an);
108   divrem(qp, rp, np, nn, c.mp, n, work);
109 }
110 
111 // rp = a * R^-1 mod m, the inverse of to_mont
112 inline constexpr void
113 from_mont(limb_t *rp, const limb_t *ap, const mont_ctx &c, limb_t *scratch) noexcept
114 {
115   limb_t *const tp = scratch;      // 2n
116   copyi(tp, ap, c.n);
117   zero(tp + c.n, c.n);
118   redc(rp, tp, c);
119 }

121 inline constexpr void
122 mont_one(limb_t *rp, const mont_ctx &c, limb_t *scratch) noexcept
123 {
124   const limb_t one = 1;
125   to_mont(rp, &one, 1, c, scratch);
126 }

128 [[nodiscard, gnu::flatten]] inline constexpr usize
129 mont_mul_itch(usize n) noexcept
130 {
131   return 2u * n + __mul_itch_bound(n, n);
132 }
133 
134 [[nodiscard, gnu::flatten]] inline constexpr usize
135 mont_sqr_itch(usize n) noexcept
136 {
137   return 2u * n + __sqr_itch_bound(n);
138 }
139 
140 [[nodiscard, gnu::always_inline]] inline constexpr usize
141 from_mont_itch(usize n) noexcept
142 {
143   return 2u * n;
144 }
145 
146 [[nodiscard, gnu::flatten]] inline constexpr usize
147 to_mont_itch(usize an, usize n) noexcept
148 {
149   if ( an == 0 ) an = 1;
150   return (an + n) + (an + 1u) + divrem_cap_itch(an + n, n);
151 }
152 
153 [[nodiscard, gnu::flatten]] inline constexpr usize
154 mont_op_itch(usize n) noexcept
155 {
156   usize t = mont_mul_itch(n);
157   const usize s = mont_sqr_itch(n);
158   if ( s > t ) t = s;
159   const usize f = from_mont_itch(n);
160   if ( f > t ) t = f;
161   const usize o = to_mont_itch(n, n);
162   if ( o > t ) t = o;
163   return t;
164 }
