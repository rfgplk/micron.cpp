18 // tier crossovers
19 // (in limbs)
20 //
21 // measured on amd64 (zen+, kernel 7.1.5, -O3 -march=native -funroll-loops, taskset-pinned, median of 5)
22 //   MUL_KARATSUBA  20   karatsuba loses at 8..16 and wins from 20 on: 252 vs 309 cyc/n^2 at 20 limbs,
23 //                       257 vs 325 at 24, 177 vs 283 at 64, 99 vs 286 at 1024 and still falling.
24 //   SQR_KARATSUBA  32   the sqr crossover sits higher, because sqr_basecase already halves the
25 //                       multiply count via the a_i*a_j symmetry and karatsuba has more to beat:
26 //                       156 vs 224 cyc/n^2 at 32 limbs, 143 vs 197 at 64, 119 vs 141 at 1024.
27 //   DIV_DC         96   the recursive divider loses below it and wins above: 281 vs 340 cyc/n^2 at
28 //                       96 limbs, 231 vs 308 at 160, 156 vs 299 at 512
29 //   GET_STR_DC    128   the divide-and-conquer formatter loses below it and wins above: 178k vs 196k
30 //                       cycles at 128 limbs, 411k vs 639k at 256
31 //   SET_STR_DC  10000  in digits. the divide-and-conquer parser loses badly below it, 2.2x at 4096
32 //                       bits, 1.7x at 8192, breaks even around 32768, and wins 2.4x at 65536 bits
33 //                       (534k vs 1263k cycles)
34 //   *_COMBA             UNSELECTED
35 //
36 // TODO: we eventually want to dispatch tiers according to arch, rough references below
37 // relative refs per arch
38 //  karatsuba    12 .. 28      (haswell 20, skylake 26, zen 16, zen3 20)
39 //  toom-3       53 .. 125     (skylake 73, zen 107, zen3 89)
40 //  toom-4       93 .. 324     (skylake 208, zen3 130)
41 //  nussbaumer  180000          (Haswell, AVX2 SoA transform; final 2^20 seam wins by 21.2% mul / 23.4% sqr)
42 //  FFT          8192          (64-bit limbs; 16384 at 32 bits). Balanced products require
43 //                            >=15/16 transform occupancy; see docs/ARBINT_FFT.md and its CSV.
44 //  DC_DIV       19 .. 76      (skylake 55)
45 //  MU_DIV      855 .. 1899    (skylake 1528)

334 [[nodiscard]] inline constexpr __fft_plan
335 __fft_make_plan(usize an, usize bn) noexcept
336 {
337   return __fft_plan_for(an, bn, threshold::fft_max_log);
338 }
339 
340 inline constexpr usize __fft_block_limbs
341     = ((usize{ 1 } << (threshold::fft_max_log - 1u)) * __fft_digit_bits(threshold::fft_max_log)) / limb_bits;
342 
343 [[nodiscard]] inline constexpr bool
344 __fft_auto_shape(usize an, usize bn) noexcept
345 {
346   const auto p = __fft_make_plan(an, bn);
347   if ( !p.valid ) return false;
348   // Unequal operands otherwise fall back to Karatsuba. Balanced Toom is much
349   // stronger: only enter FFT near the full end of a power-of-two padding band.
350   if ( an != bn ) return true;
351   if constexpr ( !threshold::fft_balanced ) return false;
352   const usize bits = an * limb_bits;
353   const usize digits = bits / p.digit_bits + usize(bits % p.digit_bits != 0u);
354   return 2u * digits - 1u >= p.n - p.n / 16u;
355 }

360 // mul ladder
361 [[nodiscard, gnu::always_inline]] inline constexpr algo
362 pick_mul(usize an, usize bn) noexcept
363 {
364   const usize n = an < bn ? an : bn;
365   if ( n >= threshold::mul_nussbaumer ) return algo::nussbaumer;
366 #if defined(__micron_arbint_fft)
367   if constexpr ( MICRON_ARBINT_MUL_TIER_CAP >= 6 )
368     if ( n >= threshold::mul_fft && __fft_auto_shape(an, bn) ) return algo::fft;
369 #endif
370   if ( n < threshold::mul_comba ) return algo::basecase;
371   if ( n < threshold::mul_karatsuba ) return algo::comba;
372   if ( n < threshold::mul_toom3 ) return algo::karatsuba;
373   if ( n < threshold::mul_toom4 ) return algo::toom3;
374   if ( n < threshold::mul_nussbaumer ) return algo::toom4;
375   return algo::nussbaumer;
376 }
377 
378 // sqr ladder
379 [[nodiscard, gnu::always_inline]] inline constexpr algo
380 pick_sqr(usize n) noexcept
381 {
382   if ( n >= threshold::sqr_nussbaumer ) return algo::nussbaumer;
383 #if defined(__micron_arbint_fft)
384   if constexpr ( MICRON_ARBINT_SQR_TIER_CAP >= 6 )
385     if ( n >= threshold::sqr_fft && __fft_auto_shape(n, n) ) return algo::fft;
386 #endif
387   if ( n < threshold::sqr_comba ) return algo::basecase;
388   if ( n < threshold::sqr_karatsuba ) return algo::comba;
389   if ( n < threshold::sqr_toom3 ) return algo::karatsuba;
390   if ( n < threshold::sqr_toom4 ) return algo::toom3;
391   if ( n < threshold::sqr_nussbaumer ) return algo::toom4;
392   return algo::nussbaumer;
393 }

395 // divison ladder
396 [[nodiscard, gnu::always_inline]] inline constexpr divalgo
397 pick_div(usize nn, usize dn) noexcept
398 {
399   if ( dn < threshold::div_dc ) return divalgo::sbpi1;
400   if ( dn < threshold::div_mu ) return divalgo::dc;
401   return (nn >= dn && (nn - dn) >= dn) ? divalgo::mu : divalgo::dc;
402 }

418 [[nodiscard, gnu::always_inline]] inline constexpr usize
419 powm_window(usize ebits) noexcept
420 {
421   if ( ebits < threshold::powm_win3 ) return 1u;      // plain square-and-multiply, no table at all
422   if ( ebits < threshold::powm_win4 ) return 3u;
423   if ( ebits < threshold::powm_win5 ) return 4u;
424   if ( ebits < threshold::powm_win6 ) return 5u;
425   return 6u;

428 template<arb_solver S>
429 [[nodiscard, gnu::always_inline]] inline consteval algo
430 pinned_algo() noexcept
431 {
432   if constexpr ( micron::is_same_v<S, solver::basecase> )
433     return algo::basecase;
434   else if constexpr ( micron::is_same_v<S, solver::comba> )
435     return algo::comba;
436   else if constexpr ( micron::is_same_v<S, solver::karatsuba> )
437     return algo::karatsuba;
438   else if constexpr ( micron::is_same_v<S, solver::toom> )
439     return algo::toom3;

441     return algo::fft;
442   else
443     return algo::nussbaumer;
