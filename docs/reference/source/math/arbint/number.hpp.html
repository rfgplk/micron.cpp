31 [[nodiscard, gnu::always_inline]] inline constexpr usize
32 mod_gcd_max(usize a, usize b) noexcept
33 {
34   return a > b ? a : b;
35 }

37 template<class U> struct gcd_sizes {
38   static constexpr bool bounded = U::bounded;
39   static constexpr usize cap = U::cap_limbs;
40 
41   static constexpr usize gcd
42       = bounded ? cap + cap + (cap + 1u) + mod_gcd_max(mpn::divrem_cap_itch(cap, cap), mpn::gcd_tier_itch(cap, cap)) : 1u;
43 
44   static constexpr usize inv = bounded ? cap + mpn::invmod_itch(cap, cap) : 1u;
45 };

49 template<usize B, arb_solver S, class A>
50 [[nodiscard]] inline constexpr arbuint<B, S, A>
51 gcd(const arbuint<B, S, A> &a, const arbuint<B, S, A> &b)
52 {
53   using U = arbuint<B, S, A>;
54   using sizes = __arb::gcd_sizes<U>;
55   const usize an = a.size();
56   const usize bn = b.size();
57   if ( an == 0 ) return b;
58   if ( bn == 0 ) return a;
59 
60   const usize gn = an < bn ? an : bn;
61   __arb::scratch<U::bounded ? sizes::gcd : 1u, A, U::bounded> sc(gn + mpn::gcd_itch(an, bn));
62   mpn::limb_t *const gp = sc.get();
63   mpn::limb_t *const work = gp + gn;
64 
65   const usize rn = mpn::gcd(gp, a.limbs(), an, b.limbs(), bn, work);
66   U r;
67   r.__assign_limbs(gp, rn);
68   return r;
69 }

71 template<usize B, arb_solver S, class A>
72 [[nodiscard]] inline constexpr arbuint<B, S, A>
73 lcm(const arbuint<B, S, A> &a, const arbuint<B, S, A> &b)
74 {
75   using U = arbuint<B, S, A>;
76   if ( a.is_zero() || b.is_zero() ) return U::zero();
77   const U g = gcd(a, b);
78   return (a / g) * b;
79 }

86 template<usize B, arb_solver S, class A>
87 [[nodiscard]] inline constexpr inv_result<B, S, A>
88 invmod(const arbuint<B, S, A> &a, const arbuint<B, S, A> &m)
89 {
90   using U = arbuint<B, S, A>;
91   using sizes = __arb::gcd_sizes<U>;
92   const usize mn = m.size();
93   if ( mn == 0 ) micron::exc<micron::except::domain_error>("micron::math::invmod modulus is zero");
94 
95   const usize an = a.size();
96   __arb::scratch<U::bounded ? sizes::inv : 1u, A, U::bounded> sc(mn + mpn::invmod_itch(an != 0 ? an : 1u, mn));
97   mpn::limb_t *const rp = sc.get();
98   mpn::limb_t *const work = rp + mn;
99 
100   usize rn = 0;
101   const bool ok = mpn::invmod(rp, rn, a.limbs(), an, m.limbs(), mn, work);
102   U r;
103   if ( ok ) r.__assign_limbs(rp, rn);
104   return inv_result<B, S, A>{ r, ok };
105 }

126 template<usize B, arb_solver S, class A>
127 [[nodiscard]] inline constexpr sinv_result<B, S, A>
128 invmod(const arbint<B, S, A> &a, const arbint<B, S, A> &m)
129 {
130   using U = arbuint<B, S, A>;
131   const U mm = m.magnitude();
132   if ( mm.is_zero() ) micron::exc<micron::except::domain_error>("micron::math::invmod modulus is zero");
133 
134   U base = a.magnitude();
135   base %= mm;
136   if ( a.sign() < 0 && !base.is_zero() ) base = mm - base;
137 
138   const auto r = invmod(base, mm);
139   return sinv_result<B, S, A>{ arbint<B, S, A>(r.value, false), r.exists };
140 }
