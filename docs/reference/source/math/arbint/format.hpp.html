24 // base and the '#' prefix
25 template<typename A>
26 inline void
27 __arb_write(hstring<schar> &out, const A &v, const __impl::fmt_spec &spec, bool neg)
28 {
29   u32 base = 10;
30   bool upper = false;
31   if ( spec.type == 'x' )
32     base = 16;
33   else if ( spec.type == 'X' ) {
34     base = 16;
35     upper = true;
36   } else if ( spec.type == 'o' )
37     base = 8;
38   else if ( spec.type == 'b' )
39     base = 2;
40 
41   if ( neg ) out += '-';
42   if ( spec.alt ) {
43     if ( base == 16 ) {
44       out += '0';
45       out += (upper ? 'X' : 'x');
46     } else if ( base == 8 ) {
47       out += '0';
48     } else if ( base == 2 ) {
49       out += '0';
50       out += 'b';
51     }
52   }
53 
54   // base 2; one character per bit, plus slack
55   const usize cap = v.bit_length() + 8u;
56   micron::vector<char> tmp;
57   tmp.resize(cap);
58   const usize n = micron::math::to_chars(&tmp[0], cap, v, base, upper);
59   if ( n == 0 ) {
60     out += '0';
61     return;
62   }
63   out.append(&tmp[0], n);
64 }

90 template<typename Sk, usize B, arb_solver S, class A>
91   requires requires(Sk &s, const char *p, usize n) { s.put(p, n); }
92 max_t
93 printk(Sk &s, const arbuint<B, S, A> &v)
94 {
95   const usize cap = v.bit_length() + 8u;
96   micron::vector<char> tmp;
97   tmp.resize(cap);
98   const usize n = micron::math::to_chars(&tmp[0], cap, v, 10u, false);
99   if ( n == 0 ) return s.put("0", 1);
100   return s.put(&tmp[0], n);
101 }

103 template<typename Sk, usize B, arb_solver S, class A>
104   requires requires(Sk &s, const char *p, usize n) { s.put(p, n); }
105 max_t
106 printk(Sk &s, const arbint<B, S, A> &v)
107 {
108   const auto mag = v.magnitude();
109   const usize cap = mag.bit_length() + 9u;
110   micron::vector<char> tmp;
111   tmp.resize(cap);
112   usize off = 0;
113   if ( v.negative() ) {
114     tmp[0] = '-';
115     off = 1;
116   }
117   const usize n = micron::math::to_chars(&tmp[0] + off, cap - off, mag, 10u, false);
118   if ( n == 0 ) return s.put("0", 1);
119   return s.put(&tmp[0], n + off);
120 }
