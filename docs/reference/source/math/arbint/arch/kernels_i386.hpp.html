33 [[gnu::always_inline]] inline limb_t
34 __aors_n_idx(limb_t *rp, const limb_t *ap, const limb_t *bp, usize n, bool sub) noexcept
35 {
36   limb_t t0;
37   rp += n;
38   ap += n;
39   bp += n;
40   usize i = static_cast<usize>(0) - n;
41 
42   if ( sub ) {
43     __asm__ volatile("xorl %[t0], %[t0]\n\t"      // CF = 0, and t0 = 0
44                      "1:\n\t"
45                      "movl (%[ap],%[i],4), %[t0]\n\t"
46                      "sbbl (%[bp],%[i],4), %[t0]\n\t"
47                      "movl %[t0], (%[rp],%[i],4)\n\t"
48                      "incl %[i]\n\t"      // inc leaves CF alone; that is the entire trick
49                      "jnz 1b\n\t"
50                      "setc %b[t0]\n\t"      // CF here is still the chain's borrow
51                      "movzbl %b[t0], %[t0]\n\t"
52                      : [t0] "=&q"(t0), [i] "+r"(i)
53                      : [rp] "r"(rp), [ap] "r"(ap), [bp] "r"(bp)
54                      : "cc", "memory");
55   } else {
56     __asm__ volatile("xorl %[t0], %[t0]\n\t"
57                      "1:\n\t"
58                      "movl (%[ap],%[i],4), %[t0]\n\t"
59                      "adcl (%[bp],%[i],4), %[t0]\n\t"
60                      "movl %[t0], (%[rp],%[i],4)\n\t"
61                      "incl %[i]\n\t"
62                      "jnz 1b\n\t"
63                      "setc %b[t0]\n\t"
64                      "movzbl %b[t0], %[t0]\n\t"
65                      : [t0] "=&q"(t0), [i] "+r"(i)
66                      : [rp] "r"(rp), [ap] "r"(ap), [bp] "r"(bp)
67                      : "cc", "memory");
68   }
69   return t0;
70 }

72 [[nodiscard]] inline limb_t
73 add_n(limb_t *rp, const limb_t *ap, const limb_t *bp, usize n) noexcept
74 {
75   if ( n >= kern_min_limbs ) return __aors_n_idx(rp, ap, bp, n, false);
76   limb_t cy = 0;
77   for ( usize i = 0; i < n; ++i ) cy = addc(ap[i], bp[i], cy, rp[i]);
78   return cy;
79 }
80 
81 [[nodiscard]] inline limb_t
82 sub_n(limb_t *rp, const limb_t *ap, const limb_t *bp, usize n) noexcept
83 {
84   if ( n >= kern_min_limbs ) return __aors_n_idx(rp, ap, bp, n, true);
85   limb_t bw = 0;
86   for ( usize i = 0; i < n; ++i ) bw = subb(ap[i], bp[i], bw, rp[i]);
87   return bw;
88 }
