41 // add / subtract
42 [[gnu::always_inline]] inline limb_t
43 __add_n_blocks(limb_t *rp, const limb_t *ap, const limb_t *bp, usize blocks) noexcept
44 {
45   limb_t cy = 0, a0, a1, b0, b1;
46   __asm__ volatile("adds xzr, xzr, xzr\n\t"      // C = 0
47                    "1:\n\t"
48                    "ldp %[a0], %[a1], [%[ap]], #16\n\t"
49                    "ldp %[b0], %[b1], [%[bp]], #16\n\t"
50                    "adcs %[a0], %[a0], %[b0]\n\t"
51                    "adcs %[a1], %[a1], %[b1]\n\t"
52                    "stp %[a0], %[a1], [%[rp]], #16\n\t"
53                    "sub %[n], %[n], #1\n\t"
54                    "cbnz %[n], 1b\n\t"
55                    "cset %[cy], cs\n\t"
56                    : [cy] "+r"(cy), [rp] "+r"(rp), [ap] "+r"(ap), [bp] "+r"(bp), [n] "+r"(blocks), [a0] "=&r"(a0), [a1] "=&r"(a1),
57                      [b0] "=&r"(b0), [b1] "=&r"(b1)
58                    :
59                    : "cc", "memory");
60   return cy;
61 }

63 [[gnu::always_inline]] inline limb_t
64 __sub_n_blocks(limb_t *rp, const limb_t *ap, const limb_t *bp, usize blocks) noexcept
65 {
66   limb_t bw = 0, a0, a1, b0, b1;
67   __asm__ volatile("subs xzr, xzr, xzr\n\t"      // C = 1, i.e. no borrow
68                    "1:\n\t"
69                    "ldp %[a0], %[a1], [%[ap]], #16\n\t"
70                    "ldp %[b0], %[b1], [%[bp]], #16\n\t"
71                    "sbcs %[a0], %[a0], %[b0]\n\t"
72                    "sbcs %[a1], %[a1], %[b1]\n\t"
73                    "stp %[a0], %[a1], [%[rp]], #16\n\t"
74                    "sub %[n], %[n], #1\n\t"
75                    "cbnz %[n], 1b\n\t"
76                    "cset %[bw], cc\n\t"      // carry CLEAR means a borrow happened
77                    : [bw] "+r"(bw), [rp] "+r"(rp), [ap] "+r"(ap), [bp] "+r"(bp), [n] "+r"(blocks), [a0] "=&r"(a0), [a1] "=&r"(a1),
78                      [b0] "=&r"(b0), [b1] "=&r"(b1)
79                    :
80                    : "cc", "memory");
81   return bw;
82 }

84 [[nodiscard]] inline limb_t
85 add_n(limb_t *rp, const limb_t *ap, const limb_t *bp, usize n) noexcept
86 {
87   limb_t cy = 0;
88   const usize blocks = n >= kern_min_limbs ? (n >> 1) : 0u;
89   if ( blocks != 0 ) {
90     cy = __add_n_blocks(rp, ap, bp, blocks);
91     const usize done = blocks << 1;
92     rp += done;
93     ap += done;
94     bp += done;
95     n -= done;
96   }
97   for ( usize i = 0; i < n; ++i ) cy = addc(ap[i], bp[i], cy, rp[i]);
98   return cy;
99 }

121 [[gnu::always_inline]] inline limb_t
122 __mul_1_blocks(limb_t *rp, const limb_t *ap, usize blocks, limb_t b, limb_t cy) noexcept
123 {
124   limb_t a0, a1, l0, l1, h0, h1;
125   __asm__ volatile("1:\n\t"
126                    "ldp %[a0], %[a1], [%[ap]], #16\n\t"
127                    "mul %[l0], %[a0], %[b]\n\t"
128                    "umulh %[h0], %[a0], %[b]\n\t"
129                    "mul %[l1], %[a1], %[b]\n\t"
130                    "umulh %[h1], %[a1], %[b]\n\t"
131                    "adds %[l0], %[l0], %[cy]\n\t"
132                    "cinc %[h0], %[h0], cs\n\t"
133                    "adds %[l1], %[l1], %[h0]\n\t"
134                    "cinc %[cy], %[h1], cs\n\t"
135                    "stp %[l0], %[l1], [%[rp]], #16\n\t"
136                    "sub %[n], %[n], #1\n\t"
137                    "cbnz %[n], 1b\n\t"
138                    : [cy] "+r"(cy), [rp] "+r"(rp), [ap] "+r"(ap), [n] "+r"(blocks), [a0] "=&r"(a0), [a1] "=&r"(a1), [l0] "=&r"(l0),
139                      [l1] "=&r"(l1), [h0] "=&r"(h0), [h1] "=&r"(h1)
140                    : [b] "r"(b)
141                    : "cc", "memory");
142   return cy;
143 }

145 // rp += ap * b; two accumulations per limb
146 [[gnu::always_inline]] inline limb_t
147 __addmul_1_blocks(limb_t *rp, const limb_t *ap, usize blocks, limb_t b, limb_t cy) noexcept
148 {
149   limb_t a0, a1, l0, l1, h0, h1, d0, d1;
150   __asm__ volatile("1:\n\t"
151                    "ldp %[a0], %[a1], [%[ap]], #16\n\t"
152                    "ldp %[d0], %[d1], [%[rp]]\n\t"
153                    "mul %[l0], %[a0], %[b]\n\t"
154                    "umulh %[h0], %[a0], %[b]\n\t"
155                    "mul %[l1], %[a1], %[b]\n\t"
156                    "umulh %[h1], %[a1], %[b]\n\t"
157                    "adds %[l0], %[l0], %[cy]\n\t"      // fold the incoming carry
158                    "cinc %[h0], %[h0], cs\n\t"
159                    "adds %[l1], %[l1], %[h0]\n\t"      // fold product 0's high
160                    "cinc %[h1], %[h1], cs\n\t"
161                    "adds %[l0], %[l0], %[d0]\n\t"      // now add rp[], a fresh chain
162                    "adcs %[l1], %[l1], %[d1]\n\t"
163                    "cinc %[cy], %[h1], cs\n\t"
164                    "stp %[l0], %[l1], [%[rp]], #16\n\t"
165                    "sub %[n], %[n], #1\n\t"
166                    "cbnz %[n], 1b\n\t"
167                    : [cy] "+r"(cy), [rp] "+r"(rp), [ap] "+r"(ap), [n] "+r"(blocks), [a0] "=&r"(a0), [a1] "=&r"(a1), [l0] "=&r"(l0),
168                      [l1] "=&r"(l1), [h0] "=&r"(h0), [h1] "=&r"(h1), [d0] "=&r"(d0), [d1] "=&r"(d1)
169                    : [b] "r"(b)
170                    : "cc", "memory");
171   return cy;
172 }

174 // rp -= ap * b
175 [[gnu::always_inline]] inline limb_t
176 __submul_1_blocks(limb_t *rp, const limb_t *ap, usize blocks, limb_t b, limb_t cy) noexcept
177 {
178   limb_t a0, a1, l0, l1, h0, h1, d0, d1;
179   __asm__ volatile("1:\n\t"
180                    "ldp %[a0], %[a1], [%[ap]], #16\n\t"
181                    "ldp %[d0], %[d1], [%[rp]]\n\t"
182                    "mul %[l0], %[a0], %[b]\n\t"
183                    "umulh %[h0], %[a0], %[b]\n\t"
184                    "mul %[l1], %[a1], %[b]\n\t"
185                    "umulh %[h1], %[a1], %[b]\n\t"
186                    "adds %[l0], %[l0], %[cy]\n\t"
187                    "cinc %[h0], %[h0], cs\n\t"
188                    "adds %[l1], %[l1], %[h0]\n\t"
189                    "cinc %[h1], %[h1], cs\n\t"
190                    "subs %[d0], %[d0], %[l0]\n\t"
191                    "sbcs %[d1], %[d1], %[l1]\n\t"
192                    "cinc %[cy], %[h1], cc\n\t"      // carry clear == borrow out
193                    "stp %[d0], %[d1], [%[rp]], #16\n\t"
194                    "sub %[n], %[n], #1\n\t"
195                    "cbnz %[n], 1b\n\t"
196                    : [cy] "+r"(cy), [rp] "+r"(rp), [ap] "+r"(ap), [n] "+r"(blocks), [a0] "=&r"(a0), [a1] "=&r"(a1), [l0] "=&r"(l0),
197                      [l1] "=&r"(l1), [h0] "=&r"(h0), [h1] "=&r"(h1), [d0] "=&r"(d0), [d1] "=&r"(d1)
198                    : [b] "r"(b)
199                    : "cc", "memory");
200   return cy;
201 }

203 [[nodiscard]] inline limb_t
204 mul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
205 {
206   limb_t cy = 0;
207   const usize blocks = n >= kern_min_limbs ? (n >> 1) : 0u;
208   if ( blocks != 0 ) {
209     cy = __mul_1_blocks(rp, ap, blocks, b, cy);
210     const usize done = blocks << 1;
211     rp += done;
212     ap += done;
213     n -= done;
214   }
215   for ( usize i = 0; i < n; ++i ) {
216     limb_t lo = 0, hi = 0;
217     muladd_wide(ap[i], b, cy, lo, hi);
218     rp[i] = lo;
219     cy = hi;
220   }
221   return cy;
222 }

224 [[nodiscard]] inline limb_t
225 addmul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
226 {
227   limb_t cy = 0;
228   const usize blocks = n >= kern_min_limbs ? (n >> 1) : 0u;
229   if ( blocks != 0 ) {
230     cy = __addmul_1_blocks(rp, ap, blocks, b, cy);
231     const usize done = blocks << 1;
232     rp += done;
233     ap += done;
234     n -= done;
235   }
236   for ( usize i = 0; i < n; ++i ) {
237     limb_t lo = 0, hi = 0;
238     muladd2_wide(ap[i], b, cy, rp[i], lo, hi);
239     rp[i] = lo;
240     cy = hi;
241   }
242   return cy;

245 [[nodiscard]] inline limb_t
246 submul_1(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
247 {
248   limb_t cy = 0;
249   const usize blocks = n >= kern_min_limbs ? (n >> 1) : 0u;
250   if ( blocks != 0 ) {
251     cy = __submul_1_blocks(rp, ap, blocks, b, cy);
252     const usize done = blocks << 1;
253     rp += done;
254     ap += done;
255     n -= done;
256   }
257   for ( usize i = 0; i < n; ++i ) {
258     limb_t lo = 0, hi = 0;
259     muladd_wide(ap[i], b, cy, lo, hi);
260     limb_t d = 0;
261     const limb_t bw = subb(rp[i], lo, 0, d);
262     rp[i] = d;
263     cy = static_cast<limb_t>(hi + bw);
264   }
265   return cy;
266 }
