13 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
14 // SIMD MULTIPLY EXPERIMENT
15 //
16 // decomposition, per limb, with M = 2^32 - 1:
17 //     m0 = a_lo*b_lo   m1 = a_lo*b_hi   m2 = a_hi*b_lo   m3 = a_hi*b_hi
18 //
19 //     t  = m1 + (m0 >> 32)          <= 2^64 - 2^33 + 2^32, so it cannot wrap
20 //     u  = (t & M) + m2             same bound, same reason
21 //     lo = (u << 32) | (m0 & M)
22 //     hi = m3 + (t >> 32) + (u >> 32)
23

48 // rp = ap * b + cy
49 [[nodiscard]] inline limb_t
50 mul_1_avx2(limb_t *rp, const limb_t *ap, usize n, limb_t b) noexcept
51 {
52   const __ai_v4du mask = { 0xFFFFFFFFull, 0xFFFFFFFFull, 0xFFFFFFFFull, 0xFFFFFFFFull };
53   const limb_t bl = b & 0xFFFFFFFFull;
54   const limb_t bh = b >> 32;
55   const __ai_v4du vbl = { bl, bl, bl, bl };
56   const __ai_v4du vbh = { bh, bh, bh, bh };
57 
58   limb_t cy = 0;
59   usize i = 0;
60   alignas(32) limb_t lo[4], hi[4];
61 
62   for ( ; i + 4u <= n; i += 4u ) {
63     __ai_v4du va;
64     __builtin_memcpy(&va, ap + i, sizeof(va));
65     const __ai_v4du val = va & mask;
66     const __ai_v4du vah = va >> 32;
67 
68     const __ai_v4du m0 = __vmul32(val, vbl);
69     const __ai_v4du m1 = __vmul32(val, vbh);
70     const __ai_v4du m2 = __vmul32(vah, vbl);
71     const __ai_v4du m3 = __vmul32(vah, vbh);
72 
73     const __ai_v4du t = m1 + (m0 >> 32);
74     const __ai_v4du u = (t & mask) + m2;
75     const __ai_v4du vlo = (u << 32) | (m0 & mask);
76     const __ai_v4du vhi = m3 + (t >> 32) + (u >> 32);
77 
78     __builtin_memcpy(lo, &vlo, sizeof(vlo));
79     __builtin_memcpy(hi, &vhi, sizeof(vhi));
80 
81     for ( usize j = 0; j < 4u; ++j ) {
82       limb_t s = 0;
83       const limb_t c = addc(lo[j], cy, 0, s);
84       rp[i + j] = s;
85       cy = static_cast<limb_t>(hi[j] + c);
86     }
87   }
88 
89   for ( ; i < n; ++i ) {
90     limb_t l = 0, h = 0;
91     muladd_wide(ap[i], b, cy, l, h);
92     rp[i] = l;
93     cy = h;
94   }
95   return cy;
96 }
