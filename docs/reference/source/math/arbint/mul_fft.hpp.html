20 // binary64 convolution
21 // 2*q + log2(N) <= 35 leaves 18 mantissa bits for transform error
22 // re-anchor twiddle recurrences every 32 coefficients
23 // Larger products use overlap-add; no floating coefficient grows with the
24 // number of tiles; all sizes below are in limbs, including scratch
25 
26 struct __fft_shape {
27   usize an, bn;
28   __fft_plan plan;
29 };
30 
31 [[nodiscard]] inline constexpr __fft_shape
32 __fft_pick_shape(usize an, usize bn) noexcept
33 {
34   if ( !an || !bn ) return {};
35   usize a = an, b = bn;
36   if ( an / bn >= 8u ) a = __nuss_size_mul(3u, bn);
37   if ( bn / an >= 8u ) b = __nuss_size_mul(3u, an);
38   const auto whole = __fft_make_plan(a, b);
39   if ( whole.valid ) return { a, b, whole };
40   a = a < __fft_block_limbs ? a : __fft_block_limbs;
41   b = b < __fft_block_limbs ? b : __fft_block_limbs;
42   return { a, b, __fft_make_plan(a, b) };
43 }

45 [[nodiscard]] inline constexpr usize
46 fft_itch(usize an, usize bn) noexcept
47 {
48   if ( !an || !bn ) return 0u;
49   if ( __nuss_size_add(an, bn) == __nuss_size_max ) return __nuss_size_max;
50   const usize exact = nussbaumer_itch(an, bn);
51 #if defined(__micron_arbint_fft)
52   const auto s = __fft_pick_shape(an, bn);
53   // three half-length arrays, complex/real twiddles, alignment slack, and a tile
54   const usize work = __nuss_size_add(__nuss_size_mul(s.plan.n / 4u, 144u / limb_bytes), __nuss_size_add(s.an + s.bn, 24u / limb_bytes));
55   return work > exact ? work : exact;
56 #else
57   return exact;
58 #endif
59 }

65 inline void
66 __fft_roots(__fft_complex *tw, usize count, usize j, usize n) noexcept
67 {
68   const auto step = fft::__impl::__root<double>(j, n);
69   __fft_complex powers[20];
70   for ( usize bit = 5u; (usize{ 1 } << bit) < count; ++bit ) powers[bit - 5u] = fft::__impl::__root<double>(j * (usize{ 1 } << bit), n);
71   for ( usize k = 0u; k < count; k += 32u ) {
72     // each anchor uses at most log2(N) independently evaluated roots;
73     // recurrence error cannot accumulate across the whole table
74     __fft_complex w{ 1.0, 0.0 };
75     for ( usize bits = k >> 5u, bit = 0u; bits; bits >>= 1u, ++bit )
76       if ( bits & 1u ) w *= powers[bit];
77     const usize end = k + 32u < count ? k + 32u : count;
78     for ( usize i = k; i < end; ++i ) {
79       tw[i] = w;
80       w *= step;
81     }
82   }
83 }

85 inline void
86 __fft_twiddles(__fft_complex *tw, usize n) noexcept
87 {
88   while ( n > 8u ) {
89     const usize m = n / 4u;
90     for ( usize j = 1u; j <= 3u; ++j ) __fft_roots(tw + (j - 1u) * m, m, j, n);
91     tw += 3u * m;
92     n = m;
93   }
94 }

96 template<bool Inverse>
97 inline void
98 __fft_transform(const __fft_complex *in, __fft_complex *out, const __fft_complex *tw, usize n, usize stride = 1u) noexcept
99 {
100   if ( n == 2u ) return fft::__impl::__leaf<2, Inverse>(in, out, stride);
101   if ( n == 4u ) return fft::__impl::__leaf<4, Inverse>(in, out, stride);
102   if ( n == 8u ) return fft::__impl::__leaf<8, Inverse>(in, out, stride);
103   const usize m = n / 4u;
104   for ( usize j = 0u; j < 4u; ++j ) __fft_transform<Inverse>(in + j * stride, out + j * m, tw + 3u * m, m, stride * 4u);
105   fft::__impl::__stage<4, Inverse>(out, tw, m, 0u, m);
106 }

108 template<u64 P>
109 [[nodiscard]] inline u64
110 __fft_residue(const limb_t *p, usize n) noexcept
111 {
112   u64 r = 0u;
113   for ( usize i = n; i-- > 0u; ) {
114     if constexpr ( limb_bits == 64u ) r = ((r << 32u) | u32(u64(p[i]) >> 32u)) % P;
115     r = ((r << 32u) | u32(p[i])) % P;
116   }
117   return r;
118 }

120 template<u64 P>
121 [[nodiscard]] inline bool
122 __fft_check(const limb_t *r, const limb_t *a, usize an, const limb_t *b, usize bn) noexcept
123 {
124   return __fft_residue<P>(r, an + bn) == (__fft_residue<P>(a, an) * __fft_residue<P>(b, bn)) % P;
125 }

127 // pack even/odd real samples into an N/2 complex transform
128 // slot zero stores the independent real DC and Nyquist bins in its real and imaginary fields
129 template<bool Inverse>
130 inline void
131 __fft_real_pass(__fft_complex *b, const __fft_complex *roots, usize h) noexcept
132 {
133   const auto z = b[0];
134   b[0] = { z.real + z.imag, z.real - z.imag };
135   if constexpr ( Inverse ) b[0] *= 0.5;
136   for ( usize k = 1u; k < h / 2u; ++k ) {
137     const auto u = b[k], v = conj(b[h - k]);
138     const auto sum = (u + v) * 0.5;
139     auto w = roots[k];
140     if constexpr ( Inverse ) w = conj(w);
141     const auto diff = (u - v) * w * 0.5;
142     const __fft_complex rotated = Inverse ? __fft_complex{ -diff.imag, diff.real } : __fft_complex{ diff.imag, -diff.real };
143     b[k] = sum + rotated;
144     b[h - k] = conj(sum - rotated);
145   }
146   b[h / 2u] = conj(b[h / 2u]);
147 }

149 inline void
150 __fft_load(__fft_complex *a, usize h, const limb_t *ap, usize an, usize q) noexcept
151 {
152   const usize digits = __nuss_ceil_div(an * limb_bits, q);
153   for ( usize k = 0u; k < h; ++k )
154     a[k] = { 2u * k < digits ? double(__nuss_digit(ap, an, 2u * k, q)) : 0.0,
155              2u * k + 1u < digits ? double(__nuss_digit(ap, an, 2u * k + 1u, q)) : 0.0 };
156 }

158 template<bool Square>
159 [[nodiscard]] inline bool
160 __fft_product(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn, const __fft_plan &p, __fft_complex *a, __fft_complex *b,
161               __fft_complex *c, const __fft_complex *tw, const __fft_complex *roots) noexcept
162 {
163   const usize ad = __nuss_ceil_div(an * limb_bits, p.digit_bits), bd = __nuss_ceil_div(bn * limb_bits, p.digit_bits);
164   const usize h = p.n / 2u;
165   __fft_load(a, h, ap, an, p.digit_bits);
166   __fft_transform<false>(a, b, tw, h);
167   __fft_real_pass<false>(b, roots, h);
168   if constexpr ( Square ) {
169     b[0] = { b[0].real * b[0].real, b[0].imag * b[0].imag };
170     fft::__impl::__multiply(b + 1u, b + 1u, h - 1u);
171   } else {
172     __fft_load(a, h, bp, bn, p.digit_bits);
173     __fft_transform<false>(a, c, tw, h);
174     __fft_real_pass<false>(c, roots, h);
175     b[0] = { b[0].real * c[0].real, b[0].imag * c[0].imag };
176     fft::__impl::__multiply(b + 1u, c + 1u, h - 1u);
177   }
178   __fft_real_pass<true>(b, roots, h);
179   __fft_transform<true>(b, a, tw, h);
180   const double scale = 1.0 / double(h);
181   const u64 mask = (u64{ 1 } << p.digit_bits) - 1u;
182   const double bound = double(ad < bd ? ad : bd) * double(mask * mask);
183   __nuss_packer pack{ rp, an + bn, 0u, 0u, 0u, true };
184   u64 carry = 0u;
185   for ( usize k = 0u; k < ad + bd - 1u; ++k ) {
186     const double v = ((k & 1u) ? a[k / 2u].imag : a[k / 2u].real) * scale;
187     if ( !(v >= -0.125 && v <= bound + 0.125) ) return false;
188     const u64 nearest = static_cast<u64>(v + 0.5);
189     const double error = v - double(nearest);
190     if ( error < -0.125 || error > 0.125 ) return false;
191     const u64 word = nearest + carry;
192     pack.push(static_cast<limb_t>(word & mask), p.digit_bits);
193     carry = word >> p.digit_bits;
194   }
195   while ( carry ) {
196     pack.push(static_cast<limb_t>(carry & mask), p.digit_bits);
197     carry >>= p.digit_bits;
198   }
199   pack.finish();
200   // consistency guards supplement the bounded rounding error
201   return pack.exact && __fft_check<4294967291ull>(rp, ap, an, bp, bn) && __fft_check<4294967279ull>(rp, ap, an, bp, bn);
202 }

204 template<bool Square>
205 [[nodiscard]] inline bool
206 __fft_run(limb_t *rp, const limb_t *ap, usize an, const limb_t *bp, usize bn, limb_t *scratch) noexcept
207 {
208   const auto s = __fft_pick_shape(an, bn);
209   limb_t *aligned = scratch;
210   while ( reinterpret_cast<usize>(aligned) % 8u ) ++aligned;
211   const usize h = s.plan.n / 2u, count = 9u * s.plan.n / 4u + 1u;
212   auto *a = ::new (static_cast<void *>(aligned)) __fft_complex[count];
213   auto *b = a + h, *c = b + h, *tw = c + h, *roots = tw + h;
214   limb_t *part = aligned + count * sizeof(__fft_complex) / limb_bytes;
215   __fft_twiddles(tw, h);
216   __fft_roots(roots, h / 2u + 1u, 1u, s.plan.n);
217   usize prepared = s.plan.n;
218   bool ok = true;
219   if ( s.an == an && s.bn == bn ) {
220     ok = __fft_product<Square>(rp, ap, an, bp, bn, s.plan, a, b, c, tw, roots);
221   } else {
222     zero(rp, an + bn);
223     for ( usize ai = 0u; ai < an && ok; ai += s.an ) {
224       const usize atake = an - ai < s.an ? an - ai : s.an;
225       for ( usize bi = 0u; bi < bn && ok; bi += s.bn ) {
226         if constexpr ( Square ) {
227           if ( bi > ai ) break;
228         }
229         const usize btake = bn - bi < s.bn ? bn - bi : s.bn;
230         if ( atake <= 2u || btake <= 2u ) {
231           if ( atake >= btake )
232             mul_basecase(part, ap + ai, atake, bp + bi, btake);
233           else
234             mul_basecase(part, bp + bi, btake, ap + ai, atake);
235         } else {
236           const auto p = __fft_make_plan(atake, btake);
237           if ( p.n != prepared ) {
238             __fft_twiddles(tw, p.n / 2u);
239             __fft_roots(roots, p.n / 4u + 1u, 1u, p.n);
240             prepared = p.n;
241           }
242           if ( Square && ai == bi )
243             ok = __fft_product<true>(part, ap + ai, atake, bp + bi, btake, p, a, b, c, tw, roots);
244           else
245             ok = __fft_product<false>(part, ap + ai, atake, bp + bi, btake, p, a, b, c, tw, roots);
246         }
247         if ( !ok ) break;
248         __nuss_add_partial(rp, an + bn, ai + bi, part, atake + btake);
249         if ( Square && ai != bi ) __nuss_add_partial(rp, an + bn, ai + bi, part, atake + btake);
250       }
251     }
252   }
253   // restore limb lifetimes before another mpn operation reuses this scratch
254   (void)::new (static_cast<void *>(aligned)) limb_t[count * sizeof(__fft_complex) / limb_bytes];
255   return ok;
256 }

259 inline constexpr void
260 mul_fft(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn, limb_t *scratch) noexcept
261 {
262   if ( !an || !bn ) return zero(rp, an + bn);
263 #if defined(__micron_arbint_fft)
264   if !consteval {
265     if ( __fft_run<false>(rp, ap, an, bp, bn, scratch) ) return;
266   }
267 #endif
268   mul_nussbaumer(rp, ap, an, bp, bn, scratch);
269 }

271 inline constexpr void
272 sqr_fft(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch) noexcept
273 {
274   if ( !n ) return;
275 #if defined(__micron_arbint_fft)
276   if !consteval {
277     if ( __fft_run<true>(rp, ap, n, ap, n, scratch) ) return;
278   }
279 #endif
280   sqr_nussbaumer(rp, ap, n, scratch);
281 }
