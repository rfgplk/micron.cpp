16 // ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
17 // extended Euclid
18 //
19 // cofactors accumulate as non negative magnitudes under a checkerboard convention
20 //
21 // ..r_{-1} = m, r_0 = a mod m               t_{-1} = 0, t_0 = 1
22 // ..r_{i+1} = r_{i-1} - q_i * r_i           t_{i+1} = t_{i-1} + q_i * t_i
23 // ..(-1)^i * t_i * a  ==  r_i  (mod m)      for every i

32 [[nodiscard, gnu::flatten]] inline constexpr usize
33 invmod_itch(usize an, usize mn) noexcept
34 {
35   const usize s = mn + 3u;
36   usize w = divrem_cap_itch(2u * s, s);
37   const usize m = __mul_itch_bound(s, s);
38   if ( m > w ) w = m;
39   if ( an > mn ) {
40     const usize h = (an - mn + 1u) + divrem_itch(an, mn);
41     if ( h > w ) w = h;
42   }
43   // r0 r1 r2 t0 t1 t2 q, the q*t product, the Lehmer path's second staging slot, then the tail
44   return 7u * s + 2u * s + s + w;
45 }

47 [[nodiscard]] inline constexpr bool
48 invmod(limb_t *rp, usize &rn, const limb_t *ap, usize an, const limb_t *mp, usize mn, limb_t *scratch) noexcept
49 {
50   mn = normalize(mp, mn);
51   an = normalize(ap, an);
52   rn = 0;
53   if ( mn == 0 ) return false;
54 
55   if ( mn == 1 && mp[0] == 1 ) return true;
56 
57   const usize s = mn + 3u;
58   limb_t *r0 = scratch;                  // s
59   limb_t *r1 = r0 + s;                   // s
60   limb_t *r2 = r1 + s;                   // s
61   limb_t *t0 = r2 + s;                   // s
62   limb_t *t1 = t0 + s;                   // s
63   limb_t *t2 = t1 + s;                   // s
64   limb_t *const q = t2 + s;              // s
65   limb_t *const prod = q + s;            // 2s
66   limb_t *const w2 = prod + 2u * s;      // s
67   limb_t *const work = w2 + s;
68 
69   usize r0n = mn, r1n = 0, t0n = 0, t1n = 1;
70   copyi(r0, mp, mn);
71   t0[0] = 0;
72   t1[0] = 1;
73 
74   // r_0 = a mod m
75   if ( an == 0 ) {
76     r1n = 0;
77   } else if ( an < mn || cmp_var(ap, an, mp, mn) < 0 ) {
78     copyi(r1, ap, an);
79     r1n = an;
80   } else {
81     limb_t *const rq = work;
82     limb_t *const rw = rq + (an - mn + 1u);
83     divrem(rq, r1, ap, an, mp, mn, rw);
84     r1n = normalize(r1, mn);
85   }
86 
87   // k counts the index of r1
88   usize k = 0;
89   while ( r1n != 0 ) {
90     // Lehmer
91     if ( r0n == r1n && r0n >= 2u && r0n >= threshold::gcd_lehmer ) {
92       const usize sh = limb_clz(r0[r0n - 1u]);
93       limb_t A = r0[r0n - 1u];
94       limb_t B = r1[r0n - 1u];
95       if ( sh != 0 ) {
96         A = static_cast<limb_t>((A << sh) | (r0[r0n - 2u] >> (limb_bits - sh)));
97         B = static_cast<limb_t>((B << sh) | (r1[r0n - 2u] >> (limb_bits - sh)));
98       }
99 
100       mat1 mm{}, nn{};
101       const usize steps = hgcd2_ext(mm, nn, A, B);
102       if ( steps != 0 ) {
103         usize pn = 0;
104         if ( mat1_mul_inverse_vector(mm, r2, q, r0, r1, r0n, pn) ) {
105           // length is of the cofactors, not the operands
106           usize tn = t0n > t1n ? t0n : t1n;
107           if ( tn == 0 ) tn = 1u;
108           if ( t0n < tn ) zero(t0 + t0n, tn - t0n);
109           if ( t1n < tn ) zero(t1 + t1n, tn - t1n);
110           (void)mat1_mul_vector(nn, prod, w2, t0, t1, tn);
111 
112           const usize a0 = normalize(r2, pn);
113           const usize a1 = normalize(q, pn);
114           const usize c0 = normalize(prod, tn + 2u);
115           const usize c1 = normalize(w2, tn + 2u);
116 
117           const bool odd = (steps & 1u) != 0;
118           copyi(r0, odd ? q : r2, odd ? a1 : a0);
119           copyi(r1, odd ? r2 : q, odd ? a0 : a1);
120           copyi(t0, odd ? w2 : prod, odd ? c1 : c0);
121           copyi(t1, odd ? prod : w2, odd ? c0 : c1);
122           r0n = odd ? a1 : a0;
123           r1n = odd ? a0 : a1;
124           t0n = odd ? c1 : c0;
125           t1n = odd ? c0 : c1;
126           k += steps;
127           continue;
128         }
129         // high limbs disagreed; the matrix was not valid for the full operands
130       }
131     }
132 
133     divrem(q, r2, r0, r0n, r1, r1n, work);
134     const usize qn = normalize(q, r0n - r1n + 1u);
135     const usize r2n = normalize(r2, r1n);
136 
137     // t2 = t0 + q*t1
138     usize t2n = 0;
139     if ( qn == 0 || t1n == 0 ) {
140       copyi(t2, t0, t0n);
141       t2n = t0n;
142     } else {
143       const usize pn = qn + t1n;
144       if ( qn >= t1n )
145         mul(prod, q, qn, t1, t1n, work);
146       else
147         mul(prod, t1, t1n, q, qn, work);
148       const usize pnn = normalize(prod, pn);
149       if ( pnn >= t0n ) {
150         const limb_t cy = add(t2, prod, pnn, t0, t0n);
151         t2[pnn] = cy;
152         t2n = normalize(t2, pnn + 1u);
153       } else {
154         const limb_t cy = add(t2, t0, t0n, prod, pnn);
155         t2[t0n] = cy;
156         t2n = normalize(t2, t0n + 1u);
157       }
158     }
159 
160     limb_t *sw = r0;
161     r0 = r1;
162     r1 = r2;
163     r2 = sw;
164     r0n = r1n;
165     r1n = r2n;
166 
167     sw = t0;
168     t0 = t1;
169     t1 = t2;
170     t2 = sw;
171     t0n = t1n;
172     t1n = t2n;
173     ++k;
174   }
175 
176   // g == r0
177   if ( r0n != 1 || r0[0] != 1 ) return false;
178 
179   // (-1)^(k-1) * t0
180   const bool negative = ((k - 1u) & 1u) != 0;
181   if ( !negative || t0n == 0 ) {
182     copyi(rp, t0, t0n);
183     rn = t0n;
184     return true;
185   }
186   (void)sub(rp, mp, mn, t0, t0n);
187   rn = normalize(rp, mn);
188   return true;
189 }
