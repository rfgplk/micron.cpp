49 [[nodiscard, gnu::flatten]] inline constexpr usize
50 mul_itch(usize an, usize bn, usize cutoff = threshold::mul_karatsuba) noexcept
51 {
52   const usize n = an < bn ? an : bn;
53   if ( clamp_algo(pick_mul(an, bn)) == algo::fft ) return fft_itch(an, bn);
54   if ( n < cutoff ) return 0;
55   if ( clamp_algo(pick_mul(an, bn)) == algo::nussbaumer ) return nussbaumer_itch(an, bn);
56   usize base = karatsuba_itch(n, cutoff);
57   if ( an == bn && n >= threshold::mul_toom3 ) {
58     const usize t3 = toom3_itch(n);
59     if ( t3 > base ) base = t3;
60   }
61   if ( an == bn && n >= threshold::mul_toom4 ) {
62     const usize t4 = toom4_itch(n);
63     if ( t4 > base ) base = t4;
64   }
65   return (an == bn) ? base : (unbalanced_stage_itch(an, bn, cutoff) + base);
66 }

68 [[nodiscard, gnu::flatten]] inline constexpr usize
69 sqr_itch(usize n, usize cutoff = threshold::sqr_karatsuba) noexcept
70 {
71   if ( clamp_to(pick_sqr(n), sqr_tier_cap) == algo::fft ) return fft_itch(n, n);
72   if ( clamp_to(pick_sqr(n), sqr_tier_cap) == algo::nussbaumer ) return sqr_nussbaumer_itch(n);
73   usize base = sqr_karatsuba_itch(n, cutoff);
74   if ( n >= threshold::sqr_toom3 ) {
75     const usize t3 = sqr_toom3_itch(n);
76     if ( t3 > base ) base = t3;
77   }
78   if ( n >= threshold::sqr_toom4 ) {
79     const usize t4 = sqr_toom4_itch(n);
80     if ( t4 > base ) base = t4;
81   }
82   return base;
83 }

85 [[nodiscard, gnu::flatten]] inline constexpr usize
86 mul_itch_forced(usize an, usize bn) noexcept
87 {
88   usize base = mul_itch(an, bn, karatsuba_force);
89   // mul_with<toom3> runs the Toom kernel wherever toom3_applies() allows
90   if ( an == bn && an >= 3u ) {
91     const usize t3 = toom3_itch(an, true);
92     if ( t3 > base ) base = t3;
93     const usize t4 = toom4_itch(an, true);
94     if ( t4 > base ) base = t4;
95   }
96   const usize ns = nussbaumer_itch(an, bn);
97   if ( ns > base ) base = ns;
98   const usize fs = fft_itch(an, bn);
99   if ( fs > base ) base = fs;
100   return base;
101 }

103 [[nodiscard, gnu::flatten]] inline constexpr usize
104 sqr_itch_forced(usize n) noexcept
105 {
106   usize base = sqr_itch(n, karatsuba_force);
107   if ( sqr_toom3_applies(n) ) {
108     const usize t3 = sqr_toom3_itch(n, true);
109     if ( t3 > base ) base = t3;
110   }
111   const usize t4 = sqr_toom4_itch(n, true);
112   if ( t4 > base ) base = t4;
113   const usize ns = sqr_nussbaumer_itch(n);
114   if ( ns > base ) base = ns;
115   const usize fs = fft_itch(n, n);
116   if ( fs > base ) base = fs;
117   return base;
118 }

120 template<algo K>
121 [[nodiscard, gnu::flatten]] inline constexpr usize
122 mul_itch_with(usize an, usize bn) noexcept
123 {
124   if constexpr ( K == algo::basecase || K == algo::comba )
125     return 0u;
126   else if constexpr ( K == algo::karatsuba ) {
127     if ( bn < 2u ) return 0u;
128     if ( an != bn ) {
129       usize stage = unbalanced_stage_itch(an, bn, threshold::mul_karatsuba);
130       if ( stage == 0u ) stage = 2u * bn;
131       return stage + karatsuba_itch(bn);
132     }
133     const usize k = (an + 1u) / 2u;
134     return 6u * k + 1u + karatsuba_itch(k);
135   } else if constexpr ( K == algo::toom3 ) {
136     if ( toom3_applies(an, bn) ) return toom3_itch(an, true);
137     return mul_itch_with<algo::karatsuba>(an, bn);
138   } else if constexpr ( K == algo::toom4 ) {
139     if ( toom4_applies(an, bn) ) return toom4_itch(an, true);
140     if ( toom3_applies(an, bn) ) return toom3_itch(an, true);
141     return mul_itch_with<algo::karatsuba>(an, bn);
142   } else if constexpr ( K == algo::nussbaumer )
143     return nussbaumer_itch(an, bn);
144   else if constexpr ( K == algo::fft )
145     return fft_itch(an, bn);
146 }

148 template<algo K>
149 [[nodiscard, gnu::flatten]] inline constexpr usize
150 sqr_itch_with(usize n) noexcept
151 {
152   if constexpr ( K == algo::basecase || K == algo::comba )
153     return 0u;
154   else if constexpr ( K == algo::karatsuba ) {
155     if ( n < 2u ) return 0u;
156     const usize k = (n + 1u) / 2u;
157     return 5u * k + 1u + sqr_karatsuba_itch(k);
158   } else if constexpr ( K == algo::toom3 ) {
159     if ( sqr_toom3_applies(n) ) return sqr_toom3_itch(n, true);
160     return sqr_itch_with<algo::karatsuba>(n);
161   } else if constexpr ( K == algo::toom4 ) {
162     if ( sqr_toom4_applies(n) ) return sqr_toom4_itch(n, true);
163     if ( sqr_toom3_applies(n) ) return sqr_toom3_itch(n, true);
164     return sqr_itch_with<algo::karatsuba>(n);
165   } else if constexpr ( K == algo::nussbaumer )
166     return sqr_nussbaumer_itch(n);
167   else if constexpr ( K == algo::fft )
168     return fft_itch(n, n);
169 }

197 template<arb_solver S>
198 [[nodiscard, gnu::flatten]] inline constexpr usize
199 mul_solver_cap_itch(usize cap) noexcept
200 {
201   if constexpr ( micron::is_same_v<S, solver::basecase> || micron::is_same_v<S, solver::comba> ) {
202     return 0u;
203   } else if constexpr ( micron::is_same_v<S, solver::nussbaumer> ) {
204     return nussbaumer_itch(cap, cap);
205   } else if constexpr ( micron::is_same_v<S, solver::fft> ) {
206     return fft_itch(cap, cap);
207   } else {
208     // Every unbalanced Karatsuba stage follows an Euclidean remainder chain.  The sum of that
209     // chain is below 3*cap and each stage reserves 2*bn, so this bounds every an,bn <= cap.
210     usize out = 0u;
211     if constexpr ( !micron::is_same_v<S, solver::automatic> )
212       out = __nuss_size_add(__nuss_size_mul(6u, cap), karatsuba_itch(cap));
213     else if ( cap >= threshold::mul_karatsuba )
214       out = __nuss_size_add(__nuss_size_mul(6u, cap), karatsuba_itch(cap));
215     if constexpr ( micron::is_same_v<S, solver::toom> ) {
216       out = __mul_itch_max(out, toom3_itch(cap, true));
217     } else if constexpr ( micron::is_same_v<S, solver::automatic> ) {
218       if constexpr ( static_cast<u8>(mul_tier_cap) >= static_cast<u8>(algo::toom3) )
219         if ( cap >= threshold::mul_toom3 ) out = __mul_itch_max(out, toom3_itch(cap));
220       if constexpr ( static_cast<u8>(mul_tier_cap) >= static_cast<u8>(algo::toom4) )
221         if ( cap >= threshold::mul_toom4 ) out = __mul_itch_max(out, toom4_itch(cap));
222       if constexpr ( static_cast<u8>(mul_tier_cap) >= static_cast<u8>(algo::nussbaumer) )
223         if ( cap >= threshold::mul_nussbaumer ) out = __mul_itch_max(out, nussbaumer_itch(cap, cap));
224 #if defined(__micron_arbint_fft)
225       if constexpr ( static_cast<u8>(mul_tier_cap) >= static_cast<u8>(algo::fft) )
226         if ( cap >= threshold::mul_fft ) out = __mul_itch_max(out, fft_itch(cap, cap));
227 #endif
228     }
229     return out;
230   }
231 }

234 [[nodiscard, gnu::flatten]] inline constexpr usize
235 sqr_solver_cap_itch(usize cap) noexcept
236 {
237   if constexpr ( micron::is_same_v<S, solver::automatic> ) {

248 // A normalized intermediate can enter FFT even when its maximum shape cannot.
249 // Consumers reserving for a range of operand lengths must include that workspace.
250 [[nodiscard]] inline constexpr usize
251 __mul_itch_bound(usize an, usize bn) noexcept
252 {
253 #if defined(__micron_arbint_fft)
254   if constexpr ( MICRON_ARBINT_MUL_TIER_CAP >= 6 )
255     if ( an >= threshold::mul_fft && bn >= threshold::mul_fft ) return mul_solver_cap_itch<solver::automatic>(an > bn ? an : bn);
256 #endif
257   return mul_itch(an, bn);
258 }

268 template<algo K>
269 inline constexpr void
270 mul_with(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn,
271          limb_t *scratch) noexcept
272 {
273   static_assert(static_cast<u8>(K) <= static_cast<u8>(tiers_built),
274                 "arbint: this multiplication tier is not implemented yet -- see mpn::tiers_built");
275   if constexpr ( K == algo::basecase )
276     mul_basecase(rp, ap, an, bp, bn);
277   else if constexpr ( K == algo::comba )
278     mul_comba(rp, ap, an, bp, bn);
279   else if constexpr ( K == algo::karatsuba )
280     mul_karatsuba_top(rp, ap, an, bp, bn, scratch);
281   else if constexpr ( K == algo::toom3 ) {
282     if ( toom3_applies(an, bn) )
283       mul_toom3(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
284     else
285       mul_karatsuba_top(rp, ap, an, bp, bn, scratch);
286   } else if constexpr ( K == algo::toom4 ) {
287     if ( toom4_applies(an, bn) )
288       mul_toom4(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
289     else if ( toom3_applies(an, bn) )
290       mul_toom3(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
291     else
292       mul_karatsuba_top(rp, ap, an, bp, bn, scratch);
293   } else if constexpr ( K == algo::fft ) {
294     mul_fft(rp, ap, an, bp, bn, scratch);
295   } else {
296     mul_nussbaumer(rp, ap, an, bp, bn, scratch);
297   }
298 }

300 template<algo K>
301 inline constexpr void
302 sqr_with(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch) noexcept
303 {
304   static_assert(static_cast<u8>(K) <= static_cast<u8>(tiers_built),
305                 "arbint: this squaring tier is not implemented yet -- see mpn::tiers_built");
306   if constexpr ( K == algo::basecase )
307     sqr_basecase(rp, ap, n);
308   else if constexpr ( K == algo::comba )
309     sqr_comba(rp, ap, n);
310   else if constexpr ( K == algo::karatsuba )
311     sqr_karatsuba_top(rp, ap, n, scratch);
312   else if constexpr ( K == algo::toom3 ) {
313     if ( sqr_toom3_applies(n) )
314       sqr_toom3(rp, ap, n, scratch, threshold::sqr_karatsuba);
315     else
316       sqr_karatsuba_top(rp, ap, n, scratch);
317   } else if constexpr ( K == algo::toom4 ) {
318     if ( sqr_toom4_applies(n) )
319       sqr_toom4(rp, ap, n, scratch, threshold::sqr_karatsuba);
320     else if ( sqr_toom3_applies(n) )
321       sqr_toom3(rp, ap, n, scratch, threshold::sqr_karatsuba);
322     else
323       sqr_karatsuba_top(rp, ap, n, scratch);
324   } else if constexpr ( K == algo::fft ) {
325     sqr_fft(rp, ap, n, scratch);
326   } else {
327     sqr_nussbaumer(rp, ap, n, scratch);
328   }
329 }

334 inline constexpr void
335 mul(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn, limb_t *scratch) noexcept
336 {
337   if ( bn == 1 ) {
338     rp[an] = mul_1(rp, ap, an, bp[0]);
339     return;
340   }
341   switch ( clamp_algo(pick_mul(an, bn)) ) {
342   case algo::basecase:
343     mul_basecase(rp, ap, an, bp, bn);
344     return;
345   case algo::comba:
346     mul_comba(rp, ap, an, bp, bn);
347     return;
348   case algo::karatsuba:
349     mul_karatsuba(rp, ap, an, bp, bn, scratch);
350     return;
351   case algo::toom3:
352     if ( toom3_applies(an, bn) )
353       mul_toom3(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
354     else
355       mul_karatsuba(rp, ap, an, bp, bn, scratch);
356     return;
357   case algo::toom4:
358     if ( toom4_applies(an, bn) )
359       mul_toom4(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
360     else if ( toom3_applies(an, bn) )
361       mul_toom3(rp, ap, bp, an, scratch, threshold::mul_karatsuba);
362     else
363       mul_karatsuba(rp, ap, an, bp, bn, scratch);
364     return;
365   case algo::nussbaumer:
366     mul_nussbaumer(rp, ap, an, bp, bn, scratch);
367     return;
368   case algo::fft:
369     mul_fft(rp, ap, an, bp, bn, scratch);
370     return;
371   }
372 }

374 inline constexpr void
375 sqr(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch) noexcept
376 {
377   if ( n == 1 ) {
378     mul_wide(ap[0], ap[0], rp[0], rp[1]);
379     return;
380   }
381   switch ( clamp_to(pick_sqr(n), sqr_tier_cap) ) {
382   case algo::basecase:
383     sqr_basecase(rp, ap, n);
384     return;
385   case algo::comba:
386     sqr_comba(rp, ap, n);
387     return;
388   case algo::karatsuba:
389     sqr_karatsuba(rp, ap, n, scratch);
390     return;
391   case algo::toom3:
392     if ( sqr_toom3_applies(n) )
393       sqr_toom3(rp, ap, n, scratch, threshold::sqr_karatsuba);
394     else
395       sqr_karatsuba(rp, ap, n, scratch);
396     return;
397   case algo::toom4:
398     if ( sqr_toom4_applies(n) )
399       sqr_toom4(rp, ap, n, scratch, threshold::sqr_karatsuba);
400     else if ( sqr_toom3_applies(n) )
401       sqr_toom3(rp, ap, n, scratch, threshold::sqr_karatsuba);
402     else
403       sqr_karatsuba(rp, ap, n, scratch);
404     return;
405   case algo::nussbaumer:
406     sqr_nussbaumer(rp, ap, n, scratch);
407     return;
408   case algo::fft:
409     sqr_fft(rp, ap, n, scratch);
410     return;
411   }
412 }

417 template<usize AN, usize BN, arb_solver S>
418 [[gnu::always_inline]] inline constexpr void
419 mul_fixed(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize an, const limb_t *__restrict__ bp, usize bn,
420           limb_t *scratch) noexcept
421 {
422   if constexpr ( !micron::is_same_v<S, solver::automatic> ) {
423     mul_with<pinned_algo<S>()>(rp, ap, an, bp, bn, scratch);
424   } else if constexpr ( AN <= 16 && BN <= 16 && AN + BN <= 24 ) {
425     // small enough that the branch to pick a shape costs more than always taking the unrolled one
426     if ( an == AN && bn == BN )
427       mul_comba_fixed<AN, BN>(rp, ap, bp);
428     else
429       mul(rp, ap, an, bp, bn, scratch);
430   } else {
431     mul(rp, ap, an, bp, bn, scratch);
432   }
433 }

435 template<usize N, arb_solver S>
436 [[gnu::always_inline]] inline constexpr void
437 sqr_fixed(limb_t *__restrict__ rp, const limb_t *__restrict__ ap, usize n, limb_t *scratch) noexcept
438 {
439   if constexpr ( !micron::is_same_v<S, solver::automatic> ) {
440     sqr_with<pinned_algo<S>()>(rp, ap, n, scratch);
441   } else if constexpr ( N <= 16 ) {
442     if ( n == N )
443       sqr_comba_fixed<N>(rp, ap);
444     else
445       sqr(rp, ap, n, scratch);
446   } else {
447     sqr(rp, ap, n, scratch);
448   }
