15 constexpr u32
16 __rne(u32 value, u32 shift) noexcept
17 {
18   const u32 retained = value >> shift, mask = (u32(1) << shift) - 1;
19   const u32 rest = value & mask, half = u32(1) << (shift - 1);
20   return retained + u32(rest > half || (rest == half && (retained & 1)));
21 }

23 constexpr u16
24 __encode16(u32 raw) noexcept
25 {
26   const u32 sign = (raw >> 16) & 0x8000;
27   const u32 fraction = raw & 0x7fffff, exponent = (raw >> 23) & 0xff;
28   if ( exponent == 255 ) return u16(sign | (fraction ? 0x7e00 : 0x7c00));
29   const i32 e = i32(exponent) - 112;
30   if ( e >= 31 ) return u16(sign | 0x7c00);
31   if ( e < -10 ) return u16(sign);
32   if ( e <= 0 ) return u16(sign | __rne(fraction | 0x800000, u32(14 - e)));
33   return u16(sign | ((u32(e) << 10) + __rne(fraction, 13)));
34 }

36 constexpr f32
37 __decode16(u16 value) noexcept
38 {
39   const u32 sign = u32(value & 0x8000) << 16;
40   u32 e = (value >> 10) & 31, f = value & 1023;
41   if ( e == 31 ) return __builtin_bit_cast(f32, sign | 0x7f800000u | (f ? (f << 13) | 0x400000u : 0u));
42   if ( e == 0 ) {
43     if ( f == 0 ) return __builtin_bit_cast(f32, sign);
44     e = 113;
45     while ( !(f & 1024) ) {
46       f <<= 1;
47       --e;
48     }
49     f &= 1023;
50   } else
51     e += 112;
52   return __builtin_bit_cast(f32, sign | (e << 23) | (f << 13));
53 }

55 template<usize Mantissa, i32 Bias>
56 constexpr f64
57 __decode64(u16 value) noexcept
58 {
59   const u64 sign = u64(value & 0x8000) << 48;
60   constexpr u32 mask = (u32(1) << Mantissa) - 1, max_exp = (u32(1) << (15 - Mantissa)) - 1;
61   const u32 exponent = (value >> Mantissa) & max_exp;
62   u32 fraction = value & mask;
63   if ( exponent == max_exp )
64     return __builtin_bit_cast(f64,
65                               sign | 0x7ff0000000000000ull | (fraction ? (u64(fraction) << (52 - Mantissa)) | 0x8000000000000ull : 0ull));
66   i32 e = i32(exponent ? exponent : 1) - Bias;
67   if ( !exponent ) {
68     if ( !fraction ) return __builtin_bit_cast(f64, sign);
69     while ( !(fraction & (mask + 1)) ) {
70       fraction <<= 1;
71       --e;
72     }
73     fraction &= mask;
74   }
75   return __builtin_bit_cast(f64, sign | (u64(e + 1023) << 52) | (u64(fraction) << (52 - Mantissa)));
76 }

79 // Conversion from f32 rounds to nearest, ties to even; overflow becomes signed
80 // infinity, subnormals survive, and NaNs become a signed canonical quiet NaN.
81 // Wider source values must be explicitly narrowed to f32 by the caller.
82 // References preserve nonfinite object bits under Clang finite-math call attributes.
83 struct float16 {
84   u16 bits{};
85 
86   constexpr float16() noexcept = default;
87 
88   constexpr bool
89   is_finite() const noexcept
90   {
91     return (bits & 0x7c00) != 0x7c00;
92   }
93 
94   template<class F>
95     requires(micron::is_floating_point_v<F> && sizeof(F) == 4)
96   explicit constexpr float16(const F &value) noexcept : bits(__compact::__encode16(__builtin_bit_cast(u32, value)))
97   {
98   }
99 
100   static constexpr float16
101   from_bits(u16 value) noexcept
102   {
103     float16 out;
104     out.bits = value;
105     return out;
106   }
107 
108   constexpr f32
109   to_float() const noexcept
110   {
111     return __compact::__decode16(bits);
112   }
113 
114   constexpr f64
115   to_double() const noexcept
116   {
117     return __compact::__decode64<10, 15>(bits);
118   }
119 };

121 struct bfloat16 {
122   u16 bits{};
123 
124   constexpr bfloat16() noexcept = default;
125 
126   constexpr bool
127   is_finite() const noexcept
128   {
129     return (bits & 0x7f80) != 0x7f80;
130   }
131 
132   template<class F>
133     requires(micron::is_floating_point_v<F> && sizeof(F) == 4)
134   explicit constexpr bfloat16(const F &value) noexcept
135   {
136     const u32 raw = __builtin_bit_cast(u32, value);
137     bits = (raw & 0x7fffffff) > 0x7f800000 ? u16((raw >> 16 & 0x8000) | 0x7fc0) : u16((raw + 0x7fff + ((raw >> 16) & 1)) >> 16);
138   }
139 
140   static constexpr bfloat16
141   from_bits(u16 value) noexcept
142   {
143     bfloat16 out;
144     out.bits = value;
145     return out;
146   }
147 
148   constexpr f32
149   to_float() const noexcept
150   {
151     u32 raw = u32(bits) << 16;
152     if ( (raw & 0x7fffffff) > 0x7f800000 ) raw |= 0x400000;
153     return __builtin_bit_cast(f32, raw);
154   }
155 
156   constexpr f64
157   to_double() const noexcept
158   {
159     return __compact::__decode64<7, 127>(bits);
160   }
161 };
