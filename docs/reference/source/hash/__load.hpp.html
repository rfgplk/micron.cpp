11 // unaligned little-endian punning loads for the checksum kernels
12 namespace micron
13 {
14 namespace hashes
15 {
16 
17 constexpr u32
18 __load32(const u8 *p) noexcept
19 {
20 #if defined(__micron_endian_little)
21   if !consteval {
22     u32 v;
23     __builtin_memcpy(&v, p, 4);
24     return v;
25   }
26 #endif
27   return u32(p[0]) | (u32(p[1]) << 8) | (u32(p[2]) << 16) | (u32(p[3]) << 24);
28 }
29 
30 constexpr u64
31 __load64(const u8 *p) noexcept
32 {
33 #if defined(__micron_endian_little)
34   if !consteval {
35     u64 v;
36     __builtin_memcpy(&v, p, 8);
37     return v;
38   }
39 #endif
40   return u64(p[0]) | (u64(p[1]) << 8) | (u64(p[2]) << 16) | (u64(p[3]) << 24) | (u64(p[4]) << 32) | (u64(p[5]) << 40) | (u64(p[6]) << 48)
41          | (u64(p[7]) << 56);
42 }

44 };      // namespace hashes
