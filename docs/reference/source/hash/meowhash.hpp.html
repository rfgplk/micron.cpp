110 [[gnu::target("aes,sse4.1")]] static inline __m128i
111 __meow_hash(const void *Seed128Init, usize Len, const void *SourceInit) noexcept
112 {
113   __m128i xmm0, xmm1, xmm2, xmm3, xmm4, xmm5, xmm6, xmm7;            // hash accumulation lanes
114   __m128i xmm8, xmm9, xmm10, xmm11, xmm12, xmm13, xmm14, xmm15;      // residual / length ingests
115 
116   const u8 *rax = (const u8 *)SourceInit;
117   const u8 *rcx = (const u8 *)Seed128Init;
118 
119   // seed the eight hash registers
120   movdqu(xmm0, rcx + 0x00);
121   movdqu(xmm1, rcx + 0x10);
122   movdqu(xmm2, rcx + 0x20);
123   movdqu(xmm3, rcx + 0x30);
124   movdqu(xmm4, rcx + 0x40);
125   movdqu(xmm5, rcx + 0x50);
126   movdqu(xmm6, rcx + 0x60);
127   movdqu(xmm7, rcx + 0x70);
128 
129   // hash all full 256-byte blocks
130   usize BlockCount = (Len >> 8);
131   if ( BlockCount > MEOW_PREFETCH_LIMIT ) {
132     while ( BlockCount-- ) {
133       prefetcht0(rax + MEOW_PREFETCH + 0x00);
134       prefetcht0(rax + MEOW_PREFETCH + 0x40);
135       prefetcht0(rax + MEOW_PREFETCH + 0x80);
136       prefetcht0(rax + MEOW_PREFETCH + 0xc0);
137 
138       MEOW_MIX(xmm0, xmm4, xmm6, xmm1, xmm2, rax + 0x00);
139       MEOW_MIX(xmm1, xmm5, xmm7, xmm2, xmm3, rax + 0x20);
140       MEOW_MIX(xmm2, xmm6, xmm0, xmm3, xmm4, rax + 0x40);
141       MEOW_MIX(xmm3, xmm7, xmm1, xmm4, xmm5, rax + 0x60);
142       MEOW_MIX(xmm4, xmm0, xmm2, xmm5, xmm6, rax + 0x80);
143       MEOW_MIX(xmm5, xmm1, xmm3, xmm6, xmm7, rax + 0xa0);
144       MEOW_MIX(xmm6, xmm2, xmm4, xmm7, xmm0, rax + 0xc0);
145       MEOW_MIX(xmm7, xmm3, xmm5, xmm0, xmm1, rax + 0xe0);
146 
147       rax += 0x100;
148     }
149   } else {
150     while ( BlockCount-- ) {
151       MEOW_MIX(xmm0, xmm4, xmm6, xmm1, xmm2, rax + 0x00);
152       MEOW_MIX(xmm1, xmm5, xmm7, xmm2, xmm3, rax + 0x20);
153       MEOW_MIX(xmm2, xmm6, xmm0, xmm3, xmm4, rax + 0x40);
154       MEOW_MIX(xmm3, xmm7, xmm1, xmm4, xmm5, rax + 0x60);
155       MEOW_MIX(xmm4, xmm0, xmm2, xmm5, xmm6, rax + 0x80);
156       MEOW_MIX(xmm5, xmm1, xmm3, xmm6, xmm7, rax + 0xa0);
157       MEOW_MIX(xmm6, xmm2, xmm4, xmm7, xmm0, rax + 0xc0);
158       MEOW_MIX(xmm7, xmm3, xmm5, xmm0, xmm1, rax + 0xe0);
159 
160       rax += 0x100;
161     }
162   }
163 
164   // load any less-than-32-byte residual
165   pxor_clear(xmm9, xmm9);
166   pxor_clear(xmm11, xmm11);
167 
168   // the part that is _not_ 16-byte aligned
169   const u8 *Last = (const u8 *)SourceInit + (Len & ~(usize)0xf);
170   unsigned Len8 = (Len & 0xf);
171   if ( Len8 ) {
172     movdqu(xmm8, &__meow_mask_len[0x10 - Len8]);
173 
174     const u8 *LastOk = (const u8 *)((((usize)(((const u8 *)SourceInit) + Len - 1)) | (MEOW_PAGESIZE - 1)) - 16);
175     int Align = (Last > LastOk) ? ((int)(usize)Last) & 0xf : 0;
176     movdqu(xmm10, &__meow_shift_adjust[Align]);
177     movdqu(xmm9, Last - Align);
178     pshufb(xmm9, xmm10);
179 
180     // and off the extra bytes (jeffr)
181     pand(xmm9, xmm8);
182   }
183 
184   // the part that _is_ 16-byte aligned
185   if ( Len & 0x10 ) {
186     xmm11 = xmm9;
187     movdqu(xmm9, Last - 0x10);
188   }
189 
190   // construct the residual and length ingests
191   xmm8 = xmm9;
192   xmm10 = xmm9;
193   palignr(xmm8, xmm11, 15);
194   palignr(xmm10, xmm11, 1);
195 
196   // room for a 128-bit + 64-bit nonce here; left zeroed by design
197   pxor_clear(xmm12, xmm12);
198   pxor_clear(xmm13, xmm13);
199   pxor_clear(xmm14, xmm14);
200   movq(xmm15, Len);
201   palignr(xmm12, xmm15, 15);
202   palignr(xmm14, xmm15, 1);
203 
204   // always mix the less-than-32-byte residual, even if empty, to keep the pattern
205   MEOW_MIX_REG(xmm0, xmm4, xmm6, xmm1, xmm2, xmm8, xmm9, xmm10, xmm11);
206 
207   // append the length, to avoid problems with the 32-byte padding
208   MEOW_MIX_REG(xmm1, xmm5, xmm7, xmm2, xmm3, xmm12, xmm13, xmm14, xmm15);
209 
210   // hash all full 32-byte blocks
211   unsigned LaneCount = (Len >> 5) & 0x7;
212   if ( LaneCount == 0 ) goto MixDown;
213   MEOW_MIX(xmm2, xmm6, xmm0, xmm3, xmm4, rax + 0x00);
214   --LaneCount;
215   if ( LaneCount == 0 ) goto MixDown;
216   MEOW_MIX(xmm3, xmm7, xmm1, xmm4, xmm5, rax + 0x20);
217   --LaneCount;
218   if ( LaneCount == 0 ) goto MixDown;
219   MEOW_MIX(xmm4, xmm0, xmm2, xmm5, xmm6, rax + 0x40);
220   --LaneCount;
221   if ( LaneCount == 0 ) goto MixDown;
222   MEOW_MIX(xmm5, xmm1, xmm3, xmm6, xmm7, rax + 0x60);
223   --LaneCount;
224   if ( LaneCount == 0 ) goto MixDown;
225   MEOW_MIX(xmm6, xmm2, xmm4, xmm7, xmm0, rax + 0x80);
226   --LaneCount;
227   if ( LaneCount == 0 ) goto MixDown;
228   MEOW_MIX(xmm7, xmm3, xmm5, xmm0, xmm1, rax + 0xa0);
229   --LaneCount;
230   if ( LaneCount == 0 ) goto MixDown;
231   MEOW_MIX(xmm0, xmm4, xmm6, xmm1, xmm2, rax + 0xc0);
232   --LaneCount;
233 
234   // mix the eight lanes down to one 128-bit hash
235 MixDown:
236 
237   MEOW_SHUFFLE(xmm0, xmm1, xmm2, xmm4, xmm5, xmm6);
238   MEOW_SHUFFLE(xmm1, xmm2, xmm3, xmm5, xmm6, xmm7);
239   MEOW_SHUFFLE(xmm2, xmm3, xmm4, xmm6, xmm7, xmm0);
240   MEOW_SHUFFLE(xmm3, xmm4, xmm5, xmm7, xmm0, xmm1);
241   MEOW_SHUFFLE(xmm4, xmm5, xmm6, xmm0, xmm1, xmm2);
242   MEOW_SHUFFLE(xmm5, xmm6, xmm7, xmm1, xmm2, xmm3);
243   MEOW_SHUFFLE(xmm6, xmm7, xmm0, xmm2, xmm3, xmm4);
244   MEOW_SHUFFLE(xmm7, xmm0, xmm1, xmm3, xmm4, xmm5);
245   MEOW_SHUFFLE(xmm0, xmm1, xmm2, xmm4, xmm5, xmm6);
246   MEOW_SHUFFLE(xmm1, xmm2, xmm3, xmm5, xmm6, xmm7);
247   MEOW_SHUFFLE(xmm2, xmm3, xmm4, xmm6, xmm7, xmm0);
248   MEOW_SHUFFLE(xmm3, xmm4, xmm5, xmm7, xmm0, xmm1);
249 
250   paddq(xmm0, xmm2);
251   paddq(xmm1, xmm3);
252   paddq(xmm4, xmm6);
253   paddq(xmm5, xmm7);
254   pxor(xmm0, xmm1);
255   pxor(xmm4, xmm5);
256   paddq(xmm0, xmm4);
257 
258   return xmm0;
259 }

280 inline void
281 __meow_make_seed(u8 seed128[128], u64 seed) noexcept
282 {
283   micron::cbytecpy<128>(seed128, __meow_default_seed);
284   u64 w0;
285   micron::cbytecpy<sizeof(u64)>(&w0, seed128 + 0);
286   w0 ^= seed;
287   micron::cbytecpy<sizeof(u64)>(seed128 + 0, &w0);
288   u64 w1;
289   micron::cbytecpy<sizeof(u64)>(&w1, seed128 + 8);
290   w1 ^= seed * 0x9E3779B97F4A7C15ull;
291   micron::cbytecpy<sizeof(u64)>(seed128 + 8, &w1);
292 }

295 [[gnu::target("aes,sse4.1")]] inline micron::pair<u64, u64>
296 meowhash128(const byte *data, usize sz) noexcept
297 {
298   alignas(16) u8 seed128[128];
299   __meow_make_seed(seed128, Seed);
300   __m128i h = __meow_hash(seed128, sz, data);
301   return { (u64)_mm_extract_epi64(h, 0), (u64)_mm_extract_epi64(h, 1) };
302 }
