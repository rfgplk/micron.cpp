55 inline __attribute__((always_inline)) void
56 rapid_mum(u64 *A, u64 *B) noexcept
57 {
58   u128 r = *A;
59   r *= *B;
60   *A = (u64)r;
61   *B = (u64)(r >> 64);
62 }
63 
64 inline __attribute__((always_inline)) u64
65 rapid_mix(u64 A, u64 B) noexcept
66 {
67   rapid_mum(&A, &B);
68   return A ^ B;
69 }

71 inline __attribute__((always_inline)) u64
72 rapid_read64(const u8 *p) noexcept
73 {
74   u64 v;
75   micron::cbytecpy<sizeof(u64)>(&v, p);
76 #if defined(__BYTE_ORDER__) && __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
77   v = __builtin_bswap64(v);
78 #endif
79   return v;
80 }

93 inline __attribute__((always_inline)) u64
94 rapid_finalize(u64 a, u64 b, u64 seed, usize i, const u64 *secret) noexcept
95 {
96   a ^= secret[1];
97   b ^= seed;
98   rapid_mum(&a, &b);
99   return rapid_mix(a ^ secret[7], b ^ secret[1] ^ i);
100 }

102 inline __attribute__((always_inline)) void
103 rapid_short(const u8 *p, usize len, u64 &seed, u64 &a, u64 &b) noexcept
104 {
105   if ( len >= 4 ) {
106     seed ^= len;
107     if ( len >= 8 ) {
108       a = rapid_read64(p);
109       b = rapid_read64(p + len - 8);
110     } else {
111       a = rapid_read32(p);
112       b = rapid_read32(p + len - 4);
113     }
114   } else if ( len > 0 ) {
115     a = (((u64)p[0]) << 45) | p[len - 1];
116     b = p[len >> 1];
117   } else
118     a = b = 0;
119 }

121 inline u64
122 rapidhash_internal(const void *key, usize len, u64 seed, const u64 *secret) noexcept
123 {
124   const u8 *p = (const u8 *)key;
125   seed ^= rapid_mix(seed ^ secret[2], secret[1]);
126   u64 a = 0, b = 0;
127   usize i = len;
128   if ( __rapid_likely(len <= 16) ) {
129     rapid_short(p, len, seed, a, b);
130   } else {
131     if ( len > 112 ) {
132       u64 see1 = seed, see2 = seed;
133       u64 see3 = seed, see4 = seed;
134       u64 see5 = seed, see6 = seed;
135       do {
136         seed = rapid_mix(rapid_read64(p) ^ secret[0], rapid_read64(p + 8) ^ seed);
137         see1 = rapid_mix(rapid_read64(p + 16) ^ secret[1], rapid_read64(p + 24) ^ see1);
138         see2 = rapid_mix(rapid_read64(p + 32) ^ secret[2], rapid_read64(p + 40) ^ see2);
139         see3 = rapid_mix(rapid_read64(p + 48) ^ secret[3], rapid_read64(p + 56) ^ see3);
140         see4 = rapid_mix(rapid_read64(p + 64) ^ secret[4], rapid_read64(p + 72) ^ see4);
141         see5 = rapid_mix(rapid_read64(p + 80) ^ secret[5], rapid_read64(p + 88) ^ see5);
142         see6 = rapid_mix(rapid_read64(p + 96) ^ secret[6], rapid_read64(p + 104) ^ see6);
143         p += 112;
144         i -= 112;
145       } while ( i > 112 );
146       seed ^= see1;
147       see2 ^= see3;
148       see4 ^= see5;
149       seed ^= see6;
150       see2 ^= see4;
151       seed ^= see2;
152     }
153     if ( i > 16 ) {
154       seed = rapid_mix(rapid_read64(p) ^ secret[2], rapid_read64(p + 8) ^ seed);
155       if ( i > 32 ) {
156         seed = rapid_mix(rapid_read64(p + 16) ^ secret[2], rapid_read64(p + 24) ^ seed);
157         if ( i > 48 ) {
158           seed = rapid_mix(rapid_read64(p + 32) ^ secret[1], rapid_read64(p + 40) ^ seed);
159           if ( i > 64 ) {
160             seed = rapid_mix(rapid_read64(p + 48) ^ secret[1], rapid_read64(p + 56) ^ seed);
161             if ( i > 80 ) {
162               seed = rapid_mix(rapid_read64(p + 64) ^ secret[2], rapid_read64(p + 72) ^ seed);
163               if ( i > 96 ) {
164                 seed = rapid_mix(rapid_read64(p + 80) ^ secret[1], rapid_read64(p + 88) ^ seed);
165               }
166             }
167           }
168         }
169       }
170     }
171     a = rapid_read64(p + i - 16) ^ i;
172     b = rapid_read64(p + i - 8);
173   }
174   return rapid_finalize(a, b, seed, i, secret);
175 }
