16 inline __attribute__((always_inline)) u64
17 rotl64(const u64 x, const i8 r)
18 {
19   return (x << r) | (x >> (64 - r));
20 }

22 // NOTE: this is an idiomatic port of murmur3 (x64), due to this library being designed for exclusive x64 use, the 32-bit
23 // variants of this hash are being omitted. credits to smhasher
24 template<typename T = byte>
25 micron::pair<u64, u64>
26 murmur(const T *key, const usize len, const u32 seed)
27 {
28   const usize nblocks = len / 16;
29   u64 s1 = seed;
30   u64 s2 = seed;
31 
32   const u64 c1 = 0x87c37b91114253d5;
33   const u64 c2 = 0x4cf5ad432745937f;
34 
35   const u64 *blcks = reinterpret_cast<const u64 *>(key);
36 
37   for ( usize i = 0; i < nblocks; i++ ) {
38     u64 k1 = blcks[i * 2];
39     u64 k2 = blcks[i * 2 + 1];
40 
41     k1 *= c1;
42     k1 = rotl64(k1, 31);
43     k1 *= c2;
44     s1 ^= k1;
45 
46     s1 = rotl64(s1, 27);
47     s1 += s2;
48     s1 = s1 * 5 + 0x52dce729;
49 
50     k2 *= c2;
51     k2 = rotl64(k2, 33);
52     k2 *= c1;
53     s2 ^= k2;
54 
55     // fix
56     s2 = rotl64(s2, 31);
57     s2 += s1;
58     s2 = s2 * 5 + 0x38495ab5;
59   }
60   const u8 *tail = (const u8 *)(key + nblocks * 16);
61 
62   u64 k1 = 0;
63   u64 k2 = 0;
64 
65   switch ( len & 15 ) {
66   case 15:
67     k2 ^= ((u64)tail[14]) << 48;
68     [[fallthrough]];
69   case 14:
70     k2 ^= ((u64)tail[13]) << 40;
71     [[fallthrough]];
72   case 13:
73     k2 ^= ((u64)tail[12]) << 32;
74     [[fallthrough]];
75   case 12:
76     k2 ^= ((u64)tail[11]) << 24;
77     [[fallthrough]];
78   case 11:
79     k2 ^= ((u64)tail[10]) << 16;
80     [[fallthrough]];
81   case 10:
82     k2 ^= ((u64)tail[9]) << 8;
83     [[fallthrough]];
84   case 9:
85     k2 ^= ((u64)tail[8]) << 0;
86     k2 *= c2;
87     k2 = rotl64(k2, 33);
88     k2 *= c1;
89     s2 ^= k2;
90 
91     [[fallthrough]];
92   case 8:
93     k1 ^= ((u64)tail[7]) << 56;
94     [[fallthrough]];
95   case 7:
96     k1 ^= ((u64)tail[6]) << 48;
97     [[fallthrough]];
98   case 6:
99     k1 ^= ((u64)tail[5]) << 40;
100     [[fallthrough]];
101   case 5:
102     k1 ^= ((u64)tail[4]) << 32;
103     [[fallthrough]];
104   case 4:
105     k1 ^= ((u64)tail[3]) << 24;
106     [[fallthrough]];
107   case 3:
108     k1 ^= ((u64)tail[2]) << 16;
109     [[fallthrough]];
110   case 2:
111     k1 ^= ((u64)tail[1]) << 8;
112     [[fallthrough]];
113   case 1:
114     k1 ^= ((u64)tail[0]) << 0;
115     k1 *= c1;
116     k1 = rotl64(k1, 31);
117     k1 *= c2;
118     s1 ^= k1;
119   };
120   s1 ^= len;
121   s2 ^= len;
122 
123   s1 += s2;
124   s2 += s1;
125 
126   s1 ^= s1 >> 33;
127   s1 *= 0xff51afd7ed558ccd;
128   s1 ^= s1 >> 33;
129   s1 *= 0xc4ceb9fe1a85ec53;
130   s1 ^= s1 >> 33;
131 
132   s2 ^= s2 >> 33;
133   s2 *= 0xff51afd7ed558ccd;
134   s2 ^= s2 >> 33;
135   s2 *= 0xc4ceb9fe1a85ec53;
136   s2 ^= s2 >> 33;
137 
138   s1 += s2;
139   s2 += s1;
140   return { s1, s2 };
141 }
