34 // Bounded incremental XXH64. The byte tail also makes the state usable during
35 // constant evaluation; no pointer punning or caller-provided padding is needed.
36 struct xxhash64_stream {
37   u64 total = 0, seed = 0, lanes[4]{};
38   byte tail[32]{};
39   usize buffered = 0;
40 
41   constexpr explicit xxhash64_stream(u64 initial = 0) noexcept { reset(initial); }
42 
43   static constexpr u64
44   rotate(u64 x, u32 n) noexcept
45   {
46     return (x << n) | (x >> (64 - n));
47   }
48 
49   static constexpr u64
50   round(u64 a, u64 x) noexcept
51   {
52     return rotate(a + x * xxprime64b, 31) * xxprime64a;
53   }
54 
55   static constexpr u64
56   load(const byte *p, usize n) noexcept
57   {
58     u64 x = 0;
59     for ( usize i = 0; i < n; ++i ) x |= u64(p[i]) << (8 * i);
60     return x;
61   }
62 
63   constexpr void
64   reset(u64 initial = 0) noexcept
65   {
66     total = 0;
67     seed = initial;
68     buffered = 0;
69     lanes[0] = seed + xxprime64a + xxprime64b;
70     lanes[1] = seed + xxprime64b;
71     lanes[2] = seed;
72     lanes[3] = seed - xxprime64a;
73   }
74 
75   constexpr void
76   stripe(const byte *p) noexcept
77   {
78     for ( usize i = 0; i < 4; ++i ) lanes[i] = round(lanes[i], load(p + 8 * i, 8));
79   }
80 
81   constexpr void
82   update(const byte *p, usize n) noexcept
83   {
84     total += n;
85     usize i = 0;
86     if ( buffered ) {
87       while ( buffered < 32 && i < n ) tail[buffered++] = p[i++];
88       if ( buffered < 32 ) return;
89       stripe(tail);
90       buffered = 0;
91     }
92     for ( ; n - i >= 32; i += 32 ) stripe(p + i);
93     while ( i < n ) tail[buffered++] = p[i++];
94   }
95 
96   constexpr u64
97   digest() const noexcept
98   {
99     u64 h = seed + xxprime64e;
100     if ( total >= 32 ) {
101       h = rotate(lanes[0], 1) + rotate(lanes[1], 7) + rotate(lanes[2], 12) + rotate(lanes[3], 18);
102       for ( auto v : lanes ) h = (h ^ round(0, v)) * xxprime64a + xxprime64d;
103     }
104     h += total;
105     usize i = 0;
106     for ( ; buffered - i >= 8; i += 8 ) h = rotate(h ^ round(0, load(tail + i, 8)), 27) * xxprime64a + xxprime64d;
107     if ( buffered - i >= 4 ) {
108       h = rotate(h ^ (load(tail + i, 4) * xxprime64a), 23) * xxprime64b + xxprime64c;
109       i += 4;
110     }
111     while ( i < buffered ) h = rotate(h ^ (u64(tail[i++]) * xxprime64e), 11) * xxprime64a;
112     h ^= h >> 33;
113     h *= xxprime64b;
114     h ^= h >> 29;
115     h *= xxprime64c;
116     return h ^ (h >> 32);
117   }
118 };

183 inline __attribute__((always_inline)) u64
184 xxmergeround(u64 (&v)[4])
185 {
186   u64 xxh = rotl64(v[0], 1) + rotl64(v[1], 7) + rotl64(v[2], 12) + rotl64(v[3], 18);
187   v[0] = xxround(0, v[0]);
188   xxh ^= v[0];
189   xxh = xxh * xxprime64a + xxprime64d;
190 
191   v[1] = xxround(0, v[1]);
192   xxh ^= v[1];
193   xxh = xxh * xxprime64a + xxprime64d;
194 
195   v[2] = xxround(0, v[2]);
196   xxh ^= v[2];
197   xxh = xxh * xxprime64a + xxprime64d;
198 
199   v[3] = xxround(0, v[3]);
200   xxh ^= v[3];
201   xxh = xxh * xxprime64a + xxprime64d;
202   return xxh;
203 }

205 inline u64
206 xxhash_final(u64 xxhash, const byte *ptr, usize len)
207 {
208   len &= 31;
209   while ( len >= 8 ) {
210     const u64 k1 = xxround(0, __load64(ptr));
211     ptr += 8;
212     xxhash ^= k1;
213     xxhash = rotl64(xxhash, 27) * xxprime64a + xxprime64d;
214     len -= 8;
215   }
216   if ( len >= 4 ) {
217     xxhash ^= u64(__load32(ptr)) * xxprime64a;
218     ptr += 4;
219     xxhash = rotl64(xxhash, 23) * xxprime64b + xxprime64c;
220     len -= 4;
221   }
222   while ( len > 0 ) {
223     xxhash ^= (*ptr++) * xxprime64e;
224     xxhash = rotl64(xxhash, 11) * xxprime64a;
225     --len;
226   }
227 
228   xxhash ^= xxhash >> 33;
229   xxhash *= xxprime64b;
230   xxhash ^= xxhash >> 29;
231   xxhash *= xxprime64c;
232   xxhash ^= xxhash >> 32;
233 
234   return xxhash;
235 }

237 // for lz4 czlib code (spliced in from there)
238 constexpr u32
239 xxhash32(const byte *src, usize len, u32 seed = 0) noexcept
240 {
241   usize i = 0;
242   u32 h = 0;
243   if ( len >= 16 ) {
244     u32 v1 = seed + xxprime32a + xxprime32b;
245     u32 v2 = seed + xxprime32b;
246     u32 v3 = seed;
247     u32 v4 = seed - xxprime32a;
248     for ( ; i + 16 <= len; i += 16 ) {
249       v1 = xxround_32(v1, __load32(src + i));
250       v2 = xxround_32(v2, __load32(src + i + 4));
251       v3 = xxround_32(v3, __load32(src + i + 8));
252       v4 = xxround_32(v4, __load32(src + i + 12));
253     }
254     h = __xxrotl32(v1, 1) + __xxrotl32(v2, 7) + __xxrotl32(v3, 12) + __xxrotl32(v4, 18);
255   } else {
256     h = seed + xxprime32e;
257   }
258   h += static_cast<u32>(len);
259   for ( ; i + 4 <= len; i += 4 ) h = __xxrotl32(h + __load32(src + i) * xxprime32c, 17) * xxprime32d;
260   for ( ; i < len; ++i ) h = __xxrotl32(h + u32(src[i]) * xxprime32e, 11) * xxprime32a;
261   h ^= h >> 15;
262   h *= xxprime32b;
263   h ^= h >> 13;
264   h *= xxprime32c;
265   h ^= h >> 16;
266   return h;
267 }

269 template<u64 seed>
270 inline u64
271 xxhash64(const byte *src, usize len)
272 {
273   // xstate state = init_seed<seed>();
274   u64 xxhash = 0;
275   if ( len >= 32 ) [[likely]] {
276     const byte *const end = src + len;
277     const byte *const limit = end - 31;
278     u64 v[4];
279     v[0] = seed + xxprime64a + xxprime64b;
280     v[1] = seed + xxprime64b;
281     v[2] = seed;
282     v[3] = seed - xxprime64a;
283 
284     do {
285       v[0] = xxround(v[0], __load64(src));
286       v[1] = xxround(v[1], __load64(src + 8));
287       v[2] = xxround(v[2], __load64(src + 16));
288       v[3] = xxround(v[3], __load64(src + 24));
289       src += 32;
290     } while ( src < limit );
291 
292     xxhash = xxmergeround(v);
293   } else {
294     xxhash = seed + xxprime64e;
295   }
296   xxhash += len;
297   return xxhash_final(xxhash, src, len);
298 }
