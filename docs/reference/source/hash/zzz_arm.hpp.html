186 __zload_tail(const u8 *ptr, usize remaining)
187 {
188   alignas(16) u8 tmp[32];

194 // hard seed
195 template<i64 Seed>
196 void
197 z(const u8 *__restrict data, usize sz, u64 *__restrict out)
198 {
199   const __m256_neon IV = __zset_epi64(iv64_3, iv64_2, iv64_1, iv64_0);
200   const __m256_neon seedv = __zset1_epi64(Seed);
201 
202   __m256_neon S0 = __zxor(IV, seedv);
203   __m256_neon S1 = __zxor(__zperm_swap(IV), seedv);
204 
205   const u8 *ptr = data;
206   const u8 *end = data + sz;
207 
208   while ( ptr < end ) {
209     __m256_neon block;
210     usize remaining = (usize)(end - ptr);
211     if ( remaining >= 32 ) {
212       block = __zload(ptr);
213       ptr += 32;
214     } else {
215       block = __zload_tail(ptr, remaining);
216       ptr += remaining;
217     }
218 
219     __m256_neon S0_next = __zxor(S0, block);
220     __m256_neon S1_next = __zxor(S1, block);
221 
222     for ( u64 r = 0; r < __zzzrounds; ++r ) {
223       __m256_neon T0 = __zslli64<3>(S0_next);
224       __m256_neon A0 = __zsub64(S0_next, T0);
225       S0_next = A0;
226 
227       __m256_neon T1 = __zslli64<2>(S1_next);
228       __m256_neon A1 = __zsub64(S1_next, T1);
229       S1_next = A1;
230     }
231 
232     S0 = S0_next;
233     S1 = S1_next;
234   }
235 
236   __m256_neon S = __zxor(S0, S1);
237   __zstore(reinterpret_cast<u8 *>(out), S);
238 }

296 // hard seed version
297 template<i64 Seed>
298 void
299 zz(const u8 *__restrict data, usize sz, u64 *__restrict out)
300 {
301   const __m256_neon shift_vec = __zz_shift_vec();
302   const __m256_neon IV = __zset_epi64(iv64_3, iv64_2, iv64_1, iv64_0);
303 
304   __m256_neon S0 = IV;
305   __m256_neon S1 = __zperm_swap(IV);
306   __m256_neon S2 = __zxor(IV, __zslli64<1>(IV));
307   __m256_neon S3 = __zadd64(IV, __zslli64<2>(IV));
308 
309   const __m256_neon seedv = __zset1_epi64(Seed);
310   S0 = __zxor(S0, seedv);
311   S1 = __zxor(S1, seedv);
312   S2 = __zxor(S2, seedv);
313   S3 = __zxor(S3, seedv);
314 
315   const u8 *ptr = data;
316   const u8 *end = data + sz;
317 
318   while ( ptr < end ) {
319     __m256_neon block;
320     usize remaining = (usize)(end - ptr);
321     if ( remaining >= 32 ) {
322       block = __zload(ptr);
323       ptr += 32;
324     } else {
325       block = __zload_tail(ptr, remaining);
326       ptr += remaining;
327     }
328 
329     S0 = __zxor(S0, block);
330     S1 = __zadd64(S1, block);
331     S2 = __zxor(S2, __zslli64<1>(block));
332     S3 = __zadd64(S3, __zsrli64<1>(block));
333 
334     for ( u64 r = 0; r < __zzzrounds; ++r ) {
335       __m256_neon T, U;
336 
337       // s0
338       T = __zsllv64(S0, shift_vec);
339       S0 = __zxor(S0, T);
340       U = __zperm_swap(S0);
341       S0 = __zsubs_epi8(S0, U);
342       T = __zslli16<3>(S0);
343       S0 = __zxor(S0, T);
344 
345       // s1
346       T = __zsllv64(S1, shift_vec);
347       S1 = __zxor(S1, T);
348       U = __zperm_swap(S1);
349       S1 = __zsubs_epi8(S1, U);
350       T = __zslli16<3>(S1);
351       S1 = __zxor(S1, T);
352 
353       // s2
354       T = __zsllv64(S2, shift_vec);
355       S2 = __zxor(S2, T);
356       U = __zperm_swap(S2);
357       S2 = __zsubs_epi8(S2, U);
358       T = __zslli16<3>(S2);
359       S2 = __zxor(S2, T);
360 
361       // s3
362       T = __zsllv64(S3, shift_vec);
363       S3 = __zxor(S3, T);
364       U = __zperm_swap(S3);
365       S3 = __zsubs_epi8(S3, U);
366       T = __zslli16<3>(S3);
367       S3 = __zxor(S3, T);
368     }
369   }
370 
371   __m256_neon S_final = __zxor(__zxor(S0, S1), __zxor(S2, S3));
372   __zstore(reinterpret_cast<u8 *>(out), S_final);
373 }

531 zzz(const u8 *__restrict data, usize sz, u64 *__restrict out)
532 {
533   const __m256_neon IV = __zset_epi64(iv64_3, iv64_2, iv64_1, iv64_0);
534   const __m256_neon seedv = __zset1_epi64(Seed);
535 
536   __m256_neon S0 = __zxor(IV, seedv);
537   __m256_neon S1 = __zxor(__zperm_swap(IV), seedv);
538   __m256_neon S2 = __zxor(__zslli64<1>(IV), seedv);
539   __m256_neon S3 = __zxor(__zsrli64<1>(IV), seedv);
540 
541   const u8 *ptr = data;
542   const u8 *end = data + sz;
543 
544   while ( ptr < end ) {
545     __m256_neon block;
546     usize remaining = (usize)(end - ptr);
547     if ( remaining >= 32 ) {
548       block = __zload(ptr);
549       ptr += 32;
550     } else {
551       block = __zload_tail(ptr, remaining);
552       ptr += remaining;
553     }
554 
555     __m256_neon S0_next = __zxor(S0, block);
556     __m256_neon S1_next = __zxor(S1, block);
557     __m256_neon S2_next = __zxor(S2, block);
558     __m256_neon S3_next = __zxor(S3, block);
559 
560     for ( u64 r = 0; r < __zzzrounds; ++r ) {
561       __m256_neon T0 = __zslli64<3>(S0_next);
562       S0_next = __zsub64(S0_next, T0);
563       T0 = __zsrli64<2>(S0_next);
564       S0_next = __zsub64(S0_next, T0);
565       T0 = __zslli64<5>(S0_next);
566       S0_next = __zadd64(S0_next, T0);
567       T0 = __zsrli64<4>(S0_next);
568       S0_next = __zsub64(S0_next, T0);
569 
570       __m256_neon T1 = __zslli64<3>(S1_next);
571       S1_next = __zsub64(S1_next, T1);
572       T1 = __zsrli64<2>(S1_next);
573       S1_next = __zsub64(S1_next, T1);
574       T1 = __zslli64<5>(S1_next);
575       S1_next = __zadd64(S1_next, T1);
576       T1 = __zsrli64<4>(S1_next);
577       S1_next = __zsub64(S1_next, T1);
578 
579       __m256_neon T2 = __zslli64<2>(S2_next);
580       S2_next = __zsub64(S2_next, T2);
581       T2 = __zsrli64<3>(S2_next);
582       S2_next = __zsub64(S2_next, T2);
583       T2 = __zslli64<4>(S2_next);
584       S2_next = __zadd64(S2_next, T2);
585       T2 = __zsrli64<5>(S2_next);
586       S2_next = __zsub64(S2_next, T2);
587 
588       __m256_neon T3 = __zslli64<3>(S3_next);
589       S3_next = __zsub64(S3_next, T3);
590       T3 = __zsrli64<1>(S3_next);
591       S3_next = __zsub64(S3_next, T3);
592       T3 = __zslli64<6>(S3_next);
593       S3_next = __zadd64(S3_next, T3);
594       T3 = __zsrli64<3>(S3_next);
595       S3_next = __zsub64(S3_next, T3);
596     }
597 
598     S0 = S0_next;
599     S1 = S1_next;
600     S2 = S2_next;
601     S3 = S3_next;
602   }
603 
604   __m256_neon S = __zxor(__zxor(S0, S1), __zxor(S2, S3));
605   __zstore(reinterpret_cast<u8 *>(out), S);
606 }

680 template<int A, int B, int C, int D>
681 __attribute__((always_inline)) static inline __m256_neon
682 __zzzf_mix(__m256_neon s)
683 {
684   __m256_neon t = __zslli64<A>(s);
685   s = __zsub64(s, t);
686   t = __zsrli64<B>(s);
687   s = __zxor(s, t);
688   s = __zbyterev64(s);
689   t = __zslli64<C>(s);
690   s = __zadd64(s, t);
691   t = __zsrli64<D>(s);
692   s = __zxor(s, t);
693   return s;
694 }

705   __m256_neon S2 = __zxor(__zslli64<1>(IV), seedv);
706   __m256_neon S3 = __zxor(__zsrli64<1>(IV), seedv);
707 
708   const u8 *ptr = data;
709   const u8 *end = data + sz;
710 
711   while ( ptr < end ) {
712     __m256_neon block;
713     usize remaining = (usize)(end - ptr);

716       ptr += 32;
717     } else {
718       block = __zload_tail(ptr, remaining);
719       ptr += remaining;
720     }
721 
722     S0 = __zzzf_mix<3, 31, 5, 29>(__zxor(S0, block));
723     S1 = __zzzf_mix<5, 29, 7, 31>(__zxor(S1, block));
724     S2 = __zzzf_mix<7, 27, 3, 33>(__zxor(S2, block));
