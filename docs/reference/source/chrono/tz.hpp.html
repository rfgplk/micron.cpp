45 constexpr u32
46 be32(const u8 *p) noexcept
47 {
48   return (static_cast<u32>(p[0]) << 24) | (static_cast<u32>(p[1]) << 16) | (static_cast<u32>(p[2]) << 8) | static_cast<u32>(p[3]);
49 }
50 
51 constexpr i64
52 be64(const u8 *p) noexcept
53 {
54   u64 v = 0;
55   for ( usize i = 0; i < 8; ++i ) v = (v << 8) | static_cast<u64>(p[i]);
56   return static_cast<i64>(v);
57 }

59 struct header {
60   u8 version;
61   u32 isutcnt;
62   u32 isstdcnt;
63   u32 leapcnt;
64   u32 timecnt;
65   u32 typecnt;
66   u32 charcnt;
67 };

69 constexpr bool
70 read_header(const u8 *p, usize avail, header &h) noexcept
71 {
72   if ( avail < 44 ) return false;
73   if ( p[0] != 'T' || p[1] != 'Z' || p[2] != 'i' || p[3] != 'f' ) return false;
74   h.version = p[4];
75   h.isutcnt = be32(p + 20);
76   h.isstdcnt = be32(p + 24);
77   h.leapcnt = be32(p + 28);
78   h.timecnt = be32(p + 32);
79   h.typecnt = be32(p + 36);
80   h.charcnt = be32(p + 40);
81   return true;
82 }
83 
84 constexpr u64
85 block_size(const header &h, u64 ts) noexcept
86 {
87   return static_cast<u64>(h.timecnt) * ts + static_cast<u64>(h.timecnt) + static_cast<u64>(h.typecnt) * 6u + static_cast<u64>(h.charcnt)
88          + static_cast<u64>(h.leapcnt) * (ts + 4u) + static_cast<u64>(h.isstdcnt) + static_cast<u64>(h.isutcnt);
89 }

91 inline bool
92 parse_v2(const u8 *p, usize avail, i64 *trans, i32 *off, usize cap, usize &n_out, i32 &first_out) noexcept
93 {
94   header h{};
95   if ( !read_header(p, avail, h) ) return false;
96   if ( h.typecnt == 0 ) return false;
97   const u64 need = 44u + block_size(h, 8u);
98   if ( need > avail ) return false;
99 
100   const u8 *times = p + 44;
101   const u8 *idx = times + static_cast<u64>(h.timecnt) * 8u;
102   const u8 *ttinfo = idx + h.timecnt;
103 
104   u32 firstt = 0;
105   for ( u32 i = 0; i < h.typecnt; ++i ) {
106     if ( ttinfo[i * 6u + 4u] == 0 ) {
107       firstt = i;
108       break;
109     }
110   }
111   first_out = static_cast<i32>(be32(ttinfo + firstt * 6u));
112 
113   usize n = 0;
114   for ( u32 i = 0; i < h.timecnt && n < cap; ++i ) {
115     const u32 ti = idx[i];
116     if ( ti >= h.typecnt ) continue;
117     trans[n] = be64(times + static_cast<u64>(i) * 8u);
118     off[n] = static_cast<i32>(be32(ttinfo + static_cast<u64>(ti) * 6u));
119     ++n;
120   }
121   n_out = n;
122   return true;
123 }

127 template<usize N>
128 inline bool
129 parse(const u8 *data, usize len, tz_storage<N> &out) noexcept
130 {
131   out.loaded = false;
132   out.n = 0;
133   out.first = 0;
134   if ( !data || len < 44 ) return false;
135 
136   __impl::header h{};
137   if ( !__impl::read_header(data, len, h) ) return false;
138 
139   if ( h.version == '2' || h.version == '3' || h.version == '4' ) {
140     const u64 skip = 44u + __impl::block_size(h, 4u);
141     if ( skip >= len ) return false;
142     if ( !__impl::parse_v2(data + skip, len - static_cast<usize>(skip), out.trans, out.off, N, out.n, out.first) ) return false;
143     out.loaded = true;
144     return true;
145   }
146   return false;
147 }

152 inline max_t
153 slurp(const char *path, u8 *buf, usize cap) noexcept
154 {
155   const long fd = micron::syscall(SYS_openat, micron::posix::at_fdcwd, path, 0 /* O_RDONLY */, 0);
156   if ( micron::syscall_failed(fd) ) return static_cast<max_t>(fd);
157   usize got = 0;
158   for ( ;; ) {
159     if ( got >= cap ) break;
160     const long r = micron::syscall(SYS_read, fd, buf + got, cap - got);
161     if ( r < 0 ) {
162       if ( -r == static_cast<long>(error::interrupted) ) continue;
163       micron::syscall(SYS_close, fd);
164       return static_cast<max_t>(r);
165     }
166     if ( r == 0 ) break;
167     got += static_cast<usize>(r);
168   }
169   micron::syscall(SYS_close, fd);
170   return static_cast<max_t>(got);
171 }

173 constexpr bool
174 safe_zone(const char *z) noexcept
175 {
176   if ( !z || z[0] == '\0' ) return false;
177   if ( z[0] == '/' || z[0] == ':' ) return false;
178   usize n = 0;
179   for ( ; z[n]; ++n ) {
180     const char c = z[n];
181     if ( c == '.' ) return false;
182     const bool ok
183         = (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '/' || c == '_' || c == '-' || c == '+';
184     if ( !ok ) return false;
185     if ( n > 200 ) return false;
186   }
187   return n != 0;
188 }

192 template<usize N>
193 inline bool
194 load_named(const char *zone, u8 *scratch, usize scratch_len, tz_storage<N> &out) noexcept
195 {
196   out.loaded = false;
197   if ( !__impl::safe_zone(zone) ) return false;
198 
199   char path[256];
200   const char pre[] = "/usr/share/zoneinfo/";
201   usize k = 0;
202   for ( ; pre[k]; ++k ) path[k] = pre[k];
203   usize j = 0;
204   for ( ; zone[j] && k + 1 < sizeof(path); ++j, ++k ) path[k] = zone[j];
205   if ( zone[j] != '\0' ) return false;
206   path[k] = '\0';
207 
208   const max_t n = __impl::slurp(path, scratch, scratch_len);
209   if ( n <= 0 ) return false;
210   return parse(scratch, static_cast<usize>(n), out);
211 }

213 template<usize N>
214 inline bool
215 load_localtime(u8 *scratch, usize scratch_len, tz_storage<N> &out) noexcept
216 {
217   const max_t n = __impl::slurp("/etc/localtime", scratch, scratch_len);
218   if ( n <= 0 ) {
219     out.loaded = false;
220     return false;
221   }
222   return parse(scratch, static_cast<usize>(n), out);
223 }

225 template<usize N>
226 inline bool
227 load_local(const char *tz_env, u8 *scratch, usize scratch_len, tz_storage<N> &out) noexcept
228 {
229   if ( tz_env && __impl::safe_zone(tz_env) ) {
230     if ( load_named(tz_env, scratch, scratch_len, out) ) return true;
231   }
232   return load_localtime(scratch, scratch_len, out);
233 }
