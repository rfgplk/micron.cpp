74 struct scan_out {
75   elf::dyn_info_t dyn{};
76   const u8 *bias = nullptr;
77   bool ok = false;
78 };

80 [[gnu::cold, gnu::noinline]] inline scan_out
81 __scan(unsigned long base_addr) noexcept
82 {
83   scan_out o{};
84   if ( base_addr == 0 ) return o;
85   const u8 *base = reinterpret_cast<const u8 *>(base_addr);
86 
87   const elf::nehdr_t *eh = micron::ptr_cast<const elf::nehdr_t *>(base);
88   if ( eh->ident[0] != 0x7f || eh->ident[1] != 'E' || eh->ident[2] != 'L' || eh->ident[3] != 'F' ) return o;
89   if ( eh->phoff == 0 || eh->phentsize == 0 || eh->phnum == 0 ) return o;
90 
91   const u8 *ph = base + eh->phoff;
92   const elf::ndyn_t *dyn = nullptr;
93 
94   for ( elf::half i = 0; i < eh->phnum; ++i ) {
95     const elf::nphdr_t *p = micron::ptr_cast<const elf::nphdr_t *>(ph + static_cast<usize>(i) * eh->phentsize);
96     if ( p->type == elf::pt_load && o.bias == nullptr ) {
97       o.bias = reinterpret_cast<const u8 *>(base_addr + static_cast<unsigned long>(p->offset) - static_cast<unsigned long>(p->vaddr));
98     } else if ( p->type == elf::pt_dynamic ) {
99       dyn = micron::ptr_cast<const elf::ndyn_t *>(base + p->offset);
100     }
101   }
102   if ( !dyn || !o.bias ) return o;
103 
104   for ( const elf::ndyn_t *d = dyn; d->tag != elf::dt_null; ++d ) {
105     switch ( d->tag ) {
106     case elf::dt_strtab:
107       o.dyn.strtab = reinterpret_cast<const char *>(o.bias + d->un.ptr);
108       break;
109     case elf::dt_symtab:
110       o.dyn.symtab = micron::ptr_cast<const elf::nsym_t *>(o.bias + d->un.ptr);
111       break;
112     case elf::dt_hash:
113       o.dyn.hash = micron::ptr_cast<const elf::word *>(o.bias + d->un.ptr);
114       break;
115     case elf::dt_gnu_hash:
116       o.dyn.gnu_hash = micron::ptr_cast<const elf::word *>(o.bias + d->un.ptr);
117       break;
118     default:
119       break;
120     }
121   }
122   if ( !o.dyn.symtab || !o.dyn.strtab ) return o;
123   if ( !o.dyn.hash && !o.dyn.gnu_hash ) return o;
124   o.dyn.symcount = elf::count_dynsyms(o.dyn);
125   o.ok = true;
126   return o;
127 }

129 [[gnu::cold, gnu::noinline]] inline void
130 __resolve(void) noexcept
131 {
132   resolved r{};
133   r.tried = true;
134 
135   const scan_out o = __scan(micron::getauxval(micron::at_sysinfo_ehdr));
136   if ( o.ok ) {
137     for ( const char *n : __cgt_names ) {
138       const elf::nsym_t *s = elf::lookup_sym(o.dyn, n);
139       if ( s && s->value != 0 ) {
140         r.clock_gettime = reinterpret_cast<clock_gettime_fn>(const_cast<u8 *>(o.bias) + s->value);
141         break;
142       }
143     }
144     for ( const char *n : __getcpu_names ) {
145       if ( n[0] == 0 ) break;      // this arch's vDSO exports no getcpu
146       const elf::nsym_t *s = elf::lookup_sym(o.dyn, n);
147       if ( s && s->value != 0 ) {
148         r.getcpu = reinterpret_cast<getcpu_fn>(const_cast<u8 *>(o.bias) + s->value);
149         break;
150       }
151     }
152   }
153 
154   __slot = r;
155   __atomic_store_n(&__state, 1u, __ATOMIC_RELEASE);
156 }

160 inline const resolved &
161 info(void) noexcept
162 {
163   if ( __atomic_load_n(&__impl::__state, __ATOMIC_ACQUIRE) == 0u ) [[unlikely]]
164     __impl::__resolve();
165   return __impl::__slot;
166 }

168 inline bool
169 available(void) noexcept
170 {
171   return info().clock_gettime != nullptr;
172 }

174 [[gnu::always_inline]] inline ssize_t
175 clock_gettime(clockid_t clk, timespec_t &ts) noexcept
176 {
177   const clock_gettime_fn f = info().clock_gettime;
178   if ( f ) [[likely]]
179     return static_cast<ssize_t>(f(clk, &ts));
180   return micron::clock_gettime(clk, ts);
181 }
