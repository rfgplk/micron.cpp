25 // NOTE: the first three are semantic levels, not instruction names; we use amd64 spelling for convenience, other arches use their own
26 // respective insts
27 // ..none           no barrier at all
28 // ..lfence         serialise the instruction stream    x86: lfence          arm: isb
29 // ..mfence_lfence  drain memory, then serialise        x86: mfence+lfence   arm: dsb sy + isb
30 //
31 enum class serial : u8 {
32   none,               // bare counter read
33   lfence,             // serialise around it
34   mfence_lfence,      // drain the store buffer first, then serialise
35   cpuid,              // x86: fully serialising, EXPENSIVE and VM-exits
36   rdtscp,             // x86: waits for prior instructions to retire; also yields the cpu id
37   serialize,          // x86: serialize instruction, (Sapphire Rapids and later)
38   isb,                // arm: instruction-synchronisation barrier
39 };

166 [[gnu::always_inline]] inline u64
167 __tick_bare() noexcept
168 {
169 #if defined(__micron_arch_x86_any)
170   u32 lo = 0, hi = 0;
171   __asm__ __volatile__("rdtsc" : "=a"(lo), "=d"(hi));
172   return (static_cast<u64>(hi) << 32) | static_cast<u64>(lo);
173 #elif defined(__micron_arch_arm64)
174   u64 v = 0;
175   __asm__ __volatile__("mrs %0, cntvct_el0" : "=r"(v));
176   return v;
177 #elif defined(__micron_arch_arm32) && defined(MICRON_CHRONO_ARM32_CNTVCT)
178   u32 lo = 0, hi = 0;
179   __asm__ __volatile__("mrrc p15, 1, %0, %1, c14" : "=r"(lo), "=r"(hi));
180   return (static_cast<u64>(hi) << 32) | static_cast<u64>(lo);
181 #else
182   // the counter is nanoseconds here, and tick_hz() says so
183   const i64 v = clock_ns(clock_monotonic_raw);
184   return v < 0 ? 0ull : static_cast<u64>(v);
185 #endif
186 }

188 #if defined(__micron_arch_x86_any)
189 [[gnu::always_inline]] inline u64
190 __tick_p(u32 &aux) noexcept
191 {
192   u32 lo = 0, hi = 0, a = 0;
193   __asm__ __volatile__("rdtscp" : "=a"(lo), "=d"(hi), "=c"(a));
194   aux = a;
195   return (static_cast<u64>(hi) << 32) | static_cast<u64>(lo);
196 }

199 template<serial S>
200 [[gnu::always_inline]] inline void
201 __fence_before() noexcept
202 {
203   if constexpr ( S == serial::none ) {
204     fence_compiler();
205   } else if constexpr ( S == serial::mfence_lfence ) {
206     fence_mem();
207     fence_load();
208   } else if constexpr ( S == serial::cpuid || S == serial::serialize ) {
209     fence_serialising();
210   } else if constexpr ( S == serial::rdtscp ) {
211     fence_compiler();
212   } else {
213     fence_load();
214   }
215 }

219 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
220 // public reads
221 //
222 // tick()        fence, read          cheapest
223 // tick_start()  fence, read, fence   region cannot begin before the read retires
224 // tick_end()    fence, read, fence   region cannot leak past the read
225 
226 template<serial S = default_serial>
227 [[gnu::always_inline]] inline u64
228 tick() noexcept
229 {
230   static_assert(serial_supported<S>, "micron::chrono: that serialisation policy is not available on this target");
231 #if defined(__micron_arch_x86_any)
232   if constexpr ( S == serial::rdtscp ) {
233     u32 aux = 0;
234     __impl::__fence_before<S>();
235     return __impl::__tick_p(aux);
236   } else
237 #endif
238   {
239     __impl::__fence_before<S>();
240     return __impl::__tick_bare();
241   }
242 }
243 
244 template<serial S = default_serial>
245 [[gnu::always_inline]] inline u64
246 tick_start() noexcept
247 {
248   const u64 v = tick<S>();
249   fence_load();
250   return v;
251 }
252 
253 template<serial S = default_serial>
254 [[gnu::always_inline]] inline u64
255 tick_end() noexcept
256 {
257   fence_load();
258   const u64 v = tick<S>();
259   fence_load();
260   return v;
261 }

286 // rdpmc
287 // (instruction only)
288 //
289 // WARNING: this faults unless /sys/bus/event_source/devices/cpu/rdpmc allows it && the counter is actually scheduled
290 // never call it speculatively
291 
292 #if defined(__micron_arch_x86_any)
293 [[gnu::always_inline]] inline u64
294 rdpmc(u32 counter) noexcept
295 {
296   u32 lo = 0, hi = 0;
297   __asm__ __volatile__("rdpmc" : "=a"(lo), "=d"(hi) : "c"(counter));
298   return (static_cast<u64>(hi) << 32) | static_cast<u64>(lo);
299 }

301 inline constexpr u64
302 pmc_delta(u64 now, u64 then, u32 width) noexcept
303 {
304   if ( width == 0 || width >= 64 ) return now - then;
305   const u64 mask = (1ull << width) - 1ull;
306   return (now - then) & mask;
307 }
