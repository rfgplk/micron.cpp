68 constexpr bool
69 __is_digit(char c) noexcept
70 {
71   return c >= '0' && c <= '9';

74 constexpr bool
75 __fixed_digits(const char *s, usize n, usize &i, u32 count, u32 &out) noexcept
76 {
77   out = 0;
78   if ( i + count > n ) return false;
79   for ( u32 k = 0; k < count; ++k ) {
80     if ( !__is_digit(s[i]) ) return false;
81     out = out * 10u + static_cast<u32>(s[i] - '0');
82     ++i;
83   }
84   return true;
85 }

87 // 1..max digits, greedy
88 constexpr bool
89 __some_digits(const char *s, usize n, usize &i, u32 max, u64 &out, u32 &taken) noexcept
90 {
91   out = 0;
92   taken = 0;
93   while ( i < n && taken < max && __is_digit(s[i]) ) {
94     out = out * 10ull + static_cast<u64>(s[i] - '0');
95     ++i;
96     ++taken;
97   }
98   return taken != 0;
99 }

101 constexpr bool
102 __ieq(const char *s, usize n, usize i, const char *lit, usize k) noexcept
103 {
104   if ( i + k > n ) return false;
105   for ( usize j = 0; j < k; ++j ) {
106     char a = s[i + j];
107     char b = lit[j];
108     if ( a >= 'A' && a <= 'Z' ) a = static_cast<char>(a + 32);
109     if ( b >= 'A' && b <= 'Z' ) b = static_cast<char>(b + 32);
110     if ( a != b ) return false;
111   }
112   return true;
113 }

115 // [+-]YYYY[Y...] or exactly YYYY
116 constexpr bool
117 __year(const char *s, usize n, usize &i, i32 &out) noexcept
118 {
119   bool neg = false;
120   bool expanded = false;
121   if ( i < n && (s[i] == '+' || s[i] == '-') ) {
122     neg = s[i] == '-';
123     expanded = true;
124     ++i;
125   }
126   u64 v = 0;
127   u32 taken = 0;
128   if ( !__some_digits(s, n, i, expanded ? 9u : 4u, v, taken) ) return false;
129   if ( !expanded && taken != 4 ) return false;
130   if ( expanded && taken < 4 ) return false;
131   if ( v > 2'000'000'000ull ) return false;
132   out = static_cast<i32>(neg ? -static_cast<i64>(v) : static_cast<i64>(v));
133   return true;
134 }

136 // Z | z | +HH:MM | +HHMM | +HH
137 constexpr bool
138 __offset(const char *s, usize n, usize &i, i32 &out) noexcept
139 {
140   if ( i >= n ) return false;
141   if ( s[i] == 'Z' || s[i] == 'z' ) {
142     out = 0;
143     ++i;
144     return true;
145   }
146   if ( s[i] != '+' && s[i] != '-' ) return false;
147   const bool neg = s[i] == '-';
148   ++i;
149   u32 hh = 0, mm = 0;
150   if ( !__fixed_digits(s, n, i, 2, hh) ) return false;
151   if ( i < n && s[i] == ':' ) {
152     ++i;
153     if ( !__fixed_digits(s, n, i, 2, mm) ) return false;
154   } else if ( i < n && __is_digit(s[i]) ) {
155     if ( !__fixed_digits(s, n, i, 2, mm) ) return false;
156   }
157   if ( hh > 23 || mm > 59 ) return false;
158   const i32 mag = static_cast<i32>(hh) * 3600 + static_cast<i32>(mm) * 60;
159   out = neg ? -mag : mag;
160   return true;
161 }

169 inline constexpr civil_result
170 parse_iso8601(const char *s, usize n) noexcept
171 {
172   civil_result r{};
173   if ( !s || n == 0 ) {
174     r.err = parse_err::empty;
175     return r;
176   }
177   usize i = 0;
178   i32 y = 0;
179   if ( !__impl::__year(s, n, i, y) ) {
180     r.err = parse_err::bad_year;
181     return r;
182   }
183   if ( i >= n || s[i] != '-' ) {
184     r.err = parse_err::malformed;
185     return r;
186   }
187   ++i;
188   u32 mo = 0, d = 0;
189   if ( !__impl::__fixed_digits(s, n, i, 2, mo) ) {
190     r.err = parse_err::bad_month;
191     return r;
192   }
193   if ( i >= n || s[i] != '-' ) {
194     r.err = parse_err::malformed;
195     return r;
196   }
197   ++i;
198   if ( !__impl::__fixed_digits(s, n, i, 2, d) ) {
199     r.err = parse_err::bad_day;
200     return r;
201   }
202   if ( mo < 1 || mo > 12 ) {
203     r.err = parse_err::bad_month;
204     return r;
205   }
206   if ( d < 1 || d > days_in_month(y, mo) ) {
207     r.err = parse_err::bad_day;
208     return r;
209   }
210 
211   r.c.y = y;
212   r.c.mo = mo;
213   r.c.d = d;
214 
215   if ( i == n ) return r;      // date only -> midnight UTC
216 
217   if ( s[i] != 'T' && s[i] != 't' && s[i] != ' ' ) {
218     r.err = parse_err::malformed;
219     return r;
220   }
221   ++i;
222 
223   u32 h = 0, mi = 0, sec = 0;
224   if ( !__impl::__fixed_digits(s, n, i, 2, h) ) {
225     r.err = parse_err::bad_hour;
226     return r;
227   }
228   if ( i >= n || s[i] != ':' ) {
229     r.err = parse_err::malformed;
230     return r;
231   }
232   ++i;
233   if ( !__impl::__fixed_digits(s, n, i, 2, mi) ) {
234     r.err = parse_err::bad_minute;
235     return r;
236   }
237   if ( i < n && s[i] == ':' ) {
238     ++i;
239     if ( !__impl::__fixed_digits(s, n, i, 2, sec) ) {
240       r.err = parse_err::bad_second;
241       return r;
242     }
243   }
244   if ( h > 23 ) {
245     r.err = parse_err::bad_hour;
246     return r;
247   }
248   if ( mi > 59 ) {
249     r.err = parse_err::bad_minute;
250     return r;
251   }
252   // 60 is a leap second, which the calendar cannot represent but the grammar must accept
253   if ( sec > 60 ) {
254     r.err = parse_err::bad_second;
255     return r;
256   }
257   r.c.h = h;
258   r.c.mi = mi;
259   r.c.s = sec;
260 
261   if ( i < n && (s[i] == '.' || s[i] == ',') ) {
262     ++i;
263     u64 frac = 0;
264     u32 taken = 0;
265     if ( !__impl::__some_digits(s, n, i, 9u, frac, taken) ) {
266       r.err = parse_err::malformed;
267       return r;
268     }
269     for ( u32 k = taken; k < 9u; ++k ) frac *= 10ull;
270     r.c.ns = static_cast<u32>(frac);
271     // anything past nine digits is below the resolution we carry; consume and drop it
272     while ( i < n && __impl::__is_digit(s[i]) ) ++i;
273   }
274 
275   if ( i < n ) {
276     i32 off = 0;
277     if ( !__impl::__offset(s, n, i, off) ) {
278       r.err = parse_err::bad_offset;
279       return r;
280     }
281     r.c.off = off;
282     r.has_offset = true;
283   }
284 
285   if ( i != n ) {
286     r.err = parse_err::trailing;
287     return r;
288   }
289   return r;
290 }

315 inline constexpr civil_result
316 parse_rfc2822(const char *s, usize n) noexcept
317 {
318   civil_result r{};
319   if ( !s || n == 0 ) {
320     r.err = parse_err::empty;
321     return r;
322   }
323   usize i = 0;
324 
325   if ( i + 4 <= n && !__impl::__is_digit(s[i]) ) {
326     bool found = false;
327     for ( u32 k = 0; k < 7; ++k ) {
328       if ( __impl::__ieq(s, n, i, weekday_abbr[k], 3) ) {
329         found = true;
330         break;
331       }
332     }
333     if ( !found ) {
334       r.err = parse_err::bad_weekday;
335       return r;
336     }
337     i += 3;
338     if ( i < n && s[i] == ',' ) ++i;
339     while ( i < n && s[i] == ' ' ) ++i;
340   }
341 
342   u64 dv = 0;
343   u32 taken = 0;
344   if ( !__impl::__some_digits(s, n, i, 2u, dv, taken) ) {
345     r.err = parse_err::bad_day;
346     return r;
347   }
348   while ( i < n && s[i] == ' ' ) ++i;
349 
350   u32 mo = 0;
351   for ( u32 k = 1; k <= 12; ++k ) {
352     if ( __impl::__ieq(s, n, i, month_abbr[k], 3) ) {
353       mo = k;
354       break;
355     }
356   }
357   if ( mo == 0 ) {
358     r.err = parse_err::bad_month;
359     return r;
360   }
361   i += 3;
362   while ( i < n && s[i] == ' ' ) ++i;
363 
364   u64 yv = 0;
365   if ( !__impl::__some_digits(s, n, i, 4u, yv, taken) ) {
366     r.err = parse_err::bad_year;
367     return r;
368   }
369   if ( taken == 2 ) yv += (yv < 50 ? 2000 : 1900);
370   while ( i < n && s[i] == ' ' ) ++i;
371 
372   u32 h = 0, mi = 0, sec = 0;
373   if ( !__impl::__fixed_digits(s, n, i, 2, h) ) {
374     r.err = parse_err::bad_hour;
375     return r;
376   }
377   if ( i >= n || s[i] != ':' ) {
378     r.err = parse_err::malformed;
379     return r;
380   }
381   ++i;
382   if ( !__impl::__fixed_digits(s, n, i, 2, mi) ) {
383     r.err = parse_err::bad_minute;
384     return r;
385   }
386   if ( i < n && s[i] == ':' ) {
387     ++i;
388     if ( !__impl::__fixed_digits(s, n, i, 2, sec) ) {
389       r.err = parse_err::bad_second;
390       return r;
391     }
392   }
393   while ( i < n && s[i] == ' ' ) ++i;
394 
395   i32 off = 0;
396   if ( i < n ) {
397     if ( __impl::__ieq(s, n, i, "GMT", 3) || __impl::__ieq(s, n, i, "UTC", 3) ) {
398       i += 3;
399     } else if ( __impl::__ieq(s, n, i, "UT", 2) ) {
400       i += 2;
401     } else if ( !__impl::__offset(s, n, i, off) ) {
402       r.err = parse_err::bad_offset;
403       return r;
404     }
405     r.has_offset = true;
406   }
407   while ( i < n && (s[i] == ' ' || s[i] == '\r' || s[i] == '\n') ) ++i;
408   if ( i != n ) {
409     r.err = parse_err::trailing;
410     return r;
411   }
412 
413   const i32 y = static_cast<i32>(yv);
414   const u32 d = static_cast<u32>(dv);
415   if ( d < 1 || d > days_in_month(y, mo) ) {
416     r.err = parse_err::bad_day;
417     return r;
418   }
419   if ( h > 23 ) {
420     r.err = parse_err::bad_hour;
421     return r;
422   }
423   if ( mi > 59 ) {
424     r.err = parse_err::bad_minute;
425     return r;
426   }
427   if ( sec > 60 ) {
428     r.err = parse_err::bad_second;
429     return r;
430   }
431 
432   r.c.y = y;
433   r.c.mo = mo;
434   r.c.d = d;
435   r.c.h = h;
436   r.c.mi = mi;
437   r.c.s = sec;
438   r.c.off = off;
439   return r;
440 }

442 struct until_spec {
443   civil c{};
444   i64 secs = 0;      // @ form's second count
445   const char *err = nullptr;
446   bool utc = false;            // a trailing Z

448   bool time_only = false;      // HH:MM[:SS] with no date, so it rolls to the next occurrence
449 
450   constexpr bool
451   ok() const noexcept
452   {
453     return err == nullptr;
454   }

457 inline constexpr until_spec
458 parse_until(const char *s, usize n) noexcept
459 {
460   until_spec u{};
461   if ( !s || n == 0 ) {
462     u.err = parse_err::empty;
463     return u;
464   }
465   usize i = 0;
466 
467   if ( s[0] == '@' ) {
468     i = 1;
469     bool neg = false;
470     if ( i < n && (s[i] == '-' || s[i] == '+') ) {
471       neg = s[i] == '-';
472       ++i;
473     }
474     u64 v = 0;
475     u32 taken = 0;
476     if ( !__impl::__some_digits(s, n, i, 19u, v, taken) ) {
477       u.err = parse_err::malformed;
478       return u;
479     }
480     if ( i != n ) {
481       u.err = parse_err::trailing;
482       return u;
483     }
484     if ( v > 0x7FFF'FFFF'FFFF'FFFFull ) {
485       u.err = parse_err::overflow;
486       return u;
487     }
488     u.epoch = true;
489     u.secs = neg ? -static_cast<i64>(v) : static_cast<i64>(v);
490     return u;
491   }
492 
493   usize len = n;
494   if ( len > 0 && (s[len - 1] == 'Z' || s[len - 1] == 'z') ) {
495     u.utc = true;
496     --len;
497   }
498 
499   // HH:MM[:SS]
500   bool has_dash = false;
501   for ( usize k = 0; k < len; ++k ) {
502     if ( s[k] == '-' ) {
503       has_dash = true;
504       break;
505     }
506   }
507 
508   if ( !has_dash ) {
509     u32 h = 0, mi = 0, sec = 0;
510     if ( !__impl::__fixed_digits(s, len, i, 2, h) ) {
511       u.err = parse_err::bad_hour;
512       return u;
513     }
514     if ( i >= len || s[i] != ':' ) {
515       u.err = parse_err::malformed;
516       return u;
517     }
518     ++i;
519     if ( !__impl::__fixed_digits(s, len, i, 2, mi) ) {
520       u.err = parse_err::bad_minute;
521       return u;
522     }
523     if ( i < len && s[i] == ':' ) {
524       ++i;
525       if ( !__impl::__fixed_digits(s, len, i, 2, sec) ) {
526         u.err = parse_err::bad_second;
527         return u;
528       }
529     }
530     if ( i != len ) {
531       u.err = parse_err::trailing;
532       return u;
533     }
534     if ( h > 23 ) {
535       u.err = parse_err::bad_hour;
536       return u;
537     }
538     if ( mi > 59 ) {
539       u.err = parse_err::bad_minute;
540       return u;
541     }
542     if ( sec > 59 ) {
543       u.err = parse_err::bad_second;
544       return u;
545     }
546     u.time_only = true;
547     u.c.h = h;
548     u.c.mi = mi;
549     u.c.s = sec;
550     return u;
551   }
552 
553   const civil_result r = parse_iso8601(s, len);
554   if ( !r.ok() ) {
555     u.err = r.err;
556     return u;
557   }
558   if ( r.has_offset ) {
559     u.err = parse_err::trailing;
560     return u;
561   }
562   u.c = r.c;
563   return u;
564 }

566 inline constexpr i64
567 until_epoch(const until_spec &u, i64 now, const tz_table &t) noexcept
568 {
569   if ( u.epoch ) return u.secs;
570   civil c = u.c;
571   if ( u.time_only ) {
572     const civil today = to_civil(now, u.utc ? 0 : offset_at(t, now));
573     c.y = today.y;
574     c.mo = today.mo;
575     c.d = today.d;
576   }
577   i64 e = u.utc ? civil_secs(c) : from_civil(c, t);
578   if ( u.time_only && e <= now ) {
579     civil mid = c;
580     mid.h = 0;
581     mid.mi = 0;
582     mid.s = 0;
583     const civil next = to_civil(civil_secs(mid) + 86400, 0);
584     c.y = next.y;
585     c.mo = next.mo;
586     c.d = next.d;
587     e = u.utc ? civil_secs(c) : from_civil(c, t);
588   }
589   return e;
590 }

608 // WARNING: longest match first; ms is milliseconds and m is minutes
609 constexpr u64
610 __dur_unit(const char *s, usize n, usize i, u32 &len) noexcept
611 {
612   if ( i + 1 < n ) {
613     if ( s[i] == 'n' && s[i + 1] == 's' ) {
614       len = 2;
615       return 1ull;
616     }
617     if ( s[i] == 'u' && s[i + 1] == 's' ) {
618       len = 2;
619       return ns_per_us;
620     }
621     if ( s[i] == 'm' && s[i + 1] == 's' ) {
622       len = 2;
623       return ns_per_ms;
624     }
625   }
626   if ( i < n ) {
627     len = 1;
628     switch ( s[i] ) {
629     case 's':
630       return ns_per_s;
631     case 'm':
632       return ns_per_min;
633     case 'h':
634       return ns_per_hour;
635     case 'd':
636       return ns_per_day;
637     case 'w':
638       return ns_per_week;
639     default:
640       break;
641     }
642   }
643   len = 0;
644   return ns_per_s;
645 }

665 inline constexpr dur_result
666 parse_duration_ns(const char *s, usize n) noexcept
667 {
668   dur_result r{};
669   if ( !s || n == 0 ) {
670     r.err = parse_err::empty;
671     return r;
672   }
673 
674   if ( __impl::__ieq(s, n, 0, "infinity", 8) && n == 8 ) {
675     r.forever = true;
676     r.ns = ~0ull;
677     return r;
678   }
679   if ( __impl::__ieq(s, n, 0, "inf", 3) && n == 3 ) {
680     r.forever = true;
681     r.ns = ~0ull;
682     return r;
683   }
684 
685   usize i = 0;
686   u64 total = 0;
687   bool any = false;
688 
689   while ( i < n ) {
690     u64 whole = 0;
691     u32 wtaken = 0;
692     if ( !__impl::__some_digits(s, n, i, 19u, whole, wtaken) ) {
693       r.err = parse_err::malformed;
694       return r;
695     }
696 
697     u64 frac = 0;
698     u32 ftaken = 0;
699     if ( i < n && s[i] == '.' ) {
700       ++i;
701       // NOTE: nine digits is the whole resolution; more of them cannot change a nanosecond count
702       if ( !__impl::__some_digits(s, n, i, 9u, frac, ftaken) ) {
703         r.err = parse_err::malformed;
704         return r;
705       }
706       while ( i < n && __impl::__is_digit(s[i]) ) ++i;
707     }
708 
709     u32 slen = 0;
710     const u64 scale = __impl::__dur_unit(s, n, i, slen);
711     i += slen;
712 
713     u64 part = 0;
714     if ( !__impl::__dur_mul(whole, scale, part) ) {
715       r.err = parse_err::overflow;
716       return r;
717     }
718     if ( ftaken != 0 ) {
719       u64 num = 0;
720       if ( !__impl::__dur_mul(frac, scale, num) ) {
721         r.err = parse_err::overflow;
722         return r;
723       }
724       if ( !__impl::__dur_add(part, num / __impl::__dur_pow10(ftaken)) ) {
725         r.err = parse_err::overflow;
726         return r;
727       }
728     }
729     if ( !__impl::__dur_add(total, part) ) {
730       r.err = parse_err::overflow;
731       return r;
732     }
733     any = true;
734 
735     if ( slen == 0 && i != n ) {
736       r.err = parse_err::trailing;
737       return r;
738     }
739   }
740 
741   if ( !any ) {
742     r.err = parse_err::malformed;
743     return r;
744   }
745   r.ns = total;
746   return r;
747 }
