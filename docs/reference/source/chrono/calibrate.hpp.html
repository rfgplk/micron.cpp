39 // exact tick <-> nanosecond conversion
40 //
41 // WARNING: never through f64; stops being exact after ~9e15
42 
43 inline constexpr u64
44 cycles_to_ns(u64 cycles, u64 hz) noexcept
45 {
46   if ( hz == 0 ) return 0;
47 #if defined(__micron_has_int128)
48   return static_cast<u64>((static_cast<uint128_t>(cycles) * 1'000'000'000ull) / hz);
49 #else
50   // exact and overflow-free without a wide type
51   const u64 q = cycles / hz;
52   const u64 r = cycles % hz;
53   return q * 1'000'000'000ull + (r * 1'000'000'000ull) / hz;
54 #endif
55 }
56 
57 inline constexpr u64
58 ns_to_cycles(u64 ns, u64 hz) noexcept
59 {
60 #if defined(__micron_has_int128)
61   return static_cast<u64>((static_cast<uint128_t>(ns) * hz) / 1'000'000'000ull);
62 #else
63   const u64 q = ns / 1'000'000'000ull;
64   const u64 r = ns % 1'000'000'000ull;
65   return q * hz + (r * hz) / 1'000'000'000ull;
66 #endif
67 }

69 inline constexpr u64
70 cycles_to_us(u64 cycles, u64 hz) noexcept
71 {
72   return hz == 0 ? 0 : cycles_to_ns(cycles, hz) / 1'000ull;
73 }

75 struct tsc_traits {
76   bool present = false;
77   bool invariant = false;          // CPUID.80000007H:EDX[8]
78   bool has_rdtscp = false;         // CPUID.80000001H:EDX[27]
79   bool has_serialize = false;      // CPUID.7.0:EDX[14]
80   u64 nominal_hz = 0;              // from CPUID leaf 0x15 / 0x16; 0 when the CPU does not say
81 };
82 
83 inline tsc_traits
84 counter_traits(void) noexcept
85 {
86   tsc_traits t{};
87 #if defined(__micron_arch_x86_any)
88   t.present = true;
89   if ( micron::__cpuid_has_leaf(0x8000'0007u) ) t.invariant = (micron::__cpuid_read(0x8000'0007u).edx >> 8) & 1u;
90   if ( micron::__cpuid_has_leaf(0x8000'0001u) ) t.has_rdtscp = (micron::__cpuid_read(0x8000'0001u).edx >> 27) & 1u;
91   if ( micron::__cpuid_has_leaf(7u) ) t.has_serialize = (micron::__cpuid_read(7u, 0u).edx >> 14) & 1u;
92 
93   // leaf 0x15: ecx * ebx / eax
94   if ( micron::__cpuid_has_leaf(0x15u) ) {
95     const cpuid_regs r = micron::__cpuid_read(0x15u, 0u);
96     if ( r.eax != 0 && r.ebx != 0 && r.ecx != 0 )
97       t.nominal_hz = (static_cast<u64>(r.ecx) * static_cast<u64>(r.ebx)) / static_cast<u64>(r.eax);
98   }
99   // leaf 0x16: base frequency in MHz
100   if ( t.nominal_hz == 0 && micron::__cpuid_has_leaf(0x16u) ) {
101     const u32 mhz = micron::__cpuid_read(0x16u).eax & 0xFFFFu;
102     if ( mhz != 0 ) t.nominal_hz = static_cast<u64>(mhz) * 1'000'000ull;
103   }
104 #elif defined(__micron_arch_arm64)
105   t.present = true;
106   t.invariant = true;
107   u64 f = 0;
108   __asm__ __volatile__("mrs %0, cntfrq_el0" : "=r"(f));
109   t.nominal_hz = f;
110 #elif defined(__micron_arch_arm32) && defined(MICRON_CHRONO_ARM32_CNTVCT)
111   t.present = true;
112   t.invariant = true;
113   u32 f = 0;
114   __asm__ __volatile__("mrc p15, 0, %0, c14, c0, 0" : "=r"(f));
115   t.nominal_hz = f;
116 #endif
117   return t;
118 }

122 inline void
123 __sort_u64(u64 *v, u32 n) noexcept
124 {
125   for ( u32 i = 1; i < n; ++i ) {
126     const u64 key = v[i];
127     u32 j = i;
128     while ( j > 0 && v[j - 1] > key ) {
129       v[j] = v[j - 1];
130       --j;
131     }
132     v[j] = key;
133   }
134 }
135 
136 inline u64
137 __median_u64(u64 *v, u32 n) noexcept
138 {
139   if ( n == 0 ) return 0;
140   __sort_u64(v, n);
141   return v[(n - 1) / 2];
142 }

144 inline u64
145 __min_u64(const u64 *v, u32 n) noexcept
146 {
147   if ( n == 0 ) return 0;
148   u64 m = v[0];
149   for ( u32 i = 1; i < n; ++i )
150     if ( v[i] < m ) m = v[i];
151   return m;
152 }

161 inline u64
162 calibrate_tick_hz(u64 window_ns = 20'000'000ull) noexcept
163 {
164   const i64 t0 = clock_ns(clock_monotonic);
165   if ( t0 < 0 ) return 0;
166   const u64 c0 = tick<serial::mfence_lfence>();
167   i64 t1 = t0;
168   while ( static_cast<u64>(t1 - t0) < window_ns ) {
169     t1 = clock_ns(clock_monotonic);
170     if ( t1 < 0 ) return 0;
171   }
172   const u64 c1 = tick<serial::mfence_lfence>();
173   const u64 dt = static_cast<u64>(t1 - t0);
174   if ( dt == 0 || c1 <= c0 ) return 0;
175   return static_cast<u64>((static_cast<u64>(c1 - c0) * 1'000'000'000ull) / dt);
176 }

178 inline u64
179 tick_hz(void) noexcept
180 {
181   if ( __atomic_load_n(&__impl::__tick_hz_state, __ATOMIC_ACQUIRE) != 0u ) [[likely]]
182     return __impl::__tick_hz_slot;
183 
184   u64 hz = 0;
185   if constexpr ( !counter_is_native ) {
186     hz = 1'000'000'000ull;
187   } else {
188     const tsc_traits t = counter_traits();
189     hz = t.nominal_hz;
190     if ( hz == 0 ) hz = calibrate_tick_hz();
191     if ( hz == 0 ) hz = 1'000'000'000ull;
192   }
193   __impl::__tick_hz_slot = hz;
194   __atomic_store_n(&__impl::__tick_hz_state, 1u, __ATOMIC_RELEASE);
195   return hz;
196 }

238 #if defined(__micron_arch_amd64) || defined(__micron_arch_x86) || defined(__micron_arch_arm64) || defined(__micron_arch_arm32)
239 using arch::calib_kernel;
240 inline constexpr bool __have_calib_kernel = true;

242 [[gnu::noinline]] inline void
243 calib_kernel(i64 iters) noexcept
244 {
245   if ( iters <= 0 ) return;
246   iters &= ~static_cast<i64>(3);
247   do {
248     __asm__ __volatile__("" : "+r"(iters));
249     --iters;
250     __asm__ __volatile__("" : "+r"(iters));
251     --iters;
252     __asm__ __volatile__("" : "+r"(iters));
253     --iters;
254     --iters;
255   } while ( iters > 0 );
256 }

262 inline u64
263 __calib_sample(void) noexcept
264 {
265   const i64 t0 = clock_ns(clock_monotonic);
266   __impl::calib_kernel(calib_iters);
267   const i64 t1 = clock_ns(clock_monotonic);
268   __impl::calib_kernel(calib_iters * 2);
269   const i64 t2 = clock_ns(clock_monotonic);
270   if ( t0 < 0 || t1 < 0 || t2 < 0 ) return 0;
271   const i64 d = (t2 - t1) - (t1 - t0);
272   return d > 0 ? static_cast<u64>(d) : 0;
273 }

275 // WARNING: pin the thread before running this
276 inline u64
277 calibrate_core_hz(void) noexcept
278 {
279   const i64 w0 = clock_ns(clock_monotonic);
280   if ( w0 >= 0 ) {
281     for ( ;; ) {
282       (void)__calib_sample();
283       const i64 wn = clock_ns(clock_monotonic);
284       if ( wn < 0 || static_cast<u64>(wn - w0) >= calib_warmup_ns ) break;
285     }
286   }
287 
288   u64 samples[calib_tries];
289   u32 got = 0;
290   for ( u32 i = 0; i < calib_tries; ++i ) {
291     const u64 ns = __calib_sample();
292     if ( ns != 0 ) samples[got++] = ns;
293   }
294   if ( got == 0 ) return 0;
295 
296   // WARNING: on a part that boosts, "the highest frequency observed" is the boost clock
297   const u64 fastest = __impl::__min_u64(samples, got);
298   if ( fastest == 0 ) return 0;
299   // calib_iters extra iterations cost calib_iters core cycles
300   return (static_cast<u64>(calib_iters) * 1'000'000'000ull) / fastest;
301 }

310 inline freq_spread
311 core_hz_spread(void) noexcept
312 {
313   freq_spread s{};
314   u64 v[calib_tries];
315   u32 got = 0;
316   for ( u32 i = 0; i < calib_tries; ++i ) {
317     const u64 ns = __calib_sample();
318     if ( ns != 0 ) v[got++] = (static_cast<u64>(calib_iters) * 1'000'000'000ull) / ns;
319   }
320   if ( got == 0 ) return s;
321   __impl::__sort_u64(v, got);
322   s.samples = got;
323   s.min_hz = v[0];
324   s.median_hz = v[(got - 1) / 2];
325   s.max_hz = v[got - 1];
326   return s;
327 }

329 inline u64
330 core_hz(void) noexcept
331 {
332   if ( __atomic_load_n(&__impl::__core_hz_state, __ATOMIC_ACQUIRE) != 0u ) [[likely]]
333     return __impl::__core_hz_slot;
334   u64 hz = calibrate_core_hz();
335   if ( hz == 0 ) hz = tick_hz();
336   __impl::__core_hz_slot = hz;
337   __atomic_store_n(&__impl::__core_hz_state, 1u, __ATOMIC_RELEASE);
338   return hz;
339 }

362 template<serial S = default_serial>
363 inline timer_stats
364 timer_overhead(u32 samples = 65) noexcept
365 {
366   constexpr u32 __cap = 257;
367   if ( samples > __cap ) samples = __cap;
368   if ( samples < 3 ) samples = 3;
369   u64 v[__cap];
370 
371   for ( u32 i = 0; i < 8; ++i ) {
372     const u64 a = tick_start<S>();
373     const u64 b = tick_end<S>();
374     (void)a;
375     (void)b;
376   }
377   for ( u32 i = 0; i < samples; ++i ) {
378     const u64 a = tick_start<S>();
379     const u64 b = tick_end<S>();
380     v[i] = b > a ? b - a : 0;
381   }
382 
383   timer_stats s{};
384   s.min_ticks = __impl::__min_u64(v, samples);
385   __impl::__sort_u64(v, samples);
386   s.median_ticks = v[(samples - 1) / 2];
387   s.max_ticks = v[samples - 1];
388   const u64 hz = tick_hz();
389   s.min_ns = cycles_to_ns(s.min_ticks, hz);
390   s.median_ns = cycles_to_ns(s.median_ticks, hz);
391   return s;
392 }

394 template<serial S = default_serial>
395 inline timer_stats
396 timer_resolution(u32 samples = 65) noexcept
397 {
398   constexpr u32 __cap = 257;
399   if ( samples > __cap ) samples = __cap;
400   if ( samples < 3 ) samples = 3;
401   u64 v[__cap];
402   u32 got = 0;
403 
404   for ( u32 i = 0; i < samples * 4 && got < samples; ++i ) {
405     const u64 a = tick<S>();
406     u64 b = a;
407     for ( u32 k = 0; k < 4096 && b == a; ++k ) b = tick<S>();
408     if ( b > a ) v[got++] = b - a;
409   }
410   timer_stats s{};
411   if ( got == 0 ) return s;
412   s.min_ticks = __impl::__min_u64(v, got);
413   __impl::__sort_u64(v, got);
414   s.median_ticks = v[(got - 1) / 2];
415   s.max_ticks = v[got - 1];
416   const u64 hz = tick_hz();
417   s.min_ns = cycles_to_ns(s.min_ticks, hz);
418   s.median_ns = cycles_to_ns(s.median_ticks, hz);
419   return s;
420 }

422 inline u64
423 clock_cost_ns(clockid_t clk, u32 reps = 1000) noexcept
424 {
425   if ( reps == 0 ) reps = 1;
426   const i64 t0 = clock_ns(clock_monotonic);
427   if ( t0 < 0 ) return 0;
428   i64 sink = 0;
429   for ( u32 i = 0; i < reps; ++i ) {
430     sink += clock_ns(clk);
431     __asm__ __volatile__("" : "+r"(sink));
432   }
433   const i64 t1 = clock_ns(clock_monotonic);
434   if ( t1 < 0 || t1 <= t0 ) return 0;
435   return static_cast<u64>(t1 - t0) / reps;
436 }

438 struct clock_report {
439   u64 tick_hz = 0;
440   u64 core_hz = 0;
441   bool counter_native = false;
442   bool counter_invariant = false;
443   bool has_rdtscp = false;
444   bool has_serialize = false;
445   u64 counter_overhead_ticks = 0;
446   u64 counter_resolution_ticks = 0;
447   u64 realtime_cost_ns = 0;
448   u64 monotonic_cost_ns = 0;
449   u64 monotonic_coarse_cost_ns = 0;
450   u64 monotonic_res_ns = 0;

453 inline clock_report
454 report(void) noexcept
455 {
456   clock_report r{};
457   const tsc_traits t = counter_traits();
458   r.tick_hz = tick_hz();
459   r.core_hz = core_hz();
460   r.counter_native = counter_is_native;
461   r.counter_invariant = t.invariant;
462   r.has_rdtscp = t.has_rdtscp;
463   r.has_serialize = t.has_serialize;
464   r.counter_overhead_ticks = timer_overhead<>().min_ticks;
465   r.counter_resolution_ticks = timer_resolution<>().min_ticks;
466   r.realtime_cost_ns = clock_cost_ns(clock_realtime);
467   r.monotonic_cost_ns = clock_cost_ns(clock_monotonic);
468   r.monotonic_coarse_cost_ns = clock_cost_ns(clock_monotonic_coarse);
469   const i64 res = clock_resolution_ns(clock_monotonic);
470   r.monotonic_res_ns = res < 0 ? 0 : static_cast<u64>(res);
471   return r;
472 }
