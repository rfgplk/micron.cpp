19 struct civil {
20   i32 y = 1970;
21   u32 mo = 1;
22   u32 d = 1;
23   u32 h = 0;
24   u32 mi = 0;
25   u32 s = 0;
26   u32 ns = 0;
27   i32 off = 0;
28 
29   constexpr bool
30   valid() const noexcept
31   {
32     return civil_valid(y, mo, d) && h < 24 && mi < 60 && s < 61 && ns < 1'000'000'000u;
33   }
34 
35   constexpr bool
36   operator==(const civil &o) const noexcept
37   {
38     return y == o.y && mo == o.mo && d == o.d && h == o.h && mi == o.mi && s == o.s && ns == o.ns && off == o.off;
39   }
40 
41   constexpr bool
42   operator!=(const civil &o) const noexcept
43   {
44     return !(*this == o);
45   }

73 inline constexpr i32
74 offset_at(const tz_table &t, i64 utc) noexcept
75 {
76   if ( !t.loaded || t.n == 0 || t.trans == nullptr ) return t.loaded ? t.first : 0;
77   if ( utc < t.trans[0] ) return t.first;
78   usize lo = 0, hi = t.n - 1;
79   while ( lo < hi ) {
80     const usize mid = lo + (hi - lo + 1) / 2;
81     if ( t.trans[mid] <= utc )
82       lo = mid;
83     else
84       hi = mid - 1;
85   }
86   return t.off[lo];
87 }

89 inline constexpr civil
90 to_civil(i64 utc_sec, i32 off) noexcept
91 {
92   const i64 local = utc_sec + static_cast<i64>(off);
93   const i64 days = floor_div(local, 86400);
94   const i64 rem = local - days * 86400;
95   const civil_date cd = civil_from_days(days);
96   civil c{};
97   c.y = cd.y;
98   c.mo = cd.m;
99   c.d = cd.d;
100   c.h = static_cast<u32>(rem / 3600);
101   c.mi = static_cast<u32>((rem % 3600) / 60);
102   c.s = static_cast<u32>(rem % 60);
103   c.ns = 0;
104   c.off = off;
105   return c;
106 }

136 {
137   return days_from_civil(c.y, c.mo, c.d) * 86400ll + static_cast<i64>(c.h) * 3600 + static_cast<i64>(c.mi) * 60 + static_cast<i64>(c.s);

140 inline constexpr i64
141 from_civil(const civil &c, const tz_table &t) noexcept
142 {
143   const i64 local = civil_secs(c);
144   if ( !t.loaded || t.n == 0 || t.trans == nullptr ) return local - static_cast<i64>(t.loaded ? t.first : 0);
145 
146   const i32 a = offset_at(t, local - 86400);
147   const i32 b = offset_at(t, local + 86400);
148 
149   const i64 ca = local - static_cast<i64>(a);
150   const i64 cb = local - static_cast<i64>(b);
151 
152   const bool va = offset_at(t, ca) == a;
153   const bool vb = offset_at(t, cb) == b;
154 
155   if ( va && vb ) return ca < cb ? ca : cb;
156   if ( va ) return ca;
157   if ( vb ) return cb;
158 
159   return ca > cb ? ca : cb;
160 }
