20 [[gnu::noinline]] inline void
21 calib_kernel(i64 iters64) noexcept
22 {
23   i32 iters = static_cast<i32>(iters64 > 0x3FFF'FFFF ? 0x3FFF'FFFF : iters64);
24   __asm__ __volatile__("mov r3, #1\n\t"
25                        ".p2align 5\n"
26                        "1:\n\t"
27                        "subs %0, %0, r3\n\t"
28                        "subs %0, %0, r3\n\t"
29                        "bge 1b"
30                        : "+r"(iters)
31                        :
32                        : "r3", "cc");
33 }
