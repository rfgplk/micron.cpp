47 inline constexpr chrono::civil
48 civil_of_tm(const tm_t &t) noexcept
49 {
50   chrono::civil c{};
51   c.y = t.tm_year + 1900;
52   c.mo = static_cast<u32>(t.tm_mon + 1);
53   c.d = static_cast<u32>(t.tm_mday);
54   c.h = static_cast<u32>(t.tm_hour);
55   c.mi = static_cast<u32>(t.tm_min);
56   c.s = static_cast<u32>(t.tm_sec);
57   c.off = static_cast<i32>(t.tm_gmtoff);
58   return c;
59 }
60 
61 inline constexpr void
62 tm_of_civil(const chrono::civil &c, tm_t &out) noexcept
63 {
64   out.tm_sec = static_cast<int>(c.s);
65   out.tm_min = static_cast<int>(c.mi);
66   out.tm_hour = static_cast<int>(c.h);
67   out.tm_mday = static_cast<int>(c.d);
68   out.tm_mon = static_cast<int>(c.mo) - 1;
69   out.tm_year = c.y - 1900;
70   out.tm_wday = static_cast<int>(chrono::weekday_of(c.y, c.mo, c.d));
71   out.tm_yday = static_cast<int>(chrono::day_of_year(c.y, c.mo, c.d)) - 1;
72   out.tm_gmtoff = c.off;
73 }

78 inline tm_t *
79 gmtime_r(const time_t *t, tm_t *out) noexcept
80 {
81   if ( !t || !out ) return nullptr;
82   tm_of_civil(chrono::civil_utc(static_cast<i64>(*t)), *out);
83   out->tm_isdst = 0;
84   out->tm_gmtoff = 0;
85   out->tm_zone = "UTC";
86   return out;
87 }

89 inline tm_t *
90 localtime_r(const time_t *t, tm_t *out, const chrono::tz_table &z) noexcept
91 {
92   if ( !t || !out ) return nullptr;
93   const i32 off = chrono::offset_at(z, static_cast<i64>(*t));
94   tm_of_civil(chrono::to_civil(static_cast<i64>(*t), off), *out);
95   out->tm_isdst = -1;
96   out->tm_gmtoff = off;
97   out->tm_zone = nullptr;
98   return out;
99 }

101 inline time_t
102 timegm(tm_t *t) noexcept
103 {
104   if ( !t ) return static_cast<time_t>(-1);
105   const chrono::civil in = civil_of_tm(*t);
106   const i64 secs = chrono::civil_secs(in);
107   tm_of_civil(chrono::civil_utc(secs), *t);
108   t->tm_isdst = 0;
109   t->tm_gmtoff = 0;
110   return static_cast<time_t>(secs);
111 }

113 inline time_t
114 mktime(tm_t *t, const chrono::tz_table &z) noexcept
115 {
116   if ( !t ) return static_cast<time_t>(-1);
117   const chrono::civil in = civil_of_tm(*t);
118   const i64 secs = chrono::from_civil(in, z);
119   const i32 off = chrono::offset_at(z, secs);
120   tm_of_civil(chrono::to_civil(secs, off), *t);
121   t->tm_gmtoff = off;
122   return static_cast<time_t>(secs);
123 }

128 inline constexpr usize asctime_buf = 26;
129 
130 inline char *
131 asctime_r(const tm_t *t, char *buf) noexcept
132 {
133   if ( !t || !buf ) return nullptr;
134   const chrono::civil c = civil_of_tm(*t);
135   if ( chrono::asctime_size(c) + 2 > asctime_buf ) {
136     buf[0] = '\0';
137     return nullptr;
138   }
139   const usize n = chrono::write_asctime(buf, asctime_buf - 2, c);
140   if ( n == 0 ) {
141     buf[0] = '\0';
142     return nullptr;
143   }
144   buf[n] = '\n';
145   buf[n + 1] = '\0';
146   return buf;
147 }

149 inline char *
150 ctime_r(const time_t *t, char *buf, const chrono::tz_table &z) noexcept
151 {
152   if ( !t || !buf ) return nullptr;
153   tm_t tmp{};
154   if ( !localtime_r(t, &tmp, z) ) {
155     buf[0] = '\0';
156     return nullptr;
157   }
158   return asctime_r(&tmp, buf);
159 }

161 inline usize
162 strftime(char *s, usize max, const char *format, const tm_t *t) noexcept
163 {
164   if ( !s || !format || !t || max == 0 ) return 0;
165   const chrono::civil c = civil_of_tm(*t);
166   const usize need = chrono::strftime_size(format, c);
167   if ( need + 1 > max ) return 0;
168   const usize n = chrono::write_strftime(s, max - 1, format, c);
169   s[n] = '\0';
170   return n;
171 }

179 constexpr bool
180 __sp_digits(const char *&p, const char *end, u32 max, i64 &out, bool &any) noexcept
181 {
182   out = 0;
183   any = false;
184   u32 taken = 0;
185   bool neg = false;
186   if ( p < end && (*p == '-' || *p == '+') ) {
187     neg = *p == '-';
188     ++p;
189   }
190   while ( p < end && taken < max && *p >= '0' && *p <= '9' ) {
191     out = out * 10 + (*p - '0');
192     ++p;
193     ++taken;
194     any = true;
195   }
196   if ( neg ) out = -out;
197   return any;
198 }

200 constexpr int
201 __sp_match(const char *&p, const char *end, const char *const *tbl, u32 lo, u32 hi, u32 abbr_len) noexcept
202 {
203   for ( u32 k = lo; k <= hi; ++k ) {
204     const char *w = tbl[k];
205     // the full name first, so "June" is not consumed as "Jun" leaving a stray 'e'
206     usize full = 0;
207     while ( w[full] ) ++full;
208     if ( chrono::__impl::__ieq(p, static_cast<usize>(end - p), 0, w, full) ) {
209       p += full;
210       return static_cast<int>(k);
211     }
212     if ( chrono::__impl::__ieq(p, static_cast<usize>(end - p), 0, w, abbr_len) ) {
213       p += abbr_len;
214       return static_cast<int>(k);
215     }
216   }
217   return -1;
218 }

222 inline const char *
223 strptime(const char *s, const char *format, tm_t *t) noexcept
224 {
225   if ( !s || !format || !t ) return nullptr;
226   const char *p = s;
227   const char *end = s;
228   while ( *end ) ++end;
229 
230   bool have_ymd = false;
231   bool pm = false;
232   bool have_ampm = false;
233 
234   for ( const char *f = format; *f; ++f ) {
235     if ( *f == ' ' || *f == '\t' || *f == '\n' ) {
236       while ( p < end && (*p == ' ' || *p == '\t' || *p == '\n') ) ++p;
237       continue;
238     }
239     if ( *f != '%' ) {
240       if ( p >= end || *p != *f ) return nullptr;
241       ++p;
242       continue;
243     }
244     ++f;
245     if ( *f == '\0' ) return nullptr;
246 
247     i64 v = 0;
248     bool any = false;
249     switch ( *f ) {
250     case 'Y':
251       if ( !__impl::__sp_digits(p, end, 6, v, any) ) return nullptr;
252       t->tm_year = static_cast<int>(v - 1900);
253       have_ymd = true;
254       break;
255     case 'y':
256       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
257       t->tm_year = static_cast<int>(v < 69 ? v + 100 : v);
258       have_ymd = true;
259       break;
260     case 'C':
261       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
262       t->tm_year = static_cast<int>(v * 100 - 1900);
263       break;
264     case 'm':
265       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
266       if ( v < 1 || v > 12 ) return nullptr;
267       t->tm_mon = static_cast<int>(v - 1);
268       have_ymd = true;
269       break;
270     case 'd':
271     case 'e':
272       while ( p < end && *p == ' ' ) ++p;
273       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
274       if ( v < 1 || v > 31 ) return nullptr;
275       t->tm_mday = static_cast<int>(v);
276       have_ymd = true;
277       break;
278     case 'H':
279       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
280       if ( v > 23 ) return nullptr;
281       t->tm_hour = static_cast<int>(v);
282       break;
283     case 'I':
284       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
285       if ( v < 1 || v > 12 ) return nullptr;
286       t->tm_hour = static_cast<int>(v % 12);
287       break;
288     case 'M':
289       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
290       if ( v > 59 ) return nullptr;
291       t->tm_min = static_cast<int>(v);
292       break;
293     case 'S':
294       if ( !__impl::__sp_digits(p, end, 2, v, any) ) return nullptr;
295       if ( v > 60 ) return nullptr;
296       t->tm_sec = static_cast<int>(v);
297       break;
298     case 'j':
299       if ( !__impl::__sp_digits(p, end, 3, v, any) ) return nullptr;
300       if ( v < 1 || v > 366 ) return nullptr;
301       t->tm_yday = static_cast<int>(v - 1);
302       break;
303     case 'b':
304     case 'B':
305     case 'h': {
306       const int m = __impl::__sp_match(p, end, chrono::month_name, 1, 12, 3);
307       if ( m < 0 ) return nullptr;
308       t->tm_mon = m - 1;
309       have_ymd = true;
310       break;
311     }
312     case 'a':
313     case 'A': {
314       const int w = __impl::__sp_match(p, end, chrono::weekday_name, 0, 6, 3);
315       if ( w < 0 ) return nullptr;
316       t->tm_wday = w;
317       break;
318     }
319     case 'p':
320     case 'P': {
321       if ( chrono::__impl::__ieq(p, static_cast<usize>(end - p), 0, "AM", 2) ) {
322         pm = false;
323         have_ampm = true;
324         p += 2;
325       } else if ( chrono::__impl::__ieq(p, static_cast<usize>(end - p), 0, "PM", 2) ) {
326         pm = true;
327         have_ampm = true;
328         p += 2;
329       } else {
330         return nullptr;
331       }
332       break;
333     }
334     case 'z': {
335       usize i = 0;
336       i32 off = 0;
337       if ( !chrono::__impl::__offset(p, static_cast<usize>(end - p), i, off) ) return nullptr;
338       p += i;
339       t->tm_gmtoff = off;
340       break;
341     }
342     case 's':
343       if ( !__impl::__sp_digits(p, end, 19, v, any) ) return nullptr;
344       tm_of_civil(chrono::civil_utc(v), *t);
345       have_ymd = true;
346       break;
347     case 'n':
348     case 't':
349       while ( p < end && (*p == ' ' || *p == '\t' || *p == '\n') ) ++p;
350       break;
351     case '%':
352       if ( p >= end || *p != '%' ) return nullptr;
353       ++p;
354       break;
355     case 'T':
356     case 'X': {
357       const char *r = strptime(p, "%H:%M:%S", t);
358       if ( !r ) return nullptr;
359       p = r;
360       break;
361     }
362     case 'R': {
363       const char *r = strptime(p, "%H:%M", t);
364       if ( !r ) return nullptr;
365       p = r;
366       break;
367     }
368     case 'F': {
369       const char *r = strptime(p, "%Y-%m-%d", t);
370       if ( !r ) return nullptr;
371       p = r;
372       have_ymd = true;
373       break;
374     }
375     case 'D':
376     case 'x': {
377       const char *r = strptime(p, "%m/%d/%y", t);
378       if ( !r ) return nullptr;
379       p = r;
380       have_ymd = true;
381       break;
382     }
383     case 'c': {
384       const char *r = strptime(p, "%a %b %e %H:%M:%S %Y", t);
385       if ( !r ) return nullptr;
386       p = r;
387       have_ymd = true;
388       break;
389     }
390     default:
391       return nullptr;
392     }
393   }
394 
395   if ( have_ampm ) {
396     if ( pm && t->tm_hour < 12 ) t->tm_hour += 12;
397     if ( !pm && t->tm_hour == 12 ) t->tm_hour = 0;
398   }
399   if ( have_ymd ) {
400     const chrono::civil c = civil_of_tm(*t);
401     t->tm_wday = static_cast<int>(chrono::weekday_of(c.y, c.mo, c.d));
402     t->tm_yday = static_cast<int>(chrono::day_of_year(c.y, c.mo, c.d)) - 1;
403   }
404   return p;
405 }

410 inline i32
411 usleep(u64 usec) noexcept
412 {
413   timespec_t r = chrono::ts_of_ns(static_cast<i64>(usec) * static_cast<i64>(chrono::ns_per_us));
414   timespec_t rem{};
415   for ( ;; ) {
416     const i32 v = static_cast<i32>(micron::nanosleep(r, rem));
417     if ( v == 0 ) return 0;
418     if ( -v != static_cast<i32>(error::interrupted) ) return v;
419     r = rem;
420   }
421 }
422 
423 inline i32
424 nsleep(u64 nsec) noexcept
425 {
426   timespec_t r = chrono::ts_of_ns(static_cast<i64>(nsec));
427   timespec_t rem{};
428   for ( ;; ) {
429     const i32 v = static_cast<i32>(micron::nanosleep(r, rem));
430     if ( v == 0 ) return 0;
431     if ( -v != static_cast<i32>(error::interrupted) ) return v;
432     r = rem;
433   }
434 }

448 clk_tck(void) noexcept

458 inline long
459 sysconf(int name) noexcept
460 {
461   switch ( name ) {
462   case _sc_clk_tck:
463     return clk_tck();
464   case _sc_pagesize: {
465 #if defined(__micron_arch_wasm32)
466     return 65536;
467 #else
468     const unsigned long v = micron::getauxval(micron::at_pagesz);
469     return v != 0 ? static_cast<long>(v) : static_cast<long>(__micron_page_size_default);
470 #endif
471   }
472   default:
473     return -1l;
474   }
