142 // 0 on success, -errno otherwise. loops through signals
143 inline i32
144 sleep_until(clockid_t clk, i64 deadline_ns) noexcept
145 {
146   const timespec_t ts = ts_of_ns(deadline_ns);
147   for ( ;; ) {
148     const i32 r = static_cast<i32>(micron::clock_nanosleep(clk, timer_abstime, ts));
149     if ( r == 0 ) return 0;
150     if ( -r != static_cast<i32>(error::interrupted) ) return r;
151   }
152 }

154 // as above but reports the interruption instead of swallowing it
155 inline i32
156 sleep_until_once(clockid_t clk, i64 deadline_ns) noexcept
157 {
158   const timespec_t ts = ts_of_ns(deadline_ns);
159   return static_cast<i32>(micron::clock_nanosleep(clk, timer_abstime, ts));
160 }

162 inline constexpr i64 __ns_max = 0x7FFF'FFFF'FFFF'FFFFll;
163 
164 inline i32
165 sleep_ns(i64 ns) noexcept
166 {
167   if ( ns <= 0 ) return 0;
168   const i64 start = mono_ns();
169   if ( start < 0 ) return static_cast<i32>(start);
170   // NOTE: a literal max, not `~0ll >> 1` -- that is an ARITHMETIC shift of -1 and yields -1, so the
171   // guard fired on every call and this returned -EINVAL without ever sleeping
172   if ( start > __ns_max - ns ) return -static_cast<i32>(error::invalid_arg);
173   return sleep_until(clock_monotonic, start + ns);
174 }

569   template<unit U = unit::milliseconds>
570   constexpr D
571   as() const noexcept
572   {
573     fduration_t sec = static_cast<fduration_t>(d) / 1'000.0;
574     if constexpr ( U == unit::milliseconds )
575       return static_cast<D>(d);
576     else if constexpr ( U == unit::seconds )
577       return static_cast<D>(sec);
578     else if constexpr ( U == unit::microseconds )
579       return static_cast<D>(sec * 1e6);
580     else if constexpr ( U == unit::nanoseconds )
581       return static_cast<D>(sec * 1e9);
582     else if constexpr ( U == unit::minutes )

584     else if constexpr ( U == unit::hours )
585       return static_cast<D>(sec / __dur_sec_per_hr);
586     else
587       return static_cast<D>(sec / __dur_sec_per_day);
588   }

719   return year_month_day::from_unix(micron::time());
720 }
721 
722 inline time_of_day
723 time_of_day_now(void)
724 {
725   timespec_t t;
726   if ( micron::clock_gettime(static_cast<clockid_t>(clock_realtime), t) < 0 )
727     exc<except::runtime_error>("micron::time_of_day_now failed to get time");
728   fduration_t secs_today = static_cast<fduration_t>(t.tv_sec % __dur_sec_per_day) + static_cast<fduration_t>(t.tv_nsec) * 1e-9;
729   return time_of_day(secs_today);
730 }

732 template<unit U = unit::seconds>
733 inline fduration_t
734 elapsed(const timespec_t &begin, const timespec_t &end) noexcept
735 {
736   time_t sec = end.tv_sec - begin.tv_sec;
737   long nsec = static_cast<long>(end.tv_nsec - begin.tv_nsec);
738   __impl::normalise(sec, nsec);
739   return __impl::delta_to_unit<U>(sec, nsec);
740 }

742 inline fduration_t
743 timediff(time_t t0, time_t t1) noexcept
744 {
745   return static_cast<fduration_t>(t1) - static_cast<fduration_t>(t0);
746 }
