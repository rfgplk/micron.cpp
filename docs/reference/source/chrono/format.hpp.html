22 enum class offset_style : u8 {
23   none,              // omit the offset entirely
24   z_or_colon,        // "Z" at UTC, else "+02:00"
25   colon,             // always "+00:00"
26   compact,           // always "+0200"
27   z_or_compact,      // "Z" at UTC, else "+0200"
28 };

30 enum class date_sep : u8 {
31   iso_t,      // "2026-08-14T14:22:11"
32   space,      // "2026-08-14 14:22:11"
33   none,       // "20260814T142211"
34 };

39 struct wbuf {
40   char *p;
41   usize cap;
42   usize n;
43 
44   constexpr wbuf(char *b, usize c) noexcept : p(b), cap(c), n(0) { }
45 
46   constexpr void
47   put(char ch) noexcept
48   {
49     if ( n < cap ) p[n] = ch;
50     ++n;
51   }
52 
53   constexpr void
54   put_n(const char *s, usize k) noexcept
55   {
56     for ( usize i = 0; i < k; ++i ) put(s[i]);
57   }
58 
59   constexpr void
60   put_cstr(const char *s) noexcept
61   {
62     if ( !s ) return;
63     while ( *s ) put(*s++);
64   }
65 
66   constexpr void
67   put_u(u64 v, u32 width, char fill = '0') noexcept
68   {
69     char tmp[24];
70     u32 k = 0;
71     do {
72       tmp[k++] = static_cast<char>('0' + static_cast<char>(v % 10u));
73       v /= 10u;
74     } while ( v != 0 );
75     for ( u32 i = k; i < width; ++i ) put(fill);
76     while ( k != 0 ) put(tmp[--k]);
77   }
78 
79   constexpr void
80   put_i(i64 v, u32 width, char fill = '0') noexcept
81   {
82     if ( v < 0 ) {
83       put('-');
84       put_u(static_cast<u64>(-(v + 1)) + 1u, width, fill);
85     } else {
86       put_u(static_cast<u64>(v), width, fill);
87     }
88   }
89 };

91 constexpr void
92 __put_year(wbuf &w, i32 y) noexcept
93 {
94   if ( y >= 0 && y <= 9999 ) {
95     w.put_u(static_cast<u64>(y), 4);
96   } else {
97     w.put(y < 0 ? '-' : '+');
98     const i64 mag = y < 0 ? -static_cast<i64>(y) : static_cast<i64>(y);
99     w.put_u(static_cast<u64>(mag), 4);
100   }
101 }
102 
103 // truncating, not rounding
104 constexpr void
105 __put_subsec(wbuf &w, u32 ns, u32 digits) noexcept
106 {
107   if ( digits == 0 ) return;
108   if ( digits > 9 ) digits = 9;
109   u32 scale = 1;
110   for ( u32 i = digits; i < 9; ++i ) scale *= 10u;
111   w.put('.');
112   w.put_u(static_cast<u64>(ns / scale), digits);
113 }
114 
115 // WARNING: offsets are emitted to minute resolution
116 constexpr void
117 __put_offset(wbuf &w, i32 off, offset_style os) noexcept
118 {
119   if ( os == offset_style::none ) return;
120   if ( off == 0 && (os == offset_style::z_or_colon || os == offset_style::z_or_compact) ) {
121     w.put('Z');
122     return;
123   }
124   w.put(off < 0 ? '-' : '+');
125   const u32 mag = static_cast<u32>(off < 0 ? -static_cast<i64>(off) : static_cast<i64>(off));
126   w.put_u(mag / 3600u, 2);
127   if ( os == offset_style::z_or_colon || os == offset_style::colon ) w.put(':');
128   w.put_u((mag % 3600u) / 60u, 2);
129 }

131 constexpr usize
132 __iso8601(char *buf, usize cap, const civil &c, u32 subsec_digits, date_sep sep, offset_style os) noexcept
133 {
134   wbuf w(buf, cap);
135   __put_year(w, c.y);
136   w.put('-');
137   w.put_u(c.mo, 2);
138   w.put('-');
139   w.put_u(c.d, 2);
140   w.put(sep == date_sep::space ? ' ' : 'T');
141   w.put_u(c.h, 2);
142   w.put(':');
143   w.put_u(c.mi, 2);
144   w.put(':');
145   w.put_u(c.s, 2);
146   __put_subsec(w, c.ns, subsec_digits);
147   __put_offset(w, c.off, os);
148   return w.n;
149 }

151 constexpr usize
152 __rfc2822(char *buf, usize cap, const civil &c, bool gmt_literal) noexcept
153 {
154   wbuf w(buf, cap);
155   const u32 wd = weekday_of(c.y, c.mo, c.d);
156   w.put_cstr(weekday_abbr[wd < 7 ? wd : 0]);
157   w.put(',');
158   w.put(' ');
159   w.put_u(c.d, 2);
160   w.put(' ');
161   w.put_cstr(month_abbr[c.mo <= 12 ? c.mo : 0]);
162   w.put(' ');
163   w.put_i(static_cast<i64>(c.y), 4);
164   w.put(' ');
165   w.put_u(c.h, 2);
166   w.put(':');
167   w.put_u(c.mi, 2);
168   w.put(':');
169   w.put_u(c.s, 2);
170   w.put(' ');
171   if ( gmt_literal )
172     w.put_cstr("GMT");
173   else
174     __put_offset(w, c.off, offset_style::compact);
175   return w.n;
176 }

328 constexpr usize
329 __strftime(char *buf, usize cap, const char *fmt, const civil &c) noexcept
330 {
331   wbuf w(buf, cap);
332   if ( !fmt ) return 0;
333 
334   for ( const char *f = fmt; *f; ++f ) {
335     if ( *f != '%' ) {
336       w.put(*f);
337       continue;
338     }
339     ++f;
340     if ( *f == '\0' ) {
341       w.put('%');
342       break;
343     }
344     const u32 wd = weekday_of(c.y, c.mo, c.d);
345     switch ( *f ) {
346     case 'a':
347       w.put_cstr(weekday_abbr[wd < 7 ? wd : 0]);
348       break;
349     case 'A':
350       w.put_cstr(weekday_name[wd < 7 ? wd : 0]);
351       break;
352     case 'b':
353     case 'h':
354       w.put_cstr(month_abbr[c.mo <= 12 ? c.mo : 0]);
355       break;
356     case 'B':
357       w.put_cstr(month_name[c.mo <= 12 ? c.mo : 0]);
358       break;
359     case 'c':      // the POSIX locale's date-and-time, which is asctime's shape
360       w.put_cstr(weekday_abbr[wd < 7 ? wd : 0]);
361       w.put(' ');
362       w.put_cstr(month_abbr[c.mo <= 12 ? c.mo : 0]);
363       w.put(' ');
364       w.put_u(c.d, 2, ' ');
365       w.put(' ');
366       w.put_u(c.h, 2);
367       w.put(':');
368       w.put_u(c.mi, 2);
369       w.put(':');
370       w.put_u(c.s, 2);
371       w.put(' ');
372       w.put_i(static_cast<i64>(c.y), 4);
373       break;
374     case 'C':
375       w.put_u(static_cast<u64>(c.y >= 0 ? c.y / 100 : 0), 2);
376       break;
377     case 'd':
378       w.put_u(c.d, 2);
379       break;
380     case 'D':      // %m/%d/%y
381       w.put_u(c.mo, 2);
382       w.put('/');
383       w.put_u(c.d, 2);
384       w.put('/');
385       w.put_u(static_cast<u64>(((c.y % 100) + 100) % 100), 2);
386       break;
387     case 'e':
388       w.put_u(c.d, 2, ' ');
389       break;
390     case 'F':      // %Y-%m-%d
391       __put_year(w, c.y);
392       w.put('-');
393       w.put_u(c.mo, 2);
394       w.put('-');
395       w.put_u(c.d, 2);
396       break;
397     case 'g':
398       w.put_u(static_cast<u64>(((iso_week_date(c.y, c.mo, c.d).year % 100) + 100) % 100), 2);
399       break;
400     case 'G':
401       w.put_i(static_cast<i64>(iso_week_date(c.y, c.mo, c.d).year), 4);
402       break;
403     case 'H':
404       w.put_u(c.h, 2);
405       break;
406     case 'I':
407       w.put_u(c.h % 12u == 0u ? 12u : c.h % 12u, 2);
408       break;
409     case 'j':
410       w.put_u(day_of_year(c.y, c.mo, c.d), 3);
411       break;
412     case 'k':
413       w.put_u(c.h, 2, ' ');
414       break;
415     case 'l':
416       w.put_u(c.h % 12u == 0u ? 12u : c.h % 12u, 2, ' ');
417       break;
418     case 'm':
419       w.put_u(c.mo, 2);
420       break;
421     case 'M':
422       w.put_u(c.mi, 2);
423       break;
424     case 'n':
425       w.put('\n');
426       break;
427     case 'p':
428       w.put_cstr(c.h < 12u ? "AM" : "PM");
429       break;
430     case 'P':
431       w.put_cstr(c.h < 12u ? "am" : "pm");
432       break;
433     case 'r':      // %I:%M:%S %p
434       w.put_u(c.h % 12u == 0u ? 12u : c.h % 12u, 2);
435       w.put(':');
436       w.put_u(c.mi, 2);
437       w.put(':');
438       w.put_u(c.s, 2);
439       w.put(' ');
440       w.put_cstr(c.h < 12u ? "AM" : "PM");
441       break;
442     case 'R':      // %H:%M
443       w.put_u(c.h, 2);
444       w.put(':');
445       w.put_u(c.mi, 2);
446       break;
447     case 's':      // seconds since the epoch, from the fields and the offset they carry
448       w.put_i(civil_secs(c) - static_cast<i64>(c.off), 1);
449       break;
450     case 'S':
451       w.put_u(c.s, 2);
452       break;
453     case 't':
454       w.put('\t');
455       break;
456     case 'T':      // %H:%M:%S
457       w.put_u(c.h, 2);
458       w.put(':');
459       w.put_u(c.mi, 2);
460       w.put(':');
461       w.put_u(c.s, 2);
462       break;
463     case 'u':      // 1 = Monday .. 7 = Sunday
464       w.put_u(wd == 0u ? 7u : wd, 1);
465       break;
466     case 'U':      // week of year, Sunday as the first day
467       w.put_u((day_of_year(c.y, c.mo, c.d) + 6u - wd) / 7u, 2);
468       break;
469     case 'V':
470       w.put_u(iso_week_date(c.y, c.mo, c.d).week, 2);
471       break;
472     case 'w':      // 0 = Sunday
473       w.put_u(wd, 1);
474       break;
475     case 'W':      // week of year, Monday as the first day
476       w.put_u((day_of_year(c.y, c.mo, c.d) + 6u - (wd == 0u ? 6u : wd - 1u)) / 7u, 2);
477       break;
478     case 'x':      // %m/%d/%y in the POSIX locale
479       w.put_u(c.mo, 2);
480       w.put('/');
481       w.put_u(c.d, 2);
482       w.put('/');
483       w.put_u(static_cast<u64>(((c.y % 100) + 100) % 100), 2);
484       break;
485     case 'X':      // %H:%M:%S in the POSIX locale
486       w.put_u(c.h, 2);
487       w.put(':');
488       w.put_u(c.mi, 2);
489       w.put(':');
490       w.put_u(c.s, 2);
491       break;
492     case 'y':
493       w.put_u(static_cast<u64>(((c.y % 100) + 100) % 100), 2);
494       break;
495     case 'Y':
496       __put_year(w, c.y);
497       break;
498     case 'z':
499       __put_offset(w, c.off, offset_style::compact);
500       break;
501     case 'Z':
502       if ( c.off == 0 )
503         w.put_cstr("UTC");
504       else
505         __put_offset(w, c.off, offset_style::compact);
506       break;
507     case '%':
508       w.put('%');
509       break;
510     default:      // an unknown specifier renders literally, the way glibc does
511       w.put('%');
512       w.put(*f);
513       break;
514     }
515   }
516   return w.n;
517 }

535 constexpr usize
536 __dur_hms(char *buf, usize cap, u64 secs) noexcept
537 {
538   wbuf w(buf, cap);
539   const u64 d = secs / 86400ull;
540   const u64 h = (secs / 3600ull) % 24ull;
541   const u64 m = (secs / 60ull) % 60ull;
542   if ( d != 0 ) {
543     w.put_u(d, 1);
544     w.put('d');
545     w.put(' ');
546     w.put_u(h, 2);
547     w.put(':');
548     w.put_u(m, 2);
549     return w.n;
550   }
551   w.put_u(h, 2);
552   w.put(':');
553   w.put_u(m, 2);
554   w.put(':');
555   w.put_u(secs % 60ull, 2);
556   return w.n;
557 }

559 constexpr usize
560 __dur_clock(char *buf, usize cap, u64 ticks, u64 hz) noexcept
561 {
562   wbuf w(buf, cap);
563   if ( hz == 0 ) hz = 1;
564   const u64 secs = ticks / hz;
565   if ( secs < 3600ull ) {
566     w.put_u(secs / 60ull, 1);
567     w.put(':');
568     w.put_u(secs % 60ull, 2);
569     w.put('.');
570     w.put_u(((ticks % hz) * 100ull) / hz, 2);
571     return w.n;
572   }
573   if ( secs < 86400ull ) {
574     w.put_u(secs / 3600ull, 1);
575     w.put(':');
576     w.put_u((secs / 60ull) % 60ull, 2);
577     w.put(':');
578     w.put_u(secs % 60ull, 2);
579     return w.n;
580   }
581   w.put_u(secs / 86400ull, 1);
582   w.put('d');
583   w.put(' ');
584   w.put_u((secs / 3600ull) % 24ull, 2);
585   w.put(':');
586   w.put_u((secs / 60ull) % 60ull, 2);
587   return w.n;
588 }

590 constexpr usize
591 __dur_units(char *buf, usize cap, u64 ns) noexcept
592 {
593   wbuf w(buf, cap);
594   if ( ns == 0 ) {
595     w.put('0');
596     w.put('s');
597     return w.n;
598   }
599 
600   struct comp {
601     u64 scale;
602     const char *suffix;
603   };
604 
605   constexpr comp __c[8] = { { ns_per_week, "w" }, { ns_per_day, "d" }, { ns_per_hour, "h" }, { ns_per_min, "m" },
606                             { ns_per_s, "s" },    { ns_per_ms, "ms" }, { ns_per_us, "us" },  { 1ull, "ns" } };
607   u64 rest = ns;
608   for ( usize i = 0; i < 8; ++i ) {
609     const u64 v = rest / __c[i].scale;
610     if ( v == 0 ) continue;
611     w.put_u(v, 1);
612     w.put_cstr(__c[i].suffix);
613     rest -= v * __c[i].scale;
614     if ( rest == 0 ) break;
615   }
616   return w.n;
617 }

650   return need <= cap ? need : 0;
651 }
