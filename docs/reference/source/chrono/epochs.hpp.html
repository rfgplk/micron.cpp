19 // seconds from each epoch's zero to 1970-01-01T00:00:00Z
20 inline constexpr i64 ntp_to_unix = 2'208'988'800ll;            // 1900-01-01
21 inline constexpr i64 filetime_to_unix = 11'644'473'600ll;      // 1601-01-01
22 inline constexpr i64 gps_to_unix = 315'964'800ll;              // 1980-01-06
23 inline constexpr i64 dos_to_unix = 315'532'800ll;              // 1980-01-01
24 inline constexpr i64 jdn_at_unix = 2'440'588ll;                // julian day number of 1970-01-01
25 inline constexpr i64 mjd_at_unix = 40'587ll;                   // modified julian day of 1970-01-01
26 inline constexpr i64 sec_per_gps_week = 604'800ll;
27 
28 inline constexpr i32 tai_minus_utc_default = 37;

30 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
31 // NTP (RFC 5905)
32 
33 inline constexpr i64
34 ntp_from_unix(i64 unix_sec) noexcept
35 {
36   return unix_sec + ntp_to_unix;
37 }
38 
39 inline constexpr i64
40 unix_from_ntp(i64 ntp_sec) noexcept
41 {
42   return ntp_sec - ntp_to_unix;
43 }
44 
45 inline constexpr u64
46 ntp_timestamp(i64 unix_sec, u32 nsec) noexcept
47 {
48   const u64 s = static_cast<u64>(ntp_from_unix(unix_sec)) & 0xFFFF'FFFFull;
49   // fraction = nsec * 2^32 / 1e9, computed exactly in 64 bits
50   const u64 frac = (static_cast<u64>(nsec) << 32) / 1'000'000'000ull;
51   return (s << 32) | frac;
52 }
53 
54 inline constexpr i64
55 unix_sec_from_ntp_timestamp(u64 ts, i64 era = 0) noexcept
56 {
57   return unix_from_ntp(static_cast<i64>(ts >> 32) + era * (1ll << 32));
58 }
59 
60 inline constexpr u32
61 nsec_from_ntp_timestamp(u64 ts) noexcept
62 {
63   return static_cast<u32>(((ts & 0xFFFF'FFFFull) * 1'000'000'000ull) >> 32);
64 }

66 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
67 // windows FILETIME
68 // 100-nanosecond ticks since 1601-01-01
69 
70 inline constexpr i64
71 filetime_from_unix(i64 unix_sec, u32 nsec = 0) noexcept
72 {
73   return (unix_sec + filetime_to_unix) * 10'000'000ll + static_cast<i64>(nsec / 100u);
74 }
75 
76 inline constexpr i64
77 unix_from_filetime(i64 ft) noexcept
78 {
79   return floor_div(ft, 10'000'000ll) - filetime_to_unix;
80 }
81 
82 inline constexpr u32
83 nsec_from_filetime(i64 ft) noexcept
84 {
85   return static_cast<u32>(floor_mod(ft, 10'000'000ll) * 100ll);
86 }

88 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
89 // GPS
90 //
91 // GPS time has no leap seconds at all
92 // GPS-UTC = (TAI-UTC) - 19
93 
94 struct gps_time {
95   i64 week;
96   i64 tow;      // time of week, seconds
97 };
98 
99 inline constexpr i64
100 gps_from_unix(i64 unix_sec, i32 tai_utc = tai_minus_utc_default) noexcept
101 {
102   return unix_sec - gps_to_unix + static_cast<i64>(tai_utc - 19);
103 }
104 
105 inline constexpr i64
106 unix_from_gps(i64 gps_sec, i32 tai_utc = tai_minus_utc_default) noexcept
107 {
108   return gps_sec + gps_to_unix - static_cast<i64>(tai_utc - 19);
109 }
110 
111 inline constexpr gps_time
112 gps_week_tow(i64 unix_sec, i32 tai_utc = tai_minus_utc_default) noexcept
113 {
114   const i64 g = gps_from_unix(unix_sec, tai_utc);
115   return gps_time{ floor_div(g, sec_per_gps_week), floor_mod(g, sec_per_gps_week) };
116 }
117 
118 inline constexpr i64
119 unix_from_gps_week_tow(i64 week, i64 tow, i32 tai_utc = tai_minus_utc_default) noexcept
120 {
121   return unix_from_gps(week * sec_per_gps_week + tow, tai_utc);
122 }

139 // %%%%%%%%%%%%%%%%%%%%%%%%%%%
140 // julian day
141 
142 inline constexpr i64
143 jdn(i64 unix_sec) noexcept
144 {
145   const i64 days = floor_div(unix_sec, 86400);
146   const i64 secs = unix_sec - days * 86400;
147   return days + jdn_at_unix + (secs >= 43200 ? 1 : 0);
148 }
149 
150 inline constexpr i64
151 mjd(i64 unix_sec) noexcept
152 {
153   return floor_div(unix_sec, 86400) + mjd_at_unix;
154 }
155 
156 inline constexpr i64
157 unix_from_mjd(i64 m) noexcept
158 {
159   return (m - mjd_at_unix) * 86400ll;
160 }

162 // julian date scaled by 1e6
163 inline constexpr i64
164 julian_date_x1e6(i64 unix_sec) noexcept
165 {
166   // JD = unix/86400 + 2440587.5
167   return (jdn_at_unix * 1'000'000ll) - 500'000ll + (unix_sec * 1'000'000ll) / 86400ll;
168 }

170 // %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
171 // MS-DOS / FAT packed date-time
172 
173 struct dos_datetime {
174   u16 date;
175   u16 time;
176 };
177 
178 inline constexpr dos_datetime
179 dos_from_civil(const civil &c) noexcept
180 {
181   if ( c.y < 1980 || c.y > 2107 ) return dos_datetime{ 0, 0 };
182   const u16 dt = static_cast<u16>(((static_cast<u32>(c.y - 1980) & 0x7Fu) << 9) | ((c.mo & 0x0Fu) << 5) | (c.d & 0x1Fu));
183   const u16 tm = static_cast<u16>(((c.h & 0x1Fu) << 11) | ((c.mi & 0x3Fu) << 5) | ((c.s / 2u) & 0x1Fu));
184   return dos_datetime{ dt, tm };
185 }
186 
187 inline constexpr civil
188 civil_from_dos(dos_datetime v) noexcept
189 {
190   civil c{};
191   c.y = static_cast<i32>((v.date >> 9) & 0x7Fu) + 1980;
192   c.mo = static_cast<u32>((v.date >> 5) & 0x0Fu);
193   c.d = static_cast<u32>(v.date & 0x1Fu);
194   c.h = static_cast<u32>((v.time >> 11) & 0x1Fu);
195   c.mi = static_cast<u32>((v.time >> 5) & 0x3Fu);
196   c.s = static_cast<u32>(v.time & 0x1Fu) * 2u;
197   return c;
198 }
