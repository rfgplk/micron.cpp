95 operator==(const identifier &__a, const identifier &__b) noexcept
96 {
97   if ( __a.size() != __b.size() ) return false;

104 template<typename T> class consteval_list
105 {
106   T *__mem = nullptr;
107   usize __len = 0;
108 
109 public:
110   using value_type = T;
111   using size_type = usize;
112   using const_iterator = const T *;
113 
114   constexpr consteval_list() noexcept = default;
115 
116   template<typename R> constexpr consteval_list(const R &__r) : __len(__r.size())
117   {
118     __mem = __len ? new T[__len] : nullptr;
119     usize __i = 0;
120     for ( auto __e : __r ) __mem[__i++] = __e;
121   }
122 
123   constexpr consteval_list(const consteval_list &__o) : __len(__o.__len)
124   {
125     __mem = __len ? new T[__len] : nullptr;
126     for ( usize i = 0; i < __len; ++i ) __mem[i] = __o.__mem[i];
127   }
128 
129   constexpr consteval_list &
130   operator=(const consteval_list &__o)
131   {
132     if ( this == &__o ) return *this;
133     if ( __mem ) delete[] __mem;
134     __len = __o.__len;
135     __mem = __len ? new T[__len] : nullptr;
136     for ( usize i = 0; i < __len; ++i ) __mem[i] = __o.__mem[i];
137     return *this;
138   }
139 
140   constexpr ~consteval_list()
141   {
142     if ( __mem ) delete[] __mem;
143   }
144 
145   constexpr usize
146   size() const noexcept
147   {
148     return __len;
149   }
150 
151   constexpr bool
152   empty() const noexcept
153   {
154     return __len == 0;
155   }
156 
157   constexpr const T *
158   begin() const noexcept
159   {
160     return __mem;
161   }
162 
163   constexpr const T *
164   end() const noexcept
165   {
166     return __mem + __len;
167   }
168 
169   constexpr T
170   operator[](usize __i) const noexcept
171   {
172     return __mem[__i];
173   }

1526 // [meta.define.static]
1527 template<typename R>
1528 consteval auto
1529 define_static_array(R &&__r)
1530 {
1531   using T = micron::remove_cvref_t<decltype(*__r.begin())>;
1532   // an empty range makes reflect_constant_array link against std::array, which the selfhosted branch
1533   // deliberately does not declare
1534   if ( __r.begin() == __r.end() ) return micron::raw_slice<const T>{};
1535   const auto __a = std::meta::reflect_constant_array(static_cast<R &&>(__r));
1536   const auto __t = std::meta::type_of(__a);
1537   if ( std::meta::is_array_type(__t) ) return micron::raw_slice<const T>{ std::meta::extract<const T *>(__a), std::meta::extent(__t, 0u) };
1538   return micron::raw_slice<const T>{};
1539 }
